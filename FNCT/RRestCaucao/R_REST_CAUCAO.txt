--R_REST_CAUCAO
select conp.cd_con_pag
       ,conp.dt_emissao
       ,conp.nr_documento nr_documento_pagto
       ,conp.vl_bruto_conta
       ,conp.ds_fornecedor
       ,cau.cd_caucao
       ,cau.vl_caucao
       ,ate.cd_atendimento
       ,ate.dt_atendimento
       ,rec.nr_documento nr_documento_rec
       ,rec.dt_recebimento
       ,nvl(rec.vl_recebido,0) vl_recebido
from dbamv.con_pag conp
     ,dbamv.itcon_pag it
     ,dbamv.caucao cau
     ,dbamv.atendime ate
     ,dbamv.reccon_rec rec
	 ,dbamv.usuario_multi_empresa ume
where conp.cd_con_pag = it.cd_con_pag
  and  ate.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
  and conp.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA 
  and conp.cd_atendimento = ate.cd_atendimento
  and ate.cd_atendimento = cau.cd_atendimento
  and rec.cd_caucao = cau.cd_caucao
  and ate.cd_multi_empresa =  :CAMPO_CD_MULTI_EMPRESA
  and ate.cd_multi_empresa = ume.cd_multi_empresa
  and ume.cd_id_usuario = USER
