--R_AUDIT_CONTRATO_ADIANT
SELECT * FROM (
SELECT DECODE(:CAMPO_SN_AGRUPAR, 'S', CA.CD_MULTI_EMPRESA, 'NULL') CD_MULTI_EMPRESA,
       CA.CD_CONTRATO_ADIANT,
       ICR.NR_PARCELA,
       CA.DS_CONTRATO_ADIANT,
       CA.NR_CONTRATO_ADIANT,
       CA.DT_CONTRATO_ADIANT,
       CA.TP_CONTRATO_ADIANT,
       CA.CD_CONTRATANTE,
       CA.NM_CONTRATANTE,
       CA.VL_CONTRATO_ADIANT,
       CA.VL_SALDO_RECEBIDO,
       CA.CD_ATENDIMENTO,
       CR.CD_CON_REC,
       CR.NR_DOCUMENTO,
        ICR.TP_QUITACAO,
       ICR.DT_VENCIMENTO,
       ICR.DT_PREVISTA_RECEBIMENTO,
       ICR.VL_DUPLICATA,
       RCR.DT_RECEBIMENTO,
       RCR.VL_RECEBIDO
  FROM DBAMV.CONTRATO_ADIANTAMENTO CA,
       DBAMV.CON_REC CR,
       DBAMV.ITCON_REC ICR,
       DBAMV.RECCON_REC RCR
 WHERE CA.CD_CONTRATO_ADIANT = CR.CD_CONTRATO_ADIANT
   AND CR.CD_CON_REC = ICR.CD_CON_REC
   AND RCR.CD_ITCON_REC(+) = ICR.CD_ITCON_REC
   AND CA.TP_CONTRATANTE = :CAMPO_TP_CONTRATANTE
   AND CA.DT_CONTRATO_ADIANT BETWEEN :CAMPO_DTINIC  AND  
   AND CA.ST_CONTRATO_ADIANT = :CAMPO_ST_CONTRATO_ADIANT
Group By CA.CD_MULTI_EMPRESA,
         CA.CD_CONTRATO_ADIANT,
         ICR.NR_PARCELA,
         CA.DS_CONTRATO_ADIANT,
         CA.NR_CONTRATO_ADIANT,
         CA.DT_CONTRATO_ADIANT,
         CA.TP_CONTRATO_ADIANT,
         CA.CD_CONTRATANTE,
         CA.NM_CONTRATANTE,
         CA.VL_CONTRATO_ADIANT,
         CA.VL_SALDO_RECEBIDO,
         CA.CD_ATENDIMENTO,
         CR.CD_CON_REC,
         CR.NR_DOCUMENTO,
         ICR.TP_QUITACAO,
         ICR.DT_VENCIMENTO,
         ICR.DT_PREVISTA_RECEBIMENTO,
         ICR.VL_DUPLICATA,
         RCR.DT_RECEBIMENTO,
         RCR.VL_RECEBIDO
)
ORDER BY CD_MULTI_EMPRESA,
CD_CONTRATO_ADIANT,DS_CONTRATO_ADIANT,NR_CONTRATO_ADIANT,DT_CONTRATO_ADIANT,TP_CONTRATO_ADIANT,CD_CONTRATANTE,VL_SALDO_RECEBIDO,NM_CONTRATANTE,VL_CONTRATO_ADIANT,CD_CON_REC,NR_DOCUMENTO,CD_ATENDIMENTO
