--R_RECIBO_TESOURARIA_AGRUPA
SELECT '1' VIA
       ,'1ª VIA CLIENTE' DS_VIA
       ,(SELECT 'Este recibo só terá valor de quitação após compensação bancária do cheque mencionado neste documento.' TEXTO
           FROM DBAMV.VDIC_RECIBO_TESOURARIA
          WHERE CD_CON_REC = :CAMPO_CD_CON_REC
            AND CD_TIPO_RECEB = 1
            AND ROWNUM = 1) TEXTO
       ,CAB.NM_CLIENTE NOME
       ,CAB.RPS RPS
       ,CAB.DT_EMISSAO DT_EMISSAO
       ,CAB.VL_DUPLICATA VL_TITULO
       ,(CAB.VL_DUPLICATA - CAB.VL_RECEBIDO - CAB.VL_DESCONTO ) SALDO
       ,CAB.VL_RECEBIDO VL_RECEBIDO_CAB
       ,CAB.VL_RECEBIDO VL_RECEBIDO_CAB1
       ,CAB.VL_ACRESCIMO
       ,CAB.VL_DESCONTO
       ,dbamv.PACK_FNCP.EXTENSOS((SELECT Sum(v.VL_RECEBIDO)
									FROM DBAMV.VDIC_RECIBO_TESOURARIA v
									WHERE v.CD_CON_REC_AGRUP = :CAMPO_CD_CON_REC)) EXTENSO
       ,(SELECT Sum(v.VL_RECEBIDO)
			FROM DBAMV.VDIC_RECIBO_TESOURARIA v
			WHERE v.CD_CON_REC_AGRUP = :CAMPO_CD_CON_REC) VL_TOTAL
		,(SELECT Nvl(SUM(Nvl(v.VL_RECEBIDO,0)),0) FROM DBAMV.VDIC_RECIBO_TESOURARIA v, DBAMV.RECCON_REC r
			WHERE v.CD_CON_REC_AGRUP = :CAMPO_CD_CON_REC
			AND v.cd_reccon_rec = r.cd_reccon_rec
			AND r.CD_CONTRATO_ADIANT IS NOT NULL)  VL_ADIANTAMENTO
FROM (SELECT v.NM_CLIENTE
             ,v.RPS
             ,V.DT_EMISSAO
             ,(SELECT Sum(VL_DUPLICATA) FROM (SELECT DISTINCT v.CD_CON_REC, v.CD_ITCON_REC, Nvl(v.VL_DUPLICATA,0) VL_DUPLICATA
												FROM DBAMV.VDIC_RECIBO_TESOURARIA v
												WHERE v.CD_CON_REC_AGRUP = :CAMPO_CD_CON_REC)) VL_DUPLICATA
             ,Sum(Nvl(v.VL_RECEBIDO,0)) VL_RECEBIDO
             ,Sum(NVL(v.VL_ACRESCIMO,0)) VL_ACRESCIMO
             ,Sum(NVL(v.VL_DESCONTO,0)) VL_DESCONTO
			 FROM DBAMV.VDIC_RECIBO_TESOURARIA v
			WHERE v.CD_CON_REC_AGRUP = :CAMPO_CD_CON_REC
			GROUP BY NM_CLIENTE
			 ,RPS
			 ,DT_EMISSAO
			 ,VL_DUPLICATA) CAB
UNION ALL
SELECT '2' VIA
       ,'2ª VIA FINANCEIRO' DS_VIA
       ,(SELECT 'Este recibo só terá valor de quitação após compensação bancária do cheque mencionado neste documento.' TEXTO
           FROM DBAMV.VDIC_RECIBO_TESOURARIA
          WHERE CD_CON_REC = :CAMPO_CD_CON_REC
            AND CD_TIPO_RECEB = 1
            AND ROWNUM = 1) TEXTO
       ,CAB.NM_CLIENTE NOME
       ,CAB.RPS RPS
       ,CAB.DT_EMISSAO DT_EMISSAO
       ,CAB.VL_DUPLICATA VL_TITULO
       ,(CAB.VL_DUPLICATA - CAB.VL_RECEBIDO - CAB.VL_DESCONTO ) SALDO
       ,CAB.VL_RECEBIDO VL_RECEBIDO_CAB
       ,CAB.VL_RECEBIDO VL_RECEBIDO_CAB1
       ,CAB.VL_ACRESCIMO
       ,CAB.VL_DESCONTO
       ,dbamv.PACK_FNCP.EXTENSOS((SELECT Sum(v.VL_RECEBIDO)
									FROM DBAMV.VDIC_RECIBO_TESOURARIA v
									WHERE v.CD_CON_REC_AGRUP = :CAMPO_CD_CON_REC)) EXTENSO
       ,(SELECT Sum(v.VL_RECEBIDO)
			FROM DBAMV.VDIC_RECIBO_TESOURARIA v
			WHERE v.CD_CON_REC_AGRUP = :CAMPO_CD_CON_REC) VL_TOTAL
		,(SELECT Nvl(SUM(Nvl(v.VL_RECEBIDO,0)),0) FROM DBAMV.VDIC_RECIBO_TESOURARIA v, DBAMV.RECCON_REC r
					   WHERE v.cd_con_rec_agrup = :CAMPO_CD_CON_REC
						AND v.CD_RECCON_REC = r.cd_reccon_rec
						AND r.CD_CONTRATO_ADIANT IS NOT NULL)  VL_ADIANTAMENTO
FROM (SELECT v.NM_CLIENTE
             ,v.RPS
             ,v.DT_EMISSAO
             ,(SELECT Sum(VL_DUPLICATA) FROM (SELECT DISTINCT v.CD_CON_REC, v.CD_ITCON_REC, Nvl(v.VL_DUPLICATA,0) VL_DUPLICATA
                                                FROM DBAMV.VDIC_RECIBO_TESOURARIA v
                                               WHERE v.CD_CON_REC_AGRUP = :CAMPO_CD_CON_REC )) VL_DUPLICATA
             ,Sum(Nvl(v.VL_RECEBIDO,0)) VL_RECEBIDO
             ,Sum(NVL(v.VL_ACRESCIMO,0)) VL_ACRESCIMO
             ,Sum(NVL(v.VL_DESCONTO,0)) VL_DESCONTO
	FROM DBAMV.VDIC_RECIBO_TESOURARIA v
	WHERE v.CD_CON_REC_AGRUP = :CAMPO_CD_CON_REC
GROUP BY v.NM_CLIENTE
         ,v.RPS
         ,v.DT_EMISSAO
         ,v.VL_DUPLICATA) CAB
ORDER BY VIA, DT_EMISSAO

--R_RECIBO_TESOURARIA_AGRUPA_DET
SELECT NR_DOCUMENTO_RECEB ,
  NR_PARCELA ,
  DT_VENCIMENTO ,
  VL_RECEBIDO_DET
FROM (
SELECT  NR_DOCUMENTO_RECEB
          ,ROWNUM NR_PARCELA
          ,DT_VENCIMENTO
          ,VL_RECEBIDO_DET
   FROM ( SELECT VRT.CD_CON_REC
                ,DECODE(VRT.CD_TIPO_RECEB,'2',VRT.CD_RECCON_REC,NULL)
                ,Nvl(VRT.NR_DOCUMENTO_RECEB, VRT.DS_TIPO_RECEB) NR_DOCUMENTO_RECEB
                ,TO_CHAR(DECODE(VRT.CD_TIPO_RECEB,'2',DECODE(RCR.CD_CAIXA,NULL,MC.DT_MOVIMENTACAO,DC.DT_VENCIMENTO),NVL(VRT.DT_RECEBIMENTO,VRT.DT_CHEQUE)),'DD/MM/YYYY') DT_VENCIMENTO
                ,Sum(Nvl(VRT.VL_RECEBIDO,0)) VL_RECEBIDO_DET
           FROM DBAMV.VDIC_RECIBO_TESOURARIA VRT
                ,DBAMV.RECCON_REC RCR
                ,DBAMV.DOC_CAIXA DC
                ,DBAMV.REC_MOV_CON RMC
                ,DBAMV.MOV_CONCOR MC
          WHERE RCR.CD_RECCON_REC = VRT.CD_RECCON_REC
            AND RCR.CD_DOC_CAIXA = DC.CD_DOC_CAIXA (+)
            AND RCR.CD_RECCON_REC = RMC.CD_RECCON_REC (+)
            AND RMC.CD_MOV_CONCOR = MC.CD_MOV_CONCOR (+)
            AND VRT.CD_CON_REC = :CAMPO_CD_CON_REC
            AND (Nvl(:CAMPO_CD_RECCON_REC,0) = 0 OR VRT.CD_RECCON_REC = Nvl(:CAMPO_CD_RECCON_REC,0))
            AND (Nvl(:CAMPO_CD_ITCON_REC,0) = 0 OR VRT.CD_ITCON_REC = Nvl(:CAMPO_CD_ITCON_REC,0))
            AND VRT.VL_RECEBIDO > 0
          GROUP BY VRT.CD_CON_REC
                   ,DECODE(VRT.CD_TIPO_RECEB,'2',VRT.CD_RECCON_REC,NULL)
                   ,Nvl(VRT.NR_DOCUMENTO_RECEB, VRT.DS_TIPO_RECEB)
                   ,TO_CHAR(DECODE(VRT.CD_TIPO_RECEB,'2',DECODE(RCR.CD_CAIXA,NULL,MC.DT_MOVIMENTACAO,DC.DT_VENCIMENTO),NVL(VRT.DT_RECEBIMENTO,VRT.DT_CHEQUE)),'DD/MM/YYYY')
          ORDER BY DT_VENCIMENTO
  ) ORDER BY ROWNUM
)
WHERE
