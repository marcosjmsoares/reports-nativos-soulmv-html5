--R_ENTRADAS_SAME
SELECT * FROM (
SELECT IT_SAME.DT_ENTRADA
              ,PACIENTE.CD_PACIENTE
              ,PACIENTE.NM_PACIENTE
              ,IT_SAME.CD_ATENDIMENTO
              ,IT_SAME.NR_MATRICULA_SAME
              ,IT_SAME.NR_VOLUME
              ,SALA_SAME.DS_SALA_SAME
              ,ARMARIO_SAME.DS_ARMARIO_SAME
              ,SAME.DS_COLUNA
              ,SAME.DS_LINHA
FROM   DBAMV.IT_SAME
            ,DBAMV.PACIENTE
            ,DBAMV.SAME
            ,DBAMV.ARMARIO_SAME
            ,DBAMV.SALA_SAME
WHERE IT_SAME.DT_ENTRADA >= to_date(:CAMPO_DTINIC,'dd/mm/yyyy')  AND IT_SAME.DT_ENTRADA < to_date(:CAMPO_DTFIM,'dd/mm/yyyy' )+1
AND       SAME.CD_PACIENTE  =   PACIENTE.CD_PACIENTE
AND       SAME.CD_CAD_SAME = IT_SAME.CD_CAD_SAME
AND       SAME.NR_MATRICULA_SAME = IT_SAME.NR_MATRICULA_SAME
AND       SAME.NR_VOLUME = IT_SAME.NR_VOLUME
AND       SAME.CD_ARMARIO_SAME  =  ARMARIO_SAME.CD_ARMARIO_SAME(+)
AND       ARMARIO_SAME.CD_SALA_SAME  =  SALA_SAME.CD_SALA_SAME(+)
ORDER BY  SAME.DT_CADASTRO,
                    PACIENTE.NM_PACIENTE
)
ORDER BY DT_ENTRADA
