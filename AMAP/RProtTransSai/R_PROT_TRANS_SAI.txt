--R_PROT_TRANS_SAI
SELECT * FROM (
select sol_pront.CD_SOL_PRONT
      ,sol_pront.DT_SOL_PRONT
      ,sol_pront.CD_CAD_SAME_ORIGEM
      ,sol_pront.CD_CAD_SAME_DESTINO
      ,itsol_pront.NR_MATRICULA_SAME
      ,itsol_pront.NR_volume
      ,paciente.NM_PACIENTE
  from dbamv.sol_pront
      ,dbamv.itsol_pront
      ,dbamv.same
      ,dbamv.Paciente
 where Sol_pront.cd_sol_pront = itsol_pront.cd_sol_pront
   and Same.NR_MATRICULA_SAME = itsol_pront.NR_MATRICULA_SAME
   and Same.NR_volume = itsol_pront.NR_volume
   and paciente.cd_paciente   = same.cd_paciente
   and same.cd_cad_same = :CAMPO_CD_SAME
   and sol_pront.cd_sol_pront = :CAMPO_CD_PROTOCOLO
)
ORDER BY CD_SOL_PRONT,CD_CAD_SAME_ORIGEM,CD_CAD_SAME_DESTINO,DT_SOL_PRONT
