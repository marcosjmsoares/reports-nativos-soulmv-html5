--R_LIVRO_ALTAS
SELECT ATENDIME.CD_ATENDIMENTO
              ,ATENDIME.CD_PACIENTE
              ,PACIENTE.NM_PACIENTE
              ,CONVENIO.NM_CONVENIO
              ,ATENDIME.CD_CID
              ,LEITO.DS_RESUMO
              ,MOT_ALT.DS_MOT_ALT
              ,ATENDIME.DT_ATENDIMENTO
FROM     DBAMV.ATENDIME
               ,DBAMV.PACIENTE
               ,DBAMV.CONVENIO
               ,DBAMV.CID
               ,DBAMV.LEITO
               ,DBAMV.MOT_ALT
	,DBAMV.EMPRESA_CONVENIO
WHERE    Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And          Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND         ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND         TRUNC(ATENDIME.DT_ALTA)  =  TRUNC(:CAMPO_DATA) 
AND         ATENDIME.TP_ATENDIMENTO = 'I'
AND         ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND         ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND         ATENDIME.CD_CID = CID.CD_CID(+)
AND         ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
AND         ATENDIME.CD_MOT_ALT  =  MOT_ALT.CD_MOT_ALT(+)
ORDER BY ATENDIME.CD_ATENDIMENTO
