SELECT PACIENTE.CD_PACIENTE,
               PACIENTE.NM_PACIENTE,
               PACIENTE.DT_NASCIMENTO,
               PACIENTE.TP_SEXO,
               PACIENTE.TP_COR,
               DBAMV.FNC_ESTADO_CIVIL(PACIENTE.TP_ESTADO_CIVIL) TP_ESTADO_CIVIL1,
               PACIENTE.NR_DOCUMENTO,
               PACIENTE.NM_MAE,
               PACIENTE.DS_ENDERECO,
               PACIENTE.NM_BAIRRO,
               PACIENTE.NR_CEP,
               PACIENTE.NR_FONE,
               PACIENTE.DT_CADASTRO,
               ATENDIME.CD_ATENDIMENTO,
               ATENDIME.CD_PACIENTE,
               ATENDIME.DT_ATENDIMENTO,
               ATENDIME.HR_ATENDIMENTO,
               ATENDIME.CD_SERVICO,
               ATENDIME.CD_SER_DIS,
               ATENDIME.TP_ATENDIMENTO,
               ATENDIME.DT_VAL_GUIA,
               ATENDIME.NR_GUIA,
               ATENDIME.NM_USUARIO,
               ORI_ATE.DS_ORI_ATE,
               LPAD(TO_CHAR(PRO_INT.CD_PRO_INT), 8,'0') CD_PRO_INT,
               PRO_INT.DS_PRO_INT,
               dbamv.procedias(ATENDIME.CD_CONVENIO, ATENDIME.CD_PRO_INT, 'D') PROCED,
               PRESTADOR.NM_PRESTADOR,
               PRESTADOR.DS_CODIGO_CONSELHO,
               PRESTADOR.NR_FONE_CONTATO,
               PRESTADOR.NR_FONE_COMERCIAL,
               LEITO.DS_ENFERMARIA,
               LEITO.DS_LEITO,
               UNID_INT.CD_UNID_INT,
               UNID_INT.DS_UNID_INT,
               UNID_INT.DS_LOCALIZACAO,
               CONVENIO.NM_CONVENIO,
               CID.DS_CID,
               SERVICO.DS_SERVICO,
               SERVICO.TP_ANAMNESE TP_ANAM_SERVICO,
               SER_DIS.DS_SER_DIS,
               SER_DIS.TP_ANAMNESE TP_ANAM_SER_DIS,
               TIP_ACOM.DS_TIP_ACOM,
               RESPONSA.NM_RESPONSAVEL,
               RESPONSA.NR_FONE NR_FONE_RESPONSAVEL,
               RESPONSA.DS_DOCUMENTO DS_DOCUMENTO_RESPONSAVEL,
               RESPONSA.DS_ENDERECO DS_ENDERECO_RESP,
               CARTEIRA.NR_CARTEIRA,
               CARTEIRA.NM_EMPRESA,
               CARTEIRA.NM_TITULAR,
               CON_PLA.DS_CON_PLA  DS_CON_PLA,
               SUBSTR( CIDADE.NM_CIDADE, 1, 100) NM_CIDADE,
               CIDADE.CD_UF,
               SAME.NR_MATRICULA_SAME
FROM DBAMV.PACIENTE        PACIENTE,
           DBAMV.CIDADE            CIDADE,
           DBAMV.ATENDIME       ATENDIME,
           DBAMV.ORI_ATE          ORI_ATE,
           DBAMV.PRO_INT          PRO_INT,
           DBAMV.PRESTADOR   PRESTADOR,
           DBAMV.LEITO               LEITO,
           DBAMV.UNID_INT        UNID_INT,
           DBAMV.CONVENIO       CONVENIO,
           DBAMV.CID                    CID,
           DBAMV.SERVICO          SERVICO,
           DBAMV.SER_DIS          SER_DIS,
           DBAMV.TIP_ACOM       TIP_ACOM,
           DBAMV.RESPONSA      RESPONSA,
           DBAMV.CARTEIRA        CARTEIRA,
           DBAMV.CON_PLA          CON_PLA,
           DBAMV.SAME                SAME
	   ,Dbamv.Empresa_Con_Pla
Where Empresa_Con_Pla.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND  ATENDIME.CD_MULTI_EMPRESA =  DBAMV.PKG_MV2000.LE_EMPRESA 
And Empresa_Con_Pla.Cd_Convenio = Con_Pla.Cd_Convenio
And Empresa_Con_Pla.Cd_Con_Pla = Con_Pla.Cd_Con_Pla
And ATENDIME.CD_ATENDIMENTO = :CAMPO_PCD_ATENDIMENTO  AND
     ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA   AND
     ATENDIME.CD_PACIENTE    = PACIENTE.CD_PACIENTE(+) AND
     PACIENTE.CD_CIDADE      = CIDADE.CD_CIDADE(+) AND
     ATENDIME.CD_ORI_ATE     = ORI_ATE.CD_ORI_ATE(+)  AND
     ATENDIME.CD_PRO_INT     = PRO_INT.CD_PRO_INT(+) AND
     ATENDIME.CD_PRESTADOR   = PRESTADOR.CD_PRESTADOR(+) AND
     ATENDIME.CD_LEITO       = LEITO.CD_LEITO(+) AND
     LEITO.CD_UNID_INT       = UNID_INT.CD_UNID_INT(+) AND
     ATENDIME.CD_CONVENIO    = CONVENIO.CD_CONVENIO(+) AND
     ATENDIME.CD_CID         = CID.CD_CID (+) AND
     ATENDIME.CD_SERVICO     = SERVICO.CD_SERVICO (+) AND
     ATENDIME.CD_SER_DIS     = SER_DIS.CD_SER_DIS (+) AND
     ATENDIME.CD_TIP_ACOM    = TIP_ACOM.CD_TIP_ACOM (+) AND
     ATENDIME.CD_ATENDIMENTO = RESPONSA.CD_ATENDIMENTO (+) AND
     ATENDIME.CD_PACIENTE    = CARTEIRA.CD_PACIENTE (+) AND
     ATENDIME.CD_CONVENIO   = CARTEIRA.CD_CONVENIO (+) AND
     ATENDIME.CD_CONVENIO = CON_PLA.CD_CONVENIO(+) AND
     ATENDIME.CD_CON_PLA = CON_PLA.CD_CON_PLA(+) AND
     ATENDIME.CD_PACIENTE = SAME.CD_PACIENTE(+)
