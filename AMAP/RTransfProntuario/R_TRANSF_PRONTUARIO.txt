--R_TRANSF_PRONTUARIO
SELECT Sol_Pront.Cd_Sol_Pront Protocolo,
       To_Char(Sol_Pront.Dt_Sol_Pront,'DD/MM/YYYY') Data,
       To_Char(Sol_Pront.Dt_Sol_Pront,'HH24') Hora,
       Paciente.Cd_Paciente Prontuario,
       Paciente.Nm_Paciente Paciente,
       Sol_Pront.Cd_Portador Cd_Portador,
       Portadores.Ds_Portador Portador,
       ItSol_Pront.Nr_Matricula_Same Matricula,
       ItSol_Pront.Nr_Volume Volume,
       Origem.Ds_Cad_Same Origem,
       Destino.Ds_Cad_Same Destino
FROM   Dbamv.Sol_Pront,
       Dbamv.ItSol_Pront,
       Dbamv.Paciente,
       Dbamv.Same,
       Dbamv.Cad_Same Origem,
       Dbamv.Cad_Same Destino,
       Dbamv.Portadores
WHERE  Sol_Pront.Cd_Sol_Pront    = ItSol_Pront.Cd_Sol_Pront
AND    Sol_Pront.Cd_Cad_Same_Origem  = Origem.Cd_Cad_Same
AND    Sol_Pront.Cd_Cad_Same_Destino = Destino.Cd_Cad_Same
AND    Sol_Pront.Cd_Portador = Portadores.Cd_Portador(+)
AND    ItSol_Pront.Nr_Matricula_Same = Same.Nr_Matricula_Same
AND    ItSol_Pront.Nr_Volume = Same.Nr_Volume
AND    Same.Cd_Paciente = Paciente.Cd_Paciente
AND    ItSol_Pront.Cd_Atendimento IS NULL
AND    ItSol_Pront.Sn_Atendida = 'S'
Order By Data,Hora
