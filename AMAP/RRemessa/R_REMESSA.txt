--R_REMESSA
SELECT * FROM (
Select  rem_contr_mf.cd_remessa      cd_remessa
           ,rem_contr_mf.dt_remessa       dt_remessa
           ,contr_mf.cd_paciente             cd_paciente
           ,paciente.nm_paciente            nm_paciente
           ,contr_mf.tp_situacao              tp_situacao
           ,it_contr_mf.dt_internacao       dt_internacao
           ,it_contr_mf.dt_alta                  dt_alta
           ,it_contr_mf.ds_observacao    ds_observacao
  From  dbamv.rem_contr_mf     rem_contr_mf
           ,dbamv.contr_mf            contr_mf
           ,dbamv.paciente            paciente
           ,dbamv.it_contr_mf         it_contr_mf
 where rem_contr_mf.cd_remessa   LIKE :CAMPO_REMESSA
 and  rem_contr_mf.cd_remessa = contr_mf.cd_remessa
 and  contr_mf.cd_paciente  = paciente.cd_paciente
 and  contr_mf.cd_remessa  = it_contr_mf.cd_remessa(+)
 and  contr_mf.cd_paciente  = it_contr_mf.cd_paciente(+)
Order by paciente.nm_paciente
)
ORDER BY CD_REMESSA,DT_REMESSA,
CD_PACIENTE,NM_PACIENTE,TP_SITUACAO
