select    cad_same.ds_cad_same ds_cad_same,
          paciente.nm_paciente nm_paciente,
          same.nr_matricula_same nr_matricula_same,
          atendime.cd_atendimento cd_atendimento,
          convenio.nm_convenio nm_convenio,
          to_char(atendime.dt_atendimento,'dd/mm/yyyy') dt_atendimento,
          sala_same.ds_sala_same ds_sala_same,
          same.ds_coluna ds_coluna,
          armario_same.ds_armario_same ds_armario_same,
          same.ds_linha ds_linha
FROM    dbamv.cad_same
       ,dbamv.same
       ,dbamv.it_same
       ,dbamv.it_same_protocolos
       ,dbamv.paciente
       ,dbamv.sala_same
       ,dbamv.armario_same
       ,dbamv.protocolos
       ,dbamv.portadores
       ,dbamv.atendime
       ,dbamv.convenio
       ,dbamv.Empresa_Convenio
WHERE Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
  and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
  And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
  and cad_same.cd_cad_same = same.cd_cad_same
  and same.cd_paciente = paciente.cd_paciente
  and same.cd_cad_same = it_same.cd_cad_same
  and same.nr_matricula_same = it_same.nr_matricula_same
  and it_same.cd_atendimento = it_same_protocolos.cd_atendimento
  and same.cd_armario_same = armario_same.cd_armario_same
  and armario_same.cd_sala_same = sala_same.cd_sala_same
  and it_same_protocolos.cd_protocolo = protocolos.cd_protocolo
  and protocolos.cd_portador = portadores.cd_portador
  and it_same.cd_atendimento = atendime.cd_atendimento
  and atendime.cd_convenio = convenio.cd_convenio
  and protocolos.cd_protocolo = :CAMPO_CD_PROTOCOLO
  and rownum <= 3
  and atendime.cd_multi_empresa =  dbamv.Pkg_Mv2000.Le_Empresa 
 order by atendime.cd_atendimento
