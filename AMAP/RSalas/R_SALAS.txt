SELECT sala_same.cd_sala_same CODIGO   ,
       sala_same.cd_cad_same           ,
       cad_same.ds_cad_same            ,
       sala_same.ds_sala_same DESCRICAO
FROM   dbamv.sala_same,
       dbamv.cad_same
WHERE  cad_same.cd_cad_same = sala_same.cd_cad_same
AND        cad_same.cd_cad_same LIKE :CAMPO_SAME
