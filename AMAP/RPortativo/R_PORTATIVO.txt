--R_PORTATIVO
SELECT CD_CAD_SAME , DS_CAD_SAME , CD_PORTADOR , DS_PORTADOR , SUM(QTD_ITENS) QTD_ITENS
  FROM (
SELECT CAD_SAME.CD_CAD_SAME
              ,CAD_SAME.DS_CAD_SAME
              ,PORTADORES.CD_PORTADOR
              ,PORTADORES.DS_PORTADOR
              ,COUNT(PORTADORES.CD_PORTADOR) QTD_ITENS
FROM  DBAMV.IT_SAME
            ,DBAMV.CAD_SAME
            ,DBAMV.PROTOCOLOS
            ,DBAMV.PORTADORES
            ,DBAMV.IT_SAME_PROTOCOLOS
WHERE  IT_SAME.CD_CAD_SAME  =  CAD_SAME.CD_CAD_SAME
AND        PROTOCOLOS.CD_PORTADOR  =  PORTADORES.CD_PORTADOR
AND        PROTOCOLOS.CD_PROTOCOLO  =  IT_SAME_PROTOCOLOS.CD_PROTOCOLO
AND        IT_SAME.CD_CAD_SAME  =  IT_SAME_PROTOCOLOS.CD_CAD_SAME
AND        IT_SAME.CD_ATENDIMENTO = IT_SAME_PROTOCOLOS.CD_ATENDIMENTO
AND        IT_SAME.NR_MATRICULA_SAME  =  IT_SAME_PROTOCOLOS.NR_MATRICULA_SAME
AND        TRUNC(PROTOCOLOS.DT_SAIDA) BETWEEN :CAMPO_DTINIC GROUP BY CAD_SAME.CD_CAD_SAME
                   ,CAD_SAME.DS_CAD_SAME
                   ,IT_SAME_PROTOCOLOS.CD_CAD_SAME
                   ,PORTADORES.CD_PORTADOR
                   ,PORTADORES.DS_PORTADOR
UNION ALL
SELECT CAD_SAME.CD_CAD_SAME
              ,CAD_SAME.DS_CAD_SAME
              ,PORTADORES.CD_PORTADOR
              ,PORTADORES.DS_PORTADOR
              ,COUNT(PORTADORES.CD_PORTADOR) QTD_ITENS
FROM  DBAMV.SAME
           ,DBAMV.CAD_SAME
           ,DBAMV.PROTOCOLOS
           ,DBAMV.PORTADORES
           ,DBAMV.IT_SAME_PROTOCOLOS
WHERE  SAME.CD_CAD_SAME  =  CAD_SAME.CD_CAD_SAME
AND        SAME.NR_MATRICULA_SAME = IT_SAME_PROTOCOLOS.NR_MATRICULA_SAME
AND        PROTOCOLOS.CD_PORTADOR  =  PORTADORES.CD_PORTADOR
AND        PROTOCOLOS.CD_PROTOCOLO  =  IT_SAME_PROTOCOLOS.CD_PROTOCOLO
AND        TRUNC(PROTOCOLOS.DT_SAIDA) BETWEEN :CAMPO_DTINIC  GROUP BY CAD_SAME.CD_CAD_SAME
                   ,CAD_SAME.DS_CAD_SAME
                   ,IT_SAME_PROTOCOLOS.CD_CAD_SAME
                   ,PORTADORES.CD_PORTADOR
                   ,PORTADORES.DS_PORTADOR
)
GROUP BY CD_CAD_SAME , DS_CAD_SAME , CD_PORTADOR , DS_PORTADOR
ORDER BY CD_CAD_SAME , CD_PORTADOR , QTD_ITENS DESC
