--R_ETIQUETA_SAME_6
SELECT PACIENTE.CD_PACIENTE,
               SAME.CD_CAD_SAME,
               PACIENTE.NM_PACIENTE,
               PROFISSAO.NM_PROFISSAO PROFISSAO,
               PACIENTE.NR_CPF,
               PACIENTE.DS_ENDERECO,
               PACIENTE.NM_BAIRRO BAIRRO,
               PACIENTE.NR_ENDERECO,
               PACIENTE.NR_CEP,
               UPPER(DECODE(PACIENTE.TP_ESTADO_CIVIL, 'S', 'Solteiro', 'C', 'Casado', 'V', 'Viúvo', 'D', 'Desquitado', 'I', 'Divordicado')) ESTADO_CIVIL,
               PACIENTE.NM_MAE,
               PACIENTE.NR_FONE,
               to_date(SYSDATE,'dd/mm/yyyy hh24') DT_ATUAL
    FROM DBAMV.PACIENTE,
               DBAMV.SAME,
               DBAMV.PROFISSAO
WHERE PACIENTE.CD_PACIENTE      = SAME.CD_PACIENTE
      AND PACIENTE.CD_PROFISSAO   = PROFISSAO.CD_PROFISSAO(+)
      AND PACIENTE.CD_PACIENTE      = :CAMPO_CD_PACIENTE
      AND SAME.CD_CAD_SAME            = :CAMPO_CD_CAD_SAME
