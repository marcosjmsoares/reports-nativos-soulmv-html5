--R_LIVRO_INTERNACOES
SELECT ATENDIME.CD_ATENDIMENTO
              ,ATENDIME.CD_PACIENTE
              ,PACIENTE.NM_PACIENTE
              ,LEITO.DS_RESUMO
              ,PRESTADOR.NM_PRESTADOR
              ,PRESTADOR.DS_CODIGO_CONSELHO
              ,ATENDIME.CD_SERVICO
              ,SERVICO.DS_SERVICO
FROM     DBAMV.ATENDIME
               ,DBAMV.PACIENTE
               ,DBAMV.LEITO
               ,DBAMV.PRESTADOR
               ,DBAMV.SERVICO
WHERE  TRUNC(ATENDIME.DT_ATENDIMENTO)  =  :CAMPO_DATA
AND         ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND         ATENDIME.TP_ATENDIMENTO = 'I'
AND         ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND         ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
AND         ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
AND         ATENDIME.CD_SERVICO = SERVICO.CD_SERVICO(+)
ORDER BY ATENDIME.CD_ATENDIMENTO
