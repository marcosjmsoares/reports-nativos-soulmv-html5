--R_LOG_EXCLUSAO_SAME
SELECT (SELECT CD_CAD_SAME || ' - ' || DS_CAD_SAME FROM DBAMV.CAD_SAME C WHERE C.CD_CAD_SAME = LOG_IT_SAME_PROTOCOLOS.CD_CAD_SAME) CD_CAD_SAME,            
               NR_MATRICULA_SAME,
               CD_PROTOCOLO,
               CD_ATENDIMENTO,
               NR_SEQUENCIA,
               NR_VOLUME,
               HR_EXCLUSAO,
               DT_EXCLUSAO,
               NM_USUARIO,
              (SELECT DT_SAIDA FROM DBAMV.PROTOCOLOS P WHERE P.CD_PROTOCOLO = LOG_IT_SAME_PROTOCOLOS.CD_PROTOCOLO) DT_SAIDA
              FROM  DBAMV.LOG_IT_SAME_PROTOCOLOS
              WHERE TRUNC(DT_EXCLUSAO) BETWEEN TO_DATE(:CAMPO_DT_INICIAL,'DD/MM/YYYY')   AND TO_DATE(:CAMPO_DT_FINAL,'DD/MM/YYYY')
