--R_INTERNACAO_DIA
Select atendime.cd_atendimento,
       paciente.nm_paciente,
	   prestador.nm_prestador,
	   nvl( leito.ds_resumo, leito.ds_leito ) ds_leito,
	   convenio.nm_convenio,
	   tipo_internacao.ds_tipo_internacao
  From dbamv.atendime,
       dbamv.prestador,
	   dbamv.convenio,
	   dbamv.paciente,
	   dbamv.tipo_internacao,
	   dbamv.leito,
	   dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
  And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
  And atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
  And atendime.cd_prestador = prestador.cd_prestador
  And atendime.cd_convenio = convenio.cd_convenio
  And paciente.cd_paciente = atendime.cd_paciente
  And atendime.cd_leito = leito.cd_leito
  And tipo_internacao.cd_tipo_internacao = atendime.cd_tipo_internacao
  And atendime.tp_atendimento = 'I'
  And atendime.dt_atendimento >=  :CAMPO_DT_INICIO
  And atendime.dt_atendimento < :CAMPO_DT_FIM + 1
  And atendime.cd_multi_empresa =  dbamv.Pkg_Mv2000.Le_Empresa
order by convenio.nm_convenio, paciente.nm_paciente
