--R_MATRICULA
SELECT * FROM (
SELECT ARMARIO.CD_ARMARIO_SAME
             ,ARMARIO.DS_ARMARIO_SAME
             ,SALA.CD_SALA_SAME
             ,SALA.DS_SALA_SAME
             ,CAD_SAME.CD_CAD_SAME
             ,CAD_SAME.DS_CAD_SAME
             ,SAME.NR_MATRICULA_SAME
             ,SAME.NR_VOLUME
             ,PACIENTE.NM_PACIENTE
             ,SAME.DS_LINHA
             ,SAME.DS_COLUNA
             ,COUNT(IT_SAME.CD_CAD_SAME) QUANT
FROM   DBAMV.ARMARIO_SAME  ARMARIO
             ,DBAMV.SALA_SAME     SALA
             ,DBAMV.CAD_SAME
             ,DBAMV.SAME
             ,DBAMV.PACIENTE
             ,DBAMV.IT_SAME
WHERE  ARMARIO.CD_ARMARIO_SAME = SAME.CD_ARMARIO_SAME
  AND  ARMARIO.CD_SALA_SAME = SALA.CD_SALA_SAME
  AND    SALA.CD_CAD_SAME = CAD_SAME.CD_CAD_SAME
  AND   CAD_SAME.CD_CAD_SAME = SAME.CD_CAD_SAME
  AND   SAME.CD_PACIENTE = PACIENTE.CD_PACIENTE
  AND   SAME.CD_CAD_SAME = IT_SAME.CD_CAD_SAME(+)
  AND   SAME.NR_MATRICULA_SAME = IT_SAME.NR_MATRICULA_SAME(+)
  AND   SAME.NR_VOLUME = IT_SAME.NR_VOLUME(+)
GROUP BY ARMARIO.CD_ARMARIO_SAME
                   ,ARMARIO.DS_ARMARIO_SAME
                   ,SALA.CD_SALA_SAME
                   ,SALA.DS_SALA_SAME
                   ,CAD_SAME.CD_CAD_SAME
                   ,CAD_SAME.DS_CAD_SAME
                   ,SAME.NR_MATRICULA_SAME
                   ,SAME.NR_VOLUME
                   ,PACIENTE.NM_PACIENTE
                   ,SAME.DS_LINHA
                   ,SAME.DS_COLUNA
)
ORDER BY DS_CAD_SAME,CD_CAD_SAME,
CD_SALA_SAME,DS_SALA_SAME,
CD_ARMARIO_SAME,DS_ARMARIO_SAME, NR_MATRICULA_SAME
