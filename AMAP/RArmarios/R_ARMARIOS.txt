--R_ARMARIOS
SELECT cad_same.cd_cad_same,
       cad_same.ds_cad_same,
       sala_same.cd_sala_same,
       sala_same.ds_sala_same,
       armario_same.cd_armario_same CODIGO,
       armario_same.ds_armario_same DESCRICAO
FROM   dbamv.cad_same,
       dbamv.sala_same,
       dbamv.armario_same
WHERE  cad_same.cd_cad_same = sala_same.cd_cad_same
AND    sala_same.cd_sala_same = armario_same.cd_sala_same
