--R_SAMEORIEST
SELECT CAD_SAME.cd_cad_same,
       cad_same.ds_cad_same,
       atendime.cd_ori_ate,
       ori_ate.ds_ori_ate,
       count(NVL(atendime.cd_ori_ate,1)) qtd_ori
FROM dbamv.cad_same,
     dbamv.it_same,
     dbamv.ori_ate,
     dbamv.protocolos,
     dbamv.it_same_protocolos,
     dbamv.atendime
WHERE cad_same.cd_cad_same   = dbamv.it_same_protocolos.cd_cad_same
 AND dbamv.it_same.cd_cad_same(+)              = dbamv.it_same_protocolos.cd_cad_same
 AND dbamv.it_same.nr_matricula_same(+)       = dbamv.it_same_protocolos.nr_matricula_same
 AND dbamv.it_same.cd_atendimento(+)           = dbamv.it_same_protocolos.cd_atendimento
 AND dbamv.it_same_protocolos.cd_protocolo = dbamv.protocolos.cd_protocolo (+)
 AND atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
 AND   it_same.cd_atendimento = atendime.cd_atendimento (+)
 AND   atendime.cd_ori_ate    = ori_ate.cd_ori_ate(+)
 AND  cad_same.cd_cad_same  LIKE :CAMPO_SAME
 AND protocolos.dt_saida BETWEEN :CAMPO_DTINIC GROUP BY CAD_same.cd_cad_same,
         cad_same.ds_cad_same,
         atendime.cd_ori_ate,
         ori_ate.ds_ori_ate
ORDER BY CAD_SAME.cd_cad_same,
         qtd_ori DESC
