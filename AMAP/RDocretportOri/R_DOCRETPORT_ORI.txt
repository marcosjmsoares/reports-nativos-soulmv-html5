--R_DOCRETPORT_ORI
SELECT * FROM (
SELECT DISTINCT
               CAD_SAME.CD_CAD_SAME
              ,PROTOCOLOS.CD_PORTADOR
              ,PORTADORES.DS_PORTADOR
              ,PROTOCOLOS.CD_PROTOCOLO
              ,TO_CHAR(PROTOCOLOS.DT_SAIDA, 'DD/MM/YYYY')         DT_SAIDA
              ,TO_CHAR(PROTOCOLOS.HR_SAIDA, 'HH24')                  HR_SAIDA
              ,PROTOCOLOS.DS_MOTIVO
              ,TO_CHAR(PROTOCOLOS.DT_RETORNO,'DD/MM/YYYY')   DT_RETORNO
              ,TO_CHAR(IT_SAME.DT_ENTRADA,'DD/MM/YYYY')             DT_ENTRADA
              ,IT_SAME_PROTOCOLOS.CD_ATENDIMENTO
              ,IT_SAME_PROTOCOLOS.NR_MATRICULA_SAME
              ,ARMARIO_SAME.DS_ARMARIO_SAME
              ,SAME.DS_LINHA
              ,SAME.DS_COLUNA
FROM    DBAMV.SAME
             ,DBAMV.CAD_SAME
             ,DBAMV.ARMARIO_SAME
             ,DBAMV.SALA_SAME
             ,DBAMV.PROTOCOLOS
             ,DBAMV.IT_SAME_PROTOCOLOS
             ,DBAMV.PORTADORES
             ,DBAMV.IT_SAME
             ,DBAMV.ATENDIME
WHERE  PROTOCOLOS.DT_SAIDA  >= to_date( to_char( :CAMPO_DT_INICIO, 'DD/MM/YYYY' ) || ' 00', 'DD/MM/YYYY HH24' )
AND        PROTOCOLOS.DT_SAIDA  <= to_date( to_char( :CAMPO_DT_FIM   , 'DD/MM/YYYY' ) || ' 23', 'DD/MM/YYYY HH24' )
AND        ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND        SAME.CD_CAD_SAME = CAD_SAME.CD_CAD_SAME
AND        SAME.CD_ARMARIO_SAME  =  ARMARIO_SAME.CD_ARMARIO_SAME
AND        ARMARIO_SAME.CD_SALA_SAME  =  SALA_SAME.CD_SALA_SAME
AND        SAME.CD_CAD_SAME  =  IT_SAME.CD_CAD_SAME
AND        SAME.NR_MATRICULA_SAME  =  IT_SAME.NR_MATRICULA_SAME
AND        IT_SAME.CD_ATENDIMENTO  =  ATENDIME.CD_ATENDIMENTO
AND        IT_SAME.CD_PROTOCOLO_DE_SAIDA  =  PROTOCOLOS.CD_PROTOCOLO
AND        PROTOCOLOS.DT_RETORNO IS NULL
AND        IT_SAME.CD_CAD_SAME  =  IT_SAME_PROTOCOLOS.CD_CAD_SAME
AND        IT_SAME.NR_MATRICULA_SAME  =  IT_SAME_PROTOCOLOS.NR_MATRICULA_SAME
AND        IT_SAME.CD_PROTOCOLO_DE_SAIDA  =  IT_SAME_PROTOCOLOS.CD_PROTOCOLO
AND        PROTOCOLOS.CD_PORTADOR  = PORTADORES.CD_PORTADOR
AND        ATENDIME.CD_MULTI_EMPRESA =  DBAMV.PKG_MV2000.LE_EMPRESA 
UNION ALL
SELECT  DISTINCT
               CAD_SAME.CD_CAD_SAME
              ,PROTOCOLOS.CD_PORTADOR
              ,PORTADORES.DS_PORTADOR
              ,PROTOCOLOS.CD_PROTOCOLO
              ,TO_CHAR(PROTOCOLOS.DT_SAIDA, 'DD/MM/YYYY')         DT_SAIDA
              ,TO_CHAR(PROTOCOLOS.HR_SAIDA, 'HH24')                  HR_SAIDA
              ,PROTOCOLOS.DS_MOTIVO
              ,TO_CHAR(PROTOCOLOS.DT_RETORNO,'DD/MM/YYYY')   DT_RETORNO
              ,TO_CHAR(IT_SAME.DT_ENTRADA,'DD/MM/YYYY')             DT_ENTRADA
              ,IT_SAME_PROTOCOLOS.CD_ATENDIMENTO
              ,IT_SAME_PROTOCOLOS.NR_MATRICULA_SAME
              ,ARMARIO_SAME.DS_ARMARIO_SAME
              ,SAME.DS_LINHA
              ,SAME.DS_COLUNA
FROM   DBAMV.SAME
            ,DBAMV.CAD_SAME
            ,DBAMV.ARMARIO_SAME
            ,DBAMV.SALA_SAME
            ,DBAMV.PROTOCOLOS
            ,DBAMV.IT_SAME_PROTOCOLOS
            ,DBAMV.PORTADORES
            ,DBAMV.IT_SAME
            ,DBAMV.ATENDIME
WHERE  PROTOCOLOS.DT_SAIDA  >= to_date( to_char( :CAMPO_DT_INICIO, 'DD/MM/YYYY' ) || ' 00', 'DD/MM/YYYY HH24' )
AND        PROTOCOLOS.DT_SAIDA  <= to_date( to_char( :CAMPO_DT_FIM   , 'DD/MM/YYYY' ) || ' 23', 'DD/MM/YYYY HH24' )
AND        ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND        SAME.CD_CAD_SAME = CAD_SAME.CD_CAD_SAME
AND        SAME.CD_ARMARIO_SAME  =  ARMARIO_SAME.CD_ARMARIO_SAME
AND        ARMARIO_SAME.CD_SALA_SAME  =  SALA_SAME.CD_SALA_SAME
AND        SAME.CD_PROTOCOLO_DE_SAIDA  =  PROTOCOLOS.CD_PROTOCOLO
AND        SAME.CD_CAD_SAME  =  IT_SAME_PROTOCOLOS.CD_CAD_SAME
AND        SAME.NR_MATRICULA_SAME  =  IT_SAME_PROTOCOLOS.NR_MATRICULA_SAME
AND        SAME.CD_PROTOCOLO_DE_SAIDA  =  IT_SAME_PROTOCOLOS.CD_PROTOCOLO
AND        PROTOCOLOS.CD_PORTADOR  = PORTADORES.CD_PORTADOR
AND        SAME.CD_CAD_SAME  =  IT_SAME.CD_CAD_SAME(+)
AND        SAME.NR_MATRICULA_SAME  =  IT_SAME.NR_MATRICULA_SAME(+)
AND        IT_SAME.CD_ATENDIMENTO  =  ATENDIME.CD_ATENDIMENTO(+)
:CAMPO_CF_COND_SAME
)
ORDER BY CD_CAD_SAME,
CD_PORTADOR,DS_PORTADOR,
CD_PROTOCOLO,DS_MOTIVO,DT_SAIDA,HR_SAIDA,DT_RETORNO
