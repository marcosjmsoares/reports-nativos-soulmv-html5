--R_MATRICULA_SEM_ARMARIO_SALA
SELECT * FROM (
SELECT CAD_SAME.CD_CAD_SAME
             ,CAD_SAME.DS_CAD_SAME
             ,SAME.NR_MATRICULA_SAME
             ,SAME.NR_VOLUME
             ,PACIENTE.NM_PACIENTE
             ,SAME.DS_LINHA
             ,SAME.DS_COLUNA
             ,COUNT(IT_SAME.CD_CAD_SAME) QUANT
FROM   DBAMV.CAD_SAME
             ,DBAMV.SAME
             ,DBAMV.PACIENTE
             ,DBAMV.IT_SAME
WHERE  CAD_SAME.CD_CAD_SAME = SAME.CD_CAD_SAME
  AND   SAME.CD_PACIENTE = PACIENTE.CD_PACIENTE
  AND   SAME.CD_CAD_SAME = IT_SAME.CD_CAD_SAME(+)
  AND   SAME.NR_MATRICULA_SAME = IT_SAME.NR_MATRICULA_SAME(+)
  AND   SAME.NR_VOLUME = IT_SAME.NR_VOLUME(+)
GROUP BY CAD_SAME.CD_CAD_SAME
                   ,CAD_SAME.DS_CAD_SAME
                   ,SAME.NR_MATRICULA_SAME
                   ,SAME.NR_VOLUME
                   ,PACIENTE.NM_PACIENTE
                   ,SAME.DS_LINHA
                   ,SAME.DS_COLUNA
)
ORDER BY DS_CAD_SAME,CD_CAD_SAME,NR_MATRICULA_SAME
