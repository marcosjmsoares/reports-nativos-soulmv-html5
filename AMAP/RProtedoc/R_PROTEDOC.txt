--R_PROTEDOC
SELECT * FROM (
SELECT distinct protocolos_devolucao.cd_protocolo_devolucao
               ,to_char(protocolos.dt_saida, 'dd/mm/yyyy')              dt_saida
               ,to_char(protocolos.hr_saida, 'hh24')                     hr_saida
               ,to_char(protocolos.dt_prev_retorno, 'dd/mm/yyyy')                     dt_prev_retorno
               ,to_char(protocolos_devolucao.dt_devolucao, 'dd/mm/yyyy')              dt_retorno
               ,to_char(protocolos_devolucao.hr_devolucao, 'hh24')                     hr_retorno
               ,protocolos_devolucao.cd_portador
               ,portadores.ds_portador
               ,protocolos_devolucao.ds_motivo
               ,same.nr_matricula_same
               ,it_same_protocolos.cd_atendimento
               ,sala_same.cd_sala_same
               ,sala_same.ds_sala_same
               ,same.cd_armario_same
               ,armario_same.ds_armario_same
               ,same.ds_linha
               ,same.ds_coluna
               ,paciente.nm_paciente
               ,atendime.dt_atendimento
FROM    dbamv.cad_same
             ,dbamv.same
             ,dbamv.sala_same
             ,dbamv.armario_same
             ,dbamv.protocolos_devolucao protocolos_devolucao
             ,dbamv.protocolos
             ,dbamv.portadores
             ,dbamv.it_same_protocolos
             ,dbamv.atendime
             ,dbamv.paciente
WHERE  protocolos_devolucao.cd_protocolo_devolucao  =  :CAMPO_CD_PROTOCOLO
AND       atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
AND        cad_same.cd_cad_same  =  :CAMPO_CD_SAME
AND        protocolos_devolucao.cd_portador  =  portadores.cd_portador
AND        protocolos_devolucao.cd_protocolo_devolucao  =  it_same_protocolos.cd_protocolo_devolucao
AND        it_same_protocolos.cd_cad_same  =  same.cd_cad_same
AND        it_same_protocolos.nr_matricula_same  =  same.nr_matricula_same
AND        it_same_protocolos.cd_protocolo = protocolos.cd_protocolo
AND        same.cd_cad_same  =  cad_same.cd_cad_same
AND        same.cd_armario_same  =  armario_same.cd_armario_same(+)
AND        armario_same.cd_sala_same  =  sala_same.cd_sala_same(+)
AND        it_same_protocolos.cd_atendimento    =   atendime.cd_atendimento(+)
AND        same.cd_paciente = paciente.cd_paciente(+)
AND       atendime.cd_multi_empresa =  dbamv.Pkg_Mv2000.Le_Empresa 
)
ORDER BY DT_SAIDA,DT_PREV_RETORNO,DT_RETORNO,HR_RETORNO,HR_SAIDA,CD_PORTADOR,DS_PORTADOR,DS_MOTIVO,CD_PROTOCOLO_DEVOLUCAO
