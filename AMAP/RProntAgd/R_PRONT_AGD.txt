--R_PRONT_AGD
SELECT * FROM (
SELECT   MARCACAO.CD_MARCACAO        CD_MARCACAO
        ,PRESTADOR.CD_PRESTADOR      CD_PRESTADOR
        ,PRESTADOR.NM_PRESTADOR      NM_PRESTADOR
        ,MARCACAO.DT_MARCACAO        DT_MARCACAO
        ,MARCACAO.HR_INICIAL         HR_INICIAL
        ,MARCACAO.HR_FINAL           HR_FINAL
        ,TIP_MAR.CD_TIP_MAR          CD_TIP_MAR
        ,TIP_MAR.DS_TIP_MAR          DS_TIP_MAR
        ,MARCACAO.CD_SER_DIS         CD_SER_DIS
        ,SER_DIS.DS_SER_DIS          DS_SER_DIS
        ,MARCACAO.NR_TEMPO_MEDIO     NR_TEMPO_MEDIO
        ,MARCACAO.SN_SIA             SN_SIA
        ,IT_MARCACAO.CD_PACIENTE     CD_PACIENTE
        ,IT_MARCACAO.NM_PACIENTE     NM_PACIENTE
        ,IT_MARCACAO.NR_TELEFONE     NR_TELEFONE
        ,IT_MARCACAO.CD_ATENDIMENTO  CD_ATENDIMENTO
        ,IT_MARCACAO.CD_IT_MARCACAO  CD_IT_MARCACAO
        ,ITCONTR.DS_OBSERVACAO       DS_OBSERVACAO
FROM     DBAMV.MARCACAO              MARCACAO
        ,DBAMV.PRESTADOR             PRESTADOR
        ,DBAMV.SER_DIS               SER_DIS
        ,DBAMV.IT_MARCACAO           IT_MARCACAO
        ,DBAMV.TIP_MAR               TIP_MAR
        ,DBAMV.CONTR_MF              CONTR_MF
         ,DBAMV.IT_CONTR_MF          ITCONTR
WHERE  PRESTADOR.CD_PRESTADOR      = MARCACAO.CD_PRESTADOR
AND    SER_DIS.CD_SER_DIS      = MARCACAO.CD_SER_DIS
AND    MARCACAO.CD_MARCACAO    = IT_MARCACAO.CD_MARCACAO
AND    MARCACAO.CD_MULTI_EMPRESA = NVL(DBAMV.PKG_MV2000.LE_EMPRESA, MARCACAO.CD_MULTI_EMPRESA)
AND    IT_MARCACAO.CD_TIP_MAR  = TIP_MAR.CD_TIP_MAR(+)
AND    IT_MARCACAO.CD_PACIENTE = CONTR_MF.CD_PACIENTE
AND    CONTR_MF.CD_PACIENTE     = ITCONTR.CD_PACIENTE
AND    CONTR_MF.CD_REMESSA      = ITCONTR.CD_REMESSA
AND    (IT_MARCACAO.CD_PACIENTE is not null    OR    IT_MARCACAO.NM_PACIENTE is not null )
AND    MARCACAO.CD_PRESTADOR  LIKE  :CAMPO_CD_MEDICO
AND    MARCACAO.DT_MARCACAO BETWEEN  :CAMPO_DT_INIC  AND 
AND    TRUNC(ITCONTR.DT_ALTA) =      ( SELECT MAX(TRUNC(ITCONTR_2.DT_ALTA))
                                         FROM DBAMV.IT_CONTR_MF ITCONTR_2
                                        WHERE ITCONTR_2.CD_PACIENTE = ITCONTR.CD_PACIENTE
                                          AND ITCONTR_2.CD_REMESSA = ITCONTR.CD_REMESSA )
ORDER BY MARCACAO.CD_MARCACAO
        ,PRESTADOR.CD_PRESTADOR
        ,MARCACAO.DT_MARCACAO
        ,to_date(to_char(MARCACAO.HR_INICIAL,'hh24'),'hh24')
        ,IT_MARCACAO.CD_PACIENTE
)
ORDER BY CD_MARCACAO,CD_PRESTADOR,NM_PRESTADOR,DT_MARCACAO,HR_INICIAL,HR_FINAL,CD_SER_DIS,DS_SER_DIS,NR_TEMPO_MEDIO,SN_SIA
