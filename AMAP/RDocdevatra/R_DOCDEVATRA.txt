--R_DOCDEVATRA
SELECT * FROM (
SELECT DISTINCT
               CAD_SAME.CD_CAD_SAME
              ,PROTOCOLOS.CD_PORTADOR
              ,PORTADORES.DS_PORTADOR
              ,PROTOCOLOS.CD_PROTOCOLO
              ,TO_CHAR(PROTOCOLOS.DT_SAIDA, 'DD/MM/YYYY')         DT_SAIDA
              ,TO_CHAR(PROTOCOLOS.HR_SAIDA, 'HH24')            HR_SAIDA
              ,PROTOCOLOS_DEVOLUCAO.DS_MOTIVO
              ,TO_CHAR(PROTOCOLOS.DT_PREV_RETORNO,'DD/MM/YYYY')           DT_PREV_RETORNO
              ,TO_CHAR(PROTOCOLOS_DEVOLUCAO.DT_DEVOLUCAO,'DD/MM/YYYY')    DT_DEVOLUCAO
              ,IT_SAME_PROTOCOLOS.CD_ATENDIMENTO
              ,IT_SAME_PROTOCOLOS.NR_MATRICULA_SAME
             ,PACIENTE.NM_PACIENTE
FROM    DBAMV.SAME
       ,DBAMV.CAD_SAME
       ,DBAMV.ARMARIO_SAME
       ,DBAMV.SALA_SAME
       ,DBAMV.PROTOCOLOS
       ,DBAMV.IT_SAME_PROTOCOLOS
       ,DBAMV.PORTADORES
       ,DBAMV.IT_SAME
       ,DBAMV.ATENDIME
       ,DBAMV.PACIENTE
       ,DBAMV.PROTOCOLOS_DEVOLUCAO
WHERE  TRUNC(PROTOCOLOS_DEVOLUCAO.DT_DEVOLUCAO)    >=    to_date( to_char( :CAMPO_DT_INICIO,
'DD/MM/YYYY' ) || ' 00', 'DD/MM/YYYY HH24' )
  AND  TRUNC(PROTOCOLOS_DEVOLUCAO.DT_DEVOLUCAO)       <=     to_date( to_char( :CAMPO_DT_FIM,
'DD/MM/YYYY' ) || ' 23', 'DD/MM/YYYY HH24' )
  AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
  AND  PROTOCOLOS_DEVOLUCAO.DT_DEVOLUCAO  >  PROTOCOLOS.DT_PREV_RETORNO
  AND  SAME.CD_CAD_SAME = CAD_SAME.CD_CAD_SAME
  AND  SAME.CD_ARMARIO_SAME = ARMARIO_SAME.CD_ARMARIO_SAME(+)
  AND  ARMARIO_SAME.CD_SALA_SAME = SALA_SAME.CD_SALA_SAME(+)
  AND  SAME.CD_CAD_SAME              = IT_SAME.CD_CAD_SAME
  AND  SAME.NR_MATRICULA_SAME        = IT_SAME.NR_MATRICULA_SAME
  AND  IT_SAME.CD_ATENDIMENTO        = ATENDIME.CD_ATENDIMENTO
  AND  IT_SAME_PROTOCOLOS.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
  AND  IT_SAME_PROTOCOLOS.CD_PROTOCOLO = PROTOCOLOS.CD_PROTOCOLO
  AND  IT_SAME_PROTOCOLOS.CD_PROTOCOLO_DEVOLUCAO = PROTOCOLOS_DEVOLUCAO.CD_PROTOCOLO_DEVOLUCAO
  AND  PROTOCOLOS.DT_RETORNO IS NOT NULL
  AND  SAME.CD_CAD_SAME           = IT_SAME_PROTOCOLOS.CD_CAD_SAME
  AND  SAME.NR_MATRICULA_SAME     = IT_SAME_PROTOCOLOS.NR_MATRICULA_SAME
  AND  PROTOCOLOS.CD_PORTADOR        = PORTADORES.CD_PORTADOR
  AND  PACIENTE.CD_PACIENTE          = SAME.CD_PACIENTE
  AND ATENDIME.CD_MULTI_EMPRESA =  DBAMV.PKG_MV2000.LE_EMPRESA
UNION ALL
SELECT DISTINCT
               CAD_SAME.CD_CAD_SAME
              ,PROTOCOLOS.CD_PORTADOR
              ,PORTADORES.DS_PORTADOR
              ,PROTOCOLOS.CD_PROTOCOLO
              ,TO_CHAR(PROTOCOLOS.DT_SAIDA, 'DD/MM/YYYY')         DT_SAIDA
              ,TO_CHAR(PROTOCOLOS.HR_SAIDA, 'HH24')            HR_SAIDA
              ,PROTOCOLOS_DEVOLUCAO.DS_MOTIVO
              ,TO_CHAR(PROTOCOLOS.DT_PREV_RETORNO,'DD/MM/YYYY')           DT_PREV_RETORNO
              ,TO_CHAR(PROTOCOLOS_DEVOLUCAO.DT_DEVOLUCAO,'DD/MM/YYYY')    DT_DEVOLUCAO
              ,0
              ,IT_SAME_PROTOCOLOS.NR_MATRICULA_SAME
             ,PACIENTE.NM_PACIENTE
FROM    DBAMV.SAME
       ,DBAMV.CAD_SAME
       ,DBAMV.ARMARIO_SAME
       ,DBAMV.SALA_SAME
       ,DBAMV.PROTOCOLOS
       ,DBAMV.IT_SAME_PROTOCOLOS
       ,DBAMV.PORTADORES
       ,DBAMV.PACIENTE
       ,DBAMV.PROTOCOLOS_DEVOLUCAO
WHERE  TRUNC(PROTOCOLOS_DEVOLUCAO.DT_DEVOLUCAO)    >=   to_date( to_char( :CAMPO_DT_INICIO, 'DD/MM/YYYY'
) || ' 00', 'DD/MM/YYYY HH24' )
  AND  TRUNC(PROTOCOLOS_DEVOLUCAO.DT_DEVOLUCAO)       <=    to_date( to_char( :CAMPO_DT_FIM,
'DD/MM/YYYY' ) || ' 23', 'DD/MM/YYYY HH24' )
  AND  PROTOCOLOS_DEVOLUCAO.DT_DEVOLUCAO  >  PROTOCOLOS.DT_PREV_RETORNO
  AND  SAME.CD_CAD_SAME = CAD_SAME.CD_CAD_SAME
  AND  SAME.CD_ARMARIO_SAME = ARMARIO_SAME.CD_ARMARIO_SAME(+)
  AND  ARMARIO_SAME.CD_SALA_SAME = SALA_SAME.CD_SALA_SAME(+)
  AND  IT_SAME_PROTOCOLOS.CD_PROTOCOLO = PROTOCOLOS.CD_PROTOCOLO
  AND  IT_SAME_PROTOCOLOS.CD_PROTOCOLO_DEVOLUCAO = PROTOCOLOS_DEVOLUCAO.CD_PROTOCOLO_DEVOLUCAO
  AND  IT_SAME_PROTOCOLOS.cd_atendimento IS NULL
  AND  PROTOCOLOS.DT_RETORNO IS NOT NULL
  AND  SAME.CD_CAD_SAME           = IT_SAME_PROTOCOLOS.CD_CAD_SAME
  AND  SAME.NR_MATRICULA_SAME     = IT_SAME_PROTOCOLOS.NR_MATRICULA_SAME
  AND  PROTOCOLOS.CD_PORTADOR        = PORTADORES.CD_PORTADOR
  AND  PACIENTE.CD_PACIENTE          = SAME.CD_PACIENTE
  :CAMPO_CF_COND_SAME
)
ORDER BY CD_CAD_SAME,CD_PORTADOR,DS_PORTADOR,
CD_PROTOCOLO,DT_SAIDA,HR_SAIDA,
NR_MATRICULA_SAME,DT_PREV_RETORNO,DS_MOTIVO,DT_DEVOLUCAO
