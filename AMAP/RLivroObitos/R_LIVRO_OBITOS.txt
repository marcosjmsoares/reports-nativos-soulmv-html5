SELECT ATENDIME.CD_ATENDIMENTO
              ,ATENDIME.CD_PACIENTE
              ,PACIENTE.NM_PACIENTE
              ,PRESTADOR.NM_PRESTADOR
              ,PRESTADOR.DS_CODIGO_CONSELHO
              ,TIPO_INTERNACAO.DS_TIPO_INTERNACAO
FROM     DBAMV.ATENDIME
               ,DBAMV.PACIENTE
               ,DBAMV.PRESTADOR
               ,DBAMV.MOT_ALT
               ,DBAMV.TIPO_INTERNACAO
WHERE  TRUNC(ATENDIME.DT_ALTA)  =  :CAMPO_DATA
AND         ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND         ATENDIME.TP_ATENDIMENTO  =  'I'
AND         ATENDIME.CD_PACIENTE  =  PACIENTE.CD_PACIENTE
AND         ATENDIME.CD_PRESTADOR  =  PRESTADOR.CD_PRESTADOR(+)
AND         ATENDIME.CD_MOT_ALT  =  MOT_ALT.CD_MOT_ALT
AND         MOT_ALT.TP_MOT_ALTA  =  'O'
AND         ATENDIME.CD_TIPO_INTERNACAO  =  TIPO_INTERNACAO.CD_TIPO_INTERNACAO
AND         ATENDIME.CD_MULTI_EMPRESA =  DBAMV.PKG_MV2000.LE_EMPRESA 
ORDER BY ATENDIME.CD_ATENDIMENTO
