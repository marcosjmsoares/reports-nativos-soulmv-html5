select cad_same.ds_cad_same ds_cad_same,
          paciente.nm_paciente nm_paciente,
          same.nr_matricula_same nr_matricula_same,
          same.nr_volume nr_volume,
          atendime.cd_atendimento cd_atendimento,
          convenio.nm_convenio nm_convenio,
          to_char(atendime.dt_atendimento,'dd/mm/yyyy') dt_atendimento,
          sala_same.ds_sala_same ds_sala_same,
          same.ds_coluna ds_coluna,
          armario_same.ds_armario_same ds_armario_same,
          same.ds_linha ds_linha
  from dbamv.paciente,
         dbamv.same,
         dbamv.ori_same,
         dbamv.atendime,
         dbamv.armario_same,
         dbamv.sala_same,
         dbamv.cad_same,
         dbamv.convenio,
         dbamv.Empresa_Convenio,
         dbamv.it_same
 where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
   and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
   And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
  and it_same.cd_atendimento = atendime.cd_atendimento
  and it_same.nr_matricula_same = same.nr_matricula_same
  and it_same.nr_volume = same.nr_volume
  and paciente.cd_paciente = same.cd_paciente
   and atendime.cd_multi_empresa =  :CAMPO_CD_MULTI_EMPRESA 
   and atendime.cd_paciente = paciente.cd_paciente
   and same.cd_armario_same = armario_same.cd_armario_same(+)
   and armario_same.cd_sala_same = sala_same.cd_sala_same(+)
   and same.cd_cad_same = cad_same.cd_cad_same
   and ori_same.cd_ori_ate = atendime.cd_ori_ate
   and ori_same.cd_cad_same = same.cd_cad_same
   and convenio.cd_convenio = atendime.cd_convenio
   and atendime.cd_atendimento
           IN (SELECT cd_atendimento FROM dbamv.it_same)
 order by atendime.cd_atendimento
