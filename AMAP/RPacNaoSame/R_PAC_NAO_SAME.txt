SELECT DISTINCT(ATENDIME.CD_PACIENTE),
                PACIENTE.NM_PACIENTE,
                ATENDIME.CD_ATENDIMENTO,
                ATENDIME.DT_ATENDIMENTO,
                ATENDIME.TP_ATENDIMENTO,
                TO_CHAR(ATENDIME.HR_ATENDIMENTO, 'HH24')  HR_ATENDIMENTO,
                ATENDIME.CD_CONVENIO,
                ATENDIME.CD_PRESTADOR,
                PRESTADOR.NM_PRESTADOR,
                ATENDIME.CD_ORI_ATE,
                ORI_ATE.DS_ORI_ATE,
                CONVENIO.NM_CONVENIO,
                PACIENTE.NR_FONE
FROM DBAMV.ATENDIME,
     DBAMV.PACIENTE,
     DBAMV.PRESTADOR,
     DBAMV.ORI_ATE,
     DBAMV.CONVENIO,
     DBAMV.EMPRESA_CONVENIO
WHERE Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND ATENDIME.CD_MULTI_EMPRESA =  DBAMV.PKG_MV2000.LE_EMPRESA 
AND   PACIENTE.CD_PACIENTE = ATENDIME.CD_PACIENTE
AND   ORI_ATE.CD_ORI_ATE = ATENDIME.CD_ORI_ATE
AND   ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND   PRESTADOR.CD_PRESTADOR(+) = ATENDIME.CD_PRESTADOR
AND      TRUNC( ATENDIME.DT_ATENDIMENTO)  BETWEEN :CAMPO_DATAINI  AND  
AND  NOT EXISTS(SELECT it_same.cd_atendimento
                 FROM  dbamv.it_same
                 WHERE it_same.cd_atendimento = atendime.cd_atendimento)
