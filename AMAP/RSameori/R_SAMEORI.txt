--R_SAMEORI
SELECT * FROM (
SELECT  cad_same.cd_cad_same,
        cad_same.ds_cad_same,
        ori_ate.cd_ori_ate  ,
        ori_ate.ds_ori_ate  ,
        ori_ate.tp_origem
FROM    dbamv.cad_same      ,
        dbamv.ori_ate       ,
        dbamv.ori_same
WHERE   ori_same.cd_ori_ate  = ori_ate.cd_ori_ate
  AND   ori_same.cd_cad_same = cad_same.cd_cad_same
  AND   ori_ate.cd_multi_empresa = nvl(dbamv.pkg_mv2000.le_empresa, ori_ate.cd_multi_empresa)
  AND   cad_same.cd_cad_same LIKE :CAMPO_SAME
  AND   ori_ate.cd_ori_ate   LIKE :CAMPO_ORIGEM
)
ORDER BY CD_CAD_SAME,DS_CAD_SAME
