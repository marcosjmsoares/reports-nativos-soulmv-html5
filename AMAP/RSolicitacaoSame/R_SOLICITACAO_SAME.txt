--R_SOLICITACAO_SAME
SELECT  SOLICITACAO_SAME.CD_CAD_SAME , SOLICITACAO_SAME.CD_SOLICITACAO , SOLICITACAO_SAME.DT_SOLICITACAO , SOLICITACAO_SAME.CD_MOTIVO_SOLICITACAO , SOLICITACAO_SAME.DS_JUSTIFICA_FINALIZA_CANCELA , SOLICITACAO_SAME.CD_USUARIO_SOLICITANTE , SOLICITACAO_SAME.CD_PORTADOR_SOLICITANTE , SOLICITACAO_SAME.CD_PORTADOR_ATENDIMENTO
      , decode(SOLICITACAO_SAME.TP_SITUACAO
      ,'A', :CAMPO_TOT_ATENDIDA
      ,'R', :CAMPO_PARC_ATENDIDA
      ,'F', :CAMPO_FINALIZADA
      ,'C', :CAMPO_CANCELADA
      ,'P', :CAMPO_PENDENTE ) TP_SITUACAO
      , MOTIVO_SOL_SAME.DS_MOTIVO
      , (SELECT PORTADORES.DS_PORTADOR FROM DBAMV.PORTADORES WHERE PORTADORES.CD_PORTADOR = SOLICITACAO_SAME.CD_PORTADOR_SOLICITANTE) DS_PORTADOR_DESTINO
      , (SELECT PORTADORES.DS_PORTADOR FROM DBAMV.PORTADORES WHERE PORTADORES.CD_PORTADOR = SOLICITACAO_SAME.CD_PORTADOR_ATENDIMENTO) DS_PORTADOR_ORIGEM
  FROM  DBAMV.SOLICITACAO_SAME
      , DBAMV.MOTIVO_SOL_SAME
    WHERE SOLICITACAO_SAME.CD_SOLICITACAO = :CAMPO_CD_SOLICITACAO_CME
    AND SOLICITACAO_SAME.CD_MOTIVO_SOLICITACAO = MOTIVO_SOL_SAME.CD_MOTIVO
