--R_PESQATE
SELECT ATEND.CD_ATENDIMENTO,
					   ATEND.CD_ORI_ATE,
					   ATEND.DS_ORI_ATE,
					   ATEND.CD_PACIENTE,
					   ATEND.NM_PACIENTE,
					   ATEND.DT_NASCIMENTO,
					   ATEND.TP_SEXO,
					   ATEND.DT_ATENDIMENTO,
					   ATEND.CD_PRESTADOR,
					   ATEND.NM_PRESTADOR
				  FROM (
				SELECT ATENDIME.CD_ATENDIMENTO,
						ATENDIME.CD_ORI_ATE,
						ORI_ATE.DS_ORI_ATE,
						ATENDIME.CD_PACIENTE,
						PACIENTE.NM_PACIENTE,
						PACIENTE.DT_NASCIMENTO,
						PACIENTE.TP_SEXO,
						ATENDIME.DT_ATENDIMENTO,
						ATENDIME.CD_PRESTADOR,
						PRESTADOR.NM_PRESTADOR,
						ATENDIME.CD_MULTI_EMPRESA   
				  FROM  DBAMV.ATENDIME, DBAMV.PRESTADOR, DBAMV.PACIENTE,DBAMV.ORI_ATE
				 WHERE  ATENDIME.CD_PACIENTE  = PACIENTE.CD_PACIENTE
				   AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
				   AND  ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
				   AND  ATENDIME.CD_MULTI_EMPRESA =  DBAMV.PKG_MV2000.LE_EMPRESA
				   AND  ATENDIME.CD_ORI_ATE   = ORI_ATE.CD_ORI_ATE)ATEND
				ORDER BY ATEND.CD_ATENDIMENTO
