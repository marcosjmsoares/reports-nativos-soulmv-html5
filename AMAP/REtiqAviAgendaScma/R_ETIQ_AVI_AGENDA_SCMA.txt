--R_ETIQ_AVI_AGENDA_SCMA
select it_agenda_central.cd_paciente  cd_paciente
      ,it_agenda_central.nm_paciente nm_paciente
      ,prestador.nm_prestador
      ,ser_dis.cd_ser_dis cd_ser_dis
      ,tip_mar.ds_tip_mar ds_tip_mar
      ,to_char(agenda_central.dt_agenda,'dd/mm/yyyy') dt_agenda
      ,to_char(it_agenda_central.hr_agenda,'hh24') hr_agenda
      ,unidade_atendimento.ds_unidade_atendimento ds_unidade_atendimento
      ,agenda_central.cd_recurso_central cd_recurso_central
      ,agenda_central.tp_agenda tp_agenda
  from dbamv.agenda_central
      ,dbamv.it_agenda_central
      ,dbamv.tip_mar
      ,dbamv.ser_dis
      ,dbamv.prestador
      ,dbamv.unidade_atendimento
 where agenda_central.cd_agenda_central      = it_agenda_central.cd_agenda_central
   and it_agenda_central.cd_tip_mar          = tip_mar.cd_tip_mar(+)
   and it_agenda_central.cd_ser_dis          = ser_dis.cd_ser_dis(+)
   and agenda_central.cd_prestador           = prestador.cd_prestador(+)
   and agenda_central.cd_unidade_atendimento = unidade_atendimento.cd_unidade_atendimento(+)
   and it_agenda_central.cd_it_agenda_central   = :CAMPO_CD_IT_AGENDA_CENTRAL
   and it_agenda_central.nm_paciente is not null
