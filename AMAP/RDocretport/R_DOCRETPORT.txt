--R_DOCRETPORT
SELECT * FROM (
SELECT p.cd_protocolo
      ,p.hr_saida dt_saida
      ,p.dt_prev_retorno
      ,p.ds_motivo
      ,p.cd_portador cd_portador_origem
      ,o.ds_portador ds_portador_origem
      ,ip.nr_matricula_same
      ,ip.nr_volume
      ,s.ds_linha
      ,s.ds_coluna
      ,a.ds_armario_same
      ,its.cd_atendimento
      ,its.dt_entrada
      ,pac.nm_paciente
  FROM dbamv.protocolos p
      ,dbamv.portadores o
      ,dbamv.it_same_protocolos ip
      ,dbamv.same s
      ,dbamv.it_same its
      ,dbamv.armario_same a
      ,dbamv.paciente pac
 WHERE p.cd_portador          = o.cd_portador
   AND p.cd_protocolo            = ip.cd_protocolo
   AND s.nr_matricula_same    = ip.nr_matricula_same
   AND s.nr_volume                 = ip.nr_volume
   AND s.cd_cad_same           = ip.cd_cad_same
   AND s.cd_paciente             = pac.cd_paciente
   AND s.cd_armario_same     = a.cd_armario_same(+)
   AND s.nr_matricula_same   = its.nr_matricula_same(+)
   AND s.nr_volume                = its.nr_volume(+)
   AND s.cd_cad_same          = its.cd_cad_same(+)
   AND p.dt_saida between to_date( to_char( :CAMPO_DT_INICIO, 'DD/MM/YYYY' ) || ' 00', 'DD/MM/YYYY HH24' ) and  to_date( to_char( :CAMPO_DT_FIM,    'DD/MM/YYYY' ) || ' 23', 'DD/MM/YYYY HH24' )
:CAMPO_CF_COND_PORTADOR
)
ORDER BY CD_PORTADOR_ORIGEM,DS_PORTADOR_ORIGEM,
CD_PROTOCOLO,DS_MOTIVO,DT_SAIDA,DT_PREV_RETORNO,
NR_MATRICULA_SAME,NR_VOLUME,DS_ARMARIO_SAME,DS_LINHA,DS_COLUNA
