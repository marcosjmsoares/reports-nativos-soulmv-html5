--R_ENCAIXE
SELECT TO_CHAR(HORARIO.HORA_AGENDA,'HH24') HORA_AGENDA
             ,SUBSTR(SAME.NR_MATRICULA_SAME,-2,1) DIGITO
             ,SAME.NR_MATRICULA_SAME   MATRICULA
             ,PRESTADOR.NM_PRESTADOR
             ,RECURSO.NM_RECURSO RECURSO_IMG
             ,SRECURSO.NM_RECURSO RECURSO_LAB
             ,UNIDADE_REGIONAL.DS_UNIDADE_REGIONAL
             ,SER_DIS.DS_SER_DIS
             ,PLANTAO.TIPO_AGENDA
             ,PLANTAO.CD_RECURSO
             ,HORARIO.CD_PACIENTE
             ,HORARIO.NM_PACIENTE
 FROM DBAMV.V_CENTRAL_MARCACAO_PLANTAO PLANTAO
           ,DBAMV.V_CENTRAL_MARCACAO_HORARIO HORARIO
           ,DBAMV.RECURSO
           ,DBAMV.SRECURSO
           ,DBAMV.UNIDADE_REGIONAL
           ,DBAMV.SER_DIS
           ,DBAMV.SAME
           ,DBAMV.PRESTADOR
           ,DBAMV.TIP_MAR
 WHERE PLANTAO.CODIGO_AGENDA           = HORARIO.CODIGO_AGENDA
  AND  PLANTAO.TIPO_AGENDA                     = HORARIO.TIPO_AGENDA
  AND  PLANTAO.CD_RECURSO                      = RECURSO.CD_RECURSO(+)
  AND  PLANTAO.CD_RECURSO                      = SRECURSO.CD_RECURSO(+)
  AND  PLANTAO.CD_PRESTADOR                 = PRESTADOR.CD_PRESTADOR(+)
  AND  PLANTAO.CD_SER_DIS                        = HORARIO.CD_SER_DIS(+)
  AND  PLANTAO.CD_SER_DIS                        = SER_DIS.CD_SER_DIS(+)
  AND  PLANTAO.CD_UNIDADE_REGIONAL   = UNIDADE_REGIONAL.CD_UNIDADE_REGIONAL(+)
  AND  PLANTAO.SN_ATIVO                             = 'S'
 AND  PLANTAO.CD_TIP_MAR                        = HORARIO.CD_TIP_MAR(+)
 AND  HORARIO.CD_TIP_MAR                        = TIP_MAR.CD_TIP_MAR(+)
 AND TRUNC(PLANTAO.DT_MARCACAO)    BETWEEN TO_DATE(:CAMPO_PDT_INI,'DD/MM/YYYY')   AND TO_DATE(:CAMPO_PDT_FIM,'DD/MM/YYYY')
 AND HORARIO.NM_PACIENTE IS NOT NULL
AND (HORARIO.SN_ENCAIXE = 'S'
OR TRUNC(PLANTAO.DT_MARCACAO) = TRUNC(HORARIO.DATA_DA_MARCACAO))
