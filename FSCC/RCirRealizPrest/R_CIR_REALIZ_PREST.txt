--R_CIR_REALIZ_PREST
select distinct a.cd_cen_cir
               ,trunc(a.dt_realizacao) dia
               ,count(a.dt_realizacao) total_dia
               ,p.cd_prestador
               ,p.nm_prestador
               ,ati_med.ds_ati_med
               ,cen_cir.ds_cen_cir
from dbamv.aviso_cirurgia a,
     dbamv.cirurgia_aviso b,
     dbamv.prestador p,
     dbamv.prestador_aviso pa,
     dbamv.ati_med   ati_med,
     dbamv.cen_cir    cen_cir
where a.tp_situacao = 'R'
  AND        A.CD_AVISO_CIRURGIA   =  B.CD_AVISO_CIRURGIA
  AND        B.CD_AVISO_CIRURGIA    =  PA.CD_AVISO_CIRURGIA
  AND        P.CD_MULTI_EMPRESA     =  DBAMV.PKG_MV2000.LE_EMPRESA
  AND        B.CD_CIRURGIA                =  PA.CD_CIRURGIA
  and p.cd_prestador         = pa.cd_prestador
  and pa.cd_ati_med         = ati_med.cd_ati_med
  and a.cd_cen_cir            = cen_cir.cd_cen_cir
  and a.dt_realizacao BETWEEN To_Date(Concat(To_Char(:CAMPO_DATA_INI,'dd/mm/yyyy'),'00'),'dd/mm/yyyy hh24') and  To_Date(Concat(To_Char(:CAMPO_DATA_FIM,'dd/mm/yyyy'),'23'),'dd/mm/yyyy hh24')
group by  a.cd_cen_cir,cen_cir.ds_cen_cir, p.cd_prestador,p.nm_prestador,trunc(a.dt_realizacao),ati_med.ds_ati_med
order by p.cd_prestador,dia
