--R_CIR_TRANSF
SELECT * FROM
  (SELECT TRUNC(TRANSF.DT_TRANSFERENCIA) DT_TRANSFERENCIA ,
          AVISO.CD_AVISO_CIRURGIA ,
          TRANSF.DT_ANTERIOR_AGENDAMENTO ,
          MOTIVO.DS_MOT_TRANSFERENCIA ,
          AGE_CIR.DT_INICIO_AGE_CIR NOVA_DATA ,
          CIRURGIA.DS_CIRURGIA ,
          NVL(PACIENTE.NM_PACIENTE, AVISO.NM_PACIENTE) NM_PACIENTE ,
     (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
          TRANSF.CD_TRANSFERENCIA_CIRURGIA ,
          CIRURGIA.CD_CIRURGIA ,
          Nvl(TRANSF.NM_SOLICITANTE, Transf.Nm_Usuario) NM_SOLICITANTE
   FROM DBAMV.AVISO_CIRURGIA AVISO ,
        DBAMV.TRANSFERENCIA_CIRURGIA TRANSF ,
        DBAMV.MOT_TRANSFERENCIA MOTIVO ,
        DBAMV.AGE_CIR ,
        DBAMV.CIRURGIA_AVISO CIR_AVI ,
        DBAMV.CIRURGIA ,
        DBAMV.PACIENTE
   WHERE AVISO.CD_AVISO_CIRURGIA = TRANSF.CD_AVISO_CIRURGIA
     AND TRANSF.CD_MOT_TRANSFERENCIA = MOTIVO.CD_MOT_TRANSFERENCIA
     AND AVISO.CD_AVISO_CIRURGIA = AGE_CIR.CD_AVISO_CIRURGIA
     AND AVISO.CD_AVISO_CIRURGIA = CIR_AVI.CD_AVISO_CIRURGIA
     AND CIR_AVI.CD_CIRURGIA = CIRURGIA.CD_CIRURGIA
     AND AVISO.CD_PACIENTE = PACIENTE.CD_PACIENTE(+)
     AND TRUNC(TRANSF.DT_TRANSFERENCIA)>= :CAMPO_DATA_INI
     AND TRUNC(TRANSF.DT_TRANSFERENCIA)<= :CAMPO_DATA_FIM  :CAMPO_CF_MOTIVO )
ORDER BY DT_TRANSFERENCIA
