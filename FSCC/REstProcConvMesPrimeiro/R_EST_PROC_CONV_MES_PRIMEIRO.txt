      ,gru_pro.ds_gru_pro
      ,cir_avi.cd_convenio
      ,convenio.nm_convenio

      ,count(decode(to_char(avi_cir.dt_realizacao,'dd'),2,1))  dia_2

from  dbamv.gru_pro            gru_pro
     ,dbamv.pro_fat                pro_fat
     ,dbamv.cirurgia               cirurgia
     ,dbamv.aviso_cirurgia     avi_cir
     ,dbamv.cirurgia_aviso     cir_avi
     ,dbamv.convenio            convenio
     ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = dbamv.pkg_mv2000.le_empresa
And avi_cir.tp_situacao          = 'R'
and   avi_cir.cd_aviso_cirurgia  = cir_avi.cd_aviso_cirurgia
and   cir_avi.cd_cirurgia            = cirurgia.cd_cirurgia
and   cirurgia.cd_pro_fat            = pro_fat.cd_pro_fat
and   pro_fat.cd_gru_pro           = gru_pro.cd_gru_pro
and   cir_avi.cd_convenio         = convenio.cd_convenio
and  to_char(avi_cir.dt_realizacao,'mm/yyyy') = :CAMPO_DATA_INI
group by  avi_cir.cd_aviso_cirurgia
      , gru_pro.cd_gru_pro
      , gru_pro.ds_gru_pro
      , cir_avi.cd_convenio
      , convenio.nm_convenio
      , to_char(avi_cir.dt_realizacao,'dd')
)a
 group by a.cd_gru_pro
         ,a.ds_gru_pro
         ,a.cd_convenio
         ,a.nm_convenio
order by  a.ds_gru_pro, a.nm_convenio
