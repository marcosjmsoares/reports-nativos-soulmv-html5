--R_CIR_CANCE
SELECT * FROM
  (SELECT DISTINCT AVISO_CIRURGIA.CD_PACIENTE CD_PACIENTE,
                   AVISO_CIRURGIA.CD_AVISO_CIRURGIA CD_AVISO_CIRURGIA,
                   AVISO_CIRURGIA.NM_PACIENTE NM_PACIENTE,
          (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND AVISO_CIRURGIA.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE,
                   TRUNC(AVISO_CIRURGIA.DT_CANCELAMENTO) DT_CANCELAMENTO,
                   MOT_CANC.CD_MOT_CANC CD_MOT_CANC,
                   DECODE(MOT_CANC.DS_MOT_CANC, NULL, 'NAO INFORMADO', MOT_CANC.DS_MOT_CANC) DS_MOT_CANC,
                   PRESTADOR_AVISO.CD_PRESTADOR CD_PRESTADOR,
                   DECODE(PRESTADOR.NM_PRESTADOR, NULL, 'NAO INFORMADO', PRESTADOR.NM_PRESTADOR) NM_PRESTADOR,
                   CIRURGIA_AVISO.CD_CIRURGIA CD_CIRURGIA,
                   DECODE(CIRURGIA_AVISO.DS_NPADRONIZADO, NULL , CIRURGIA.DS_CIRURGIA , 'N.P.' || CIRURGIA_AVISO.DS_NPADRONIZADO) DS_CIRURGIA,
                   CONVENIO.NM_CONVENIO NM_CONVENIO,
                   AVISO_CIRURGIA.DS_MOTIVO_CANCELAMENTO DS_MOTIVO_CANCELAMENTO,
                   TO_CHAR(AVISO_CIRURGIA.DT_CANCELAMENTO, 'HH24') HR_CANCELAMENTO,
                   AVISO_CIRURGIA.CD_USUARIO_CANCEL CD_USUARIO_CANCEL
   FROM DBAMV.AVISO_CIRURGIA,
        DBAMV.PRESTADOR_AVISO,
        DBAMV.ATI_MED,
        DBAMV.PRESTADOR,
        DBAMV.CIRURGIA_AVISO,
        DBAMV.MOT_CANC,
        DBAMV.CIRURGIA,
        DBAMV.CONVENIO,
        Dbamv.Empresa_Convenio
   WHERE Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
     AND AVISO_CIRURGIA.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
     AND AVISO_CIRURGIA.TP_SITUACAO = 'C'
     AND TRUNC(AVISO_CIRURGIA.DT_CANCELAMENTO) BETWEEN :CAMPO_DATA_INI AND 
     AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
     AND AVISO_CIRURGIA.CD_MOT_CANC = MOT_CANC.CD_MOT_CANC(+)
     AND CIRURGIA_AVISO.CD_CIRURGIA = CIRURGIA.CD_CIRURGIA
     AND CIRURGIA_AVISO.CD_CONVENIO = CONVENIO.CD_CONVENIO(+)
     AND CIRURGIA_AVISO.CD_CIRURGIA_AVISO = PRESTADOR_AVISO.CD_CIRURGIA_AVISO(+)
     AND PRESTADOR_AVISO.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
     AND PRESTADOR_AVISO.CD_ATI_MED = ATI_MED.CD_ATI_MED(+)
     AND (ATI_MED.TP_FUNCAO = 'C'
          OR PRESTADOR.CD_PRESTADOR IS NULL)    
   ORDER BY DT_AVISO_CIRURGIA,
            NM_PRESTADOR,
            DS_MOT_CANC)
ORDER BY DT_CANCELAMENTO,
         HR_CANCELAMENTO,
         CD_AVISO_CIRURGIA,
         CD_USUARIO_CANCEL,
         CD_PACIENTE,
         NM_PACIENTE,
         CD_MOT_CANC,
         DS_MOT_CANC,
         DS_MOTIVO_CANCELAMENTO
