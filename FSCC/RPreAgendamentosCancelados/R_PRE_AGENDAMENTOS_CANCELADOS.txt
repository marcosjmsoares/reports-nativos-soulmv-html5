--R_PRE_AGENDAMENTOS_CANCELADOS
SELECT aviso_cirurgia.cd_aviso_cirurgia,
       aviso_cirurgia.dt_aviso_cirurgia,
       aviso_cirurgia.cd_paciente,
       aviso_cirurgia.nm_paciente,
   (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND AVISO_CIRURGIA.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE,
       prestador_aviso.cd_prestador,
       prestador.nm_prestador,
       cirurgia_aviso.cd_cirurgia,
       cirurgia.ds_cirurgia,
       cirurgia_aviso.cd_convenio,
       convenio.nm_convenio
FROM dbamv.aviso_cirurgia,
     dbamv.cirurgia_aviso,
     dbamv.prestador_aviso,
     dbamv.cirurgia,
     dbamv.convenio,
     dbamv.prestador
WHERE aviso_cirurgia.cd_aviso_cirurgia = cirurgia_aviso.cd_aviso_cirurgia
  AND prestador_aviso.cd_aviso_cirurgia = cirurgia_aviso.cd_aviso_cirurgia
  AND prestador_aviso.cd_cirurgia_aviso = cirurgia_aviso.cd_cirurgia_aviso
  AND prestador_aviso.cd_prestador = prestador.cd_prestador
  AND cirurgia_aviso.cd_cirurgia = cirurgia.cd_cirurgia
  AND cirurgia_aviso.cd_convenio = convenio.cd_convenio
  AND cirurgia_aviso.sn_principal = 'S'
  AND prestador_aviso.sn_principal = 'S'
  AND AVISO_CIRURGIA.TP_SITUACAO='C'
  AND aviso_cirurgia.dt_pre_agendamento BETWEEN trunc(:CAMPO_DT_INICIAL) AND trunc(:CAMPO_DT_FINAL)  
  ORDER  BY dt_aviso_cirurgia DESC
