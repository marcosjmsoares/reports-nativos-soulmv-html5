--R_TOTAL_CIRURGIAS
SELECT * FROM (
SELECT  TO_CHAR(data,'MM/YY') DT_CIRURGIA
       , TAB.CDPRESTADOR           CDPRESTADOR
       ,NM_PRESTADOR          NM_PRESTADOR
       ,SUM(TOT_REALIZ)        TOT_REALIZ
      , COUNT(*)                        NUM_REGISTROS
FROM
(
SELECT  AVISO_CIRURGIA.CD_AVISO_CIRURGIA                                           cd_avi_cirurgia
       ,F_VERIF_PRESTADOR(AVISO_CIRURGIA.CD_AVISO_CIRURGIA, CIRURGIA_AVISO.CD_CIRURGIA ) cdprestador
       ,AVISO_CIRURGIA.TP_SITUACAO   tp_situacao
       ,AVISO_CIRURGIA.DT_REALIZACAO data
       ,AVISO_CIRURGIA.CD_CEN_CIR    cd_cen_cir
       ,AGE_CIR.CD_SAL_CIR                CD_SAL_CIR
       ,CONVENIO.TP_CONVENIO         TP_CONVENIO
       ,CONVENIO.CD_CONVENIO         CD_CONVENIO
       ,CONVENIO.NM_CONVENIO         NM_CONVENIO
       ,1                            TOT_REALIZ
FROM    DBAMV.AVISO_CIRURGIA
       ,DBAMV.CIRURGIA_AVISO
       ,DBAMV.AGE_CIR
       ,DBAMV.CONVENIO
       ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And AVISO_CIRURGIA.CD_AVISO_CIRURGIA  = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
AND     AVISO_CIRURGIA.CD_AVISO_CIRURGIA  = AGE_CIR.CD_AVISO_CIRURGIA(+)
AND     CIRURGIA_AVISO.CD_CONVENIO        = CONVENIO.CD_CONVENIO(+)
AND     AVISO_CIRURGIA.TP_SITUACAO        <> 'A'
AND     AVISO_CIRURGIA.TP_SITUACAO        = 'R'
 ) TAB
, DBAMV.PRESTADOR     PRESTADOR
,DBAMV.SAL_CIR      SAL_CIR
,DBAMV.CEN_CIR     CEN_CIR
WHERE  DATA BETWEEN :CAMPO_PER_INI AND 
AND TAB.CD_CEN_CIR = CEN_CIR.CD_CEN_CIR
AND    NVL(CD_PRESTADOR,0) <> 0
AND TAB.CDPRESTADOR = PRESTADOR.CD_PRESTADOR
AND TAB.CD_SAL_CIR = SAL_CIR.CD_SAL_CIR
GROUP BY   TO_CHAR(data,'MM/YY')
                     ,CDPRESTADOR
                     ,NM_PRESTADOR
ORDER BY  TO_CHAR(data,'MM/YY') , TOT_REALIZ
)
ORDER BY CDPRESTADOR,NM_PRESTADOR,
DT_CIRURGIA
