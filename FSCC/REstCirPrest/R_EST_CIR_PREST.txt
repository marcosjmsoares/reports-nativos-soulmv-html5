--R_EST_CIR_PREST
SELECT * FROM (
SELECT  CD_CEN_CIR
               ,DS_CEN_CIR
               ,CD_PRESTADOR
	       ,NM_PRESTADOR
               ,DS_PRO_FAT
               ,SUM(TOT_REALIZ)  TOT_REALIZ
               ,SUM(TOT_CANCE)   TOT_CANCE
               ,SUM(TOT_AGEND)   TOT_AGEND
FROM
           (SELECT  V.CD_CEN_CIR     			CD_CEN_CIR
                           ,V.DS_CEN_CIR     			DS_CEN_CIR
                           ,V.CD_PRESTADOR   			CD_PRESTADOR
			   ,P.NM_PRESTADOR                      NM_PRESTADOR
                           ,V.CD_PRO_FAT     			CD_PRO_FAT
                           ,V.DS_CIRURGIA     			DS_PRO_FAT
                           ,DECODE( V.TP_SITUACAO, 'R', 1, 0) 	TOT_REALIZ
                           ,DECODE( V.TP_SITUACAO, 'C', 1, 0) 	TOT_CANCE
                           , DECODE( V.TP_SITUACAO, 'G', 1, 0) 	TOT_AGEND
               FROM    DBAMV.V_CIR_PROCED  V, DBAMV.PRESTADOR P
            WHERE
      P.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA AND
	    P.CD_PRESTADOR = V.CD_PRESTADOR AND
	    V.DATA   BETWEEN To_Date(Concat(To_Char(:CAMPO_DT_INICIO,'dd/mm/yyyy'),'00'),'dd/mm/yyyy hh24') and  To_Date(Concat(To_Char(:CAMPO_DT_FIM,'dd/mm/yyyy'),'23'),'dd/mm/yyyy hh24')
            :CAMPO_CF_COND_CONVENIO
           )
GROUP BY  CD_CEN_CIR
                    ,DS_CEN_CIR
                    ,CD_PRESTADOR
                    ,NM_PRESTADOR
                    ,DS_PRO_FAT
ORDER BY  DS_CEN_CIR
                    ,DS_PRO_FAT
)
ORDER BY DS_CEN_CIR,CD_CEN_CIR,
CD_PRESTADOR
