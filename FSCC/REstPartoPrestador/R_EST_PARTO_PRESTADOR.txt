--R_EST_PARTO_PRESTADOR
SELECT * FROM (
select 	cd_convenio,
	nm_convenio,
	cd_prestador,
	nm_prestador,
	cd_tip_parto,
	ds_tip_parto,
	count(*) 	quantidade,
	(select sum(count(*))	tot_quantidade
from
(
select 	tip.cd_tip_parto,
       	tip.ds_tip_parto,
       	decode(:CAMPO_QUEBRA_CONV, 'S', con.cd_convenio, null) 	cd_convenio,
       	decode(:CAMPO_QUEBRA_CONV, 'S', con.nm_convenio, null)	nm_convenio,
       	pres.cd_prestador,
       	decode(pres.nm_prestador, null, 'Não Informado', pres.nm_prestador) nm_prestador
from   	dbamv.admissao_co	ad,
	dbamv.tip_parto    		tip,
       	dbamv.atendime     		at,
       	dbamv.paciente     		pac,
       	dbamv.convenio     		con,
       	dbamv.prestador    		pres
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Con.Cd_Convenio
And  at.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And   ad.cd_tip_parto	= tip.cd_tip_parto
and    trunc(ad.dt_parto) between :CAMPO_DATA_INI and 
and    ad.cd_atendimento	= at.cd_atendimento
and    at.cd_paciente	= pac.cd_paciente
and    at.cd_convenio	= con.cd_convenio
and    at.cd_prestador	= pres.cd_prestador(+)
:CAMPO_CF_CD_TIP_PARTO
)
group by
	cd_convenio,
	nm_convenio,
	cd_prestador,
	nm_prestador,
	cd_tip_parto,
	ds_tip_parto) TOT_QTD
from
(
select 	tip.cd_tip_parto,
       	tip.ds_tip_parto,
       	decode(:CAMPO_QUEBRA_CONV, 'S', con.cd_convenio, null) 	cd_convenio,
       	decode(:CAMPO_QUEBRA_CONV, 'S', con.nm_convenio, null)	nm_convenio,
       	pres.cd_prestador,
       	decode(pres.nm_prestador, null, 'Não Informado', pres.nm_prestador) nm_prestador
from   	dbamv.admissao_co	ad,
	dbamv.tip_parto    		tip,
       	dbamv.atendime     		at,
       	dbamv.paciente     		pac,
       	dbamv.convenio     		con,
       	dbamv.prestador    		pres
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Con.Cd_Convenio
And  at.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And   ad.cd_tip_parto	= tip.cd_tip_parto
and    trunc(ad.dt_parto) between :CAMPO_DATA_INI and 
and    ad.cd_atendimento	= at.cd_atendimento
and    at.cd_paciente	= pac.cd_paciente
and    at.cd_convenio	= con.cd_convenio
and    at.cd_prestador	= pres.cd_prestador(+)
:CAMPO_CF_CD_TIP_PARTO
)
group by
	cd_convenio,
	nm_convenio,
	cd_prestador,
	nm_prestador,
	cd_tip_parto,
	ds_tip_parto
ORDER BY quantidade desc
)
ORDER BY NM_CONVENIO,CD_CONVENIO,
NM_PRESTADOR,CD_PRESTADOR,
DS_TIP_PARTO,CD_TIP_PARTO
