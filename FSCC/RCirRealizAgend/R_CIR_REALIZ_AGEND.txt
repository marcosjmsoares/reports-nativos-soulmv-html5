--R_CIR_REALIZ_AGEND
SELECT * FROM
  (SELECT aviso_cirurgia.cd_atendimento ,
          aviso_cirurgia.tp_situacao ,
          atendime.cd_paciente ,
          paciente.nm_paciente ,
      (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
          atendime.dt_atendimento ,
          atendime.dt_alta ,
          atendime.cd_convenio ,
          convenio.nm_convenio ,
          aviso_cirurgia.cd_aviso_cirurgia ,
          age_cir.dt_inicio_age_cir ,
          aviso_cirurgia.dt_inicio_cirurgia ,
          aviso_cirurgia.cd_cen_cir ,
          cen_cir.ds_cen_cir ,
          aviso_cirurgia.cd_sal_cir ,
          sal_cir.ds_sal_cir ,
          aviso_cirurgia.cd_equipe_medica ,
          equipe_medica.ds_equipe_medica
   FROM dbamv.aviso_cirurgia ,
        dbamv.paciente ,
        dbamv.atendime ,
        dbamv.convenio ,
        dbamv.age_cir ,
        dbamv.cen_cir ,
        dbamv.sal_cir ,
        dbamv.equipe_medica
   WHERE aviso_cirurgia.cd_atendimento = atendime.cd_atendimento(+)
     AND aviso_cirurgia.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA 
     AND paciente.cd_paciente(+) = aviso_cirurgia.cd_paciente
     AND atendime.cd_convenio = convenio.cd_convenio (+)
     AND age_cir.cd_aviso_cirurgia = aviso_cirurgia.cd_aviso_cirurgia
     AND aviso_cirurgia.cd_cen_cir = cen_cir.cd_cen_cir
     AND aviso_cirurgia.cd_sal_cir = sal_cir.cd_sal_cir(+)
     AND aviso_cirurgia.cd_equipe_medica = equipe_medica.cd_equipe_medica(+)
     AND aviso_cirurgia.dt_aviso_cirurgia BETWEEN To_Date(Concat(To_Char(:CAMPO_DT_INICIAL,'dd/mm/yyyy'), '00'), 'dd/mm/yyyy hh24') AND To_Date(Concat(To_Char(:CAMPO_DT_FINAL,'dd/mm/yyyy'), '23'), 'dd/mm/yyyy hh24')    
   ORDER BY ds_cen_cir,
            cd_atendimento,
            cd_aviso_cirurgia)
