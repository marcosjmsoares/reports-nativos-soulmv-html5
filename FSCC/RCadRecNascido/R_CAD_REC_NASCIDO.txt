--R_CAD_REC_NASCIDO
SELECT ATE.CD_ATENDIMENTO,
       PACIENTE.CD_PACIENTE,
       PACIENTE.NM_PACIENTE,
  (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
       PACIENTE.NM_CONJUGE,
       PACIENTE.DS_ENDERECO,
       PACIENTE.NM_BAIRRO,
       CIDADE.CD_CIDADE,
       DECODE(CIDADE.NM_CIDADE, NULL, 'Não Informado', CIDADE.NM_CIDADE) NM_CIDADE,
       CIDADE.CD_UF,
       PACIENTE.NR_FONE,
       ATE.DT_ATENDIMENTO DT_INICIO,
       ATE.DT_ALTA,
       CONVENIO.CD_CONVENIO,
       CONVENIO.NM_CONVENIO,
       PRESTADOR.CD_PRESTADOR,
       PRESTADOR.NM_PRESTADOR,
       UNID_INT.CD_UNID_INT,
       UNID_INT.DS_UNID_INT,
       LEITO.CD_LEITO,
       LEITO.DS_LEITO,
       PRO_FAT.CD_PRO_FAT,
       PRO_FAT.DS_PRO_FAT,
       REC.CD_ADMISSAO_CO,
       REC.CD_RECEM_NASCIDO,
       REC.NM_RECEM_NASCIDO,
       DECODE(REC.TP_SEXO, 'M', 'MASCULINO', 'FEMININO') TP_SEXO,
       REC.HR_NASCIMENTO,
       REC.VL_PESO,
       REC.VL_ALTURA,
       DECODE(REC.CD_TIP_MORTE_RN, NULL, 'VIVO', 'MORTO') CONDICAO
FROM DBAMV.ADMISSAO_CO AD,
     DBAMV.RECEM_NASCIDO REC,
     DBAMV.ATENDIME ATE,
     DBAMV.CONVENIO,
     DBAMV.PRO_FAT,
     DBAMV.PRESTADOR,
     DBAMV.PACIENTE,
     DBAMV.CIDADE,
     DBAMV.LEITO,
     DBAMV.UNID_INT,
     Dbamv.Empresa_Convenio
WHERE Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
  AND Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
  AND ate.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
  AND AD.CD_ADMISSAO_CO = REC.CD_ADMISSAO_CO
  AND AD.CD_ATENDIMENTO = ATE.CD_ATENDIMENTO 
  AND ATE.CD_CONVENIO = CONVENIO.CD_CONVENIO
  AND ATE.CD_PRO_INT = PRO_FAT.CD_PRO_FAT(+)
  AND ATE.CD_PACIENTE = PACIENTE.CD_PACIENTE
  AND PACIENTE.CD_CIDADE = CIDADE.CD_CIDADE(+)
  AND ATE.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
  AND ATE.CD_LEITO = LEITO.CD_LEITO(+)
  AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)
