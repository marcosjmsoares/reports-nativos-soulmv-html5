--R_AGENDA_DIA_PRE_AGENDA
SELECT * FROM
  (SELECT TRUNC(AGE_CIR.DT_INICIO_AGE_CIR) DT_INICIO_AGE_CIR,
          TO_CHAR(AGE_CIR.DT_INICIO_AGE_CIR, 'HH24') HR_INICIO_AGE_CIR,
          TO_CHAR(AGE_CIR.VL_TEMPO_CIRURGIA, 'HH24') TEMPO_CIRURGIA,
          AVISO_CIRURGIA.CD_AVISO_CIRURGIA CD_AVISO_CIRURGIA,
          ATENDIME.CD_ATENDIMENTO CD_ATENDIMENTO,
          (CEN_CIR.DS_CEN_CIR || '/' || SAL_CIR.DS_SAL_CIR) DS_CEN_CIR,
          NVL(PACIENTE.CD_PACIENTE, AVISO_CIRURGIA.CD_PACIENTE) CD_PACIENTE,
          NVL(PACIENTE.NM_PACIENTE, AVISO_CIRURGIA.NM_PACIENTE) NM_PACIENTE,
       (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND AVISO_CIRURGIA.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
          NVL(CONVENIO.NM_CONVENIO, CONV_AVISO.NM_CONVENIO) NM_CONVENIO,
          NVL(PRESTADOR.NM_PRESTADOR, 'PRESTADOR NAO CADASTRADO') NM_PRESTADOR,
          TIP_ANEST.DS_TIP_ANEST DS_TIP_ANEST,
          AVISO_CIRURGIA.NR_TELEFONE_CONTATO NR_TELEFONE_CONTATO,
          AVISO_CIRURGIA.VL_TEMPO_PREVISTO VL_TEMPO_PREVISTO,
          PRE.CD_AVISO_INTERNACAO CD_AVISO_INTERNACAO,
          CIRURGIA_AVISO.CD_CIRURGIA COD_CIRURGIA,
          DECODE(CIRURGIA_AVISO.DS_NPADRONIZADO, NULL , CIRURGIA.DS_CIRURGIA , 'N.P.' || CIRURGIA_AVISO.DS_NPADRONIZADO) DS_CIRURGIA,
     (SELECT NM_PRESTADOR
      FROM DBAMV.PRESTADOR
      WHERE PRESTADOR.CD_PRESTADOR = AVISO_CIRURGIA.CD_PRESTADOR_INDICACAO) NM_PRESTADOR_INDICACAO
   FROM DBAMV.AVISO_CIRURGIA AVISO_CIRURGIA,
        DBAMV.AGE_CIR AGE_CIR,
        DBAMV.SAL_CIR SAL_CIR,
        DBAMV.CEN_CIR CEN_CIR,
        DBAMV.TIP_ANEST TIP_ANEST,
        DBAMV.ATENDIME ATENDIME,
        DBAMV.PACIENTE PACIENTE,
        DBAMV.CIRURGIA_AVISO CIRURGIA_AVISO,
        DBAMV.CIRURGIA CIRURGIA,
        DBAMV.CONVENIO CONVENIO,
        DBAMV.CONVENIO CONV_AVISO,
        DBAMV.PRE_INTERNACAO PRE,
        DBAMV.ESTOQUE ESTOQUE,
     (SELECT DISTINCT PRESTADOR_AVISO.CD_AVISO_CIRURGIA ,
                      PRESTADOR_AVISO.CD_PRESTADOR ,
                      PRESTADOR.NM_PRESTADOR ,
                      PRESTADOR.CD_TIP_PRESTA
      FROM DBAMV.PRESTADOR_AVISO ,
           DBAMV.ATI_MED ,
           DBAMV.PRESTADOR
      WHERE PRESTADOR_AVISO.CD_ATI_MED = ATI_MED.CD_ATI_MED
        AND ATI_MED.TP_FUNCAO = 'C'
        AND PRESTADOR_AVISO.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR ) PRESTADOR
   WHERE AVISO_CIRURGIA.TP_SITUACAO <> 'C'
     AND ESTOQUE.CD_ESTOQUE = CEN_CIR.CD_ESTOQUE
     AND ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
     AND TRUNC(AGE_CIR.DT_INICIO_AGE_CIR) BETWEEN :CAMPO_DATA_INICIO AND 
     AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = AGE_CIR.CD_AVISO_CIRURGIA
     AND AVISO_CIRURGIA.CD_CEN_CIR = CEN_CIR.CD_CEN_CIR(+)
     AND AVISO_CIRURGIA.CD_SAL_CIR = SAL_CIR.CD_SAL_CIR(+)
     AND AVISO_CIRURGIA.CD_TIP_ANEST = TIP_ANEST.CD_TIP_ANEST(+)
     AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = PRESTADOR.CD_AVISO_CIRURGIA(+)
     AND AVISO_CIRURGIA.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO(+)
     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE(+)
     AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO(+)
     AND CIRURGIA_AVISO.CD_CONVENIO = CONV_AVISO.CD_CONVENIO(+)
     AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
     AND CIRURGIA_AVISO.CD_CIRURGIA = CIRURGIA.CD_CIRURGIA(+)
     AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = PRE.CD_AVISO_INTERNACAO(+)    
   ORDER BY DT_INICIO_AGE_CIR ,
            HR_INICIO_AGE_CIR ,
            PRESTADOR.NM_PRESTADOR)
ORDER BY DT_INICIO_AGE_CIR,
         HR_INICIO_AGE_CIR,
         TEMPO_CIRURGIA,
         NM_PRESTADOR,
         CD_AVISO_CIRURGIA,
         VL_TEMPO_PREVISTO,
         CD_ATENDIMENTO,
         DS_CEN_CIR,
         CD_PACIENTE,
         NM_PACIENTE,
         NR_TELEFONE_CONTATO,
         NM_CONVENIO,
         DS_TIP_ANEST,
         CD_AVISO_INTERNACAO
