--R_PAINEL_SOLIC_AGENDA
SELECT To_Char(aviso_cirurgia.DT_PREV_INTER, 'hh24') hora,
       sal_cir.ds_sal_cir sala,
       exa_rx.ds_exa_rx exame_produto,
       aviso_cirurgia.nm_paciente paciente,
       cirurgia.ds_cirurgia cirurgia,
       aviso_cirurgia.cd_aviso_cirurgia aviso
FROM dbamv.it_av_exa_rx,
     dbamv.exa_rx,
     dbamv.aviso_cirurgia,
     dbamv.sal_cir,
     dbamv.cirurgia
WHERE it_av_exa_rx.cd_set_exa = :CAMPO_CD_SETOR
  AND to_char(aviso_cirurgia.DT_PREV_INTER, 'dd/mm/yyyy') = :CAMPO_DT_REALIZACAO
  AND it_av_exa_rx.cd_exa_rx = exa_rx.cd_exa_rx
  AND it_av_exa_rx.cd_cirurgia = cirurgia.cd_cirurgia
  AND it_av_exa_rx.cd_aviso_cirurgia = aviso_cirurgia.cd_aviso_cirurgia
  AND aviso_cirurgia.cd_sal_cir = sal_cir.cd_sal_cir
  AND 'TP_EXAME' = :CAMPO_QUERY
UNION ALL
SELECT To_Char(aviso_cirurgia.DT_PREV_INTER, 'hh24') hora,
       sal_cir.ds_sal_cir sala,
       exa_lab.nm_exa_lab exame_produto,
       aviso_cirurgia.nm_paciente paciente,
       cirurgia.ds_cirurgia cirurgia,
       aviso_cirurgia.cd_aviso_cirurgia aviso
FROM dbamv.it_av_exa_lab,
     dbamv.exa_lab,
     dbamv.aviso_cirurgia,
     dbamv.sal_cir,
     dbamv.cirurgia
WHERE it_av_exa_lab.cd_set_exa = :CAMPO_CD_SETOR
  AND to_char(aviso_cirurgia.DT_PREV_INTER, 'dd/mm/yyyy') = :CAMPO_DT_REALIZACAO
  AND it_av_exa_lab.cd_exa_lab = exa_lab.cd_exa_lab
  AND it_av_exa_lab.cd_cirurgia = cirurgia.cd_cirurgia
  AND it_av_exa_lab.cd_aviso_cirurgia = aviso_cirurgia.cd_aviso_cirurgia
  AND aviso_cirurgia.cd_sal_cir = sal_cir.cd_sal_cir
  AND 'TP_EXAME' = :CAMPO_QUERY
UNION ALL
SELECT To_Char(aviso_cirurgia.DT_PREV_INTER, 'hh24') hora,
       sal_cir.ds_sal_cir sala,
       produto.ds_produto exame_produto,
       aviso_cirurgia.nm_paciente paciente,
       cirurgia.ds_cirurgia cirurgia,
       aviso_cirurgia.cd_aviso_cirurgia aviso
FROM dbamv.it_av_produtos,
     dbamv.estoque,
     dbamv.produto,
     dbamv.aviso_cirurgia,
     dbamv.sal_cir,
     dbamv.cirurgia
WHERE it_av_produtos.cd_estoque = estoque.cd_estoque
  AND it_av_produtos.cd_produto = produto.cd_produto
  AND it_av_produtos.cd_cirurgia = cirurgia.cd_cirurgia
  AND it_av_produtos.cd_aviso_cirurgia = aviso_cirurgia.cd_aviso_cirurgia
  AND aviso_cirurgia.cd_sal_cir = sal_cir.cd_sal_cir
  AND estoque.cd_setor = :CAMPO_CD_SETOR
  AND to_char(aviso_cirurgia.DT_PREV_INTER, 'dd/mm/yyyy') = :CAMPO_DT_REALIZACAO
  AND 'TP_PRODUTO' = :CAMPO_QUERY
UNION ALL
SELECT To_Char(aviso_cirurgia.DT_PREV_INTER, 'hh24') hora,
       sal_cir.ds_sal_cir sala,
       sangue_derivados.ds_sangue_derivados exame_produto,
       aviso_cirurgia.nm_paciente paciente,
       cirurgia.ds_cirurgia cirurgia,
       aviso_cirurgia.cd_aviso_cirurgia aviso
FROM dbamv.sangue_aviso
    ,dbamv.sangue_derivados
    ,dbamv.aviso_cirurgia
    ,dbamv.sal_cir
    ,dbamv.cirurgia
WHERE sangue_aviso.cd_sangue_derivados = sangue_derivados.cd_sangue_derivados
  AND sangue_aviso.cd_aviso_cirurgia = aviso_cirurgia.cd_aviso_cirurgia
  AND sangue_aviso.cd_cirurgia = cirurgia.cd_cirurgia
  AND aviso_cirurgia.cd_sal_cir = sal_cir.cd_sal_cir
  AND to_char(aviso_cirurgia.DT_PREV_INTER, 'dd/mm/yyyy') = :CAMPO_DT_REALIZACAO
  AND 'TP_SANGUE' = :CAMPO_QUERY
