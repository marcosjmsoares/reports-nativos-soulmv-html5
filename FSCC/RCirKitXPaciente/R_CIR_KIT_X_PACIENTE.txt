--R_CIR_KIT_X_PACIENTE
SELECT * FROM
  (SELECT AVISO_CIRURGIA.CD_AVISO_CIRURGIA CD_AVISO_CIRURGIA ,
          AVISO_CIRURGIA.CD_ATENDIMENTO CD_ATENDIMENTO ,
          AVISO_CIRURGIA.NM_PACIENTE NM_PACIENTE ,
      (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND AVISO_CIRURGIA.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
          TO_CHAR(AGE_CIR.DT_INICIO_AGE_CIR, 'DD/MM/YYYY HH24') DT_INICIO ,
          CEN_CIR.DS_CEN_CIR DS_CEN_CIR ,
          SAL_CIR.DS_RESUMIDA DS_SAL_CIR ,
          CIRURGIA.CD_CIRURGIA CD_CIRURGIA ,
          CIRURGIA.DS_CIRURGIA DS_CIRURGIA ,
          CIRURGIA_KIT.CD_FORMULA CD_FORMULA ,
          PRODUTO.CD_PRODUTO CD_PRODUTO ,
          PRODUTO.DS_PRODUTO DS_PRODUTO ,
          CIRURGIA_KIT.QT_KIT QT_KIT
   FROM DBAMV.AVISO_CIRURGIA ,
        DBAMV.AGE_CIR ,
        DBAMV.CEN_CIR ,
        DBAMV.SAL_CIR ,
        DBAMV.CIRURGIA_AVISO ,
        DBAMV.CIRURGIA_KIT ,
        DBAMV.FORMULA ,
        DBAMV.PRODUTO ,
        DBAMV.CIRURGIA
   WHERE TRUNC(AGE_CIR.DT_INICIO_AGE_CIR) BETWEEN :CAMPO_DATA_INI AND 
     AND EXISTS
       (SELECT Empresa_Produto.Cd_Produto FROM Dbamv.Empresa_Produto WHERE Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
            AND Empresa_Produto.Cd_Produto = Produto.Cd_Produto )
AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = AGE_CIR.CD_AVISO_CIRURGIA
     AND AVISO_CIRURGIA.CD_CEN_CIR = CEN_CIR.CD_CEN_CIR
     AND AVISO_CIRURGIA.CD_SAL_CIR = SAL_CIR.CD_SAL_CIR
     AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
     AND CIRURGIA_AVISO.CD_CIRURGIA = CIRURGIA.CD_CIRURGIA
     AND CIRURGIA_AVISO.CD_CIRURGIA_AVISO = CIRURGIA_KIT.CD_CIRURGIA_AVISO
     AND CIRURGIA_KIT.CD_FORMULA = FORMULA.CD_FORMULA
     AND FORMULA.CD_PRODUTO = PRODUTO.CD_PRODUTO  :CAMPO_CF_COND_KIT )
ORDER BY DT_INICIO,
         CD_AVISO_CIRURGIA,
         CD_ATENDIMENTO,
         NM_PACIENTE,
         DS_CEN_CIR,
         DS_SAL_CIR,
         CD_CIRURGIA,
         DS_CIRURGIA

--R_CIR_KIT_X_PACIENTE_R_PRODUTO_Q_ITEM_KIT
SELECT * FROM (
SELECT * FROM (
SELECT  CD_FORMULA    CD_FORMULA_ITEM
               ,PRODUTO          PRODUTO
               ,CODIGO              CODIGO
               ,DESCRICAO       DESCRICAO
               ,QTDE                  QTDE
FROM (
SELECT DISTINCT ITFORMULA.CD_FORMULA       CD_FORMULA
              ,'PRODUTOS AVULSOS'                               PRODUTO
              ,TO_CHAR(PRODUTO.CD_PRODUTO)       CODIGO
              ,PRODUTO.DS_PRODUTO                          DESCRICAO
              ,ITFORMULA.QT_PADRAO                          QTDE
FROM   DBAMV.ITFORMULA
            ,DBAMV.PRODUTO
WHERE  ITFORMULA.CD_PRODUTO_TEM = PRODUTO.CD_PRODUTO
AND    PRODUTO.SN_CONSIGNADO IN ('N','R')
AND Exists ( SELECT Empresa_Produto.Cd_Produto FROM Dbamv.Empresa_Produto WHERE Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA AND Empresa_Produto.Cd_Produto = Produto.Cd_Produto )
UNION ALL
SELECT DISTINCT ITFORMULA.CD_FORMULA     CD_FORMULA
              ,'PRODUTOS CONSIGNADOS'                   PRODUTO
              ,TO_CHAR(PRODUTO.CD_PRODUTO)     CODIGO
              ,PRODUTO.DS_PRODUTO                        DESCRICAO
              ,ITFORMULA.QT_PADRAO                        QTDE
FROM   DBAMV.ITFORMULA
            ,DBAMV.PRODUTO
WHERE  ITFORMULA.CD_PRODUTO_TEM = PRODUTO.CD_PRODUTO
AND Exists ( SELECT Empresa_Produto.Cd_Produto FROM Dbamv.Empresa_Produto WHERE Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA AND Empresa_Produto.Cd_Produto = Produto.Cd_Produto )
AND    PRODUTO.SN_CONSIGNADO = 'S'
UNION ALL
SELECT KIT_PRO_FAT.CD_FORMULA     CD_FORMULA
              ,'PROCEDIMENTO'                        PRODUTO
              ,PRO_FAT.CD_PRO_FAT              CODIGO
              ,PRO_FAT.DS_PRO_FAT              DESCRICAO
              ,KIT_PRO_FAT.QT_PADRAO        QTDE
FROM   DBAMV.KIT_PRO_FAT
             ,DBAMV.PRO_FAT
WHERE  KIT_PRO_FAT.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
UNION ALL
SELECT KIT_EQUIP.CD_FORMULA                                                                          CD_FORMULA
             ,'EQUIPAMENTOS'                                                                                         PRODUTO
             ,TO_CHAR(APARELHOS_EQUIPTO.CD_APARELHO_EQUIPAMENTO)    CODIGO
             ,APARELHOS_EQUIPTO.DS_APARELHO_EQUIPAMENTO                       DESCRICAO
            ,KIT_EQUIP.QT_PADRAO                                                                              QTDE
FROM   DBAMV.KIT_EQUIP
            ,DBAMV.APARELHOS_EQUIPTO
WHERE  KIT_EQUIP.CD_APARELHO_EQUIPAMENTO = APARELHOS_EQUIPTO.CD_APARELHO_EQUIPAMENTO
UNION ALL
SELECT KIT_SANGUE.CD_FORMULA                                                         CD_FORMULA
             ,'SANGUE E DERIVADOS'                                                                PRODUTO
             ,TO_CHAR(SANGUE_DERIVADOS.CD_SANGUE_DERIVADOS)   CODIGO
             ,SANGUE_DERIVADOS.DS_SANGUE_DERIVADOS                      DESCRICAO
             ,KIT_SANGUE.QT_PADRAO                                                            QTDE
FROM   DBAMV.KIT_SANGUE
            ,DBAMV.SANGUE_DERIVADOS
WHERE  KIT_SANGUE.CD_SANGUE_DERIVADOS = SANGUE_DERIVADOS.CD_SANGUE_DERIVADOS
UNION ALL
SELECT ITCAIXA_CIRURGICA.CD_FORMULA       CD_FORMULA
             ,'CAIXA CIRURGICA'                                    PRODUTO
             ,TO_CHAR(PRODUTO.CD_PRODUTO)     CODIGO
             ,PRODUTO.DS_PRODUTO                        DESCRICAO
             ,ITCAIXA_CIRURGICA.QT_PADRAO          QTDE
FROM   DBAMV.ITCAIXA_CIRURGICA
            ,DBAMV.PRODUTO
WHERE  ITCAIXA_CIRURGICA.CD_PRODUTO = PRODUTO.CD_PRODUTO
AND Exists ( SELECT Empresa_Produto.Cd_Produto FROM Dbamv.Empresa_Produto WHERE Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA AND Empresa_Produto.Cd_Produto = Produto.Cd_Produto )
)
)
WHERE CD_FORMULA_ITEM = :CAMPO_CD_FORMULA
)
ORDER BY CD_FORMULA_ITEM,PRODUTO
