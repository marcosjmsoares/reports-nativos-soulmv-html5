--R_FICHA_CIR
SELECT * FROM
  (SELECT AVISO_CIRURGIA.CD_AVISO_CIRURGIA CD_AVISO_CIRURGIA,
          AVISO_CIRURGIA.CD_ATENDIMENTO CD_ATENDIMENTO,
          AVISO_CIRURGIA.CD_PACIENTE CD_PACIENTE,
          AVISO_CIRURGIA.NM_PACIENTE NM_PACIENTE,
      (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE,
          AVISO_CIRURGIA.DT_REALIZACAO DT_REALIZACAO,
          DECODE (CONVENIO.TP_CONVENIO,
                    'A', CIRURGIA.CD_PROCEDIMENTO_SIA,
                    'H', CIRURGIA.CD_PROCEDIMENTO_SIH,
                    'C', CIRURGIA.CD_PRO_FAT,
                    'P', CIRURGIA.CD_PRO_FAT,
                  CIRURGIA.CD_PRO_FAT) CD_CIRURGIA,
                 DECODE(CONVENIO.TP_CONVENIO, 
                          'C', decode(PRO_FAT.DS_PRO_FAT, NULL, 'Procedimento não Informado', PRO_FAT.DS_PRO_FAT), 
                          'P', decode(PRO_FAT.DS_PRO_FAT, NULL, 'Procedimento não Informado', PRO_FAT.DS_PRO_FAT), 
                          'A', decode(PROC_SUS_A.DS_PROCEDIMENTO, NULL, 'Procedimento não Informado', PROC_SUS_A.DS_PROCEDIMENTO), 
                          'H', DECODE(PROC_SUS_H.DS_PROCEDIMENTO, NULL, 'Procedimento não Informado', PROC_SUS_H.DS_PROCEDIMENTO), 
                          NVL(CIRURGIA_AVISO.DS_NPADRONIZADO, CIRURGIA.DS_CIRURGIA)) DS_CIRURGIA,
                 SAL_CIR.CD_SAL_CIR CD_SAL_CIR,
                 SAL_CIR.DS_SAL_CIR DS_SAL_CIR,
                 TO_CHAR(AVISO_CIRURGIA.DT_REALIZACAO, 'HH24') HORA_PREV_INI,
                 TO_CHAR(AVISO_CIRURGIA.VL_TEMPO_DURACAO, 'HH24') HORA_PREV_FIM,
                 SOMA_SUB_DATA(AVISO_CIRURGIA.DT_REALIZACAO, AVISO_CIRURGIA.VL_TEMPO_DURACAO, '+') DIF_TEMPO,
                 ATENDIME.CD_CONVENIO CD_CONVENIO,
                 ATENDIME.CD_LEITO CD_LEITO,
                 CONVENIO.NM_CONVENIO NM_CONVENIO,
                 LEITO.DS_LEITO DS_LEITO,
                 DECODE (CONVENIO.TP_CONVENIO,'C',ATENDIME.NR_CARTEIRA, PACIENTE.NR_CNS)                  CD_CARTEIRA,
                 CIRURGIA_AVISO.CD_CIRURGIA CIR_AVI_CIRURGIA,
                 SANGUE_AVISO.CD_SANGUE_DERIVADOS CD_SANGUE_DERIVADOS,
                 SANGUE_AVISO.QT_SOLICITADO QT_SOLICITADO,
                 SANGUE_AVISO.QT_REALIZADO QT_REALIZADO,
                 SANGUE_DERIVADOS.DS_SANGUE_DERIVADOS DS_SANGUE_DERIVADOS
   FROM DBAMV.AVISO_CIRURGIA AVISO_CIRURGIA,
        DBAMV.CIRURGIA_AVISO CIRURGIA_AVISO,
        DBAMV.CIRURGIA CIRURGIA,
        DBAMV.SAL_CIR SAL_CIR,
        DBAMV.AGE_CIR AGE_CIR,
        DBAMV.ATENDIME ATENDIME,
        DBAMV.CONVENIO CONVENIO,
        DBAMV.LEITO LEITO,
        DBAMV.PACIENTE PACIENTE,
        DBAMV.PRO_FAT PRO_FAT,
        DBAMV.PROCEDIMENTO_SUS PROC_SUS_A,
        DBAMV.PROCEDIMENTO_SUS PROC_SUS_H,
        DBAMV.SANGUE_AVISO SANGUE_AVISO,
        DBAMV.SANGUE_DERIVADOS SANGUE_DERIVADOS,
        Dbamv.Empresa_Convenio
   WHERE Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
     AND Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
     AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = :CAMPO_CD_AVISO_CIRURGIA
     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA 
     AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
     AND AVISO_CIRURGIA.CD_SAL_CIR = SAL_CIR.CD_SAL_CIR(+)
     AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = AGE_CIR.CD_AVISO_CIRURGIA(+)
     AND AVISO_CIRURGIA.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
     AND CONVENIO.CD_CONVENIO = ATENDIME.CD_CONVENIO
     AND LEITO.CD_LEITO(+) = ATENDIME.CD_LEITO
     AND CIRURGIA.CD_CIRURGIA = CIRURGIA_AVISO.CD_CIRURGIA
     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
     AND CIRURGIA.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT(+)
     AND CIRURGIA.CD_PROCEDIMENTO_SIA = PROC_SUS_A.CD_PROCEDIMENTO (+)
     AND CIRURGIA.CD_PROCEDIMENTO_SIH = PROC_SUS_H.CD_PROCEDIMENTO (+)
     AND CIRURGIA_AVISO.CD_CIRURGIA_AVISO = SANGUE_AVISO.CD_CIRURGIA_AVISO(+)
     AND SANGUE_AVISO.CD_SANGUE_DERIVADOS = SANGUE_DERIVADOS.CD_SANGUE_DERIVADOS(+)
   ORDER BY AVISO_CIRURGIA.CD_AVISO_CIRURGIA)
ORDER BY CD_AVISO_CIRURGIA,
         CD_SAL_CIR,
         DS_SAL_CIR,
         CD_PACIENTE,
         NM_PACIENTE,
         HORA_PREV_FIM,
         HORA_PREV_INI,
         DT_REALIZACAO,
         CD_CONVENIO,
         CD_LEITO,
         NM_CONVENIO,
         DS_LEITO,
         DIF_TEMPO,
         CD_ATENDIMENTO,
         CD_CARTEIRA,
         CD_CIRURGIA,
         DS_CIRURGIA,
         CIR_AVI_CIRURGIA

--R_FICHA_CIR_R_SANGUE_Q_MATMED
SELECT * FROM (
    SELECT * FROM (
        SELECT MVTO.CD_AVISO_CIRURGIA CD_AVISO_CIRURGIA3,
          SUM( ITMVTO.QT_MOVIMENTACAO * DECODE( MVTO.TP_MVTO_ESTOQUE, 'P', 1, -1 ) ) QT_MOVIMENTACAO,
              ITMVTO.CD_PRODUTO   CD_PRODUTO,
              PRODUTO.DS_PRODUTO  DS_PRODUTO,
              PRODUTO.CD_ESPECIE  CD_ESPECIE,
              ITMVTO.CD_UNI_PRO   CD_UNI_PRO,
              UNI_PRO.DS_UNIDADE  DS_UNIDADE,
              ESPECIE.DS_ESPECIE  DS_ESPECIE
        FROM  DBAMV.MVTO_ESTOQUE MVTO,
              DBAMV.ITMVTO_ESTOQUE     ITMVTO,
              DBAMV.PRODUTO            PRODUTO,
              DBAMV.UNI_PRO            UNI_PRO,
              DBAMV.ESPECIE            ESPECIE,
              DBAMV.ESTOQUE            ESTOQUE
        WHERE MVTO.CD_MVTO_ESTOQUE = ITMVTO.CD_MVTO_ESTOQUE
          AND ITMVTO.CD_PRODUTO    = PRODUTO.CD_PRODUTO
          AND ITMVTO.CD_UNI_PRO    = UNI_PRO.CD_UNI_PRO
          AND PRODUTO.CD_ESPECIE   = ESPECIE.CD_ESPECIE
          AND MVTO.CD_ESTOQUE      = ESTOQUE.CD_ESTOQUE
          AND ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
          AND MVTO.CD_AVISO_CIRURGIA = :CAMPO_CD_AVISO_CIRURGIA
        GROUP BY  MVTO.CD_AVISO_CIRURGIA,
                  ITMVTO.CD_PRODUTO,
                  PRODUTO.DS_PRODUTO,
                  PRODUTO.CD_ESPECIE,
                  ITMVTO.CD_UNI_PRO,
                  UNI_PRO.DS_UNIDADE,
                  ESPECIE.DS_ESPECIE
        HAVING SUM( ITMVTO.QT_MOVIMENTACAO * DECODE( MVTO.TP_MVTO_ESTOQUE, 'P', 1, -1 ) ) <> 0
        ORDER BY PRODUTO.DS_PRODUTO
    )
  WHERE CD_AVISO_CIRURGIA3 = :CAMPO_CD_AVISO_CIRURGIA
)
ORDER BY DS_ESPECIE,CD_ESPECIE,CD_AVISO_CIRURGIA3
