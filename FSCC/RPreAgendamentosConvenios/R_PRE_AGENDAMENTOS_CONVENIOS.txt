--R_PRE_AGENDAMENTOS_CONVENIOS
SELECT nm_convenio, 
       Count(nm_convenio) qtd_convenio 
FROM   aviso_cirurgia, 
       cirurgia_aviso, 
       convenio 
WHERE  aviso_cirurgia.tp_situacao = 'P' 
       AND aviso_cirurgia.cd_aviso_cirurgia = cirurgia_aviso.cd_aviso_cirurgia 
       AND cirurgia_aviso.sn_principal = 'S' 
       AND cirurgia_aviso.cd_convenio = convenio.cd_convenio 
       AND trunc(aviso_cirurgia.dt_aviso_cirurgia) between trunc(:CAMPO_DT_INICIAL) and trunc(:CAMPO_DT_FINAL)
GROUP  BY nm_convenio
