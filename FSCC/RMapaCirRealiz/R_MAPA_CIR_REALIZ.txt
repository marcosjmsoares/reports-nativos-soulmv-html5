--R_MAPA_CIR_REALIZ
SELECT * FROM
  (SELECT AVISO_CIRURGIA.CD_AVISO_CIRURGIA CD_AVISO_CIRURGIA ,
          ATI_MED.NR_ORDEM_APRESENTACAO , 
          PRESTADOR.CD_PRESTADOR CD_PRESTADOR ,
          NVL(PRESTADOR.NM_PRESTADOR, 'NAO INFORMADO') NM_PRESTADOR ,
          CIRURGIA_AVISO.CD_CIRURGIA_AVISO CD_CIRURGIA_AVISO ,
          CIRURGIA.DS_CIRURGIA DS_CIRURGIA ,
          TRUNC(AVISO_CIRURGIA.DT_REALIZACAO) DT_REALIZACAO ,
          AVISO_CIRURGIA.DT_REALIZACAO HR_REALIZACAO ,
          AVISO_CIRURGIA.NM_PACIENTE NM_PACIENTE ,
      (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND AVISO_CIRURGIA.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
          CEN_CIR.CD_CEN_CIR CD_CEN_CIR ,
          CEN_CIR.DS_CEN_CIR DS_CEN_CIR ,
          AVISO_CIRURGIA.CD_ATENDIMENTO CD_ATENDIMENTO ,
          CONVENIO.NM_CONVENIO NM_CONVENIO ,
          NVL(ATI_MED.DS_ATI_MED, 'NAO INFORMADO') DS_ATI_MED
   FROM
     (SELECT P_A.CD_PRESTADOR,
             A_M.CD_ATI_MED,
             A_M.DS_ATI_MED,
             P_A.CD_CIRURGIA_AVISO,
             P_A.CD_AVISO_CIRURGIA,
             P_A.CD_CIRURGIA
      FROM DBAMV.PRESTADOR_AVISO P_A,
           DBAMV.ATI_MED A_M
      WHERE A_M.CD_ATI_MED = P_A.CD_ATI_MED 
      ORDER BY A_M.NR_ORDEM_APRESENTACAO) PRESTADOR_AVISO , 
        DBAMV.ATI_MED ATI_MED ,
        DBAMV.PRESTADOR PRESTADOR ,
        DBAMV.AVISO_CIRURGIA AVISO_CIRURGIA ,
        DBAMV.CIRURGIA_AVISO CIRURGIA_AVISO ,
        DBAMV.CIRURGIA CIRURGIA ,
        DBAMV.CEN_CIR CEN_CIR ,
        DBAMV.CONVENIO CONVENIO,
        Dbamv.Empresa_Convenio
   WHERE Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
     AND Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
     AND PRESTADOR_AVISO.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
     AND PRESTADOR_AVISO.CD_ATI_MED = ATI_MED.CD_ATI_MED(+)
     AND AVISO_CIRURGIA.TP_SITUACAO = 'R'
     AND AVISO_CIRURGIA.CD_CEN_CIR = CEN_CIR.CD_CEN_CIR
     AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
     AND CIRURGIA_AVISO.CD_CIRURGIA_AVISO = PRESTADOR_AVISO.CD_CIRURGIA_AVISO(+)
     AND CIRURGIA_AVISO.CD_CIRURGIA = CIRURGIA.CD_CIRURGIA
     AND AVISO_CIRURGIA.DT_REALIZACAO BETWEEN To_Date(Concat(To_Char(:CAMPO_DATA_INI,'dd/mm/yyyy'), '00'), 'dd/mm/yyyy hh24') AND To_Date(Concat(To_Char(:CAMPO_DATA_FIM,'dd/mm/yyyy'), '23'), 'dd/mm/yyyy hh24')
     AND CIRURGIA_AVISO.CD_CONVENIO = CONVENIO.CD_CONVENIO(+)
     AND AVISO_CIRURGIA.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA 
   ORDER BY AVISO_CIRURGIA.CD_AVISO_CIRURGIA ,
            ATI_MED.NR_ORDEM_APRESENTACAO 
 )
ORDER BY DS_CEN_CIR,
         CD_CEN_CIR,
         DT_REALIZACAO,
         HR_REALIZACAO,
         CD_ATENDIMENTO,
         NM_PACIENTE,
         DS_CIRURGIA,
         CD_CIRURGIA_AVISO,
         CD_AVISO_CIRURGIA,
         NM_CONVENIO
