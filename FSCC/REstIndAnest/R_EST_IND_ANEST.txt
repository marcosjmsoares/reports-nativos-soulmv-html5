--R_EST_IND_ANEST
SELECT   DISTINCT
         AVISO_CIRURGIA.CD_AVISO_CIRURGIA             CD_AVISO_CIRURGIA
        ,PRESTADOR_AVISO.CD_PRESTADOR                 CD_PRESTADOR
        ,PRESTADOR.NM_PRESTADOR                       NM_PRESTADOR
        ,TIP_ANEST.CD_TIP_ANEST                       CD_TIP_ANEST
        ,TIP_ANEST.DS_TIP_ANEST                       DS_TIP_ANEST
        ,CEN_CIR.CD_CEN_CIR                           CD_CEN_CIR
        ,CEN_CIR.DS_CEN_CIR                           DS_CEN_CIR
FROM     DBAMV.AVISO_CIRURGIA                         AVISO_CIRURGIA
        ,DBAMV.CEN_CIR                                CEN_CIR
        ,DBAMV.TIP_ANEST_AVISO_CIRURGIA               TIP_ANEST_AVISO_CIRURGIA
        ,DBAMV.TIP_ANEST                              TIP_ANEST
        ,DBAMV.CIRURGIA_AVISO                         CIRURGIA_AVISO
        ,DBAMV.PRESTADOR_AVISO                        PRESTADOR_AVISO
        ,DBAMV.PRESTADOR                              PRESTADOR
        ,DBAMV.ATI_MED                                ATI_MED
WHERE    AVISO_CIRURGIA.TP_SITUACAO                 = 'R'
AND      AVISO_CIRURGIA.CD_CEN_CIR                  = CEN_CIR.CD_CEN_CIR
AND      AVISO_CIRURGIA.CD_AVISO_CIRURGIA           = TIP_ANEST_AVISO_CIRURGIA.CD_AVISO_CIRURGIA
AND      AVISO_CIRURGIA.CD_MULTI_EMPRESA            = DBAMV.PKG_MV2000.LE_EMPRESA
AND      TIP_ANEST_AVISO_CIRURGIA.CD_TIP_ANEST      = TIP_ANEST.CD_TIP_ANEST
AND      AVISO_CIRURGIA.CD_AVISO_CIRURGIA           = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
AND      CIRURGIA_AVISO.CD_CIRURGIA_AVISO           = PRESTADOR_AVISO.CD_CIRURGIA_AVISO
AND      PRESTADOR_AVISO.CD_PRESTADOR               = PRESTADOR.CD_PRESTADOR
AND      PRESTADOR_AVISO.CD_ATI_MED                 = ATI_MED.CD_ATI_MED
AND      ATI_MED.TP_FUNCAO                          = 'N'
AND      AVISO_CIRURGIA.DT_REALIZACAO BETWEEN To_Date(Concat(To_Char(To_Date(:CAMPO_DATA_INI,'dd/mm/yyyy'),'dd/mm/yyyy'),'00'),'dd/mm/yyyy hh24') and  To_Date(Concat(To_Char(To_Date(:CAMPO_DATA_FIM,'dd/mm/yyyy'),'dd/mm/yyyy'),'23'),'dd/mm/yyyy hh24')
ORDER BY CD_CEN_CIR, CD_AVISO_CIRURGIA
