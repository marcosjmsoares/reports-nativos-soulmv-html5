--R_ALTER_CIRURGIA
SELECT * FROM (
Select Ac.cd_aviso_cirurgia				Cd_Aviso_Cirurgia
          ,A.nr_carteira						  Nr_Carteira_Atual
          ,AcAnt.nr_carteira          Nr_Carteira_Ant
          ,A.Cd_Atendimento  Cd_Atendimento_Atual
          ,Ca.Cd_Convenio             Cd_Convenio_Atual
					,Co.Nm_Convenio							Nm_Convenio_Atual
          ,AcAnt.Cd_Convenio          Cd_Convenio_Ant
					,Co2.Nm_Convenio						Nm_Convenio_Ant
          ,Nvl(Ca.Cd_Con_Pla,0)				Cd_Con_Pla_Atual
					,Cp.Ds_Con_Pla 							Ds_Con_Pla_Atual
          ,AcAnt.Cd_Con_Plan			  	Cd_Con_Pla_Ant
					,Cp2.Ds_Con_Pla 						Ds_Con_Pla_Ant
          ,Ac.nm_paciente						  Nm_Paciente_Atual
          ,AcAnt.nm_paciente			 		Nm_Paciente_Ant
          ,Ac.ds_obs_aviso						Ds_Obs_Aviso_Atual
          ,AcAnt.ds_obs_aviso         Ds_Obs_Aviso_Ant
          ,Age_Cir.dt_inicio_age_cir					Dh_Hora_Cirurgia_Atual
          ,AcAnt.Dh_Hora_Cirurgia			Dh_Hora_Cirurgia_Ant
 from dbamv.Aviso_Cirurgia 				    Ac
        ,dbamv.Dados_Aviso_Anterior 	AcAnt
        ,dbamv.Cirurgia_aviso         Ca
        ,dbamv.Atendime  						  A
				,dbamv.Convenio								CO
				,dbamv.Convenio								CO2
				,dbamv.Con_Pla                Cp
				,dbamv.Con_Pla                Cp2
       ,dbamv.Age_Cir
Where Ac.cd_aviso_cirurgia 	= AcAnt.cd_aviso_cirurgia
  And Ac.Cd_Aviso_Cirurgia 	= Ca.Cd_Aviso_Cirurgia
  And Ca.Cd_Cirurgia_Aviso  = ( Select Max(Cd_Cirurgia_Aviso) from dbamv.Cirurgia_Aviso Where Cd_Aviso_Cirurgia = Ac.Cd_Aviso_Cirurgia )
  And Ac.cd_atendimento    = A.cd_atendimento (+)
	And Ca.Cd_Convenio				= Co.Cd_Convenio (+)
	And AcAnt.Cd_Convenio			= Co2.Cd_Convenio (+)
	And Ca.Cd_Convenio				= Cp.Cd_Convenio (+)
	And Ca.Cd_Con_Pla  				= Cp.Cd_Con_Pla (+)
	And AcAnt.Cd_Convenio			= Cp2.Cd_Convenio (+)
	And AcAnt.Cd_Con_Plan  		= Cp2.Cd_Con_Pla (+)
And Ac.cd_aviso_cirurgia 	=     Age_Cir.Cd_Aviso_Cirurgia
AND Ac.Sn_Revisado = 'N'
)
