--R_OCIO_CIR
SELECT  CEN_CIR.CD_CEN_CIR,
        CEN_CIR.DS_CEN_CIR,
        T.DATA DT_CALENDARIO,
        SAL_CIR.CD_SAL_CIR,
	SAL_CIR.DS_SAL_CIR,
        HR_SAL_CIR.HR_INICIAL,
	HR_SAL_CIR.HR_FINAL,
        SAL_CIR.DS_RESUMIDA,
	SAL_CIR.NR_TEMPO_PREPARO,
 T.TEMPO_DISPONIVEL,T.TEMPO_INTERDITADO,
  T.TEMPO_UTILIZADO AS TEMPO_UTILIZADO,
  T.TEMPO_OCIOSO AS TEMPO_OCIOSO,
  0 AS QT_FUNCIONAMENTO,
  0 AS TP_DIA_SEMANA,
  T.PERCENTUAL_OCIOSO
FROM DBAMV.TB_FSCC_REL_OCIOSIDADE T,
     CEN_CIR,
     HR_SAL_CIR,
     SAL_CIR,
     SETOR
WHERE T.SALA = SAL_CIR.CD_SAL_CIR
AND HR_SAL_CIR.CD_SAL_CIR = SAL_CIR.CD_SAL_CIR
AND SAL_CIR.CD_CEN_CIR = CEN_CIR.CD_CEN_CIR
AND HR_SAL_CIR.TP_DIA_SEMANA - Nvl(dbamv.pkg_mv2000.le_configuracao('FSCC', 'QT_DIAS_A_MENOS_DIA_SEMANA'), 0) = To_Char(T.DATA,'D')
AND SETOR.CD_SETOR = SAL_CIR.CD_SETOR
AND SETOR.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
ORDER BY CEN_CIR.CD_CEN_CIR
        ,T.DATA
        ,SAL_CIR.CD_SAL_CIR
        ,HR_SAL_CIR.HR_INICIAL

--R_OCIO_CIR_R_1_Q_ANALITICO

        CEN_CIR.DS_CEN_CIR,
        T.DATA DT_CALENDARIO,
        SAL_CIR.CD_SAL_CIR,
	SAL_CIR.DS_SAL_CIR,
        HR_SAL_CIR.HR_INICIAL,
	HR_SAL_CIR.HR_FINAL,
        SAL_CIR.DS_RESUMIDA,
	SAL_CIR.NR_TEMPO_PREPARO,
 	T.TEMPO_DISPONIVEL AS TEMPO_DISPONIVEL,
	T.TEMPO_INTERDITADO AS TEMPO_INTERDITADO,
	T.TEMPO_UTILIZADO AS TEMPO_UTILIZADO,

--R_OCIO_CIR_R_5_Q_SINTETICO
SELECT CD_CEN_CIR, CD_SAL_CIR , DS_CEN_CIR, DS_SAL_CIR,
 TO_CHAR(nvl(TRUNC(Sum( To_Char(TEMPO_DISPONIVEL,'sssss'))/3600),0),'FM9900') || ':' ||
    TO_CHAR(nvl(TRUNC(MOD(Sum( To_Char(TEMPO_DISPONIVEL,'sssss')),3600)/60),0),'FM00') AS SOMA_TEMPO_DISPONIVEL,
TO_CHAR(nvl(TRUNC(Sum( To_Char(TEMPO_UTILIZADO,'sssss'))/3600),0),'FM9900') || ':' ||
    TO_CHAR(nvl(TRUNC(MOD(Sum( To_Char(TEMPO_UTILIZADO,'sssss')),3600)/60),0),'FM00')  AS SOMA_TEMPO_UTILIZADO,
    TO_CHAR(nvl(TRUNC(Sum( To_Char(TEMPO_OCIOSO,'sssss'))/3600),0),'FM9900') || ':' ||
    TO_CHAR(nvl(TRUNC(MOD(Sum( To_Char(TEMPO_OCIOSO,'sssss')),3600)/60),0),'FM00')  AS SOMA_TEMPO_OCIOSO,
     TO_CHAR(nvl(TRUNC(Sum( To_Char(TEMPO_INTERDITADO,'sssss'))/3600),0),'FM9900') || ':' ||
    TO_CHAR(nvl(TRUNC(MOD(Sum( To_Char(TEMPO_INTERDITADO,'sssss')),3600)/60),0),'FM00') AS SOMA_TEMPO_INTERDITADO,  Avg(PERCENTUAL_OCIOSO) AS SOMA_PERCENTUAL_OCIOSO
  FROM (
SELECT  CEN_CIR.CD_CEN_CIR,
        CEN_CIR.DS_CEN_CIR,
        SAL_CIR.CD_SAL_CIR,
	SAL_CIR.DS_SAL_CIR,
        HR_SAL_CIR.HR_INICIAL,
	HR_SAL_CIR.HR_FINAL,
        SAL_CIR.DS_RESUMIDA,
	SAL_CIR.NR_TEMPO_PREPARO,
 	T.TEMPO_DISPONIVEL AS TEMPO_DISPONIVEL,
	T.TEMPO_INTERDITADO AS TEMPO_INTERDITADO,
	T.TEMPO_UTILIZADO AS TEMPO_UTILIZADO,
	T.TEMPO_OCIOSO AS TEMPO_OCIOSO,
        T.PERCENTUAL_OCIOSO AS PERCENTUAL_OCIOSO
FROM DBAMV.TB_FSCC_REL_OCIOSIDADE T,
     CEN_CIR,
     HR_SAL_CIR,
     SAL_CIR,
     SETOR
WHERE T.SALA = SAL_CIR.CD_SAL_CIR
AND HR_SAL_CIR.CD_SAL_CIR = SAL_CIR.CD_SAL_CIR
AND SAL_CIR.CD_CEN_CIR = CEN_CIR.CD_CEN_CIR
AND HR_SAL_CIR.TP_DIA_SEMANA - Nvl(dbamv.pkg_mv2000.le_configuracao('FSCC', 'QT_DIAS_A_MENOS_DIA_SEMANA'), 0) = To_Char(T.DATA,'D')
AND SETOR.CD_SETOR = SAL_CIR.CD_SETOR
AND SETOR.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
ORDER BY CEN_CIR.CD_CEN_CIR
        ,SAL_CIR.CD_SAL_CIR
        ,HR_SAL_CIR.HR_INICIAL
)
GROUP BY CD_CEN_CIR, CD_SAL_CIR, DS_CEN_CIR, DS_SAL_CIR
ORDER BY CD_CEN_CIR,CD_SAL_CIR
