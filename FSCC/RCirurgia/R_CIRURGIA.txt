--R_CIRURGIA
SELECT CIRURGIA.CD_CIRURGIA   Codigo
     , CIRURGIA.DS_CIRURGIA   Descricao
     , CIRURGIA.TP_CIRURGIA   TP_CIRURGIA
     , DECODE (:CAMPO_TIPO_PROC
               , 'Convenio', PRO_FAT.CD_PRO_FAT
               , 'SUS'     , PROCEDIMENTO_SUS_SIH.CD_PROCEDIMENTO
               , PROCEDIMENTO_SUS_SIA.CD_PROCEDIMENTO) CD_PRO_FAT
     , DECODE (:CAMPO_TIPO_PROC
                , 'Convenio', PRO_FAT.DS_PRO_FAT
                , 'SUS'     , PROCEDIMENTO_SUS_SIH.DS_PROCEDIMENTO
                , PROCEDIMENTO_SUS_SIA.DS_PROCEDIMENTO ) DS_PRO_FAT
     , CIRURGIA.DS_PREPARO     DS_PREPARO
     , CIRURGIA.DS_ORIENTACAO  DS_ORIENTACAO
     , PORTE_CIRURGIA.DS_PORTE_CIRURGIA DS_PORTE_CIRURGIA
  FROM DBAMV.CIRURGIA      CIRURGIA
     , DBAMV.PRO_FAT       PRO_FAT
     , DBAMV.PROCEDIMENTO_SUS Procedimento_sus_sih
     , DBAMV.PROCEDIMENTO_SUS Procedimento_sus_sia
     , DBAMV.PORTE_CIRURGIA   PORTE_CIRURGIA
 WHERE CIRURGIA.CD_PRO_FAT          = PRO_FAT.CD_PRO_FAT(+)
   and cirurgia.cd_procedimento_sih = procedimento_sus_sih.cd_procedimento(+)
   and cirurgia.cd_procedimento_sia = procedimento_sus_sia.cd_procedimento(+)
   AND CIRURGIA.CD_PORTE_CIRURGIA   = PORTE_CIRURGIA.CD_PORTE_CIRURGIA(+)
   ORDER BY
