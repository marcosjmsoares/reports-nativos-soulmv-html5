--R_STATUS_CIRURGIA
SELECT * FROM
  (SELECT avi_cir.cd_aviso_cirurgia cd_aviso_cirurgia ,
          avi_cir.cd_atendimento cd_atendimento ,
          avi_cir.cd_paciente cd_paciente ,
          avi_cir.nm_paciente nm_paciente ,
     (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND avi_cir.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
          trunc(avi_cir.dt_realizacao) dt_realizacao ,
          cirurgia.cd_cirurgia cd_cirurgia ,
          cirurgia.ds_cirurgia ds_cirurgia
   FROM dbamv.aviso_cirurgia avi_cir ,
        dbamv.cirurgia cirurgia ,
        dbamv.cirurgia_aviso cir_avi ,
        dbamv.atendime atendime
   WHERE avi_cir.cd_aviso_cirurgia = cir_avi.cd_aviso_cirurgia
     AND atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA 
     AND cir_avi.cd_cirurgia = cirurgia.cd_cirurgia
     AND avi_cir.tp_situacao = 'R'
     AND avi_cir.cd_atendimento = atendime.cd_atendimento
     AND trunc(avi_cir.dt_realizacao) BETWEEN :CAMPO_DATA_INI ORDER BY avi_cir.dt_realizacao,
            avi_cir.nm_paciente)
ORDER BY DT_REALIZACAO,
         NM_PACIENTE,
         CD_AVISO_CIRURGIA,
         CD_ATENDIMENTO,
         CD_PACIENTE
