--R_MOTIVO_PARTO
select tip.cd_tip_parto,
       tip.ds_tip_parto,
       mot.cd_motivo_parto,
       decode(mot.ds_motivo_parto, null, 'Não Informado',ds_motivo_parto) ds_motivo_parto,
       ad.dt_parto,
       at.cd_atendimento,
       pac.nm_paciente,
       (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND pac.CD_PACIENTE =  CD_PACIENTE) NM_SOCIAL_PACIENTE,
       pac.ds_endereco,
       cid.cd_cidade,
       decode(cid.nm_cidade, null, 'Não Informado', cid.nm_cidade) nm_cidade,
       cid.cd_uf,
       con.cd_convenio,
       con.nm_convenio,
       pres.cd_prestador,
       decode(pres.nm_prestador, null, 'Não Informado', pres.nm_prestador) nm_prestador
from   dbamv.admissao_co  ad,
       dbamv.tip_parto    tip,
       dbamv.motivo_parto mot,
       dbamv.atendime     at,
       dbamv.paciente     pac,
       dbamv.cidade       cid,
       dbamv.convenio     con,
       dbamv.prestador    pres,
       Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Con.Cd_Convenio
And  at.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And ad.cd_tip_parto     = tip.cd_tip_parto
and    ad.cd_motivo_parto  = mot.cd_motivo_parto(+)
and    trunc(ad.dt_parto) between :CAMPO_DATA_INI and 
and    ad.cd_atendimento   = at.cd_atendimento
and    at.cd_paciente      = pac.cd_paciente
and    pac.cd_cidade       = cid.cd_cidade(+)
and    at.cd_convenio      = con.cd_convenio
and    at.cd_prestador     = pres.cd_prestador(+)
ORDER BY  tip.cd_tip_parto desc, ds_motivo_parto, ad.dt_parto desc ,pac.nm_paciente
