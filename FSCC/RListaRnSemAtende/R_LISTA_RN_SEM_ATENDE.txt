--R_LISTA_RN_SEM_ATENDE
SELECT ATENDIME.CD_ATENDIMENTO CD_ATENDIMENTO ,
       PACIENTE.CD_PACIENTE CD_PACIENTE ,
       PACIENTE.NM_PACIENTE NM_PACIENTE ,
   (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
       RECEM_NASCIDO.NM_RECEM_NASCIDO NM_RECEM_NASCIDO ,
       TRUNC(ADMISSAO_CO.DT_PARTO) DT_PARTO ,
       TO_CHAR(RECEM_NASCIDO.HR_NASCIMENTO, 'HH24') HR_NASCIMENTO
FROM DBAMV.RECEM_NASCIDO ,
     DBAMV.ADMISSAO_CO ,
     DBAMV.ATENDIME ,
     DBAMV.PACIENTE
WHERE RECEM_NASCIDO.CD_ATENDIMENTO IS NULL
  AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA 
  AND RECEM_NASCIDO.CD_ADMISSAO_CO = ADMISSAO_CO.CD_ADMISSAO_CO
  AND ADMISSAO_CO.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
  AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
  AND ATENDIME.DT_ALTA IS NULL
