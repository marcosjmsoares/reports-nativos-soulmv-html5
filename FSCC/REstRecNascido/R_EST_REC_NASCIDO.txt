--R_EST_REC_NASCIDO
SELECT * FROM (
SELECT  CD_CONVENIO,
	NM_CONVENIO,
	CD_TIP_PARTO,
	DS_TIP_PARTO,
	SEXO,
	AVG(PESO_NASCIDOS)		PESO_NASCIDOS,
	SUM(NASCIDOS)			NASCIDOS,
	SUM(OBITOS)			OBITOS
FROM
(
SELECT CONVENIO.CD_CONVENIO				CD_CONVENIO,
       CONVENIO.NM_CONVENIO				NM_CONVENIO,
       TIP_PARTO.CD_TIP_PARTO				CD_TIP_PARTO,
       DECODE(TIP_PARTO.DS_TIP_PARTO, NULL, 'Não Informado', TIP_PARTO.DS_TIP_PARTO) 	DS_TIP_PARTO,
       DECODE(REC.TP_SEXO,'M', 'MASCULINO', 'FEMININO') 	SEXO,
       NVL(REC.VL_PESO,0)					PESO_NASCIDOS,
       DECODE(REC.CD_TIP_MORTE_RN, NULL, 1, 0)  		NASCIDOS,
       DECODE(REC.CD_TIP_MORTE_RN, NULL, 0, 1)  		OBITOS
FROM DBAMV.ADMISSAO_CO     AD,
     DBAMV.RECEM_NASCIDO   REC,
     DBAMV.ATENDIME        ATE,
     DBAMV.CONVENIO,
     DBAMV.TIP_PARTO,
     Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And ate.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And AD.CD_ADMISSAO_CO  = REC.CD_ADMISSAO_CO
AND AD.CD_ATENDIMENTO  = ATE.CD_ATENDIMENTO
AND TRUNC(ad.dt_parto) BETWEEN :CAMPO_DATA_INI AND 
AND ATE.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND AD.CD_TIP_PARTO = TIP_PARTO.CD_TIP_PARTO(+)
)
GROUP BY  CD_CONVENIO,
	NM_CONVENIO,
	CD_TIP_PARTO,
	DS_TIP_PARTO,
	SEXO
ORDER BY CD_CONVENIO
)
ORDER BY CD_CONVENIO,NM_CONVENIO,
CD_TIP_PARTO,DS_TIP_PARTO,
SEXO
