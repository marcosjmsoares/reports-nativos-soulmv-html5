--R_GRUPO_CIRURGIA
SELECT * FROM (
select Grupo_Cirurgia.Cd_Grupo_Cirurgia              Cd_Grupo_Cirurgia
     , Grupo_Cirurgia.Ds_Grupo_Cirurgia                  Ds_Grupo_Cirurgia
     , Grupo_Cirurgia.Sn_Ativo                                 SN_ATIVO
     , Sub_Grupo_Cirurgia.Cd_Sub_Grupo_Cirurgia CD_Sub_Grupo_Cirurgia
     , Sub_Grupo_Cirurgia.Ds_Sub_Grupo_Cirurgia Ds_Sub_Grupo_Cirurgia
  from   Dbamv.Grupo_cirurgia
         , Dbamv.Sub_grupo_cirurgia
 where Grupo_cirurgia.Cd_Grupo_cirurgia  = Sub_grupo_cirurgia.Cd_grupo_cirurgia
 order by 1,4
)
ORDER BY CD_GRUPO_CIRURGIA,CD_Sub_Grupo_Cirurgia,SN_ATIVO
