--R_COD_BARRA_AVISO
SELECT AVISO_CIRURGIA.CD_AVISO_CIRURGIA                   CD_PRODUTO
              ,AVISO_CIRURGIA.CD_AVISO_CIRURGIA                   CD_IDENTIFICADOR
              ,SUBSTR(PACIENTE.NM_PACIENTE,1,32)                  PROD_DS_PRODUTO
              ,SUBSTR(PACIENTE.NM_PACIENTE,33,13)                PROD_DS_PRODUTO2
              ,TO_CHAR(AVISO_CIRURGIA.CD_ATENDIMENTO)   DS_UNIDADE
FROM  DBAMV.AVISO_CIRURGIA
           ,DBAMV.ATENDIME
           ,DBAMV.PACIENTE
WHERE AVISO_CIRURGIA.CD_AVISO_CIRURGIA  =  :CAMPO_PCD_AVISO_CIRURGIA
AND       ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND       AVISO_CIRURGIA.CD_ATENDIMENTO  =  ATENDIME.CD_ATENDIMENTO
AND       ATENDIME.CD_PACIENTE  =  PACIENTE.CD_PACIENTE
