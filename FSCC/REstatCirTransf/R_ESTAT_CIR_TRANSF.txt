--R_ESTAT_CIR_TRANSF
SELECT * FROM (
select mot_transf.ds_mot_transferencia
      ,mot_transf.cd_mot_transferencia
      ,cen_cir.cd_cen_cir
      ,cen_cir.ds_cen_cir
      ,count(*) tot
from  dbamv.transferencia_cirurgia  transf_cir
     ,dbamv.mot_transferencia       mot_transf
     ,dbamv.aviso_cirurgia          avi_cir
     ,dbamv.cen_cir                 cen_cir
     ,dbamv.cirurgia_aviso       cir_avi
where transf_cir.cd_aviso_cirurgia    = avi_cir.cd_aviso_cirurgia
and   cen_cir.cd_cen_cir              = avi_cir.cd_cen_cir
and   mot_transf.cd_mot_transferencia = transf_cir.cd_mot_transferencia
AND   avi_cir.cd_multi_empresa = dbamv.pkg_mv2000.le_empresa
and   transf_cir.dt_transferencia between   To_Date(Concat(To_Char(:CAMPO_DATA_INI,'dd/mm/yyyy'),'00'),'dd/mm/yyyy hh24') and  To_Date(Concat(To_Char(:CAMPO_DATA_FIM,'dd/mm/yyyy'),'23'),'dd/mm/yyyy hh24')
and   avi_cir.cd_aviso_cirurgia = cir_avi.cd_aviso_cirurgia
group by cen_cir.cd_cen_cir,cen_cir.ds_cen_cir
         ,mot_transf.cd_mot_transferencia,mot_transf.ds_mot_transferencia
order by mot_transf.cd_mot_transferencia
)
