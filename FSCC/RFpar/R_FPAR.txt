--R_FPAR
SELECT * FROM
  (SELECT ADMISSAO_CO.DT_PARTO DT_PARTO ,
          ADMISSAO_CO.CD_ADMISSAO_CO CD_ADMISSAO_CO1 ,
          ADMISSAO_CO.DT_PARTO HR_PARTO ,
          ADMISSAO_CO.CD_ATENDIMENTO CD_ATENDIMENTO ,
          ADMISSAO_CO.NR_FILHOS_NO_PARTO NR_FILHOS_NO_PARTO ,
          ADMISSAO_CO.NR_CONSULTAS NR_CONSULTAS ,
          ADMISSAO_CO.QT_ABORTO QT_FILHOS_ABORTOS ,
          ADMISSAO_CO.QT_FILHOS_MORTOS QT_FILHOS_MORTOS ,
          ADMISSAO_CO.QT_FILHOS_VIVOS QT_FILHOS_VIVOS ,
          ADMISSAO_CO.VL_TEMPO_GESTACAO VL_TEMPO_GESTACAO ,
          ADMISSAO_CO.NM_PAI NM_PAI ,
          ADMISSAO_CO.DS_PARTO DS_PARTO ,
          TIP_PARTO.DS_TIP_PARTO DS_TIP_PARTO ,
          PACIENTE.CD_PACIENTE CD_PACIENTE ,
          PACIENTE.NM_PACIENTE NM_PACIENTE ,
      (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
          PACIENTE.DS_ENDERECO DS_ENDERECO ,
          PACIENTE.NM_BAIRRO NM_BAIRRO ,
          PACIENTE.NR_FONE NR_FONE ,
          LEITO.DS_ENFERMARIA DS_ENFERMARIA ,
          LEITO.DS_LEITO DS_LEITO ,
          UNID_INT.DS_UNID_INT DS_UNID_INT ,
          RECEM_NASCIDO.CD_ADMISSAO_CO CD_ADMISSAO_CO ,
          RECEM_NASCIDO.CD_RECEM_NASCIDO CD_RECEM_NASCIDO ,
          RECEM_NASCIDO.NM_RECEM_NASCIDO NM_RECEM_NASCIDO ,
          RECEM_NASCIDO.NR_PULSEIRA NR_PULSEIRA ,
          RECEM_NASCIDO.TP_RACA TP_RACA ,
          RECEM_NASCIDO.TP_SEXO TP_SEXO ,
          RECEM_NASCIDO.VL_ALTURA VL_ALTURA ,
          RECEM_NASCIDO.VL_APGAR_1MINUTO VL_APGAR_1MINUTO ,
          RECEM_NASCIDO.VL_PESO VL_PESO ,
          RECEM_NASCIDO.CD_DECLARACAO_NASCIDO_VIVO CD_DECLARACAO_NASCIDO_VIVO ,
          RECEM_NASCIDO.VL_APGAR_5MINUTO VL_APGAR_5MINUTO ,
          RECEM_NASCIDO.VL_PERIMETRO_CEFALICO VL_PERIMETRO_CEFALICO ,
          RECEM_NASCIDO.VL_PERIMETRO_TORAXICO VL_PERIMETRO_TORAXICO ,
          RECEM_NASCIDO.VL_PERIMETRO_ABDOMINAL VL_PERIMETRO_ABDOMINAL ,
          RECEM_NASCIDO.HR_NASCIMENTO HR_NASCIMENTO ,
          TIP_MOR_RN.DS_TIP_MOR_RN DS_TIP_MOR_RN ,
          CONVENIO.NM_CONVENIO NM_CONVENIO ,
          RECEM_NASCIDO.CD_PRESTADOR CD_PRESTADOR ,
          PRESTADOR.NM_PRESTADOR NM_PRESTADOR
   FROM Dbamv.RECEM_NASCIDO RECEM_NASCIDO ,
        Dbamv.ADMISSAO_CO ADMISSAO_CO ,
        Dbamv.TIP_PARTO TIP_PARTO ,
        Dbamv.ATENDIME ATENDIME ,
        Dbamv.PACIENTE PACIENTE ,
        Dbamv.LEITO LEITO ,
        Dbamv.UNID_INT UNID_INT ,
        Dbamv.TIP_MOR_RN TIP_MOR_RN ,
        Dbamv.CONVENIO CONVENIO ,
        Dbamv.PRESTADOR PRESTADOR,
        Dbamv.Empresa_Convenio
   WHERE Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
     AND Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA 
     AND ADMISSAO_CO.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
     AND ADMISSAO_CO.CD_ADMISSAO_CO = RECEM_NASCIDO.CD_ADMISSAO_CO
     AND ADMISSAO_CO.CD_TIP_PARTO = TIP_PARTO.CD_TIP_PARTO 
     AND PACIENTE.CD_PACIENTE = ATENDIME.CD_PACIENTE
     AND LEITO.CD_LEITO = ATENDIME.CD_LEITO
     AND CONVENIO.CD_CONVENIO = ATENDIME.CD_CONVENIO
     AND TIP_MOR_RN.CD_TIP_MOR_RN (+) = RECEM_NASCIDO.CD_TIP_MORTE_RN
     AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT
     AND RECEM_NASCIDO.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
   ORDER BY ADMISSAO_CO.CD_ADMISSAO_CO)
ORDER BY DT_PARTO,
         DS_TIP_PARTO,
         CD_ATENDIMENTO,
         NM_PACIENTE,
         DS_ENDERECO,
         NM_BAIRRO,
         NR_FONE,
         DS_ENFERMARIA,
         DS_LEITO,
         DS_UNID_INT,
         NR_FILHOS_NO_PARTO,
         NR_CONSULTAS,
         TP_RACA,
         VL_TEMPO_GESTACAO,
         NM_PAI,
         DS_PARTO,
         HR_PARTO,
         VL_ALTURA,
         VL_PESO,
         NM_CONVENIO,
         CD_ADMISSAO_CO,
         QT_FILHOS_ABORTOS,
         QT_FILHOS_MORTOS,
         QT_FILHOS_VIVOS,
         CD_RECEM_NASCIDO,
         CD_PACIENTE
