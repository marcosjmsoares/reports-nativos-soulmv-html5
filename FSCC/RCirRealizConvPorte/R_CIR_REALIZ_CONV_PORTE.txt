--R_CIR_REALIZ_CONV_PORTE
SELECT cd_convenio,
       nm_convenio,
       data,
       SUM(pequeno) pequeno,
       SUM(medio) medio,
       SUM(grande) grande,
       SUM(especial) especial
FROM  (SELECT  convenio.cd_convenio cd_convenio,
               convenio.nm_convenio nm_convenio,
               TRUNC(aviso_cirurgia.dt_realizacao) data,
               COUNT(cirurgia_aviso.cd_cirurgia) Pequeno,
               0 Medio,
               0 Grande,
               0 Especial
       FROM    dbamv.aviso_cirurgia
              ,dbamv.cirurgia_aviso
              ,dbamv.atendime
              ,dbamv.convenio
              ,dbamv.cirurgia
	      ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And aviso_cirurgia.tp_situacao =   'R'
       AND    aviso_cirurgia.cd_aviso_cirurgia = cirurgia_aviso.cd_aviso_cirurgia
       AND    cirurgia_aviso.cd_cirurgia = cirurgia.cd_cirurgia
       AND    aviso_cirurgia.cd_atendimento = atendime.cd_atendimento
       AND    cirurgia.tp_cirurgia = 'P'
       AND    atendime.cd_convenio = convenio.cd_convenio
       AND    aviso_cirurgia.dt_realizacao BETWEEN To_Date(Concat(To_Char(:CAMPO_DATA_INI,'dd/mm/yyyy'),'00'),'dd/mm/yyyy hh24') and  To_Date(Concat(To_Char(:CAMPO_DATA_FIM,'dd/mm/yyyy'),'23'),'dd/mm/yyyy hh24')
       GROUP BY convenio.cd_convenio,
                convenio.nm_convenio,
                TRUNC(aviso_cirurgia.dt_realizacao)
       UNION
       SELECT  convenio.cd_convenio cd_convenio,
               convenio.nm_convenio nm_convenio,
               TRUNC(aviso_cirurgia.dt_realizacao) data,
               0 Pequeno,
               COUNT(cirurgia_aviso.cd_cirurgia) Medio,
               0 Grande,
               0 Especial
       FROM    dbamv.aviso_cirurgia
              ,dbamv.cirurgia_aviso
              ,dbamv.atendime
              ,dbamv.convenio
              ,dbamv.cirurgia
	      ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And aviso_cirurgia.tp_situacao =   'R'
       AND    aviso_cirurgia.cd_aviso_cirurgia = cirurgia_aviso.cd_aviso_cirurgia
       AND    cirurgia_aviso.cd_cirurgia = cirurgia.cd_cirurgia
       AND    aviso_cirurgia.cd_atendimento = atendime.cd_atendimento
       AND    cirurgia.tp_cirurgia = 'M'
       AND    atendime.cd_convenio = convenio.cd_convenio
        AND    aviso_cirurgia.dt_realizacao BETWEEN To_Date(Concat(To_Char(:CAMPO_DATA_INI,'dd/mm/yyyy'),'00'),'dd/mm/yyyy hh24') and  To_Date(Concat(To_Char(:CAMPO_DATA_FIM,'dd/mm/yyyy'),'23'),'dd/mm/yyyy hh24')
       GROUP BY convenio.cd_convenio,
                convenio.nm_convenio,
                TRUNC(aviso_cirurgia.dt_realizacao)
       UNION
       SELECT  convenio.cd_convenio cd_convenio,
               convenio.nm_convenio nm_convenio,
               TRUNC(aviso_cirurgia.dt_realizacao) data,
               0 Pequeno,
               0 Medio,
               COUNT(cirurgia_aviso.cd_cirurgia) Grande,
               0 Especial
       FROM    dbamv.aviso_cirurgia
              ,dbamv.cirurgia_aviso
              ,dbamv.atendime
              ,dbamv.convenio
              ,dbamv.cirurgia
	      ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And aviso_cirurgia.tp_situacao =   'R'
       AND    aviso_cirurgia.cd_aviso_cirurgia = cirurgia_aviso.cd_aviso_cirurgia
       AND    cirurgia_aviso.cd_cirurgia = cirurgia.cd_cirurgia
       AND    aviso_cirurgia.cd_atendimento = atendime.cd_atendimento
       AND    cirurgia.tp_cirurgia = 'G'
       AND    atendime.cd_convenio = convenio.cd_convenio
       AND    aviso_cirurgia.dt_realizacao BETWEEN To_Date(Concat(To_Char(:CAMPO_DATA_INI,'dd/mm/yyyy'),'00'),'dd/mm/yyyy hh24') and  To_Date(Concat(To_Char(:CAMPO_DATA_FIM,'dd/mm/yyyy'),'23'),'dd/mm/yyyy hh24')
       GROUP BY convenio.cd_convenio,
                convenio.nm_convenio,
                TRUNC(aviso_cirurgia.dt_realizacao)
       UNION
       SELECT  convenio.cd_convenio cd_convenio,
               convenio.nm_convenio nm_convenio,
               TRUNC(aviso_cirurgia.dt_realizacao) data,
               0 Pequeno,
               0 Medio,
               0 Grande,
                COUNT(cirurgia_aviso.cd_cirurgia) Especial
       FROM    dbamv.aviso_cirurgia
              ,dbamv.cirurgia_aviso
              ,dbamv.atendime
              ,dbamv.convenio
              ,dbamv.cirurgia
	      ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And aviso_cirurgia.tp_situacao =   'R'
       AND    aviso_cirurgia.cd_aviso_cirurgia = cirurgia_aviso.cd_aviso_cirurgia
       AND    cirurgia_aviso.cd_cirurgia = cirurgia.cd_cirurgia
       AND    aviso_cirurgia.cd_atendimento = atendime.cd_atendimento
       AND    cirurgia.tp_cirurgia = 'E'
       AND    atendime.cd_convenio = convenio.cd_convenio
       AND    aviso_cirurgia.dt_realizacao BETWEEN To_Date(Concat(To_Char(:CAMPO_DATA_INI,'dd/mm/yyyy'),'00'),'dd/mm/yyyy hh24') and  To_Date(Concat(To_Char(:CAMPO_DATA_FIM,'dd/mm/yyyy'),'23'),'dd/mm/yyyy hh24')
       GROUP BY convenio.cd_convenio,
                convenio.nm_convenio,
                TRUNC(aviso_cirurgia.dt_realizacao)
)
GROUP BY cd_convenio,
                  nm_convenio,
                 data
ORDER BY CD_CONVENIO, DATA
