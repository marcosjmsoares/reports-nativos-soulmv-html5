--R_CIR_SALA
SELECT * FROM
  (SELECT AVISO_CIRURGIA.CD_AVISO_CIRURGIA ,
          AVISO_CIRURGIA.DT_REALIZACAO ,
          CIRURGIA.DS_CIRURGIA ,
          PACIENTE.CD_PACIENTE ,
          PACIENTE.NM_PACIENTE ,
      (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
          SAL_CIR.CD_SAL_CIR ,
          SAL_CIR.DS_SAL_CIR ,
          CEN_CIR.CD_CEN_CIR ,
          CEN_CIR.DS_CEN_CIR ,
          CIRURGIA_AVISO.CD_CIRURGIA CD_CIRURGIA
   FROM DBAMV.AVISO_CIRURGIA ,
        DBAMV.CIRURGIA_AVISO ,
        DBAMV.CIRURGIA ,
        DBAMV.SAL_CIR ,
        DBAMV.CEN_CIR ,
        DBAMV.PACIENTE ,
        DBAMV.ATENDIME ,
     (SELECT P_A.CD_PRESTADOR,
             A_M.DS_ATI_MED,
             P_A.CD_CIRURGIA_AVISO,
             P_A.CD_AVISO_CIRURGIA
      FROM DBAMV.PRESTADOR_AVISO P_A,
           DBAMV.ATI_MED A_M
      WHERE A_M.CD_ATI_MED = P_A.CD_ATI_MED
        AND SN_PRINCIPAL = 'S'
        AND A_M.TP_FUNCAO = 'C') PRESTADOR_AVISO
   WHERE AVISO_CIRURGIA.TP_SITUACAO = 'R'
     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA 
     AND AVISO_CIRURGIA.CD_SAL_CIR = SAL_CIR.CD_SAL_CIR
     AND AVISO_CIRURGIA.CD_CEN_CIR = CEN_CIR.CD_CEN_CIR
     AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
     AND CIRURGIA_AVISO.CD_CIRURGIA = CIRURGIA.CD_CIRURGIA
     AND AVISO_CIRURGIA.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
     AND AVISO_CIRURGIA.DT_REALIZACAO BETWEEN To_Date(Concat(To_Char(:CAMPO_DATA_INI,'dd/mm/yyyy'), '00'), 'dd/mm/yyyy hh24') AND To_Date(Concat(To_Char(:CAMPO_DATA_FIM,'dd/mm/yyyy'), '23'), 'dd/mm/yyyy hh24')
     AND CIRURGIA_AVISO.CD_CIRURGIA_AVISO = PRESTADOR_AVISO.CD_CIRURGIA_AVISO(+)  
   ORDER BY CD_SAL_CIR,
            DT_REALIZACAO,
            DS_CIRURGIA)
ORDER BY CD_CEN_CIR,
         DS_CEN_CIR,
         CD_SAL_CIR,
         DS_SAL_CIR,
         DT_REALIZACAO,
         CD_AVISO_CIRURGIA,
         CD_PACIENTE,
         CD_CIRURGIA,
         NM_PACIENTE

--R_CIR_SALA_R_PRESTADOR_Q_PRESTADOR
SELECT * FROM (
(SELECT AVISO.CD_AVISO_CIRURGIA
               ,PRESTADOR.CD_PRESTADOR
               ,PRESTADOR.NM_PRESTADOR
 FROM   DBAMV.AVISO_CIRURGIA          AVISO
             ,DBAMV.PRESTADOR_AVISO
             ,DBAMV.PRESTADOR
             ,DBAMV.ATI_MED
             ,DBAMV.CIRURGIA_AVISO
 WHERE   AVISO.CD_AVISO_CIRURGIA   =  CIRURGIA_AVISO.CD_AVISO_CIRURGIA
 AND        CIRURGIA_AVISO.CD_AVISO_CIRURGIA    =  PRESTADOR_AVISO.CD_AVISO_CIRURGIA
 AND        CIRURGIA_AVISO.CD_CIRURGIA                =  PRESTADOR_AVISO.CD_CIRURGIA
 AND        PRESTADOR_AVISO.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
 AND        PRESTADOR_AVISO.CD_ATI_MED = ATI_MED.CD_ATI_MED
 AND        ATI_MED.TP_FUNCAO = 'C'
 AND        AVISO.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
 AND        CIRURGIA_AVISO.CD_CIRURGIA = :CAMPO_CD_CIRURGIA
 AND        PRESTADOR_AVISO.CD_CIRURGIA = :CAMPO_CD_CIRURGIA
 MINUS
 SELECT AVISO.CD_AVISO_CIRURGIA
               ,PRESTADOR.CD_PRESTADOR
               ,PRESTADOR.NM_PRESTADOR
 FROM   DBAMV.AVISO_CIRURGIA          AVISO
              ,DBAMV.EQUIPE_MEDICA
              ,DBAMV.EQUIPE_PRESTADOR
              ,DBAMV.PRESTADOR
              ,DBAMV.ATI_MED
 WHERE  AVISO.CD_EQUIPE_MEDICA = EQUIPE_MEDICA.CD_EQUIPE_MEDICA
 AND        EQUIPE_MEDICA.CD_EQUIPE_MEDICA = EQUIPE_PRESTADOR.CD_EQUIPE_MEDICA
 AND        AVISO.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
 AND        EQUIPE_PRESTADOR.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
 AND        EQUIPE_PRESTADOR.CD_ATI_MED = ATI_MED.CD_ATI_MED
 AND        ATI_MED.TP_FUNCAO = 'C'  )
 UNION ALL
 SELECT AVISO.CD_AVISO_CIRURGIA
               ,PRESTADOR.CD_PRESTADOR
               ,PRESTADOR.NM_PRESTADOR
 FROM   DBAMV.AVISO_CIRURGIA    AVISO
             ,DBAMV.EQUIPE_MEDICA
             ,DBAMV.EQUIPE_PRESTADOR
             ,DBAMV.PRESTADOR
             ,DBAMV.ATI_MED
 WHERE  AVISO.CD_EQUIPE_MEDICA = EQUIPE_MEDICA.CD_EQUIPE_MEDICA
 AND        EQUIPE_MEDICA.CD_EQUIPE_MEDICA = EQUIPE_PRESTADOR.CD_EQUIPE_MEDICA
 AND        EQUIPE_PRESTADOR.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
 AND        EQUIPE_PRESTADOR.CD_ATI_MED = ATI_MED.CD_ATI_MED
 AND        ATI_MED.TP_FUNCAO = 'C'
 AND        AVISO.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
 AND        AVISO.CD_AVISO_CIRURGIA NOT IN
                            (SELECT AVISO_CIRURGIA.CD_AVISO_CIRURGIA
                             FROM    DBAMV.AVISO_CIRURGIA
                                          ,DBAMV.PRESTADOR_AVISO
                                          ,DBAMV.ATI_MED
                                          ,DBAMV.CIRURGIA_AVISO
                              WHERE   AVISO.CD_AVISO_CIRURGIA   =  CIRURGIA_AVISO.CD_AVISO_CIRURGIA
                              AND        CIRURGIA_AVISO.CD_AVISO_CIRURGIA    =  PRESTADOR_AVISO.CD_AVISO_CIRURGIA
                              AND        CIRURGIA_AVISO.CD_CIRURGIA                =  PRESTADOR_AVISO.CD_CIRURGIA
                             AND        PRESTADOR_AVISO.CD_ATI_MED = ATI_MED.CD_ATI_MED
                             AND        AVISO.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
                             AND        ATI_MED.TP_FUNCAO = 'C')
)
WHERE CD_AVISO_CIRURGIA = :CAMPO_CD_AVISO_CIRURGIA
