--R_CIR_REALIZ_PORTE
SELECT
dt_realizacao
,sum(tot_peq) tot_peq
,sum(tot_med) tot_med
,sum(tot_gra) tot_gra
,sum(tot_esp) tot_esp
,sum(tot_ate_peq) tot_ate_peq
,sum(tot_ate_med) tot_ate_med
,sum(tot_ate_gra) tot_ate_gra
,sum(tot_ate_esp) tot_ate_esp
 FROM (
SELECT to_date(to_char(AVISO_CIRURGIA.DT_REALIZACAO,'dd/mm/yyyy'),'dd/mm/yyyy') DT_REALIZACAO
            ,  COUNT(AVISO_CIRURGIA.CD_AVISO_CIRURGIA)     TOT_PEQ
            ,  0                                                                                      TOT_MED
            ,  0                                                                                      TOT_GRA
            ,  0                                                                                      TOT_ESP
            ,  0                                                                                      TOT_ATE_PEQ
            ,  0                                                                                      TOT_ATE_MED
            ,  0                                                                                      TOT_ATE_GRA
            ,  0                                                                                      TOT_ATE_ESP
FROM DBAMV.AVISO_CIRURGIA      AVISO_CIRURGIA
           ,DBAMV.CIRURGIA_AVISO     CIRURGIA_AVISO
           ,DBAMV.CIRURGIA                  CIRURGIA
           ,DBAMV.CONVENIO                 CONVENIO
	   ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And AVISO_CIRURGIA.TP_SITUACAO = 'R'
AND   AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
AND   CIRURGIA_AVISO.CD_CIRURGIA = CIRURGIA.CD_CIRURGIA
AND   CIRURGIA.TP_CIRURGIA = 'P'
AND   TO_CHAR(AVISO_CIRURGIA.DT_REALIZACAO, 'MM/YYYY') = :CAMPO_DATA
AND    CIRURGIA_AVISO.CD_CONVENIO = CONVENIO.CD_CONVENIO
GROUP BY DT_REALIZACAO
UNION ALL
SELECT to_date(to_char(AVISO_CIRURGIA.DT_REALIZACAO,'dd/mm/yyyy'),'dd/mm/yyyy')             DT_REALIZACAO
            ,  0                                                                                     TOT_PEQ
            ,  0                                                                                      TOT_MED
            ,  0                                                                                      TOT_GRA
            ,  0                                                                                      TOT_ESP
            ,  COUNT(DISTINCT AVISO_CIRURGIA.CD_AVISO_CIRURGIA)   TOT_ATE_PEQ
            ,  0                                                                                      TOT_ATE_MED
            ,  0                                                                                      TOT_ATE_GRA
            ,  0                                                                                      TOT_ATE_ESP
FROM DBAMV.AVISO_CIRURGIA      AVISO_CIRURGIA
           ,DBAMV.CIRURGIA_AVISO     CIRURGIA_AVISO
           ,DBAMV.CIRURGIA                  CIRURGIA
           ,DBAMV.CONVENIO                 CONVENIO
	   ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And AVISO_CIRURGIA.TP_SITUACAO = 'R'
AND   AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
AND   CIRURGIA_AVISO.CD_CIRURGIA = CIRURGIA.CD_CIRURGIA
AND   CIRURGIA.TP_CIRURGIA = 'P'
AND   TO_CHAR(AVISO_CIRURGIA.DT_REALIZACAO, 'MM/YYYY') = :CAMPO_DATA
AND    CIRURGIA_AVISO.CD_CONVENIO = CONVENIO.CD_CONVENIO
GROUP BY DT_REALIZACAO
UNION ALL
SELECT to_date(to_char(AVISO_CIRURGIA.DT_REALIZACAO,'dd/mm/yyyy'),'dd/mm/yyyy')             DT_REALIZACAO
            ,  0                                                                                   TOT_PEQ
            ,  COUNT(AVISO_CIRURGIA.CD_AVISO_CIRURGIA)  TOT_MED
            ,  0                                                                                   TOT_GRA
            ,  0                                                                                   TOT_ESP
            ,  0                                                                                      TOT_ATE_PEQ
            ,  0                                                                                      TOT_ATE_MED
            ,  0                                                                                      TOT_ATE_GRA
            ,  0                                                                                      TOT_ATE_ESP
FROM DBAMV.AVISO_CIRURGIA      AVISO_CIRURGIA
          ,DBAMV.CIRURGIA_AVISO      CIRURGIA_AVISO
          ,DBAMV.CIRURGIA                   CIRURGIA
          ,DBAMV.CONVENIO 	             CONVENIO
	  ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And AVISO_CIRURGIA.TP_SITUACAO = 'R'
AND   AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
AND   CIRURGIA_AVISO.CD_CIRURGIA = CIRURGIA.CD_CIRURGIA
AND   CIRURGIA.TP_CIRURGIA = 'M'
AND   TO_CHAR(AVISO_CIRURGIA.DT_REALIZACAO, 'MM/YYYY') = :CAMPO_DATA
AND    CIRURGIA_AVISO.CD_CONVENIO = CONVENIO.CD_CONVENIO
GROUP BY DT_REALIZACAO
UNION ALL
SELECT to_date(to_char(AVISO_CIRURGIA.DT_REALIZACAO,'dd/mm/yyyy'),'dd/mm/yyyy')             DT_REALIZACAO
            ,  0                                                                                   TOT_PEQ
            ,  0                                                                                   TOT_MED
            ,  0                                                                                   TOT_GRA
            ,  0                                                                                   TOT_ESP
            ,  0                                                                                      TOT_ATE_PEQ
            ,  COUNT(DISTINCT AVISO_CIRURGIA.CD_AVISO_CIRURGIA)     TOT_ATE_MED
            ,  0                                                                                      TOT_ATE_GRA
            ,  0                                                                                      TOT_ATE_ESP
FROM DBAMV.AVISO_CIRURGIA      AVISO_CIRURGIA
          ,DBAMV.CIRURGIA_AVISO      CIRURGIA_AVISO
          ,DBAMV.CIRURGIA                   CIRURGIA
          ,DBAMV.CONVENIO 	             CONVENIO
	  ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And AVISO_CIRURGIA.TP_SITUACAO = 'R'
AND   AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
AND   CIRURGIA_AVISO.CD_CIRURGIA = CIRURGIA.CD_CIRURGIA
AND   CIRURGIA.TP_CIRURGIA = 'M'
AND   TO_CHAR(AVISO_CIRURGIA.DT_REALIZACAO, 'MM/YYYY') = :CAMPO_DATA
AND    CIRURGIA_AVISO.CD_CONVENIO = CONVENIO.CD_CONVENIO
GROUP BY DT_REALIZACAO
UNION ALL
SELECT to_date(to_char(AVISO_CIRURGIA.DT_REALIZACAO,'dd/mm/yyyy'),'dd/mm/yyyy')             DT_REALIZACAO
              ,0                                                                                  TOT_PEQ
              ,0                                                                                  TOT_MED
              ,COUNT(AVISO_CIRURGIA.CD_AVISO_CIRURGIA)  TOT_GRA
              ,0                                                                                  TOT_ESP
            ,  0                                                                                      TOT_ATE_PEQ
            ,  0                                                                                      TOT_ATE_MED
            ,  0                                                                                      TOT_ATE_GRA
            ,  0                                                                                      TOT_ATE_ESP
FROM DBAMV.AVISO_CIRURGIA  AVISO_CIRURGIA
           ,DBAMV.CIRURGIA_AVISO  CIRURGIA_AVISO
           ,DBAMV.CIRURGIA               CIRURGIA
           ,DBAMV.CONVENIO              CONVENIO
	   ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And AVISO_CIRURGIA.TP_SITUACAO = 'R'
AND   AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
AND   CIRURGIA_AVISO.CD_CIRURGIA = CIRURGIA.CD_CIRURGIA
AND   CIRURGIA.TP_CIRURGIA = 'G'
AND   TO_CHAR(AVISO_CIRURGIA.DT_REALIZACAO, 'MM/YYYY') = :CAMPO_DATA
AND    CIRURGIA_AVISO.CD_CONVENIO = CONVENIO.CD_CONVENIO

UNION ALL
SELECT to_date(to_char(AVISO_CIRURGIA.DT_REALIZACAO,'dd/mm/yyyy'),'dd/mm/yyyy')             DT_REALIZACAO
            ,  0                                                                                  TOT_PEQ
            ,  0                                                                                  TOT_MED
            ,  0                                                                                  TOT_GRA
            ,  0                                                                                  TOT_ESP
            ,  0                                                                                                      TOT_ATE_PEQ
            ,  0                                                                                                      TOT_ATE_MED
            ,  COUNT(DISTINCT AVISO_CIRURGIA.CD_AVISO_CIRURGIA)    TOT_ATE_GRA
            ,  0                                                                                                      TOT_ATE_ESP
FROM DBAMV.AVISO_CIRURGIA  AVISO_CIRURGIA
           ,DBAMV.CIRURGIA_AVISO  CIRURGIA_AVISO
           ,DBAMV.CIRURGIA               CIRURGIA
           ,DBAMV.CONVENIO              CONVENIO
	   ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And AVISO_CIRURGIA.TP_SITUACAO = 'R'
AND   AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
AND   CIRURGIA_AVISO.CD_CIRURGIA = CIRURGIA.CD_CIRURGIA
AND   CIRURGIA.TP_CIRURGIA = 'G'
AND   TO_CHAR(AVISO_CIRURGIA.DT_REALIZACAO, 'MM/YYYY') = :CAMPO_DATA
AND    CIRURGIA_AVISO.CD_CONVENIO = CONVENIO.CD_CONVENIO
GROUP BY DT_REALIZACAO
UNION ALL
SELECT to_date(to_char(AVISO_CIRURGIA.DT_REALIZACAO,'dd/mm/yyyy'),'dd/mm/yyyy')             DT_REALIZACAO
            ,0                                                                                      TOT_PEQ
            ,0                                                                                      TOT_MED
            ,0                                                                                      TOT_GRA
              ,COUNT(AVISO_CIRURGIA.CD_AVISO_CIRURGIA)  TOT_ESP
            ,0                                                                                      TOT_ATE_PEQ
            ,0                                                                                      TOT_ATE_MED
            ,0                                                                                      TOT_ATE_GRA
            ,0                                                                                      TOT_ATE_ESP
FROM DBAMV.AVISO_CIRURGIA  AVISO_CIRURGIA
           ,DBAMV.CIRURGIA_AVISO  CIRURGIA_AVISO
           ,DBAMV.CIRURGIA               CIRURGIA
           ,DBAMV.CONVENIO              CONVENIO
	   ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And AVISO_CIRURGIA.TP_SITUACAO = 'R'
AND   AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
AND   CIRURGIA_AVISO.CD_CIRURGIA = CIRURGIA.CD_CIRURGIA
AND   CIRURGIA.TP_CIRURGIA = 'E'
AND   TO_CHAR(AVISO_CIRURGIA.DT_REALIZACAO, 'MM/YYYY') = :CAMPO_DATA
AND    CIRURGIA_AVISO.CD_CONVENIO = CONVENIO.CD_CONVENIO
GROUP BY DT_REALIZACAO
UNION ALL
SELECT to_date(to_char(AVISO_CIRURGIA.DT_REALIZACAO,'dd/mm/yyyy'),'dd/mm/yyyy')             DT_REALIZACAO
            ,  0                                                                                  TOT_PEQ
            ,  0                                                                                  TOT_MED
            ,  0                                                                                  TOT_GRA
            ,  0                                                                                  TOT_ESP
            ,  0                                                                                  TOT_ATE_PEQ
            ,  0                                                                                  TOT_ATE_MED
            ,  0                                                                                  TOT_ATE_GRA
            ,  COUNT(DISTINCT AVISO_CIRURGIA.CD_AVISO_CIRURGIA)    TOT_ATE_ESP
FROM DBAMV.AVISO_CIRURGIA  AVISO_CIRURGIA
           ,DBAMV.CIRURGIA_AVISO  CIRURGIA_AVISO
           ,DBAMV.CIRURGIA               CIRURGIA
           ,DBAMV.CONVENIO              CONVENIO
	   ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And AVISO_CIRURGIA.TP_SITUACAO = 'R'
AND   AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
AND   CIRURGIA_AVISO.CD_CIRURGIA = CIRURGIA.CD_CIRURGIA
AND   CIRURGIA.TP_CIRURGIA = 'E'
AND   TO_CHAR(AVISO_CIRURGIA.DT_REALIZACAO, 'MM/YYYY') = :CAMPO_DATA
AND    CIRURGIA_AVISO.CD_CONVENIO = CONVENIO.CD_CONVENIO
GROUP BY DT_REALIZACAO
ORDER BY DT_REALIZACAO
)
GROUP BY DT_REALIZACAO
ORDER BY DT_REALIZACAO
