--R_CIR_TAB
SELECT *
FROM  (
SELECT TRUNC(AVISO_CIRURGIA.DT_AVISO_CIRURGIA)   DATA
              ,COUNT(AVISO_CIRURGIA.CD_AVISO_CIRURGIA)  CONT_AVISO
              ,TO_NUMBER('0')                                                         CONT_MARCA
              ,TO_NUMBER('0')                                                         CONT_REALI
              ,TO_NUMBER('0')                                                         CONT_CANCE
              ,TO_NUMBER('0')                                                         CONT_TRANS
FROM     DBAMV.AVISO_CIRURGIA
              ,DBAMV.CIRURGIA_AVISO
WHERE  AVISO_CIRURGIA.TP_SITUACAO = 'A'
AND    TRUNC(AVISO_CIRURGIA.DT_AVISO_CIRURGIA) BETWEEN :CAMPO_DATA_INI AND LAST_DAY(:CAMPO_DATA_FIM)
AND    AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
AND        AVISO_CIRURGIA.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
GROUP BY TRUNC(AVISO_CIRURGIA.DT_AVISO_CIRURGIA)
UNION ALL
SELECT TRUNC(AGE_CIR.DT_INICIO_AGE_CIR)                  DATA
              ,TO_NUMBER('0')                                                        CONT_AVISO
              ,COUNT(AVISO_CIRURGIA.CD_AVISO_CIRURGIA) CONT_MARCA
              ,TO_NUMBER('0')                                                        CONT_REALI
              ,TO_NUMBER('0')                                                        CONT_CANCE
              ,TO_NUMBER('0')                                                        CONT_TRANS
FROM    DBAMV.AVISO_CIRURGIA
              ,DBAMV.AGE_CIR
              ,DBAMV.CIRURGIA_AVISO
WHERE  AVISO_CIRURGIA.CD_AVISO_CIRURGIA = AGE_CIR.CD_AVISO_CIRURGIA
AND    AVISO_CIRURGIA.TP_SITUACAO = 'G'
AND    TRUNC(AGE_CIR.DT_INICIO_AGE_CIR) BETWEEN :CAMPO_DATA_INI AND LAST_DAY(:CAMPO_DATA_FIM)
AND    AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
AND        AVISO_CIRURGIA.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
GROUP BY TRUNC(AGE_CIR.DT_INICIO_AGE_CIR)
UNION ALL
SELECT TRUNC ( AVISO_CIRURGIA.DT_REALIZACAO)        DATA
              ,TO_NUMBER('0')                                                         CONT_AVISO
              ,TO_NUMBER('0')                                                         CONT_MARCA
              ,COUNT(AVISO_CIRURGIA.CD_AVISO_CIRURGIA)  CONT_REALI
              ,TO_NUMBER('0')                                                         CONT_CANCE
              ,TO_NUMBER('0')                                                         CONT_TRANS
FROM     DBAMV.AVISO_CIRURGIA
        ,dbamv.CIRURGIA_AVISO
WHERE   AVISO_CIRURGIA.TP_SITUACAO = 'R'
AND     TRUNC(AVISO_CIRURGIA.DT_REALIZACAO) BETWEEN :CAMPO_DATA_INI AND LAST_DAY(:CAMPO_DATA_FIM)
AND     AVISO_CIRURGIA.CD_AVISO_CIRURGIA   =  CIRURGIA_AVISO.CD_AVISO_CIRURGIA
AND        AVISO_CIRURGIA.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
GROUP BY TRUNC(AVISO_CIRURGIA.DT_REALIZACAO)
UNION ALL
SELECT TRUNC ( AVISO_CIRURGIA.DT_CANCELAMENTO)   DATA
              ,TO_NUMBER('0')                                                          CONT_AVISO
              ,TO_NUMBER('0')                                                          CONT_MARCA
              ,TO_NUMBER('0')                                                          CONT_REALI
              ,COUNT(AVISO_CIRURGIA.CD_AVISO_CIRURGIA)   CONT_CANCE
              ,TO_NUMBER('0')                                                          CONT_TRANS
FROM     DBAMV.AVISO_CIRURGIA
              ,DBAMV.CIRURGIA_AVISO
WHERE  AVISO_CIRURGIA.TP_SITUACAO = 'C'
AND    TRUNC( AVISO_CIRURGIA.DT_CANCELAMENTO) BETWEEN :CAMPO_DATA_INI AND LAST_DAY(:CAMPO_DATA_FIM)
AND    AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
AND        AVISO_CIRURGIA.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
GROUP BY TRUNC(AVISO_CIRURGIA.DT_CANCELAMENTO)
UNION ALL
SELECT TRUNC( TRANSFERENCIA_CIRURGIA.DT_TRANSFERENCIA)     DATA
              ,TO_NUMBER('0')                                                                            CONT_AVISO
              ,TO_NUMBER('0')                                                                            CONT_MARCA
              ,TO_NUMBER('0')                                                                            CONT_REALI
              ,TO_NUMBER('0')                                                                            CONT_CANCE
              ,COUNT(TRANSFERENCIA_CIRURGIA.CD_AVISO_CIRURGIA)  CONT_TRANS
FROM     DBAMV.AVISO_CIRURGIA
              ,DBAMV.TRANSFERENCIA_CIRURGIA
              ,DBAMV.CIRURGIA_AVISO
WHERE   AVISO_CIRURGIA.CD_AVISO_CIRURGIA = TRANSFERENCIA_CIRURGIA.CD_AVISO_CIRURGIA
AND     TRUNC(TRANSFERENCIA_CIRURGIA.DT_TRANSFERENCIA) BETWEEN :CAMPO_DATA_INI AND LAST_DAY(:CAMPO_DATA_FIM)
AND     AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
AND        AVISO_CIRURGIA.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
GROUP BY TRUNC(TRANSFERENCIA_CIRURGIA.DT_TRANSFERENCIA)
)
ORDER BY DATA
