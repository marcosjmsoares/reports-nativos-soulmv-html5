--R_COMP_AVI_CIR_900
SELECT * FROM (
SELECT AVISO_CIRURGIA.CD_AVISO_CIRURGIA                                             CD_AVISO_CIRURGIA,
                AVISO_CIRURGIA.DT_AVISO_CIRURGIA                                            DT_AVISO_CIRURGIA,
               AVISO_CIRURGIA.CD_PACIENTE                                                         CD_PACIENTE,
               AVISO_CIRURGIA.NM_PACIENTE                                                        NM_PACIENTE,
               AVISO_CIRURGIA.CD_ATENDIMENTO                                                CD_ATENDIMENTO,
               CIRURGIA_AVISO.TP_NATUREZA                                                       TP_NATUREZA,
               AVISO_CIRURGIA.CD_USUARIO                                                          CD_USUARIO,
               CIRURGIA_AVISO.DS_OBSERVACAO                                                  DS_OBSERVACAO,
               DECODE ( CONVENIO.TP_CONVENIO,'A',TAB_SIASUS.CD_SSM,
                                                                               'H',PRO_FAT_SUS.CD_SUS,
                                                                               'C',PRO_FAT.CD_PRO_FAT,
                                                                               'P',PRO_FAT.CD_PRO_FAT,
                                                                               ' ' )                                              CD_PRO_FAT,
               DECODE ( CONVENIO.TP_CONVENIO,'A',TAB_SIASUS.DS_TITULO,
                                                                               'H',PRO_FAT_SUS.DS_SUS,
                                                                               'C',PRO_FAT.DS_PRO_FAT,
                                                                               'P',PRO_FAT.DS_PRO_FAT,
                                                                               CIRURGIA.DS_CIRURGIA )     DS_PRO_FAT,
              SAL_CIR.CD_SAL_CIR                                                                             CD_SAL_CIR,
              SAL_CIR.DS_SAL_CIR                                                                             DS_SAL_CIR,
              TO_CHAR(AGE_CIR.DT_INICIO_AGE_CIR,'HH24')                           HORA_PREV_INI,
              TO_CHAR(AGE_CIR.DT_FINAL_AGE_CIR,'HH24')                            HORA_PREV_FIM,
              DECODE(AVISO_CIRURGIA.NR_TELEFONE_CONTATO,'',PACIENTE.NR_FONE,AVISO_CIRURGIA.NR_TELEFONE_CONTATO)  NR_FONE,
              CONVENIO.NM_CONVENIO                                                                    NM_CONVENIO,
              CIRURGIA.CD_CIRURGIA                                                                       CD_CIRURGIA,
              CIRURGIA.DS_CIRURGIA                                                                        DS_CIRURGIA,
              AGE_CIR.DT_INICIO_AGE_CIR                                                               DT_INICIO_AGE_CIR,
             decode(PACIENTE.DT_NASCIMENTO,'',aviso_cirurgia.ds_idade|| ' anos', RTRIM(PACK_INTERNAMENTO.RETORNA_IDADE(PACIENTE.DT_NASCIMENTO,SysDate),' ')  )NR_IDADE,
              TIP_ANEST.DS_TIP_ANEST					DS_TIP_ANEST,
              AVISO_CIRURGIA.TP_SANGUINEO				TP_SANGUINEO,
              DECODE ( CONVENIO.TP_CONVENIO,'A',TAB_SIASUS.CD_SSM,
                                                                               'H',PRO_FAT_SUS.CD_SUS,
                                                                               'C',PRO_FAT.CD_PRO_FAT,
                                                                               'P',PRO_FAT.CD_PRO_FAT,
                                                                               ' ' )                                              CD_PRO_FAT2,
               DECODE ( CONVENIO.TP_CONVENIO,'A',TAB_SIASUS.DS_TITULO,
                                                                               'H',PRO_FAT_SUS.DS_SUS,
                                                                               'C',PRO_FAT.DS_PRO_FAT,
                                                                               'P',PRO_FAT.DS_PRO_FAT,
                                                                               CIRURGIA.DS_CIRURGIA )     DS_PRO_FAT2,
              CIRURGIA.CD_CIRURGIA                                                                       CD_CIRURGIA2,
              decode(aviso_cirurgia.sn_uti,'S','Sim','Não') sn_uti,
              decode(aviso_cirurgia.sn_exame,'S','Sim','Não') sn_exame,
              aviso_cirurgia.ds_a_seguir ds_a_seguir,
              tip_acom.ds_tip_acom ds_tip_acom,
              cen_cir.ds_cen_cir   ds_cen_cir,
              CIRURGIA_AVISO.DS_NPADRONIZADO,
              AVISO_CIRURGIA.DS_CIRURGIA DS_CIRURGIA1
FROM    DBAMV.AVISO_CIRURGIA                                                AVISO_CIRURGIA,
              DBAMV.ATENDIME                                                           ATENDIME,
              DBAMV.CONVENIO                                                           CONVENIO,
              DBAMV.CIRURGIA_AVISO                                                CIRURGIA_AVISO,
              DBAMV.CIRURGIA                                                             CIRURGIA,
              DBAMV.SAL_CIR                                                                SAL_CIR,
              DBAMV.AGE_CIR                                                               AGE_CIR,
              DBAMV.PRO_FAT                                                              PRO_FAT,
              DBAMV.TAB_SIASUS			          TAB_SIASUS,
              DBAMV.PRO_FAT				          PRO_FAT_SUS,
              DBAMV.PACIENTE                                                             PACIENTE,
              DBAMV.TIP_ANEST				          TIP_ANEST,
              DBAMV.TIP_ACOM ,
              DBAMV.CEN_CIR                                                               CEN_CIR
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And AVISO_CIRURGIA.CD_AVISO_CIRURGIA       =     CIRURGIA_AVISO.CD_AVISO_CIRURGIA
AND        CIRURGIA.CD_CIRURGIA                                =      CIRURGIA_AVISO.CD_CIRURGIA
AND        AVISO_CIRURGIA.CD_ATENDIMENTO          =      ATENDIME.CD_ATENDIMENTO(+)
AND        CIRURGIA_AVISO.CD_CONVENIO                  =      CONVENIO.CD_CONVENIO(+)
AND        AVISO_CIRURGIA.CD_SAL_CIR                      =      SAL_CIR.CD_SAL_CIR(+)
AND        AVISO_CIRURGIA.CD_AVISO_CIRURGIA       =      AGE_CIR.CD_AVISO_CIRURGIA(+)
AND        CIRURGIA.CD_PRO_FAT                                 =      PRO_FAT.CD_PRO_FAT(+)
AND        CIRURGIA.CD_SSM		          =      TAB_SIASUS.CD_SSM(+)
AND        CIRURGIA.CD_PRO_FAT_SUS                       =       PRO_FAT_SUS.CD_PRO_FAT(+)
AND        AVISO_CIRURGIA.CD_PACIENTE                   =      PACIENTE.CD_PACIENTE(+)
AND        AVISO_CIRURGIA.CD_TIP_ANEST                 =      TIP_ANEST.CD_TIP_ANEST(+)
AND        CIRURGIA_AVISO. CD_TIP_ACOM                 =       TIP_ACOM.CD_TIP_ACOM(+)
AND        AVISO_CIRURGIA.CD_CEN_CIR                     =       CEN_CIR.CD_CEN_CIR(+)
ORDER BY AVISO_CIRURGIA.CD_AVISO_CIRURGIA,  PRO_FAT.DS_PRO_FAT
)
ORDER BY CD_AVISO_CIRURGIA,CD_SAL_CIR,DS_SAL_CIR,CD_PACIENTE,NM_PACIENTE,DT_AVISO_CIRURGIA,HORA_PREV_FIM,HORA_PREV_INI,TP_NATUREZA,DS_CIRURGIA,DT_INICIO_AGE_CIR,CD_ATENDIMENTO,CD_USUARIO,NR_FONE,DS_TIP_ANEST,NR_IDADE,TP_SANGUINEO,SN_UTI,SN_EXAME,DS_TIP_ACOM,DS_CEN_CIR,DS_A_SEGUIR,DS_NPADRONIZADO,
CD_CIRURGIA,CD_PRO_FAT,DS_PRO_FAT
