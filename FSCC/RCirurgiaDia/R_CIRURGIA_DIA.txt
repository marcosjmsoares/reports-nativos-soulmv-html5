--R_CIRURGIA_DIA
SELECT * FROM (
SELECT  AVISO_CIRURGIA.CD_AVISO_CIRURGIA                                 CD_AVISO_CIRURGIA
               ,CEN_CIR.DS_CEN_CIR                                                               DS_CEN_CIR
               ,CEN_CIR.CD_CEN_CIR                                                               CD_CEN_CIR
              ,AVISO_CIRURGIA.NM_PACIENTE                                              NM_PACIENTE
              ,TO_CHAR( AGE_CIR.DT_INICIO_AGE_CIR, 'HH24')              HR_INICIO_AGE_CIR
              ,TO_CHAR( AGE_CIR.DT_INICIO_AGE_CIR, 'DD\MM\YYYY')    DT_INICIO_AGE_CIR
              ,TIP_ANEST.DS_TIP_ANEST                                                        DS_TIP_ANEST
              ,DECODE(AVISO_CIRURGIA.TP_SITUACAO, 'G',  'AGENDADA'
                                                                                       , 'R',  'REALIZADA'
                                                                                       , 'C',  'CANCELADA')  TP_SITUACAO
FROM  DBAMV.PACIENTE
           ,DBAMV.AVISO_CIRURGIA
           ,DBAMV.AGE_CIR
           ,DBAMV.CEN_CIR
           ,DBAMV.TIP_ANEST
WHERE AVISO_CIRURGIA.CD_PACIENTE  =  PACIENTE.CD_PACIENTE(+)
AND       AVISO_CIRURGIA.CD_AVISO_CIRURGIA  =  AGE_CIR.CD_AVISO_CIRURGIA(+)
AND      CEN_CIR.CD_CEN_CIR(+)  =  AVISO_CIRURGIA.CD_CEN_CIR
AND      TRUNC(AGE_CIR.DT_INICIO_AGE_CIR) BETWEEN TRUNC(:CAMPO_DT_INICIO)  AND TRUNC(:CAMPO_DT_FIM)
AND      AVISO_CIRURGIA.CD_TIP_ANEST  =  TIP_ANEST.CD_TIP_ANEST(+)
ORDER BY    DT_INICIO_AGE_CIR ,HR_INICIO_AGE_CIR
)
ORDER BY DS_CEN_CIR,CD_CEN_CIR,
DT_INICIO_AGE_CIR
