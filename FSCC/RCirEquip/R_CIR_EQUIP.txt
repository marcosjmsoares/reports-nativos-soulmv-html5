--R_CIR_EQUIP
SELECT * FROM (
SELECT   AVISO_CIRURGIA.CD_AVISO_CIRURGIA                     CD_AVISO_CIRURGIA,
                 lPad(To_Char(AVISO_CIRURGIA.CD_PACIENTE),8,'0')         CD_PACIENTE,
                 AVISO_CIRURGIA.NM_PACIENTE                                 NM_PACIENTE,
                 AVISO_CIRURGIA.DT_REALIZACAO                             DT_REALIZACAO,
                 LPad(To_Char(EQUIPE_PRESTADOR.CD_PRESTADOR),6,'0')      CD_PRESTADOR,
                 PRESTADOR.NM_PRESTADOR                                    NM_PRESTADOR,
                 LPad(To_Char(EQUIPE_MEDICA.CD_EQUIPE_MEDICA),3,'0')      CD_EQUIPE_MEDICA,
                 EQUIPE_MEDICA.DS_EQUIPE_MEDICA                       DS_EQUIPE_MEDICA,
                 CIRURGIA.CD_CIRURGIA                                              CD_CIRURGIA,
                 CIRURGIA.DS_CIRURGIA                                             DS_CIRURGIA
FROM      DBAMV.AVISO_CIRURGIA                                              AVISO_CIRURGIA,
                DBAMV.EQUIPE_MEDICA                                               EQUIPE_MEDICA,
                DBAMV.EQUIPE_PRESTADOR                                      EQUIPE_PRESTADOR,
                DBAMV.ATI_MED                                                            ATI_MED,
                DBAMV.PRESTADOR                                                     PRESTADOR,
                DBAMV.CIRURGIA_AVISO                                             CIRURGIA_AVISO,
                DBAMV.CIRURGIA                                                          CIRURGIA
 WHERE
   AVISO_CIRURGIA.TP_SITUACAO                  =   'R'
AND    AVISO_CIRURGIA.CD_EQUIPE_MEDICA       =   EQUIPE_MEDICA.CD_EQUIPE_MEDICA
AND    EQUIPE_MEDICA.CD_EQUIPE_MEDICA        =   EQUIPE_PRESTADOR.CD_EQUIPE_MEDICA
AND    EQUIPE_PRESTADOR.CD_ATI_MED             =   ATI_MED.CD_ATI_MED
AND    ATI_MED.TP_FUNCAO                                    =   'C'
AND    EQUIPE_PRESTADOR.CD_PRESTADOR      =   PRESTADOR.CD_PRESTADOR
AND    AVISO_CIRURGIA.CD_AVISO_CIRURGIA      =   CIRURGIA_AVISO.CD_AVISO_CIRURGIA
AND    CIRURGIA.CD_CIRURGIA                               =    CIRURGIA_AVISO.CD_CIRURGIA
AND    TRUNC( AVISO_CIRURGIA.DT_REALIZACAO ) BETWEEN :CAMPO_DATA_INI ORDER BY CD_EQUIPE_MEDICA, CD_PRESTADOR, DT_REALIZACAO
)
ORDER BY CD_EQUIPE_MEDICA,DS_EQUIPE_MEDICA,
CD_PRESTADOR,NM_PRESTADOR, DT_REALIZACAO
