--R_FICHA_ADMISSAO_CO_PREEN_R_FICHA1_Q_RECEM_NASCIDO
SELECT

R.NM_RECEM_NASCIDO,
R.CD_RECEM_NASCIDO,
R.VL_PESO,
R.VL_ALTURA,
R.VL_APGAR_1MINUTO,
R.VL_APGAR_5MINUTO,
R.VL_PERIMETRO_CEFALICO,
R.VL_PERIMETRO_ABDOMINAL,
R.VL_PERIMETRO_TORAXICO,
R.DS_EXAME_SUMARIO,
R.DS_DIAGNOSTICO_INICIAL,
R.DS_CONDUTA,
DECODE(NVL(R.TP_SEXO,'*'),'F','Feminino','M','Masculino','Indefinido') SEXO,
R.NR_PULSEIRA,
R.HR_NASCIMENTO,
ATE.CD_PACIENTE    CD_PACIENTE_NASC,
R.CD_DECLARACAO_NASCIDO_VIVO,
A.CD_ADMISSAO_CO,
A.DS_PARTO,
A.CD_MOTIVO_PARTO,
TO_CHAR(A.DT_PARTO, 'HH24') AS HR_DT_PARTO,
A.DT_PARTO,
A.QT_GESTACAO,
A.QT_PARIDADE,
A.QT_ABORTO,
A.QT_CESARIA,
A.QT_NATIMORTOS,
A.QT_PREMATUROS,
A.QT_FILHOS_VIVOS,
A.QT_FILHOS_MORTOS,
DECODE(A.TP_VDRL,'N','Não Reagente','R','Reagente',' ') AS TP_VDRL,
DECODE(A.TP_COMBS_DIRETO,'P','Positivo','N','Negativo',' ') AS TP_COMBS_DIRETO,
A.VL_COMBS_DIRETO,
A.NR_CONSULTAS,
A.DT_ULTIMA_MENSTRUACAO,
A.DT_PREV_PARTO,
A.NM_PAI,
A.VL_TEMPO_GESTACAO,
A.DT_INICIO_MOV_FETAL,
A.DS_MEDICAMENTOS,
A.VL_BCF,
TIP_ANEST.DS_Tip_Anest,
A.QT_RN_PESO_MAIOR_4000G,
A.QT_RN_PESO_MENOR_2500G,
A.QT_OBITOS_PRIMEIRA_SEMANA,
A.QT_OBITOS_APOS_PRIMEIRA_SEMANA,
A.DS_INTERCORRENCIA_COMPLICACAO,
A.DS_OBS_GESTACAO_ATUAL,
A.VL_INTERVALO_INTERPARTAL,
A.VL_VDRL  VDRL,
tp.DS_TIP_PARTO,
TP.DS_TIP_PARTO,
P.NM_PACIENTE,
P.TP_SANGUINEO,
P.DT_NASCIMENTO,
RN.CD_LEITO,
L.DS_RESUMO,
RN.CD_PACIENTE r_pac,
MP.DS_MOTIVO_PARTO,
PRE.NM_PRESTADOR
 FROM DBAMV.RECEM_NASCIDO R,
      DBAMV.ADMISSAO_CO A,
      dbamv.TIP_ANEST,
      DBAMV.PACIENTE P,
      DBAMV.TIP_PARTO TP,
      DBAMV.ATENDIME AT,
      DBAMV.ATENDIME RN,
      DBAMV.MOTIVO_PARTO MP,
      DBAMV.ATENDIME ATE,
      DBAMV.LEITO L,
      DBAMV.PRESTADOR PRE
WHERE MP.CD_MOTIVO_PARTO(+)     = A.CD_MOTIVO_PARTO
  AND RN.CD_ATENDIMENTO(+)      = R.CD_ATENDIMENTO
  AND R.CD_ADMISSAO_CO(+)       = A.CD_ADMISSAO_CO
  AND TIP_ANEST.CD_TIP_ANEST(+) = A.CD_TIP_ANEST
  AND AT.CD_ATENDIMENTO    = A.CD_ATENDIMENTO
  AND AT.CD_PACIENTE       = P.CD_PACIENTE
  AND A.CD_TIP_PARTO       = TP.CD_TIP_PARTO(+)
  AND R.CD_ATENDIMENTO     = ATE.CD_ATENDIMENTO(+)
  AND AT.CD_ATENDIMENTO    = :CAMPO_CD_ATENDIMENTO
  AND RN.CD_LEITO          = L.CD_LEITO(+)
  AND PRE.CD_PRESTADOR(+)     = R.CD_PRESTADOR 
  AND A.CD_ADMISSAO_CO     = :CAMPO_CD_ADMISSAO_CO
  AND  AT.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
