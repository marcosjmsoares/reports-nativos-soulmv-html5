--R_CIR_REALIZ_ANA_EVOL_ANUAL
SELECT * FROM (
select mes,ds_cirurgia,sum(qtd) qtd from
(select        cirur.ds_cirurgia                     ds_cirurgia
             ,TO_CHAR(AVISO_CIRURGIA.DT_REALIZACAO,'MM') MES
             ,COUNT(AVISO_CIRURGIA.DT_REALIZACAO)        QTD
from       dbamv.aviso_cirurgia   aviso_cirurgia
          ,dbamv.cirurgia_aviso   cir_avi
          ,dbamv.cirurgia        cirur
where      aviso_cirurgia.tp_situacao    = 'R'
and        cir_avi.cd_aviso_cirurgia = aviso_cirurgia.cd_aviso_cirurgia
and        cirur.cd_cirurgia = cir_avi.cd_cirurgia
and        to_char(aviso_cirurgia.dt_realizacao,'YYYY') = SUBSTR(:CAMPO_DATA_INI,4,4)
group by   TO_CHAR(AVISO_CIRURGIA.DT_REALIZACAO,'MM') ,cirur.ds_cirurgia
union
select cirurgia.ds_cirurgia,mes.mes,0 as qtd
from
(select distinct     TO_CHAR(AVISO_CIRURGIA.DT_REALIZACAO,'MM') MES
from       dbamv.aviso_cirurgia   aviso_cirurgia
          ,dbamv.cirurgia_aviso   cir_avi
          ,dbamv.cirurgia         cirur
where      aviso_cirurgia.tp_situacao    = 'R'
and        cir_avi.cd_aviso_cirurgia = aviso_cirurgia.cd_aviso_cirurgia
and        cirur.cd_cirurgia = cir_avi.cd_cirurgia
and        to_char(aviso_cirurgia.dt_realizacao,'YYYY') = SUBSTR(:CAMPO_DATA_INI,4,4)
:CAMPO_CF_COND_PREST1) mes,
(select distinct     cirur.ds_cirurgia ds_cirurgia
from       dbamv.aviso_cirurgia   aviso_cirurgia
          ,dbamv.cirurgia_aviso   cir_avi
          ,dbamv.cirurgia         cirur
where      aviso_cirurgia.tp_situacao    = 'R'
and        cir_avi.cd_aviso_cirurgia = aviso_cirurgia.cd_aviso_cirurgia
and        cirur.cd_cirurgia = cir_avi.cd_cirurgia
and        to_char(aviso_cirurgia.dt_realizacao,'YYYY') = SUBSTR(:CAMPO_DATA_INI,4,4)
:CAMPO_CF_COND_PREST1) cirurgia)
group by   mes ,ds_cirurgia
)
ORDER BY DS_CIRURGIA,
MES
