--R_EST_TIP_ANEST_ANUAL
SELECT * FROM (
select mes,ds_tip_anest,sum(qtd) qtd from
(select        tip_anest.ds_tip_anest                     ds_tip_anest
             ,TO_CHAR(AVISO_CIRURGIA.DT_REALIZACAO,'MM') MES
             ,COUNT(AVISO_CIRURGIA.DT_REALIZACAO)        QTD
from       dbamv.aviso_cirurgia   aviso_cirurgia
          ,dbamv.cirurgia_aviso   cir_avi
          ,dbamv.tip_anest        tip_anest
where      aviso_cirurgia.tp_situacao    = 'R'
and        cir_avi.cd_aviso_cirurgia = aviso_cirurgia.cd_aviso_cirurgia
and        tip_anest.cd_tip_anest = aviso_cirurgia.cd_tip_anest
and        to_char(aviso_cirurgia.dt_realizacao,'YYYY') = SUBSTR(:CAMPO_DATA_INI,4,4)
group by   TO_CHAR(AVISO_CIRURGIA.DT_REALIZACAO,'MM') ,tip_anest.ds_tip_anest
union
select tip_anest.ds_tip_anest,mes.mes,0 as qtd
from
(select distinct     TO_CHAR(AVISO_CIRURGIA.DT_REALIZACAO,'MM') MES
from       dbamv.aviso_cirurgia   aviso_cirurgia
          ,dbamv.cirurgia_aviso   cir_avi
          ,dbamv.tip_anest        tip_anest
where      aviso_cirurgia.tp_situacao    = 'R'
and        cir_avi.cd_aviso_cirurgia = aviso_cirurgia.cd_aviso_cirurgia
and        tip_anest.cd_tip_anest = aviso_cirurgia.cd_tip_anest
and        to_char(aviso_cirurgia.dt_realizacao,'YYYY') = SUBSTR(:CAMPO_DATA_INI,4,4)
:CAMPO_CF_CD_TIP_ANEST) mes,
(select distinct     tip_anest.ds_tip_anest ds_tip_anest
from       dbamv.aviso_cirurgia   aviso_cirurgia
          ,dbamv.cirurgia_aviso   cir_avi
          ,dbamv.tip_anest        tip_anest
where      aviso_cirurgia.tp_situacao    = 'R'
and        cir_avi.cd_aviso_cirurgia = aviso_cirurgia.cd_aviso_cirurgia
and        tip_anest.cd_tip_anest = aviso_cirurgia.cd_tip_anest
and        to_char(aviso_cirurgia.dt_realizacao,'YYYY') = SUBSTR(:CAMPO_DATA_INI,4,4)
:CAMPO_CF_CD_TIP_ANEST) tip_anest)
group by   mes ,ds_tip_anest
)
ORDER BY DS_TIP_ANEST,
MES
