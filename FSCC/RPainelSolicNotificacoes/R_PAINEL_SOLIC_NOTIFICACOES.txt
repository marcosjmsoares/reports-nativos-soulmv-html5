--R_PAINEL_SOLIC_NOTIFICACOES
SELECT * FROM (
		SELECT  CD_AVISO AVISO,
        DECODE(SN_NOTIFICACAO_VISUALIZADA, 'S', 'Sim', 'N', 'Não') CONFIRMADA,
	    (TP_SOLICITADO || ';' || TP_OPERACAO || ';' || DS_PRODUTO_EXAME || ';' || CD_AVISO || ';' || TO_CHAR(DT_REALIZACAO, 'dd/mm/yyyy') || ';' || TO_CHAR(DT_ANTERIOR, 'dd/mm/yyyy') || ';'|| QT_ANTERIOR ||';'|| QT_REALIZACAO) DADOS_MENSAGEM
FROM dbamv.notificacao_centro_cirurgico
WHERE CD_SETOR = :CAMPO_CD_SETOR
  AND to_char(DT_REALIZACAO, 'dd/mm/yyyy') = :CAMPO_DT_REALIZACAO
  AND TP_SETOR = 'P'
  AND 'TP_PRODUTO' = :CAMPO_QUERY
UNION ALL
SELECT  CD_AVISO AVISO,
        DECODE(SN_NOTIFICACAO_VISUALIZADA, 'S', 'Sim', 'N', 'Não') CONFIRMADA,
        (TP_SOLICITADO || ';' || TP_OPERACAO || ';' || DS_PRODUTO_EXAME || ';' || CD_AVISO || ';' || TO_CHAR(DT_REALIZACAO, 'dd/mm/yyyy') || ';' || TO_CHAR(DT_ANTERIOR, 'dd/mm/yyyy') || ';'|| QT_ANTERIOR ||';'|| QT_REALIZACAO) DADOS_MENSAGEM
FROM dbamv.notificacao_centro_cirurgico
WHERE CD_SETOR = :CAMPO_CD_SETOR
  AND to_char(DT_REALIZACAO, 'dd/mm/yyyy') = :CAMPO_DT_REALIZACAO
  AND TP_SETOR IN ('R', 'S')
  AND 'TP_EXAME' = :CAMPO_QUERY
UNION ALL
SELECT  CD_AVISO AVISO,
        DECODE(SN_NOTIFICACAO_VISUALIZADA, 'S', 'Sim', 'N', 'Não') CONFIRMADA,
	    (TP_SOLICITADO || ';' || TP_OPERACAO || ';' || DS_PRODUTO_EXAME || ';' || CD_AVISO || ';' || TO_CHAR(DT_REALIZACAO, 'dd/mm/yyyy') || ';' || TO_CHAR(DT_ANTERIOR, 'dd/mm/yyyy') || ';'|| QT_ANTERIOR ||';'|| QT_REALIZACAO) DADOS_MENSAGEM
FROM dbamv.notificacao_centro_cirurgico
WHERE CD_SETOR = :CAMPO_CD_SETOR
  AND to_char(DT_REALIZACAO, 'dd/mm/yyyy') = :CAMPO_DT_REALIZACAO
  AND TP_SETOR = 'B'
  AND 'TP_SANGUE' = :CAMPO_QUERY
  :CAMPO_SN_CONFIRMADO_QUERY)
