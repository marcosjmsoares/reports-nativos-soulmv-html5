--R_TEXPAD
SELECT  CTP.CD_CIRURGIA_TEXTO_PADRAO              Codigo,
                CTP.DS_CIRURGIA_TEXTO_PADRAO              Descricao,
                CTP.DS_TEXTO                                                  DS_TEXTO,
                CTP.CD_PRESTADOR                                        CD_PRESTADOR,
		CTP.TP_CAMPOS					TP_CAMPOS,
                PRESTADOR.NM_PRESTADOR                        NM_PRESTADOR,
                PRESTADOR.CD_CONSELHO                           CD_CONSELHO,
                PRESTADOR.DS_CODIGO_CONSELHO           NR_CONSELHO,
                CONSELHO.DS_CONSELHO                              DS_CONSELHO
FROM      DBAMV.CIRURGIA_TEXTO_PADRAO               CTP,
                DBAMV.PRESTADOR                                         PRESTADOR,
                DBAMV.CONSELHO                                            CONSELHO
WHERE  CTP.CD_PRESTADOR                      =    PRESTADOR.CD_PRESTADOR (+)
AND        PRESTADOR.CD_CONSELHO         =    CONSELHO.CD_CONSELHO (+)
AND	CTP.TP_CAMPOS = :CAMPO_TP_CAMPOS
ORDER BY
