--R_PAC_ATEND_CIR
SELECT * FROM (
select unique atendime.cd_atendimento
                    ,atendime.dt_atendimento
                    ,prestador.cd_prestador
                    ,prestador.nm_prestador
                    ,aviso_cirurgia.cd_paciente
                    ,aviso_cirurgia.nm_paciente
                    ,cen_cir.cd_cen_cir
                    ,cen_cir.ds_cen_cir
from  dbamv.atendime
       ,dbamv.prestador
       ,dbamv.aviso_cirurgia
       ,dbamv.cen_cir
where atendime.cd_atendimento    = aviso_cirurgia.cd_atendimento
And  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and    atendime.cd_prestador      = prestador.cd_prestador
and    nvl(prestador.sn_cirurgiao,'N') <> 'S'
and    nvl(prestador.sn_anestesista,'N') <> 'S'
and    aviso_cirurgia.cd_cen_cir  = cen_cir.cd_cen_cir
and    aviso_cirurgia.tp_situacao = 'R'
and    trunc(atendime.dt_atendimento) between  :CAMPO_DATA_INI  order by cd_prestador, dt_atendimento
)
ORDER BY DS_CEN_CIR,CD_CEN_CIR,
CD_PRESTADOR,NM_PRESTADOR,
DT_ATENDIMENTO
