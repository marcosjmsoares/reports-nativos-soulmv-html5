--R_CIR_REA_SIN
SELECT * FROM (
SELECT COUNT(*) VALOR
,TO_CHAR(DT_REALIZACAO,'MM') MES
,TO_CHAR(DT_REALIZACAO,'YYYY') ANO
FROM DBAMV.AVISO_CIRURGIA,
           DBAMV.CIRURGIA_AVISO
WHERE TP_SITUACAO = 'R'
     AND CIRURGIA_AVISO.CD_AVISO_CIRURGIA = AVISO_CIRURGIA.CD_AVISO_CIRURGIA
AND AVISO_CIRURGIA.cd_multi_empresa = Pkg_mv2000.Le_empresa
GROUP BY
TO_CHAR(DT_REALIZACAO,'YYYY')
,TO_CHAR(DT_REALIZACAO,'MM')
)
ORDER BY MES,
ANO
