--R_FICHA_PARA_CIR_R_1_Q_AVISO_CIRURGIA
SELECT * FROM
  (SELECT AVISO_CIRURGIA.CD_AVISO_CIRURGIA CD_AVISO_CIRURGIA,
          AVISO_CIRURGIA.CD_PACIENTE CD_PACIENTE,
          CASE WHEN dbamv.aviso_cirurgia.cd_paciente IS NULL THEN dbamv.aviso_cirurgia.nm_paciente
	           WHEN dbamv.aviso_cirurgia.cd_paciente IS NOT NULL THEN (SELECT nm_paciente FROM dbamv.paciente WHERE cd_paciente = dbamv.aviso_cirurgia.cd_paciente)
	      END AS nm_paciente,
      (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND AVISO_CIRURGIA.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE,
          DECODE(AGE_CIR.DT_INICIO_AGE_CIR, '', AVISO_CIRURGIA.DT_AVISO_CIRURGIA, AGE_CIR.DT_INICIO_AGE_CIR) DT_AVISO_CIRURGIA,
          CIRURGIA_AVISO.TP_NATUREZA TP_NATUREZA,
          AVISO_CIRURGIA.DS_CIRURGIA DS_CIRURGIA,
          AVISO_CIRURGIA.DS_RISCO_CIRURGICO DS_RISCO_CIRURGICO,
          AVISO_CIRURGIA.DS_ACIDENTES_DURANTE_A_OPERACA DS_ACIDENTES_DURANTE_A_OPERACA,
          DECODE (CONVENIO.TP_CONVENIO,
                  'A', PROCEDIMENTO_SUS_A.CD_PROCEDIMENTO,
                  'H', PROCEDIMENTO_SUS_H.CD_PROCEDIMENTO,
                  'C', PRO_FAT.CD_PRO_FAT,
                  'P', PRO_FAT.CD_PRO_FAT,
                  ' ') CD_PRO_FAT,
          DECODE (CONVENIO.TP_CONVENIO,
                  'A', PROCEDIMENTO_SUS_A.DS_PROCEDIMENTO,
                  'H', PROCEDIMENTO_SUS_H.DS_PROCEDIMENTO,
                  'C', PRO_FAT.DS_PRO_FAT,
                  'P', PRO_FAT.DS_PRO_FAT,
                  CIRURGIA.DS_CIRURGIA) DS_PRO_FAT,
          SAL_CIR.CD_SAL_CIR CD_SAL_CIR,
          SAL_CIR.DS_SAL_CIR DS_SAL_CIR,
          TO_CHAR(AGE_CIR.DT_INICIO_AGE_CIR, 'HH24') HORA_PREV_INI,
          TO_CHAR(AGE_CIR.DT_FINAL_AGE_CIR, 'HH24') HORA_PREV_FIM,
          CIRURGIA.CD_PROCEDIMENTO_SIH CD_SIH_SUS
   FROM DBAMV.AVISO_CIRURGIA AVISO_CIRURGIA,
        DBAMV.ATENDIME ATENDIME,
        DBAMV.CONVENIO CONVENIO,
        DBAMV.CIRURGIA_AVISO CIRURGIA_AVISO,
        DBAMV.CIRURGIA CIRURGIA,
        DBAMV.SAL_CIR SAL_CIR,
        DBAMV.AGE_CIR AGE_CIR,
        DBAMV.PRO_FAT PRO_FAT,
        DBAMV.PROCEDIMENTO_SUS PROCEDIMENTO_SUS_A,
        DBAMV.PROCEDIMENTO_SUS PROCEDIMENTO_SUS_H
   WHERE AVISO_CIRURGIA.CD_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
     AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
     AND AVISO_CIRURGIA.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO(+)
     AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO(+) 
     AND AVISO_CIRURGIA.CD_SAL_CIR = SAL_CIR.CD_SAL_CIR(+)
     AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = AGE_CIR.CD_AVISO_CIRURGIA(+)
     AND CIRURGIA.CD_CIRURGIA = CIRURGIA_AVISO.CD_CIRURGIA
     AND CIRURGIA.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT(+)
     AND CIRURGIA.CD_PROCEDIMENTO_SIH = PROCEDIMENTO_SUS_H.CD_PROCEDIMENTO(+)
     AND CIRURGIA.CD_PROCEDIMENTO_SIA = PROCEDIMENTO_SUS_A.CD_PROCEDIMENTO(+)
   ORDER BY AVISO_CIRURGIA.CD_AVISO_CIRURGIA ,
            PRO_FAT.DS_PRO_FAT)
ORDER BY CD_AVISO_CIRURGIA,
         CD_SAL_CIR,
         DS_SAL_CIR,
         CD_PACIENTE,
         NM_PACIENTE,
         DT_AVISO_CIRURGIA,
         HORA_PREV_FIM,
         HORA_PREV_INI,
         TP_NATUREZA,
         DS_RISCO_CIRURGICO,
         DS_ACIDENTES_DURANTE_A_OPERACA,
         CD_SIH_SUS

--R_FICHA_PARA_CIR_R_EQUIPE_MEDICA1_Q_EQUIPE_MEDICA
SELECT * FROM (
SELECT   DISTINCT CD_AVISO_CIRURGIA,
                 CD_PRESTADOR,
	NM_PRESTADOR,
	DS_ATI_MED,
                P_ORDEM
FROM
(
SELECT  PRESTADOR_AVISO.CD_AVISO_CIRURGIA          CD_AVISO_CIRURGIA,
                PRESTADOR.CD_PRESTADOR                              CD_PRESTADOR,
                PRESTADOR.NM_PRESTADOR                             NM_PRESTADOR,
                ATI_MED.DS_ATI_MED                                           DS_ATI_MED,
                DECODE(ATI_MED.TP_FUNCAO,'C','0','N','1','2')     P_ORDEM
FROM      DBAMV.PRESTADOR_AVISO                                 PRESTADOR_AVISO,
                 DBAMV.ATI_MED                                                    ATI_MED,
                 DBAMV.PRESTADOR                                             PRESTADOR
WHERE  PRESTADOR_AVISO.CD_PRESTADOR        =  PRESTADOR.CD_PRESTADOR
AND        PRESTADOR_AVISO.CD_ATI_MED               =  ATI_MED.CD_ATI_MED
UNION ALL
SELECT AVISO_CIRURGIA.CD_AVISO_CIRURGIA              CD_AVISO_CIRURGIA,
              PRESTADOR.CD_PRESTADOR                              CD_PRESTADOR,
              PRESTADOR.NM_PRESTADOR                              NM_PRESTADOR,
              ATI_MED.DS_ATI_MED                                            DS_ATI_MED,
              DECODE(ATI_MED.TP_FUNCAO,'C','0','N','1','2')      P_ORDEM
FROM   DBAMV.EQUIPE_MEDICA                                          EQUIPE_MEDICA,
             DBAMV.EQUIPE_PRESTADOR                                  EQUIPE_PRESTADOR,
             DBAMV.ATI_MED                                                        ATI_MED,
             DBAMV.PRESTADOR                                                 PRESTADOR,
             DBAMV.AVISO_CIRURGIA                                         AVISO_CIRURGIA
WHERE  EQUIPE_MEDICA.CD_EQUIPE_MEDICA      =   AVISO_CIRURGIA.CD_EQUIPE_MEDICA
AND        EQUIPE_MEDICA.CD_EQUIPE_MEDICA      =  EQUIPE_PRESTADOR.CD_EQUIPE_MEDICA
AND        PRESTADOR.CD_PRESTADOR                   =  EQUIPE_PRESTADOR.CD_PRESTADOR
AND        ATI_MED.CD_ATI_MED                                =  EQUIPE_PRESTADOR.CD_ATI_MED
AND        ATI_MED.CD_ATI_MED NOT IN
                      ( SELECT  PRESTADOR_AVISO.CD_ATI_MED      CD_ATI_MED
                        FROM     DBAMV.PRESTADOR_AVISO                PRESTADOR_AVISO,
                                       DBAMV.ATI_MED                                    ATI_MED,
                                       DBAMV.PRESTADOR                             PRESTADOR,
                                       DBAMV.AVISO_CIRURGIA                     AVISO_VIRURGIA,
                                       DBAMV.CIRURGIA_AVISO                     CIRURGIA_AVISO
                        WHERE   PRESTADOR_AVISO.CD_PRESTADOR        =    PRESTADOR.CD_PRESTADOR
                        AND        AVISO_CIRURGIA.CD_AVISO_CIRURGIA      =  CIRURGIA_AVISO.CD_AVISO_CIRURGIA
                        AND        CIRURGIA_AVISO.CD_AVISO_CIRURGIA      =  PRESTADOR_AVISO.CD_AVISO_CIRURGIA
                        AND        CIRURGIA_AVISO.CD_CIRURGIA                   =  PRESTADOR_AVISO.CD_CIRURGIA
                        AND        AVISO_CIRURGIA.CD_AVISO_CIRURGIA       =   :CAMPO_CD_AVISO_CIRURGIA
                        AND        PRESTADOR_AVISO.CD_ATI_MED               =    ATI_MED.CD_ATI_MED)
)
)
WHERE CD_AVISO_CIRURGIA = :CAMPO_CD_AVISO_CIRURGIA
ORDER BY P_ORDEM, DS_ATI_MED
