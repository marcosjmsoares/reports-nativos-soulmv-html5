--R_EQUIPAMENTO
SELECT EMP_APA_EQUIP.CD_APARELHO_EQUIPAMENTO
      ,DS_APARELHO_EQUIPAMENTO
      ,DECODE(EMP_APA_EQUIP.TP_REGISTRO,'T','Taxa','E','Equipamento','Ambos') TIPO
      ,NVL(EMP_APA_EQUIP.QT_APARELHO_EQUIPAMENTO,0) QT_APARELHO_EQUIPAMENTO
      ,NVL(EMP_APA_EQUIP.QT_INATIVO,0) QT_INATIVO
      ,NVL(EMP_APA_EQUIP.QT_APARELHO_EQUIPAMENTO,0) - NVL(EMP_APA_EQUIP.QT_INATIVO,0) QT_DISPONIVEL
FROM DBAMV.APARELHOS_EQUIPTO APA_EQUIP
         ,DBAMV.EMPRESA_APARELHOS_EQUIPTO EMP_APA_EQUIP
where  APA_EQUIP.CD_APARELHO_EQUIPAMENTO = EMP_APA_EQUIP.CD_APARELHO_EQUIPAMENTO
       and EMP_APA_EQUIP.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
ORDER BY EMP_APA_EQUIP.CD_APARELHO_EQUIPAMENTO
