--R_PRE_AGENDAMENTOS_SUSPENSOS_SINTETICO
SELECT * FROM (
SELECT AVISO_CIRURGIA.CD_AVISO_CIRURGIA                     CD_AVISO_CIRURGIA
      ,PRESTADOR.CD_PRESTADOR                                    CD_PRESTADOR
      ,PRESTADOR.NM_PRESTADOR                                    NM_PRESTADOR
      ,CIRURGIA.CD_CIRURGIA                                              CD_CIRURGIA
      ,CIRURGIA.DS_CIRURGIA                                              DS_CIRURGIA
      ,TRUNC( AVISO_CIRURGIA.DT_HR_SUSPENSAO)   DT_HR_SUSPENSAO
      ,AVISO_CIRURGIA.NM_PACIENTE                                NM_PACIENTE
      ,AVISO_CIRURGIA.CD_CEN_CIR                                   CD_CEN_CIR
      ,CEN_CIR.DS_CEN_CIR                                                  DS_CEN_CIR
      ,AVISO_CIRURGIA.CD_MOTIVO_SUSPENSAO                               CD_MOTIVO_SUSPENSAO
      ,DECODE(MOTIVO_SUSPENSAO.DS_MOTIVO_SUSPENSAO, NULL, 'NAO INFORMADO', MOTIVO_SUSPENSAO.DS_MOTIVO_SUSPENSAO)  DS_MOTIVO_SUSPENSAO
      ,CONVENIO.NM_CONVENIO NM_CONVENIO
      ,CIRURGIA_AVISO.CD_CIRURGIA_AVISO
FROM    DBAMV.AVISO_CIRURGIA
              ,DBAMV.CIRURGIA_AVISO
              ,DBAMV.CEN_CIR
              ,DBAMV.CIRURGIA
              ,DBAMV.MOTIVO_SUSPENSAO
              ,DBAMV.CONVENIO
              ,DBAMV.PRESTADOR_AVISO
              ,DBAMV.PRESTADOR
              ,DBAMV.ATI_MED
              ,(SELECT CD_CONVENIO
                  FROM DBAMV.EMPRESA_CONVENIO
                 WHERE CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA) EMPRESA_CONVENIO
WHERE AVISO_CIRURGIA.SN_SUSPENSAO = 'S'
AND        AVISO_CIRURGIA.CD_AVISO_CIRURGIA   =  CIRURGIA_AVISO.CD_AVISO_CIRURGIA
AND        AVISO_CIRURGIA.CD_CEN_CIR  = CEN_CIR.CD_CEN_CIR
AND        CIRURGIA_AVISO.CD_CIRURGIA = CIRURGIA.CD_CIRURGIA
AND        AVISO_CIRURGIA.CD_MOTIVO_SUSPENSAO = MOTIVO_SUSPENSAO.CD_MOTIVO_SUSPENSAO(+)
AND        CIRURGIA_AVISO.CD_CONVENIO  = CONVENIO.CD_CONVENIO(+)
AND        CIRURGIA_AVISO.CD_CIRURGIA_AVISO  =  PRESTADOR_AVISO.CD_CIRURGIA_AVISO(+)
AND        PRESTADOR_AVISO.CD_PRESTADOR =  PRESTADOR.CD_PRESTADOR(+)
AND        PRESTADOR_AVISO.CD_ATI_MED =  ATI_MED.CD_ATI_MED(+)
AND        (ATI_MED.TP_FUNCAO = 'C'  OR PRESTADOR_AVISO.CD_PRESTADOR IS NULL)
AND EMPRESA_CONVENIO.CD_CONVENIO(+) = CONVENIO.CD_CONVENIO
ORDER BY CD_PRESTADOR, DT_AVISO_CIRURGIA
)
ORDER BY CD_CEN_CIR,DS_CEN_CIR,
CD_MOTIVO_SUSPENSAO,DS_MOTIVO_SUSPENSAO,
CD_AVISO_CIRURGIA,NM_PACIENTE,NM_CONVENIO
