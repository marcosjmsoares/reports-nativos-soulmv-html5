--R_FICHA_ADMISSAO_UNIMED
SELECT * FROM (
SELECT PERGUNTAS.CD_PERGUNTAS                                  CD_PERGUNTAS,
              PERGUNTAS.DS_PERGUNTAS                                   DS_PERGUNTAS,
              RESPOSTAS_FECHADAS.CD_RESPOSTAS               CD_RESPOSTAS,
              '(  ) ' || RESPOSTAS_FECHADAS.DS_RESPOSTAS     DS_RESPOSTAS,
              GRUPO_PERGUNTAS.CD_GRUPO_PERGUNTAS     CD_GRUPO_PERGUNTAS,
              GRUPO_PERGUNTAS.DS_GRUPO_PERGUNTAS     DS_GRUPO_PERGUNTAS
FROM   DBAMV.QUESTIONARIO_PERGUNTAS                       QUESTIONARIO_PERGUNTAS,
             DBAMV.GRUPO_PERGUNTAS                                      GRUPO_PERGUNTAS,
             DBAMV.PERGUNTAS                                                     PERGUNTAS,
             DBAMV.RESPOSTAS_FECHADAS                                 RESPOSTAS_FECHADAS
WHERE  PERGUNTAS.CD_SISTEMA                                  = 'FSCC'
AND    PERGUNTAS.CD_PERGUNTAS                               = RESPOSTAS_FECHADAS.CD_PERGUNTAS
AND    PERGUNTAS.CD_PERGUNTAS                               = QUESTIONARIO_PERGUNTAS.CD_PERGUNTA
AND    GRUPO_PERGUNTAS.CD_GRUPO_PERGUNTAS = QUESTIONARIO_PERGUNTAS.CD_GRUPO_PERGUNTAS
ORDER BY DS_GRUPO_PERGUNTAS, QUESTIONARIO_PERGUNTAS.CD_ORDEM, CD_RESPOSTAS
)
ORDER BY CD_GRUPO_PERGUNTAS,DS_GRUPO_PERGUNTAS,
CD_PERGUNTAS,DS_PERGUNTAS
