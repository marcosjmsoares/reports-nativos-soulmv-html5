--R_HIST_AVISO_CIRURGIA_SUSP
SELECT a.cd_aviso_cirurgia ,
       Nvl(p.nm_paciente, a.nm_paciente) nm_paciente ,
  (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND a.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
       a.dt_aviso_cirurgia ,
       a.dt_prev_inter ,
       l.dt_agendamento ,
       l.dt_hr_suspensao ,
       l.cd_usuario_suspensao ,
       l.cd_motivo_suspensao ,
       m.ds_motivo_suspensao ,
       l.dt_hr_ret_suspensao ,
       l.cd_usuario_ret_suspensao ,
       l.dt_cancelamento ,
       l.cd_usuario_cancelamento ,
       mc.CD_MOT_CANC ,
       mc.DS_MOT_CANC
FROM dbamv.log_aviso_cirurgia_suspensao l ,
     dbamv.aviso_cirurgia a ,
     dbamv.motivo_suspensao m ,
     dbamv.sal_cir s ,
     dbamv.mot_canc mc ,
     dbamv.cen_cir c ,
     dbamv.atendime ate,
     dbamv.paciente p
WHERE l.cd_aviso_cirurgia = a.cd_aviso_cirurgia
  AND m.cd_motivo_suspensao = l.cd_motivo_suspensao
  AND a.cd_sal_cir = s.cd_sal_cir
  AND mc.cd_mot_canc(+) = l.cd_mot_canc
  AND s.cd_cen_cir = c.cd_cen_cir
  AND a.cd_atendimento = ate.cd_atendimento(+)
  AND a.cd_paciente = p.cd_paciente(+)    
ORDER BY a.cd_aviso_cirurgia + l.cd_log_avi_cir_suspensao DESC
