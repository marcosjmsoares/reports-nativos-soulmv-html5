--R_CIR_REALIZ
select distinct decode(length(c.dia),1,'0' || c.dia,c.dia) dia , nvl(b.total_dia,0) total_dia
,nvl(acum_cir_realiz_dia('01',decode(length(c.dia),1,'0' || c.dia,c.dia),:CAMPO_MES_ANO),0) total_acum_dia
from dbamv.v_realiz b,
      dbamv.dia_aux c
where decode(length(c.dia),1,'0' || c.dia,c.dia)||'/'||:CAMPO_MES_ANO =  to_char(trunc(b.dt_realizacao (+)) ,'dd/mm/yyyy')
  and decode(length(c.dia),1,'0' || c.dia,c.dia) <= to_char(last_day(to_date('01/'||:CAMPO_MES_ANO,'dd/mm/yyyy')),'dd')
order by decode(length(c.dia),1,'0' || c.dia,c.dia)
