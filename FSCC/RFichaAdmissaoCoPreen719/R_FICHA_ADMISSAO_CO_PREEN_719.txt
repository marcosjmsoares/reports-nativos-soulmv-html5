DECODE(NVL(R.TP_RACA,'*') ,'A', 'AMARELA','B','BRANCA','I','ÍNDIGENA','P','PARDA','T','PRETA') AS TP_RACA,
(SELECT T.DS_TIP_PARTO FROM DBAMV.TIP_PARTO T WHERE T.CD_TIP_PARTO = R.CD_TIP_PARTO)AS TIP_PARTO_RN ,
(SELECT m.DS_motivo_PARTO FROM DBAMV.MOTIVO_PARTO M WHERE M.cd_motivo_parto = R.cd_motivo_parto)AS motivo_parto_rn ,
MORTE.DS_TIP_MOR_RN DS_TIP_MOR_RN,
(SELECT UI.DS_UNID_INT FROM DBAMV.UNID_INT UI, DBAMV.LEITO LT WHERE UI.CD_UNID_INT = LT.CD_UNID_INT AND LT.CD_LEITO =  RN.CD_LEITO ) AS DS_UNID_INT_RN
 FROM DBAMV.RECEM_NASCIDO R,
      DBAMV.ADMISSAO_CO A,
      dbamv.TIP_ANEST,
      DBAMV.PACIENTE P,
      DBAMV.TIP_PARTO TP,
      DBAMV.ATENDIME AT,
      DBAMV.ATENDIME RN,
      DBAMV.MOTIVO_PARTO MP,
      DBAMV.ATENDIME ATE,
      DBAMV.LEITO L,
      DBAMV.PRESTADOR PRE,
      DBAMV.TIP_MOR_RN MORTE 
WHERE MP.CD_MOTIVO_PARTO(+)     = A.CD_MOTIVO_PARTO
  AND RN.CD_ATENDIMENTO(+)      = R.CD_ATENDIMENTO
  AND R.CD_ADMISSAO_CO(+)       = A.CD_ADMISSAO_CO
  AND TIP_ANEST.CD_TIP_ANEST(+) = A.CD_TIP_ANEST
  AND AT.CD_ATENDIMENTO    = A.CD_ATENDIMENTO
  AND AT.CD_PACIENTE       = P.CD_PACIENTE
  AND A.CD_TIP_PARTO       = TP.CD_TIP_PARTO(+)
  AND R.CD_ATENDIMENTO     = ATE.CD_ATENDIMENTO(+)
  AND AT.CD_ATENDIMENTO    = :CAMPO_CD_ATENDIMENTO
  AND RN.CD_LEITO          = L.CD_LEITO(+)
  AND MORTE.CD_TIP_MOR_RN(+) = R.CD_TIP_MORTE_RN
  AND PRE.CD_PRESTADOR(+)     = R.CD_PRESTADOR 
  AND A.CD_ADMISSAO_CO     = :CAMPO_CD_ADMISSAO_CO
  AND AT.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
