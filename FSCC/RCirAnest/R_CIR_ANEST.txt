--R_CIR_ANEST
SELECT * FROM (
Select tip_anest_aviso_cirurgia.cd_tip_anest,
       aviso_cirurgia.dt_realizacao,
       aviso_cirurgia.cd_cid,
       Decode( aviso_cirurgia.tp_natureza, 'L', 'Limpa', 'C', 'Contaminada', 'I', 'Infectada', 'P', 'Potencialmente Contaminada', 'N','Não especificado','Não Informada' ) tp_natureza,
       cirurgia_aviso.cd_cirurgia,
       cirurgia_aviso.cd_convenio,
       cirurgia_aviso.cd_especialid,
       prestador_aviso.cd_prestador,
       sal_cir.cd_setor,
       cirurgia.ds_cirurgia,
       Decode( especialid.ds_especialid, Null, 'Não Informada', especialid.ds_especialid ) ds_especialid,
       convenio.nm_convenio,
       tip_anest.ds_tip_anest,
       prestador.nm_prestador,
       Decode( cid.ds_cid, Null, 'Não Informado', cid.ds_cid ) ds_cid,
       setor.nm_setor
  From dbamv.aviso_cirurgia,
       dbamv.tip_anest_aviso_cirurgia,
       dbamv.cirurgia_aviso,
       dbamv.cirurgia,
       dbamv.especialid,
       dbamv.convenio,
       dbamv.tip_anest,
       dbamv.prestador_aviso,
       dbamv.prestador,
       dbamv.ati_med,
       dbamv.cid,
       dbamv.setor,
       dbamv.sal_cir,
       Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And aviso_cirurgia.cd_aviso_cirurgia = cirurgia_aviso.cd_aviso_cirurgia
   and cirurgia_aviso.cd_cirurgia = cirurgia.cd_cirurgia
   and cirurgia_aviso.cd_convenio = convenio.cd_convenio
   and cirurgia_aviso.cd_especialid = especialid.cd_especialid (+)
   and tip_anest_aviso_cirurgia.cd_tip_anest = tip_anest.cd_tip_anest(+)
   and tip_anest_aviso_cirurgia.cd_aviso_cirurgia(+) = aviso_cirurgia.cd_aviso_cirurgia
   and cirurgia_aviso.cd_cirurgia = prestador_aviso.cd_cirurgia
   and cirurgia_aviso.cd_aviso_cirurgia = prestador_aviso.cd_aviso_cirurgia
   and prestador_aviso.cd_prestador = prestador.cd_prestador
   and prestador_aviso.cd_ati_med = ati_med.cd_ati_med
   and aviso_cirurgia.cd_cid = cid.cd_cid (+)
   and aviso_cirurgia.cd_sal_cir = sal_cir.cd_sal_cir
   and sal_cir.cd_setor = setor.cd_setor
   and aviso_cirurgia.tp_situacao = 'R'
   and ati_med.tp_funcao = 'N'
   and trunc(aviso_cirurgia.dt_realizacao) between  :CAMPO_DT_INICIAL  and  
:CAMPO_CF_CID
)
ORDER BY
