--R_PRE_AGENDAMENTOS_MEDICOS
SELECT cd_prestador, nm_prestador, quantidade FROM (
SELECT cd_prestador cd_prestador_principal, 
       Count(cd_prestador) quantidade  
FROM   dbamv.prestador_aviso, aviso_cirurgia 
WHERE  sn_principal = 'S' 
       AND trunc(aviso_cirurgia.dt_aviso_cirurgia) between trunc(:CAMPO_DT_INICIAL) and trunc(:CAMPO_DT_FINAL)
       AND prestador_aviso.cd_aviso_cirurgia = aviso_cirurgia.cd_aviso_cirurgia 
	   and aviso_cirurgia.tp_situacao = 'P'
GROUP  BY cd_prestador 
) , prestador
WHERE cd_prestador_principal = cd_prestador
