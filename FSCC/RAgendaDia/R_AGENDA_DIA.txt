--R_AGENDA_DIA
SELECT DISTINCT AGE_CIR.CD_AGE_CIR CD_AGE_CIR,
                AGE_CIR.DT_INICIO_AGE_CIR DT_INICIO_AGE_CIR,
                TO_CHAR(AGE_CIR.DT_INICIO_AGE_CIR, 'HH24') HR_INICIO_AGE_CIR,
                AVISO_CIRURGIA.CD_AVISO_CIRURGIA CD_AVISO_CIRURGIA,
                AVISO_CIRURGIA.DS_A_SEGUIR DS_A_SEGUIR,
                ATENDIME.CD_ATENDIMENTO CD_ATENDIMENTO,
                SAL_CIR.CD_SAL_CIR CD_SAL_CIR,
                SAL_CIR.DS_SAL_CIR DS_SAL_CIR,
                CEN_CIR.CD_CEN_CIR CD_CEN_CIR,
                CEN_CIR.DS_CEN_CIR DS_CEN_CIR,
                NVL(PACIENTE.CD_PACIENTE, AVISO_CIRURGIA.CD_PACIENTE) CD_PACIENTE,
                NVL(PACIENTE.NM_PACIENTE, AVISO_CIRURGIA.NM_PACIENTE) NM_PACIENTE,
        (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
                NVL(CONVENIO.CD_CONVENIO, CONV_AVISO.CD_CONVENIO) CD_CONVENIO,
                NVL(CONVENIO.NM_CONVENIO, CONV_AVISO.NM_CONVENIO) NM_CONVENIO,
                TIP_ANEST.CD_TIP_ANEST CD_TIP_ANEST,
                TIP_ANEST.DS_TIP_ANEST DS_TIP_ANEST,
                AVISO_CIRURGIA.DS_OBS_AVISO DS_OBS_AVISO,
                AVISO_CIRURGIA.VL_IDADE VL_IDADE,
                AVISO_CIRURGIA.NR_TELEFONE_CONTATO NR_TELEFONE_CONTATO,
                AVISO_CIRURGIA.DT_PREV_INTER DT_PREV_INTER,
                AVISO_CIRURGIA.VL_TEMPO_PREVISTO VL_TEMPO_PREVISTO,
                CIRURGIA_AVISO.CD_CIRURGIA COD_CIRURGIA,
                CIRURGIA_AVISO.DS_OBSERVACAO DS_OBS_CIR,
                DECODE(CIRURGIA_AVISO.SN_PACOTE, 'S', 'SIM', 'NÃO') SN_PACOTE,
                DECODE(CIRURGIA_AVISO.DS_NPADRONIZADO, NULL , CIRURGIA.DS_CIRURGIA , 'N.P.' || CIRURGIA_AVISO.DS_NPADRONIZADO) DS_CIRURGIA,
                AVISO_CIRURGIA.SN_UTI SN_UTI,
                TIP_ACOM.DS_TIP_ACOM DS_TIP_ACOM,
                AVISO_CIRURGIA.cd_unid_int ,
                unid_int.ds_unid_int,
  (SELECT NM_PRESTADOR
   FROM DBAMV.PRESTADOR
   WHERE PRESTADOR.CD_PRESTADOR = AVISO_CIRURGIA.CD_PRESTADOR_INDICACAO) NM_PRESTADOR_INDICACAO
FROM DBAMV.AGE_CIR AGE_CIR,
     DBAMV.SAL_CIR SAL_CIR,
     DBAMV.CEN_CIR CEN_CIR,
     DBAMV.AVISO_CIRURGIA AVISO_CIRURGIA,
     DBAMV.CIRURGIA_AVISO CIRURGIA_AVISO,
     DBAMV.CIRURGIA CIRURGIA,
     DBAMV.CONVENIO CONVENIO,
     DBAMV.CONVENIO CONV_AVISO,
     DBAMV.ATENDIME ATENDIME,
     DBAMV.TIP_ANEST TIP_ANEST,
     DBAMV.PACIENTE PACIENTE,
     DBAMV.TIP_ACOM TIP_ACOM,
     DBAMV.UNID_INT UNID_INT 
WHERE AVISO_CIRURGIA.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
  AND AVISO_CIRURGIA.TP_SITUACAO <> 'C'
  AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = AGE_CIR.CD_AVISO_CIRURGIA
  AND AVISO_CIRURGIA.CD_TIP_ANEST = TIP_ANEST.CD_TIP_ANEST(+)
  AND AVISO_CIRURGIA.CD_CEN_CIR = CEN_CIR.CD_CEN_CIR(+)
  AND AVISO_CIRURGIA.CD_SAL_CIR = SAL_CIR.CD_SAL_CIR(+)
  AND AVISO_CIRURGIA.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO(+)
  AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE(+)
  AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO(+)
  AND CIRURGIA_AVISO.CD_CONVENIO = CONV_AVISO.CD_CONVENIO(+)
  AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
  AND CIRURGIA_AVISO.CD_CIRURGIA = CIRURGIA.CD_CIRURGIA(+)
  AND TO_CHAR(AGE_CIR.DT_INICIO_AGE_CIR, 'DD/MM/YYYY') = :CAMPO_DATA_AGENDA
  AND AVISO_CIRURGIA.CD_TIPO_ACOMODACAO = TIP_ACOM.CD_TIP_ACOM(+)
  AND aviso_cirurgia.cd_unid_int = unid_int.cd_unid_int (+)  
ORDER BY DT_INICIO_AGE_CIR ,
         HR_INICIO_AGE_CIR

--R_AGENDA_DIA_R_PROD_Q_PRODUTO
SELECT * FROM (
SELECT AV.CD_AVISO_CIRURGIA  CD_AVISO_PROD
              ,AV.CD_CIRURGIA              CD_CIR_PROD
              ,AV.CD_PRODUTO              CD_PROD
              ,AV.DS_PRODUTO              DS_PROD
              ,AV.QT_SOLICITADA          QT_SOLIC_PROD
FROM  DBAMV.IT_AV_PRODUTOS    AV
           ,DBAMV.PRODUTO                   PROD
           ,DBAMV.UNI_PRO                     UNI_PRO
WHERE AV.CD_PRODUTO = PROD.CD_PRODUTO(+)
AND   AV.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO(+)
AND Exists ( SELECT Empresa_Produto.Cd_Produto
                        FROM Dbamv.Empresa_Produto
                        WHERE Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
                              AND Empresa_Produto.Cd_Produto = PROD.Cd_Produto )
AND (AV.CD_PRODUTO IS NULL
OR        AV.CD_PRODUTO NOT IN (SELECT ITFORMULA.CD_PRODUTO_TEM             CD_PRODUTO
FROM        DBAMV.CIRURGIA_KIT            CIRURGIA_KIT,
            DBAMV.FORMULA                    FORMULA,
            DBAMV.PRODUTO                   PRODUTO,
            DBAMV.ITFORMULA,
            DBAMV.PRODUTO PROD_ITEM,
            DBAMV.UNI_PRO
 WHERE      CIRURGIA_KIT.CD_AVISO_CIRURGIA = AV.CD_AVISO_CIRURGIA
      AND   CIRURGIA_KIT.CD_FORMULA                   =      FORMULA.CD_FORMULA
     AND   FORMULA.CD_PRODUTO                           =      PRODUTO.CD_PRODUTO
    AND Exists ( SELECT Empresa_Produto.Cd_Produto FROM Dbamv.Empresa_Produto WHERE Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA AND Empresa_Produto.Cd_Produto = Produto.Cd_Produto )
    AND   FORMULA.CD_FORMULA                            = ITFORMULA.CD_FORMULA
     AND   ITFORMULA.CD_PRODUTO_TEM              = PROD_ITEM.CD_PRODUTO
    AND   ITFORMULA.CD_UNI_PRO                           = UNI_PRO.CD_UNI_PRO) )
ORDER BY AV.DS_PRODUTO
)
WHERE CD_AVISO_PROD = :CAMPO_CD_AVISO_CIRURGIA
