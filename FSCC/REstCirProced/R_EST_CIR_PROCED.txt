--R_EST_CIR_PROCED
SELECT   V_CIR_PROCED.CD_PRO_FAT       	CD_PRO_FAT
	,V_CIR_PROCED.DS_PRO_FAT       	DS_PRO_FAT
       	,COUNT( * ) 			TOT_REALI
FROM    	DBAMV.V_CIR_PROCED
WHERE TRUNC(V_CIR_PROCED.DATA)  BETWEEN :CAMPO_DATA_INI AND 
AND        V_CIR_PROCED.TP_SITUACAO = 'R'
GROUP BY V_CIR_PROCED.CD_PRO_FAT
                   ,V_CIR_PROCED.DS_PRO_FAT
ORDER BY  V_CIR_PROCED.DS_PRO_FAT
