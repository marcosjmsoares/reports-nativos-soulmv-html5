--R_EST_CIR_ESP
SELECT CD_PRESTADOR
       ,NM_PRESTADOR
       ,CD_ESPECIALID
       ,DS_ESPECIALID
       ,TOT_REALI
       ,TOT_CANCE
       ,TOT_AGEND
       ,TOT_REALI+TOT_CANCE+TOT_AGEND AS TOT_GERAL
       FROM (
SELECT  V_CIR_ESPEC.CD_PRESTADOR     CD_PRESTADOR
       ,PRESTADOR.NM_PRESTADOR          NM_PRESTADOR
       ,V_CIR_ESPEC.CD_ESPECIALID          CD_ESPECIALID
       ,V_CIR_ESPEC.DS_ESPECIALID          DS_ESPECIALID
       ,SUM( DECODE(V_CIR_ESPEC.TP_SITUACAO, 'R', 1, 0) ) TOT_REALI
       ,SUM( DECODE(V_CIR_ESPEC.TP_SITUACAO, 'C', 1, 0) ) TOT_CANCE
       ,SUM( DECODE(V_CIR_ESPEC.TP_SITUACAO, 'G', 1, 0) ) TOT_AGEND
FROM      DBAMV.V_CIR_ESPEC
               ,DBAMV.PRESTADOR
WHERE  TRUNC(DATA) BETWEEN :CAMPO_DATA_INI AND 
AND        V_CIR_ESPEC.CD_PRESTADOR   = PRESTADOR.CD_PRESTADOR
GROUP BY  V_CIR_ESPEC.CD_PRESTADOR
                 ,PRESTADOR.NM_PRESTADOR
            ,V_CIR_ESPEC.CD_ESPECIALID
            ,V_CIR_ESPEC.DS_ESPECIALID
ORDER BY V_CIR_ESPEC.DS_ESPECIALID
         , PRESTADOR.NM_PRESTADOR
)
ORDER BY DS_ESPECIALID,CD_ESPECIALID,NM_PRESTADOR
