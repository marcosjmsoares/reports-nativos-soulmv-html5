--R_CIR_CEN_CUS_CONV
SELECT * FROM (
select setor.cd_setor cd_setor
         ,setor.nm_setor nm_setor
         ,convenio.cd_convenio cd_convenio
         ,convenio.nm_convenio nm_convenio
         ,cirurgia.cd_cirurgia cd_cirurgia
         ,cirurgia.ds_cirurgia ds_cirurgia
         ,decode(cirurgia.tp_cirurgia,'P','PEQUENO',
                                                    'M', 'MEDIO',
                                                    'G','GRANDE') tp_cirurgia
         ,count(cirurgia.tp_cirurgia) total_cirurgia
from  dbamv.cirurgia cirurgia
     ,dbamv.aviso_cirurgia  avi_cir
     ,dbamv.cirurgia_aviso  cir_avi
     ,dbamv.sal_cir sal_cir
     ,dbamv.setor setor
     ,dbamv.convenio convenio
     ,dbamv.atendime atendime,
     Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
And avi_cir.tp_situacao        = 'R'
and   avi_cir.cd_aviso_cirurgia  = cir_avi.cd_aviso_cirurgia
and   avi_cir.cd_sal_cir = sal_cir.cd_sal_cir
and   sal_cir.cd_setor = setor.cd_setor
and   cir_avi.cd_cirurgia = cirurgia.cd_cirurgia
and   cir_avi.cd_convenio = convenio.cd_convenio
and   avi_cir.cd_atendimento = atendime.cd_atendimento
and       to_char(avi_cir.dt_realizacao,'MM/YYYY') = :CAMPO_DT_COMPET
group by setor.cd_setor
        ,setor.nm_setor
        ,convenio.cd_convenio
        ,convenio.nm_convenio
        ,cirurgia.cd_cirurgia
        ,cirurgia.ds_cirurgia
        ,cirurgia.tp_cirurgia
order by nm_setor,
         convenio.nm_convenio,
         cirurgia.ds_cirurgia,
         cirurgia.tp_cirurgia
)
ORDER BY NM_SETOR,CD_SETOR,
NM_CONVENIO,CD_CONVENIO
