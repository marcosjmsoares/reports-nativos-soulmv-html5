--R_ANEST_REALIZ
SELECT * FROM (
SELECT TIP_ANEST.CD_TIP_ANEST CD_TIP_ANEST
      ,TIP_ANEST.DS_TIP_ANEST DS_TIP_ANEST
      ,AVISO_CIRURGIA.CD_CEN_CIR CD_CEN_CIR
      ,CEN_CIR.DS_CEN_CIR DS_CEN_CIR
      ,TRUNC(AVISO_CIRURGIA.DT_REALIZACAO) DT_REALIZACAO
      ,TO_CHAR(AVISO_CIRURGIA.DT_REALIZACAO,'D') DIA
      ,COUNT(TIP_ANEST.CD_TIP_ANEST) TOT_DIA_ANEST
,(SELECT SUM(TOT_DIA_ANEST) FROM (
            SELECT TIP_ANEST.CD_TIP_ANEST CD_TIP_ANEST
                  ,TIP_ANEST.DS_TIP_ANEST DS_TIP_ANEST
                  ,AVISO_CIRURGIA.CD_CEN_CIR CD_CEN_CIR
                  ,CEN_CIR.DS_CEN_CIR DS_CEN_CIR
                  ,TRUNC(AVISO_CIRURGIA.DT_REALIZACAO) DT_REALIZACAO
                  ,TO_CHAR(AVISO_CIRURGIA.DT_REALIZACAO,'D') DIA
                  ,COUNT(TIP_ANEST.CD_TIP_ANEST) TOT_DIA_ANEST
            FROM DBAMV.AVISO_CIRURGIA AVISO_CIRURGIA
                ,DBAMV.TIP_ANEST TIP_ANEST
                ,DBAMV.CEN_CIR CEN_CIR
                ,DBAMV.TIP_ANEST_AVISO_CIRURGIA
WHERE AVISO_CIRURGIA.TP_SITUACAO                   = 'R'
  AND AVISO_CIRURGIA.CD_MULTI_EMPRESA              = DBAMV.pkg_mv2000.le_empresa
  AND TIP_ANEST_AVISO_CIRURGIA.CD_TIP_ANEST        = TIP_ANEST.CD_TIP_ANEST(+)
  AND TIP_ANEST_AVISO_CIRURGIA.CD_AVISO_CIRURGIA(+)   = AVISO_CIRURGIA.CD_AVISO_CIRURGIA
  AND AVISO_CIRURGIA.CD_CEN_CIR                    = CEN_CIR.CD_CEN_CIR
  AND      TRUNC(AVISO_CIRURGIA.DT_REALIZACAO)  BETWEEN  To_Date(Concat(To_Char(TO_DATE(:CAMPO_DATA_INI, 'DD/MM/YYYY'),'DD/MM/YYYY'),'00'),'DD/MM/YYYY HH24')  AND To_Date(Concat(To_Char(TO_DATE(:CAMPO_DATA_FIM, 'DD/MM/YYYY'),'DD/MM/YYYY'),'23'),'DD/MM/YYYY HH24')
GROUP BY AVISO_CIRURGIA.CD_CEN_CIR
                   ,CEN_CIR.DS_CEN_CIR
                   ,TIP_ANEST.CD_TIP_ANEST
                   ,TIP_ANEST.DS_TIP_ANEST
                   ,TRUNC(AVISO_CIRURGIA.DT_REALIZACAO)
                   ,TO_CHAR(AVISO_CIRURGIA.DT_REALIZACAO,'D'))) TOTAL
  FROM DBAMV.AVISO_CIRURGIA AVISO_CIRURGIA
      ,DBAMV.TIP_ANEST_AVISO_CIRURGIA
      ,DBAMV.TIP_ANEST TIP_ANEST
      ,DBAMV.CEN_CIR CEN_CIR
WHERE AVISO_CIRURGIA.TP_SITUACAO         = 'R'
  AND AVISO_CIRURGIA.CD_MULTI_EMPRESA              = DBAMV.pkg_mv2000.le_empresa
  AND TIP_ANEST_AVISO_CIRURGIA.CD_TIP_ANEST        = TIP_ANEST.CD_TIP_ANEST(+)
  AND TIP_ANEST_AVISO_CIRURGIA.CD_AVISO_CIRURGIA(+)   = AVISO_CIRURGIA.CD_AVISO_CIRURGIA
  AND AVISO_CIRURGIA.CD_CEN_CIR          = CEN_CIR.CD_CEN_CIR
  AND      TRUNC(AVISO_CIRURGIA.DT_REALIZACAO)  BETWEEN  To_Date(Concat(To_Char(TO_DATE(:CAMPO_DATA_INI, 'DD/MM/YYYY'),'DD/MM/YYYY'),'00'),'DD/MM/YYYY HH24')  AND To_Date(Concat(To_Char(TO_DATE(:CAMPO_DATA_FIM, 'DD/MM/YYYY'),'DD/MM/YYYY'),'23'),'DD/MM/YYYY HH24')
GROUP BY AVISO_CIRURGIA.CD_CEN_CIR
                   ,CEN_CIR.DS_CEN_CIR
                   ,TIP_ANEST.CD_TIP_ANEST
                   ,TIP_ANEST.DS_TIP_ANEST
                   ,TRUNC(AVISO_CIRURGIA.DT_REALIZACAO)
                   ,TO_CHAR(AVISO_CIRURGIA.DT_REALIZACAO,'D')
)
ORDER BY CD_CEN_CIR,DS_CEN_CIR,
CD_TIP_ANEST,DS_TIP_ANEST, DT_REALIZACAO
