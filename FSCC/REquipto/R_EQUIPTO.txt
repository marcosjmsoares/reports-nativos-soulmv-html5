--R_EQUIPTO
SELECT APA_EQUI.CD_APARELHO_EQUIPAMENTO                  Codigo,
               APA_EQUI.DS_APARELHO_EQUIPAMENTO                  Descricao,
               EMP_APA_EQUI.NR_TEMPO_ASSEPSIA                                 NR_TEMPO_ASSEPSIA,
               EMP_APA_EQUI.TP_MODO_DE_COBRANCA                          TP_MODO_DE_COBRANCA,
               APA_EQUI.CD_ESP_EQPTO                                            CD_ESP_EQPTO,
               Nvl(EMP_APA_EQUI.QT_INATIVO, 0)                                                   QT_INATIVO,
               Nvl(EMP_APA_EQUI.QT_APARELHO_EQUIPAMENTO, 0)                 QT_APARELHO_EQUIPAMENTO,
               ESP_EQPTO.DS_ESP_EQPTO                                        DS_ESP_EQPTO
FROM    DBAMV.APARELHOS_EQUIPTO                                      APA_EQUI,
              DBAMV.EMPRESA_APARELHOS_EQUIPTO                   EMP_APA_EQUI,
              DBAMV.ESP_EQPTO                                                         ESP_EQPTO
WHERE ESP_EQPTO.CD_ESP_EQPTO = APA_EQUI.CD_ESP_EQPTO
       and  APA_EQUI.CD_APARELHO_EQUIPAMENTO = EMP_APA_EQUI.CD_APARELHO_EQUIPAMENTO(+)
       and  (EMP_APA_EQUI.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA OR EMP_APA_EQUI.CD_MULTI_EMPRESA IS NULL)
ORDER BY
