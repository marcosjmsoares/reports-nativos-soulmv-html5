select sysdate from dual


/*----------------------------------------------------------------------------*/
--R_FICHA_CIR_ANEST_R_1_Q_1
SELECT * FROM
  (SELECT DISTINCT AVISO_CIRURGIA.CD_PACIENTE CD_PACIENTE ,
                   AVISO_CIRURGIA.NM_PACIENTE NM_PACIENTE ,
          (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND AVISO_CIRURGIA.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE,
                   PRESTADOR_AVISO.CD_AVISO_CIRURGIA CD_AVISO_CIRURGIA ,
                   ATI_MED.DS_ATI_MED DS_ATI_MED ,
                   PRESTADOR.CD_PRESTADOR CD_PRESTADOR ,
                   PRESTADOR.NM_PRESTADOR NM_PRESTADOR ,
                   PRESTADOR.DS_CODIGO_CONSELHO DS_CONSELHO ,
                   CIRURGIA.CD_CIRURGIA CD_CIRURGIA ,
                   CIRURGIA.DS_CIRURGIA DS_CIRURGIA ,
                   AVISO_CIRURGIA.CD_ATENDIMENTO CD_ATENDIMENTO ,
                   DECODE(ATI_MED.TP_FUNCAO, 'C', 1, 'A', 2, 'N', 3, 4) ORDEM ,
                   TRUNC(AVISO_CIRURGIA.DT_INICIO_CIRURGIA) DATA_CIRURGIA ,
                   AVISO_CIRURGIA.DT_INICIO_CIRURGIA DT_INICIO_CIRURGIA ,
                   AVISO_CIRURGIA.DT_FIM_CIRURGIA DT_FIM_CIRURGIA ,
                   AVISO_CIRURGIA.VL_TEMPO_DURACAO DURACAO ,
                   AVISO_CIRURGIA.CD_ASA CD_ASA ,
                   AVISO_CIRURGIA.CD_SAL_CIR CD_SAL_CIR
   FROM DBAMV.PRESTADOR_AVISO PRESTADOR_AVISO ,
        DBAMV.ATI_MED ATI_MED ,
        DBAMV.PRESTADOR PRESTADOR ,
        DBAMV.AVISO_CIRURGIA AVISO_CIRURGIA ,
        DBAMV.CIRURGIA_AVISO CIRURGIA_AVISO ,
        DBAMV.CIRURGIA CIRURGIA
   WHERE :CAMPO_CD_AVISO_CIRURGIA <> '%'
     AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA =:CAMPO_CD_AVISO_CIRURGIA
     AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
     AND CIRURGIA_AVISO.CD_AVISO_CIRURGIA = PRESTADOR_AVISO.CD_AVISO_CIRURGIA
     AND CIRURGIA_AVISO.CD_CIRURGIA = PRESTADOR_AVISO.CD_CIRURGIA
     AND CIRURGIA_AVISO.CD_CIRURGIA = CIRURGIA.CD_CIRURGIA
     AND PRESTADOR_AVISO.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
     AND PRESTADOR_AVISO.CD_ATI_MED = ATI_MED.CD_ATI_MED 
   UNION ALL SELECT TO_NUMBER(NULL) CD_PACIENTE ,
                    NULL NM_PACIENTE ,
                    NULL NM_SOCIAL_PACIENTE ,
                    TO_NUMBER(NULL) CD_AVISO_CIRURGIA ,
                    NULL DS_ATI_MED ,
                    TO_NUMBER(NULL) CD_PRESTADOR ,
                    NULL NM_PRESTADOR ,
                    NULL DS_CONSELHO ,
                    TO_NUMBER(NULL) CD_CIRURGIA ,
                    NULL DS_CIRURGIA ,
                    TO_NUMBER(NULL) CD_ATENDIMENTO ,
                    TO_NUMBER(NULL) ORDEM ,
                    TO_DATE(NULL) DATA_CIRURGIA ,
                    TO_DATE(NULL) DT_INICIO_CIRURGIA ,
                    TO_DATE(NULL) DT_FIM_CIRURGIA ,
                    TO_DATE(NULL) DURACAO ,
                    NULL CD_ASA ,
                    NULL CD_SAL_CIR
   FROM DUAL
   WHERE :CAMPO_CD_AVISO_CIRURGIA = '%' )
ORDER BY CD_CIRURGIA,
         DS_CIRURGIA,
         CD_PACIENTE,
         NM_PACIENTE,
         CD_AVISO_CIRURGIA,
         CD_ATENDIMENTO,
         DATA_CIRURGIA,
         DT_INICIO_CIRURGIA,
         DT_FIM_CIRURGIA,
         DURACAO,
         CD_ASA,
         CD_SAL_CIR,
         ORDEM,
         DS_ATI_MED
