--R_CIR_PER
SELECT * FROM (
SELECT  CEN_CIR.CD_CEN_CIR              CD_CEN_CIR,
                CEN_CIR.DS_CEN_CIR              DS_CEN_CIR,
                CIRURGIA.CD_CIRURGIA          CODIGO,
                CIRURGIA.DS_CIRURGIA          DESCRICAO,
                COUNT(*)                                    QUANTIDADE
FROM      DBAMV.CIRURGIA           CIRURGIA,
                DBAMV.CIRURGIA_AVISO     CIRURGIA_AVISO,
                DBAMV.AVISO_CIRURGIA     AVISO_CIRURGIA,
                DBAMV.CEN_CIR            CEN_CIR
WHERE  CIRURGIA_AVISO.CD_CIRURGIA  =  CIRURGIA.CD_CIRURGIA
AND        CIRURGIA_AVISO.CD_AVISO_CIRURGIA  =  AVISO_CIRURGIA.CD_AVISO_CIRURGIA
AND        AVISO_CIRURGIA.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
AND        AVISO_CIRURGIA.TP_SITUACAO = 'R'
AND        AVISO_CIRURGIA.CD_CEN_CIR  =  CEN_CIR.CD_CEN_CIR
AND        AVISO_CIRURGIA.DT_REALIZACAO BETWEEN To_Date(Concat(To_Char(:CAMPO_DATA_INI,'dd/mm/yyyy'),'00'),'dd/mm/yyyy hh24') and  To_Date(Concat(To_Char(:CAMPO_DATA_FIM,'dd/mm/yyyy'),'23'),'dd/mm/yyyy hh24')
GROUP BY  CEN_CIR.CD_CEN_CIR,
                     CEN_CIR.DS_CEN_CIR,
                     CIRURGIA.CD_CIRURGIA,
                     CIRURGIA.DS_CIRURGIA
)
ORDER BY CD_CEN_CIR,DS_CEN_CIR, :CAMPO_ORDEM
