--R_BALANCETE_SETOR
SELECT * FROM (
SELECT    A.CD_CONTABIL           CD_CONTABIL
         ,A.DS_CONTA              DS_CONTA
         ,A.CD_GRAU               CD_GRAU
         ,A.TP_CONTA              TP_CONTA
         ,SUM( A.VL_SALDO_ANT )   VL_SALDO_ANT
         ,SUM( A.VL_DEBITO )      VL_DEBITO
         ,SUM( A.VL_CREDITO )     VL_CREDITO
         ,SUM( A.VL_SALDO_ATU )   VL_SALDO_ATU
         ,A.CD_SETOR
         ,A.DT_SALDO_MES
FROM (
SELECT  PC.CD_CONTABIL             CD_CONTABIL
       ,PC.DS_CONTA                DS_CONTA
       ,PC.CD_GRAU_DA_CONTA        CD_GRAU
       ,PC.TP_CONTA                           TP_CONTA
       ,DBAMV.FCCT_SALDO_CONTA_SETOR(  :CAMPO_CD_MULTI_EMPRESA, :CAMPO_CD_GRAU_CONTA, :CAMPO_DT_ANO_ANT, :CAMPO_DT_MES_ANT, PC.CD_CONTABIL, SMS.CD_SETOR )   VL_SALDO_ANT
       ,nvl( SMS.VL_DEBITO, 0 )     VL_DEBITO
       ,nvl( SMS.VL_CREDITO, 0 )    VL_CREDITO
       ,DBAMV.FCCT_SALDO_CONTA_SETOR(  :CAMPO_CD_MULTI_EMPRESA, :CAMPO_CD_GRAU_CONTA, :CAMPO_DT_ANO, :CAMPO_DT_MES, PC.CD_CONTABIL, SMS.CD_SETOR )   VL_SALDO_ATU
       ,SMS.CD_SETOR
       ,SMS.DT_SALDO_MES
FROM    DBAMV.PLANO_CONTAS                PC
             ,DBAMV.SALDO_MENSAL_SETOR  SMS
WHERE   PC.CD_MULTI_EMPRESA                  =  :CAMPO_CD_MULTI_EMPRESA
AND     PC.CD_GRAU_DA_CONTA                   <= :CAMPO_CD_GRAU_CONTA
AND     PC.CD_REDUZIDO                                  = SMS.CD_REDUZIDO(+)
:CAMPO_CF_COND_SETOR
) A
GROUP BY  A.CD_SETOR,
          A.CD_CONTABIL
         ,A.DS_CONTA
         ,A.CD_GRAU
         ,A.TP_CONTA
         ,A.DT_SALDO_MES
ORDER BY  A.CD_SETOR, A.CD_CONTABIL
)
ORDER BY CD_SETOR
