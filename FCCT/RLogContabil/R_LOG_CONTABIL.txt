--R_LOG_CONTABIL
SELECT L.CD_MULTI_EMPRESA,
       L.CD_LOG,
       L.DT_GERACAO,
       L.DT_COMPETENCIA,
       L.NM_USUARIO,
       L.DS_MACHINE,
       L.TP_EVENTO,
       L.NM_OSUSER,
       I.DT_MENSAGEM,
       I.DS_MENSAGEM,
       I.TP_MENSAGEM
FROM
   DBAMV.LOG_CONTABIL L,
   DBAMV.ITEM_LOG_CONTABIL I
WHERE L.CD_LOG = I.CD_LOG(+)
AND L.CD_LOG = :CAMPO_CD_LOG
AND L.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
ORDER BY I.DT_MENSAGEM, I.CD_ITEM_LOG_CONTABIL
