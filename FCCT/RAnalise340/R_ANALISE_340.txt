SELECT * FROM (
select p.cd_reduzido
        , p.cd_contabil
        , p.ds_conta									CONTA
       , decode(substr(p.cd_contabil,1,1),m.cd_receita,'RECEITA',m.cd_despesa,'DESPESA') 	TIPO
       , s_ant_1.dt_saldo_mes		dt_saldo_mes_1
       , s_ant_2.dt_saldo_mes		dt_saldo_mes_2
       , nvl(s_ant_1.vl_saldo_mes,0)     	vl_saldo_mes_1
       , nvl(s_ant_2.vl_saldo_mes,0)     	vl_saldo_mes_2
from     (select  cd_reduzido, dt_saldo_mes, vl_Saldo_mes
         from    dbamv.saldo_mensal s
         where s.dt_saldo_mes >= to_date(:CAMPO_DT_MES|| '/' || :CAMPO_DT_ANO, 'MM/YYYY') ) s_ant_1
        ,(select  cd_reduzido, dt_saldo_mes, vl_saldo_mes
         from    dbamv.saldo_mensal s
         where   s.dt_saldo_mes = add_months(To_Date(:CAMPO_DT_MES|| '/' || :CAMPO_DT_ANO, 'MM/YYYY'),-1) ) s_ant_2
        ,dbamv.plano_contas   p
        ,dbamv.multi_empresas m
where   s_ant_1.cd_reduzido = p.cd_reduzido
and      s_ant_2.cd_reduzido (+) = p.cd_reduzido
and      m.cd_multi_empresa  = p.cd_multi_empresa
and      substr(p.cd_contabil,1,1) in (m.cd_receita,m.cd_despesa)
AND     P.CD_GRAU_DA_CONTA        <= :CAMPO_CD_GRAU_CONTA
AND     P.SN_IMPRIME		IN		('S',:CAMPO_SN_IMPRIME)
UNION
select p.cd_reduzido
        , p.cd_contabil
        , p.ds_conta									CONTA
       , decode(substr(p.cd_contabil,1,1),m.cd_receita,'RECEITA',m.cd_despesa,'DESPESA') 	TIPO
       , s_ant_1.dt_saldo_mes		dt_saldo_mes_1
       , s_ant_2.dt_saldo_mes		dt_saldo_mes_2
       , nvl(s_ant_1.vl_saldo_mes,0)     	vl_saldo_mes_1
       , nvl(s_ant_2.vl_saldo_mes,0)     	vl_saldo_mes_2
from     (select  cd_reduzido, dt_saldo_mes, vl_Saldo_mes
         from    dbamv.saldo_mensal s
         where   s.dt_saldo_mes >= to_date(:CAMPO_DT_MES|| '/' || :CAMPO_DT_ANO, 'MM/YYYY') ) s_ant_1
        ,(select  cd_reduzido, dt_saldo_mes, vl_saldo_mes
         from    dbamv.saldo_mensal s
         where   s.dt_saldo_mes = add_months(To_Date(:CAMPO_DT_MES|| '/' || :CAMPO_DT_ANO, 'MM/YYYY'),-1) ) s_ant_2
        ,dbamv.plano_contas   p
        ,dbamv.multi_empresas m
where   s_ant_1.cd_reduzido (+) = p.cd_reduzido
and      s_ant_2.cd_reduzido = p.cd_reduzido
and      m.cd_multi_empresa  = p.cd_multi_empresa
and      substr(p.cd_contabil,1,1) in (m.cd_receita,m.cd_despesa)
AND     P.CD_GRAU_DA_CONTA        <= :CAMPO_CD_GRAU_CONTA
AND     P.SN_IMPRIME		IN		('S',:CAMPO_SN_IMPRIME)
order by tipo, CONTA
)
ORDER BY TIPO,DT_SALDO_MES_1,DT_SALDO_MES_2
