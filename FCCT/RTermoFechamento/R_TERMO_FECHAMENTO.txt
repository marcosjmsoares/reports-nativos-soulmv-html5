--R_TERMO_FECHAMENTO
select multi.cd_multi_empresa
          ,multi.ds_multi_empresa
          ,multi.ds_nome_contador
          ,multi.cd_cpf_contador
          ,multi.cd_crc_contador
          ,multi.cd_cgc
          ,multi.cd_iest
          ,multi.cd_imun
          ,multi.cd_ijunta
          ,multi.ds_endereco
          ,multi.cd_cidade
          ,cidade.nm_cidade
          ,multi.cd_uf
          ,termos.ds_termo
          ,hospital.ds_titular
from  dbamv.multi_empresas multi
        ,dbamv.termos_multi_emp termos
        ,dbamv.cidade cidade
        ,dbamv.hospital hospital
where multi.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
          and multi.cd_cidade = cidade.cd_cidade
          and multi.cd_multi_empresa = termos.cd_multi_empresa
          and termos.tp_termo = 'F'
	  and Hospital.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
