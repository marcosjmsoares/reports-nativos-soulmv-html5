--R_CONFLOTE_SETOR
SELECT * FROM (
select  lcto_contabil.cd_lote cd_lote
          ,lote.ds_lote ds_lote
          ,lote.cd_multi_empresa 
          ,lote.cd_lote_pai 
          ,multi_empresas.ds_multi_empresa 
          ,to_char(lcto_contabil.dt_lcto, 'dd/mm/yyyy') data
          ,lcto_contabil.cd_lcto_contabil cd_lcto_contabil
          ,lcto_contabil.cd_lcto_movimento cd_lcto_movimento
          ,lcto_contabil.cd_reduzido_debito red_debito
          ,lcto_contabil.cd_reduzido_credito red_credito
      ,decode(nvl(lcto_setor.vl_lcto_setor,-1),-1,lcto_contabil.vl_lancado,lcto_setor.vl_lcto_setor ) vl_lancado
          ,lcto_contabil.ds_complemento ds_complemento
          ,plano_contas.cd_reduzido
         ,decode(lcto_contabil.cd_reduzido_debito,plano_contas.cd_reduzido,plano_contas.cd_contabil) contabil_deb
         ,decode(lcto_contabil.cd_reduzido_credito,plano_contas1.cd_reduzido,plano_contas1.cd_contabil) contabil_cre
         ,historico_padrao.ds_historico_padrao ds_historico
         ,decode(lcto_setor.cd_setor_debito,setor.cd_setor,setor.cd_setor) cen_deb 
         ,decode(lcto_setor.cd_setor_credito,setor1.cd_setor,setor1.cd_setor) cen_cre 
from   dbamv.lcto_contabil
       , dbamv.lcto_setor
        ,dbamv.setor
       ,dbamv.setor setor1
       ,dbamv.plano_contas
       ,dbamv.plano_contas plano_contas1
       ,dbamv.lote
       ,dbamv.historico_padrao
       ,dbamv.multi_empresas
where  Lote.cd_multi_empresa = multi_empresas.cd_multi_empresa
       and lcto_contabil.cd_historico_padrao = historico_padrao.cd_historico_padrao(+)
       and lcto_contabil.cd_reduzido_debito = plano_contas.cd_reduzido(+)
       and lcto_contabil.cd_reduzido_credito = plano_contas1.cd_reduzido(+)
       and lcto_contabil.cd_lote = lote.cd_lote
       and lcto_contabil.cd_lcto_contabil = lcto_setor.cd_lcto_contabil(+)
       and lcto_setor.cd_setor_debito = setor.cd_setor(+)
       and lcto_setor.cd_setor_credito = setor1.cd_setor(+)
       and plano_contas.cd_multi_empresa =  :CAMPO_CD_MULTI_EMPRESA
union all
select  lcto_contabil.cd_lote cd_lote
          ,lote.ds_lote ds_lote
          ,lote.cd_multi_empresa 
          ,lote.cd_lote_pai 
          ,multi_empresas.ds_multi_empresa 
          ,to_char(lcto_contabil.dt_lcto, 'dd/mm/yyyy') data
         ,lcto_contabil.cd_lcto_contabil cd_lcto_contabil
          ,lcto_contabil.cd_lcto_movimento cd_lcto_movimento
         ,lcto_contabil.cd_reduzido_debito red_debito
         ,lcto_contabil.cd_reduzido_credito red_credito
      ,decode(nvl(lcto_setor.vl_lcto_setor,-1),-1,lcto_contabil.vl_lancado,lcto_setor.vl_lcto_setor ) vl_lancado
         ,lcto_contabil.ds_complemento ds_complemento
         ,plano_contas.cd_reduzido
         ,decode(lcto_contabil.cd_reduzido_debito,plano_contas.cd_reduzido,plano_contas.cd_contabil) contabil_deb
         ,decode(lcto_contabil.cd_reduzido_credito,plano_contas.cd_reduzido,plano_contas.cd_contabil) contabil_cre
         ,historico_padrao.ds_historico_padrao ds_historico
        ,decode(lcto_setor.cd_setor_debito,setor.cd_setor,setor.cd_setor) cen_deb 
        ,decode(lcto_setor.cd_setor_credito,setor1.cd_setor,setor1.cd_setor) cen_cre 
from   dbamv.lcto_contabil
       , dbamv.lcto_setor
       ,dbamv.setor
       ,dbamv.setor setor1
       ,dbamv.plano_contas
       ,dbamv.lote
       ,dbamv.historico_padrao
       ,dbamv.multi_empresas
where  lote.cd_multi_empresa = multi_empresas.cd_multi_empresa
       and lcto_contabil.cd_historico_padrao = historico_padrao.cd_historico_padrao(+)
       and lcto_contabil.cd_reduzido_credito = plano_contas.cd_reduzido(+)
       and lcto_contabil.cd_reduzido_debito is null
       and lcto_contabil.cd_lote = lote.cd_lote
       and lcto_contabil.cd_lcto_contabil = lcto_setor.cd_lcto_contabil(+)
       and lcto_setor.cd_setor_debito = setor.cd_setor(+)
       and lcto_setor.cd_setor_credito = setor1.cd_setor(+)
       and plano_contas.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
:CAMPO_CF_CD_SETOR_NULO
)
