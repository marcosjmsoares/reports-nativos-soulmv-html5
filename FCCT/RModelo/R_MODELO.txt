--R_MODELO
SELECT CD_NIVEL
       , DS_EVENTO
       , DBAMV.APLICA_ESCALA_VALOR(COMP_ATUAL,:CAMPO_ESCALA) COMP_ATUAL
       , DBAMV.APLICA_ESCALA_VALOR(COMP_ANT,:CAMPO_ESCALA) COMP_ANT
       , DBAMV.APLICA_ESCALA_VALOR(COMP_DOIS_ANT,:CAMPO_ESCALA) COMP_DOIS_ANT
       , CD_CONTABIL_GRUPO
       , DS_CONTABIL_GRUPO
       , AH_ATUAL
       , AH_ANT
       , AH_DOIS_ANT
       , AV_ATUAL
       , AV_ANT
       , AV_DOIS_ANT
       , GRUPO_ANT
	   , DS_OPERACAO
FROM (
SELECT   MOVIMENTO.CD_NIVEL,
         MOVIMENTO.DS_EVENTO,
         DBAMV.CONVERTE_MOEDA_VALOR(:CAMPO_CD_MULTI_EMPRESA,:CAMPO_CD_MOEDA,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),MOVIMENTO.COMP_ATUAL,NULL,NULL) COMP_ATUAL,
         DBAMV.CONVERTE_MOEDA_VALOR(:CAMPO_CD_MULTI_EMPRESA,:CAMPO_CD_MOEDA,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),MOVIMENTO.COMP_ANT,NULL,NULL) COMP_ANT,
         DBAMV.CONVERTE_MOEDA_VALOR(:CAMPO_CD_MULTI_EMPRESA,:CAMPO_CD_MOEDA,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),MOVIMENTO.COMP_DOIS_ANT,NULL,NULL) COMP_DOIS_ANT,
		     MOVIMENTO.CD_CONTABIL_GRUPO,
		     GRUPO.DS_CONTABIL_GRUPO,
         CASE
                  WHEN MOVIMENTO.COMP_DOIS_ANT != 0 THEN ROUND(((MOVIMENTO.COMP_ATUAL/MOVIMENTO.COMP_DOIS_ANT))*100,2)
                  ELSE 0
         END AH_ATUAL,
         CASE
                  WHEN MOVIMENTO.COMP_DOIS_ANT != 0 THEN ROUND(((MOVIMENTO.COMP_ANT/MOVIMENTO.COMP_DOIS_ANT))*100,2)
                  ELSE 0
         END AH_ANT,
         CASE
                  WHEN MOVIMENTO.COMP_DOIS_ANT != 0 THEN ROUND(((MOVIMENTO.COMP_DOIS_ANT/MOVIMENTO.COMP_DOIS_ANT))*100,2)
                  ELSE 0
         END AH_DOIS_ANT,
         CASE
                  WHEN AV.COMP_ATUAL != 0 THEN ROUND(((MOVIMENTO.COMP_ATUAL/AV.COMP_ATUAL))*100,2)
                  ELSE 0
         END AV_ATUAL,
         CASE
                  WHEN AV.COMP_ANT != 0 THEN ROUND(((MOVIMENTO.COMP_ANT/AV.COMP_ANT))*100,2)
                  ELSE 0
         END AV_ANT,
         CASE
                  WHEN AV.COMP_DOIS_ANT != 0 THEN ROUND(((MOVIMENTO.COMP_DOIS_ANT/AV.COMP_DOIS_ANT))*100,2)
                  ELSE 0
         END AV_DOIS_ANT,
	 NVL((SELECT ME.CD_CONTABIL_GRUPO CD_CONTABIL_GRUPO
                FROM   (
                          SELECT CD_MODELO_CONTABIL, SN_VISIVEL, CD_ORDEM, CD_CONTABIL_GRUPO
                          FROM DBAMV.MODELO_CONTABIL_EVENTO
                          ORDER BY CD_ORDEM DESC, CD_MODELO_CONTABIL_EVENTO, CD_MODELO_CONTABIL
                       ) ME,
                       DBAMV.MODELO_CONTABIL M
                WHERE  M.CD_MODELO_CONTABIL = ME.CD_MODELO_CONTABIL
                AND    ME.CD_MODELO_CONTABIL = :CAMPO_CD_MODELO
                AND    M.SN_ATIVO = 'S'
                AND    M.TP_STATUS <> 'D'
                AND    ME.SN_VISIVEL = 'S'
                AND    ME.CD_ORDEM < MOVIMENTO.CD_ORDEM
                AND    ROWNUM = 1),'0') GRUPO_ANT,
		MOVIMENTO.DS_OPERACAO
FROM     (
                SELECT LPAD(ME.DS_MODELO_CONTABIL_EVENTO,LENGTH(ME.DS_MODELO_CONTABIL_EVENTO) + (ME.CD_NIVEL*3),' ')                      DS_EVENTO,
                       DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'), NVL(ME.TP_APRESENTACAO_VALOR, :CAMPO_TP_SALDO),:CAMPO_PER_COMP, :CAMPO_LIST_CD_MULTI_EMPRESA) COMP_ATUAL,
                       DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(1*:CAMPO_PER_COMP)), NVL(ME.TP_APRESENTACAO_VALOR, :CAMPO_TP_SALDO), :CAMPO_PER_COMP, :CAMPO_LIST_CD_MULTI_EMPRESA) COMP_ANT,
                       DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(2*:CAMPO_PER_COMP)), NVL(ME.TP_APRESENTACAO_VALOR, :CAMPO_TP_SALDO), :CAMPO_PER_COMP, :CAMPO_LIST_CD_MULTI_EMPRESA) COMP_DOIS_ANT,
                       ME.CD_NIVEL,
                       ME.CD_ORDEM,
					   ME.DS_OPERACAO,
		       NVL(ME.CD_CONTABIL_GRUPO,'0') CD_CONTABIL_GRUPO
                FROM   DBAMV.MODELO_CONTABIL M,
                       DBAMV.MODELO_CONTABIL_EVENTO ME
                WHERE  M.CD_MODELO_CONTABIL = ME.CD_MODELO_CONTABIL
                AND    ME.CD_MODELO_CONTABIL = :CAMPO_CD_MODELO
                AND    M.SN_ATIVO = 'S'
                AND    M.TP_STATUS <> 'D'
                AND    ME.SN_VISIVEL = 'S') MOVIMENTO,
         (      SELECT SUM(DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'), NVL(ME.TP_APRESENTACAO_VALOR, :CAMPO_TP_SALDO),:CAMPO_PER_COMP, :CAMPO_LIST_CD_MULTI_EMPRESA)) COMP_ATUAL,
                       SUM(DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(1*:CAMPO_PER_COMP)), NVL(ME.TP_APRESENTACAO_VALOR, :CAMPO_TP_SALDO), :CAMPO_PER_COMP, :CAMPO_LIST_CD_MULTI_EMPRESA)) COMP_ANT,
                       SUM(DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(2*:CAMPO_PER_COMP)), NVL(ME.TP_APRESENTACAO_VALOR, :CAMPO_TP_SALDO), :CAMPO_PER_COMP, :CAMPO_LIST_CD_MULTI_EMPRESA)) COMP_DOIS_ANT,
                       ME.CD_NIVEL,
                       NVL(ME.CD_CONTABIL_GRUPO,'0') CD_CONTABIL_GRUPO
                FROM   DBAMV.MODELO_CONTABIL M,
                       DBAMV.MODELO_CONTABIL_EVENTO ME
                WHERE  M.CD_MODELO_CONTABIL = ME.CD_MODELO_CONTABIL
                AND    ME.CD_MODELO_CONTABIL = :CAMPO_CD_MODELO
                AND    M.SN_ATIVO = 'S'
                AND    M.TP_STATUS <> 'D'
                AND    ME.SN_VISIVEL = 'S'
                GROUP BY ME.CD_NIVEL, NVL(ME.CD_CONTABIL_GRUPO,'0')) AV,
		 DBAMV.MODELO_CONTABIL_GRUPO GRUPO
         WHERE AV.CD_NIVEL = MOVIMENTO.CD_NIVEL
           AND AV.CD_CONTABIL_GRUPO = MOVIMENTO.CD_CONTABIL_GRUPO
		   AND GRUPO.CD_MODELO_CONTABIL(+) = :CAMPO_CD_MODELO
		   AND MOVIMENTO.CD_CONTABIL_GRUPO = GRUPO.CD_CONTABIL_GRUPO(+)
ORDER BY CD_ORDEM)
