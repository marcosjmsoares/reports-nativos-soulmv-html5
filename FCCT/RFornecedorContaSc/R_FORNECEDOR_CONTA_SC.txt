--R_FORNECEDOR_CONTA_SC
SELECT f.cd_fornecedor codigo
	  ,f.nm_fornecedor
	  ,f.nm_fantasia fantasia
	  ,pe.cd_reduzido
	  ,pe.cd_contabil
	  ,pe.ds_conta
     FROM dbamv.fornecedor f
LEFT JOIN dbamv.forn_conta fc
       ON fc.cd_fornecedor = f.cd_fornecedor
	  AND fc.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
     JOIN dbamv.plano_contabil pconta
	   ON pconta.cd_reduzido = fc.cd_reduzido
     JOIN dbamv.plano_estr pe
       ON pe.cd_plano_estr = pconta.cd_plano_estr
	  AND pe.cd_plano = :CAMPO_CD_PLANO
WHERE 1 = 1
