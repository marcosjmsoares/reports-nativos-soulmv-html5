--R_MODELO_P_ORCA
SELECT CD_NIVEL ,
  DS_EVENTO ,
  DBAMV.APLICA_ESCALA_VALOR(COMP_ATUAL,:CAMPO_ESCALA) COMP_ATUAL ,
  DBAMV.APLICA_ESCALA_VALOR(COMP_ANT,:CAMPO_ESCALA) COMP_ANT ,
  DBAMV.APLICA_ESCALA_VALOR(COMP_DOIS_ANT,:CAMPO_ESCALA) COMP_DOIS_ANT ,
  DBAMV.APLICA_ESCALA_VALOR(COMP_TRES_ANT,:CAMPO_ESCALA) COMP_TRES_ANT ,
  DBAMV.APLICA_ESCALA_VALOR(COMP_QUATRO_ANT,:CAMPO_ESCALA) COMP_QUATRO_ANT,
  DBAMV.APLICA_ESCALA_VALOR(COMP_CINCO_ANT,:CAMPO_ESCALA) COMP_CINCO_ANT,
  DBAMV.APLICA_ESCALA_VALOR(COMP_SEIS_ANT,:CAMPO_ESCALA) COMP_SEIS_ANT,
  DBAMV.APLICA_ESCALA_VALOR(COMP_SETE_ANT,:CAMPO_ESCALA) COMP_SETE_ANT,
  DBAMV.APLICA_ESCALA_VALOR(COMP_OITO_ANT,:CAMPO_ESCALA) COMP_OITO_ANT,
  DBAMV.APLICA_ESCALA_VALOR(COMP_NOVE_ANT,:CAMPO_ESCALA) COMP_NOVE_ANT,
  DBAMV.APLICA_ESCALA_VALOR(COMP_DEZ_ANT,:CAMPO_ESCALA) COMP_DEZ_ANT,
  DBAMV.APLICA_ESCALA_VALOR(COMP_ONZE_ANT,:CAMPO_ESCALA) COMP_ONZE_ANT,
  CD_CONTABIL_GRUPO ,
  DS_CONTABIL_GRUPO ,
  AH_ATUAL,
  AH_ANT,
  AH_DOIS_ANT,
  AH_TRES_ANT,
  AH_QUATRO_ANT,
  AH_CINCO_ANT,
  AV_ATUAL,
  AV_ANT,
  AV_DOIS_ANT,
  AV_TRES_ANT,
  AV_QUATRO_ANT,
  AV_CINCO_ANT,
  GRUPO_ANT ,
  DS_OPERACAO
FROM
  (SELECT MOVIMENTO.CD_NIVEL ,
    MOVIMENTO.DS_EVENTO ,
    DBAMV.CONVERTE_MOEDA_VALOR(:CAMPO_CD_MULTI_EMPRESA,:CAMPO_CD_MOEDA,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),MOVIMENTO.COMP_ATUAL,NULL,NULL) COMP_ATUAL,
    DBAMV.CONVERTE_MOEDA_VALOR(:CAMPO_CD_MULTI_EMPRESA,:CAMPO_CD_MOEDA,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),MOVIMENTO.COMP_ANT,NULL,NULL) COMP_ANT,
    DBAMV.CONVERTE_MOEDA_VALOR(:CAMPO_CD_MULTI_EMPRESA,:CAMPO_CD_MOEDA,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),MOVIMENTO.COMP_DOIS_ANT,NULL,NULL) COMP_DOIS_ANT,
    DBAMV.CONVERTE_MOEDA_VALOR(:CAMPO_CD_MULTI_EMPRESA,:CAMPO_CD_MOEDA,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),MOVIMENTO.COMP_TRES_ANT,NULL,NULL) COMP_TRES_ANT,
    DBAMV.CONVERTE_MOEDA_VALOR(:CAMPO_CD_MULTI_EMPRESA,:CAMPO_CD_MOEDA,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),MOVIMENTO.COMP_QUATRO_ANT,NULL,NULL) COMP_QUATRO_ANT,
    DBAMV.CONVERTE_MOEDA_VALOR(:CAMPO_CD_MULTI_EMPRESA,:CAMPO_CD_MOEDA,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),MOVIMENTO.COMP_CINCO_ANT,NULL,NULL) COMP_CINCO_ANT,
    DBAMV.CONVERTE_MOEDA_VALOR(:CAMPO_CD_MULTI_EMPRESA,:CAMPO_CD_MOEDA,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),MOVIMENTO.COMP_SEIS_ANT,NULL,NULL) COMP_SEIS_ANT,
    DBAMV.CONVERTE_MOEDA_VALOR(:CAMPO_CD_MULTI_EMPRESA,:CAMPO_CD_MOEDA,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),MOVIMENTO.COMP_SETE_ANT,NULL,NULL) COMP_SETE_ANT,
    DBAMV.CONVERTE_MOEDA_VALOR(:CAMPO_CD_MULTI_EMPRESA,:CAMPO_CD_MOEDA,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),MOVIMENTO.COMP_OITO_ANT,NULL,NULL) COMP_OITO_ANT,
    DBAMV.CONVERTE_MOEDA_VALOR(:CAMPO_CD_MULTI_EMPRESA,:CAMPO_CD_MOEDA,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),MOVIMENTO.COMP_NOVE_ANT,NULL,NULL) COMP_NOVE_ANT,
    DBAMV.CONVERTE_MOEDA_VALOR(:CAMPO_CD_MULTI_EMPRESA,:CAMPO_CD_MOEDA,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),MOVIMENTO.COMP_DEZ_ANT,NULL,NULL) COMP_DEZ_ANT,
    DBAMV.CONVERTE_MOEDA_VALOR(:CAMPO_CD_MULTI_EMPRESA,:CAMPO_CD_MOEDA,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),MOVIMENTO.COMP_ONZE_ANT,NULL,NULL) COMP_ONZE_ANT,
    MOVIMENTO.CD_CONTABIL_GRUPO,
    GRUPO.DS_CONTABIL_GRUPO,
    0 AH_ATUAL,
    0 AH_ANT,
    0 AH_DOIS_ANT,
    0 AH_TRES_ANT,
    0 AH_QUATRO_ANT,
    0 AH_CINCO_ANT,
    DBAMV.CONVERTE_MOEDA_VALOR(:CAMPO_CD_MULTI_EMPRESA,:CAMPO_CD_MOEDA,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),MOVIMENTO.COMP_ATUAL_REAL,NULL,NULL) AV_ATUAL,
    DBAMV.CONVERTE_MOEDA_VALOR(:CAMPO_CD_MULTI_EMPRESA,:CAMPO_CD_MOEDA,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),MOVIMENTO.COMP_ANT_REAL,NULL,NULL) AV_ANT,
    DBAMV.CONVERTE_MOEDA_VALOR(:CAMPO_CD_MULTI_EMPRESA,:CAMPO_CD_MOEDA,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),MOVIMENTO.COMP_DOIS_ANT_REAL,NULL,NULL) AV_DOIS_ANT,
    DBAMV.CONVERTE_MOEDA_VALOR(:CAMPO_CD_MULTI_EMPRESA,:CAMPO_CD_MOEDA,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),MOVIMENTO.COMP_TRES_ANT_REAL,NULL,NULL) AV_TRES_ANT,
    DBAMV.CONVERTE_MOEDA_VALOR(:CAMPO_CD_MULTI_EMPRESA,:CAMPO_CD_MOEDA,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),MOVIMENTO.COMP_QUATRO_ANT_REAL,NULL,NULL) AV_QUATRO_ANT,
    DBAMV.CONVERTE_MOEDA_VALOR(:CAMPO_CD_MULTI_EMPRESA,:CAMPO_CD_MOEDA,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),MOVIMENTO.COMP_CINCO_ANT_REAL,NULL,NULL) AV_CINCO_ANT,
    NVL(
    (SELECT ME.CD_CONTABIL_GRUPO CD_CONTABIL_GRUPO
    FROM
      (SELECT CD_MODELO_CONTABIL,
        SN_VISIVEL,
        CD_ORDEM,
        CD_CONTABIL_GRUPO
      FROM DBAMV.MODELO_CONTABIL_EVENTO
      ORDER BY CD_ORDEM DESC,
        CD_MODELO_CONTABIL_EVENTO,
        CD_MODELO_CONTABIL
      ) ME,
      DBAMV.MODELO_CONTABIL M
    WHERE M.CD_MODELO_CONTABIL = ME.CD_MODELO_CONTABIL
    AND ME.CD_MODELO_CONTABIL  = :CAMPO_CD_MODELO
    AND M.SN_ATIVO             = 'S'
    AND M.TP_STATUS           <> 'D'
    AND ME.SN_VISIVEL          = 'S'
    AND ME.CD_ORDEM            < MOVIMENTO.CD_ORDEM
    AND ROWNUM                 = 1
    ),'0') GRUPO_ANT,
    MOVIMENTO.DS_OPERACAO
  FROM
    (SELECT LPAD(ME.DS_MODELO_CONTABIL_EVENTO,LENGTH(ME.DS_MODELO_CONTABIL_EVENTO) + (ME.CD_NIVEL * 3), ' ') DS_EVENTO,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),'SF') COMP_ATUAL,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(1*:CAMPO_PER_COMP)),'SF') COMP_ANT,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(2*:CAMPO_PER_COMP)),'SF') COMP_DOIS_ANT,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(3*:CAMPO_PER_COMP)),'SF') COMP_TRES_ANT,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(4*:CAMPO_PER_COMP)),'SF') COMP_QUATRO_ANT,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(5*:CAMPO_PER_COMP)),'SF') COMP_CINCO_ANT,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(6*:CAMPO_PER_COMP)),'SF') COMP_SEIS_ANT,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(7*:CAMPO_PER_COMP)),'SF') COMP_SETE_ANT,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(8*:CAMPO_PER_COMP)),'SF') COMP_OITO_ANT,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(9*:CAMPO_PER_COMP)),'SF') COMP_NOVE_ANT,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(10*:CAMPO_PER_COMP)),'SF') COMP_DEZ_ANT,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(11*:CAMPO_PER_COMP)),'SF') COMP_ONZE_ANT,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),'SR') COMP_ATUAL_REAL,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(1*:CAMPO_PER_COMP)),'SR') COMP_ANT_REAL,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(2*:CAMPO_PER_COMP)),'SR') COMP_DOIS_ANT_REAL,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(3*:CAMPO_PER_COMP)),'SR') COMP_TRES_ANT_REAL,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(4*:CAMPO_PER_COMP)),'SR') COMP_QUATRO_ANT_REAL,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(5*:CAMPO_PER_COMP)),'SR') COMP_CINCO_ANT_REAL,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(6*:CAMPO_PER_COMP)),'SR') COMP_SEIS_ANT_REAL,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(7*:CAMPO_PER_COMP)),'SR') COMP_SETE_ANT_REAL,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(8*:CAMPO_PER_COMP)),'SR') COMP_OITO_ANT_REAL,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(9*:CAMPO_PER_COMP)),'SR') COMP_NOVE_ANT_REAL,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(10*:CAMPO_PER_COMP)),'SR') COMP_DEZ_ANT_REAL,
      DBAMV.FNC_FCCT_CALC_FORMULA_MODELO(ME.CD_MODELO_CONTABIL_EVENTO,ADD_MONTHS(TO_DATE(:CAMPO_DT_COMP,'DD/MM/YYYY'),-(11*:CAMPO_PER_COMP)),'SR') COMP_ONZE_ANT_REAL,
      ME.CD_NIVEL,
      ME.CD_ORDEM,
      ME.DS_OPERACAO,
      NVL(ME.CD_CONTABIL_GRUPO,'0') CD_CONTABIL_GRUPO
    FROM DBAMV.MODELO_CONTABIL M,
      DBAMV.MODELO_CONTABIL_EVENTO ME
    WHERE M.CD_MODELO_CONTABIL = ME.CD_MODELO_CONTABIL
    AND ME.CD_MODELO_CONTABIL  = :CAMPO_CD_MODELO
    AND M.SN_ATIVO             = 'S'
    AND M.TP_STATUS           <> 'D'
    AND ME.SN_VISIVEL          = 'S'
    ) MOVIMENTO,
    DBAMV.MODELO_CONTABIL_GRUPO GRUPO
  WHERE GRUPO.CD_MODELO_CONTABIL(+) = :CAMPO_CD_MODELO
  AND MOVIMENTO.CD_CONTABIL_GRUPO = GRUPO.CD_CONTABIL_GRUPO(+)
  ORDER BY CD_ORDEM
  )
