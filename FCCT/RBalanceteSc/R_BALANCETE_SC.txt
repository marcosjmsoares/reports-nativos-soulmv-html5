--R_BALANCETE_SC
SELECT
		cd_reduzido
	       ,cd_contabil
	       ,tp_conta_contabil
	       ,cd_grupo_conta
	       ,tp_natureza
	       ,total_devedor_ant
	       ,total_credor_ant
	       ,ds_conta
	       ,cd_grau
	       ,tp_conta
	       ,cd_plano_estr
	       ,tp_natureza_saldo_ant
	       ,tp_natureza_saldo_atu
	       ,tp_conta_contabil
	       ,DBAMV.PKG_FCCT_BALANCETE_PERIODO.FNC_TRATA_SINAL(vl_saldo_ant, :CAMPO_SINAL) vl_saldo_ant
	       ,vl_debito
	       ,vl_credito
	       ,vl_movimento
           ,DBAMV.PKG_FCCT_BALANCETE_PERIODO.FNC_TRATA_SINAL(vl_saldo_atu, :CAMPO_SINAL) vl_saldo_atu
           FROM DBAMV.DADOS_BALANCETE
           WHERE dbamv.PKG_FCCT_BALANCETE_PERIODO.FNC_CONTAR_GRAU(cd_contabil) <= :CAMPO_CD_GRAU_CONTA
           AND cd_usuario = :CAMPO_CD_USUARIO
           AND dt_hora_inicio_carga IS NULL
           AND dt_hora_fim_carga IS NULL
           AND dt_hora_inicio_leitura IS NULL
           AND dt_hora_fim_leitura IS NULL
           AND tempo_leitura IS NULL
           AND tempo_carga IS NULL
           AND campo_auxiliar IS NULL
           ORDER BY cd_contabil
