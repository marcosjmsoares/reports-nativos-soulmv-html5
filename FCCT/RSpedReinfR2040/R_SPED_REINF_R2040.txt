--R_SPED_REINF_R2040
SELECT recAssDesp.CD_REINF_REC_ASS_DESPORTIVA,
       recAssDesp.CD_REINF_EVENTOS,
       To_Char(recAssDesp.CD_MULTI_EMPRESA_EST_REP||' - '||empresa.DS_MULTI_EMPRESA) DS_MULTI_EMPRESA_EST_REP,
       To_Char(empresa.CD_MULTI_EMPRESA||' - '||empresa.DS_MULTI_EMPRESA) DS_MULTI_EMPRESA,
       recAssDesp.CD_FORNECEDOR,
       To_Char(fornecedores.CD_FORNECEDOR||' - '||fornecedores.NM_FORNECEDOR) NM_FORNECEDOR,
       empresa.NR_CNPJCPF_REPRES_LEGAL,
       fornecedores.NR_CGC_CPF,
       eventos.DT_COMPETENCIA,
       retorno.DS_PROTOCOLO recibo,
	   recAssDesp.VL_TOTAL_REPASSE,
	   recAssDesp.VL_TOTAL_RETENCAO,
	   recAssDesp.VL_TOTAL_NAO_RETIDO
  FROM dbamv.REINF_REC_REP_ASS_DESPORTIVA recAssDesp,
       dbamv.MULTI_EMPRESAS empresa,
       dbamv.FORNECEDOR fornecedores,
       dbamv.REINF_EVENTOS eventos,
       dbamv.REINF_TRANS_RETORNO retorno
 WHERE recAssDesp.CD_MULTI_EMPRESA_EST_REP = empresa.CD_MULTI_EMPRESA
   AND recAssDesp.CD_FORNECEDOR = fornecedores.CD_FORNECEDOR
   AND recAssDesp.CD_REINF_EVENTOS = eventos.CD_REINF_EVENTOS
   AND eventos.CD_REINF_EVENTOS = retorno.CD_REINF_EVENTOS
   AND recAssDesp.CD_REINF_REC_ASS_DESPORTIVA = retorno.CD_REINF_REGISTRO(+)
   AND eventos.CD_EVENTO = 'R-2040'
   AND eventos.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
   AND eventos.DT_COMPETENCIA = to_date('','mm/yyyy')
