--R_PLACON
SELECT pe.cd_reduzido
      ,pe.cd_contabil
      ,pe.ds_conta
      ,(nvl(Length(REPLACE(TRANSLATE(pe.cd_contabil,'0123456789',' '),' ','')),0) + 1) cd_grau_da_conta
      ,DECODE( pe.TP_CONTA , 'S', 'Sintética', 'A', 'Analítica' ) TP_CONTA
FROM dbamv.plano_estr pe
WHERE pe.cd_plano = :CAMPO_CD_PLANO
