--R_PLACON_SINT
SELECT ALL PLANO_CONTAS.CD_REDUZIDO
             ,PLANO_CONTAS.CD_CONTABIL
             ,PLANO_CONTAS.DS_CONTA
             ,PLANO_CONTAS.CD_GRAU_DA_CONTA
             ,DECODE( PLANO_CONTAS.TP_CONTA , 'S', 'Sintética', 'A', 'Analítica' ) TP_CONTA
FROM DBAMV.PLANO_CONTAS
           ,DBAMV.MULTI_EMPRESAS
WHERE PLANO_CONTAS.CD_MULTI_EMPRESA  =  :CAMPO_CD_MULTI_EMPRESA
AND       PLANO_CONTAS.CD_MULTI_EMPRESA  =  MULTI_EMPRESAS.CD_MULTI_EMPRESA
AND      PLANO_CONTAS.TP_CONTA = 'S'
