   , dbamv.setor           setor_custo

  and substr(pc_result.cd_contabil,1,Length(g.ds_grau_da_conta)) <> me.cd_passivo

       , setor.nm_setor
       , setor_custo.cd_setor
       , setor_custo.nm_setor
       , trunc(lcto_cont.dt_lcto,'month')) lanca
