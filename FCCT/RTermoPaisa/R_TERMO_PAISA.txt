--R_TERMO_PAISA
select  multi.cd_multi_empresa		Cd_Multi_Empresa
          ,multi.ds_multi_empresa 	Ds_Multi_Empresa
          ,multi.ds_nome_contador	Ds_Nome_Contador
          ,multi.cd_cpf_contador	Cd_Cpf_Contador
          ,multi.cd_crc_contador	Cd_Crc_Contador
          ,multi.cd_cgc			Cd_Cgc
          ,multi.cd_iest		Cd_Iest
          ,multi.cd_imun		Cd_Imun
          ,multi.cd_ijunta 		Cd_Ijunta
          ,multi.ds_endereco		Ds_Endereco
          ,multi.nm_bairro          Nm_Bairro 
          ,multi.cd_cidade		Cd_Cidade
          ,cidade.nm_cidade		Nm_Cidade
          ,multi.cd_uf			Cd_Uf
          ,termos.ds_termo    		Ds_Termo
          ,hospital.ds_titular		Ds_Titular
          ,termos.tp_termo		Tp_Termo
from  dbamv.multi_empresas multi
        ,dbamv.termos_multi_emp termos
        ,dbamv.cidade cidade
        ,dbamv.hospital hospital
where multi.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
          and multi.cd_cidade = cidade.cd_cidade
          and multi.cd_multi_empresa = termos.cd_multi_empresa
          and termos.tp_termo = :CAMPO_TIPO_TERMO
          and Hospital.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
          and termos.cd_esp_termo = :CAMPO_ESP_TERMO
