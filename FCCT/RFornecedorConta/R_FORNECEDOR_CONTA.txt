--R_FORNECEDOR_CONTA
select f.cd_fornecedor codigo,
f.nm_fornecedor nm_fornecedor,
f.nm_fantasia fantasia,
fc.cd_reduzido cd_reduzido,
pc.cd_contabil cd_contabil,
pc.ds_conta ds_conta
from dbamv.fornecedor f,
     dbamv.forn_conta fc,
     dbamv.plano_contas pc
where f.cd_fornecedor = fc.cd_fornecedor (+)
  and fc.cd_reduzido = pc.cd_reduzido(+)
  and fc.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
