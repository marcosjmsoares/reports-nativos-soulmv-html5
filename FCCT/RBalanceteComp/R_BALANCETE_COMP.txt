--R_BALANCETE_COMP
SELECT PC.CD_CONTABIL           CD_CONTABIL
      ,PC.DS_CONTA              DS_CONTA
      ,PC.CD_GRAU_DA_CONTA      CD_GRAU
      ,PC.TP_CONTA              TP_CONTA
      ,PC.VL_SALDO_ANTERIOR     VL_SALDO_ANT
      ,PC.VL_VALOR_DEBITO       VL_DEBITO
      ,PC.VL_VALOR_CREDITO      VL_CREDITO
      ,PC.VL_SALDO_ATUAL        VL_SALDO_ATU
      ,PC.VL_SALDO_MOVIMENTO    VL_MOVIMENTO
      ,PC.DT_SALDO_MES          DT_SALDO_MES
FROM dbamv.temp_balancete PC
WHERE cd_grau_da_conta <= :CAMPO_CD_GRAU_CONTA
  AND sn_imprime IN ('S', :CAMPO_SN_IMPRIME)
  AND ((:CAMPO_PSNCTRLUSUPLCONTAS = 'N') OR
       (:CAMPO_PSNCTRLUSUPLCONTAS = 'S'
       AND EXISTS (SELECT CD_REDUZIDO
	                FROM DBAMV.PLANO_USUARIO_MULTI_EMPRESA PU
   	              WHERE PU.CD_REDUZIDO       = PC.CD_REDUZIDO
	                AND PU.CD_MULTI_EMPRESA  = :CAMPO_CD_MULTI_EMPRESA
                  AND PU.CD_ID_USUARIO     = USER  )))
ORDER BY  PC.CD_CONTABIL
