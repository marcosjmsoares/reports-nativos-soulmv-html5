--R_CONTAS_AUSENTES_GERENCIAL
SELECT P.CD_MULTI_EMPRESA, P.CD_REDUZIDO, P.CD_CONTABIL, P.DS_CONTA, P.TP_CONTA
FROM DBAMV.V_NATUREZA_PLANO_CONTAS P
, DBAMV.PLANO_CONTAS_GERENCIAL_COMP C
WHERE P.CD_REDUZIDO = C.CD_REDUZIDO_COMP(+)
AND C.DT_ANO(+) = TO_DATE('01/01/'||:CAMPO_DT_ANO,'DD/MM/YYYY')
AND C.CD_MULTI_EMPRESA(+) = P.CD_MULTI_EMPRESA
AND P.TP_CONTA = 'A'
AND C.CD_REDUZIDO_COMP IS NULL
AND P.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
ORDER BY P.CD_CONTABIL
