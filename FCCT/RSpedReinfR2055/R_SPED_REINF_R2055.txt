--R_SPED_REINF_R2055
SELECT R.cd_reinf_eventos
     , R.cd_reinf_prod_rural
     , R.cd_multi_empresa_adquir_prod ||' - '|| M.ds_multi_empresa cd_multi_empresa_adquir_prod
     , To_Char(fornecedores.CD_FORNECEDOR||' - '||fornecedores.NM_FORNECEDOR) NM_FORNECEDOR
     , FORNECEDORES.NR_CGC_CPF
     , R.tp_origem_informacao
     , Decode (E.TP_STATUS,                                                                                     
           'P', 'Pendente',
           'V', 'Validado',
           'A', 'Advertência',
           'C', 'Crítica',
           'T', 'Transmitido') TP_STATUS
     , To_Char(E.dt_competencia, 'MM/YYYY') dt_competencia
     , T.DS_PROTOCOLO RECIBO
  FROM dbamv.REINF_PROD_RURAL R
     , dbamv.REINF_EVENTOS E
     , dbamv.REINF_TRANS_RETORNO T
     , dbamv.FORNECEDOR FORNECEDORES
     , dbamv.MULTI_EMPRESAS M
 WHERE E.cd_reinf_eventos  = R.cd_reinf_eventos
   AND E.cd_reinf_eventos  = T.cd_reinf_eventos
   AND T.cd_reinf_registro = R.cd_reinf_prod_rural
   AND R.cd_fornecedor     = FORNECEDORES.cd_fornecedor
   AND R.cd_multi_empresa_adquir_prod = M.cd_multi_empresa
   AND E.cd_evento         = 'R-2055'
   AND E.DT_COMPETENCIA = to_date('','mm/yyyy')
