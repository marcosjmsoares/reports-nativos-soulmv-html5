--R_ACOMP_MENSAL_CARACT
select setor.cd_cen_cus           cd_cen_cus
       ,setor.cd_cen_cus             cd_cen_cus1
       ,rtrim(setor.nm_setor)||' (Especifico)'           nm_setor
       ,setor.cd_setor               cd_setor
       ,pc.cd_contabil             cd_contabil
       ,substr(pc.cd_contabil,1,1)  cd_contabil_quebra
       ,pc.cd_reduzido                  cd_reduzido_quebra
       ,pc.ds_conta                ds_conta
       ,pc.cd_reduzido            cd_reduzido
       ,sum(nvl(orc.vl_orcamento,0) )   vl_orcamento
       ,sum(abs(nvl(sms.vl_saldo_mes,0))) vl_realizado
       ,setor.classificacao            classificacao
from dbamv.plano_contas pc
      ,dbamv.setor        setor
      ,dbamv.saldo_mensal_setor  sms
      ,dbamv.orcamento    orc
where  setor.cd_setor          = sms.cd_setor
and     sms.cd_reduzido            = pc.cd_reduzido
and    orc.cd_reduzido(+)         = pc.cd_reduzido
and    pc.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
and    pc.cd_reduzido_resultado is not null
and    to_char(sms.dt_saldo_mes,'yyyy/mm')  between :CAMPO_CF_DATA and  to_char(:CAMPO_SALDO_MES,'yyyy/mm')
group by setor.cd_setor, cd_cen_cus, nm_setor, cd_contabil, ds_conta, pc.cd_reduzido, setor.classificacao
order by  setor.classificacao desc, pc.cd_contabil
