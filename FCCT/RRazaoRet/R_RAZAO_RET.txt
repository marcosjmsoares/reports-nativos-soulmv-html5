--R_RAZAO_RET_R_12_Q_PROCESSO
SELECT *
FROM
  (SELECT TRUNC( LCTO_CONTABIL.DT_LCTO ) DT_LANC_PROC ,
    TRUNC( LCTO_CONTABIL.DT_LCTO ) DT_LANC_PROC_AUX ,
    LCTO_CONTABIL.CD_LOTE CD_LOTE_PROC ,
    'AGRUPAMENTO DE PROCESSO' DS_COMPLEMENTO_PROC ,
    SUM(LCTO_CONTABIL.VL_LANCADO) VL_LANCADO_DEB_PROC ,
    to_number(NULL) VL_LANCADO_CRE_PROC ,
    SUM(LCTO_CONTABIL.VL_LANCADO) VL_LANCADO_CTA_PROC ,
    LCTO_CONTABIL.CD_PROCESSO CD_PROCESSO_PROC ,
    PLANO_CONTAS.CD_REDUZIDO CD_REDUZIDO_PROC ,
    PLANO_CONTAS.CD_CONTABIL CD_CONTABIL_PROC ,
    PLANO_CONTAS.DS_CONTA DS_CONTABIL_PROC ,
    PLANO_CONTAS.TP_CONTA TP_CONTA_PROC ,
    PROCESSO.CD_ESTRUTURAL
    || ' - '
    || PROCESSO.DS_PROCESSO DS_PROCESSO_PROC ,
    PROCESSO.CD_ESTRUTURAL CD_ESTRUTURAL,
    PLANO_CONTAS.TP_NATUREZA TP_NATUREZA
  FROM DBAMV.LOTE LT ,
    DBAMV.LCTO_CONTABIL ,
    DBAMV.HISTORICO_PADRAO,
    DBAMV.V_NATUREZA_PLANO_CONTAS PLANO_CONTAS ,
    DBAMV.PROCESSO
  WHERE LCTO_CONTABIL.CD_HISTORICO_PADRAO    = HISTORICO_PADRAO.CD_HISTORICO_PADRAO(+)
  AND LCTO_CONTABIL.CD_REDUZIDO_DEBITO       = PLANO_CONTAS.CD_REDUZIDO
  AND LCTO_CONTABIL.CD_PROCESSO              = PROCESSO.CD_PROCESSO (+) 
  AND LCTO_CONTABIL.CD_LOTE                  = LT.CD_LOTE (+)
  AND NVL(LCTO_CONTABIL.SN_TRANSITORIA,'N') IN ('N', NVL(:CAMPO_INC_CONTA_TRANSITORIA,'N')) 
  AND :CAMPO_SALDO_DIA                        = 'N'
  AND :CAMPO_SN_AGRUPA_PROC                   = 'S'     
  GROUP BY TRUNC( LCTO_CONTABIL.DT_LCTO ) ,
    LCTO_CONTABIL.CD_LOTE ,
    LCTO_CONTABIL.CD_PROCESSO ,
    PLANO_CONTAS.CD_REDUZIDO ,
    PLANO_CONTAS.CD_CONTABIL ,
    PLANO_CONTAS.DS_CONTA ,
    PLANO_CONTAS.TP_CONTA ,
    PROCESSO.DS_PROCESSO ,
    PROCESSO.CD_ESTRUTURAL,
    PLANO_CONTAS.TP_NATUREZA
  UNION ALL
  SELECT TRUNC( LCTO_CONTABIL.DT_LCTO ) DT_LANC_PROC ,
    TRUNC( LCTO_CONTABIL.DT_LCTO ) DT_LANC_PROC_AUX ,
    LCTO_CONTABIL.CD_LOTE CD_LOTE_PROC ,
    'AGRUPAMENTO DE PROCESSO' DS_COMPLEMENTO_PROC ,
    to_number(NULL) VL_LANCADO_DEB_PROC ,
    SUM(LCTO_CONTABIL.VL_LANCADO) VL_LANCADO_CRE_PROC ,
    SUM(LCTO_CONTABIL.VL_LANCADO) VL_LANCADO_CTA_PROC ,
    LCTO_CONTABIL.CD_PROCESSO CD_PROCESSO_PROC ,
    PLANO_CONTAS.CD_REDUZIDO CD_REDUZIDO_PROC ,
    PLANO_CONTAS.CD_CONTABIL CD_CONTABIL_PROC ,
    PLANO_CONTAS.DS_CONTA DS_CONTABIL_PROC ,
    PLANO_CONTAS.TP_CONTA TP_CONTA_PROC ,
    PROCESSO.DS_PROCESSO DS_PROCESSO_PROC ,
    PROCESSO.CD_ESTRUTURAL CD_ESTRUTURAL,
    PLANO_CONTAS.TP_NATUREZA TP_NATUREZA
  FROM DBAMV.LOTE LT ,
    DBAMV.LCTO_CONTABIL ,
    DBAMV.HISTORICO_PADRAO,
    DBAMV.V_NATUREZA_PLANO_CONTAS PLANO_CONTAS ,
    DBAMV.PROCESSO
  WHERE LCTO_CONTABIL.CD_HISTORICO_PADRAO    = HISTORICO_PADRAO.CD_HISTORICO_PADRAO(+)
  AND LCTO_CONTABIL.CD_REDUZIDO_CREDITO      = PLANO_CONTAS.CD_REDUZIDO
  AND LCTO_CONTABIL.CD_PROCESSO              = PROCESSO.CD_PROCESSO (+) 
  AND LCTO_CONTABIL.CD_LOTE                  = LT.CD_LOTE (+)
  AND NVL(LCTO_CONTABIL.SN_TRANSITORIA,'N') IN ('N', NVL(:CAMPO_INC_CONTA_TRANSITORIA,'N')) 
  AND :CAMPO_SALDO_DIA                        = 'N'
  AND :CAMPO_SN_AGRUPA_PROC                   = 'S'     
  GROUP BY TRUNC( LCTO_CONTABIL.DT_LCTO ) ,
    LCTO_CONTABIL.CD_LOTE ,
    LCTO_CONTABIL.CD_PROCESSO ,
    PLANO_CONTAS.CD_REDUZIDO ,
    PLANO_CONTAS.CD_CONTABIL ,
    PLANO_CONTAS.DS_CONTA ,
    PLANO_CONTAS.TP_CONTA ,
    PROCESSO.DS_PROCESSO ,
    PROCESSO.CD_ESTRUTURAL,
    PLANO_CONTAS.TP_NATUREZA
  UNION ALL
  SELECT TO_DATE(NULL) DT_LANC_PROC ,
    TO_DATE(NULL) DT_LANC_PROC_AUX ,
    TO_NUMBER(NULL) CD_LOTE_PROC ,
    TO_CHAR(NULL) DS_COMPLEMENTO_PROC ,
    TO_NUMBER(NULL) VL_LANCADO_DEB_PROC ,
    TO_NUMBER(NULL) VL_LANCADO_CRE_PROC ,
    TO_NUMBER(NULL) VL_LANCADO_CTA_PROC ,
    TO_NUMBER(NULL) CD_PROCESSO_PROC ,
    PLANO_CONTAS.CD_REDUZIDO CD_REDUZIDO_PROC ,
    PLANO_CONTAS.CD_CONTABIL CD_CONTABIL_PROC ,
    PLANO_CONTAS.DS_CONTA DS_CONTABIL_PROC ,
    PLANO_CONTAS.TP_CONTA TP_CONTA_PROC ,

    TO_CHAR(NULL) CD_ESTRUTURAL,

  AND Dbamv.Pkg_Transacao_Entre_Empresas.CompartilhaPlContas(:CAMPO_CD_MULTI_EMPRESA)                    = 'S' ))
  AND                                                                                    = 'S'
  AND                                                                             = 'S'
  AND PLANO_CONTAS.TP_CONTA                                                                           = 'A'
  AND                                                                                  = 'N'  
  ORDER BY CD_CONTABIL_PROC ,
    DT_LANC_PROC ,
    CD_ESTRUTURAL ,
    VL_LANCADO_DEB_PROC ,
    VL_LANCADO_CRE_PROC

ORDER BY CD_CONTABIL_PROC,
  DS_CONTABIL_PROC ,
  CD_REDUZIDO_PROC ,
  TP_CONTA_PROC ,

  DT_LANC_PROC ,

--R_RAZAO_RET_R_1_Q_MOV_CT_PARTIDA
SELECT * FROM (
SELECT  CD_REDUZIDO_CPART
        ,CD_CONTABIL_CPART
       ,VL_CPART_CRE
       ,VL_CPART_DEB
       ,CD_LOTE_CPART
       ,CD_LCT_MOV_CPART
       ,CD_LCTO_CONTABIL_CPART
        ,DS_HISTORICO_PADRAO_CPART
       ,DS_COMPLEMENTO_CPART
       ,cd_multi_empresa_cpart
  FROM (SELECT   plano.cd_reduzido      CD_REDUZIDO_CPART
                 , plano.cd_contabil      CD_CONTABIL_CPART
                 ,lcto.vl_lancado        vl_cpart_cre
                 ,TO_NUMBER (NULL)       vl_cpart_deb
                 ,LCTO.CD_LOTE           CD_LOTE_CPART
                 ,lcto.cd_lcto_movimento CD_LCT_MOV_CPART
                 ,lcto.cd_lcto_contabil  CD_LCTO_CONTABIL_CPART
                 ,HISTORICO_PADRAO.DS_historico_padrao  DS_HISTORICO_PADRAO_CPART
                 ,lcto.ds_complemento     DS_COMPLEMENTO_CPART
                 ,lcto.cd_multi_empresa   cd_multi_empresa_cpart
          FROM   dbamv.lcto_contabil lcto
                 ,dbamv.plano_contas plano
                  ,DBAMV.HISTORICO_PADRAO
         WHERE    lcto.CD_HISTORICO_PADRAO  = HISTORICO_PADRAO.CD_HISTORICO_PADRAO(+)
               AND lcto.cd_reduzido_credito = plano.cd_reduzido
          UNION
         SELECT  plano.cd_reduzido      CD_REDUZIDO_CPART
                  , plano.cd_contabil       CD_CONTABIL_CPART
                 ,TO_NUMBER (NULL)        vl_cpart_cre
                 ,lcto.vl_lancado         vl_cpart_deb
                 ,LCTO.CD_LOTE            CD_LOTE_CPART
                 ,lcto.cd_lcto_movimento  CD_LCT_MOV_CPART
                 ,lcto.cd_lcto_contabil   CD_LCTO_CONTABIL_CPART
                   ,HISTORICO_PADRAO.DS_historico_padrao  DS_HISTORICO_PADRAO_CPART
                 ,lcto.ds_complemento     DS_COMPLEMENTO_CPART
                 ,lcto.cd_multi_empresa   cd_multi_empresa_cpart
          FROM    dbamv.lcto_contabil lcto
                 ,dbamv.plano_contas plano
                 ,DBAMV.HISTORICO_PADRAO
         WHERE    lcto.CD_HISTORICO_PADRAO  = HISTORICO_PADRAO.CD_HISTORICO_PADRAO(+)
               AND lcto.cd_reduzido_debito = plano.cd_reduzido
                 :CAMPO_CF_LOTE
        )  A
order by  A.CD_LOTE_cpart
         ,A.CD_LCT_MOV_CPART
         ,A.cd_lcto_contabil_cpart
)
WHERE CD_LCT_MOV_CPART = :CAMPO_CD_LCTO_MOVIMENTO
AND (CD_LCTO_CONTABIL_CPART != :CAMPO_CD_LCTO_CONTABIL
OR (CD_LCTO_CONTABIL_CPART = :CAMPO_CD_LCTO_CONTABIL
AND DECODE(:CAMPO_VL_LANCADO_CRE,NULL, VL_CPART_CRE,VL_CPART_DEB) > 0)
)
