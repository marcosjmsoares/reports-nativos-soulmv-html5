--R_DRE_LISTA
SELECT * FROM (
select d.cd_dre
     , d.ds_dre
     , s.cd_sequencia
     , s.cd_nivel
     , s.ds_nivel
     , s.cd_seq_nivel
     , s.sinal
     , decode(s.sn_mostra_dre,'S','SIM','NÃO')        sn_mostra_dre
     , s.cd_reduzido
     , p.cd_contabil
     , p.ds_conta
from   dbamv.dre_seq_nivel s
     , dbamv.dre d
     , dbamv.plano_contas p
where  d.cd_dre = s.cd_dre
and    s.cd_reduzido = p.cd_reduzido (+)
:CAMPO_CF_DRE
)
ORDER BY CD_SEQUENCIA,
         cd_nivel,
         cd_seq_nivel
