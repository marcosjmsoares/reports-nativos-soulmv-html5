--R_DRE_DMPL
SELECT * FROM (
select d.cd_dre
     , d.ds_dre
     , s.cd_sequencia
     , s.ds_nivel
     , sum(t.valor)  valor
     , dc.ds_dre_coluna
     , dc.cd_dre_coluna
from   dbamv.dre_seq_nivel s
     , dbamv.dre d
     , dbamv.dre_coluna dc
     , dbamv.plano_contas p
     , dbamv.dre_seq_nivel_temp t
where d.cd_dre = s.cd_dre
  and dc.cd_dre = d.cd_dre
  and dc.cd_dre_coluna = s.cd_dre_coluna
  and s.sn_mostra_dre = 'S'
  and s.cd_reduzido = p.cd_reduzido(+)
  and t.cd_sequencia = s.cd_sequencia
  and t.cd_nivel     = s.cd_nivel
  and s.vl_lancamento is null
  and t.cd_seq_nivel = s.cd_seq_nivel
  and s.cd_dre = :CAMPO_CD_DRE
group by
       d.cd_dre
     , d.ds_dre
     , s.cd_sequencia
     , s.ds_nivel
     , dc.ds_dre_coluna
     , dc.cd_dre_coluna
union all
select d.cd_dre
     , d.ds_dre
     , s.cd_sequencia
     , s.ds_nivel
     , s.vl_lancamento valor
     , dc.ds_dre_coluna
     , dc.cd_dre_coluna
from   dbamv.dre_seq_nivel s
     , dbamv.dre d
     , dbamv.dre_coluna dc
     , dbamv.plano_contas p
where d.cd_dre = s.cd_dre
  and dc.cd_dre = d.cd_dre
  and dc.cd_dre_coluna = s.cd_dre_coluna
  and s.sn_mostra_dre = 'S'
  and s.cd_reduzido = p.cd_reduzido(+)
  and s.vl_lancamento is not null
  and s.cd_dre = :CAMPO_CD_DRE
)
ORDER BY DS_NIVEL,
DS_DRE_COLUNA
