SELECT lcto.cd_multi_empresa,
CD_LCTO_CONTABIL,
dt_lcto,
cd_lcto_movimento,
vl_lancado,
cd_reduzido_credito,
cd_reduzido_debito,
cd_lote,
ds_nome_contador,
ds_complemento,
ds_razao_social,
nr_endereco,
ds_endereco,
cd_uf,
nm_bairro,
logomarca_relatorio
FROM dbamv.lcto_contabil lcto,
     dbamv.multi_empresas emp,
     dbamv.multi_empresas_logomarca logo
WHERE emp.cd_multi_empresa = lcto.cd_multi_empresa
      and logo.cd_multi_empresa = emp.cd_multi_empresa
      And lcto.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
      And lcto.cd_lcto_contabil = :CAMPO_CD_LCTO_CONTABIL
