--R_DRE
SELECT * FROM (
select d.cd_dre
     , d.ds_dre
     , t.cd_sequencia
     , t.cd_nivel
     , s.ds_nivel
     , t.cd_seq_nivel
     , s.sinal
     , t.valor
     ,s.cd_reduzido
from   dbamv.dre_seq_nivel_temp t
     , dbamv.dre_seq_nivel s
     , dbamv.dre d
where  d.cd_dre = s.cd_dre
and    t.cd_sequencia = s.cd_sequencia
and    t.cd_nivel     = s.cd_nivel
and    t.cd_seq_nivel = s.cd_seq_nivel
and    s.sn_mostra_dre = 'S'
and    s.cd_dre = :CAMPO_CD_DRE
order by d.cd_dre, s.cd_sequencia, s.cd_nivel , s.cd_seq_nivel
)
ORDER BY CD_DRE,DS_DRE, cd_sequencia,cd_nivel,cd_seq_nivel
