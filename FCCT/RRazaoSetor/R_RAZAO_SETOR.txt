--R_RAZAO_SETOR_R_1_Q_RAZAO_SETOR
SELECT CD_SETOR
,NM_SETOR
,CD_CONTABIL_PAI
,DS_CONTABIL_PAI
,CD_CONTABIL
,DS_CONTABIL
,DT_LCTO
,CD_LOTE
,CD_LCTO
,CD_HIST_PADRAO
,DS_HIST_PADRAO
,DS_COMPLEMENTO
,VL_DEBITO
,VL_CREDITO
,TP_NATUREZA
,CD_LCTO_SETOR
,VL_SALDO
,Sum(VL_SALDO) OVER (PARTITION BY NM_SETOR,CD_CONTABIL ORDER BY DT_LCTO,CD_LCTO_SETOR) SALDO_ACUM
,CD_MULTI_EMPRESA
FROM (
SELECT X.*, DECODE(TP_NATUREZA,'D',-1,1) * (VL_CREDITO-VL_DEBITO) VL_SALDO FROM (
SELECT   LCTO_SETOR.CD_SETOR_DEBITO               CD_SETOR
        ,SETOR.NM_SETOR                           NM_SETOR
        ,PC.CD_CONTABIL                           CD_CONTABIL_PAI
        ,PC.DS_CONTA                              DS_CONTABIL_PAI
        ,PLANO_CONTAS.CD_CONTABIL                 CD_CONTABIL
        ,PLANO_CONTAS.DS_CONTA                    DS_CONTABIL
        ,TRUNC( LCTO_CONTABIL.DT_LCTO )           DT_LCTO
        ,LCTO_CONTABIL.CD_LOTE                    CD_LOTE
        ,LCTO_CONTABIL.CD_LCTO_movimento          CD_LCTO
        ,LCTO_CONTABIL.CD_HISTORICO_PADRAO        CD_HIST_PADRAO
        ,HIST_P.DS_HISTORICO_PADRAO               DS_HIST_PADRAO
        ,LCTO_CONTABIL.DS_COMPLEMENTO             DS_COMPLEMENTO
        ,LCTO_SETOR.VL_LCTO_SETOR                 VL_DEBITO
        ,0                                        VL_CREDITO
        ,PLANO_CONTAS.TP_NATUREZA                 TP_NATUREZA
        ,LCTO_SETOR.CD_LCTO_SETOR                 CD_LCTO_SETOR
        ,LCTO_CONTABIL.CD_MULTI_EMPRESA           CD_MULTI_EMPRESA
FROM     DBAMV.LCTO_SETOR                         LCTO_SETOR
        ,DBAMV.SETOR                              SETOR
        ,DBAMV.LCTO_CONTABIL                      LCTO_CONTABIL
        ,DBAMV.V_NATUREZA_PLANO_CONTAS            PLANO_CONTAS
        ,DBAMV.HISTORICO_PADRAO                   HIST_P
        ,DBAMV.PLANO_CONTAS                       PC
WHERE  TRUNC(LCTO_CONTABIL.DT_LCTO)  BETWEEN :CAMPO_DT_INICIO AND 
AND      LCTO_SETOR.CD_SETOR_DEBITO                  =  SETOR.CD_SETOR
AND      LCTO_SETOR.CD_LCTO_CONTABIL                 =  LCTO_CONTABIL.CD_LCTO_CONTABIL
AND      Nvl(LCTO_CONTABIL.CD_REDUZIDO_DEBITO, LCTO_CONTABIL.CD_REDUZIDO_CREDITO) =  PLANO_CONTAS.CD_REDUZIDO 
AND      LCTO_CONTABIL.CD_HISTORICO_PADRAO           =  HIST_P.CD_HISTORICO_PADRAO(+)
AND      PLANO_CONTAS.CD_REDUZIDO_PAI                =  PC.CD_REDUZIDO(+)
AND      :CAMPO_SALDO_DIA = 'N'
  AND (:CAMPO_PSNCTRLUSUPLCONTAS = 'N' or :CAMPO_PSNCTRLUSUPLCONTAS = 'S'
  AND EXISTS
 (SELECT CD_REDUZIDO
   FROM DBAMV.PLANO_USUARIO_MULTI_EMPRESA PU
   WHERE PU.CD_REDUZIDO       = PLANO_CONTAS.CD_REDUZIDO
   AND PU.CD_MULTI_EMPRESA  = :CAMPO_CD_MULTI_EMPRESA
   AND PU.CD_ID_USUARIO     = USER  ))
UNION ALL
SELECT   LCTO_SETOR.CD_SETOR_CREDITO              CD_SETOR
        ,SETOR.NM_SETOR                           NM_SETOR
        ,PC.CD_CONTABIL                           CD_CONTABIL_PAI
        ,PC.DS_CONTA                              DS_CONTABIL_PAI
        ,PLANO_CONTAS.CD_CONTABIL                 CD_CONTABIL
        ,PLANO_CONTAS.DS_CONTA                    DS_CONTABIL
        ,TRUNC( LCTO_CONTABIL.DT_LCTO )           DT_LCTO
        ,LCTO_CONTABIL.CD_LOTE                    CD_LOTE
        ,LCTO_CONTABIL.CD_LCTO_movimento          CD_LCTO
        ,LCTO_CONTABIL.CD_HISTORICO_PADRAO        CD_HIST_PADRAO
	    ,HIST_P.DS_HISTORICO_PADRAO               DS_HIST_PADRAO
        ,LCTO_CONTABIL.DS_COMPLEMENTO             DS_COMPLEMENTO
        ,0                                        VL_DEBITO
        ,LCTO_SETOR.VL_LCTO_SETOR                 VL_CREDITO
        ,PLANO_CONTAS.TP_NATUREZA                 TP_NATUREZA
        ,LCTO_SETOR.CD_LCTO_SETOR                 CD_LCTO_SETOR
        ,LCTO_CONTABIL.CD_MULTI_EMPRESA           CD_MULTI_EMPRESA
FROM     DBAMV.LCTO_SETOR                         LCTO_SETOR
        ,DBAMV.SETOR                              SETOR
        ,DBAMV.LCTO_CONTABIL                      LCTO_CONTABIL
        ,DBAMV.V_NATUREZA_PLANO_CONTAS            PLANO_CONTAS
        ,DBAMV.HISTORICO_PADRAO                   HIST_P
        ,DBAMV.PLANO_CONTAS                       PC
WHERE  TRUNC(LCTO_CONTABIL.DT_LCTO)  BETWEEN :CAMPO_DT_INICIO AND 
AND      LCTO_SETOR.CD_SETOR_CREDITO                 =  SETOR.CD_SETOR
AND      LCTO_SETOR.CD_LCTO_CONTABIL                 =  LCTO_CONTABIL.CD_LCTO_CONTABIL
AND      Nvl(LCTO_CONTABIL.CD_REDUZIDO_CREDITO, LCTO_CONTABIL.CD_REDUZIDO_DEBITO) =  PLANO_CONTAS.CD_REDUZIDO 
AND      LCTO_CONTABIL.CD_HISTORICO_PADRAO           =  HIST_P.CD_HISTORICO_PADRAO(+)
AND      PLANO_CONTAS.CD_REDUZIDO_PAI                =  PC.CD_REDUZIDO(+)
AND    :CAMPO_SALDO_DIA = 'N'
AND (:CAMPO_PSNCTRLUSUPLCONTAS = 'N' or :CAMPO_PSNCTRLUSUPLCONTAS= 'S'
AND EXISTS
(SELECT CD_REDUZIDO
   FROM DBAMV.PLANO_USUARIO_MULTI_EMPRESA PU
   WHERE PU.CD_REDUZIDO       = PLANO_CONTAS.CD_REDUZIDO
   AND PU.CD_MULTI_EMPRESA  = :CAMPO_CD_MULTI_EMPRESA
   AND PU.CD_ID_USUARIO     = USER  ))
) X )
ORDER BY  NM_SETOR,
          CD_CONTABIL,
          DT_LCTO
