--R_DRE_MUTACAO
SELECT * FROM (
select d.cd_dre
     , d.ds_dre
     , t.cd_sequencia
     , t.cd_nivel
     , s.ds_nivel
     , t.cd_seq_nivel
     , s.sinal
     , t.valor
     , s.cd_reduzido
     , s.cd_dre_coluna
     , c.ds_dre_coluna
     , c.cd_ordem
     , s.vl_lancamento
from   dbamv.dre_seq_nivel_temp t
     , dbamv.dre_seq_nivel s
     , dbamv.dre d
     , dbamv.dre_coluna c
where  d.cd_dre = s.cd_dre
and    t.cd_sequencia = s.cd_sequencia
and    t.cd_nivel     = s.cd_nivel
and    t.cd_seq_nivel = s.cd_seq_nivel
AND    s.cd_dre_coluna =  c.cd_dre_coluna
and    s.sn_mostra_dre = 'S'
and    s.cd_dre = :CAMPO_CD_DRE
order by d.cd_dre, s.cd_sequencia, s.cd_nivel , s.cd_seq_nivel
)
ORDER BY cd_sequencia,cd_nivel,cd_seq_nivel
