--R_LALUR_PARTEB
SELECT * FROM (
select dt_referencia
  	   ,cd_reduzido
  	   ,dt_lancamento
  	   ,ds_lancamento
  	   ,vl_lancamento
  	   ,vl_a_corrigir
  	   ,vl_coeficiente
  	   ,tp_natureza
  	   ,cd_multi_empresa
from dbamv.digitacao_parte_b_lalur
where to_char(dt_referencia, 'mm/yyyy')  <= :CAMPO_DT_FINAL
:CAMPO_CF_COND_CONTA
)
ORDER BY CD_MULTI_EMPRESA,
CD_REDUZIDO
