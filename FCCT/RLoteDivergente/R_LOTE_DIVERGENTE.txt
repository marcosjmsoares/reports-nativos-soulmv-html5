--R_LOTE_DIVERGENTE
SELECT * FROM (
SELECT     LOTE.CD_LOTE                 CD_LOTE
          ,LOTE.DS_LOTE                 DS_LOTE
          ,SUM( DECODE( LCTO.CD_REDUZIDO_DEBITO,  NULL, 0, NVL( LCTO.VL_LANCADO, 0 ) ) ) VL_DEB
          ,SUM( DECODE( LCTO.CD_REDUZIDO_CREDITO, NULL, 0, NVL( LCTO.VL_LANCADO, 0 ) ) ) VL_CRE
FROM       DBAMV.LOTE                              LOTE
                ,DBAMV.LCTO_CONTABIL          LCTO
WHERE      LOTE.CD_MULTI_EMPRESA        = :CAMPO_CD_MULTI_EMPRESA
AND       to_char(  LCTO.DT_LCTO, 'yyyy/mm' )  >= to_char( :CAMPO_DT_INICIAL, 'yyyy/mm' )
AND       to_char(  LCTO.DT_LCTO, 'yyyy/mm' )   <=to_char( :CAMPO_DT_FINAL, 'yyyy/mm' )
AND        LCTO.CD_LOTE                 = LOTE.CD_LOTE
GROUP BY   LOTE.CD_LOTE
          ,LOTE.DS_LOTE
HAVING
SUM( DECODE( LCTO.CD_REDUZIDO_DEBITO,  NULL, 0, NVL( LCTO.VL_LANCADO, 0 ) ) ) <> SUM( DECODE( LCTO.CD_REDUZIDO_CREDITO, NULL, 0, NVL( LCTO.VL_LANCADO, 0 ) ) )
ORDER BY   CD_LOTE
)
ORDER BY CD_LOTE,DS_LOTE
