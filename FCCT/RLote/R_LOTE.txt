--R_LOTE
SELECT CD_LOTE
      ,DS_LOTE
      ,DT_INICIAL_LCTO
      ,DT_FINAL_LCTO
      ,SUM(VALOR_DEBITO) SUM_VALOR_DEBITO
      ,SUM(VALOR_CREDITO) SUM_VALOR_CREDITO
      ,VL_LOTE
 FROM(
SELECT LOTE.CD_LOTE
      ,LOTE.DS_LOTE
      ,LOTE.DT_INICIAL_LCTO
      ,LOTE.DT_FINAL_LCTO
      ,SUM(DECODE(LCTO_CONTABIL.CD_REDUZIDO_CREDITO,NULL,LCTO_CONTABIL.VL_LANCADO,0)) VALOR_DEBITO
      ,SUM(DECODE(LCTO_CONTABIL.CD_REDUZIDO_DEBITO ,NULL,LCTO_CONTABIL.VL_LANCADO,0)) VALOR_CREDITO
      ,LOTE.VL_LOTE
  FROM DBAMV.LOTE
      ,DBAMV.LCTO_CONTABIL
WHERE LOTE.CD_LOTE = LCTO_CONTABIL.CD_LOTE
GROUP BY LOTE.CD_LOTE
        ,LOTE.DS_LOTE
        ,LOTE.DT_INICIAL_LCTO
        ,LOTE.DT_FINAL_LCTO
        ,LOTE.VL_LOTE
UNION ALL
SELECT LOTE.CD_LOTE
      ,LOTE.DS_LOTE
      ,LOTE.DT_INICIAL_LCTO
      ,LOTE.DT_FINAL_LCTO
      ,SUM(VL_LANCADO) VALOR_DEBITO
      ,SUM(VL_LANCADO) VALOR_CREDITO
      ,LOTE.VL_LOTE
  FROM DBAMV.LOTE
      ,DBAMV.LCTO_CONTABIL
WHERE LOTE.CD_LOTE = LCTO_CONTABIL.CD_LOTE
  AND  CD_REDUZIDO_CREDITO IS NOT NULL
  AND  CD_REDUZIDO_DEBITO  IS NOT NULL
GROUP BY LOTE.CD_LOTE
        ,LOTE.DS_LOTE
        ,LOTE.DT_INICIAL_LCTO
        ,LOTE.DT_FINAL_LCTO
        ,LOTE.VL_LOTE         )
WHERE DT_INICIAL_LCTO BETWEEN :CAMPO_DT_INICIAL GROUP BY CD_LOTE
        ,DS_LOTE
        ,DT_INICIAL_LCTO
        ,DT_FINAL_LCTO
        ,VL_LOTE
ORDER BY CD_LOTE
