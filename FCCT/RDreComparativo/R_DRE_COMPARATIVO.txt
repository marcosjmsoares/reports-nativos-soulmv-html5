--R_DRE_COMPARATIVO
SELECT * FROM (
select d.cd_dre
     , d.ds_dre
     , t.cd_sequencia
     , t.cd_nivel
     , s.ds_nivel
     , t.cd_seq_nivel
     , s.sinal
     , t.valor
     , s.vl_lancamento
     , t.dt_saldo_mes
from   dbamv.dre_seq_nivel_temp t
          , dbamv.dre_seq_nivel s
          , dbamv.dre d
where  d.cd_dre = s.cd_dre
and    t.cd_sequencia = s.cd_sequencia
and    t.cd_nivel     = s.cd_nivel
and    t.cd_seq_nivel = s.cd_seq_nivel
and    s.sn_mostra_dre = 'S'
and    d.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
and    ( ( ( t.dt_saldo_mes between :CAMPO_DT_INICIAL and :CAMPO_DT_FINAL )  OR  t.dt_saldo_mes is null ) )
order by s.cd_sequencia, s.cd_nivel, s.cd_seq_nivel, d.cd_dre , t.dt_saldo_mes
)
ORDER BY CD_SEQUENCIA,CD_SEQ_NIVEL,CD_NIVEL,DS_NIVEL,SINAL,
DT_SALDO_MES
