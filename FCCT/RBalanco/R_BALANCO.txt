--R_BALANCO
SELECT DBAMV.MASCARA_CONTABIL(CD_CONTABIL,CD_MULTI_EMPRESA, 1 )     NIVEL1
       ,DBAMV.MASCARA_CONTABIL(CD_CONTABIL,CD_MULTI_EMPRESA, 2 )     NIVEL2
       ,DBAMV.MASCARA_CONTABIL(CD_CONTABIL,CD_MULTI_EMPRESA, 3 )     NIVEL3
       ,DBAMV.MASCARA_CONTABIL(CD_CONTABIL,CD_MULTI_EMPRESA, 4 )     NIVEL4
       ,DBAMV.MASCARA_CONTABIL(CD_CONTABIL,CD_MULTI_EMPRESA, 5 )     NIVEL5
       ,DBAMV.MASCARA_CONTABIL(CD_CONTABIL,CD_MULTI_EMPRESA, 6 )     NIVEL6
       ,DBAMV.MASCARA_CONTABIL(CD_CONTABIL,CD_MULTI_EMPRESA, 7 )     NIVEL7
       , dsp
       , vl_saldo_mes
       , GRAU_DA_CONTA
FROM (
SELECT p.cd_contabil, LPAD(' ',5 * (CD_GRAU_DA_CONTA-1)) || P.DS_CONTA dsp, S.VL_SALDO_MES
, P.CD_GRAU_DA_CONTA GRAU_DA_CONTA, p.cd_multi_empresa
FROM SALDO_MENSAL S
, v_natureza_plano_contas P
WHERE S.CD_REDUZIDO = P.CD_REDUZIDO
AND P.CD_GRAU_DA_CONTA <= :CAMPO_CD_GRAU_CONTA
AND (:CAMPO_SN_CONTA_ZERADA = 'S' OR (:CAMPO_SN_CONTA_ZERADA = 'N' AND S.VL_SALDO_MES <> 0))
AND P.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND P.SN_IMPRIME IN ('S',:CAMPO_SN_IMPRIME)
AND (:CAMPO_PSNCTRLUSUPLCONTAS = 'N' OR
        (:CAMPO_PSNCTRLUSUPLCONTAS = 'S' AND
        EXISTS (SELECT CD_REDUZIDO
                  FROM DBAMV.PLANO_USUARIO_MULTI_EMPRESA PU
                 WHERE PU.CD_REDUZIDO = P.CD_REDUZIDO
                   AND PU.CD_MULTI_EMPRESA  = :CAMPO_CD_MULTI_EMPRESA
                   AND PU.CD_ID_USUARIO     = USER  )))
AND tp_conta_contabil NOT IN ('RECEITA','DESPESA')
ORDER BY cd_contabil
)
