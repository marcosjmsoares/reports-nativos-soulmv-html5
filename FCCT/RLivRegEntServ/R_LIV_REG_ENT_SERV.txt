--R_LIV_REG_ENT_SERV
SELECT SEQ,
  NM_PACIENTE,
  DS_ENDERECO,
  NM_BAIRRO,
  NR_ENDERECO,
  CIDADE,
  FONE_PAC,
  NR_FONE,
  CD_PACIENTE,
  CD_ATENDIMENTO,
  DATA_ATEND,
  NR_CPF,
  CD_PRO_INT,
  DS_PRO_FAT,
  NM_CONVENIO,
  NR_NOTA_FISCAL_NFE,
  dt_emissao,
  dt_emissao,
  VL_TOTAL_NOTA,
  OBSERVACAO,
  CD_MULTI_EMPRESA,
  DS_MULTI_EMPRESA,
  CGC,
  NR_CNES,
  FONE,
  VL_ATENDIMENTO
  FROM DBAMV.VDIC_LIVRO_REGISTRO
WHERE data_atend between :CAMPO_DATA_INICIAL AND :CAMPO_DATA_FINAL + 0.99999
AND (
CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
OR (:CAMPO_EMITIR_POR = 'C' AND PKG_TRANSACAO_ENTRE_EMPRESAS.ISFILIAL(CD_MULTI_EMPRESA, :CAMPO_CD_MULTI_EMPRESA) = 1)
)
ORDER BY DATA_ATEND
