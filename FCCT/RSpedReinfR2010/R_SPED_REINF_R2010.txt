--R_SPED_REINF_R2010
SELECT servicoTomado.CD_REINF_SERVICO_TOMADO,
       servicoTomado.CD_REINF_EVENTOS,
       To_Char(servicoTomado.CD_MULTI_EMPRESA_EST_TOMADOR||' - '||empresa.DS_MULTI_EMPRESA) DS_MULTI_EMPRESA_EST_TOMADOR,
       servicoTomado.CD_NUM_INSCR_ESTABELECIMENTO,
       servicoTomado.CD_TIP_INSCR_ESTABELECIMENTO,
       servicoTomado.CD_IND_OBRA,
       servicoTomado.TP_ORIGEM_INFORMACAO,
       servicoTomado.CD_CNPJ_PRESTADOR,
       servicoTomado.NR_IND_CPRB,
       servicoTomado.VL_TOTAL_BRUTO_NF,
       servicoTomado.VL_TOTAL_BASE_RETENCAO,
       servicoTomado.VL_TOTAL_RET_PRINCIPAL,
       servicoTomado.VL_TOTAL_RET_ADICIONAL,
       servicoTomado.VL_TOTAL_NAO_RET_PRINCIPAL,
       servicoTomado.VL_TOTAL_NAO_RET_ADICIONAL,
       To_Char(empresa.CD_MULTI_EMPRESA||' - '||empresa.DS_MULTI_EMPRESA) DS_MULTI_EMPRESA,
       To_Char(fornecedores.CD_FORNECEDOR||' - '||fornecedores.NM_FORNECEDOR) NM_FORNECEDOR,
       eventos.DT_COMPETENCIA,
       Decode (eventos.TP_STATUS,
                'P', 'Pendente',
                'V', 'Validado',
                'A', 'Advertência',
                'C', 'Crítica',
                'T', 'Transmitido') TP_STATUS,
       Decode (servicoTomado.CD_IND_OBRA,
                0, 'Não é obra de construção civil ou não está sujeita a matrícula de obra',
                1, 'Obra de Construção Civil - Empreitada Total',
              2, 'Obra de Construção Civil - Empreitada Parcial') NM_IND_OBRA,
       retorno.DS_PROTOCOLO recibo
  FROM dbamv.REINF_SERVICO_TOMADO servicoTomado,
       dbamv.MULTI_EMPRESAS empresa,
       dbamv.FORNECEDOR fornecedores,
       dbamv.REINF_EVENTOS eventos,
       dbamv.REINF_TRANS_RETORNO retorno
 WHERE servicoTomado.CD_MULTI_EMPRESA_EST_TOMADOR = empresa.CD_MULTI_EMPRESA
   AND servicoTomado.CD_FORNECEDOR = fornecedores.CD_FORNECEDOR
   AND servicoTomado.CD_REINF_EVENTOS = eventos.CD_REINF_EVENTOS
   AND eventos.CD_REINF_EVENTOS = retorno.CD_REINF_EVENTOS(+)
   AND servicoTomado.CD_REINF_SERVICO_TOMADO = retorno.CD_REINF_REGISTRO
   AND eventos.CD_EVENTO = 'R-2010'
   AND empresa.CD_MULTI_EMPRESA = :CAMPO_MULTI_EMPRESA
   AND eventos.DT_COMPETENCIA = to_date('','mm/yyyy')
   AND retorno.cd_reinf_trans_retorno IN
        (SELECT Max(cd_reinf_trans_retorno)
                FROM dbamv.reinf_trans_retorno
                WHERE cd_reinf_eventos = eventos.CD_REINF_EVENTOS
                  AND cd_reinf_registro = servicoTomado.CD_REINF_SERVICO_TOMADO)
UNION ALL
SELECT servicoTomado.CD_REINF_SERVICO_TOMADO,
       servicoTomado.CD_REINF_EVENTOS,
       To_Char(servicoTomado.CD_MULTI_EMPRESA_EST_TOMADOR||' - '||empresa.DS_MULTI_EMPRESA) DS_MULTI_EMPRESA_EST_TOMADOR,
       servicoTomado.CD_NUM_INSCR_ESTABELECIMENTO,
       servicoTomado.CD_TIP_INSCR_ESTABELECIMENTO,
       servicoTomado.CD_IND_OBRA,
       servicoTomado.TP_ORIGEM_INFORMACAO,
       servicoTomado.CD_CNPJ_PRESTADOR,
       servicoTomado.NR_IND_CPRB,
       servicoTomado.VL_TOTAL_BRUTO_NF,
       servicoTomado.VL_TOTAL_BASE_RETENCAO,
       servicoTomado.VL_TOTAL_RET_PRINCIPAL,
       servicoTomado.VL_TOTAL_RET_ADICIONAL,
       servicoTomado.VL_TOTAL_NAO_RET_PRINCIPAL,
       servicoTomado.VL_TOTAL_NAO_RET_ADICIONAL,
       To_Char(empresa.CD_MULTI_EMPRESA||' - '||empresa.DS_MULTI_EMPRESA) DS_MULTI_EMPRESA,
       To_Char(fornecedores.CD_FORNECEDOR||' - '||fornecedores.NM_FORNECEDOR) NM_FORNECEDOR,
       eventos.DT_COMPETENCIA,
       Decode (eventos.TP_STATUS,
                'P', 'Pendente',
                'V', 'Validado',
                'A', 'Advertência',
                'C', 'Crítica',
                'T', 'Transmitido') TP_STATUS,
       Decode (servicoTomado.CD_IND_OBRA,
                0, 'Não é obra de construção civil ou não está sujeita a matrícula de obra',
                1, 'Obra de Construção Civil - Empreitada Total',
              2, 'Obra de Construção Civil - Empreitada Parcial') NM_IND_OBRA,
       retorno.DS_PROTOCOLO recibo
  FROM dbamv.REINF_SERVICO_TOMADO servicoTomado,
       dbamv.MULTI_EMPRESAS empresa,
       dbamv.FORNECEDOR fornecedores,
       dbamv.REINF_EVENTOS eventos,
       dbamv.REINF_TRANS_RETORNO retorno
 WHERE servicoTomado.CD_MULTI_EMPRESA_EST_TOMADOR = empresa.CD_MULTI_EMPRESA
   AND servicoTomado.CD_FORNECEDOR = fornecedores.CD_FORNECEDOR
   AND servicoTomado.CD_REINF_EVENTOS = eventos.CD_REINF_EVENTOS
   AND eventos.CD_REINF_EVENTOS = retorno.CD_REINF_EVENTOS(+)
   AND servicoTomado.CD_REINF_SERVICO_TOMADO = retorno.CD_REINF_REGISTRO
   AND eventos.CD_EVENTO = 'R-2010'
   AND pkg_transacao_entre_empresas.Isfilial(empresa.cd_multi_empresa, :CAMPO_MULTI_EMPRESA) = 1 AND '' = 'S'
   AND eventos.DT_COMPETENCIA = to_date('','mm/yyyy')
   AND retorno.cd_reinf_trans_retorno IN
        (SELECT Max(cd_reinf_trans_retorno)
                FROM dbamv.reinf_trans_retorno
                WHERE cd_reinf_eventos = eventos.CD_REINF_EVENTOS
                  AND cd_reinf_registro = servicoTomado.CD_REINF_SERVICO_TOMADO)
