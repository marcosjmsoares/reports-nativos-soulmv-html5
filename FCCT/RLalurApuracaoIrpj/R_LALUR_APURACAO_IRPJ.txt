--R_LALUR_APURACAO_IRPJ
SELECT * FROM (
select  to_char(dig_acres.dt_referencia, 'mm/yyyy')  dt_referencia_apu
       ,dig_acres.vl_lucro vl_lucro
       ,dig_acres.vl_acrescimo_csll
       ,dig_acres.vl_perc_aliquota_csll
       ,dig_acres.vl_perc_aliquota_irpj
       ,dig_acres.cd_multi_empresa
from dbamv.digitacao_lucro_acresc_lalur dig_acres
where to_char(dig_acres.dt_referencia, 'mm/yyyy')  = :CAMPO_DT_FINAL
:CAMPO_CF_COND_EMPRESA
)
ORDER BY CD_MULTI_EMPRESA
