SELECT * FROM (
SELECT    A.CD_CONTABIL           CD_CONTABIL
         ,A.DS_CONTA              DS_CONTA
         ,A.CD_GRAU               CD_GRAU
        ,DECODE( :CAMPO_TP_PERIODO
                ,'D', TRUNC(DATA_MOV)
                ,'S', TRUNC(DATA_MOV, 'D' )
                ,'M', TRUNC(DATA_MOV, 'MM' )
                ,'A', TRUNC(DATA_MOV, 'YYYY' )) DT_MOV
         ,A.TP_CONTA              TP_CONTA
         ,A.VL_SALDO_ANT  VL_SALDO_ANTERIOR
         ,A.CD_REDUZIDO    CD_REDUZIDO
         ,SUM(NVL( A.VL_DEBITO, 0) )     VL_DEBITO
         ,SUM(NVL( A.VL_CREDITO, 0) )    VL_CREDITO

FROM (
SELECT pc.cd_contabil cd_contabil
       ,pc.ds_conta ds_conta
       ,pc.cd_grau_da_conta cd_grau
       ,pc.tp_conta tp_conta
       , 0 vl_saldo_ant
       ,decode(:CAMPO_TP_PERIODO
               ,'D', trunc( lc.dt_lcto )
               ,'S', trunc( lc.dt_lcto, 'd' )
               ,'M', trunc( lc.dt_lcto, 'mm' )
               ,'A', trunc( lc.dt_lcto, 'yyyy' )) data_mov
       ,pc.cd_reduzido  cd_reduzido
       ,DECODE(PC.TP_CONTA,'A',( lc.vl_lancado ),
                                                             (SELECT SUM(NVL(LC.VL_LANCADO,0))
                                                                 FROM DBAMV.LCTO_CONTABIL LC
                                                              WHERE LC.CD_REDUZIDO_DEBITO IN (SELECT PL1.CD_REDUZIDO
                                                                                                                                     FROM DBAMV.PLANO_CONTAS PL1
                                                                                                                                  WHERE PL1.cd_grau_da_conta > (SELECT PL2.CD_GRAU_DA_CONTA
                                                                                                                                                                                              FROM DBAMV.PLANO_CONTAS PL2
                                                                                                                                                                                          WHERE PL2.cd_reduzido = PC.CD_REDUZIDO)
						                                        and PL1.cd_contabil like PC.CD_CONTABIL || '%' )
                                                                 AND LC.DT_LCTO >= :CAMPO_DT_INI
                                                                 AND LC.DT_LCTO <= :CAMPO_DT_FIM
                                                                 )) vl_debito
               ,to_number(NULL) vl_credito
               ,0 vl_saldo_atu
               ,lc.dt_lcto dt_saldo_mes
   FROM dbamv.plano_contas pc, dbamv.lcto_contabil lc
 WHERE pc.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
  AND pc.cd_reduzido  =   lc.cd_reduzido_debito(+)
   AND LC.DT_LCTO(+) >= :CAMPO_DT_INI
   AND LC.DT_LCTO(+) <= :CAMPO_DT_FIM
   AND pc.cd_grau_da_conta <= :CAMPO_CD_GRAU_CONTA
   AND pc.sn_imprime IN ('S',:CAMPO_SN_IMPRIME)
   AND (:CAMPO_PSNCTRLUSUPLCONTAS = 'N' or :CAMPO_PSNCTRLUSUPLCONTAS = 'S'
   AND EXISTS (SELECT CD_REDUZIDO
	               FROM DBAMV.PLANO_USUARIO_MULTI_EMPRESA PU
   	            WHERE PU.CD_REDUZIDO       = PC.CD_REDUZIDO
	                AND PU.CD_MULTI_EMPRESA  = :CAMPO_CD_MULTI_EMPRESA
                  AND PU.CD_ID_USUARIO     = USER  ))
   AND :CAMPO_TP_ORIGEM  = 'L'
   AND :CAMPO_TP_PERIODO = 'A'
UNION ALL

                                                                                                                                                                                         FROM DBAMV.PLANO_CONTAS PL2

                                                            AND LC.DT_LCTO <= :CAMPO_DT_FIM ))  vl_credito

                                                                                                                                                                                         FROM DBAMV.PLANO_CONTAS PL2
                                                                                                                                                                                     WHERE PL2.cd_reduzido = PC.CD_REDUZIDO)
                                                                                                                                    and PL1.cd_contabil like PC.CD_CONTABIL || '%' )
                                                            AND LC.DT_LCTO >= :CAMPO_DT_INI
                                                            AND LC.DT_LCTO <= :CAMPO_DT_FIM ))  vl_credito
               ,0 vl_saldo_atu
   FROM dbamv.plano_contas pc, dbamv.lcto_contabil lc
 WHERE pc.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
   AND lc.cd_reduzido_credito(+) = pc.cd_reduzido
   AND LC.DT_LCTO(+) >= :CAMPO_DT_INI
   AND LC.DT_LCTO(+) <= :CAMPO_DT_FIM
   AND pc.cd_grau_da_conta <= :CAMPO_CD_GRAU_CONTA
   AND pc.sn_imprime IN ('S',:CAMPO_SN_IMPRIME)
   AND (:CAMPO_PSNCTRLUSUPLCONTAS = 'N' or :CAMPO_PSNCTRLUSUPLCONTAS = 'S'
   AND EXISTS (SELECT CD_REDUZIDO
	               FROM DBAMV.PLANO_USUARIO_MULTI_EMPRESA PU
   	            WHERE PU.CD_REDUZIDO       = PC.CD_REDUZIDO
	                AND PU.CD_MULTI_EMPRESA  = :CAMPO_CD_MULTI_EMPRESA
                  AND PU.CD_ID_USUARIO     = USER  ))
   AND :CAMPO_TP_ORIGEM  = 'L'
   AND :CAMPO_TP_PERIODO = 'D'
:CAMPO_CF_VISAO
) D
GROUP BY  D.CD_CONTABIL
         ,D.DS_CONTA
         ,D.CD_GRAU
         ,D.TP_CONTA
          ,D.DATA_MOV
         ,TO_CHAR(D.DT_SALDO_MES, 'DDMMYYYY')
         ,D.CD_REDUZIDO
         ,D.VL_SALDO_ANT
ORDER BY  CD_CONTABIL
)
ORDER BY CD_CONTABIL,DS_CONTA,
DT_MOV
