--R_BALANCETE_SET_SC
SELECT
            cd_contabil
            ,cd_setor
            ,nm_setor
            ,ds_conta
            ,cd_grau
            ,tp_conta
            ,Min(tp_conta_contabil) tp_conta_contabil
            ,Sum(vl_saldo_ant) vl_saldo_ant
            ,Sum(vl_debito) vl_debito
            ,Sum(vl_credito) vl_credito
            ,Sum(vl_saldo_atu) vl_saldo_atu
            ,Min(tp_natureza) tp_natureza
            ,Min(tp_natureza_saldo_ant) tp_natureza_saldo_ant
	        ,Min(tp_natureza_saldo_atu) tp_natureza_saldo_atu
        FROM
        (
            SELECT
              cd_contabil
              ,cd_setor
              ,nm_setor
              ,ds_conta
              ,cd_grau
              ,tp_conta
              ,tp_conta_contabil
              ,vl_saldo_ant
              ,vl_debito
              ,vl_credito
              ,vl_saldo_atu
              ,tp_natureza
              ,tp_natureza_saldo_ant
	          ,tp_natureza_saldo_atu
            FROM
              DBAMV.DADOS_BALANCETE_SETOR
            WHERE
              dbamv.PKG_FCCT_BALANCETE_SETOR.FNC_CONTAR_GRAU(cd_contabil) <= :CAMPO_CD_GRAU_CONTA
            AND
              cd_usuario = :CAMPO_CD_USUARIO
        )
        GROUP BY
          cd_contabil
          ,cd_setor
          ,nm_setor
          ,ds_conta
          ,cd_grau
          ,tp_conta
        ORDER BY
          cd_setor,
          cd_contabil
