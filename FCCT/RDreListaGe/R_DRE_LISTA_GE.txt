--R_DRE_LISTA_GE
SELECT * FROM (
select d.cd_dre
     , d.ds_dre
     , s.cd_sequencia
     , s.cd_nivel
     , s.ds_nivel
     , s.cd_seq_nivel
     , s.ds_sinal sinal
     , decode(s.sn_mostra_dre,'S','SIM','NÃO') sn_mostra_dre
     , s.cd_reduzido
     , p.cd_contabil
     , p.ds_conta
from   dbamv.dre_seq_nivel_gerencial s
     , dbamv.dre_gerencial d
     , dbamv.plano_contas_gerencial p
where  d.cd_dre = s.cd_dre
and    s.cd_reduzido = p.cd_reduzido (+)
and    d.dt_ano = p.dt_ano
:CAMPO_CF_DRE
)
ORDER BY CD_DRE,DS_DRE
