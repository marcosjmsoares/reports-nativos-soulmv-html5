--R_CCSS_QIV_MANUT
SELECT cd_ccss,cd_ccss||' - '||nm_ccss ds_plano
   FROM dbamv.ffch_ccss
WHERE ffch_ccss.cd_multi_empresa =PKG_MV2000.LE_EMPRESA


/*----------------------------------------------------------------------------*/
--R_CCSS_QIV_MANUT_R_1_Q_SETOR_MANU
SELECT * FROM (
SELECT   dt_comp dt_comp_manut
        ,cd_ccss
        ,SUM ( vl_item_res_manut ) vl_item_res_manut
        ,SUM ( vl_set_man ) vl_set_man
    FROM ( SELECT  to_char ( meses.dt_comp, 'mm/yyyy' ) dt_comp
                  ,ffch_setor_ccss.cd_ccss
                  ,0 vl_item_res_manut
                  ,SUM ( nvl ( custo_final.vl_custo_primario, 0 )) vl_set_man
              FROM dbamv.setor
                  ,dbamv.custo_final
                  ,dbamv.ffch_setor_ccss
                  , ( SELECT add_months ( TO_DATE ( :CAMPO_DT_INICIAL, 'dd/mm/yyyy' ), rownum - 1 ) dt_comp
                       FROM dbamv.setor
                      WHERE rownum <= 3 ) meses
             WHERE setor.cd_setor = custo_final.cd_setor
               AND setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
               AND to_char ( custo_final.dt_competencia, 'mm/yyyy' ) = to_char ( meses.dt_comp, 'mm/yyyy' )
               AND ffch_setor_ccss.cd_setor = setor.cd_setor
               AND setor.cd_multi_empresa = pkg_mv2000.le_empresa
          GROUP BY ffch_setor_ccss.cd_ccss
                            ,meses.dt_comp
          UNION
          SELECT   to_char ( meses.dt_comp, 'mm/yyyy' ) dt_comp
                  ,ffch_setor_ccss.cd_ccss
                  ,SUM ( nvl ( custo_primario.vl_custo_setor, 0 )) vl_item_res_manut
                  ,0 vl_set_man
              FROM dbamv.custo_primario
                  ,dbamv.setor
                  ,dbamv.ffch_setor_ccss
                  , ( SELECT add_months ( TO_DATE ( :CAMPO_DT_INICIAL, 'dd/mm/yyyy' ), rownum - 1 ) dt_comp
                       FROM dbamv.setor
                      WHERE rownum <= 3 ) meses
             WHERE custo_primario.cd_setor = setor.cd_setor
               AND setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
               AND to_char ( custo_primario.dt_competencia, 'mm/yyyy' ) = to_char ( meses.dt_comp, 'mm/yyyy' )
               AND ffch_setor_ccss.cd_setor = setor.cd_setor
               AND setor.cd_multi_empresa = pkg_mv2000.le_empresa
          GROUP BY ffch_setor_ccss.cd_ccss, meses.dt_comp )
GROUP BY cd_ccss,dt_comp
ORDER BY dt_comp
)
WHERE CD_CCSS = :CAMPO_CD_CCSS
