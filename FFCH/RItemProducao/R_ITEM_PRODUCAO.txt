--R_ITEM_PRODUCAO
SELECT  EMPRE.DS_MULTI_EMPRESA,
       ITEM_PRODUCAO.CD_ITEM_PRODUCAO,
       ITEM_PRODUCAO.DS_ITEM_PRODUCAO,
       ITEM_PRODUCAO.FATOR,
       TO_CHAR(ITEM_PRODUCAO.TP_DIRECIONADOR) TP_DIRECIONADOR,
       ROTINAS_CUSTO.DS_ROTINA,
       NULL
FROM DBAMV.ITEM_PRODUCAO,
     DBAMV.FICHA_ITEM,
     DBAMV.UNI_PRO,
     DBAMV.PRODUTO,
     DBAMV.ROTINAS_CUSTO,
     DBAMV.MULTI_EMPRESAS EMPRE
WHERE FICHA_ITEM.CD_ITEM_PRODUCAO(+) = ITEM_PRODUCAO.CD_ITEM_PRODUCAO
  AND FICHA_ITEM.CD_PRODUTO = PRODUTO.CD_PRODUTO(+)
  AND FICHA_ITEM.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO(+)
  AND ITEM_PRODUCAO.CD_ROTINA = ROTINAS_CUSTO.CD_ROTINA(+)
  AND EMPRE.CD_MULTI_EMPRESA IN (:CAMPO_CF_COND_EMPRESA)
