--R_LOG_CALC_CUSTO
SELECT   lcc.DT_GERACAO
        ,lcc.DT_COMPETENCIA
        ,lcc.NM_USUARIO
	,lcc.DS_MACHINE
	,lcc.NM_OSUSER
	,Decode(lcc.TP_CALCULO
			,'AR',:CAMPO_TRAD_AR
			,'PR',:CAMPO_TRAD_PR
			,'RT',:CAMPO_TRAD_RT
			,'RE',:CAMPO_TRAD_RE
			,'CN',:CAMPO_TRAD_CN
			,'CP',:CAMPO_TRAD_CP
			,'AC',:CAMPO_TRAD_AC
			,'FC',:CAMPO_TRAD_FC
			,'TA',:CAMPO_TRAD_TA
			,'GA',:CAMPO_TRAD_GA
			,'R1',:CAMPO_TRAD_R1
			,'IA',:CAMPO_TRAD_IA
			,'IX',:CAMPO_TRAD_IX
			,'CX',:CAMPO_TRAD_CX
			,'GH',:CAMPO_TRAD_GH
		) TP_CALCULO
FROM  dbamv.log_calc_custo lcc
WHERE  lcc.CD_LOG = :CAMPO_CD_LOG
