--R_RATEIO_SETOR
select nr_ordem_grupo,
       grupo_de_custo.ds_grupo_de_custo  ds_grupo_de_custo,
       grupo_de_custo.cd_grupo_de_custo  cd_grupo_de_custo,
       grupo_de_custo.sn_rateio_custo_final,              to_char(dt_competencia,'mm/yyyy') dt_comp,
       setor_origem.cd_setor origem,
       Initcap(setor_origem.nm_setor) nm_origem,
       setor_destino.cd_setor destino,
       setor_destino.nm_setor nm_destino,
       sum(nvl(vl_custo_absorvido,0) - nvl(vl_custo_rateado,0)) valor,
       nvl(setor_destino.nr_ordem_rateio,999999) nr_ordem_rateio_dest,
       nvl(setor_origem.nr_ordem_rateio,999999) nr_ordem_rateio_orig
from   dbamv.custo_absorvido,
       dbamv.setor setor_destino,
       dbamv.setor setor_origem,
       dbamv.grupo_de_custo
where custo_absorvido.cd_setor_origem  = setor_origem.cd_setor and
      custo_absorvido.cd_setor_destino = setor_destino.cd_setor and
      setor_destino.cd_grupo_de_custo = grupo_de_custo.cd_grupo_de_custo and
      to_char(dt_competencia,'mm/yyyy')= :CAMPO_DT_INI
and    setor_destino.cd_multi_empresa   = :CAMPO_CD_MULTI_EMPRESA and
      setor_origem.cd_multi_empresa    = :CAMPO_CD_MULTI_EMPRESA and
      custo_absorvido.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA and
      grupo_de_custo.tp_grupo_de_custo in ('P','T','O')
group by grupo_de_custo.ds_grupo_de_custo,
          grupo_de_custo.cd_grupo_de_custo,                            to_char(dt_competencia,'mm/yyyy') ,
         setor_origem.cd_setor ,
         setor_origem.nm_setor ,
         setor_destino.cd_setor ,
         setor_destino.nm_setor ,
         setor_destino.nr_ordem_rateio,
         setor_origem.nr_ordem_rateio,
         grupo_de_custo.sn_rateio_custo_final,
         nr_ordem_grupo
