--R_ANALISE_CONTA_CUSTO
SELECT * FROM (
select To_Char(item_res.cd_item_res||' - '||ds_item_res) ds_item_res,
       nm_setor,
       trunc(dt_competencia) dt_comp,
       to_char(dt_competencia,'yyyy/mm') dt_comp_string,
       decode('','A', item_res.ds_item_res,to_char(item_res.cd_item_res,'0000')) nr_ordem
from   dbamv.custo_primario,
       dbamv.setor,
       dbamv.item_res
where  dbamv.custo_primario.cd_setor = dbamv.setor.cd_setor and
            dbamv.custo_primario.cd_item_res = dbamv.item_res.cd_item_res and
            dt_competencia >= to_date('','yyyy/mm') and
            dt_competencia <= to_date('','yyyy/mm') and
            setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
group by To_Char(item_res.cd_item_res||' - '||ds_item_res), nm_setor,
         to_char(dt_competencia,'yyyy/mm'), trunc(dt_competencia),
         decode('','A', item_res.ds_item_res,to_char(item_res.cd_item_res,'0000'))
)
ORDER BY nr_ordem
