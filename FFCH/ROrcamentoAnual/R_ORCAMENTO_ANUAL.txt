--R_ORCAMENTO_ANUAL_RESUMO_CT
SELECT cd_setor_origem cd_setor_origem_resum,
       dt_comp dt_comp_resum,
       Round(Sum(soma_vl_realiz),2) soma_vl_realiz_resum,
       Round(Sum(soma_vl_orcado),2) soma_vl_orcado_resum,
       Round(Sum(resul_vl_real),2) resul_vl_real_resum,
       Round(Sum(resul_vl_orca), 2) resul_vl_orca_resum
FROM (
SELECT cd_setor_origem
, dt_comp
, soma_vl_realiz
, soma_vl_orcado
, resul_vl_real
, resul_vl_orca
FROM
  (SELECT Base.cd_setor cd_setor_origem ,
          TO_CHAR (TO_DATE (dt_comp, 'mm/yyyy'), 'mm/yyyy') dt_comp ,
          SUM(decode(nsecao,'Receitas',0,vl_realizado)) soma_vl_realiz ,
          SUM(decode(nsecao,'Receitas',0,vl_orcado)) soma_vl_orcado ,
          SUM(decode(nsecao,'Receitas',vl_realizado,-1*vl_realizado)) resul_vl_real ,
          SUM(decode(nsecao,'Receitas',vl_orcado,-1*vl_orcado)) resul_vl_orca
   FROM
     ( SELECT temp_setor.cd_setor cd_setor ,
              'Receitas' nsecao ,
              TO_CHAR (receita_setor.dt_competencia, 'mm/yyyy') dt_comp ,
              SUM (receita_setor.vl_receita_setor) vl_realizado ,
              0 vl_orcado
      FROM dbamv.setor ,
           dbamv.item_res ,
           dbamv.temp_Setor ,
           dbamv.receita_setor
      WHERE receita_setor.dt_competencia >= TO_DATE (:CAMPO_DATA_INI, 'mm/yyyy')
        AND receita_setor.dt_competencia <= TO_DATE (:CAMPO_DATA_FIM, 'mm/yyyy')
        AND NVL (receita_setor.vl_receita_setor, 0) <> 0
        AND NVL (receita_setor.tp_origem_faturamento, 'NO') IN ('CH','RA','FA','FS','NO')
        AND receita_setor.cd_setor = temp_setor.cd_setor_filho
        AND Temp_setor.cd_setor_filho = setor.cd_setor
        AND setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
        AND receita_setor.cd_item_res = item_res.cd_item_res
        AND receita_setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
        AND temp_setor.cd_setor = :CAMPO_CD_SETOR_ORIGEM
      GROUP BY Temp_Setor.cd_setor ,
               TO_CHAR (receita_setor.dt_competencia, 'mm/yyyy')
      UNION ALL SELECT Temp_Setor.cd_setor cd_setor ,
                       'Receitas' nsecao ,
                       TO_CHAR (compt.dt_competencia, 'mm/yyyy') dt_comp ,
                       0 vl_realizado ,
                       SUM (item_orcamento.vl_orcamento) vl_orcado
      FROM dbamv.setor ,dbamv.item_res ,dbamv.Temp_setor ,dbamv.item_orcamento ,
        (SELECT competencia.dt_competencia dt_competencia
         FROM dbamv.competencia
         WHERE competencia.dt_competencia BETWEEN TO_DATE (:CAMPO_DATA_INI, 'mm/yyyy') AND TO_DATE (:CAMPO_DATA_FIM, 'mm/yyyy') AND competencia.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA) compt
      WHERE NVL (item_orcamento.vl_orcamento, 0) <> 0
        AND NVL (item_orcamento.tp_meta, 'NO') IN ('CH','RA','FA','FS','NO')
        AND item_orcamento.tp_orcamento = 'I'
        AND setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
        AND item_orcamento.cd_setor_origem = Temp_setor.cd_setor_filho
        AND Temp_setor.cd_setor_filho = setor.cd_setor
        AND item_orcamento.cd_item_res = item_res.cd_item_res
        AND item_orcamento.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
        AND item_orcamento.dt_competencia =
          (SELECT MAX(io.dt_competencia) DATA
           FROM dbamv.item_orcamento io
           WHERE io.dt_competencia <= compt.dt_competencia
             AND NVL (io.cd_setor_origem, item_orcamento.cd_setor_origem) = item_orcamento.cd_setor_origem
             AND NVL (io.cd_item_res, item_orcamento.cd_item_res) = item_orcamento.cd_item_res
             AND NVL (io.tp_meta, item_orcamento.tp_meta) = item_orcamento.tp_meta
             AND NVL (io.cd_multi_empresa, item_orcamento.cd_multi_empresa) = item_orcamento.cd_multi_empresa)
        AND temp_setor.cd_setor = :CAMPO_CD_SETOR_ORIGEM
      GROUP BY temp_Setor.cd_setor,
               TO_CHAR (compt.dt_competencia, 'mm/yyyy')
      UNION ALL SELECT Temp_Setor.cd_setor cd_setor ,
                       'Custos' nsecao ,
                       TO_CHAR (dt_competencia, 'mm/yyyy') dt_comp ,
                       SUM (custo_primario.vl_custo_setor) vl_realizado ,
                       0 vl_orcado
      FROM dbamv.item_res ,
           dbamv.custo_primario ,
           dbamv.setor ,
           dbamv.temp_setor
      WHERE dt_competencia >= TO_DATE (:CAMPO_DATA_INI, 'mm/yyyy')
        AND dt_competencia <= TO_DATE (:CAMPO_DATA_FIM, 'mm/yyyy')
        AND custo_primario.cd_item_res = item_res.cd_item_res
        AND custo_primario.cd_setor = Temp_setor.cd_setor_filho
        AND Temp_setor.cd_setor_filho = Setor.cd_setor
        AND setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
        AND temp_setor.cd_setor = :CAMPO_CD_SETOR_ORIGEM
      GROUP BY Temp_Setor.cd_setor ,
               TO_CHAR (dt_competencia, 'mm/yyyy')
      UNION ALL SELECT Temp_Setor.cd_setor cd_setor ,
                       'Custos' nsecao ,
                       TO_CHAR (compt.dt_competencia, 'mm/yyyy') dt_comp ,
                       0 vl_realizado ,
                       SUM (item_orcamento.vl_orcamento) vl_orcado
      FROM dbamv.setor ,dbamv.Temp_Setor ,dbamv.item_res ,dbamv.item_orcamento ,
        (SELECT competencia.dt_competencia dt_competencia
         FROM dbamv.competencia
         WHERE competencia.dt_competencia BETWEEN TO_DATE (:CAMPO_DATA_INI, 'mm/yyyy') AND TO_DATE (:CAMPO_DATA_FIM, 'mm/yyyy') AND competencia.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA) compt
      WHERE NVL (item_orcamento.vl_orcamento, 0) <> 0
        AND item_orcamento.cd_item_res = item_res.cd_item_res
        AND item_orcamento.tp_orcamento = 'I'
        AND item_orcamento.tp_meta IN ('OC')
        AND item_orcamento.cd_setor_origem = Temp_setor.cd_setor_filho
        AND Temp_setor.cd_setor_filho = Setor.cd_setor
        AND setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
        AND item_orcamento.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
        AND item_orcamento.dt_competencia =
          (SELECT MAX (io.dt_competencia) DATA
           FROM dbamv.item_orcamento io
           WHERE io.dt_competencia <= compt.dt_competencia
             AND NVL (io.cd_setor_origem, item_orcamento.cd_setor_origem) = item_orcamento.cd_setor_origem
             AND NVL (io.cd_item_res, item_orcamento.cd_item_res) = item_orcamento.cd_item_res
             AND NVL (io.tp_meta, item_orcamento.tp_meta) = item_orcamento.tp_meta
             AND NVL (io.tp_orcamento, item_orcamento.tp_orcamento) = item_orcamento.tp_orcamento
             AND NVL (io.cd_multi_empresa, item_orcamento.cd_multi_empresa) = item_orcamento.cd_multi_empresa)
        AND temp_setor.cd_setor = :CAMPO_CD_SETOR_ORIGEM
      GROUP BY Temp_Setor.cd_setor ,
               TO_CHAR (compt.dt_competencia, 'mm/yyyy')
      UNION ALL SELECT Temp_Setor.cd_setor cd_setor ,
                       'Rateios' nsecao ,
                       TO_CHAR (custo_absorvido.dt_competencia, 'mm/yyyy') dt_comp ,
                       SUM (custo_absorvido.vl_custo_absorvido) vl_realizado ,
                       0 vl_orcado
      FROM dbamv.custo_absorvido ,
           dbamv.setor destino ,
           dbamv.setor origem ,
           dbamv.temp_setor
      WHERE custo_absorvido.dt_competencia >= TO_DATE (:CAMPO_DATA_INI, 'mm/yyyy')
        AND custo_absorvido.dt_competencia <= TO_DATE (:CAMPO_DATA_FIM, 'mm/yyyy')
        AND custo_absorvido.cd_setor_destino = temp_setor.cd_setor_filho
        AND temp_setor.cd_setor = destino.cd_setor
        AND custo_absorvido.cd_setor_origem = origem.cd_setor
        AND custo_absorvido.cd_setor_origem NOT IN
          (SELECT cd_setor_filho
           FROM dbamv.temp_setor)
        AND origem.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
        AND destino.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
        AND custo_absorvido.vl_custo_absorvido <> 0
        AND temp_setor.cd_setor = :CAMPO_CD_SETOR_ORIGEM
      GROUP BY Temp_Setor.cd_setor ,
               TO_CHAR (custo_absorvido.dt_competencia, 'mm/yyyy')
      UNION ALL SELECT temp_setor.cd_setor cd_setor ,
                       'Rateios' nsecao ,
                       TO_CHAR(compt.dt_competencia, 'mm/yyyy') dt_comp ,
                       0 vl_realizado ,
                       SUM (item_orcamento.vl_orcamento) vl_orcado
      FROM dbamv.setor origem ,dbamv.setor destino ,dbamv.item_orcamento ,dbamv.temp_setor ,
        (SELECT competencia.dt_competencia dt_competencia
         FROM dbamv.competencia
         WHERE competencia.dt_competencia BETWEEN TO_DATE(:CAMPO_DATA_INI, 'mm/yyyy') AND TO_DATE(:CAMPO_DATA_FIM, 'mm/yyyy') AND competencia.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA) compt
      WHERE NVL(item_orcamento.vl_orcamento, 0) <> 0
        AND origem.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
        AND destino.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
        AND item_orcamento.cd_setor_origem = origem.cd_setor
        AND item_orcamento.cd_setor_destino = destino.cd_setor
        AND temp_setor.cd_setor_filho = origem.cd_setor
        AND item_orcamento.cd_setor_destino NOT IN
          (SELECT cd_setor_filho
           FROM dbamv.temp_setor)
        AND item_orcamento.tp_orcamento = 'R'
        AND item_orcamento.dt_competencia =
          (SELECT MAX(io.dt_competencia) DATA
           FROM dbamv.item_orcamento io
           WHERE io.dt_competencia <= compt.dt_competencia
             AND NVL(io.cd_setor_origem, item_orcamento.cd_setor_origem) = item_orcamento.cd_setor_origem
             AND NVL(io.tp_orcamento, item_orcamento.tp_orcamento) = item_orcamento.tp_orcamento
             AND NVL(io.cd_multi_empresa, item_orcamento.cd_multi_empresa) = item_orcamento.cd_multi_empresa)
        AND temp_setor.cd_setor = :CAMPO_CD_SETOR_ORIGEM
      GROUP BY temp_setor.cd_setor ,
               TO_CHAR(compt.dt_competencia, 'mm/yyyy')) Base ,
                                                         dbamv.setor
   WHERE NOT (vl_orcado = 0
              AND vl_realizado = 0)
     AND Base.cd_setor = Setor.cd_setor
   GROUP BY Base.cd_setor ,
            dt_comp
   ORDER BY Base.cd_setor ,
            dt_comp) resultado
)
GROUP BY cd_setor_origem,
         dt_comp
ORDER BY cd_setor_origem,
         dt_comp

                                                               sum(decode(nsecao,'Receitas',vl_orcado,-1*vl_orcado)) resul_vl_orca
   FROM
     ( SELECT '1' Ordenacao

              'Receitas' nsecao ,
                         Decode(:CAMPO_SN_RESUMIR_RECEITAS,'N',DECODE (NVL (receita_setor.tp_origem_faturamento, 'NO') ,'CH', 'Conta Hospitalar' ,'H', 'Conta Hospitalar' ,'RA', 'Remessa Ambulatorial' ,'FA', 'Faturamento AIH' ,'FS', 'Faturamento SUS' ,'NO', 'Não Operacional' ,tp_origem_faturamento ),NULL) ds_grupo ,

                                                                'RA',
                                                                'FA',
                                                                'FS',
                                                                'NO')

        AND setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA

               item_res.ds_item_res ,
               TO_CHAR (receita_setor.dt_competencia, 'mm/yyyy')
      UNION ALL SELECT '1' Ordenacao
              , Temp_Setor.cd_setor cd_setor ,

                                    SUM (item_orcamento.vl_orcamento) vl_orcado
      FROM dbamv.setor ,dbamv.item_res ,dbamv.Temp_setor ,dbamv.item_orcamento ,
        (SELECT competencia.dt_competencia dt_competencia
         FROM dbamv.competencia
         WHERE competencia.dt_competencia BETWEEN TO_DATE (:CAMPO_DATA_INI, 'mm/yyyy') AND TO_DATE (:CAMPO_DATA_FIM, 'mm/yyyy') AND competencia.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA) compt
      WHERE NVL (item_orcamento.vl_orcamento, 0) <> 0
        AND NVL (item_orcamento.tp_meta, 'NO') IN ('CH',
                                                   'RA',
                                                   'FA',
                                                   'FS',
                                                   'NO')
        AND item_orcamento.tp_orcamento = 'I'
        AND setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
        AND item_orcamento.cd_setor_origem = Temp_setor.cd_setor_filho
        AND Temp_setor.cd_setor_filho = setor.cd_setor
        AND item_orcamento.cd_item_res = item_res.cd_item_res
        AND item_orcamento.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
        AND item_orcamento.dt_competencia =
          (SELECT MAX(io.dt_competencia) DATA
           FROM dbamv.item_orcamento io
           WHERE io.dt_competencia <= compt.dt_competencia
             AND NVL (io.cd_setor_origem, item_orcamento.cd_setor_origem) = item_orcamento.cd_setor_origem
             AND NVL (io.cd_item_res, item_orcamento.cd_item_res) = item_orcamento.cd_item_res
             AND NVL (io.tp_meta, item_orcamento.tp_meta) = item_orcamento.tp_meta
             AND NVL (io.cd_multi_empresa, item_orcamento.cd_multi_empresa) = item_orcamento.cd_multi_empresa)
        AND temp_setor.cd_setor = :CAMPO_CD_SETOR_ORIGEM
      GROUP BY temp_Setor.cd_setor ,
               DECODE (NVL (item_orcamento.tp_meta, 'NO'), 'CH', 'Conta Hospitalar', 'RA', 'Remessa Ambulatorial', 'FA', 'Faturamento AIH', 'FS', 'Faturamento SUS', 'NO', 'Não Operacional') ,
               item_res.ds_item_res ,
               TO_CHAR (compt.dt_competencia, 'mm/yyyy')
      UNION ALL SELECT '2' Ordenacao
              , Temp_Setor.cd_setor cd_setor ,
                       'Custos' nsecao ,
                                Decode(:CAMPO_SN_RESUMIR_CUSTO_PRIMARIO,'N', gru_res.ds_gru_res,NULL) ds_grupo ,
                                Decode(:CAMPO_SN_RESUMIR_CUSTO_PRIMARIO,'N', TO_CHAR (item_res.ds_item_res),'Dados Agrupados') ds_item ,
                                TO_CHAR (dt_competencia, 'mm/yyyy') dt_comp ,
                                SUM (custo_primario.vl_custo_setor) vl_realizado ,
                                                                    0 vl_orcado
      FROM dbamv.gru_res ,
           dbamv.item_res ,
           dbamv.custo_primario ,
           dbamv.setor ,
           dbamv.temp_setor
      WHERE dt_competencia >= TO_DATE (:CAMPO_DATA_INI, 'mm/yyyy')
        AND dt_competencia <= TO_DATE (:CAMPO_DATA_FIM, 'mm/yyyy')
        AND item_res.cd_gru_res = gru_res.cd_gru_res
        AND custo_primario.cd_item_res = item_res.cd_item_res
        AND custo_primario.cd_setor = Temp_setor.cd_setor_filho
        AND Temp_setor.cd_setor_filho = Setor.cd_setor
        AND setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
        AND temp_setor.cd_setor = :CAMPO_CD_SETOR_ORIGEM
      GROUP BY Temp_Setor.cd_setor ,
               gru_res.ds_gru_res ,
               item_res.ds_item_res ,
               TO_CHAR (dt_competencia, 'mm/yyyy')
      UNION ALL SELECT '2' Ordenacao
              , Temp_Setor.cd_setor cd_setor ,
                       'Custos' nsecao ,
                                Decode(:CAMPO_SN_RESUMIR_CUSTO_PRIMARIO,'N', gru_res.ds_gru_res,NULL) ds_grupo ,
                                Decode(:CAMPO_SN_RESUMIR_CUSTO_PRIMARIO,'N', TO_CHAR (item_res.ds_item_res),'Dados Agrupados') ds_item ,

      WHERE NVL (item_orcamento.vl_orcamento, 0) <> 0
        AND item_orcamento.cd_item_res = item_res.cd_item_res

        AND item_orcamento.dt_competencia =
          (SELECT MAX (io.dt_competencia) DATA
           FROM dbamv.item_orcamento io
           WHERE io.dt_competencia <= compt.dt_competencia
             AND NVL (io.cd_setor_origem, item_orcamento.cd_setor_origem) = item_orcamento.cd_setor_origem
             AND NVL (io.cd_item_res, item_orcamento.cd_item_res) = item_orcamento.cd_item_res
             AND NVL (io.tp_meta, item_orcamento.tp_meta) = item_orcamento.tp_meta
             AND NVL (io.tp_orcamento, item_orcamento.tp_orcamento) = item_orcamento.tp_orcamento
             AND NVL (io.cd_multi_empresa, item_orcamento.cd_multi_empresa) = item_orcamento.cd_multi_empresa)
        AND temp_setor.cd_setor = :CAMPO_CD_SETOR_ORIGEM
      GROUP BY Temp_Setor.cd_setor ,
               gru_res.ds_gru_res ,
               item_orcamento.cd_item_res ,
               item_res.ds_item_res ,
               TO_CHAR (compt.dt_competencia, 'mm/yyyy')
      UNION ALL SELECT '3' Ordenacao

                                 Decode(:CAMPO_SN_RESUMIR_RATEIOS,'N', DECODE (Origem.classificacao, 'G', 'Genéricos', 'E', 'Específicos', 'Não Informado'),NULL) ds_grupo ,

                                 TO_CHAR (custo_absorvido.dt_competencia, 'mm/yyyy') dt_comp ,
                                 SUM (custo_absorvido.vl_custo_absorvido) vl_realizado ,
                                                                          0 vl_orcado

           dbamv.setor destino ,

           dbamv.temp_setor

        AND custo_absorvido.vl_custo_absorvido <> 0
        AND temp_setor.cd_setor = :CAMPO_CD_SETOR_ORIGEM
      GROUP BY Temp_Setor.cd_setor ,
               DECODE (Origem.classificacao, 'G', 'Genéricos', 'E', 'Específicos', 'Não Informado') ,
               origem.nm_setor ,
               TO_CHAR (custo_absorvido.dt_competencia, 'mm/yyyy')

        (SELECT competencia.dt_competencia dt_competencia
         FROM dbamv.competencia
         WHERE competencia.dt_competencia BETWEEN TO_DATE(:CAMPO_DATA_INI, 'mm/yyyy') AND TO_DATE(:CAMPO_DATA_FIM, 'mm/yyyy') AND competencia.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA) compt
      WHERE NVL(item_orcamento.vl_orcamento, 0) <> 0
        AND origem.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
        AND destino.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
        AND item_orcamento.cd_setor_origem = origem.cd_setor
        AND item_orcamento.cd_setor_destino = destino.cd_setor
        AND temp_setor.cd_setor_filho = origem.cd_setor
        AND item_orcamento.cd_setor_destino NOT IN
          (SELECT cd_setor_filho
           FROM dbamv.temp_setor)
        AND item_orcamento.tp_orcamento = 'R'
        AND item_orcamento.dt_competencia =
          (SELECT MAX(io.dt_competencia) DATA
           FROM dbamv.item_orcamento io
           WHERE io.dt_competencia <= compt.dt_competencia
             AND NVL(io.cd_setor_origem, item_orcamento.cd_setor_origem) = item_orcamento.cd_setor_origem
             AND NVL(io.tp_orcamento, item_orcamento.tp_orcamento) = item_orcamento.tp_orcamento
             AND NVL(io.cd_multi_empresa, item_orcamento.cd_multi_empresa) = item_orcamento.cd_multi_empresa)
        AND temp_setor.cd_setor = :CAMPO_CD_SETOR_ORIGEM
      GROUP BY temp_setor.cd_setor ,
               DECODE(destino.classificacao, 'G', 'Genéricos', 'E', 'Específicos', 'Não Informado') ,
               destino.nm_setor ,
               TO_CHAR(compt.dt_competencia, 'mm/yyyy')) Base ,
                                                         dbamv.setor
   WHERE NOT (vl_orcado = 0
              AND vl_realizado = 0)
     AND Base.cd_setor = Setor.cd_setor
   GROUP BY Ordenacao

            UPPER (setor.nm_setor) ,nsecao ,

                                    dt_comp
   ORDER BY Base.cd_setor ,
            nm_setor ,
            nsecao ,

            ds_item ,
