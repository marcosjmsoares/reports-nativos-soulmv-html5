--R_ITEM_PROCEDIMENTO
SELECT * FROM (
select  ds_gru_pro,
        pro_fat.cd_pro_fat,
        ds_pro_fat,
        item_pro_fat.cd_produto,
        ds_produto,
        uni_pro.ds_unidade,
        qt_padrao,
        vl_tempo
from   dbamv.item_pro_fat,
       dbamv.pro_fat,
       dbamv.uni_pro,
       dbamv.produto,
       dbamv.gru_pro
where  item_pro_fat.cd_pro_fat(+) = pro_fat.cd_pro_fat and
       item_pro_fat.cd_produto = produto.cd_produto(+) and
       item_pro_fat.cd_uni_pro = uni_pro.cd_uni_pro(+) and
       pro_fat.cd_gru_pro = gru_pro.cd_gru_pro(+)
:CAMPO_CF_GRU_PRO
)
ORDER BY DS_PRO_FAT,CD_PRO_FAT,VL_TEMPO,DS_GRU_PRO
