--R_CONTA_CUSTO
SELECT * FROM (
SELECT
            item_res.cd_item_res  cd_item_res,
            item_res.ds_item_res  ds_item_res,
            item_res.tp_custo tp_custo,
            item_res.tp_rateio_con_pag tp_rateio_con_pag,
            gru_res.cd_gru_res   cd_gru_res,
            gru_res.ds_gru_res   ds_gru_res,
            item_res.tp_receita tp_receita,
            item_res.tp_item_res tp_item_res,
            item_res.tp_modelo_custo tp_modelo_custo,
            item_res.sn_importa_estoque sn_importa_estoque,
            item_res.sn_custo_evento sn_custo_evento,
            item_res.tp_analise_desempenho tp_analise_desempenho
FROM
            dbamv.item_res,
            dbamv.gru_res
WHERE
           item_res.cd_gru_res = gru_res.cd_gru_res
:CAMPO_CF_ORDEM
)
