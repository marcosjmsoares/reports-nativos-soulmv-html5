--R_ANA_RESULTADO_ATENDIME
SELECT * FROM (
SELECT   A.ANALISE
        , SUM ( A.RECEITA ) RECEITA
        , SUM ( A.DIARIA ) DIARIA
        , SUM ( A.QTD_DIARIA ) QTD_DIARIA
        , SUM ( A.HONORARIO ) HONORARIO
        , SUM ( A.PROCEDIMENTO ) PROCEDIMENTO
        , SUM ( A.MATMED ) MATMED
        , SUM ( A.TERCEIRO ) TERCEIRO
        , MAX ( A.TX_ADM ) TX_ADM
        , DECODE(:CAMPO_SN_CALCULO_DEDUCAO_RECEITA, 'N', MAX(A.TX_IMPOSTO), 'S', SUM(A.TX_IMPOSTO)) TX_IMPOSTO
        , 'X' QUEBRA
        , COUNT ( A.QTD_ATENDIMENTO ) QTD_ATENDIMENTO
    FROM ( SELECT DECODE (
                    :CAMPO_ANALISE
                   ,'1', NVL ( ATE.CD_PRO_INT, NVL ( ATE.CD_SSM_SIA, ATE.CD_PROCEDIMENTO ))
                   ,'2', ATE.CD_CID
                   ,'3', ATE.CD_CONVENIO
                   ,'4', ATE.CD_SERVICO
                   ,'5', ATE.CD_ESPECIALID
                   ,'6', ATE.CD_PRESTADOR
                   ,'7', ATE.TP_ATENDIMENTO
                   ,'8', ATE.CD_TIPO_INTERNACAO
                   ,'0', ATE.CD_PACIENTE
                   ,'10', SGRU_CID.CD_GRU_CID
                   ,'11', CID.CD_SGRU_CID
                  ) ANALISE
                 , DBAMV.PACK_FFCH.RECEITA(ATE.CD_ATENDIMENTO) RECEITA
                 , DBAMV.PACK_FFCH.DIARIA(ATE.CD_ATENDIMENTO, 'R') DIARIA
                 , DBAMV.PACK_FFCH.QTD_DIARIA(ATE.CD_ATENDIMENTO) QTD_DIARIA
                 , DBAMV.PACK_FFCH.HONORARIO(ATE.CD_ATENDIMENTO ) HONORARIO
                 , DBAMV.PACK_FFCH.PROCEDIMENTO(ATE.CD_ATENDIMENTO, 'R', :CAMPO_CD_MULTI_EMPRESA) PROCEDIMENTO
                 , DBAMV.PACK_FFCH.MATMED(ATE.CD_ATENDIMENTO, ATE.DT_ATENDIMENTO, :CAMPO_CD_MULTI_EMPRESA) MATMED
                 , DBAMV.PACK_FFCH.TERCEIRO(ATE.CD_ATENDIMENTO, 'R') TERCEIRO
                 , DBAMV.PACK_FFCH.TX_ADM(TO_CHAR(ATE.DT_ATENDIMENTO, 'MM/YYYY'), :CAMPO_CD_MULTI_EMPRESA) TX_ADM
                 , DECODE(:CAMPO_SN_CALCULO_DEDUCAO_RECEITA, 'N', DBAMV.PACK_FFCH.TX_IMPOSTO(TO_CHAR(ATE.DT_ATENDIMENTO, 'MM/YYYY'), :CAMPO_CD_MULTI_EMPRESA), 'S', DBAMV.PACK_FFCH.FNC_CUSTO_DEDUCAO_CONTAB_ATEND(ATE.CD_ATENDIMENTO, 'N', 'S', 'N', 'N')) TX_IMPOSTO
                 , 'X' DUMMY
                 , ROWNUM QTD_ATENDIMENTO
            FROM DBAMV.ATENDIME ATE
                ,DBAMV.PACIENTE PC
                ,DBAMV.PRO_FAT PF
                ,DBAMV.TAB_SIASUS SIA
                ,DBAMV.CONVENIO CV
                ,DBAMV.PRESTADOR PR
                ,DBAMV.CID
                ,DBAMV.TIPO_INTERNACAO TP_INT
                ,DBAMV.GRU_CID
                ,DBAMV.SGRU_CID
           WHERE ATE.DT_ATENDIMENTO BETWEEN :CAMPO_DATA_INI AND 
             AND ATE.CD_PACIENTE = PC.CD_PACIENTE
             AND ATE.CD_PRO_INT = PF.CD_PRO_FAT(+)
             AND ATE.CD_SSM_SIA = SIA.CD_SSM(+)
             AND CID.CD_CID(+) = ATE.CD_CID
             AND CID.CD_SGRU_CID = SGRU_CID.CD_SGRU_CID(+)
             AND SGRU_CID.CD_GRU_CID = GRU_CID.CD_GRU_CID(+)
             AND ATE.CD_CONVENIO = CV.CD_CONVENIO
             AND ATE.CD_PRESTADOR = PR.CD_PRESTADOR(+)
             AND ATE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
             AND ATE.CD_TIPO_INTERNACAO = TP_INT.CD_TIPO_INTERNACAO(+)
             AND TRUNC(ATE.DT_ATENDIMENTO, 'MM') >= TO_DATE(:CAMPO_DT_ATENDIMENTO, 'YYYY/MM/DD')
             :CAMPO_CF_CIRURGIA
         ) A
GROUP BY A.ANALISE
)
ORDER BY DECODE(''
     , '1'
     , NVL((NVL(RECEITA,0) - NVL(HONORARIO,0) - NVL(NVL(DIARIA,0) + NVL(PROCEDIMENTO,0) + NVL(MATMED,0) + NVL(TERCEIRO,0),0) - DECODE(:CAMPO_SN_CALCULO_DEDUCAO_RECEITA, 'N', (NVL(RECEITA,0) * (NVL(TX_IMPOSTO,0)/100)), 'S', TX_IMPOSTO) - NVL(NVL(DIARIA,0) + NVL(PROCEDIMENTO,0) + NVL(MATMED,0) + NVL(TERCEIRO,0),0) * (NVL(TX_ADM,0)/100)),0)
     , '2'
     , NVL((NVL(RECEITA,0) - NVL(HONORARIO,0) - NVL(NVL(DIARIA,0) + NVL(PROCEDIMENTO,0) + NVL(MATMED,0) + NVL(TERCEIRO,0),0) - DECODE(:CAMPO_SN_CALCULO_DEDUCAO_RECEITA, 'N', (NVL(RECEITA,0) * (NVL(TX_IMPOSTO,0)/100)), 'S', TX_IMPOSTO) - NVL(NVL(DIARIA,0) + NVL(PROCEDIMENTO,0) + NVL(MATMED,0) + NVL(TERCEIRO,0),0) * (NVL(TX_ADM,0)/100)) / DECODE(QTD_DIARIA,0,1),0)
     , '3'
     , NVL((NVL(RECEITA,0) - NVL(HONORARIO,0) - NVL(NVL(DIARIA,0) + NVL(PROCEDIMENTO,0) + NVL(MATMED,0) + NVL(TERCEIRO,0),0) - DECODE(:CAMPO_SN_CALCULO_DEDUCAO_RECEITA, 'N', (NVL(RECEITA,0) * (NVL(TX_IMPOSTO,0)/100)), 'S', TX_IMPOSTO) - NVL(NVL(DIARIA,0) + NVL(PROCEDIMENTO,0) + NVL(MATMED,0) + NVL(TERCEIRO,0),0) * (NVL(TX_ADM,0)/100)) / DECODE(QTD_ATENDIMENTO,0,1),0)
     , '4'
     , NVL(((NVL(RECEITA,0) - NVL(HONORARIO,0) - NVL(NVL(DIARIA,0) + NVL(PROCEDIMENTO,0) + NVL(MATMED,0) + NVL(TERCEIRO,0),0) - DECODE(:CAMPO_SN_CALCULO_DEDUCAO_RECEITA, 'N', (NVL(RECEITA,0) * (NVL(TX_IMPOSTO,0)/100)), 'S', TX_IMPOSTO) - NVL(NVL(DIARIA,0) + NVL(PROCEDIMENTO,0) + NVL(MATMED,0) + NVL(TERCEIRO,0),0) * (NVL(TX_ADM,0)/100)) / RECEITA )*100,0)
) DESC
