--R_LANCAMENTOS_ERR
SELECT * FROM (
select ratcon_pag.cd_con_pag,
       ratcon_pag.cd_item_res,
       item_res.ds_item_res,
       con_pag.dt_lancamento,
       setor.nm_setor,
       con_pag.ds_fornecedor,
       ratcon_pag.nr_meses_amortizacao,
       ratcon_pag.vl_rateio
from dbamv.ratcon_pag,
     dbamv.item_res,
     dbamv.setor,
     dbamv.con_pag
where ratcon_pag.dt_competencia = to_date(:CAMPO_DTCOMP,'mm/yyyy') and
          dt_lancamento >= nvl(to_date (:CAMPO_DT_INI,'yyyy/mm/dd'),to_date('1980/01/01','yyyy/mm/dd'))and
          dt_lancamento <= nvl(to_date (:CAMPO_DT_FIM,'yyyy/mm/dd'),to_date('2200/12/31','yyyy/mm/dd')) and
          item_res.tp_rateio_con_pag = 'C' and
          ratcon_pag.cd_setor is null and
          ratcon_pag.cd_item_res is not null and
          item_res.cd_item_res = ratcon_pag.cd_item_res and
          setor.cd_setor(+) = ratcon_pag.cd_setor and
          con_pag.cd_con_pag(+) = ratcon_pag.cd_con_pag   and
          setor.cd_multi_empresa(+) = :CAMPO_CD_MULTI_EMPRESA
order by ratcon_pag.cd_item_res,
       item_res.ds_item_res,
       con_pag.dt_lancamento,
       setor.nm_setor,
       con_pag.ds_fornecedor
)
ORDER BY DS_ITEM_RES,CD_ITEM_RES
