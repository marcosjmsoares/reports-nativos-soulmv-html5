select setor_unidade_operacional.cd_setor,
          setor_unidade_operacional.sn_calcula_ponto_de_equilibrio sn_ponto,
          nm_setor,
          ds_item_producao,
          setor_unidade_operacional.cd_item_producao,
          ds_unidade_mensuracao,
          setor_unidade_operacional.cd_unidade_mensuracao
from dbamv.setor_unidade_operacional,
        dbamv.setor,
        dbamv.item_producao,
        dbamv.unidade_mensuracao
where setor_unidade_operacional.cd_setor = setor.cd_setor(+) and
           setor_unidade_operacional.cd_setor in (select cd_setor from dbamv.setor where
           cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA) and
           setor_unidade_operacional.cd_item_producao = item_producao.cd_item_producao(+) and
           setor_unidade_operacional.cd_unidade_mensuracao = unidade_mensuracao.cd_unidade_mensuracao(+)
ORDER BY NM_SETOR
