--R_CUSTO_FINAL_PAC_CONS
SELECT * FROM (
select fa.cd_atendimento cd_atendimento,
          pc.cd_paciente  cd_paciente,
          Initcap(pc.nm_paciente) nm_paciente,
          fa.vl_receita     receita,
          fa.vl_diaria       diaria,
         '('||ltrim(fa.qtd_diaria)||')' qtd_diaria,
          fa.qtd_diaria               qtd_cont_diaria,
          fa.vl_honorario                honorario,
          fa.vl_procedimento         procedimento,
          fa.vl_matmed                  matmed,
          fa.vl_terceiro                   terceiro,
          fa.tx_adm                        tx_adm,
          fa.tx_imposto                   tx_imposto,
          fa.vl_resultado                 vl_resultado,
          fa.vl_rentabilidade  vl_rentabilidade,
          'X'  dummy,
          decode( :CAMPO_SN_UTI,'%','-','T',fa.sn_uti,nvl ( fa.sn_uti, 'N' )) sn_uti
from   dbamv.fa_custo_atendimento  fa,
         dbamv.atendime ate,
         dbamv.paciente pc,
         dbamv.pro_fat pf,
         dbamv.convenio cv,
         dbamv.prestador pr,
         dbamv.cid,
         dbamv.cirurgia,
         dbamv.gru_cid,
         dbamv.sgru_cid
where  to_char(fa.dt_alta,'yyyy/mm/dd') >=  :CAMPO_DT_INI and
            to_char(fa.dt_alta,'yyyy/mm/dd') <=  :CAMPO_DT_FIM and
            to_char(ate.dt_atendimento,'yyyy/mm/dd') >=  :CAMPO_DT_ATENDIMENTO and
            fa.cd_atendimento = ate.cd_atendimento and
            ate.cd_paciente = pc.cd_paciente and
            ate.cd_pro_int = pf.cd_pro_fat (+) and
            ate.cd_cid = cid.cd_cid (+) and
            cid.cd_sgru_cid = sgru_cid.cd_sgru_cid (+) and
            sgru_cid.cd_gru_cid = gru_cid.cd_gru_cid (+) and
            fa.cd_cirurgia = cirurgia.cd_cirurgia (+) and
            ate.cd_convenio = cv.cd_convenio and
            ate.cd_prestador = pr.cd_prestador(+) and
            ate.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
order by fa.vl_resultado  desc , fa.cd_atendimento
)
ORDER BY VL_RESULTADO DESC,CD_ATENDIMENTO,RECEITA,DIARIA,PROCEDIMENTO,MATMED,TERCEIRO,CD_PACIENTE,TX_ADM,HONORARIO,TX_IMPOSTO,QTD_DIARIA,QTD_CONT_DIARIA,VL_RENTABILIDADE,NM_PACIENTE
