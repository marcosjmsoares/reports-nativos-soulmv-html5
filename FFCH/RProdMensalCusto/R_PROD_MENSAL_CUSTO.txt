--R_PROD_MENSAL_CUSTO
SELECT * FROM (
SELECT   INITCAP (ds_item_producao) ds_item_producao
        ,TO_CHAR (custo_unitario.dt_competencia, 'mm/yyyy') dt_comp
        ,SUM (NVL (custo_unitario.vl_primario, 0)  +
          decode(:CAMPO_VGEN,'S',NVL (custo_unitario.vl_genericos, 0)   , 0) +
          decode(:CAMPO_VESP,'S',NVL (custo_unitario.vl_especificos, 0) , 0) +
          decode(:CAMPO_VMAT,'S',NVL (custo_unitario.vl_materiais_diretos, 0) , 0) ) custo_unitario
        ,SUM (NVL (custo_unitario.qt_producao, 0)) qt_producao
    FROM dbamv.item_producao
               ,dbamv.custo_unitario
   WHERE item_producao.cd_item_producao = custo_unitario.cd_item_producao(+)
        AND custo_unitario.dt_competencia BETWEEN :CAMPO_DT_INI AND 
        AND custo_unitario.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
GROUP BY ds_item_producao
        ,TO_CHAR (custo_unitario.dt_competencia, 'mm/yyyy')
ORDER BY ds_item_producao
        ,TO_CHAR (custo_unitario.dt_competencia, 'mm/yyyy')
   )
ORDER BY DS_ITEM_PRODUCAO,DT_COMP
