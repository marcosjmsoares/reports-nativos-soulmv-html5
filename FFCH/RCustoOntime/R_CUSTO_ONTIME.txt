--R_CUSTO_ONTIME
SELECT * FROM (
select io.cd_setor                   cd_setor
         ,s.nm_setor                    nm_setor
         ,itp.ds_item_producao   ds_item_producao
         ,io.dt_competencia        dt_competencia
         ,io.vl_producao1           vl_producao1
         ,io.vl_producao2           vl_producao2
         ,io.vl_producao3           vl_producao3
         ,io.vl_producao4           vl_producao4
         ,io.vl_receita1               vl_receita1
         ,io.vl_receita2               vl_receita2
         ,io.vl_receita3               vl_receita3
         ,io.vl_receita4               vl_receita4
         ,hc.vl_avg_custo_fixo_total      custo_fixo_total
         ,hc.vl_avg_custeio_var_unit     vl_custo_var_unit
         ,nvl(grafico.maxima,0)        vl_cap_max_prod
         ,(nvl(grafico.maxima,0)/4)  vl_cap_semana
         ,nvl(grafico.meta,0)         vl_cap_esperada
         ,itp.cd_item_producao
  from dbamv.ind_ontime       io
      ,dbamv.setor                   s
      ,dbamv.item_producao   itp
      ,dbamv.historico_custo   hc
      ,dbamv.ontime_grafico    grafico
 where io.cd_setor               = s.cd_setor
   and io.cd_multi_empresa  = :CAMPO_CD_MULTI_EMPRESA
   and trunc(io.dt_competencia)  between trunc(:CAMPO_DATA_INI) and trunc(:CAMPO_DATA_FIM)
   and io.cd_item_producao = itp.cd_item_producao
   and io.cd_setor                 = hc.cd_setor
   and io.cd_item_producao = hc.cd_item_producao
   and io.cd_setor                 = grafico.cd_setor
   and io.cd_item_producao = grafico.cd_item_producao
   and io.dt_competencia     = grafico.dt_competencia
   and grafico.semana = 4
order by nm_setor, io.dt_competencia
)
ORDER BY NM_SETOR,CD_SETOR,
DS_ITEM_PRODUCAO,CD_ITEM_PRODUCAO
