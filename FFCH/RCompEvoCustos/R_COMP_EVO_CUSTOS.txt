--R_COMP_EVO_CUSTOS
SELECT * FROM (
select gru_res.ds_gru_res  ds_gru_res,
       item_res.cd_item_res cd_item_res,
       Decode('', '1', item_res.ds_item_res, to_char(to_char(item_res.cd_item_res,'0000')||'    '||item_res.ds_item_res))  ds_item_res,
       trunc(dt_competencia) dt_comp,
       decode('', 'A', gru_res.ds_gru_res,to_char(gru_res.nr_ordem_grupo,'0000')) nr_ordem_grupo,
       decode('', 'A', item_res.ds_item_res, item_res.ds_item_res) nr_ordem_item,
       sum(vl_custo_setor) valor
from dbamv.gru_res,
     dbamv.item_res,
     dbamv.custo_primario,
     dbamv.setor,
     dbamv.grupo_de_custo
where   to_char(dt_competencia,'yyyy/mm') >= :CAMPO_DT_INI and
             to_char(dt_competencia,'yyyy/mm') <= :CAMPO_DT_FIM and
              item_res.cd_gru_res = gru_res.cd_gru_res and
             custo_primario.cd_item_res = item_res.cd_item_res and
             custo_primario.cd_setor = setor.cd_setor and
	    setor.cd_grupo_de_custo = grupo_de_custo.cd_grupo_de_custo and
              setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
group by  decode('', 'A', gru_res.ds_gru_res,to_char(gru_res.nr_ordem_grupo,'0000')) ,
                decode('', 'A', item_res.ds_item_res, item_res.ds_item_res),
                gru_res.ds_gru_res,
                item_res.cd_item_res,
                Decode('', '1', item_res.ds_item_res, to_char(to_char(item_res.cd_item_res,'0000')||'    '||item_res.ds_item_res)),
                trunc(dt_competencia)
)
