--R_ANALISE_CUSTO_CENCIR
SELECT * FROM (
select setor.cd_setor,
       Initcap(nm_setor) nm_setor,
       atendime.cd_atendimento  cd_atendimento,
       initcap(nm_paciente) nm_paciente,
       dt_realizacao,
       sal_cir.cd_sal_cir,
       ((nvl( dt_fim_limpeza, dbamv.soma_sub_data(dt_realizacao, vl_tempo_duracao,'+')) - dt_realizacao) *24 *60 ) hr_sala,
       aviso_cirurgia.cd_aviso_cirurgia
from dbamv.aviso_cirurgia,
     dbamv.atendime,
     dbamv.setor,
     dbamv.sal_cir
where aviso_cirurgia.dt_realizacao between To_Date('01/'||:CAMPO_DT_COMP, 'dd/mm/yyyy') and
      Last_Day(To_Date('01/'||:CAMPO_DT_COMP, 'dd/mm/yyyy'))+0.99999 and
      setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA and
      aviso_cirurgia.cd_atendimento = atendime.cd_atendimento and
      aviso_cirurgia.cd_sal_cir = sal_cir.cd_sal_cir and
      sal_cir.cd_setor = setor.cd_setor and
      aviso_cirurgia.tp_situacao = 'R'
order by        Initcap(nm_setor) ,
       initcap(nm_paciente)
)
ORDER BY NM_SETOR,CD_SETOR,NM_PACIENTE,CD_SAL_CIR
