--R_ANA_TRAT_ATEND
select atendime.dt_atendimento  dt_atendimento_ate,
       atendime.dt_alta  dt_alta_Ate,
       atendime.cd_atendimento  cd_Atendimento_ate,
       atendime.tp_atendimento   tp_atendimento_Ate,
       paciente.nm_paciente  nm_paciente_ate,
       nm_prestador  nm_prestador_ate,
       ds_cid  ds_cid_ate,
       especialid.ds_especialid ds_especialid_Ate,
       convenio.nm_convenio  nm_convenio_ate,
       servico.ds_servico  ds_servico_ate,
       decode(atendime.cd_pro_int,null,tab_siasus.ds_titulo,pro_fat.ds_pro_fat ) ds_pro_fat_Ate
from   dbamv.atendime,
       dbamv.paciente,
       dbamv.prestador,
       dbamv.cid,
       dbamv.especialid,
       dbamv.convenio,
       dbamv.servico,
       dbamv.pro_fat,
      dbamv.tab_Siasus
where atendime.cd_paciente = paciente.cd_paciente(+) and
      atendime.cd_prestador = prestador.cd_prestador(+) and
      atendime.cd_cid = cid.cd_cid(+) and
      atendime.cd_atendimento = :CAMPO_CD_ATENDIME and
      atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA and
      atendime.cd_pro_int = pro_fat.cd_pro_fat(+) and
      atendime.cd_ssm_sia  = tab_SiaSus.cd_ssm(+) and
      atendime.cd_especialid = especialid.cd_especialid(+) and
      atendime.cd_convenio = convenio.cd_convenio(+) and
      atendime.cd_servico = servico.cd_servico(+)


/*----------------------------------------------------------------------------*/
--R_ANA_TRAT_ATEND_R_2_Q_1
SELECT   ft.dt_fase_trata dt_atendimento_ft
        ,ft.dt_alta dt_alta_ft
        ,ft.cd_atendimento cd_atendimento_ft
        ,ft.tp_atendimento tp_atendimento_ft
        ,paciente.nm_paciente nm_paciente_ft
        ,nm_prestador nm_prestador_ft
        ,ds_cid ds_cid_ft
        ,especialid.ds_especialid ds_especialid_ft
        ,servico.ds_servico ds_servico_ft
        ,ft.cd_fa_tratamento cd_fa_tratamento_ft
        ,pro_fat.ds_pro_fat ds_pro_fat_ft
        ,convenio.nm_convenio nm_convenio_ft
        ,ft.vl_receita
        ,ft.vl_receita_liquida
        ,ft.vl_cus_diaria
        ,ft.vl_cus_matmed
        ,ft.vl_terceiro
        ,ft.vl_cus_honorario
        ,ft.vl_total_custo
        ,ft.vl_total_DEDUCAO
        ,ft.vl_resultado
        ,ft.vl_procedimento
        ,ft.vl_rentabilidade
        ,ft.vl_cus_cencir
        ,ft.fase
        ,setor.nm_setor
    FROM dbamv.temp_ffch_fa_tratamento ft
        ,dbamv.paciente
        ,dbamv.prestador
        ,dbamv.cid
        ,dbamv.especialid
        ,dbamv.convenio
        ,dbamv.servico
        ,dbamv.pro_fat
        ,dbamv.setor
WHERE    ft.cd_paciente = paciente.cd_paciente(+)
     AND ft.cd_prestador = prestador.cd_prestador(+)
     AND ft.cd_cid = cid.cd_cid(+)
     AND ft.cd_especialid = especialid.cd_especialid(+)
     AND ft.cd_convenio = convenio.cd_convenio(+)
     AND ft.cd_servico = servico.cd_servico(+)
     AND ft.cd_atendimento = :CAMPO_CD_ATENDIMENTO_ATE
     AND ft.cd_pro_fat = pro_fat.cd_pro_fat(+)
     AND ft.cd_setor = setor.cd_setor
ORDER BY ft.cd_atendimento
        ,ft.cd_fa_tratamento
