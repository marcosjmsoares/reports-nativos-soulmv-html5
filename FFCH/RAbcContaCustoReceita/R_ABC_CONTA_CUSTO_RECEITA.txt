--R_ABC_CONTA_CUSTO_RECEITA
SELECT DISTINCT 
  a.nm_setor,  
  a.ds_item_res, 
  b.dt_comp,  
  a.dt_comp_string, 
  a.nr_ordem_item, 
  CASE WHEN a.dt_comp = b.dt_comp then a.valor ELSE 0 END valor FROM (
select setor.nm_setor,
		  Decode('', 'A', Initcap(item_res.ds_item_res), to_char(to_char(item_res.cd_item_res,'0000')||' - '||Initcap(item_res.ds_item_res)))  ds_item_res,
          trunc(receita_setor.dt_competencia) dt_comp,
          to_char(receita_setor.dt_competencia, 'mm/yyyy') dt_comp_string,
          decode('', 'A', item_res.ds_item_res,to_char(item_res.nr_ordem_item,'0000')) nr_ordem_item,
          sum(receita_setor.vl_receita_setor) valor
from   dbamv.setor,
          dbamv.item_res,
          dbamv.receita_setor
where receita_setor.dt_competencia >= to_date(:CAMPO_DT_INI,'yyyy/mm') and
           receita_setor.dt_competencia <= to_date(:CAMPO_DT_FIM,'yyyy/mm') and
          setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA and
           nvl(receita_setor.vl_receita_setor,0) <>0 and
          receita_setor.cd_setor = setor.cd_setor and
          receita_setor.cd_item_res = item_res.cd_item_res  and
          receita_setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
group by  setor.nm_setor,
          Decode('', 'A', Initcap(item_res.ds_item_res), to_char(to_char(item_res.cd_item_res,'0000')||' - '||Initcap(item_res.ds_item_res))),
          trunc(receita_setor.dt_competencia),
          to_char(receita_setor.dt_competencia, 'mm/yyyy'),
          decode('', 'A', item_res.ds_item_res,to_char(item_res.nr_ordem_item,'0000'))
) a,
(
select setor.nm_setor,
		  Decode('', 'A', Initcap(item_res.ds_item_res), to_char(to_char(item_res.cd_item_res,'0000')||' - '||Initcap(item_res.ds_item_res)))  ds_item_res,
          trunc(receita_setor.dt_competencia) dt_comp,
          to_char(receita_setor.dt_competencia, 'mm/yyyy') dt_comp_string,
          decode('', 'A', item_res.ds_item_res,to_char(item_res.nr_ordem_item,'0000')) nr_ordem_item,
          sum(receita_setor.vl_receita_setor) valor
from   dbamv.setor,
          dbamv.item_res,
          dbamv.receita_setor
where receita_setor.dt_competencia >= to_date(:CAMPO_DT_INI,'yyyy/mm') and
           receita_setor.dt_competencia <= to_date(:CAMPO_DT_FIM,'yyyy/mm') and
          setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA and
           nvl(receita_setor.vl_receita_setor,0) <>0 and
          receita_setor.cd_setor = setor.cd_setor and
          receita_setor.cd_item_res = item_res.cd_item_res  and
          receita_setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
group by  setor.nm_setor,
          Decode('', 'A', Initcap(item_res.ds_item_res), to_char(to_char(item_res.cd_item_res,'0000')||' - '||Initcap(item_res.ds_item_res))),
          trunc(receita_setor.dt_competencia),
          to_char(receita_setor.dt_competencia, 'mm/yyyy'),
          decode('', 'A', item_res.ds_item_res,to_char(item_res.nr_ordem_item,'0000'))
) b
WHERE 1=1
  AND a.NM_SETOR = b.NM_SETOR
ORDER BY a.NM_SETOR,a.DS_ITEM_RES, b.DT_COMP
