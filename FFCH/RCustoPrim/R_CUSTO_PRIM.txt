--R_CUSTO_PRIM
SELECT *
  FROM (  SELECT ds_item_res,
                 nm_setor,
                 tp_origem_informacao,
                 tp_lancamento_custos,
                 SUM(vl_rateio) vl_rateio
            FROM dbamv.ratcon_pag,
                 dbamv.setor,
                 dbamv.item_res
           WHERE item_res.cd_item_res(+)           = ratcon_pag.cd_item_res
             AND setor.cd_setor (+)                = ratcon_pag.cd_setor
             AND dt_competencia                    = to_date(:CAMPO_DTCOMP,'mm/yyyy')
             AND setor.cd_multi_empresa            = :CAMPO_EMPRESA
             AND ratcon_pag.cd_multi_empresa       = :CAMPO_EMPRESA
        GROUP BY ds_item_res,
                 nm_setor,
                 tp_origem_informacao,
                 tp_lancamento_custos
        )
ORDER BY DS_ITEM_RES
