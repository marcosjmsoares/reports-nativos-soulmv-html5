--R_ANALISE_CONTA_RECEITA
SELECT DISTINCT
  a.ds_item_res,
  a.nm_setor,  
  b.dt_comp ,
  a.dt_comp_string,
  a.nr_ordem_item,
  CASE WHEN a.dt_comp = b.dt_comp then a.vl_custo_setor ELSE 0 END vl_custo_setor
   FROM (
select ds_item_res,
       nm_setor,
       trunc(dt_competencia) dt_comp,
       to_char(dt_competencia,'mm/yyyy') dt_comp_string,
       decode('', 'A', item_res.ds_item_res,to_char(item_res.nr_ordem_item,'0000')) nr_ordem_item,
       sum(vl_receita_setor) vl_custo_setor
from   dbamv.receita_setor,
       dbamv.setor,
       dbamv.item_res
where   	dt_competencia >= :CAMPO_DT_INI and
            dt_competencia <= :CAMPO_DT_FIM and
            setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA and
            dbamv.receita_setor.cd_setor = dbamv.setor.cd_setor and
            dbamv.receita_setor.cd_item_res = dbamv.item_res.cd_item_res and
            receita_setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
group by  decode('', 'A', item_res.ds_item_res,to_char(item_res.nr_ordem_item,'0000')),
                ds_item_res , nm_setor,trunc(dt_competencia),to_char(dt_competencia,'mm/yyyy')
)a, 
(
select ds_item_res,
       nm_setor,
       trunc(dt_competencia) dt_comp,
       to_char(dt_competencia,'mm/yyyy') dt_comp_string,
       decode('', 'A', item_res.ds_item_res,to_char(item_res.nr_ordem_item,'0000')) nr_ordem_item,
       sum(vl_receita_setor) vl_custo_setor
from   dbamv.receita_setor,
       dbamv.setor,
       dbamv.item_res
where   	dt_competencia >= :CAMPO_DT_INI and
            dt_competencia <= :CAMPO_DT_FIM and
            setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA and
            dbamv.receita_setor.cd_setor = dbamv.setor.cd_setor and
            dbamv.receita_setor.cd_item_res = dbamv.item_res.cd_item_res and
            receita_setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
group by  decode('', 'A', item_res.ds_item_res,to_char(item_res.nr_ordem_item,'0000')),
                ds_item_res , nm_setor,trunc(dt_competencia),to_char(dt_competencia,'mm/yyyy')
)b
WHERE 1=1
  AND a.ds_item_res = b.ds_item_res       
  ORDER BY a.DS_ITEM_RES, a.NM_SETOR, b.DT_COMP
