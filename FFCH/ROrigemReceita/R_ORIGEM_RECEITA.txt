--R_ORIGEM_RECEITA
SELECT * FROM (
SELECT tmp.cd_convenio cd_convenio
      ,nm_convenio nm_convenio
      ,nvl(tmp.cd_setor_produziu,tmp.cd_setor) cd_setor
      ,nm_setor nm_setor
      ,tmp.cd_gru_fat cd_gru_fat
      ,ds_gru_fat ds_gru_fat
      ,tmp.cd_gru_pro cd_gru_pro
      ,ds_gru_pro ds_gru_pro
      ,tmp.cd_atendimento cd_atendimento
      ,atendime.cd_paciente cd_paciente
      ,nm_paciente nm_paciente
      ,tmp.cd_item_res cd_item_res
      ,ds_item_res ds_item_res
      ,tmp.tp_origem_faturamento tp_origem_faturamento
      ,tmp.cd_pro_fat cd_pro_fat
      ,ds_pro_fat ds_pro_fat
      ,SUM (qt_lancamento) qt_lancamento
      ,SUM (vl_total_conta) vl_total_conta
  FROM dbamv.temp_ffch_receita_unitaria tmp,
       dbamv.convenio,
       dbamv.setor,
       dbamv.gru_fat,
       dbamv.gru_pro,
       dbamv.atendime,
       dbamv.paciente,
       dbamv.item_res,
       dbamv.pro_fat pro_fat
 WHERE tmp.cd_convenio = convenio.cd_convenio
   AND nvl(tmp.cd_setor_produziu,tmp.cd_setor) = setor.cd_setor(+)
   AND tmp.cd_gru_fat = gru_fat.cd_gru_fat
   AND tmp.cd_gru_pro = gru_pro.cd_gru_pro
   AND tmp.cd_atendimento = atendime.cd_atendimento
   AND atendime.cd_paciente = paciente.cd_paciente
   AND tmp.cd_item_res = item_res.cd_item_res
   AND tmp.cd_pro_fat = pro_fat.cd_pro_fat
   AND tmp.tp_origem_faturamento IN ('CH','RA')
GROUP BY tmp.cd_convenio
        ,nm_convenio
        ,nvl(tmp.cd_setor_produziu,tmp.cd_setor)
        ,nm_setor
        ,tmp.cd_gru_fat
        ,ds_gru_fat
        ,tmp.cd_gru_pro
        ,ds_gru_pro
        ,tmp.cd_atendimento
        ,atendime.cd_paciente
        ,nm_paciente
        ,tmp.cd_item_res
        ,ds_item_res
        ,tmp.tp_origem_faturamento
        ,tmp.cd_pro_fat
        ,ds_pro_fat
UNION ALL
SELECT tmp.cd_convenio cd_convenio
      ,nm_convenio nm_convenio
      ,nvl(tmp.cd_setor_produziu,tmp.cd_setor) cd_setor
      ,nm_setor nm_setor
      ,tmp.cd_gru_fat cd_gru_fat
      ,ds_gru_fat ds_gru_fat
      ,tmp.cd_gru_pro cd_gru_pro
      ,ds_gru_pro ds_gru_pro
      ,tmp.cd_atendimento cd_atendimento
      ,atendime.cd_paciente cd_paciente
      ,nm_paciente nm_paciente
      ,tmp.cd_item_res cd_item_res
      ,ds_item_res ds_item_res
      ,tmp.tp_origem_faturamento tp_origem_faturamento
      ,tmp.cd_pro_fat cd_pro_fat
      ,ds_pro_fat ds_pro_fat
      ,SUM (qt_lancamento) qt_lancamento
      ,SUM (vl_total_conta) vl_total_conta
  FROM dbamv.temp_ffch_receita_unitaria tmp,
       dbamv.convenio,
       dbamv.setor,
       dbamv.gru_fat,
       dbamv.gru_pro,
       dbamv.atendime,
       dbamv.paciente,
       dbamv.item_res,
       dbamv.pro_fat pro_fat
 WHERE tmp.cd_convenio = convenio.cd_convenio
   AND nvl(tmp.cd_setor_produziu,tmp.cd_setor) = setor.cd_setor(+)
   AND tmp.cd_gru_fat = gru_fat.cd_gru_fat
   AND pro_fat.cd_gru_pro = gru_pro.cd_gru_pro
   AND tmp.cd_atendimento = atendime.cd_atendimento
   AND atendime.cd_paciente = paciente.cd_paciente
   AND tmp.cd_item_res = item_res.cd_item_res
   AND tmp.cd_pro_fat = pro_fat.cd_pro_fat
   AND tmp.tp_origem_faturamento IN ('FS','FA')
   AND tmp.dt_lancamento <= to_date('31/12/2007 23','dd/mm/yyyy hh24')
GROUP BY tmp.cd_convenio
        ,nm_convenio
        ,nvl(tmp.cd_setor_produziu,tmp.cd_setor)
        ,nm_setor
        ,tmp.cd_gru_fat
        ,ds_gru_fat
        ,tmp.cd_gru_pro
        ,ds_gru_pro
        ,tmp.cd_atendimento
        ,atendime.cd_paciente
        ,nm_paciente
        ,tmp.cd_item_res
        ,ds_item_res
        ,tmp.tp_origem_faturamento
        ,tmp.cd_pro_fat
        ,ds_pro_fat
UNION ALL
SELECT tmp.cd_convenio cd_convenio
      ,nm_convenio nm_convenio
      ,nvl(tmp.cd_setor_produziu,tmp.cd_setor) cd_setor
      ,nm_setor nm_setor
      ,gru_fat.cd_gru_fat
      ,gru_fat.ds_gru_fat
      ,to_number(tmp.cd_grupo_procedimento) cd_gru_pro
      ,ds_grupo_procedimento ds_gru_pro
      ,tmp.cd_atendimento cd_atendimento
      ,atendime.cd_paciente cd_paciente
      ,nm_paciente nm_paciente
      ,tmp.cd_item_res cd_item_res
      ,ds_item_res ds_item_res
      ,tmp.tp_origem_faturamento tp_origem_faturamento
      ,tmp.cd_procedimento_sus cd_pro_fat
      ,ds_procedimento ds_pro_fat
      ,SUM (qt_lancamento) qt_lancamento
      ,SUM (vl_total_conta) vl_total_conta
  FROM dbamv.temp_ffch_receita_unitaria tmp,
       dbamv.convenio,
       dbamv.setor,
       dbamv.grupo_procedimentos,
       dbamv.atendime,
       dbamv.paciente,
       dbamv.item_res,
       dbamv.procedimento_sus,
       dbamv.procedimento_sus_detalhe,
       dbamv.gru_fat,
       dbamv.gru_pro
 WHERE tmp.cd_convenio = convenio.cd_convenio
   AND nvl(tmp.cd_setor_produziu,tmp.cd_setor) = setor.cd_setor(+)
   AND tmp.cd_grupo_procedimento = grupo_procedimentos.cd_grupo_procedimento
   AND tmp.cd_atendimento = atendime.cd_atendimento(+)
   AND atendime.cd_paciente = paciente.cd_paciente(+)
   AND tmp.cd_item_res = item_res.cd_item_res
   AND tmp.cd_procedimento_sus = procedimento_sus.cd_procedimento
   AND procedimento_sus.cd_procedimento = procedimento_sus_detalhe.cd_procedimento
   AND tmp.tp_origem_faturamento IN ('FS','FA')
   AND tmp.dt_lancamento > to_date('31/12/2007 23','dd/mm/yyyy hh24')
   AND procedimento_sus_detalhe.cd_gru_pro = gru_pro.cd_gru_pro(+)
   AND gru_pro.cd_gru_fat = gru_fat.cd_gru_fat(+)
GROUP BY tmp.cd_convenio
      ,nm_convenio
      ,nvl(tmp.cd_setor_produziu,tmp.cd_setor)
      ,nm_setor
      ,gru_fat.cd_gru_fat
      ,gru_fat.ds_gru_fat
      ,tmp.cd_grupo_procedimento
      ,ds_grupo_procedimento
      ,tmp.cd_atendimento
      ,atendime.cd_paciente
      ,nm_paciente
      ,tmp.cd_item_res
      ,ds_item_res
      ,tmp.tp_origem_faturamento
      ,tmp.cd_procedimento_sus
      ,ds_procedimento
)
ORDER BY CD_CONVENIO,NM_CONVENIO,
CD_SETOR,NM_SETOR,
CD_GRU_FAT,DS_GRU_FAT
