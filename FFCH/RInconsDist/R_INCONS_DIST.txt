--R_INCONS_DIST
SELECT * FROM (
SELECT idf.cd_atendimento,
       atendime.dt_atendimento,
       paciente.nm_paciente,
       idf.cd_reg_fat,
       itreg_fat.cd_pro_fat,
       pro_fat.ds_pro_fat,
       itreg_fat.cd_pro_fat||' - '||pro_fat.ds_pro_fat desc_pro_fat,
       itreg_fat.dt_lancamento,
       log_itcalc_custo.ds_mensagem,
       to_char(itreg_fat.hr_lancamento,'hh24') hr_lancamento,
       to_char(dt_lancamento,'dd/mm/yyyy')||' '||to_char(itreg_fat.hr_lancamento,'hh24') ds_data_hora,
       idf.vl_faturado_pacote,
       idf.vl_faturado_sem_pacote,
       idf.vl_faturado_pacote+idf.vl_faturado_sem_pacote vl_total_faturado,
       idf.vl_distribuido_pacote,
       idf.vl_distribuido_sem_pacote,
       idf.vl_distribuido_pacote+idf.vl_distribuido_sem_pacote vl_total_distribuido,
       abs((idf.vl_faturado_pacote+idf.vl_faturado_sem_pacote)-(idf.vl_distribuido_pacote+idf.vl_distribuido_sem_pacote)) vl_diferenca
  FROM dbamv.inconsist_distribuicao_ffch idf,
       dbamv.atendime,
       dbamv.reg_fat,
       dbamv.itreg_fat,
       dbamv.pro_fat,
       dbamv.paciente,
       dbamv.log_itcalc_custo
 WHERE idf.dt_competencia = to_date(:CAMPO_DT_COMPETENCIA,'MM/YYYY')
   AND idf.cd_multi_empresa = dbamv.pkg_mv2000.le_empresa
   AND idf.cd_atendimento = atendime.cd_atendimento
   AND idf.cd_atendimento = reg_fat.cd_atendimento
   AND idf.cd_reg_fat = itreg_fat.cd_reg_fat
   AND idf.cd_lancamento = itreg_fat.cd_lancamento
   AND idf.cd_log = log_itcalc_custo.cd_log
   AND idf.cd_log_itcalc_custo = log_itcalc_custo.cd_log_itcalc_custo
   AND itreg_fat.cd_reg_fat = reg_fat.cd_reg_fat
   AND itreg_fat.cd_pro_fat = pro_fat.cd_pro_fat
   AND atendime.cd_paciente = paciente.cd_paciente
)
ORDER BY CD_ATENDIMENTO,DT_ATENDIMENTO,NM_PACIENTE,
DS_MENSAGEM
