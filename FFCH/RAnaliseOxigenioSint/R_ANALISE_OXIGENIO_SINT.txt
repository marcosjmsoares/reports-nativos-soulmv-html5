--R_ANALISE_OXIGENIO_SINT
select a.cd_item_res  cd_item_res
      ,a.ds_item_res      ds_item_res
      ,sum(a.custo) custo
 from(
select custo_primario.cd_item_res cd_item_res,
       item_res.ds_item_res ds_item_res,
       nvl((custo_primario.vl_custo_setor),0) custo
from dbamv.item_res,
     dbamv.custo_primario,
     dbamv.setor
where custo_primario.cd_item_res = item_res.cd_item_res and
      item_res.sn_gases = 'S' AND
      custo_primario.dt_competencia = To_Date(:CAMPO_DT_INI, 'MM/YYYY') and
      custo_primario.cd_setor = setor.cd_setor and
      setor.cd_multi_empresa =  :CAMPO_CD_MULTI_EMPRESA
union all
select receita_setor.cd_item_res,
       item_res.ds_item_res,
       0 custo
from dbamv.item_res,
     dbamv.receita_setor,
     dbamv.setor
where receita_setor.cd_item_res = item_res.cd_item_res and
      item_res.sn_gases = 'S' AND
      receita_setor.dt_competencia = To_Date(:CAMPO_DT_INI, 'MM/YYYY') and
      receita_setor.cd_setor = setor.cd_setor and
      setor.cd_multi_empresa =  :CAMPO_CD_MULTI_EMPRESA
)a
group by a.cd_item_res
        ,a.ds_item_res
