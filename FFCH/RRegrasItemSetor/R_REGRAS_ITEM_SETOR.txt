--R_REGRAS_ITEM_SETOR
SELECT * FROM (
select item_res.cd_gru_res,
       gru_res.ds_gru_res,
       regra_item.cd_item_res,
       item_res.ds_item_res,
       regra_item.cd_unidade_mensuracao,
       unidade_mensuracao.ds_unidade_mensuracao,
       regra_item.tp_rateio,
       regra_item.cd_regra_item,
       regra_item.dt_vigencia,
       regra_item.ds_formula
from   dbamv.regra_item,
       dbamv.item_res,
       dbamv.gru_res,
       dbamv.unidade_mensuracao
where  regra_item.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA and
       item_res.cd_gru_res = gru_res.cd_gru_res and
       regra_item.cd_item_res = item_res.cd_item_res and
       regra_item.cd_unidade_mensuracao = unidade_mensuracao.cd_unidade_mensuracao(+) and
       to_char(dt_vigencia,'MM/YYYY') = (select to_char(max(ri.dt_vigencia),'MM/YYYY')
                                         from dbamv.regra_item ri
                                         where to_char(ri.dt_vigencia,'MM/YYYY')<= :CAMPO_DATA_VIRG
                                           and ri.cd_item_res = item_res.cd_item_res)
:CAMPO_CF_GRUPO
)
ORDER BY DS_GRU_RES,CD_GRU_RES,
CD_ITEM_RES,DS_ITEM_RES
