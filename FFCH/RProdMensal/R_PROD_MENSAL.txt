--R_PROD_MENSAL
SELECT * FROM (
select
           to_date( to_char( producao_mensal.dt_competencia, 'mm/yyyy'), 'mm/yyyy') dt_comp,
           to_date( to_char( producao_mensal.dt_competencia, 'mm/yyyy'), 'mm/yyyy') dt_comp1,
           producao_mensal.cd_setor_origem   cd_setor_origem,
           setor1.nm_setor origem,
           item_prod_mensal.cd_item_producao   cd_item_producao,
           initcap(item_producao.ds_item_producao) ds_item_producao,
           item_prod_mensal.cd_setor_destino  cd_setor_destino,
           setor2.nm_setor item,
           sum(item_prod_mensal.vl_producao_mensal) vlr_prod,
           (months_between(:CAMPO_DT_FIM, :CAMPO_DT_INI)+1) qtd_meses
  from
           dbamv.producao_mensal,
           dbamv.setor setor1,
           dbamv.setor setor2,
           dbamv.item_prod_mensal,
           dbamv.item_producao
where producao_mensal.cd_setor_origem         = setor1.cd_setor
   and setor1.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
   and producao_mensal.dt_competencia between :CAMPO_DT_INI and 
   and producao_mensal.cd_producao_mensal = item_prod_mensal.cd_producao_mensal
   and producao_mensal.cd_setor_origem is not null
   and item_prod_mensal.cd_item_producao     = item_producao.cd_item_producao
   and item_prod_mensal.cd_setor_destino       = setor2.cd_setor
group by
       to_date( to_char( producao_mensal.dt_competencia, 'mm/yyyy'), 'mm/yyyy'),
       producao_mensal.cd_setor_origem,
       setor1.nm_setor,
       item_prod_mensal.cd_item_producao,
       item_producao.ds_item_producao,
       item_prod_mensal.cd_setor_destino,
       setor2.nm_setor
order by
         to_date( to_char( producao_mensal.dt_competencia, 'mm/yyyy'), 'mm/yyyy'),
         setor1.nm_setor,
         item_producao.ds_item_producao,
         setor2.nm_setor
)
ORDER BY CD_SETOR_ORIGEM,ORIGEM,
DS_ITEM_PRODUCAO,
DT_COMP,DT_COMP1,CD_ITEM_PRODUCAO,CD_SETOR_DESTINO,ITEM
