--R_CUS_EST_PRI_GRU
SELECT * FROM (
 select setor.cd_setor  cd_setor,
           setor.nm_setor  nm_setor,
           gru_res.cd_gru_res  cd_gru_res,
           gru_res.ds_gru_res  ds_gru_res,
           sum(vl_custo_setor) sum_vl_custo_setor,
           to_char( custo_primario.dt_competencia,'yyyy') TO_CHAR_CUSTO_PRIMARIO_DT_COM1,
           custo_primario.dt_competencia  dt_competencia,
           dbamv.ffch_vl_conta_custo( gru_res.cd_gru_res, :CAMPO_CD_MULTI_EMPRESA, to_char( custo_primario.dt_competencia, 'MM/YYYY' ) ) FFCH_VL_CONTA_CUSTO_GRU_RES_CD
from    dbamv.item_res,
           dbamv.gru_res,
           dbamv.setor,
           dbamv.custo_primario
where  setor.cd_setor                             =  custo_primario.cd_setor
and      custo_primario.cd_item_res         =  item_res.cd_item_res
and      setor.cd_multi_empresa              = :CAMPO_CD_MULTI_EMPRESA
and      TO_CHAR( custo_primario .dt_competencia, 'MM/YYYY' ) = TO_CHAR( :CAMPO_PCOMP, 'MM/YYYY' )
and     gru_res.cd_gru_res                      = item_res.cd_gru_res
Group by  setor.cd_cen_cus,
                setor.cd_setor,
                setor.nm_setor,
                 gru_res.cd_gru_res,
                 gru_res.ds_gru_res,
                custo_primario.dt_competencia
Order by   dbamv.ffch_vl_conta_custo( gru_res.cd_gru_res, :CAMPO_CD_MULTI_EMPRESA, to_char( custo_primario.dt_competencia, 'MM/YYYY' )) desc
               ,sum( vl_custo_setor ) desc)
