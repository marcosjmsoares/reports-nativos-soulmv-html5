--R_ESPECIE_CLASS
SELECT * FROM (
select distinct sub_clas.cd_especie,
       especie.ds_especie,
       item_res.ds_item_res,
       sub_clas.cd_classe,
       classe.ds_classe,
       sub_clas.cd_sub_cla,
       sub_clas.ds_sub_cla,
       sub_clas.sn_custo_direto,
       sub_clas.cd_item_res item_res_sub,
       classe.cd_item_res item_res_classe,
       ds_item_res_classe,
       ds_item_res_sub
from dbamv.sub_clas,
     dbamv.especie,
     dbamv.classe,
     dbamv.item_res,
    (
    SELECT cd_item_res, ds_item_res ds_item_res_classe
      FROM dbamv.item_res
      ) item_res_classe,
    (
    SELECT cd_item_res, ds_item_res ds_item_res_sub
      FROM dbamv.item_res
    ) item_res_sub
where sub_clas.cd_especie = especie.cd_especie and
      sub_clas.cd_classe = classe.cd_classe and
      classe.cd_especie = especie.cd_especie and
      especie.cd_item_res =  item_res.cd_item_res(+) AND
      classe.cd_item_res = item_res_classe.cd_item_res(+) and
      sub_clas.cd_item_res = item_res_sub.cd_item_res(+)
order by  sub_clas.cd_especie,
               especie.ds_especie,
               sub_clas.cd_classe,
               classe.ds_classe
)
ORDER BY CD_ESPECIE,DS_ESPECIE,DS_ITEM_RES,
CD_CLASSE,DS_CLASSE,CD_SUB_CLA
