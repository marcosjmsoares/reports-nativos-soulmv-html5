--R_CCSS_QIII_RANKING
SELECT   temp_rank.cd_ccss
        ,temp_rank.cd_ccss||' - '||ffch_ccss.nm_ccss ds_ccss
        ,temp_rank.cd_setor cd_setor
        ,setor.nm_setor nm_setor
        ,temp_rank.dt_competencia dt_competencia
        ,temp_rank.num_ordem num_ordem
        ,SUM ( temp_rank.vl_custo_setor ) vl_custo_setor
    FROM dbamv.temp_ffch_ranking temp_rank
        ,dbamv.setor
        ,dbamv.ffch_ccss
        ,dbamv.ffch_setor_ccss
   WHERE setor.cd_setor = temp_rank.cd_setor
     AND setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
     AND ffch_ccss.cd_ccss = temp_rank.cd_ccss
     AND ffch_ccss.cd_ccss = ffch_setor_ccss.cd_ccss
     AND ffch_setor_ccss.tp_config_q3 = 'S'
     AND ffch_setor_ccss.cd_setor = setor.cd_setor
GROUP BY temp_rank.cd_ccss
        ,temp_rank.cd_ccss||' - '||ffch_ccss.nm_ccss
        ,temp_rank.cd_setor
        ,setor.nm_setor
        ,temp_rank.dt_competencia
        ,temp_rank.num_ordem
ORDER BY  temp_rank.num_ordem
	,temp_rank.dt_competencia
