--R_MAPA_CUS_PRI_GRU
SELECT   setor.cd_setor                           cd_setor
               ,setor.nm_setor                           nm_setor
               ,gru_res.cd_gru_res                   cd_gru_res
               ,gru_res.ds_gru_res                    ds_gru_res
               ,custo_primario.vl_custo_setor   vl_custo_setor
FROM    dbamv.item_res,
               dbamv.gru_res,
               dbamv.setor,
               dbamv.custo_primario
WHERE setor.cd_setor                                                                     =  custo_primario.cd_setor
and         custo_primario.cd_item_res                                                 =  item_res.cd_item_res
and         setor.cd_multi_empresa                                                       = :CAMPO_CD_MULTI_EMPRESA
and         TO_CHAR( custo_primario .dt_competencia, 'MM/YYYY' ) = TO_CHAR( :CAMPO_PCOMP, 'MM/YYYY' )
and         gru_res.cd_gru_res                                                              = item_res.cd_gru_res
ORDER BY setor.nm_setor,
                    gru_res.ds_gru_res
