--R_DET_CUSTO_OPERAC
SELECT   grupo_de_custo.cd_grupo_de_custo cd_grupo_de_custo
        ,grupo_de_custo.ds_grupo_de_custo ds_grupo_de_custo
 ,DECODE(tp_grupo_de_custo,'A',:CAMPO_TRAD_APOIO
     ,'P',:CAMPO_TRAD_PRODUTIVO
     ,'D',:CAMPO_TRAD_ADMINISTRATIVO
     ,'T',:CAMPO_TRAD_TERCEIRO
                                 ,'O',:CAMPO_TRAD_OUTROS
                                 ,'I',:CAMPO_TRAD_ORDEM_INTERNA
                                 ,:CAMPO_TRAD_SEM_CLASSIFICACAO) tp_grupo_de_custo
        ,setor.nm_setor nm_setor
        ,setor.cd_setor cd_setor
        ,setor.nr_ordem_rateio nr_ordem_rateio
        ,nvl(custo_final.vl_custo_primario, 0) - nvl( custo_final.vl_custo_matmed,0) primario
        ,nvl(custo_final.vl_custo_genericos,0) generico
        ,nvl(custo_final.vl_custo_especificos,0) especifico
        ,nvl(custo_final.vl_custo_matmed,0)    matmed
        ,dbamv.pack_ffch.custo_operacional_setor( :CAMPO_DT_COMPETENCIA , setor.cd_setor )  Custo_Operacional
    FROM dbamv.grupo_de_custo
        ,dbamv.setor
        ,dbamv.custo_final
   WHERE setor.cd_grupo_de_custo = grupo_de_custo.cd_grupo_de_custo
     AND setor.cd_setor = custo_final.cd_setor
	 AND setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
     AND custo_final.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
     AND custo_final.dt_competencia = To_Date(:CAMPO_DT_COMPETENCIA, 'mm/yyyy' )
ORDER BY grupo_de_custo.cd_grupo_de_custo
        ,setor.nr_ordem_rateio
        ,setor.nm_setor
