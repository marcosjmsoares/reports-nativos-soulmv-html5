--R_ANA_CUSTO_SETOR_SINT_R_3_Q_1
SELECT * FROM (
SELECT  cd_setor_base_padrao
      ,setor_nivel.cd_setor cd_setor_filho
      ,setor_nivel.nm_setor nm_setor_filho
  FROM (SELECT s1.cd_setor
              ,s1.nm_setor nm_setor
        FROM   dbamv.setor s1
              ,dbamv.setor
         WHERE s1.cd_setor = setor.cd_setor
           AND setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
           AND setor.cd_setor = :CAMPO_CD_SETOR_BASE
        UNION ALL
        SELECT s2.cd_setor
              ,s2.nm_setor nm_setor
        FROM   dbamv.setor s2
              ,dbamv.setor
         WHERE s2.cd_setor_custo = setor.cd_setor
           AND setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
           AND setor.cd_setor = :CAMPO_CD_SETOR_BASE
        UNION ALL
        SELECT s3.cd_setor
              ,s3.nm_setor nm_setor
        FROM   dbamv.setor s3
         WHERE s3.cd_setor_custo IN (SELECT s1.cd_setor
                                       FROM dbamv.setor s1
                                           ,dbamv.setor
                                      WHERE s1.cd_setor_custo = setor.cd_setor
                                        AND setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
                                        AND setor.cd_setor = :CAMPO_CD_SETOR_BASE)
        UNION ALL
        SELECT s4.cd_setor
              ,s4.nm_setor nm_setor
        FROM   dbamv.setor s4
         WHERE s4.cd_setor_custo IN (SELECT cd_setor
                                       FROM dbamv.setor s3
                                      WHERE s3.cd_setor_custo IN (SELECT s1.cd_setor
                                                                    FROM dbamv.setor s1
                                                                        ,dbamv.setor
                                                                   WHERE s1.cd_setor_custo = setor.cd_setor
                                                                     AND setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
                                                                     AND setor.cd_setor = :CAMPO_CD_SETOR_BASE)
                                                                                                         )
        UNION ALL
        SELECT s5.cd_setor
              ,s5.nm_setor
        FROM   dbamv.setor s5
         WHERE s5.cd_setor_custo IN (SELECT cd_setor
                                       FROM dbamv.setor s4
                                      WHERE s4.cd_setor_custo IN (SELECT cd_setor
                                                                    FROM dbamv.setor s3
                                                                   WHERE s3.cd_setor_custo IN (SELECT s1.cd_setor
                                                                                                 FROM dbamv.setor s1
                                                                                                     ,dbamv.setor
                                                                                                WHERE s1.cd_setor_custo = setor.cd_setor
                                                                                                  AND setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
                                                                                                  AND setor.cd_setor = :CAMPO_CD_SETOR_BASE)
                                                                                                                                      )
                                                                                                                                       )) setor_nivel
)
WHERE CD_SETOR_BASE_PADRAO = :CAMPO_CD_SETOR_BASE

                                                        FROM Dbamv.Setor S1
                                                             ,Dbamv.Setor

                                                         AND Setor.Cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA

                                              UNION ALL
                                             SELECT S4.Cd_setor
                                               FROM Dbamv.Setor S4
                                              WHERE S4.Cd_setor_custo IN(
                                                       SELECT Cd_setor
                                                         FROM Dbamv.Setor S3
                                                         WHERE S3.Cd_setor_custo IN(
                                                                  SELECT S1.Cd_setor
                                                                    FROM Dbamv.Setor S1
                                                                        ,Dbamv.Setor
                                                                   WHERE S1.Cd_setor_custo = Setor.Cd_setor
                                                                     AND Setor.Cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
                                                                     and setor.cd_setor = :CAMPO_CD_SETOR_BASE  )
                                                                        )
                                              UNION ALL

                                               FROM Dbamv.Setor S5
                                              WHERE S5.Cd_setor_custo IN(
                                                       SELECT Cd_setor
                                                         FROM Dbamv.Setor S4
                                                        WHERE S4.Cd_setor_custo IN(
                                                                 SELECT Cd_setor
                                                                   FROM Dbamv.Setor S3
                                                                  WHERE S3.Cd_setor_custo IN(
                                                                           SELECT S1.Cd_setor
                                                                             FROM Dbamv.Setor S1, dbamv.setor
                                                                            WHERE S1.Cd_setor_custo = Setor.Cd_setor

                                                                              and Setor.cd_setor = :CAMPO_CD_SETOR_BASE )
                                                                                   )

                                      and setor.cd_setor = :CAMPO_CD_SETOR_BASE
                                  )
GROUP BY setor_base.cd_setor
       ,Setor.Cd_setor
       ,setor.nm_setor
       ,setor.cd_cen_cus

       ,Gru_res.Ds_gru_res
       ,Gru_res.nr_ordem_grupo
       ,item_res.cd_item_res
