--R_GRUPO_ORIGEM_DESTINO
SELECT * FROM (
select
           grupo_de_custo.cd_grupo_de_custo,
           grupo_de_custo.ds_grupo_de_custo,
           setor_origem.cd_setor cd_setor_origem,
           setor_origem.nm_setor nm_setor_origem,
           nvl(item_prod_mensal.cd_pro_fat, item_prod_mensal.cd_item_producao)  cd_pro_fat,
           nvl(pro_fat.ds_pro_fat,ds_item_producao) ds_pro_fat,
           v_prod_centro.total,
           item_prod_mensal.vl_producao_mensal,
           setor_destino.cd_setor cd_setor_destino,
           setor_destino.nm_setor nm_setor_destino,
           producao_mensal.dt_competencia
  from
       dbamv.grupo_de_custo,
       dbamv.setor setor_origem,
       dbamv.item_prod_mensal,
       dbamv.setor setor_destino,
       dbamv.producao_mensal,
       dbamv.v_prod_centro,
       dbamv.pro_fat,
       dbamv.item_producao
 where
       grupo_de_custo.cd_grupo_de_custo         = setor_origem.cd_grupo_de_custo
   and setor_origem.cd_setor                    = producao_mensal.cd_setor_origem
   and setor_origem.cd_multi_empresa            = :CAMPO_CD_MULTI_EMPRESA
   and to_date(to_char(producao_mensal.dt_competencia, 'mm/yyyy'), 'mm/yyyy') = :CAMPO_COMPETENCIA
   and producao_mensal.cd_producao_mensal = item_prod_mensal.cd_producao_mensal
   and item_prod_mensal.cd_pro_fat                  = pro_fat.cd_pro_fat(+)
   and item_prod_mensal.cd_item_producao                  = item_producao.cd_item_producao(+)
   and item_prod_mensal.cd_setor_destino        = setor_destino.cd_setor
   and v_prod_centro.cd_item_producao     = item_prod_mensal.cd_item_producao
   and producao_mensal.dt_competencia  = v_prod_centro.dt_competencia
   and producao_mensal.cd_setor_origem = v_prod_centro.cd_setor_origem
   and producao_mensal.cd_producao_mensal = item_prod_mensal.cd_producao_mensal
  order by
           grupo_de_custo.ds_grupo_de_custo,
           setor_origem.nm_setor,
           pro_fat.ds_pro_fat,
           setor_destino.nm_setor
)
ORDER BY CD_GRUPO_DE_CUSTO,DS_GRUPO_DE_CUSTO,
CD_SETOR_ORIGEM,NM_SETOR_ORIGEM,
CD_PRO_FAT,DS_PRO_FAT,TOTAL
