from dbamv.ffch_ccss
where 1 = 1


/*----------------------------------------------------------------------------*/
--R_CCSS_QI_EVOLUCAO_R_2_Q_SERVICO
SELECT * FROM (
SELECT * FROM (
SELECT   tp_config_q1
        ,cd_ccss
        ,dt_competencia
        ,SUM ( TRUNC( vl_total_unidade, 2 ) ) vl_tot_unidade
        ,SUM ( vl_total_exame ) vl_tot_exame
        ,SUM ( tot_fator ) qtd_fator
    FROM ( SELECT  tmp.tp_config_q1
                  ,tmp.cd_ccss
                  ,tmp.dt_competencia dt_competencia
                  ,tmp.vl_tot_unidade vl_total_unidade
                  ,tmp.vl_tot_exame vl_total_exame
                  ,tmp.qtd_fator tot_fator
              FROM dbamv.tmpmv_ffch_quadro1 tmp
          UNION
          SELECT tp_config_q1
                ,null cd_ccss
                ,null dt_comp
                ,0 vl_unidade
                ,0 vl_exame
                ,0 qtd_fator
            FROM ( SELECT '1' tp_config_q1 FROM dual UNION
                  SELECT '2' tp_config_q1  FROM dual UNION
                  SELECT '3' tp_config_q1  FROM dual UNION
                  SELECT '4' tp_config_q1  FROM dual UNION
                  SELECT '5' tp_config_q1  FROM dual UNION
                  SELECT '6' tp_config_q1  FROM dual UNION
                  SELECT '7' tp_config_q1  FROM dual UNION
                  SELECT '8' tp_config_q1  FROM dual UNION
                  SELECT '9' tp_config_q1  FROM dual
                  ))
 WHERE tp_config_q1 IS NOT NULL
GROUP BY tp_config_q1, cd_ccss
        ,dt_competencia
ORDER BY tp_config_q1
        ,dt_competencia
)
WHERE CD_CCSS = :CAMPO_CD_CCSS
)
ORDER BY CD_CCSS,TP_CONFIG_Q1,DT_COMPETENCIA
