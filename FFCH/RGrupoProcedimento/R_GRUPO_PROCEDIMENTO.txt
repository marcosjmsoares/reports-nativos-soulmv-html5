--R_GRUPO_PROCEDIMENTO
SELECT * FROM (
select  ds_gru_pro,
        pro_fat.cd_pro_fat ,
        ds_pro_fat,
        item_pro_fat.cd_produto,
        ds_produto,
        uni_pro.ds_unidade,
        qt_padrao,
        vl_tempo
from   dbamv.item_pro_fat,
       dbamv.pro_fat,
       dbamv.uni_pro,
       dbamv.produto,
       dbamv.gru_pro
where  item_pro_fat.cd_pro_fat(+) = pro_fat.cd_pro_fat and
       item_pro_fat.cd_produto = produto.cd_produto(+) and
       item_pro_fat.cd_uni_pro = uni_pro.cd_uni_pro(+) and
       pro_fat.cd_gru_pro = gru_pro.cd_gru_pro(+) and
       produto.ds_produto is not null
       AND Exists ( SELECT Empresa_Produto.Cd_Produto
                      FROM Dbamv.Empresa_Produto
                     WHERE Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
                       AND Empresa_Produto.Cd_Produto = Produto.Cd_Produto )
       :CAMPO_CF_FICHA
)
ORDER BY DS_GRU_PRO,
DS_PRO_FAT,DS_PRO_FAT,DS_PRODUTO
