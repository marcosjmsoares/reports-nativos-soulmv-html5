--ATENDIMENTO_DETALHADO
SELECT atendime.dt_atendimento,
    atendime.dt_alta,
    atendime.cd_atendimento,
    atendime.tp_atendimento,
    paciente.nm_paciente,
    NVL(prestador.nm_prestador, '-') nm_prestador,
    NVL(cid.ds_cid, '-') ds_cid,
    NVL(especialid.ds_especialid, '-') ds_especialid,
    NVL(convenio.nm_convenio, '-') nm_convenio,
    NVL(servico.ds_servico, '-') ds_servico,
    atendime.cd_multi_empresa
   FROM dbamv.atendime,
    dbamv.paciente,
    dbamv.prestador,
    dbamv.cid,
    dbamv.especialid,
    dbamv.convenio,
    dbamv.servico,
    dbamv.pro_fat,
    dbamv.tab_siasus
  WHERE atendime.cd_paciente      = paciente.cd_paciente(+)
    AND atendime.cd_prestador     = prestador.cd_prestador(+)
    AND atendime.cd_cid           = cid.cd_cid(+)
    AND atendime.cd_atendimento   = :CAMPO_CD_ATENDIME
    AND atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
    AND atendime.cd_pro_int       = pro_fat.cd_pro_fat(+)
    AND atendime.cd_ssm_sia       = tab_siasus.cd_ssm(+)
    AND atendime.cd_especialid    = especialid.cd_especialid(+)
    AND atendime.cd_convenio      = convenio.cd_convenio(+)
    AND atendime.cd_servico       = servico.cd_servico(+)

--ATENDIMENTO_DETALHADO_DIARIA_1
SELECT cd_tip_acom,
    cd_setor,
    cd_atendimento_diaria,
    dt_periodo,
    dias,
    ds_tip_acom,
    nm_setor,
    Decode(Sign(custo_diaria), -1, 0, custo_diaria) custo_diaria,
    Decode(Sign(custo_diaria), -1, 0, NVL(dias, 0) * NVL(custo_diaria, 0)) custo_diaria_total,
    dt_referencia
   FROM
    (SELECT resumo.cd_tip_acom,
        resumo.cd_setor,
        resumo.cd_atendimento_diaria,
        resumo.dt_periodo,
        resumo.dias,
        tip_acom.ds_tip_acom,
        setor.nm_setor,
        dbamv.pack_ffch.custo_diaria(resumo.cd_setor, resumo.cd_tip_acom, resumo.dt_referencia) custo_diaria,
        to_date(resumo.dt_periodo, 'mm/yyyy') data_periodo_ordenacao,
        resumo.dt_referencia
       from
        (select geral.cd_tip_acom,
            geral.cd_setor,
            geral.cd_atendimento_diaria,
            geral.dt_periodo,
            geral.dias,
            (SELECT TO_CHAR(NVL(MAX(dt_competencia), to_date(geral.dt_periodo, 'mm/yyyy')), 'mm/yyyy')
               FROM dbamv.custo_final
              WHERE cd_setor        = geral.cd_setor
                AND dt_competencia <= to_date(geral.dt_periodo, 'mm/yyyy')
            ) dt_referencia
           FROM
            (SELECT a.cd_tip_acom,
                a.cd_setor,
                a.cd_atendimento_diaria cd_atendimento_diaria,
                a.dt_periodo,
                ROUND(SUM(to_number(a.dias))) dias
               FROM
                (SELECT leito.cd_tip_acom,
                    unid_int.cd_setor,
                    mov_int.cd_atendimento cd_atendimento_diaria,
                    TO_CHAR(mov_int.dt_mov_int, 'mm/yyyy') dt_int,
                    TO_CHAR(NVL(mov_int.dt_lib_mov, sysdate), 'mm/yyyy') dt_lib,
                    TO_CHAR(add_months(NVL(mov_int.dt_lib_mov, sysdate),(linhas.lin - 12)), 'mm/yyyy') dt_periodo,
                    dbamv.ret_dias_def(To_Date(TO_CHAR(mov_int.dt_mov_int, 'dd/mm/yyyy') || ' ' || TO_CHAR(mov_int.hr_mov_int, 'hh24'), 'dd/mm/yyyy hh24'), NVL(mov_int.dt_lib_mov, sysdate), mov_int.tp_mov, mov_int.cd_leito, mov_int.cd_leito_anterior, mov_int.cd_atendimento, mov_int.cd_mov_int, linhas.lin) dias
                   FROM dbamv.mov_int,
                    dbamv.leito,
                    dbamv.unid_int,
                    dbamv.setor,
                    (SELECT rownum lin FROM dbamv.pro_fat WHERE rownum <= 12
                    ) linhas
                  WHERE mov_int.dt_mov_int                                         IS NOT NULL
                    AND mov_int.cd_leito                                            = leito.cd_leito
                    AND leito.cd_unid_int                                           = unid_int.cd_unid_int
                    AND setor.cd_setor                                              = unid_int.cd_setor
                    AND cd_atendimento                                              = :CAMPO_CD_ATENDIMENTO
                    AND setor.tp_composicao_calculo                                 = '4'
                    AND TO_DATE(TO_CHAR(mov_int.dt_mov_int, 'mm/yyyy'), 'mm/yyyy') <= TO_DATE(TO_CHAR(NVL(mov_int.dt_lib_mov, sysdate), 'mm/yyyy'), 'mm/yyyy')
                ) a
              WHERE TO_DATE(a.dt_periodo, 'mm/yyyy') BETWEEN TO_DATE(a.dt_int, 'mm/yyyy') AND TO_DATE(a.dt_lib, 'mm/yyyy')
                AND NVL(a.dias, 0)       > 0
                AND (SELECT tp_custo_diaria FROM dbamv.config_fcch WHERE cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA) NOT IN ('P')
           GROUP BY a.cd_tip_acom,
                a.cd_setor,
                a.cd_atendimento_diaria,
                a.dt_periodo,
                a.dt_int,
                a.dt_lib,
                0
              UNION
             SELECT b.cd_tip_acom,
                b.cd_setor,
                b.cd_atendimento cd_atendimento_diaria,
                b.dt_periodo,
                SUM(b.tt_pac_dia) dias
               FROM
                (SELECT cd_atendimento,
                    dt_periodo,
                    cd_setor,
                    cd_tip_acom,
                    ((SUM(pac_int_00h) + SUM(ent_internados) + SUM(ent_transf)) -(SUM(sai_altas) + SUM(sai_transfpara) + SUM(sai_obitos))) tt_pac_dia
                   FROM
                    (SELECT atendime.cd_atendimento,
                        TO_CHAR(NVL(contador.DATA, sysdate), 'mm/yyyy') dt_periodo,
                        unid_int.cd_setor cd_setor,
                        leito.cd_tip_acom cd_tip_acom,
                        COUNT( *) pac_int_00h,
                        0 ent_internados,
                        0 ent_transf,
                        0 sai_altas,
                        0 sai_transfpara,
                        0 sai_obitos
                       FROM dbamv.mov_int,
                        dbamv.unid_int,
                        dbamv.leito,
                        dbamv.atendime,
                        (SELECT(:CAMPO_DT_ATENDIMENTO - 1) + rownum DATA
                           FROM dbamv.cid
                          WHERE(:CAMPO_DT_ATENDIMENTO - 1) + rownum <= NVL(:CAMPO_DT_ALTA, sysdate)
                        ) contador
                      WHERE TRUNC(mov_int.dt_mov_int)              <= contador.DATA - 1
                        AND TRUNC(NVL(mov_int.dt_lib_mov, sysdate)) > contador.DATA - 1
                        AND mov_int.tp_mov                                         IN('O', 'I')
                        AND leito.cd_unid_int                       = unid_int.cd_unid_int
                        AND mov_int.cd_atendimento                  = atendime.cd_atendimento
                        AND(atendime.tp_atendimento                IN('I', 'H'))
                        AND mov_int.cd_leito                        = leito.cd_leito
                        AND atendime.cd_atendimento                 = :CAMPO_CD_ATENDIMENTO
                   GROUP BY atendime.cd_atendimento,
                        TO_CHAR(NVL(contador.DATA, sysdate), 'mm/yyyy'),
                        unid_int.cd_setor,
                        leito.cd_tip_acom
                      UNION
                     SELECT atendime.cd_atendimento,
                        TO_CHAR(NVL(contador.DATA, sysdate), 'mm/yyyy') dt_periodo,
                        unid_int.cd_setor cd_setor,
                        leito.cd_tip_acom cd_tip_acom,
                        COUNT( *) pac_int_00h,
                        0 ent_internados,
                        0 ent_transf,
                        0 sai_altas,
                        0 sai_transfpara,
                        0 sai_obitos
                       FROM dbamv.atendime,
                        dbamv.leito,
                        dbamv.unid_int,
                        dbamv.convenio,
                      (SELECT cd_atendimento, tp_mov, cd_leito, dt_mov_int
                         FROM dbamv.mov_int
                        WHERE cd_atendimento = :CAMPO_CD_ATENDIMENTO
                          AND cd_mov_int IN (SELECT cd_mov_int
                                               FROM dbamv.mov_int mov_int_filho
                                              WHERE 'INTERNACAO' IN (SELECT Nvl(tp_contagem_hospitaldia, 'INTERNACAO')
                                                                       FROM dbamv.config_fcch
                                                                      WHERE cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA)
                                                AND mov_int_filho.cd_atendimento = :CAMPO_CD_ATENDIMENTO
                                                AND mov_int_filho.tp_mov = 'I'
                                             UNION ALL
                                             SELECT Max(cd_mov_int)
                                               FROM dbamv.mov_int mov_int_filho
                                              WHERE 'ALTA' IN (SELECT Nvl(tp_contagem_hospitaldia, 'INTERNACAO')
                                                                       FROM dbamv.config_fcch
                                                                      WHERE cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA)
                                                AND mov_int_filho.cd_atendimento = :CAMPO_CD_ATENDIMENTO
                                            )) mov_int,
                        (SELECT(:CAMPO_DT_ATENDIMENTO - 1) + ROWNUM DATA
                           FROM dbamv.cid
                          WHERE(:CAMPO_DT_ATENDIMENTO - 1) + ROWNUM <= :CAMPO_DT_ALTA
                        ) contador
                      WHERE  mov_int.cd_leito               = leito.cd_leito
                        AND leito.cd_unid_int              = unid_int.cd_unid_int
                        AND atendime.cd_atendimento        = mov_int.cd_atendimento
                        AND(atendime.tp_atendimento       IN('I', 'H'))
                        AND atendime.dt_alta              IS NOT NULL
                        AND TRUNC(atendime.dt_atendimento) = TRUNC(atendime.dt_alta)
                        AND TRUNC(mov_int.dt_mov_int)      = contador.DATA
                        AND atendime.cd_convenio           = convenio.cd_convenio
                        AND EXISTS
                        (SELECT 1
                           FROM DBAMV.CONFIG_FCCH
                          WHERE CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
                            AND SN_HOSPITAL_DIA  = 'S'
                        )
                        AND atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
                        AND atendime.cd_atendimento   = :CAMPO_CD_ATENDIMENTO
                   GROUP BY atendime.cd_atendimento,
                        TO_CHAR(NVL(contador.DATA, sysdate), 'mm/yyyy'),
                        unid_int.cd_setor,
                        leito.cd_tip_acom
                      UNION
                     SELECT atendime.cd_atendimento,
                        TO_CHAR(NVL(contador.DATA, sysdate), 'mm/yyyy') dt_periodo,
                        unid_int.cd_setor cd_setor,
                        leito.cd_tip_acom cd_tip_acom,
                        0 pac_int_00h,
                        COUNT( *) ent_internados,
                        0 ent_transf,
                        0 sai_altas,
                        0 sai_transfpara,
                        0 sai_obitos
                       FROM dbamv.atendime,
                        dbamv.unid_int,
                        dbamv.leito,
                        dbamv.mov_int,
                        (SELECT(:CAMPO_DT_ATENDIMENTO - 1) + rownum DATA
                           FROM dbamv.cid
                          WHERE(:CAMPO_DT_ATENDIMENTO - 1) + rownum <= NVL(:CAMPO_DT_ALTA, sysdate)
                        ) contador
                      WHERE leito.cd_unid_int        = unid_int.cd_unid_int
                        AND leito.cd_leito           = mov_int.cd_leito
                        AND TRUNC(dt_mov_int)        = TRUNC(contador.DATA)
                        AND mov_int.tp_mov           = 'I'
                        AND atendime.cd_atendimento  = mov_int.cd_atendimento
                        AND(atendime.tp_atendimento IN('I', 'H'))
                        AND atendime.cd_atendimento  = :CAMPO_CD_ATENDIMENTO
                   GROUP BY atendime.cd_atendimento,
                        TO_CHAR(NVL(contador.DATA, sysdate), 'mm/yyyy'),
                        unid_int.cd_setor,
                        leito.cd_tip_acom
                      UNION
                     SELECT atendime.cd_atendimento,
                        TO_CHAR(NVL(contador.DATA, sysdate), 'mm/yyyy') dt_periodo,
                        unid_int.cd_setor cd_setor,
                        leito.cd_tip_acom cd_tip_acom,
                        0 pac_int_00h,
                        0 ent_internados,
                        COUNT( *) ent_transf,
                        0 sai_altas,
                        0 sai_transfpara,
                        0 sai_obitos
                       FROM dbamv.mov_int,
                        dbamv.unid_int,
                        dbamv.unid_int unid_int1,
                        dbamv.leito,
                        dbamv.leito leito1,
                        dbamv.atendime,
                        (SELECT(:CAMPO_DT_ATENDIMENTO - 1) + rownum DATA
                           FROM dbamv.cid
                          WHERE(:CAMPO_DT_ATENDIMENTO - 1) + rownum <= NVL(:CAMPO_DT_ALTA, sysdate)
                        ) contador
                      WHERE mov_int.tp_mov            = 'O'
                        AND TRUNC(mov_int.dt_mov_int) = contador.DATA
                        AND mov_int.cd_leito          = leito.cd_leito
                        AND mov_int.cd_leito_anterior = leito1.cd_leito
                        AND leito1.cd_unid_int        = unid_int1.cd_unid_int
                        AND unid_int.cd_unid_int     <> unid_int1.cd_unid_int
                        AND leito.cd_unid_int         = unid_int.cd_unid_int
                        AND atendime.cd_atendimento   = mov_int.cd_atendimento
                        AND(atendime.tp_atendimento  IN('I', 'H'))
                        AND atendime.cd_atendimento   = :CAMPO_CD_ATENDIMENTO
                   GROUP BY atendime.cd_atendimento,
                        TO_CHAR(NVL(contador.DATA, sysdate), 'mm/yyyy'),
                        unid_int.cd_setor,
                        leito.cd_tip_acom
                      UNION
                     SELECT atendime.cd_atendimento,
                        TO_CHAR(NVL(contador.DATA, sysdate), 'mm/yyyy') dt_periodo,
                        unid_int.cd_setor cd_setor,
                        leito.cd_tip_acom cd_tip_acom,
                        0 pac_int_00h,
                        0 ent_internados,
                        0 ent_transf,
                        COUNT( *) sai_altas,
                        0 sai_transfpara,
                        0 sai_obitos
                       FROM dbamv.atendime,
                        dbamv.unid_int,
                        dbamv.leito,
                        dbamv.mot_alt,
                        (SELECT(:CAMPO_DT_ATENDIMENTO - 1) + rownum DATA
                           FROM dbamv.cid
                          WHERE(:CAMPO_DT_ATENDIMENTO - 1) + rownum <= NVL(:CAMPO_DT_ALTA, sysdate)
                        ) contador
                      WHERE leito.cd_leito          = atendime.cd_leito
                        AND mot_alt.cd_mot_alt      = atendime.cd_mot_alt
                        AND unid_int.cd_unid_int    = leito.cd_unid_int
                        AND TRUNC(atendime.dt_alta) = contador.DATA
                        AND mot_alt.tp_mot_alta    <> 'O'
                        AND atendime.tp_atendimento = 'I'
                        AND atendime.cd_atendimento = :CAMPO_CD_ATENDIMENTO
                   GROUP BY atendime.cd_atendimento,
                        TO_CHAR(NVL(contador.DATA, sysdate), 'mm/yyyy'),
                        unid_int.cd_setor,
                        leito.cd_tip_acom
                      UNION
                     SELECT atendime.cd_atendimento,
                        TO_CHAR(NVL(contador.DATA, sysdate), 'mm/yyyy') dt_periodo,
                        unid_int.cd_setor cd_setor,
                        leito.cd_tip_acom cd_tip_acom,
                        0 pac_int_00h,
                        0 ent_internados,
                        0 ent_transf,
                        0 sai_altas,
                        COUNT( *) sai_transfpara,
                        0 sai_obitos
                       FROM dbamv.mov_int,
                        dbamv.unid_int,
                        dbamv.unid_int unid_int1,
                        dbamv.leito,
                        dbamv.leito leito1,
                        dbamv.atendime,
                        (SELECT(:CAMPO_DT_ATENDIMENTO - 1) + rownum DATA
                           FROM dbamv.cid
                          WHERE(:CAMPO_DT_ATENDIMENTO - 1) + rownum <= NVL(:CAMPO_DT_ALTA, sysdate)
                        ) contador
                      WHERE TRUNC(mov_int.dt_mov_int) = contador.DATA
                        AND mov_int.tp_mov            = 'O'
                        AND mov_int.cd_leito_anterior = leito.cd_leito
                        AND mov_int.cd_leito          = leito1.cd_leito
                        AND leito1.cd_unid_int        = unid_int1.cd_unid_int
                        AND unid_int.cd_unid_int     <> unid_int1.cd_unid_int
                        AND leito.cd_unid_int         = unid_int.cd_unid_int
                        AND atendime.cd_atendimento   = mov_int.cd_atendimento
                        AND(atendime.tp_atendimento  IN('I', 'H'))
                        AND atendime.cd_atendimento   = :CAMPO_CD_ATENDIMENTO
                   GROUP BY atendime.cd_atendimento,
                        TO_CHAR(NVL(contador.DATA, sysdate), 'mm/yyyy'),
                        unid_int.cd_setor,
                        leito.cd_tip_acom
                      UNION
                     SELECT atendime.cd_atendimento,
                        TO_CHAR(NVL(contador.DATA, sysdate), 'mm/yyyy') dt_periodo,
                        unid_int.cd_setor cd_setor,
                        leito.cd_tip_acom cd_tip_acom,
                        0 pac_int_00h,
                        0 ent_internados,
                        0 ent_transf,
                        0 sai_altas,
                        0 sai_transfpara,
                        COUNT( *) sai_obitos
                       FROM dbamv.atendime,
                        dbamv.unid_int,
                        dbamv.leito,
                        dbamv.mot_alt,
                        (SELECT(:CAMPO_DT_ATENDIMENTO - 1) + rownum DATA
                           FROM dbamv.cid
                          WHERE(:CAMPO_DT_ATENDIMENTO - 1) + rownum <= NVL(:CAMPO_DT_ALTA, sysdate)
                        ) contador
                      WHERE leito.cd_leito          = atendime.cd_leito
                        AND mot_alt.cd_mot_alt      = atendime.cd_mot_alt
                        AND leito.cd_unid_int       = unid_int.cd_unid_int
                        AND TRUNC(atendime.dt_alta) = contador.DATA
                        AND mot_alt.tp_mot_alta     = 'O'
                        AND atendime.tp_atendimento = 'I'
                        AND atendime.cd_atendimento = :CAMPO_CD_ATENDIMENTO
                   GROUP BY atendime.cd_atendimento,
                        TO_CHAR(NVL(contador.DATA, sysdate), 'mm/yyyy'),
                        unid_int.cd_setor,
                        leito.cd_tip_acom
                    )
               GROUP BY cd_atendimento,
                    dt_periodo,
                    cd_setor,
                    cd_tip_acom
                ) b,
                dbamv.setor
              WHERE b.tt_pac_dia                > 0
                AND setor.cd_setor              = b.cd_setor
                AND setor.tp_composicao_calculo = '4'
                AND (SELECT tp_custo_diaria FROM dbamv.config_fcch WHERE cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA) IN ('P')
           GROUP BY b.cd_atendimento,
                b.dt_periodo,
                b.cd_setor,
                b.cd_tip_acom,
                0
            ) geral
        ) resumo,
        dbamv.tip_acom,
        dbamv.setor
      WHERE resumo.cd_tip_acom    = tip_acom.cd_tip_acom (+)
        AND resumo.cd_setor       = setor.cd_setor (+)
        AND cd_atendimento_diaria = :CAMPO_CD_ATENDIMENTO
    )
ORDER BY cd_tip_acom,
    data_periodo_ordenacao,
    cd_setor

--ATENDIMENTO_DETALHADO_DIARIA_2
SELECT * FROM (
SELECT cd_tip_acom
        ,cd_setor
        ,nm_atividade_gen
        ,cd_atendimento_diaria
        ,dt_periodo
        ,dias
        ,Decode(Sign(vl_atividade), -1, 0, vl_atividade) vl_atividade
        ,Decode(Sign(vl_atividade), -1, 0, NVL(dias, 0) * NVL(vl_atividade, 0)) vl_total FROM (
SELECT   cd_tip_acom
        ,cd_setor
        ,nm_atividade_gen
        ,cd_atendimento_diaria
        ,dt_periodo
        ,NVL(dias, 0) dias
        ,NVL(vl_atividade, 0) vl_atividade
    FROM (
        SELECT   atividade_xml.cd_tip_acom_mv cd_tip_acom
                ,atividade_xml.cd_setor_produziu_mv cd_setor
                ,atividade_xml.nm_atividade_gen
                ,atendimento_xml.cd_atendimento_mv cd_atendimento_diaria
                ,TO_CHAR( atividade_xml.dt_atividade, 'MM/YYYY' ) dt_periodo
                ,SUM( atividade_xml.qt_realizada ) dias
                ,SUM(
                 dbamv.pack_ffch.custo_procedimento(
                                             TO_CHAR( atividade_xml.dt_atividade, 'mm/yyyy' )
                                    ,NULL
                                    ,atividade_xml.cd_setor_produziu_mv
                                    ,:CAMPO_CD_MULTI_EMPRESA
                                    ,NULL
                                    ,NULL
                                    ,atendimento_xml.cd_atendimento_mv
                                    ,NULL
                                    ,NULL
                                    ,NULL
                                    ,atividade_xml.cd_atividade_gen )) vl_atividade
        FROM     dbamv.atendimento_xml
                ,dbamv.atividade_xml
                ,dbamv.item_atividade
                ,dbamv.custo_unitario
        WHERE    atendimento_xml.cd_atendimento_mv = :CAMPO_CD_ATENDIMENTO
        AND      atividade_xml.cd_atendimento_gen = atendimento_xml.cd_atendimento_gen
        AND      item_atividade.cd_atividade = atividade_xml.cd_atividade_gen
        AND      custo_unitario.cd_atividade = item_atividade.cd_atividade
        AND      TO_CHAR( atividade_xml.dt_atividade, 'MM/YYYY' ) = TO_CHAR( custo_unitario.dt_competencia, 'MM/YYYY' )
        AND      custo_unitario.cd_setor = atividade_xml.cd_setor_produziu_mv
        AND      atividade_xml.tp_atividade = '6'
        GROUP BY atividade_xml.cd_tip_acom_mv
                ,atividade_xml.cd_setor_produziu_mv
                ,atividade_xml.nm_atividade_gen
                ,atendimento_xml.cd_atendimento_mv
                ,TO_CHAR( atividade_xml.dt_atividade, 'MM/YYYY' )
)
WHERE CD_ATENDIMENTO_DIARIA = :CAMPO_CD_ATENDIMENTO
ORDER BY dt_periodo
        ,cd_setor
)
ORDER BY CD_TIP_ACOM
)

--ATENDIMENTO_DETALHADO_HONORARIO_1
SELECT NVL(valor, 0) valor,
       NVL(qt_lancamento_hon, 0 ) qt_lancamento_hon,
       NVL(vl_unitario_hon, 0 ) vl_unitario_hon,
       cd_pro_fat_hon,
       ds_pro_fat_hon,
       cd_atendimento_hon
 FROM (
SELECT SUM( NVL( valor, 0 )) valor
      ,SUM(DECODE( valor, 0, 0, 1 )) qt_lancamento_hon
      ,SUM( NVL( valor, 0 )) vl_unitario_hon
      ,a.cd_pro_fat_hon cd_pro_fat_hon
      ,a.ds_pro_fat_hon ds_pro_fat_hon
      ,a.cd_atendimento cd_atendimento_hon
  FROM (
        SELECT NVL(dbamv.fnc_fnrm_consulta_rep_atend( :CAMPO_CD_ATENDIMENTO ), 0) valor
              ,'FNRM-REPASSE' cd_pro_fat_hon
              ,'PON' ds_pro_fat_hon
              ,:CAMPO_CD_ATENDIMENTO cd_atendimento
          FROM DUAL
         WHERE EXISTS (SELECT 1 FROM config_fcch WHERE cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA AND NVL(tp_calc_honor_atend, 'D') = 'R')

                                ,p.cd_pro_fat

                            AND regra.cd_produto = p.cd_produto )) bf

        ,a.vl_unitario vl_unitario
        ,a.ds_pro_fat ds_pro_fat
        ,a.cd_atendimento cd_atendimento_receita
        ,SUM( a.qt_lancamento ) qt_lancamento_receita
        ,SUM( a.vl_total_conta ) vl_total
    FROM ( SELECT  tmp.cd_atendimento
                  ,DECODE( tmp.cd_pro_fat, NULL, ps.cd_procedimento, pro_fat.cd_pro_fat ) cd_pro_fat
                  ,DECODE( tmp.cd_pro_fat, NULL, ps.ds_procedimento, pro_fat.ds_pro_fat ) ds_pro_fat
                  ,SUM( tmp.qt_lancamento ) qt_lancamento
                  ,( SUM( NVL( tmp.vl_total_conta, 0 )) / sum( nvl(tmp.qt_lancamento,0) )) vl_unitario
                  ,SUM( NVL( tmp.vl_total_conta, 0 )) vl_total_conta
              FROM dbamv.temp_ffch_receita_unitaria tmp
                  ,dbamv.pro_fat
                  ,dbamv.procedimento_sus ps
             WHERE tmp.cd_atendimento = :CAMPO_CD_ATENDIMENTO
                   AND tmp.cd_pro_fat = pro_fat.cd_pro_fat(+)
                   AND tmp.cd_procedimento_sus = ps.cd_procedimento(+)
                   AND NVL(tmp.sn_pertence_pacote,'N') = 'N' AND NVL(tmp.sn_pacote,'N') = 'N'
          GROUP BY tmp.cd_atendimento
                  ,DECODE( tmp.cd_pro_fat, NULL, ps.cd_procedimento, pro_fat.cd_pro_fat )
                  ,DECODE( tmp.cd_pro_fat, NULL, ps.ds_procedimento, pro_fat.ds_pro_fat )
           UNION
           SELECT  to_char(atendimento_xml.cd_atendimento_mv) cd_atendimento
                  ,atividade_xml.cd_atividade_gen cd_pro_fat
                  ,item_atividade.nm_item_atividade ds_pro_fat
                  ,SUM( atividade_xml.qt_realizada ) qt_lancamento
                  ,SUM(NVL( atividade_xml.vl_atividade, 0)*nvl(atividade_xml.qt_realizada,0))/SUM(nvl(atividade_xml.qt_realizada,0)) vl_unitario
                  ,SUM( NVL( atividade_xml.vl_atividade, 0 ) * nvl(atividade_xml.qt_realizada,0) ) vl_total_conta
              FROM dbamv.atendimento_xml,
                   dbamv.atividade_xml,

		           dbamv.atendime
             WHERE atendimento_xml.cd_atendimento_mv = :CAMPO_CD_ATENDIMENTO
                   AND atendime.cd_atendimento = atendimento_xml.cd_atendimento_mv
                   AND atividade_xml.cd_atendimento_gen = atendimento_xml.cd_atendimento_gen
                   AND atividade_xml.cd_atividade_gen = item_atividade.cd_atividade
                   AND atendime.tp_atendimento NOT IN ('S','A')
                   AND atividade_xml.tp_atividade IN ( '1', '3', '4', '5', '6', '7', '8', '9', '10', '12', '14')
                   AND NVL( atividade_xml.vl_atividade, 0 ) <> 0
          GROUP BY atendimento_xml.cd_atendimento_mv
                  ,atividade_xml.cd_atividade_gen
                  ,item_atividade.nm_item_atividade
            UNION ALL
           SELECT  to_char(atendimento_xml.cd_atendimento_mv) cd_atendimento
                  ,atividade_xml.cd_atividade_gen cd_pro_fat
                  ,item_atividade.nm_item_atividade ds_pro_fat
                  ,SUM( atividade_xml.qt_realizada ) qt_lancamento
                  ,SUM(NVL( atividade_xml.vl_atividade, 0)) / SUM(nvl(atividade_xml.qt_realizada,1)) vl_unitario

              FROM dbamv.atendimento_xml,
                   dbamv.atividade_xml,
                   dbamv.item_atividade,
                   dbamv.atendime
             WHERE atendimento_xml.cd_atendimento_mv = :CAMPO_CD_ATENDIMENTO

                   AND atividade_xml.cd_atendimento_gen = atendimento_xml.cd_atendimento_gen

                   AND atendime.tp_atendimento IN ('S','A') AND atividade_xml.tp_atividade = '15'
                   AND NVL( atividade_xml.vl_atividade, 0 ) <> 0
          GROUP BY atendimento_xml.cd_atendimento_mv
                  ,atividade_xml.cd_atividade_gen
                  ,item_atividade.nm_item_atividade
           UNION ALL
           SELECT '1916' cd_atendimento, 'abc' cd_pro_fat, 'abcdef' ds_pro_fat, 2 qt_lancamento, 10 vl_unitario, 20 vl_total_conta FROM sys.dual
         ) a
GROUP BY a.cd_pro_fat
        ,a.vl_unitario
        ,a.ds_pro_fat
        ,a.cd_atendimento
ORDER BY a.ds_pro_fat
)
WHERE CD_ATENDIMENTO_RECEITA = :CAMPO_CD_ATENDIMENTO

            dbamv.setor
          WHERE atendimento_xml.cd_atendimento_mv = :CAMPO_CD_ATENDIMENTO
            AND atividade_xml.cd_atendimento_gen  = atendimento_xml.cd_atendimento_gen
            AND item_atividade.cd_atividade       = atividade_xml.cd_atividade_gen
            AND setor.cd_setor                    = atividade_xml.cd_setor_produziu_mv
            AND atividade_xml.tp_atividade        = 7
       GROUP BY item_atividade.nm_item_atividade,
            setor.nm_setor,
            setor.cd_setor,
            atendimento_xml.cd_atendimento_mv,
            atividade_xml.dt_atividade,
            atividade_xml.cd_atividade_gen
        ) resumo_2
    ) geral
  WHERE geral.cd_atendimento_cirurgia = :CAMPO_CD_ATENDIMENTO
ORDER BY geral.dt_realizacao,
    geral.ds_sal_cir,
    geral.nm_setor_cirurgia

--ATENDIMENTO_DETALHADO_TERCEIRO_1
SELECT geral.cd_pro_fat_terc,
    geral.ds_pro_fat_terc,
    geral.cd_setor_terc,
    geral.cd_atendimento_terc,
    geral.dt_lancamento_terc,
    (
    CASE
        WHEN geral.dt_lancamento_terc IS NOT NULL
            AND geral.cd_pro_fat_terc IS NOT NULL
        THEN
            Decode(
			       Sign(NVL(dbamv.pack_ffch.custo_procedimento(geral.dt_lancamento_terc, geral.cd_pro_fat_terc, geral.cd_setor_terc, :CAMPO_CD_MULTI_EMPRESA), 0)),
			       -1,
			        0,
			       NVL(dbamv.pack_ffch.custo_procedimento(geral.dt_lancamento_terc, geral.cd_pro_fat_terc, geral.cd_setor_terc,:CAMPO_CD_MULTI_EMPRESA), 0 )
			       )
        ELSE 0
    END) vl_custo_terc,
    (SELECT TO_CHAR(NVL(MAX(dt_competencia), to_date(geral.dt_lancamento_terc, 'mm/yyyy')), 'mm/yyyy')
       FROM dbamv.custo_unitario
      WHERE cd_setor        = geral.cd_setor_terc
        AND cd_pro_fat      = geral.cd_pro_fat_terc
        AND dt_competencia <= geral.dt_lancamento
     ) data_referencia
   FROM
    (SELECT itreg_fat.cd_pro_fat cd_pro_fat_terc,
        pro_fat.ds_pro_fat ds_pro_fat_terc,
        nvl(itreg_fat.cd_setor_produziu, itreg_fat.cd_setor) cd_setor_terc,
        reg_fat.cd_atendimento cd_atendimento_terc,
        to_char(itreg_fat.dt_lancamento, 'mm/yyyy') dt_lancamento_terc,
        itreg_fat.dt_lancamento dt_lancamento
       FROM dbamv.itreg_fat,
        dbamv.reg_fat,
        dbamv.pro_fat,
        dbamv.setor,
        dbamv.grupo_de_custo,
        dbamv.gru_pro
      WHERE reg_fat.cd_atendimento                               = :CAMPO_CD_ATENDIMENTO
        AND reg_fat.cd_multi_empresa                             = :CAMPO_CD_MULTI_EMPRESA
        AND reg_fat.cd_reg_fat                                   = itreg_fat.cd_reg_fat
        AND nvl(itreg_fat.cd_setor_produziu, itreg_fat.cd_setor) = setor.cd_setor
        AND setor.cd_grupo_de_custo                              = grupo_de_custo.cd_grupo_de_custo
        AND grupo_de_custo.tp_grupo_de_custo                     = 'T'
        AND itreg_fat.cd_pro_fat                                 = pro_fat.cd_pro_fat(+)
        AND pro_fat.cd_gru_pro                                   = gru_pro.cd_gru_pro(+)
        AND nvl(gru_pro.tp_custo, 1)                            IN(1, 3)
  UNION ALL
     SELECT itreg_amb.cd_pro_fat cd_pro_fat_terc,
        pro_fat.ds_pro_fat ds_pro_fat_terc,
        nvl(itreg_amb.cd_setor_produziu, itreg_amb.cd_setor) cd_setor_terc,
        itreg_amb.cd_atendimento cd_atendimento_terc,
        to_char(itreg_amb.hr_lancamento, 'mm/yyyy') dt_lancamento_terc,
        itreg_amb.hr_lancamento dt_lancamento
       FROM dbamv.itreg_amb,
        dbamv.reg_amb,
        dbamv.setor,
        dbamv.pro_fat,
        dbamv.grupo_de_custo,
        dbamv.gru_pro
      WHERE itreg_amb.cd_atendimento                             = :CAMPO_CD_ATENDIMENTO
        AND reg_amb.cd_multi_empresa                             = :CAMPO_CD_MULTI_EMPRESA
        AND reg_amb.cd_reg_amb                                   = itreg_amb.cd_reg_amb
        AND nvl(itreg_amb.cd_setor_produziu, itreg_amb.cd_setor) = setor.cd_setor
        AND setor.cd_grupo_de_custo                              = grupo_de_custo.cd_grupo_de_custo
        AND grupo_de_custo.tp_grupo_de_custo                     = 'T'
        AND itreg_amb.cd_pro_fat                                 = pro_fat.cd_pro_fat(+)
        AND pro_fat.cd_gru_pro                                   = gru_pro.cd_gru_pro(+)
        AND nvl(gru_pro.tp_custo, 1)                            IN(1, 3)
    ) geral
  WHERE geral.cd_atendimento_terc = :CAMPO_CD_ATENDIMENTO
  ORDER BY geral.dt_lancamento, geral.ds_pro_fat_terc

SELECT atividade_xml.cd_atividade_gen Cd_pro_fat_terc
      ,item_atividade.nm_item_atividade  DS_PRO_fAT_TERC
      ,atividade_xml.Cd_setor_produziu_mv Cd_setor_terc
      ,atendimento_xml.cd_atendimento_mv Cd_atendimento_terc
      ,TO_CHAR( atividade_xml.dt_atividade, 'mm/yyyy' ) Dt_lancamento_terc
      ,Decode(
              Sign(nvl(dbamv.pack_ffch.custo_procedimento(TO_CHAR( atividade_xml.dt_atividade, 'mm/yyyy' ),
                                                          null,
                                                          atividade_xml.cd_setor_produziu_mv,
                                                          :CAMPO_CD_MULTI_EMPRESA,
                                                          atividade_xml.cd_atividade_gen),0)),
              -1,
               0,
                   nvl(dbamv.pack_ffch.custo_procedimento(TO_CHAR( atividade_xml.dt_atividade, 'mm/yyyy' ),
                                                          null,
                                                          atividade_xml.cd_setor_produziu_mv,
                                                          :CAMPO_CD_MULTI_EMPRESA,
                                                          atividade_xml.cd_atividade_gen),0)
              ) vl_custo_terc,
    (SELECT TO_CHAR(NVL(MAX(dt_competencia), sysdate), 'mm/yyyy')
       FROM dbamv.custo_unitario
      WHERE cd_setor        = atividade_xml.cd_setor_produziu_mv
        AND cd_atividade    = atividade_xml.cd_atividade_gen
        AND dt_competencia <= atividade_xml.dt_atividade
     ) data_referencia
  FROM  dbamv.atendimento_xml,
        dbamv.atividade_xml,
        dbamv.item_atividade
 WHERE  atendimento_xml.cd_atendimento_gen = atividade_xml.cd_atendimento_gen
   AND  item_atividade.cd_atividade = atividade_xml.cd_atividade_gen
   AND  atendimento_xml.cd_atendimento_mv = :CAMPO_CD_ATENDIMENTO
   AND  atividade_xml.tp_atividade = '4'
ORDER BY atividade_xml.dt_atividade, item_atividade.nm_item_atividade
