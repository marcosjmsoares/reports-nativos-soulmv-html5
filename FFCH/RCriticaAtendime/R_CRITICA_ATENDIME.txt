--R_CRITICA_ATENDIME
SELECT  ate.cd_atendimento        cd_atendimento
                ,paciente.nm_paciente    nm_paciente
                ,trunc((sysdate - paciente.dt_nascimento)/365.25,0)  idade
                ,ate.dt_atendimento         dt_atendimento
                ,ate.dt_alta                       dt_alta
                ,ate.cd_prestador             cd_prestador
                ,ate.cd_especialid            cd_especialid
                ,ate.cd_pro_int                 cd_pro_int
    FROM  dbamv.atendime ate
               ,dbamv.paciente
 WHERE  ate.cd_paciente = paciente.cd_paciente
       AND  ate.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
ORDER BY nm_paciente
