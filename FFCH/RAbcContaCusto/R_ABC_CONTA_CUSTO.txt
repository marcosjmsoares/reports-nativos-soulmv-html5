--R_ABC_CONTA_CUSTO
SELECT * FROM (
select setor.nm_setor nm_setor,
          item_res.ds_item_res,
          to_char(custo_primario.dt_competencia, 'yyyy/mm') dt_comp,
          sum(custo_primario.vl_custo_setor) valor,
          decode('','D', setor.nm_setor,to_char(setor.cd_setor,'00000000')) nr_ordem_setor,
          decode('','A', item_res.ds_item_res,to_char(item_res.nr_ordem_item,'0000')) nr_ordem
from   dbamv.setor,
          dbamv.item_res,
          dbamv.custo_primario
where to_char(custo_primario.dt_competencia,'yyyy/mm') >= to_char(:CAMPO_DT_INI, 'yyyy/mm') and
           to_char(custo_primario.dt_competencia,'yyyy/mm') <= to_char(:CAMPO_DT_FIM, 'yyyy/mm') and
          custo_primario.cd_setor = setor.cd_setor and
          custo_primario.cd_item_res = item_res.cd_item_res and
          setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
group by  setor.nm_setor,
          item_res.ds_item_res,
          to_char(custo_primario.dt_competencia, 'yyyy/mm'),
          decode('','D', setor.nm_setor,to_char(setor.cd_setor,'00000000')) ,
          decode('','A', item_res.ds_item_res,to_char(item_res.nr_ordem_item,'0000'))
)
ORDER BY NR_ORDEM_SETOR,NM_SETOR,
NR_ORDEM,DS_ITEM_RES,
DT_COMP
