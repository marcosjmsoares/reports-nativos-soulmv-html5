--R_SINTETICO_CENCIR
select cd_setor,
          nvl(vl_custo_evento,0) custo_prim,
          nvl(vl_custo_genericos,0) vl_gen,
          nvl(vl_custo_especificos,0) vl_esp
from dbamv.custo_final
where dt_competencia = to_date(:CAMPO_DT_COMP, 'mm/yyyy') and
           custo_final.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA and
           cd_setor in (select distinct cd_setor from dbamv.sal_cir)
