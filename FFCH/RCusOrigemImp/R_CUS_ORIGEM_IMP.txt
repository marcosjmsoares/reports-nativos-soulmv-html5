--R_CUS_ORIGEM_IMP
SELECT * FROM (
SELECT   setor.cd_setor cd_setor
       , setor.nm_setor nm_setor
       , NVL (setor.nr_ordem_rateio, 9999) nr_ordem_rateio
       , gru_res.cd_gru_res
       , gru_res.ds_gru_res
       , NVL (gru_res.nr_ordem_grupo, 9999) nr_ordem_gru_res
       , item_res.ds_item_res ds_item_res
       , item_res.cd_item_res cd_item_res
       , ratcon_pag.cd_recebe cd_recebe
       , ratcon_pag.nr_meses_amortizacao
       , ratcon_pag.vl_rateio vl_rateio
       , ratcon_pag.dt_competencia dt_competencia
       , ratcon_pag.cd_log_importacao_custos
    FROM dbamv.item_res
       , dbamv.setor
       , dbamv.gru_res
       , dbamv.ratcon_pag
       , dbamv.recebe_conta_custo recebe
   WHERE TO_CHAR (ratcon_pag.dt_competencia, 'MM/YYYY') = TO_CHAR (:CAMPO_PCOMP, 'MM/YYYY')
     AND ratcon_pag.cd_setor    = setor.cd_setor
     AND ratcon_pag.cd_item_res = item_res.cd_item_res
     AND setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
     AND ratcon_pag.cd_recebe   = recebe.cd_recebe
     AND (ratcon_pag.cd_recebe = :CAMPO_CD_RECEBE OR :CAMPO_CD_RECEBE = '%')
     AND gru_res.cd_gru_res     = item_res.cd_gru_res
ORDER BY nm_setor, ds_gru_res, ds_item_res, cd_recebe
)
ORDER BY NM_SETOR,CD_SETOR,NR_ORDEM_RATEIO,
DS_GRU_RES,CD_GRU_RES,NR_ORDEM_GRU_RES,
DS_ITEM_RES,CD_ITEM_RES,
CD_RECEBE,CD_LOG_IMPORTACAO_CUSTOS
