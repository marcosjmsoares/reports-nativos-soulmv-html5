--R_RESULTADO_SETOR
SELECT * FROM (
select setor.nm_setor                               nm_setor,
          setor.nr_ordem_rateio                     nr_ordem_rateio,
          setor.cd_cen_cus                          	cd_cen_cus,
          custo_final.cd_setor                       cd_setor,
          setor.sn_deducao_honorario         sn_deducao_honorario,
          grupo_de_custo.ds_grupo_de_custo,
          (nvl(custo_final.vl_custo_primario,0)+
          nvl(custo_final.vl_custo_genericos,0)+
          nvl(custo_final.vl_custo_especificos,0))   vl_custo,
          nvl(custo_final.vl_receita_sh,0)                vl_receita_sh,
          nvl(custo_final.vl_receita_honorarios,0)   vl_receita_honorarios,
          nvl(custo_final.vl_receita_sd,0)                vl_receita_sd,
          nvl(custo_final.vl_receita_matmed,0)       vl_receita_matmed,
          nvl(custo_final.vl_receita_no,0)               vl_receita_no
  from dbamv.setor,
          dbamv.custo_final,
          dbamv.grupo_de_custo
where to_char(dt_competencia,'mm/yyyy') = :CAMPO_DT_INI and
          setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA and
          setor.cd_setor = custo_final.cd_setor  and
          setor.cd_grupo_de_custo = grupo_de_custo.cd_grupo_de_custo (+) and
          grupo_de_custo.tp_grupo_de_custo not in ('A','D')
:CAMPO_CF_GRUPO_DE_CUSTO
)
ORDER BY
DECODE('',
                'A',
                    NM_SETOR,
                'R',
                    nr_ordem_rateio
                ,
                    cd_cen_cus)
