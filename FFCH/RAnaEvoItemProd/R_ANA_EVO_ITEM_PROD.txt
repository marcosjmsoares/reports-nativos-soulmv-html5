--R_ANA_EVO_ITEM_PROD
SELECT * FROM (
select
           to_date( to_char( producao_mensal.dt_competencia, 'mm/yyyy'), 'mm/yyyy') dt_comp,
           item_prod_mensal.cd_item_producao  cd_item_producao  ,
           item_producao.ds_item_producao  ds_item_producao,
           item_prod_mensal.cd_setor_destino  cd_setor_destino,
           setor.nm_setor destino ,
           SUM(item_prod_mensal.vl_producao_mensal) valor
  from
           dbamv.producao_mensal,
           dbamv.setor,
           dbamv.item_prod_mensal,
           dbamv.item_producao
where producao_mensal.dt_competencia between to_date(:CAMPO_DT_INI,'mm/yyyy') and to_date(:CAMPO_DT_FIM,'mm/yyyy')
   and setor.cd_multi_empresa                           = :CAMPO_CD_MULTI_EMPRESA
   and producao_mensal.cd_producao_mensal = item_prod_mensal.cd_producao_mensal
   and item_prod_mensal.cd_item_producao     = item_producao.cd_item_producao
   and item_prod_mensal.cd_setor_destino       = setor.cd_setor
   and producao_mensal.cd_setor_origem        is not null
Group by            to_date( to_char( producao_mensal.dt_competencia, 'mm/yyyy'), 'mm/yyyy') ,
           item_prod_mensal.cd_item_producao ,
           item_producao.ds_item_producao,
           item_prod_mensal.cd_setor_destino,
           setor.nm_setor
order by
         to_date( to_char( producao_mensal.dt_competencia, 'mm/yyyy'), 'mm/yyyy'),
         item_producao.ds_item_producao,
         setor.nm_setor
)
ORDER BY DS_ITEM_PRODUCAO,DESTINO,CD_ITEM_PRODUCAO,
CD_SETOR_DESTINO
