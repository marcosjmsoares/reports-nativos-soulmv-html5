--R_ORIGEM_CONTA_CUS
SELECT * FROM (
SELECT item_res.cd_item_res,
       item_res.ds_item_res,
       item_res.cd_item_res||' - '||item_res.ds_item_res item_res,
       recebe_conta_custo.cd_recebe,
        SUM(vl_custo_setor) vl_custo_conta
  FROM dbamv.item_res,
       dbamv.custo_primario,
       dbamv.recebe_conta_custo
 WHERE TRUNC(custo_primario.dt_competencia,'MM') = TRUNC(:CAMPO_DT_INICIAL,'MM')
   AND item_res.cd_item_res = recebe_conta_custo.cd_item_res
   AND item_res.cd_item_res = custo_primario.cd_item_res
GROUP BY item_res.cd_item_res,
       item_res.ds_item_res,
       recebe_conta_custo.cd_recebe
ORDER BY ds_item_res
)
ORDER BY CD_ITEM_RES,DS_ITEM_RES,ITEM_RES,VL_CUSTO_CONTA
