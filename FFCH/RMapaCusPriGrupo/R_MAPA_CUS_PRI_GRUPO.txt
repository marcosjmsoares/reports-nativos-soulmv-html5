--R_MAPA_CUS_PRI_GRUPO
SELECT * FROM (
SELECT   nr_ordem_grupo
        ,setor.cd_grupo_de_custo
        ,Initcap(grupo_de_custo.ds_grupo_de_custo) ds_grupo_de_custo
        ,Initcap(item_res.ds_item_res)                 ds_item_res
        ,sum(custo_primario.vl_custo_setor)        vl_custo_setor
FROM    dbamv.item_res,
        dbamv.setor,
        dbamv.custo_primario,
        dbamv.grupo_de_custo
WHERE   setor.cd_setor                                       =  custo_primario.cd_setor
and     custo_primario.cd_item_res                           =  item_res.cd_item_res
and     grupo_de_custo.cd_grupo_de_custo = setor.cd_grupo_de_custo
and      setor.cd_multi_empresa              = :CAMPO_CD_MULTI_EMPRESA
and      TO_CHAR( custo_primario .dt_competencia, 'MM/YYYY' ) = to_char(:CAMPO_PCOMP,'mm/yyyy')
group by  nr_ordem_grupo
        ,setor.cd_grupo_de_custo
        ,grupo_de_custo.ds_grupo_de_custo
        ,item_res.ds_item_res
)
ORDER BY NR_ORDEM_GRUPO,DS_ITEM_RES,CD_GRUPO_DE_CUSTO
