--R_ANA_CUSTO_SETOR
select setor.cd_setor     cd_setor,
           setor.nm_setor    nm_setor,
           setor.cd_cen_cus   cd_cen_cus
   from dbamv.setor
where setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
  and exists    (SELECT DISTINCT retorno FROM (
                select 1 retorno
                from dbamv.custo_primario
                 where custo_primario.dt_competencia = to_date(:CAMPO_DT_COMP,'mm/yyyy')
                   and custo_primario.vl_custo_setor <> 0
                UNION ALL
                select 1 retorno
                   from dbamv.custo_absorvido
                  where custo_absorvido.dt_competencia = to_date(:CAMPO_DT_COMP,'mm/yyyy')
                    and custo_absorvido.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
                    and custo_absorvido.vl_custo_absorvido <> 0
                UNION ALL
                select 1 retorno
                  from dbamv.ind_produtividade ip
                 where ip.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
                    and ip.dt_competencia = to_date(:CAMPO_DT_COMP,'mm/yyyy')
                ))

--R_ANA_CUSTO_SETOR_R_CLA_ABSORCAO_Q_3
SELECT * FROM (
SELECT * FROM (
 select destino.cd_setor   cd_setor_destino,
        custo_absorvido.cd_setor_origem,
        Initcap(origem.nm_setor)          origem,
        decode(origem.classificacao, 'G',:CAMPO_TRAD_GEN, 'E', :CAMPO_TRAD_ESP) cla_origem,
        custo_absorvido.vl_custo_absorvido vlr_origem,
        ds_multi_empresa ds_multi_empresa
   from dbamv.custo_absorvido,
        dbamv.setor   destino,
        dbamv.setor   origem,
        dbamv.multi_empresas
  where custo_absorvido.cd_setor_destino   = destino.cd_setor
    and custo_absorvido.cd_setor_origem    = origem.cd_setor
    and destino.cd_multi_empresa           = multi_empresas.cd_multi_empresa
    and origem.cd_multi_empresa            = multi_empresas.cd_multi_empresa
    and destino.cd_multi_empresa           = :CAMPO_CD_MULTI_EMPRESA
    and custo_absorvido.dt_competencia = to_date(:CAMPO_DT_COMP,'mm/yyyy')
    and custo_absorvido.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
    and destino.cd_setor = :CAMPO_CD_SETOR
    and custo_absorvido.vl_custo_absorvido <> 0
  order by  origem.nm_setor,
           custo_absorvido.cd_setor_destino,
           decode(origem.classificacao, 'G',:CAMPO_TRAD_GEN, 'E', :CAMPO_TRAD_ESP),
           custo_absorvido.cd_setor_origem,
           custo_absorvido.vl_custo_absorvido,
           ds_multi_empresa
)
WHERE CD_SETOR_DESTINO = :CAMPO_CD_SETOR
)
ORDER BY CD_SETOR_DESTINO,CLA_ORIGEM,origem, cd_setor_origem, vlr_origem

--R_ANA_CUSTO_SETOR_R_GRU_RES_Q_2
SELECT * FROM (
SELECT * FROM (
select custo_primario.cd_setor                cd_setor_primario
       ,gru_res.cd_gru_res          cd_gru_res
      ,gru_res.ds_gru_res          ds_gru_res
      ,custo_primario.cd_item_res  cd_item_res
      ,item_res.ds_item_res        ds_item_res
      ,sum(vl_custo_setor)         vl_custo_setor
,gru_res.nr_ordem_grupo nr_ordem_grupo
  from dbamv.custo_primario
      ,dbamv.item_res
      ,dbamv.gru_res
      ,dbamv.setor
 where custo_primario.dt_competencia = to_date(:CAMPO_DT_COMP, 'mm/yyyy')
   and custo_primario.cd_setor = setor.cd_setor
   and setor.cd_setor = :CAMPO_CD_SETOR
   and setor.cd_multi_empresa  = :CAMPO_CD_MULTI_EMPRESA
   and custo_primario.cd_item_res = item_res.cd_item_res
   and item_res.cd_gru_res = gru_res.cd_gru_res
   and custo_primario.vl_custo_setor <> 0
 group by custo_primario.cd_setor
         ,gru_res.cd_gru_res
         ,gru_res.ds_gru_res
         ,custo_primario.cd_item_res
         ,item_res.ds_item_res
,gru_res.nr_ordem_grupo
:CAMPO_CF_ORDEM_GRU_RES
)
WHERE CD_SETOR_PRIMARIO = :CAMPO_CD_SETOR
)
ORDER BY
DECODE('',
        '1',
          TO_CHAR(cd_gru_res,'0009'),
        '2',
	      ds_gru_res,
	    '3',
	      TO_CHAR(nr_ordem_grupo,'00009')
)
