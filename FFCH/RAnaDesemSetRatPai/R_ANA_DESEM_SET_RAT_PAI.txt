--R_ANA_DESEM_SET_RAT_PAI
SELECT filho.cd_setor_custo
      ,pai.nm_setor nm_pai
      ,filho.cd_setor  cd_setor_filho
      ,initcap (filho.nm_setor) nm_setor
from dbamv.setor pai,
     dbamv.setor filho
where pai.cd_setor = :CAMPO_SETOR_PAI and
      filho.cd_setor_custo = :CAMPO_SETOR_PAI and
      pai.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
ORDER BY CD_SETOR_CUSTO,NM_PAI


/*----------------------------------------------------------------------------*/
--R_ANA_DESEM_SET_RAT_PAI_R_17_Q_2
SELECT   custo_absorvido.cd_setor_origem
        ,initcap (origem.nm_setor) origem
        ,decode (origem.classificacao
                ,'G', 'P_TRAD_GENERICOS'
                ,'E', 'P_TRAD_ESPECIFICOS'
                ,'', 'P_TRAD_NAO_INFORMADO'
                ) cla_origem
        ,SUM (custo_absorvido.vl_custo_absorvido) vlr_origem
        ,destino.cd_setor_custo
    FROM dbamv.custo_absorvido
        ,dbamv.setor destino
        ,dbamv.setor origem
   WHERE custo_absorvido.cd_setor_destino = destino.cd_setor
     AND custo_absorvido.cd_setor_origem = origem.cd_setor
     AND destino.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
     AND custo_absorvido.dt_competencia = to_date(:CAMPO_DT_COMP, 'mm/yyyy')
     AND cd_setor_destino IN (SELECT cd_setor
                                FROM dbamv.setor
                               WHERE cd_setor_custo = :CAMPO_SETOR_PAI)
     AND destino.CD_SETOR_CUSTO = :CAMPO_CD_SETOR_CUSTO
     AND custo_absorvido.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
GROUP BY destino.cd_setor_custo
        ,custo_absorvido.cd_setor_origem
        ,origem.nm_setor
        ,decode (origem.classificacao
                ,'G', 'P_TRAD_GENERICOS'
                ,'E', 'P_TRAD_ESPECIFICOS'
                ,'', 'P_TRAD_NAO_INFORMADO'
                )
ORDER BY decode (origem.classificacao
                ,'G', 'P_TRAD_GENERICOS'
                ,'E', 'P_TRAD_ESPECIFICOS'
                ,'', 'P_TRAD_NAO_INFORMADO'
                )
        ,vlr_origem DESC
        ,custo_absorvido.cd_setor_origem
        ,destino.cd_setor_custo
        ,origem
