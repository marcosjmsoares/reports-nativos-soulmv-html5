--R_CONTA_CUSTO_GRAF
SELECT * FROM (
select * from (select ds_item_res,
       nm_setor,
       to_char(dt_competencia,'mm/yyyy') dt_comp,
       sum(vl_custo_setor) valor
from dbamv.custo_primario,
     dbamv.setor,
     dbamv.item_res
where setor.cd_setor = custo_primario.cd_setor and
      item_res.cd_item_res = custo_primario.cd_item_res and
      to_char(dt_competencia,'mm/yyyy')= :CAMPO_DT_INI and
      setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
group by ds_item_res,
       nm_setor,
       to_char(dt_competencia,'mm/yyyy') )
order by valor desc
)
ORDER BY DT_COMP,DS_ITEM_RES,VALOR DESC,NM_SETOR
