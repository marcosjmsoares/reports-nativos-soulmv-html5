--R_LEVANTAMENTO_CONVENIO
SELECT cd_convenio,
       nm_convenio,
       mes,
       qt_atende,
       (SELECT COUNT(*) dia
          FROM DBAMV.ATENDIME ATE
         WHERE trunc(ate.dt_atendimento) BETWEEN To_Date('01/'||mes, 'dd/mm/yyyy')
                                    AND Last_Day(To_Date('01/'||mes, 'dd/mm/yyyy'))
                                           ) total_mes
  FROM (SELECT CONVENIO.NM_CONVENIO
              ,CONVENIO.CD_CONVENIO
              ,to_char(ate.DT_ATENDIMENTO,'MM/YYYY') MES
              ,COUNT(*)  QT_ATENDE
          FROM DBAMV.ATENDIME ate
              ,DBAMV.CONVENIO
        WHERE TO_CHAR(ATE.DT_ATENDIMENTO,'YYYY/MM') BETWEEN TO_CHAR(to_date(:CAMPO_DT_INI,'YYYY/MM'), 'YYYY/MM') AND TO_CHAR(to_date(:CAMPO_DT_FIM,'YYYY/MM'), 'YYYY/MM')
           AND ate.CD_CONVENIO = CONVENIO.CD_CONVENIO
        GROUP BY CONVENIO.NM_CONVENIO
                ,CONVENIO.CD_CONVENIO
                ,to_char(ate.DT_ATENDIMENTO,'MM/YYYY'))
GROUP BY cd_convenio,
       nm_convenio, mes, qt_atende
ORDER BY  NM_CONVENIO,CD_CONVENIO,MES
