--R_ESTMOTENT
SELECT
        dbamv.ate_motivo.cd_mot_ent,
        NVL(dbamv.mot_ent.ds_mot_ent,'Motivo não declarado') ds_mot_ent,
        count(*) qtd_ent
 FROM
       dbamv.atendime
       ,dbamv.des_ate
       ,dbamv.prestador
       ,dbamv.ori_ate
       ,dbamv.paciente
       ,dbamv.convenio
       ,dbamv.ate_motivo
       ,dbamv.mot_ent
 WHERE
             dbamv.atendime.cd_des_ate     = dbamv.des_ate.cd_des_ate(+)
AND     dbamv.atendime.cd_prestador   = dbamv.prestador.cd_prestador (+)
AND     dbamv.atendime.cd_ori_ate     = dbamv.ori_ate.cd_ori_ate
AND     dbamv.atendime.cd_convenio    = dbamv.convenio.cd_convenio
AND     dbamv.atendime.cd_paciente    = dbamv.paciente.cd_paciente
AND     dbamv.atendime.cd_atendimento = dbamv.ate_motivo.cd_atendimento(+)
AND     dbamv.ate_motivo.cd_mot_ent   = dbamv.mot_ent.cd_mot_ent(+)
AND     dbamv.ori_ate.tp_origem     = 'U'
AND     dbamv.atendime.cd_atendimento_pai IS NULL
AND     dbamv.atendime.dt_atendimento BETWEEN
to_date(to_char(:CAMPO_DTINIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND to_date(to_char(:CAMPO_DTFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
and       dbamv.atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
GROUP BY
        dbamv.ate_motivo.cd_mot_ent
        ,NVL(dbamv.mot_ent.ds_mot_ent,'Motivo não declarado')
ORDER BY qtd_ent DESC
