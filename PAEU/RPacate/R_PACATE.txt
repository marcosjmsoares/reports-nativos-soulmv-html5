--R_PACATE
SELECT ATEN.DT_ATENDIMENTO
      ,ATEN.CD_ORI_ATE
      ,ORIG.DS_ORI_ATE
      ,ATEN.CD_PRESTADOR
      ,PRES.NM_PRESTADOR
      ,ATEN.CD_ATENDIMENTO
      ,PACI.NM_PACIENTE
      ,(SELECT P.NM_SOCIAL_PACIENTE
			  FROM DBAMV.PACIENTE P
			  WHERE P.CD_PACIENTE = PACI.CD_PACIENTE
				  AND NVL(P.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE
      ,CART.NR_CARTEIRA
      ,CONV.NM_CONVENIO
      ,TIPO.DS_TIPO_INTERNACAO
FROM DBAMV.ATENDIME ATEN
    ,DBAMV.ORI_ATE ORIG
    ,DBAMV.PACIENTE PACI
    ,DBAMV.CARTEIRA CART
    ,DBAMV.CONVENIO CONV
    ,DBAMV.PRESTADOR PRES
    ,DBAMV.TIPO_INTERNACAO TIPO
WHERE ATEN.DT_ATENDIMENTO BETWEEN TO_DATE(TO_CHAR(:CAMPO_DTINIC, 'DD/MM/YYYY') || ' 00', 'DD/MM/YYYY HH24') 
    AND TO_DATE(TO_CHAR(:CAMPO_DTFIM, 'DD/MM/YYYY') || ' 23', 'DD/MM/YYYY HH24')
  AND ATEN.TP_ATENDIMENTO = 'U'
  AND ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
  AND ATEN.CD_ORI_ATE   = ORIG.CD_ORI_ATE (+)
  AND ATEN.CD_PACIENTE  = PACI.CD_PACIENTE
  AND ATEN.CD_PACIENTE  = CART.CD_PACIENTE (+)
  AND ATEN.CD_CON_PLA = CART.CD_CON_PLA(+)
  AND ATEN.CD_CONVENIO  = CART.CD_CONVENIO (+)
  AND ATEN.CD_CONVENIO  = CONV.CD_CONVENIO (+)
  AND ATEN.CD_PRESTADOR = PRES.CD_PRESTADOR (+)
  AND ATEN.NR_CARTEIRA = CART.NR_CARTEIRA (+)
  AND ATEN.CD_TIPO_INTERNACAO = TIPO.CD_TIPO_INTERNACAO(+)
ORDER BY ATEN.DT_ATENDIMENTO,
         ORIG.DS_ORI_ATE,
         PRES.NM_PRESTADOR,
         PACI.NM_PACIENTE
