--R_ESTDESTATE
SELECT
        dbamv.atendime.cd_des_ate,
        decode(dbamv.des_ate.ds_des_ate , Null, 'NÃO INFORMADO', dbamv.des_ate.ds_des_ate) ds_des_ate,
        count(*) qtd_des_ate
 FROM
       dbamv.atendime     ,
       dbamv.des_ate     ,
       dbamv.prestador     ,
       dbamv.ori_ate     ,
       dbamv.paciente     ,
       dbamv.convenio
 WHERE
        dbamv.atendime.cd_des_ate   = dbamv.des_ate.cd_des_ate(+)
AND     dbamv.atendime.cd_prestador = dbamv.prestador.cd_prestador (+)
AND     atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
AND     dbamv.atendime.cd_ori_ate   = dbamv.ori_ate.cd_ori_ate
AND     dbamv.atendime.cd_convenio  = dbamv.convenio.cd_convenio
AND     dbamv.atendime.cd_paciente  = dbamv.paciente.cd_paciente
AND     dbamv.atendime.dt_atendimento BETWEEN to_date(to_char(:CAMPO_DTINIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND to_date(to_char(:CAMPO_DTFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
GROUP BY  dbamv.atendime.cd_des_ate,
                     dbamv.des_ate.ds_des_ate
ORDER BY  qtd_des_ate DESC
