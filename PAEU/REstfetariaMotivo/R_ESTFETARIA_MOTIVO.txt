--R_ESTFETARIA_MOTIVO
SELECT
       SUM(PACK_INTERNAMENTO.F_RETORNA_F_HORARIO( RTRIM( LTRIM( SUBSTR( TO_CHAR( (SYSDATE - PACIENTE.DT_NASCIMENTO) / 365 , '999.99'), 1,4 ) ) )  , TO_CHAR(:CAMPO_HRINIC1) , TO_CHAR(:CAMPO_HRFIM1) )) Faixa1,
       SUM(PACK_INTERNAMENTO.F_RETORNA_F_HORARIO( RTRIM( LTRIM( SUBSTR( TO_CHAR( (SYSDATE - PACIENTE.DT_NASCIMENTO) / 365 , '999.99'), 1,4 ) ) )  , TO_CHAR(:CAMPO_HRINIC2) , TO_CHAR(:CAMPO_HRFIM2) )) Faixa2,
       SUM(PACK_INTERNAMENTO.F_RETORNA_F_HORARIO( RTRIM( LTRIM( SUBSTR( TO_CHAR( (SYSDATE - PACIENTE.DT_NASCIMENTO) / 365 , '999.99'), 1,4 ) ) )  , TO_CHAR(:CAMPO_HRINIC3) , TO_CHAR(:CAMPO_HRFIM3) )) Faixa3,
       SUM(PACK_INTERNAMENTO.F_RETORNA_F_HORARIO( RTRIM( LTRIM( SUBSTR( TO_CHAR( (SYSDATE - PACIENTE.DT_NASCIMENTO) / 365 , '999.99'), 1,4 ) ) )  , TO_CHAR(:CAMPO_HRINIC4) , TO_CHAR(:CAMPO_HRFIM4) )) Faixa4,
       SUM(PACK_INTERNAMENTO.F_RETORNA_F_HORARIO( RTRIM( LTRIM( SUBSTR( TO_CHAR( (SYSDATE - PACIENTE.DT_NASCIMENTO) / 365 , '999.99'), 1,4 ) ) )  , TO_CHAR(:CAMPO_HRINIC5) , TO_CHAR(:CAMPO_HRFIM5) )) Faixa5,
       SUM(PACK_INTERNAMENTO.F_RETORNA_F_HORARIO( RTRIM( LTRIM( SUBSTR( TO_CHAR( (SYSDATE - PACIENTE.DT_NASCIMENTO) / 365 , '999.99'), 1,4 ) ) )  , TO_CHAR(:CAMPO_HRINIC6) , TO_CHAR(:CAMPO_HRFIM6) )) Faixa6,
       SUM(PACK_INTERNAMENTO.F_RETORNA_F_HORARIO( RTRIM( LTRIM( SUBSTR( TO_CHAR( (SYSDATE - PACIENTE.DT_NASCIMENTO) / 365 , '999.99'), 1,4 ) ) )  , TO_CHAR(:CAMPO_HRINIC7) , TO_CHAR(:CAMPO_HRFIM7) )) Faixa7,
       SUM(PACK_INTERNAMENTO.F_RETORNA_F_HORARIO( RTRIM( LTRIM( SUBSTR( TO_CHAR( (SYSDATE - PACIENTE.DT_NASCIMENTO) / 365 , '999.99'), 1,4 ) ) )  , TO_CHAR(:CAMPO_HRINIC8) , TO_CHAR(:CAMPO_HRFIM8) )) Faixa8
FROM
       dbamv.atendime    ,
       dbamv.ate_motivo  ,
       dbamv.mot_ent  ,
       dbamv.paciente
 WHERE atendime.dt_atendimento BETWEEN
to_date(to_char(:CAMPO_DTINIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND to_date(to_char(:CAMPO_DTFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND      atendime.cd_atendimento = ate_motivo.cd_atendimento(+)
AND      atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
AND      ate_motivo.cd_mot_ent   = mot_ent.cd_mot_ent(+)
 AND     atendime.cd_paciente    = paciente.cd_paciente
