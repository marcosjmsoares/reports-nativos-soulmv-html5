--R_TRIAGEM
SELECT ATENDIME.CD_ATENDIMENTO        CD_ATENDIMENTO
             ,PACIENTE.NM_PACIENTE                  NM_PACIENTE
             ,DECODE(PACIENTE.TP_SEXO,'M','MASCULINO','F','FEMININO','AMBOS')   TP_SEXO
             ,TRUNC(((TRUNC(SYSDATE) - TRUNC(PACIENTE.DT_NASCIMENTO)) / 365.25))   IDADE
             ,TRIAGEM.DT_TRIAGEM                      DT_TRIAGEM
             ,to_char(TRIAGEM.HR_TRIAGEM,'HH24')   HR_TRIAGEM
             ,TRIAGEM.NR_PA_INI                           NR_PA_INI
             ,TRIAGEM.NR_PA_FIM                         NR_PA_FIM
             ,TRIAGEM.NR_PULSO                          NR_PULSO
             ,TRIAGEM.NR_TEMPERATURA           NR_TEMPERATURA
             ,TRIAGEM.DS_QUEIXA_PRINCIPAL     DS_QUEIXA_PRINCIPAL
             ,TRIAGEM.SN_PRIMEIRO_ATENDE     SN_PRIMEIRO_ATENDE
             ,TRIAGEM.SN_RETORNO_QUEIXA      SN_RETORNO_QUEIXA
             ,TRIAGEM.DS_DIABETES                      DS_DIABETES
             ,TRIAGEM.DS_HIPERTENSAO              DS_HIPERTENSAO
             ,TRIAGEM.DS_ULCERA                         DS_ULCERA
             ,TRIAGEM.DS_ASMA                             DS_ASMA
             ,TRIAGEM.DS_OUTROS                        DS_OUTROS
             ,TRIAGEM.DS_ALERGIA                        DS_ALERGIA
             ,nvl(TRIAGEM.DS_OBSERVACAO,'NÃO TEM OBSERVAÇÃO')               DS_OBSERVACAO
             ,PRESTADOR.NM_PRESTADOR          NM_PRESTADOR
             ,ESPECIALID.DS_ESPECIALID              DS_ESPECIALID
FROM   DBAMV.ATENDIME
            ,DBAMV.ESPECIALID
            ,DBAMV.PRESTADOR
            ,DBAMV.PACIENTE
            ,DBAMV.TRIAGEM
WHERE  ATENDIME.CD_ESPECIALID = ESPECIALID.CD_ESPECIALID(+)
AND        ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
AND        ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND        ATENDIME.CD_ATENDIMENTO = TRIAGEM.CD_ATENDIMENTO
AND        ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND        TRIAGEM.CD_TRIAGEM = :CAMPO_PCD_TRIAGEM
