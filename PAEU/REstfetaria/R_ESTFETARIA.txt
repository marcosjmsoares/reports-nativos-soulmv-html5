--R_ESTFETARIA
SELECT  DBAMV.RETORNA_FAIXA_ETARIA_RELATORIO( PACIENTE.DT_NASCIMENTO, 'D' )   FAIXA,
               COUNT(*)   QTDE
FROM    dbamv.atendime
       ,dbamv.des_ate
       ,dbamv.prestador
       ,dbamv.ori_ate
       ,dbamv.paciente
       ,dbamv.convenio
WHERE
             dbamv.atendime.cd_des_ate     = dbamv.des_ate.cd_des_ate(+)
AND     dbamv.atendime.cd_prestador   = dbamv.prestador.cd_prestador (+)
AND     dbamv.atendime.cd_ori_ate     = dbamv.ori_ate.cd_ori_ate
AND     dbamv.atendime.cd_convenio    = dbamv.convenio.cd_convenio
AND     dbamv.atendime.cd_paciente    = dbamv.paciente.cd_paciente
AND     dbamv.atendime.cd_atendimento_pai IS NULL
AND     ATENDIME.DT_ATENDIMENTO BETWEEN to_date(to_char(:CAMPO_DTINIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND to_date(to_char(:CAMPO_DTFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
and       atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
GROUP BY DBAMV.RETORNA_FAIXA_ETARIA_RELATORIO( PACIENTE.DT_NASCIMENTO, 'D' )
ORDER BY QTDE DESC
