--R_CIDP
SELECT  COUNT(*) COUNT_CID
       ,DECODE(ATENDIME.CD_CID, null,  '0', ATENDIME.CD_CID) CD_CID
       ,DECODE(CID.DS_CID, null, 'NÃO INFORMADO', CID.DS_CID)  DS_CID
FROM    Dbamv.ATENDIME ATENDIME
       ,Dbamv.CID CID
WHERE CID.CD_CID(+) = ATENDIME.CD_CID AND
      ATENDIME.DT_ATENDIMENTO BETWEEN to_date(to_char(:CAMPO_DT_INIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND to_date(to_char(:CAMPO_DT_FIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY ATENDIME.CD_CID, CID.DS_CID
ORDER BY  CID.DS_CID
