--R_COMP_MOV_HOSP
select atendime.cd_atendimento
      ,decode(mot_alt.cd_mot_alt||' - '||mot_alt.ds_mot_alt,' - ',decode(tip_res.cd_tip_res||' - '||tip_res.ds_tip_res,' - ',null,tip_res.cd_tip_res||' - '||tip_res.ds_tip_res),mot_alt.cd_mot_alt||' - '||mot_alt.ds_mot_alt) cd_nm_mot_alt
      ,to_char(atendime.dt_atendimento,'DD/MM/YYYY') dt_atendimento
      ,to_char(atendime.hr_atendimento,'HH24') hr_atendimento
      ,to_char(atendime.dt_alta,'DD/MM/YYYY') dt_alta
      ,to_char(atendime.hr_alta,'HH24') hr_alta
      ,paciente.cd_paciente||' - '||paciente.nm_paciente cd_nm_paciente
      ,convenio.cd_convenio||' - '||convenio.nm_convenio cd_nm_convenio
      ,ori_ate.cd_ori_ate||' - '||ori_ate.ds_ori_ate cd_nm_ori_ate
      ,setor.cd_setor||' - '||setor.nm_setor cd_nm_setor
      ,con_pla.cd_con_pla||' - '||con_pla.ds_con_pla cd_nm_con_pla
      ,mov_hosp.cd_mov_hosp cd_mov_hosp
      ,to_char(mov_hosp.dt_mov_hosp,'DD/MM/YYYY') dt_mov_hosp
      ,mov_hosp.cd_tip_situacao cd_tip_situacao
      ,mov_hosp.cd_des_ate cd_des_ate
      ,mov_hosp.cd_tip_situacao cd_tip_situacao
      ,mov_hosp.cd_loc_trans cd_loc_trans
      ,mov_hosp.cd_mot_canc cd_mot_canc
	  ,(SELECT P.NM_SOCIAL_PACIENTE
FROM DBAMV.PACIENTE P
WHERE P.CD_PACIENTE = paciente.CD_PACIENTE 
	AND NVL(P.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE
from dbamv.atendime
      ,dbamv.paciente
      ,dbamv.convenio
      ,dbamv.ori_ate
      ,dbamv.mot_alt
      ,dbamv.setor
      ,dbamv.con_pla
      ,dbamv.mov_hosp
      ,dbamv.tip_res
	  ,Dbamv.Empresa_Con_Pla
Where Empresa_Con_Pla.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
	And Empresa_Con_Pla.Cd_Convenio = Con_Pla.Cd_Convenio
	And Empresa_Con_Pla.Cd_Con_Pla = Con_Pla.Cd_Con_Pla
	And atendime.cd_convenio = convenio.cd_convenio
   	and atendime.cd_paciente = paciente.cd_paciente
   	and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
   	and atendime.cd_con_pla = con_pla.cd_con_pla
   	and con_pla.cd_convenio = convenio.cd_convenio
   	and atendime.cd_ori_ate = ori_ate.cd_ori_ate
   	and atendime.cd_mot_alt = mot_alt.cd_mot_alt(+)
   	and ori_ate.cd_setor = setor.cd_setor(+)
   	and mov_hosp.cd_atendimento = atendime.cd_atendimento
   	and atendime.cd_tip_res = tip_res.cd_tip_res(+)
   	and mov_hosp.cd_mov_hosp = :CAMPO_CD_MOV_HOSP
