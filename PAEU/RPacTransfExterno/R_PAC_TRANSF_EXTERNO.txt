--R_PAC_TRANSF_EXTERNO
SELECT * FROM (
  SELECT PACIENTE.CD_PACIENTE       CD_PACIENTE
        ,PACIENTE.NM_PACIENTE       NM_PACIENTE
        ,(SELECT P.NM_SOCIAL_PACIENTE
			    FROM DBAMV.PACIENTE P
			    WHERE P.CD_PACIENTE = PACIENTE.CD_PACIENTE
				    AND NVL(P.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE
        ,ATENDIME.CD_ATENDIMENTO    CD_ATENDIMENTO
        ,ATENDIME.DT_ATENDIMENTO    DT_ATENDIMENTO
        ,ATENDIME.HR_ATENDIMENTO    HR_ATENDIMENTO
        ,ESPECIALID.DS_ESPECIALID   CD_ESPECIALID
        ,LOC_TRANS.DS_LOC_TRANS     DS_LOC_TRANS
        ,MOV_HOSP.DT_MOV_HOSP       DT_MOV_HOSP
        ,MOV_HOSP.HR_MOV_HOSP       HR_MOV_HOSP
  FROM DBAMV.ATENDIME
      ,DBAMV.PACIENTE
      ,DBAMV.ESPECIALID
      ,DBAMV.MOV_HOSP
      ,DBAMV.LOC_TRANS
  WHERE ATENDIME.DT_ATENDIMENTO BETWEEN to_date(to_char(:CAMPO_DATINI, 'dd/mm/yyyy') || ' 00', 'dd/mm/yyyy HH24') 
    AND to_date(to_char(:CAMPO_DATFIM, 'dd/mm/yyyy') || ' 23', 'dd/mm/yyyy HH24')
    AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
    AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
    AND ATENDIME.CD_ESPECIALID = ESPECIALID.CD_ESPECIALID(+)
    AND ATENDIME.CD_ATENDIMENTO = MOV_HOSP.CD_ATENDIMENTO
    AND MOV_HOSP.CD_LOC_TRANS = LOC_TRANS.CD_LOC_TRANS
  ORDER BY ATENDIME.DT_ATENDIMENTO DESC
          ,MOV_HOSP.HR_MOV_HOSP
          ,PACIENTE.NM_PACIENTE
)
ORDER BY DT_MOV_HOSP
