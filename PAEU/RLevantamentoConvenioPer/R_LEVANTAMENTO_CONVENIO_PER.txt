--R_LEVANTAMENTO_CONVENIO_PER
SELECT * FROM (
SELECT CONVENIO.NM_CONVENIO
              ,CONVENIO.CD_CONVENIO
              ,TRUNC (ATENDIME.DT_ATENDIMENTO)  MES
	            ,TO_NUMBER(TO_CHAR(ATENDIME.DT_ATENDIMENTO, 'DD'))  MESG
              ,COUNT(*)  QT_ATENDE
              ,a.dia
  FROM   DBAMV.ATENDIME
        ,DBAMV.CONVENIO
        ,(SELECT a.dt_atendimento, COUNT(*) dia
            FROM DBAMV.ATENDIME A
           GROUP BY a.dt_atendimento) a
WHERE ATENDIME.DT_ATENDIMENTO BETWEEN to_date(to_char(:CAMPO_PER_INI,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND to_date(to_char(:CAMPO_PER_FIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND       ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND a.dt_atendimento = atendime.dt_atendimento
GROUP BY CONVENIO.NM_CONVENIO
        ,CONVENIO.CD_CONVENIO
        ,TRUNC (ATENDIME.DT_ATENDIMENTO)
	,TO_NUMBER(TO_CHAR(ATENDIME.DT_ATENDIMENTO, 'DD')), a.dia
)
ORDER BY MESG,NM_CONVENIO,CD_CONVENIO
