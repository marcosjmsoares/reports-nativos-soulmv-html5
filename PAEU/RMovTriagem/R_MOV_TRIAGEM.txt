--R_MOV_TRIAGEM
SELECT TRUNC(TRIAGEM.DT_TRIAGEM)              DT_TRIAGEM
      ,TO_CHAR(TRIAGEM.HR_TRIAGEM, 'HH24') HR_TRIAGEM
      ,CONVENIO.CD_CONVENIO                   CD_CONVENIO
      ,CONVENIO.NM_CONVENIO                   NM_CONVENIO
      ,ATENDIME.CD_ATENDIMENTO                CD_ATENDIMENTO
      ,PACIENTE.CD_PACIENTE                   CD_PACIENTE
      ,PACIENTE.NM_PACIENTE                   NM_PACIENTE
      ,(SELECT P.NM_SOCIAL_PACIENTE
			  FROM DBAMV.PACIENTE P
			  WHERE P.CD_PACIENTE = PACIENTE.CD_PACIENTE
				  AND NVL(P.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE
      ,ESPECIALID.CD_ESPECIALID               CD_ESPECIALID
      ,ESPECIALID.DS_ESPECIALID               DS_ESPECIALID
      ,TRIAGEM.NM_USUARIO
FROM DBAMV.ATENDIME
    ,DBAMV.PACIENTE
    ,DBAMV.CONVENIO
    ,DBAMV.ESPECIALID
    ,DBAMV.TRIAGEM
WHERE  TRIAGEM.DT_TRIAGEM BETWEEN TO_DATE(TO_CHAR(:CAMPO_DATINI, 'DD/MM/YYYY') || ' 00', 'DD/MM/YYYY HH24')
    AND TO_DATE(TO_CHAR(:CAMPO_DATFIM, 'DD/MM/YYYY') || ' 23', 'DD/MM/YYYY HH24')
  AND ATENDIME.CD_ATENDIMENTO   = TRIAGEM.CD_ATENDIMENTO
  AND ATENDIME.CD_PACIENTE      = PACIENTE.CD_PACIENTE
  AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
  AND ATENDIME.CD_CONVENIO      = CONVENIO.CD_CONVENIO
  AND ATENDIME.CD_ESPECIALID    = ESPECIALID.CD_ESPECIALID(+)
ORDER BY TRIAGEM.HR_TRIAGEM
        ,PACIENTE.NM_PACIENTE
