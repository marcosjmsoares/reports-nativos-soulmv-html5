--F_PROCEDIMENTOS_ORIGEM
SELECT * FROM (
SELECT   ORI_ATE.DS_ORI_ATE
         ,DECODE(CONVENIO.TP_CONVENIO,
                         'C', 'CONVÊNIO',
                         'P', 'PARTICULAR',
                         'H', 'SIH/SUS',
                         'A', 'SIA/SUS'  , 'ZERO') TP_CONVENIO
         ,ESPECIALID.DS_ESPECIALID DS_ESPECIALID
         ,PRO_FAT.DS_PRO_FAT
         ,COUNT(*) QTD
FROM     DBAMV.ATENDIME
         ,DBAMV.ESPECIALID
         ,DBAMV.CONVENIO
         ,DBAMV.PRO_FAT
         ,DBAMV.ORI_ATE
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
And    ATENDIME.CD_ESPECIALID  =  ESPECIALID.CD_ESPECIALID
AND   ATENDIME.CD_CONVENIO    =  CONVENIO.CD_CONVENIO
AND   ATENDIME.CD_PRO_INT     =  PRO_FAT.CD_PRO_FAT
AND   ORI_ATE.CD_ORI_ATE      = ATENDIME.CD_ORI_ATE
AND   ATENDIME.DT_ATENDIMENTO between to_date(to_char(:CAMPO_DTINIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND to_date(to_char(:CAMPO_DTFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
GROUP BY ORI_ATE.DS_ORI_ATE
         ,DECODE(CONVENIO.TP_CONVENIO,
                         'C', 'CONVÊNIO',
                         'P', 'PARTICULAR',
                         'H', 'SIH/SUS',
                         'A', 'SIA/SUS'  , 'ZERO')
         ,ESPECIALID.DS_ESPECIALID
         ,PRO_FAT.DS_PRO_FAT
ORDER BY DS_ORI_ATE, TP_CONVENIO, DS_ESPECIALID,DS_PRO_FAT
)
ORDER BY DS_ORI_ATE,
TP_CONVENIO,
DS_ESPECIALID
