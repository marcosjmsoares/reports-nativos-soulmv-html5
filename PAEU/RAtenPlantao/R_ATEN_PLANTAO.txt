--R_ATEN_PLANTAO
SELECT PRES.CD_PRESTADOR,
       PRES.NM_PRESTADOR,
       ATEN.CD_ATENDIMENTO,
       PACI.NM_PACIENTE,
       PACI.CD_PACIENTE,
       ATEN.DT_ATENDIMENTO,
       ATEN.HR_ATENDIMENTO,
       COPL.DS_CON_PLA,
       CONV.NM_CONVENIO,
       PROC.DS_PRO_FAT,
       (SELECT P.NM_SOCIAL_PACIENTE
			 FROM DBAMV.PACIENTE P
			 WHERE P.CD_PACIENTE = PACI.CD_PACIENTE
				 AND NVL(P.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE
FROM DBAMV.ATENDIME     ATEN,
       DBAMV.PACIENTE   PACI,
       DBAMV.CON_PLA    COPL,
       DBAMV.PRESTADOR  PRES,
       DBAMV.CONVENIO   CONV,
       DBAMV.PRO_FAT    PROC
WHERE ATEN.DT_ATENDIMENTO between to_date(to_char(:CAMPO_DATINI,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') 
  AND to_date(to_char(:CAMPO_DATFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24') 
  AND ATEN.HR_ATENDIMENTO BETWEEN TO_DATE(to_char(:CAMPO_DATINI,'dd/mm/yyyy')||' '||:CAMPO_HORINI, 'dd/mm/yyyy HH24') AND TO_DATE(to_char(:CAMPO_DATFIM,'dd/mm/yyyy')||' '||:CAMPO_HORFIM,'dd/mm/yyyy HH24')
  AND ATEN.CD_PACIENTE  =  PACI.CD_PACIENTE
  AND ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
  AND ATEN.CD_CONVENIO  =  CONV.CD_CONVENIO
  AND ATEN.CD_CONVENIO  =  COPL.CD_CONVENIO(+)
  AND ATEN.CD_CON_PLA   =  COPL.CD_CON_PLA(+)
  AND ATEN.CD_PRESTADOR =  PRES.CD_PRESTADOR(+)
  AND ATEN.CD_PRO_INT   =  PROC.CD_PRO_FAT(+)
ORDER BY PRES.CD_PRESTADOR,
         TO_CHAR(ATEN.DT_ATENDIMENTO,'DD/MM/YYYY'),
         TO_CHAR(ATEN.HR_ATENDIMENTO,'HH24MI')
