--R_ATEND_TP_RES
SELECT ATENDIME.CD_ATENDIMENTO CD_ATENDIMENTO,
      TO_CHAR(ATENDIME.DT_ATENDIMENTO,'DD/MM/YYYY') || ' ' || TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24') DATA,
      PACIENTE.CD_PACIENTE CD_PACIENTE,
      PACIENTE.NM_PACIENTE NM_PACIENTE,
      (SELECT P.NM_SOCIAL_PACIENTE
			 FROM DBAMV.PACIENTE P
			 WHERE P.CD_PACIENTE = PACIENTE.CD_PACIENTE
				 AND NVL(P.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE,
      CONVENIO.NM_CONVENIO NM_CONVENIO,
      CON_PLA.DS_CON_PLA DS_CON_PLA,
      ORI_ATE.DS_ORI_ATE DS_ORI_ATE,
      PRESTADOR.NM_PRESTADOR NM_PRESTADOR,
      TIP_RES.DS_TIP_RES,
      TIPO_INTERNACAO.DS_TIPO_INTERNACAO
FROM DBAMV.ATENDIME,
    DBAMV.PACIENTE,
    DBAMV.ORI_ATE,
    DBAMV.CONVENIO,
    DBAMV.CON_PLA,
    DBAMV.PRESTADOR,
    DBAMV.TIP_RES,
    DBAMV.TIPO_INTERNACAO
WHERE ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
  AND ATENDIME.CD_PACIENTE      = PACIENTE.CD_PACIENTE
  AND ATENDIME.CD_ORI_ATE       = ORI_ATE.CD_ORI_ATE
  AND ATENDIME.CD_CONVENIO      = CONVENIO.CD_CONVENIO
  AND ATENDIME.CD_CON_PLA       = CON_PLA.CD_CON_PLA(+)
  AND ATENDIME.CD_PRESTADOR     = PRESTADOR.CD_PRESTADOR
  AND ATENDIME.CD_CONVENIO      = CON_PLA.CD_CONVENIO(+)
  AND TO_DATE(TO_CHAR(ATENDIME.DT_ATENDIMENTO,'DD/MM/YYYY'),'DD/MM/YYYY HH24') BETWEEN TO_DATE(:CAMPO_DT_INICIAL||' 00','DD/MM/YYYY HH24') 
  AND TO_DATE(:CAMPO_DT_FINAL||' 23','DD/MM/YYYY HH24')
  AND TIP_RES.CD_TIP_RES(+)     = ATENDIME.CD_TIP_RES
  AND TIPO_INTERNACAO.CD_TIPO_INTERNACAO(+) = ATENDIME.CD_TIPO_INTERNACAO
  AND ((ATENDIME.TP_ATENDIMENTO = 'I' AND CD_ATENDIMENTO_PAI IS NULL) OR ATENDIME.TP_ATENDIMENTO <> 'I')
ORDER BY ATENDIME.CD_ATENDIMENTO, 
         ATENDIME.DT_ATENDIMENTO, 
         ATENDIME.HR_ATENDIMENTO
