--R_FICHA_TRIAGEM
SELECT TRIAGEM.CD_TRIAGEM ,
       TRIAGEM.CD_ATENDIMENTO ,
       ATENDIME.DT_ATENDIMENTO ,
       ATENDIME.HR_ATENDIMENTO ,
       ATENDIME.TP_PRIORIDADE ,
       ATENDIME.CD_PACIENTE ,
       PACIENTE.NM_PACIENTE ,
       (
              SELECT P.NM_SOCIAL_PACIENTE
              FROM   DBAMV.PACIENTE P
              WHERE  P.CD_PACIENTE = PACIENTE.CD_PACIENTE
              AND    Nvl(P.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE ,
       PACIENTE.TP_SEXO ,
       PACIENTE.DT_NASCIMENTO ,
       ATENDIME.CD_PRESTADOR ,
       PRESTADOR.NM_PRESTADOR ,
       ATENDIME.CD_ESPECIALID ,
       ESPECIALID.DS_ESPECIALID ,
       ATENDIME.CD_ORI_ATE ,
       ORI_ATE.DS_ORI_ATE ,
       ATENDIME.CD_SERVICO ,
       SERVICO.DS_SERVICO ,
       TRIAGEM.NM_USUARIO ,
       Decode(TRIAGEM.SN_PRIMEIRO_ATENDE,
              'S','X',
              NULL) SN_PRIMEIRO_ATENDE_S ,
       Decode(TRIAGEM.SN_PRIMEIRO_ATENDE,
              'N','X',
              NULL) SN_PRIMEIRO_ATENDE_N ,
       Decode(TRIAGEM.SN_RETORNO_QUEIXA,
              'S','X',
              NULL) SN_RETORNO_QUEIXA_S ,
       Decode(TRIAGEM.SN_RETORNO_QUEIXA,
              'N','X',
              NULL) SN_RETORNO_QUEIXA_N ,
       TRIAGEM.DS_QUEIXA_PRINCIPAL ,
       TRIAGEM.DS_DIABETES ,
       TRIAGEM.DS_ULCERA ,
       TRIAGEM.DS_HIPERTENSAO ,
       TRIAGEM.DS_ASMA ,
       TRIAGEM.DS_OUTROS ,
       TRIAGEM.DS_ALERGIA ,
       TRIAGEM.NR_PA_INI ,
       TRIAGEM.NR_PA_FIM ,
       TRIAGEM.NR_PULSO ,
       TRIAGEM.NR_TEMPERATURA ,
       TRIAGEM.DS_OBSERVACAO ,
       Decode(TRIAGEM.TP_NASCIMENTO,
              'D','Distócito',
              'E','Eutócito',
              NULL) TP_NASCIMENTO ,
       TRIAGEM.NR_MENARCA ,
       TRIAGEM.NR_ULT_MENSTRUACAO ,
       TRIAGEM.NR_PARTOS ,
       TRIAGEM.NR_PRI_RELACAO_SEXUAL ,
       Decode(TRIAGEM.TP_HIGIENE_BUCAL,
              'R','Ruim',
              'G','Regular',
              'B','Boa',
              NULL) TP_HIGIENE_BUCAL ,
       Decode(TRIAGEM.SN_PROTESE_DENTARIA,
              'S','X',
              NULL) SN_PROTESE_DENTARIA_S ,
       Decode(TRIAGEM.SN_PROTESE_DENTARIA,
              'N','X',
              NULL) SN_PROTESE_DENTARIA_N ,
       Decode(TRIAGEM.SN_BEBE_ALCOOL,
              'S','X',
              NULL) SN_BEBE_ALCOOL_S ,
       Decode(TRIAGEM.SN_BEBE_ALCOOL,
              'N','X',
              NULL) SN_BEBE_ALCOOL_N ,
       TRIAGEM.NR_ANOS_BEBE_ALCOOL ,
       TRIAGEM.NR_MESES_BEBE_ALCOOL ,
       Decode(TRIAGEM.SN_FUMANTE,
              'S','X',
              NULL) SN_FUMANTE_S ,
       Decode(TRIAGEM.SN_FUMANTE,
              'N','X',
              NULL) SN_FUMANTE_N ,
       TRIAGEM.NR_ANOS_FUMANTE ,
       TRIAGEM.NR_MESES_FUMANTE ,
       Decode(TRIAGEM.SN_DOR,
              'S','X',
              NULL) SN_DOR_S ,
       Decode(TRIAGEM.SN_DOR,
              'N','X',
              NULL) SN_DOR_N ,
       TRIAGEM.NR_ANOS_DOR ,
       TRIAGEM.NR_MESES_DOR ,
       Decode(TRIAGEM.SN_CANCER_FAMILIA,
              'S','X',
              NULL) SN_CANCER_FAMILIA_S ,
       Decode(TRIAGEM.SN_CANCER_FAMILIA,
              'N','X',
              NULL) SN_CANCER_FAMILIA_N ,
       TRIAGEM.CD_TIP_PAREN ,
       TIP_PAREN.DS_TIP_PAREN ,
       Decode(TRIAGEM.SN_CONSULTOU_MEDICO,
              'S','X',
              NULL) SN_CONSULTOU_MEDICO_S ,
       Decode(TRIAGEM.SN_CONSULTOU_MEDICO,
              'N','X',
              NULL) SN_CONSULTOU_MEDICO_N ,
       TRIAGEM.NR_ANOS_CONSULTA ,
       TRIAGEM.NR_MESES_CONSULTA ,
       TRIAGEM.CD_CIDADE ,
       CIDADE.NM_CIDADE ,
       CIDADE.CD_UF ,
       TRIAGEM.DS_HOSPITAL ,
       Decode(TRIAGEM.SN_TRATAMENTO,
              'S','X',
              NULL) SN_TRATAMENTO_S ,
       Decode(TRIAGEM.SN_TRATAMENTO,
              'N','X',
              NULL) SN_TRATAMENTO_N ,
       TRIAGEM.DS_TRATAMENTO ,
       TRIAGEM.DS_MEDICAMENTOS ,
       TRIAGEM.DS_ANTECEDENTES ,
       TRIAGEM.DS_HISTORICO_DOENCA ,
       Decode(PACIENTE.SN_UTIL_IDENTIDADE_GENERO,
              'S',Decode(
                          (
                          SELECT Max(1)
                          FROM   DBASGU.CONFIG_ITEM_MODULO
                          WHERE  CD_MODULO = 'CAD_PAC'
                          AND    SN_VISIBLE = 'S'
                          AND    CD_ITEM = 'DSP_DS_IDENTIDADE_GENERO'
                          AND    CD_BLOCO = 'PACIENTE') ,
                         1,IDENTIDADE_GENERO.DS_IDENTIDADE_GENERO ,
                         NULL),
              NULL) AS IDENTIDADE_GENERO ,
       Decode(PACIENTE.SN_UTIL_ORIENTACAO_SEXUAL,
              'S',Decode(
                          (
                          SELECT Max(1)
                          FROM   DBASGU.CONFIG_ITEM_MODULO
                          WHERE  CD_MODULO = 'CAD_PAC'
                          AND    SN_VISIBLE = 'S'
                          AND    CD_ITEM = 'DSP_DS_ORIENTACAO_SEXUAL'
                          AND    CD_BLOCO = 'PACIENTE' ),
                         1,ORIENTACAO_SEXUAL.DS_ORIENTACAO_SEXUAL,
                         NULL),
              NULL) AS ORIENTACAO_SEXUAL
FROM   DBAMV.TRIAGEM ,
       DBAMV.ATENDIME ,
       DBAMV.PACIENTE ,
       DBAMV.PRESTADOR ,
       DBAMV.ESPECIALID ,
       DBAMV.ORI_ATE ,
       DBAMV.SERVICO ,
       DBAMV.CIDADE ,
       DBAMV.TIP_PAREN ,
       DBAMV.IDENTIDADE_GENERO ,
       DBAMV.ORIENTACAO_SEXUAL
WHERE  TRIAGEM.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
AND    ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND    ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
AND    ATENDIME.CD_ESPECIALID = ESPECIALID.CD_ESPECIALID(+)
AND    ATENDIME.CD_ORI_ATE = ORI_ATE.CD_ORI_ATE
AND    ATENDIME.CD_SERVICO = SERVICO.CD_SERVICO(+)
AND    TRIAGEM.CD_CIDADE = CIDADE.CD_CIDADE(+)
AND    TRIAGEM.CD_TIP_PAREN = TIP_PAREN.CD_TIP_PAREN(+)
AND    PACIENTE.CD_ORIENTACAO_SEXUAL = ORIENTACAO_SEXUAL.CD_ORIENTACAO_SEXUAL(+)
AND    PACIENTE.CD_IDENTIDADE_GENERO = IDENTIDADE_GENERO.CD_IDENTIDADE_GENERO(+)
AND    TRIAGEM.CD_TRIAGEM = :CAMPO_PCD_TRIAGEM
