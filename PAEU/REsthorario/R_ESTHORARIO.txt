--R_ESTHORARIO
SELECT * FROM (
SELECT distinct
       faixa.cd_faixa_hor_rel,
       to_char(faixa.hr_ini,'hh24') h_ini,
       to_char(faixa.hr_fim,'hh24') h_fim,
       atendime.cd_des_ate,
       des_ate.ds_des_ate,
       atendime.dt_atendimento
FROM
       dbamv.atendime,
       dbamv.des_ate,
       dbamv.prestador,
       dbamv.faixa_hor_rel faixa,
       dbamv.ori_ate ori_ate
WHERE  atendime.cd_des_ate    =  des_ate.cd_des_ate
and    atendime.cd_prestador  =  prestador.cd_prestador (+)
and    atendime.cd_ori_ate = ori_ate.cd_ori_ate
and    atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and    atendime.tp_atendimento  =  'U'
and    atendime.dt_atendimento between to_date(to_char(:CAMPO_DTINIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND to_date(to_char(:CAMPO_DTFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
order by 1,6
)
ORDER BY DT_ATENDIMENTO,
CD_DES_ATE,DS_DES_ATE,
H_INI,H_FIM
