Select
      Atendime.Cd_Paciente                Cd_Paciente
      ,Atendime.Cd_Atendimento         Cd_Atendimento
      ,Atendime.Dt_Atendimento          Data
      ,Atendime.Hr_Atendimento          Hrora
      ,Atendime.Cd_Prestador              Cd_Prestador
      ,Atendime.Cd_Convenio              Cd_Convenio
      ,Atendime.Dt_Alta                        Dt_Alta
      ,Atendime.Hr_Alta                        Hr_Alta
      ,Atendime.Ds_Atendimento          Diagnostico
      ,Atendime.Cd_Ori_Ate                  Cd_Ori_Ate
      ,Atendime.Cd_Especialid              Cd_Especialid
      ,Paciente.nm_paciente                nm_paciente
      ,Paciente.ds_endereco                ds_endereco
      ,Fn_Idade (Paciente.Dt_nascimento)       Idade
      ,Paciente.nm_bairro                     nm_bairro
      ,Paciente.nr_cep                         nr_cep
      ,Paciente.nr_fone                        nr_fone
      ,Cidade.Nm_Cidade                     Nm_Cidade
      ,Cidade.Cd_UF                            Cd_UF
      ,Convenio.Nm_convenio             Nm_convenio
      ,Prestador.cd_prestador              Cd_prestador1
      ,Prestador.Nm_prestador             Nm_prestador
      ,Prestador.ds_Codigo_conselho  CRM
      ,Atendime.Cd_Cid || ' - ' || Cid.Ds_cid     Ds_cid
      ,Tip_Res.ds_tip_res                     Tipo_Resultado
      ,Con_pla.Ds_con_pla                   Plano_Convenio
From
        Dbamv.ATENDIME
       ,Dbamv.PACIENTE
       ,DbaMv.CIDADE
       ,DbaMv.CONVENIO
       ,Dbamv.PRESTADOR
       ,Dbamv.CID
       ,Dbamv.TIP_RES
       ,Dbamv.CON_PLA
Where
        Atendime.Cd_Paciente                    =  Paciente.Cd_Paciente
        And  Atendime.Cd_prestador            =  Prestador.cd_prestador
        And  Atendime.cd_convenio            =  Convenio.cd_convenio(+)
        And  Atendime.cd_tip_res                = Tip_Res.cd_tip_res(+)
        And  Atendime.Cd_cid                     =  Cid.Cd_cid (+)
        And  Paciente.cd_cidade                =  Cidade.cd_cidade(+)
        And  Atendime.Cd_convenio           =  Con_pla.Cd_Convenio(+)
        And  Atendime.Cd_con_pla             =  Con_pla.Cd_Con_pla(+)
