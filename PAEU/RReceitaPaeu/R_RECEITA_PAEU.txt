--R_RECEITA_PAEU
SELECT TEXTOS_ATD.CD_TEXTOS_ATD      CD_TEXTOS_ATD,
       'Para:' || ' ' || PACIENTE.NM_PACIENTE                         NM_PACIENTE,
       CASOS_ATD.CD_PRESTADOR                    CD_PRESTADOR,
       'Dr(a):' || ' ' || PRESTADOR.NM_PRESTADOR                NM_PRESTADOR,
       PRESTADOR.DS_CODIGO_CONSELHO   DS_CODIGO_CONSELHO,
       SYSDATE                                                     DATA_RECEITA,
       TEXTOS_ATD.DS_TEXTOS_ATD              DS_TEXTOS_ATD
FROM   DBAMV.TEXTOS_ATD,
       DBAMV.PRESTADOR,
       DUAL,
       DBAMV.PACIENTE,
       DBAMV.CASOS_ATD
WHERE TEXTOS_ATD.TP_TEXTOS_ATD    = 'R'
AND    TEXTOS_ATD.CD_ATENDIMENTO =  CASOS_ATD.CD_ATENDIMENTO
AND    TEXTOS_ATD.CD_CASOS_ATD      = CASOS_ATD.CD_CASOS_ATD
AND    CASOS_ATD.CD_PRESTADOR       =  PRESTADOR.CD_PRESTADOR
AND    CASOS_ATD.CD_PACIENTE            =  PACIENTE.CD_PACIENTE
