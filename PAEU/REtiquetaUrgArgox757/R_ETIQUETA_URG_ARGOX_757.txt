       '1' A
FROM DBAMV.CIDADE WHERE ROWNUM <= :CAMPO_ETIQ

--R_ETIQUETA_URG_ARGOX_757_RF_ETIQ2_Q_ATENDIMENTO
SELECT * FROM (
SELECT '1' A1,
       paci.NM_PACIENTE,
       paci.CD_PACIENTE,
       paci.nr_fone,
       aten.DT_ATENDIMENTO,
       aten.HR_ATENDIMENTO,
       paci.TP_SEXO,
       PACK_INTERNAMENTO.RETORNA_IDADE(paci.DT_NASCIMENTO, SYSDATE) IDADE,
      aten.CD_ATENDIMENTO,
       copl.DS_CON_PLA,
       cart.NM_EMPRESA,
       cart.NR_CARTEIRA,
       cart.NM_TITULAR,
       desa.DS_DES_ATE,
       guia.cd_senha,
       convenio.nm_convenio,
       DECODE(sub_plano.DS_SUB_PLANO, NULL, NULL,'/'||sub_plano.DS_SUB_PLANO)  DS_SUB_PLANO
  FROM DBAMV.ATENDIME aten,
       DBAMV.PACIENTE paci,
       DBAMV.CON_PLA  copl,
       DBAMV.CARTEIRA cart,
       DBAMV.DES_ATE desa,
       DBAMV.guia guia,
       DBAMV.convenio,
       DBAMV.sub_plano
,Dbamv.Empresa_Con_Pla
Where Empresa_Con_Pla.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And Empresa_Con_Pla.Cd_Convenio = CoPl.Cd_Convenio
And  aten.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
And Empresa_Con_Pla.Cd_Con_Pla = CoPl.Cd_Con_Pla
And aten.CD_PACIENTE = paci.CD_PACIENTE
   AND aten.CD_CONVENIO = copl.CD_CONVENIO(+)
   AND convenio.cd_convenio = aten.cd_convenio
   AND  aten.cd_atendimento = guia.cd_atendimento(+)
   AND aten.CD_CON_PLA  = copl.CD_CON_PLA (+)
   AND aten.CD_CONVENIO = cart.CD_CONVENIO(+)
   AND aten.CD_PACIENTE = cart.CD_PACIENTE(+)
   AND aten.NR_CARTEIRA = cart.NR_CARTEIRA(+)
   AND aten.CD_DES_ATE  = desa.CD_DES_ATE(+)
   AND aten.CD_CON_PLA = cart.CD_CON_PLA(+)
   AND aten.cd_convenio = sub_plano.cd_convenio(+)
   AND aten.cd_con_pla = sub_plano.cd_con_pla(+)
   AND aten.cd_sub_plano = sub_plano.cd_sub_plano(+)
group by        paci.NM_PACIENTE,
       paci.CD_PACIENTE,
       paci.nr_fone,
       aten.DT_ATENDIMENTO,
       aten.HR_ATENDIMENTO,
       paci.TP_SEXO,
       PACK_INTERNAMENTO.RETORNA_IDADE(paci.DT_NASCIMENTO, SYSDATE),
      aten.CD_ATENDIMENTO,
       copl.DS_CON_PLA,
       SUB_PLANO.DS_SUB_PLANO,
       cart.NM_EMPRESA,
       cart.NR_CARTEIRA,
       cart.NM_TITULAR,
       desa.DS_DES_ATE,
       guia.cd_senha,
       convenio.nm_convenio
)
WHERE A1 = :CAMPO_A
