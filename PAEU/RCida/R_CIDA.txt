--R_CIDA
SELECT * FROM (
SELECT  nm_cidade
              ,'' CD_UF
              ,COUNT(*)   QT_CIDADE
FROM    DBAMV.ATENDIME
             ,DBAMV.PACIENTE
WHERE  PACIENTE.CD_PACIENTE = ATENDIME.CD_PACIENTE
AND paciente.cd_cidade IS NULL
AND        ATENDIME.DT_ATENDIMENTO  BETWEEN  to_date(to_char(:CAMPO_DT_INIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND to_date(to_char(:CAMPO_DT_FIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
GROUP BY 0
              , 'NAO_INFORMADA'
                   ,''
UNION ALL
SELECT
               DISTINCT(CIDADE.NM_CIDADE) nm_cidade
              ,CIDADE.CD_UF
              ,COUNT(*)   QT_CIDADE
FROM    DBAMV.ATENDIME
             ,DBAMV.PACIENTE
             ,DBAMV.CIDADE
WHERE  PACIENTE.CD_PACIENTE = ATENDIME.CD_PACIENTE
AND        CIDADE.CD_CIDADE = PACIENTE.CD_CIDADE
AND        ATENDIME.DT_ATENDIMENTO  BETWEEN  to_date(to_char(:CAMPO_DT_INIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND to_date(to_char(:CAMPO_DT_FIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
GROUP BY
               CIDADE.NM_CIDADE
                   ,CIDADE.CD_UF
)
ORDER BY QT_CIDADE DESC
