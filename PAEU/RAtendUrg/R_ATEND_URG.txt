--R_ATEND_URG
SELECT ATENDIME.DT_ATENDIMENTO DT_ATENDIMENTO
      ,ATENDIME.HR_ATENDIMENTO HR_ATENDIMENTO
      ,ATENDIME.CD_ATENDIMENTO CD_ATENDIMENTO
      ,ATENDIME.CD_PACIENTE CD_PACIENTE
      ,PACIENTE.NM_PACIENTE NM_PACIENTE
      ,TRUNC(((TO_DATE(TO_CHAR(NVL(ATENDIME.DT_ALTA,SYSDATE),'DD/MM/YYYY') || ' ' || TO_CHAR(NVL(ATENDIME.HR_ALTA,SYSDATE),'HH24'),'DD/MM/YYYY HH24')
               -  TO_DATE(TO_CHAR(ATENDIME.DT_ATENDIMENTO,'DD/MM/YYYY') || ' ' || TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'),'DD/MM/YYYY HH24')) * 24)) PERMANENCIA
      ,ATENDIME.CD_CONVENIO CD_CONVENIO
      ,CONVENIO.NM_CONVENIO NM_CONVENIO
      ,ATENDIME.CD_CON_PLA
      ,ATENDIME.NR_CARTEIRA
      ,ATENDIME.CD_SERVICO
      ,SERVICO.DS_SERVICO
      ,(SELECT P.NM_SOCIAL_PACIENTE
FROM DBAMV.PACIENTE P
WHERE P.CD_PACIENTE = ATENDIME.CD_PACIENTE 
  AND NVL(P.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE
FROM DBAMV.ATENDIME
    ,DBAMV.PACIENTE
    ,DBAMV.CONVENIO
    ,DBAMV.SERVICO
WHERE ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
  AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
  AND ATENDIME.CD_SERVICO = SERVICO.CD_SERVICO(+)
  AND ATENDIME.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
  AND ATENDIME.TP_ATENDIMENTO = 'U'
ORDER BY PERMANENCIA DESC
