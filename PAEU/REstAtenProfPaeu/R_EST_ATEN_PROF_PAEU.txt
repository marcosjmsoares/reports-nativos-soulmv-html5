--R_EST_ATEN_PROF_PAEU
SELECT DECODE(PRESTADOR.CD_PRESTADOR, NULL, 0, PRESTADOR.CD_PRESTADOR)    CD_PRESTADOR
              ,DECODE(PRESTADOR.NM_PRESTADOR, NULL, 'MÉDICO NÃO INFORMADO', PRESTADOR.NM_PRESTADOR)   NM_PRESTADOR
             ,COUNT(*)     ATENDIMENTOS
FROM  DBAMV.ATENDIME
            ,DBAMV.PRESTADOR
WHERE TRUNC(ATENDIME.DT_ATENDIMENTO)  BETWEEN 
to_date(to_char(:CAMPO_DATINI,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND to_date(to_char(:CAMPO_DATFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND       ATENDIME.TP_ATENDIMENTO = 'U'
AND       ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND       ATENDIME.CD_PRESTADOR  =  PRESTADOR.CD_PRESTADOR(+)
GROUP BY PRESTADOR.CD_PRESTADOR
                  ,PRESTADOR.NM_PRESTADOR
ORDER BY ATENDIMENTOS  DESC
