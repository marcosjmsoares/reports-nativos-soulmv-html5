SELECT
          PACIENTE.CD_PACIENTE,
          PACIENTE.NM_PACIENTE,
          (SELECT PA.NM_SOCIAL_PACIENTE
        	FROM DBAMV.PACIENTE PA
        	WHERE PA.CD_PACIENTE = PACIENTE.CD_PACIENTE
          		AND NVL(PA.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE,
          PACIENTE.DT_NASCIMENTO,
          NVL(PACIENTE.TP_SEXO, ' ') TP_SEXO,
          NVL(PACIENTE.TP_COR, ' ') TP_COR,
          DBAMV.FNC_ESTADO_CIVIL(NVL(PACIENTE.TP_ESTADO_CIVIL, 'S')) TP_ESTADO_CIVIL1,
          nvl(PACIENTE.NR_DOCUMENTO, ' ') NR_DOCUMENTO,
          nvl(PACIENTE.NM_MAE, ' ') NM_MAE,
          nvl(PACIENTE.NM_PAI, ' ') NM_PAI,
          nvl(PACIENTE.NM_CONJUGE, ' ') NM_CONJUGE,
          nvl(PACIENTE.DS_ENDERECO, ' ') DS_ENDERECO,
          nvl(PACIENTE.NR_ENDERECO, 0) NR_ENDERECO,
          nvl(PACIENTE.DS_COMPLEMENTO, ' ') DS_COMPLEMENTO,
          nvl(PACIENTE.NM_BAIRRO, ' ') NM_BAIRRO,
          nvl(PACIENTE.NR_CEP, ' ') NR_CEP,
          nvl(PACIENTE.NR_FONE, ' ') NR_FONE,
          PACIENTE.DT_CADASTRO, 
          nvl(PACIENTE.NR_CPF,  ' ') NR_CPF,
          nvl(PACIENTE.NR_IDENTIDADE, ' ') NR_IDENTIDADE, 
          nvl(PACIENTE.DS_OM_IDENTIDADE, ' ') DS_OM_IDENTIDADE, 
          nvl(PACIENTE.NR_RG_NASC, 0) NR_RG_NASC,
          nvl(PACIENTE.NR_CNS, ' ') NR_CNS,
          DECODE( NVL(PACIENTE.SN_DOADOR, 'N'), 'N', 'NÃO', 'SIM')   SN_DOADOR,
          nvl(PACIENTE.NM_MNEMONICO, ' ') NM_MNEMONICO,  
          nvl(PACIENTE.CD_NATURALIDADE, 0) CD_NATURALIDADE,
          nvl(ATENDIME.CD_ATENDIMENTO, 0) CD_ATENDIMENTO,
          ('*'  ||  TO_CHAR(ATENDIME.CD_ATENDIMENTO)  ||  '*')   CD_ATENDE_BARRAS,
          ATENDIME.DT_ATENDIMENTO,
	  TO_CHAR ( ATENDIME.HR_ATENDIMENTO, 'HH24') HR_ATENDIMENTO,
          nvl(ATENDIME.CD_SERVICO, 0) CD_SERVICO,
          nvl(ATENDIME.CD_SER_DIS, 0) CD_SER_DIS,
          nvl(SER_DIS.DS_SER_DIS, ' ') DS_SER_DIS,
          nvl(ATENDIME.CD_CID, ' ') CD_CID,
          nvl(ATENDIME.TP_ATENDIMENTO, ' ') TP_ATENDIMENTO, 
          nvl(ATENDIME.CD_DES_ATE, 0) CD_DES_ATE,
          nvl(ATENDIME.NM_USUARIO, ' ') NM_USUARIO,
          nvl(ATENDIME.CD_PRO_INT, ' ') CD_PRO_INT,
          nvl(ATENDIME.NR_CHAMADA_PAINEL, ' ') NR_CHAMADA_PAINEL,
          nvl(CONVENIO.CD_CONVENIO, 0) CD_CONVENIO,
          DECODE( NVL(ATENDIME.SN_RETORNO, 'N'), 'N', 'NÃO', 'SIM')   SN_RETORNO,
          nvl(ORI_ATE.DS_ORI_ATE, ' ') DS_ORI_ATE,
          nvl(PRESTADOR.CD_PRESTADOR, 0) CD_PRESTADOR,
          nvl(PRESTADOR.NM_PRESTADOR, ' ') NM_PRESTADOR,
          nvl(PRESTADOR.DS_CODIGO_CONSELHO,  ' ') DS_CODIGO_CONSELHO,
          nvl(CONVENIO.NM_CONVENIO, ' ') NM_CONVENIO,
          nvl(CID.DS_CID, ' ') DS_CID,
          nvl(SERVICO.DS_SERVICO, ' ') DS_SERVICO, 
          nvl(DES_ATE.TP_ANAMNESE, ' ') TP_ANAM_SERVICO,
          nvl(DES_ATE.TP_ANAMNESE, ' ') TP_ANAM_SER_DIS,
          nvl(DES_ATE.DS_DES_ATE, ' ') DS_DES_ATE,
          nvl(RESPONSA.NM_RESPONSAVEL, ' ') NM_RESPONSAVEL,
          nvl(RESPONSA.NR_FONE, ' ') NR_FONE_RESPONSAVEL,
          nvl(RESPONSA.DS_DOCUMENTO, ' ') DS_DOCUMENTO_RESPONSAVEL,
          nvl(RESPONSA.DS_ENDERECO, ' ') DS_ENDERECO_RESPONSAVEL,
          nvl(RESPONSA.NR_CPF, ' ') NR_CPF_RESP,
          nvl(RESPONSA.DS_CONTATO_FAMILIA, ' ') DS_CONTATO_FAMILIA,
          nvl(RESPONSA.NR_ENDERECO, 0) NR_ENDERECO_RESPONSAVEL,
          nvl(RESPONSA.DS_COMPLEMENTO, ' ') DS_COMPLEMENTO_RESPONSAVEL,
          nvl(RESPONSA.NM_BAIRRO, ' ') NM_BAIRRO_RESPONSAVEL,
          nvl(RESPONSA.DS_NACIONALIDADE, ' ') DS_NACIONALIDADE_RESPONSAVEL,
          nvl(CARTEIRA.NR_CARTEIRA, ' ') NR_CARTEIRA,
          nvl(CARTEIRA.NM_EMPRESA, ' ') NM_EMPRESA,
          nvl(CARTEIRA.NM_TITULAR, ' ') NM_TITULAR,
          CARTEIRA.DT_VALIDADE,
          CARTEIRA.DT_ULT_PGTO,
          DECODE( NVL(CARTEIRA.SN_TITULAR, 'N'), 'N', 'NÃO', 'SIM')   SN_TITULAR,
          DECODE( NVL(CARTEIRA.SN_PENSIONISTA, 'N'), 'N', 'NÃO', 'SIM')   SN_PENSIONISTA,
          nvl(CON_PLA.CD_CON_PLA, 0) CD_CON_PLA,
          nvl(CON_PLA.DS_CON_PLA, ' ') DS_CON_PLA,
          nvl(ATENDIME.CD_SUB_PLANO, ' ') CD_SUB_PLANO,
          nvl(SUB_PLANO.DS_SUB_PLANO, ' ') DS_SUB_PLANO,
          nvl(CIDADE.NM_CIDADE, ' ') NM_CIDADE,
          nvl(CIDADE.CD_UF, ' ') CD_UF,
          nvl(TIPO_INTERNACAO.CD_TIPO_INTERNACAO, 0) CD_TIPO_INTERNACAO,
          nvl(TIPO_INTERNACAO.DS_TIPO_INTERNACAO, ' ') DS_TIPO_INTERNACAO,
          ATENDIME_INFO.DS_INFO_ATENDIMENTO,  
          nvl(ESPECIALID.DS_ESPECIALID, ' ') DS_ESPECIALID, 
          nvl(LOC_PROCED.DS_LOC_PROCED, ' ') DS_LOC_PROCED,
          nvl(CIDADE2.NM_CIDADE, ' ') NM_NATURALIDADE, 
          nvl(CIDADE2.CD_UF, ' ') UF_NATURALIDADE,
          nvl(CATEGORIA_PLANO.CD_CATEGORIA_PLANO, 0) CD_CATEGORIA_PLANO,
          nvl(CATEGORIA_PLANO.DS_CATEGORIA_PLANO, ' ') DS_CATEGORIA_PLANO,
          nvl(GRAU_INS.DS_GRAU_INS, ' ') DS_GRAU_INS,
          nvl(RELIGIAO.DS_RELIGIAO, ' ') DS_RELIGIAO,
          nvl(MOT_ALT.DS_MOT_ALT, ' ') DS_MOT_ALT,
          nvl(TIP_PAREN.DS_TIP_PAREN, ' ') DS_TIP_PAREN,
          nvl(CIDADANIAS.DS_CIDADANIA, ' ') DS_CIDADANIA,
          nvl(PROFISSAO.NM_PROFISSAO, ' ') NM_PROFISSAO,
          nvl(Procedimento_Sus.Ds_Procedimento, ' ') DS_TITULO,
          nvl(Atendime.Cd_Procedimento, ' ') CD_SSM_SIA
FROM  DBAMV.PACIENTE
           ,DBAMV.CIDADE
           ,DBAMV.CIDADE  CIDADE2
           ,DBAMV.ATENDIME
           ,DBAMV.ORI_ATE
           ,DBAMV.PRESTADOR
           ,DBAMV.CONVENIO
           ,DBAMV.CID
           ,DBAMV.SERVICO
           ,DBAMV.RESPONSA
           ,DBAMV.CARTEIRA
           ,DBAMV.CON_PLA
           ,DBAMV.DES_ATE
           ,DBAMV.TIPO_INTERNACAO
           ,DBAMV.ATENDIME_INFO
           ,DBAMV.ESPECIALID
           ,DBAMV.LOC_PROCED
           ,DBAMV.CATEGORIA_PLANO
           ,DBAMV.GRAU_INS
           ,DBAMV.RELIGIAO
           ,DBAMV.MOT_ALT
           ,DBAMV.TIP_PAREN
           ,DBAMV.CIDADANIAS
           ,DBAMV.PROFISSAO
           ,DBAMV.SER_DIS
           ,DBAMV.procedimento_sus
           ,DBAMV.SUB_PLANO
Where Atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
    AND ATENDIME.CD_ATENDIMENTO  =  TO_CHAR( :CAMPO_PCD_ATENDIMENTO )
    AND ATENDIME.CD_PACIENTE  =  PACIENTE.CD_PACIENTE
   AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
   AND PACIENTE.CD_CIDADE        =  CIDADE.CD_CIDADE(+)
   AND DES_ATE.CD_DES_ATE(+)     =  ATENDIME.CD_DES_ATE
   AND ATENDIME.CD_ORI_ATE       =  ORI_ATE.CD_ORI_ATE (+)
   AND ATENDIME.CD_PRESTADOR     =  PRESTADOR.CD_PRESTADOR(+)
   AND ATENDIME.CD_CONVENIO  = SUB_PLANO.CD_CONVENIO(+)
   AND ATENDIME.CD_CON_PLA   = SUB_PLANO.CD_CON_PLA(+)
   AND ATENDIME.CD_SUB_PLANO = SUB_PLANO.CD_SUB_PLANO(+)
   AND ATENDIME.CD_CID         =  CID.CD_CID (+)
   AND ATENDIME.CD_SERVICO     =  SERVICO.CD_SERVICO (+)
   AND ATENDIME.CD_ATENDIMENTO =  RESPONSA.CD_ATENDIMENTO (+)
   AND ATENDIME.CD_SER_DIS     =  SER_DIS.CD_SER_DIS(+)
   AND CARTEIRA.CD_PACIENTE(+) = ATENDIME.CD_PACIENTE
   AND ATENDIME.CD_TIPO_INTERNACAO  =  TIPO_INTERNACAO.CD_TIPO_INTERNACAO(+)
   AND CARTEIRA.CD_CON_PLA          =  CON_PLA.CD_CON_PLA (+)
   AND CARTEIRA.CD_CONVENIO         =  CON_PLA.CD_CONVENIO(+)
   AND CARTEIRA.CD_CATEGORIA_PLANO  =  CATEGORIA_PLANO.CD_CATEGORIA_PLANO(+)
   AND ATENDIME.CD_ATENDIMENTO  = ATENDIME_INFO.CD_ATENDIMENTO(+)
   AND ATENDIME.CD_ESPECIALID   = ESPECIALID.CD_ESPECIALID(+)
   AND ATENDIME.CD_LOC_PROCED   =  LOC_PROCED.CD_LOC_PROCED(+)
   AND PACIENTE.CD_NATURALIDADE =  CIDADE2.CD_CIDADE(+)
   AND PACIENTE.CD_GRAU_INS     = GRAU_INS.CD_GRAU_INS(+)
   AND PACIENTE.CD_RELIGIAO     = RELIGIAO.CD_RELIGIAO(+)
   AND ATENDIME.CD_MOT_ALT      = MOT_ALT.CD_MOT_ALT(+)
   AND RESPONSA.CD_TIP_PAREN    = TIP_PAREN.CD_TIP_PAREN(+)
   AND PACIENTE.CD_CIDADANIA    = CIDADANIAS.CD_CIDADANIA(+)
   AND PACIENTE.CD_PROFISSAO    = PROFISSAO.CD_PROFISSAO(+)
   AND ATENDIME.cd_procedimento = procedimento_sus.cd_procedimento (+)
