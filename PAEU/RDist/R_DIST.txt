--R_DIST
SELECT  COUNT(*)   QUANT
               ,DIS_SAN.CD_DIS_SAN CD_DIS_SAN
               ,decode(DIS_SAN.NM_DIS_SAN, null, 'Não Informado', DIS_SAN.NM_DIS_SAN) NM_DIS_SAN
FROM      DBAMV.ATENDIME
               ,DBAMV.PACIENTE
               ,DBAMV.DIS_SAN
WHERE PACIENTE.CD_PACIENTE  =  ATENDIME.CD_PACIENTE
AND       DIS_SAN.CD_DIS_SAN(+) = PACIENTE.CD_DIS_SAN
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND  ATENDIME.DT_ATENDIMENTO  BETWEEN
to_date(to_char(:CAMPO_DT_INIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND to_date(to_char(:CAMPO_DT_FIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
GROUP BY  DIS_SAN.CD_DIS_SAN
                  , DIS_SAN.NM_DIS_SAN
ORDER BY  QUANT DESC
