--R_ESCALAS_PRESTADOR_PAEU
SELECT * FROM (
SELECT  ESC_DIA_EXT.CD_ESPECIALID                   CD_ESPECIALID
               ,ESPECIALID.DS_ESPECIALID                      DS_ESPECIALID
               ,ESC_DIA_EXT.DT_ESC_DIA                        DT_ESC_DIA
               ,ESC_DIA_EXT.HR_ENTRADA                     HR_ENTRADA
               ,ESC_DIA_EXT.HR_SAIDA                            HR_SAIDA
               ,ESC_DIA_EXT.CD_PRESTADOR                 CD_PRESTADOR
               ,PRESTADOR.NM_PRESTADOR                  NM_PRESTADOR
               ,PRESTADOR.DS_CODIGO_CONSELHO     DS_CODIGO_CONSELHO
               ,ESC_DIA_EXT.DS_OBSERVACAO               DS_OBSERVACAO
FROM  DBAMV.ESC_DIA_EXT
           ,DBAMV.PRESTADOR
           ,DBAMV.ESPECIALID
WHERE ESC_DIA_EXT.DT_ESC_DIA 
between to_date(to_char(:CAMPO_DATINI,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND to_date(to_char(:CAMPO_DATFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24') 
AND       ESC_DIA_EXT.CD_ESPECIALID  =  ESPECIALID.CD_ESPECIALID
AND       ESC_DIA_EXT.CD_PRESTADOR  =  PRESTADOR.CD_PRESTADOR
ORDER BY  ESC_DIA_EXT.HR_ENTRADA   DESC
                   ,PRESTADOR.NM_PRESTADOR
)
ORDER BY DT_ESC_DIA,
CD_ESPECIALID,DS_ESPECIALID
