       '1' A
FROM DBAMV.CIDADE WHERE ROWNUM <= :CAMPO_QUANT

--R_ETIQUETA_URG_RF_ETIQ1_Q_ATENDIMENTO
SELECT * FROM (
  SELECT '1' A1
        ,ATENDIME.CD_ATENDIMENTO
        ,('*'  ||  TO_CHAR(ATENDIME.CD_ATENDIMENTO)  ||  '*')   CD_ATENDIMENTO_BAR
        ,ATENDIME.DT_ATENDIMENTO
        ,ATENDIME.HR_ATENDIMENTO
        ,ATENDIME.CD_PACIENTE
        ,PACIENTE.NM_PACIENTE
        ,(SELECT P.NM_SOCIAL_PACIENTE
			    FROM DBAMV.PACIENTE P
			    WHERE P.CD_PACIENTE = PACIENTE.CD_PACIENTE
				    AND NVL(P.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE
        ,CONVENIO.NM_CONVENIO
        ,CON_PLA.DS_CON_PLA
        ,DES_ATE.DS_DES_ATE
  FROM DBAMV.ATENDIME
      ,DBAMV.PACIENTE
      ,DBAMV.CONVENIO
      ,DBAMV.CON_PLA
      ,DBAMV.DES_ATE
      ,DBAMV.EMPRESA_CON_PLA
  WHERE EMPRESA_CON_PLA.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
    AND EMPRESA_CON_PLA.CD_CONVENIO = CON_PLA.CD_CONVENIO
    AND EMPRESA_CON_PLA.CD_CON_PLA = CON_PLA.CD_CON_PLA
    AND ATENDIME.CD_ATENDIMENTO = :CAMPO_PCD_ATENDIMENTO
    AND ATENDIME.TP_ATENDIMENTO =  'U'
    AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
    AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
    AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
    AND ATENDIME.CD_DES_ATE = DES_ATE.CD_DES_ATE(+)
    AND ATENDIME.CD_CONVENIO = CON_PLA.CD_CONVENIO(+)
    AND ATENDIME.CD_CON_PLA = CON_PLA.CD_CON_PLA(+)
)
WHERE A1 = :CAMPO_A
