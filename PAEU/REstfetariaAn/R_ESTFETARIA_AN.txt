SELECT PACIENTE.CD_PACIENTE
      ,PACIENTE.NM_PACIENTE
      ,(SELECT P.NM_SOCIAL_PACIENTE
			  FROM DBAMV.PACIENTE P
			  WHERE P.CD_PACIENTE = PACIENTE.CD_PACIENTE
				  AND NVL(P.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE
      ,PACIENTE.DT_NASCIMENTO
      ,ATENDIME.DT_ATENDIMENTO
      ,ATENDIME.CD_ATENDIMENTO
      ,CIDADE.NM_CIDADE
      ,DECODE(CONVENIO.TP_CONVENIO, 'C', PRO_FAT.CD_PRO_FAT
                                  , 'P', PRO_FAT.CD_PRO_FAT
                                  , 'A', PROCEDIMENTO_SUS.CD_PROCEDIMENTO
                                  , 'H', PROCEDIMENTO_SUS.CD_PROCEDIMENTO) CD_PRO_FAT
      ,DECODE(CONVENIO.TP_CONVENIO, 'C', PRO_FAT.DS_PRO_FAT
                                  , 'P', PRO_FAT.DS_PRO_FAT
                                  , 'A', PROCEDIMENTO_SUS.DS_PROCEDIMENTO
                                  , 'H', PROCEDIMENTO_SUS.DS_PROCEDIMENTO) DS_PRO_FAT
      ,DBAMV.RETORNA_FAIXA_ETARIA_RELATORIO(PACIENTE.DT_NASCIMENTO, 'C') CD_FAIXA
      ,DBAMV.RETORNA_FAIXA_ETARIA_RELATORIO(PACIENTE.DT_NASCIMENTO, 'D') DS_FAIXA
FROM DBAMV.ATENDIME
    ,DBAMV.DES_ATE
    ,DBAMV.PRESTADOR
    ,DBAMV.ORI_ATE
    ,DBAMV.PACIENTE
    ,DBAMV.CONVENIO
    ,DBAMV.LEITO
    ,DBAMV.CIDADE
    ,DBAMV.PRO_FAT
    ,DBAMV.PROCEDIMENTO_SUS
WHERE DBAMV.ATENDIME.CD_DES_ATE   = DBAMV.DES_ATE.CD_DES_ATE(+)
  AND DBAMV.ATENDIME.CD_PRESTADOR = DBAMV.PRESTADOR.CD_PRESTADOR(+)
  AND DBAMV.ATENDIME.CD_ORI_ATE   = DBAMV.ORI_ATE.CD_ORI_ATE
  AND DBAMV.ATENDIME.CD_CONVENIO  = DBAMV.CONVENIO.CD_CONVENIO
  AND DBAMV.ATENDIME.CD_PACIENTE  = DBAMV.PACIENTE.CD_PACIENTE
  AND ATENDIME.CD_LEITO           = LEITO.CD_LEITO(+)
  AND PACIENTE.CD_CIDADE          = CIDADE.CD_CIDADE(+)
  AND ATENDIME.CD_PRO_INT         = PRO_FAT.CD_PRO_FAT(+)
  AND ATENDIME.CD_PROCEDIMENTO    = PROCEDIMENTO_SUS.CD_PROCEDIMENTO(+)
  AND DBAMV.ATENDIME.CD_ATENDIMENTO_PAI IS NULL
  AND ATENDIME.DT_ATENDIMENTO BETWEEN TO_DATE(TO_CHAR(:CAMPO_DTINIC,'DD/MM/YYYY')||' 00','DD/MM/YYYY HH24') 
    AND TO_DATE(TO_CHAR(:CAMPO_DTFIM,'DD/MM/YYYY')||' 23','DD/MM/YYYY HH24')
ORDER BY TO_NUMBER(DBAMV.RETORNA_FAIXA_ETARIA_RELATORIO(PACIENTE.DT_NASCIMENTO, 'C'))
