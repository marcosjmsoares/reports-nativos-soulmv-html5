--R_BOLETIM_DIARIO_POR
SELECT ATENDIME.CD_ATENDIMENTO
              ,ATENDIME.CD_PACIENTE
              ,PACIENTE.NM_PACIENTE
              ,TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24')  HORA
              ,CONVENIO.NM_CONVENIO
              ,PRESTADOR.NM_MNEMONICO
              ,ATENDIME.CD_PRO_INT
              ,DBAMV.PROCEDIAS(ATENDIME.CD_CONVENIO, ATENDIME.CD_PRO_INT, 'D')  DS_PRO_INT
              ,1  PLANTAO
FROM     DBAMV.ATENDIME
               ,DBAMV.PACIENTE
               ,DBAMV.CONVENIO
               ,DBAMV.PRESTADOR
WHERE  TO_DATE(TO_CHAR(ATENDIME.DT_ATENDIMENTO,'DD/MM/YYYY ')  || TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'),'DD/MM/YYYY HH24')
BETWEEN
                TO_DATE((TO_CHAR(:CAMPO_DATA_INI,'dd/mm/yyyy')  || :CAMPO_FLUXO1), 'DD/MM/YYYY HH24')  AND
                TO_DATE(DECODE('', NULL, to_char(:CAMPO_DATA_INI,'dd/mm/yyyy'), to_char(:CAMPO_DATA_FIM,'dd/mm/yyyy'))  || DECODE('', NULL, :CAMPO_FLUXO2, :CAMPO_FLUXO4), 'DD/MM/YYYY HH24')
AND         TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24') BETWEEN :CAMPO_FLUXO1  AND  
AND         ATENDIME.TP_ATENDIMENTO IN ( 'U', 'I')
AND         ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND         ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND         ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND         ATENDIME.CD_PRESTADOR  =  PRESTADOR.CD_PRESTADOR(+)
union all
SELECT ATENDIME.CD_ATENDIMENTO
              ,ATENDIME.CD_PACIENTE
              ,PACIENTE.NM_PACIENTE
              ,TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24')  HORA
              ,CONVENIO.NM_CONVENIO
              ,PRESTADOR.NM_MNEMONICO
              ,ATENDIME.CD_PRO_INT
              ,DBAMV.PROCEDIAS(ATENDIME.CD_CONVENIO, ATENDIME.CD_PRO_INT, 'D')  DS_PRO_INT
              ,2  PLANTAO
FROM     DBAMV.ATENDIME
               ,DBAMV.PACIENTE
               ,DBAMV.CONVENIO
               ,DBAMV.PRESTADOR
WHERE  TO_DATE(TO_CHAR(ATENDIME.DT_ATENDIMENTO,'DD/MM/YYYY ')  || TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'),'DD/MM/YYYY HH24')
BETWEEN
                TO_DATE(TO_CHAR(:CAMPO_DATA_INI,'dd/mm/yyyy ')  || :CAMPO_FLUXO1, 'DD/MM/YYYY HH24')  AND
                TO_DATE(DECODE('', NULL, to_char(:CAMPO_DATA_INI,'dd/mm/yyyy '), to_char(:CAMPO_DATA_FIM,'dd/mm/yyyy ')) || DECODE('', NULL, :CAMPO_FLUXO2, :CAMPO_FLUXO4), 'DD/MM/YYYY HH24')
AND         TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24')  BETWEEN :CAMPO_FLUXO3  AND  
AND         ATENDIME.TP_ATENDIMENTO IN ( 'U', 'I')
AND         ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND         ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND         ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND         ATENDIME.CD_PRESTADOR  =  PRESTADOR.CD_PRESTADOR(+)
ORDER BY  HORA
