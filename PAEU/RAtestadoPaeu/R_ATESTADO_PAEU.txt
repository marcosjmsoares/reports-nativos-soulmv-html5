--R_ATESTADO_PAEU
SELECT TEXTOS_ATD.DS_TEXTOS_ATD              DS_TEXTOS_ATD,
               ATENDIME.CD_CID                                    CD_CID,
               ATENDIME.CD_PRESTADOR                    CD_PRESTADOR,
              CID.DS_CID                                                  DS_CID,
              PRESTADOR.NM_PRESTADOR                 NM_PRESTADOR,
              PRESTADOR.DS_CODIGO_CONSELHO    DS_CODIGO_CONSELHO,
              TO_CHAR(SYSDATE,'DD/MM/YYYY')                                                     DATA_ATESTADO,
              PACIENTE.NM_PACIENTE                         NM_PACIENTE,
              TO_CHAR(ATENDIME.DT_ATENDIMENTO,'DD/MM/YYYY') DT_ATENDIMENTO
FROM    DBAMV.TEXTOS_ATD,
              DBAMV.ATENDIME,
              DBAMV.CID,
              DBAMV.PRESTADOR,
              DBAMV.PACIENTE,
              DBAMV.CASOS_ATD
WHERE  TEXTOS_ATD.TP_TEXTOS_ATD    = 'A'
AND        TEXTOS_ATD.CD_CASOS_ATD      =  CASOS_ATD.CD_CASOS_ATD
AND        ATENDIME.CD_CID                           = CID.CD_CID
AND        CASOS_ATD.CD_PRESTADOR          = PRESTADOR.CD_PRESTADOR
AND       ATENDIME.CD_PACIENTE                = PACIENTE.CD_PACIENTE
AND      ATENDIME.CD_ATENDIMENTO       = CASOS_ATD.CD_ATENDIMENTO
AND        ATENDIME.CD_MULTI_EMPRESA  = :CAMPO_CD_MULTI_EMPRESA
