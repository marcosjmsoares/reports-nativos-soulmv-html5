SELECT ATENDIME.CD_ATENDIMENTO
             ,ATENDIME.DT_ATENDIMENTO
             ,ATENDIME.HR_ATENDIMENTO
             ,PACIENTE.CD_PACIENTE
             ,PACIENTE.NM_PACIENTE
             ,PACIENTE.DT_NASCIMENTO
             ,DECODE(PACIENTE.TP_SEXO,'M', 'MASCULINO', 'F', 'FEMININO', 'AMBOS') DS_SEXO
             ,PRESTADOR.CD_PRESTADOR
             ,PRESTADOR.NM_PRESTADOR
             ,ATENDIME.DS_ATE_OCO
             ,DECODE(PACIENTE.SN_UTIL_IDENTIDADE_GENERO,'S',DECODE((SELECT MAX(1) FROM DBASGU.CONFIG_ITEM_MODULO
             WHERE CD_MODULO = 'CAD_PAC'
             AND SN_VISIBLE = 'S'
             AND CD_ITEM = 'DSP_DS_IDENTIDADE_GENERO'
             AND CD_BLOCO = 'PACIENTE')
             ,1,IDENTIDADE_GENERO.DS_IDENTIDADE_GENERO
             ,NULL),NULL) AS IDENTIDADE_GENERO
             ,DECODE(PACIENTE.SN_UTIL_ORIENTACAO_SEXUAL,'S',DECODE((SELECT MAX(1) FROM DBASGU.CONFIG_ITEM_MODULO
             WHERE CD_MODULO = 'CAD_PAC'
             AND SN_VISIBLE = 'S'
             AND CD_ITEM = 'DSP_DS_ORIENTACAO_SEXUAL'
             AND CD_BLOCO = 'PACIENTE'
             ),1,ORIENTACAO_SEXUAL.DS_ORIENTACAO_SEXUAL,NULL),NULL) AS ORIENTACAO_SEXUAL
FROM   DBAMV.ATENDIME
            ,DBAMV.PACIENTE
            ,DBAMV.PRESTADOR
            ,DBAMV.IDENTIDADE_GENERO
            ,DBAMV.ORIENTACAO_SEXUAL
WHERE    ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND        ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
AND        ATENDIME.DS_ATE_OCO IS NOT NULL
AND       PACIENTE.CD_IDENTIDADE_GENERO = IDENTIDADE_GENERO.CD_IDENTIDADE_GENERO(+)
AND       PACIENTE.CD_ORIENTACAO_SEXUAL = ORIENTACAO_SEXUAL.CD_ORIENTACAO_SEXUAL(+)
AND     ATENDIME.CD_ATENDIMENTO = :CAMPO_PCD_ATENDIMENTO
