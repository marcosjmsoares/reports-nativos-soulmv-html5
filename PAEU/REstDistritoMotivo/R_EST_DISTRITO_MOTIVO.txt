--R_EST_DISTRITO_MOTIVO
SELECT  COUNT(*)                       QUANT
               ,NVL(DIS_SAN.CD_DIS_SAN,'0')                                        CD_DIS_SAN
               ,NVL(DIS_SAN.NM_DIS_SAN,'NÃO INFORMADO')           NM_DIS_SAN
               ,NVL(MOT_ENT.CD_MOT_ENT,'0')                                    CD_MOT_ENT
               ,NVL(MOT_ENT.DS_MOT_ENT,'NÃO INFORMADO')        DS_MOT_ENT
FROM      DBAMV.ATENDIME
               ,DBAMV.PACIENTE
               ,DBAMV.DIS_SAN
               ,DBAMV.ATE_MOTIVO
               ,DBAMV.MOT_ENT
WHERE ATENDIME.CD_ATENDIMENTO = ATE_MOTIVO.CD_ATENDIMENTO(+)
AND       ATE_MOTIVO.CD_MOT_ENT = MOT_ENT.CD_MOT_ENT(+)
AND       ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND       ATENDIME.TP_ATENDIMENTO = 'U'
AND       ATENDIME.DT_ATENDIMENTO  BETWEEN
to_date(to_char(:CAMPO_DTINIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND to_date(to_char(:CAMPO_DTFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND       ATENDIME.CD_PACIENTE =  PACIENTE.CD_PACIENTE
AND       PACIENTE.CD_DIS_SAN  =  DIS_SAN.CD_DIS_SAN(+)
GROUP BY  DIS_SAN.CD_DIS_SAN
                  , DIS_SAN.NM_DIS_SAN
                  , MOT_ENT.CD_MOT_ENT
                  , MOT_ENT.DS_MOT_ENT
ORDER BY  QUANT DESC
