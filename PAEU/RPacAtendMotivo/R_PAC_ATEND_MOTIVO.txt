--R_PAC_ATEND_MOTIVO
SELECT atendime.cd_paciente
      ,atendime.cd_atendimento
      ,paciente.nm_paciente
      ,(SELECT P.NM_SOCIAL_PACIENTE
			  FROM DBAMV.PACIENTE P
			  WHERE P.CD_PACIENTE = PACIENTE.CD_PACIENTE
				  AND NVL(P.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE
      ,atendime.dt_atendimento
      ,prestador.nm_prestador
      ,nvl(mot_ent.ds_mot_ent, :CAMPO_CF_MOTIVO_NAO_INFORMADO) Ds_mot_ent
FROM dbamv.atendime
    ,dbamv.paciente
    ,dbamv.prestador
    ,dbamv.ate_motivo
    ,dbamv.mot_ent
    ,dbamv.convenio
WHERE paciente.cd_paciente = atendime.cd_paciente
  AND atendime.cd_convenio = convenio.cd_convenio
  AND atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
  AND atendime.cd_atendimento = ate_motivo.cd_atendimento(+)
  AND ate_motivo.cd_mot_ent = mot_ent.cd_mot_ent(+)
  AND atendime.dt_atendimento BETWEEN to_date(to_char(:CAMPO_DATAINI, 'dd/mm/yyyy') || ' 00', 'dd/mm/yyyy HH24')
    AND to_date(to_char(:CAMPO_DATAFIM, 'dd/mm/yyyy') || ' 23', 'dd/mm/yyyy HH24')
  AND prestador.cd_prestador(+) = atendime.cd_prestador
ORDER BY ds_mot_ent
