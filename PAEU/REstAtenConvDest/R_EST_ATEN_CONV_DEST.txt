--R_EST_ATEN_CONV_DEST
SELECT CONVENIO.CD_CONVENIO
             ,CONVENIO.NM_CONVENIO
             ,COUNT(*)  QUANT_CONV
             ,( SELECT Sum(QUANT_CONV) FROM ( SELECT COUNT(*)  QUANT_CONV
                                                FROM  DBAMV.ATENDIME
                                                    ,DBAMV.CONVENIO
                                                    ,DBAMV.DES_ATE
                                               WHERE  ATENDIME.TP_ATENDIMENTO  =  'U'
                                               AND        ATENDIME.DT_ATENDIMENTO  BETWEEN
                                               to_date(to_char(:CAMPO_DT_INICIO,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND to_date(to_char(:CAMPO_DT_FINAL,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
                                               AND        ATENDIME.CD_CONVENIO  =  CONVENIO.CD_CONVENIO
                                               AND        ATENDIME.CD_DES_ATE  =  DES_ATE.CD_DES_ATE
                                               AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
             )) TOT_QUANT_CONV
FROM   DBAMV.ATENDIME
            ,DBAMV.CONVENIO
            ,DBAMV.DES_ATE
WHERE  ATENDIME.TP_ATENDIMENTO  =  'U'
AND        TRUNC(ATENDIME.DT_ATENDIMENTO)  BETWEEN
to_date(to_char(:CAMPO_DT_INICIO,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND to_date(to_char(:CAMPO_DT_FINAL,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND        ATENDIME.CD_CONVENIO  =  CONVENIO.CD_CONVENIO
AND        ATENDIME.CD_DES_ATE  =  DES_ATE.CD_DES_ATE
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY   CONVENIO.CD_CONVENIO
                    ,CONVENIO.NM_CONVENIO
ORDER BY  QUANT_CONV  DESC
