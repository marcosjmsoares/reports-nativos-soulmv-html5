                       ,0        TOTAL_LIMPEZA
                       ,0        TOTAL_OCUPADO
                       ,0        TOTAL_RESERVA
                       ,0        TOTAL_VAGO
                       ,0        TOTAL_INFECTADO
                       ,0        TOTAL_ACOMPANHANTE
                       ,COUNT(*) TOTAL_EXTRA
           FROM   DBAMV.UNID_INT
                       ,DBAMV.LEITO
                       ,DBAMV.SETOR
           WHERE  LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT
          AND         UNID_INT.CD_SETOR = SETOR.CD_SETOR
          AND         UNID_INT.SN_ATIVO = 'S'
          AND   UNID_INT.TP_UNID_INT = 'U'
          AND         LEITO.TP_SITUACAO = 'A'
          AND         LEITO.SN_EXTRA    = 'S'
          AND TP_OCUPACAO NOT IN ('V','L')
             GROUP BY DS_UNID_INT
 )
 GROUP BY UNIDADE
 ORDER BY UNIDADE
