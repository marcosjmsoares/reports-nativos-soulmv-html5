--R_PAC_TEMPO_PERMANENCIA
SELECT PACIENTE .CD_PACIENTE, PACIENTE.NM_PACIENTE,
       (SELECT P.NM_SOCIAL_PACIENTE
			  FROM DBAMV.PACIENTE P
			  WHERE P.CD_PACIENTE = PACIENTE.CD_PACIENTE
				  AND NVL(P.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE,
       ATENDIME.CD_ATENDIMENTO, ATENDIME.TP_ATENDIMENTO,
       TO_DATE(TO_CHAR(DT_ATENDIMENTO,'DD/MM/YYYY') || TO_CHAR(HR_ATENDIMENTO,'HH24') ,'DD/MM/YYYY HH24') DT_ATENDIMENTO,
       TO_DATE(TO_CHAR(DT_ALTA_MEDICA,'DD/MM/YYYY') || TO_CHAR(HR_ALTA_MEDICA,'HH24') ,'DD/MM/YYYY HH24') DT_ALTA_MEDICA,
       TO_DATE(TO_CHAR(DT_ALTA,'DD/MM/YYYY') || TO_CHAR(HR_ALTA,'HH24') ,'DD/MM/YYYY HH24') DT_ALTA,
       ATENDIME.HR_ATENDIMENTO,
       ATENDIME.HR_ALTA_MEDICA,
       CONVENIO.NM_CONVENIO,
       CONVENIO.CD_CONVENIO,
       NULL TEMP_ALTA_MEDICA,
       ATENDIME.HR_ALTA,
       NULL TEMP_ALTA_HOSPITALAR,
       ORI_ATE.DS_ORI_ATE
FROM DBAMV.PACIENTE,
     DBAMV.ATENDIME,
     DBAMV.CONVENIO,
     DBAMV.ORI_ATE
WHERE PACIENTE.CD_PACIENTE = ATENDIME.CD_PACIENTE
  AND CONVENIO.CD_CONVENIO = ATENDIME.CD_CONVENIO
  AND ATENDIME.CD_ORI_ATE = ORI_ATE.CD_ORI_ATE
  AND ATENDIME.CD_MULTI_EMPRESA = dbamv.pkg_mv2000.le_empresa
  AND (ATENDIME.DT_ALTA IS NOT NULL OR ATENDIME.DT_ALTA_MEDICA IS NOT NULL)
  AND ATENDIME.DT_ATENDIMENTO between to_date(to_char(:CAMPO_DATINI, 'dd/mm/yyyy') || ' 00', 'dd/mm/yyyy HH24')
    AND to_date(to_char(:CAMPO_DATFIM,'dd/mm/yyyy') || ' 23', 'dd/mm/yyyy HH24')
ORDER BY PACIENTE.NM_PACIENTE
