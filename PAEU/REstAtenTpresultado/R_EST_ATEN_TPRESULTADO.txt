--R_EST_ATEN_TPRESULTADO
SELECT tipo.DS_TIP_RES,
              COUNT(*) TotalAtend
FROM  DBAMV.ATENDIME aten,
             DBAMV.TIP_RES  tipo,
             DBAMV.CONVENIO conv
WHERE aten.DT_ATENDIMENTO between to_date(to_char(:CAMPO_DATINI,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND to_date(to_char(:CAMPO_DATFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND       aten.TP_ATENDIMENTO  =  'U'
AND       aten.CD_TIP_RES  =  tipo.CD_TIP_RES
AND       ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND       aten.CD_CONVENIO = conv.CD_CONVENIO
GROUP BY tipo.DS_TIP_RES
ORDER BY  TotalAtend  DESC
