--R_FAT_SANGUE_SETOR_P321
select sysdate from dual


/*----------------------------------------------------------------------------*/
--R_FAT_SANGUE_SETOR_P321_FR_SETOR_ANA_Q_RECEITA_ANALIT
SELECT * FROM (
SELECT decode (SETOR.CD_SETOR, null, '00', SETOR.CD_SETOR )  CD_SET_SAN_ANA,
       decode (SETOR.NM_SETOR, null, 'SETOR NÃO ENCONTRADO',SETOR.NM_SETOR)  NM_SET_SAN_ANA,
       PRESTADOR.CD_PRESTADOR,
       PRESTADOR.NM_PRESTADOR,
       ATENDIME.CD_ATENDIMENTO,
       REG_FAT.NR_GUIA,
       PACIENTE.NM_PACIENTE,
       TO_CHAR(REG_FAT.DT_INICIO,'DD/MM') DT_INICIO,
       TO_CHAR(REG_FAT.DT_FINAL,'DD/MM')DT_FINAL,
       PROCEDIMENTO_SUS.CD_PROCEDIMENTO,
       PROCEDIMENTO_SUS.DS_PROCEDIMENTO,
       ITREG_FAT.CD_TIPO_VINCULO,
       nvl(ITREG_FAT.QT_LANCAMENTO,0) Quant,
       nvl(ITREG_FAT.QT_PONTOS,0) TOT_PONTOS,
       nvl(nvl(ITREG_FAT.VL_SH,ITREG_FAT.VL_ATO),0) VL_ATO
  FROM DBAMV.ITREG_FAT,
       DBAMV.PROCEDIMENTO_SUS,
       DBAMV.PROCEDIMENTO_SUS_DETALHE,
       DBAMV.REG_FAT,
       DBAMV.CONVENIO,
       DBAMV.REMESSA_FATURA,
       DBAMV.FATURA,
       DBAMV.PRESTADOR,
       DBAMV.ATENDIME,
       DBAMV.PACIENTE,
       DBAMV.SETOR
 WHERE CONVENIO.TP_CONVENIO     = 'H'
   AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   AND PROCEDIMENTO_SUS_DETALHE.SN_HEMOTERAPIA  = 'S'
   AND ITREG_FAT.CD_PROCEDIMENTO   = PROCEDIMENTO_SUS_DETALHE.CD_PROCEDIMENTO
   AND PROCEDIMENTO_SUS_DETALHE.CD_PROCEDIMENTO = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
   AND ITREG_FAT.CD_REG_FAT        = REG_FAT.CD_REG_FAT
   AND ITREG_FAT.CD_SETOR          = SETOR.CD_SETOR(+)
   AND REG_FAT.CD_REMESSA       = REMESSA_FATURA.CD_REMESSA
   AND REMESSA_FATURA.CD_FATURA = FATURA.CD_FATURA
   AND REG_FAT.CD_CONVENIO      = CONVENIO.CD_CONVENIO
   AND ITREG_FAT.CD_PRESTADOR   = PRESTADOR.CD_PRESTADOR
   AND REG_FAT.CD_ATENDIMENTO   = ATENDIME.CD_ATENDIMENTO
   AND ATENDIME.CD_PACIENTE     = PACIENTE.CD_PACIENTE
   AND to_char(FATURA.DT_COMPETENCIA, 'MM/YYYY') = to_char(:CAMPO_COMPET, 'MM/YYYY')
   :CAMPO_CF_REAPRESENTADAS
)
ORDER BY NM_SET_SAN_ANA,CD_SET_SAN_ANA,
NM_PRESTADOR,CD_PRESTADOR,
CD_ATENDIMENTO,NR_GUIA,NM_PACIENTE


/*----------------------------------------------------------------------------*/
--R_FAT_SANGUE_SETOR_P321_FR_SETOR_SIN_Q_RECEITA_SINT
SELECT decode (SETOR.CD_SETOR, null, '00', SETOR.CD_SETOR )  CD_SET_SAN_SIN,
       decode (SETOR.NM_SETOR, null, 'SETOR NÃO ENCONTRADO',SETOR.NM_SETOR)  NM_SET_SAN_SIN,
       DBAMV.PRESTADOR.CD_PRESTADOR,
       DBAMV.PRESTADOR.NM_PRESTADOR,
       DBAMV.PROCEDIMENTO_SUS.CD_PROCEDIMENTO,
       DBAMV.PROCEDIMENTO_SUS.DS_PROCEDIMENTO,
       DBAMV.ITREG_FAT.CD_TIPO_VINCULO,
       nvl(SUM(DBAMV.ITREG_FAT.QT_LANCAMENTO),0) PONTOS,
       nvl(SUM(DBAMV.ITREG_FAT.QT_PONTOS),0) TOT_PONTOS,
      (nvl(SUM(DBAMV.ITREG_FAT.QT_PONTOS)/SUM(DBAMV.ITREG_FAT.QT_LANCAMENTO),0)) UNI_PONTOS,
       sum(nvl(NVL(DBAMV.ITREG_FAT.VL_SH,DBAMV.ITREG_FAT.VL_ATO),0)) VL_ATO
  FROM DBAMV.ITREG_FAT,
       DBAMV.PROCEDIMENTO_SUS,
       DBAMV.PROCEDIMENTO_SUS_DETALHE,
       DBAMV.REG_FAT,
       DBAMV.CONVENIO,
       DBAMV.REMESSA_FATURA,
       DBAMV.FATURA,
       DBAMV.PRESTADOR,
       DBAMV.ATENDIME,
       DBAMV.PACIENTE,
       DBAMV.SETOR
 WHERE CONVENIO.TP_CONVENIO     = 'H'
   AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   AND PROCEDIMENTO_SUS_DETALHE.SN_HEMOTERAPIA  = 'S'
   AND ITREG_FAT.CD_PROCEDIMENTO   = PROCEDIMENTO_SUS_DETALHE.CD_PROCEDIMENTO
   AND PROCEDIMENTO_SUS_DETALHE.CD_PROCEDIMENTO = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
   AND ITREG_FAT.CD_REG_FAT        = REG_FAT.CD_REG_FAT
   AND ITREG_FAT.CD_SETOR          = SETOR.CD_SETOR(+)
   AND REG_FAT.CD_REMESSA       = REMESSA_FATURA.CD_REMESSA
   AND REMESSA_FATURA.CD_FATURA = FATURA.CD_FATURA
   AND REG_FAT.CD_CONVENIO      = CONVENIO.CD_CONVENIO
   AND ITREG_FAT.CD_PRESTADOR   = PRESTADOR.CD_PRESTADOR
   AND REG_FAT.CD_ATENDIMENTO   = ATENDIME.CD_ATENDIMENTO
   AND ATENDIME.CD_PACIENTE     = PACIENTE.CD_PACIENTE
   AND to_char(FATURA.DT_COMPETENCIA, 'MM/YYYY') = to_char(:CAMPO_COMPET, 'MM/YYYY')
GROUP BY SETOR.CD_SETOR,
         SETOR.NM_SETOR,
         PRESTADOR.NM_PRESTADOR,
         PRESTADOR.CD_PRESTADOR,
         PROCEDIMENTO_SUS.DS_PROCEDIMENTO,
         PROCEDIMENTO_SUS.CD_PROCEDIMENTO,
         ITREG_FAT.CD_TIPO_VINCULO
ORDER BY decode (SETOR.CD_SETOR, null, '00', SETOR.CD_SETOR ),PRESTADOR.CD_PRESTADOR,PROCEDIMENTO_SUS.CD_PROCEDIMENTO
