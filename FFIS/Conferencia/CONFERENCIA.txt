--CONFERENCIA
SELECT * FROM (
Select distinct reg_fat.cd_atendimento
     , itrg.cd_reg_fat
     , itrg.cd_procedimento
     , procSus.ds_procedimento
     , itrg.qt_lancamento
     , itrg.vl_sh
     , proc.vl_servico_hospitalar
     , proc.vl_servico_hospitalar * (nvl(mtem.vl_ivhe,0) / 100) vl_ivhe
     , Decode(procDet.sn_urgencia, 'S', proc.vl_servico_hospitalar * (nvl(mtem.vl_urgencia,0) / 100)
                                      , 0) vl_urgencia
  from dbamv.itreg_fat                  itrg
     , dbamv.reg_fat
     , dbamv.procedimento_sus_valor     proc
     , dbamv.procedimento_sus_detalhe   procDet
     , dbamv.procedimento_sus           procSus
     , dbamv.multi_empresas             mtem
 where itrg.cd_reg_fat          = reg_fat.cd_reg_fat
   and reg_fat.cd_multi_empresa = mtem.cd_multi_empresa
   and proc.cd_procedimento     = itrg.cd_procedimento
   and itrg.cd_procedimento     = procDet.cd_procedimento
   and itrg.cd_procedimento     = procSus.cd_procedimento
 Order by 1
)
ORDER BY CD_ATENDIMENTO,
CD_REG_FAT
