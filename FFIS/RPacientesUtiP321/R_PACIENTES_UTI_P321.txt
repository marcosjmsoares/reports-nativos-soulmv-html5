--R_PACIENTES_UTI_P321
select sysdate from dual


/*----------------------------------------------------------------------------*/
--R_PACIENTES_UTI_P321_RF_GUIA_ESP_Q_UTI_ESP
SELECT REG_FAT.NR_GUIA NR_GUI_ESP
     , REG_FAT.CD_PROCEDIMENTO_REALIZADO
     , REG_FAT.DT_INICIO
     , REG_FAT.DT_FINAL
     , PACIENTE.NM_PACIENTE NM_PAC_ESP
     , NVL(SUM(ITREG_FAT.QT_LANCAMENTO), 0) DIAS_ESP
     , NVL(SUM(ITREG_FAT.VL_SH), 0) SH_ESP
     , NVL(SUM(ITREG_FAT.VL_SP), 0) SP_ESP
     , NVL((NVL(SUM(ITREG_FAT.VL_SH),0)+
            NVL(SUM(ITREG_FAT.VL_SP),0)),0) TOTAL_ESP
     , PRESTADOR.NM_PRESTADOR
     , PROCSUS.CD_PROCEDIMENTO
  FROM DBAMV.REG_FAT
     , DBAMV.ATENDIME
     , DBAMV.REMESSA_FATURA
     , DBAMV.FATURA
     , DBAMV.PACIENTE
     , DBAMV.ITREG_FAT
     , DBAMV.PRESTADOR
     , DBAMV.PROCEDIMENTO_SUS PROCSUS
     , DBAMV.PROCEDIMENTO_SUS_DETALHE PROCSD
 WHERE REG_FAT.Cd_Multi_Empresa  = :CAMPO_CD_MULTI_EMPRESA
   AND REG_FAT.CD_REMESSA        = REMESSA_FATURA.CD_REMESSA
   AND REMESSA_FATURA.CD_FATURA  = FATURA.CD_FATURA
   AND ATENDIME.CD_PACIENTE      = PACIENTE.CD_PACIENTE
   AND ITREG_FAT.CD_REG_FAT      = REG_FAT.CD_REG_FAT
   AND ITREG_FAT.cd_procedimento = PROCSD.cd_procedimento
   AND ITREG_FAT.CD_PRESTADOR    = PRESTADOR.CD_PRESTADOR
   AND ATENDIME.CD_ATENDIMENTO   = REG_FAT.CD_ATENDIMENTO
   AND PROCSUS.CD_PROCEDIMENTO   = PROCSD.CD_PROCEDIMENTO
   AND PROCSD.sn_diaria_uti      = 'S'
   AND dbamv.pkt_config_ffis.retorna_campo('CD_CONVENIO') = FATURA.CD_CONVENIO
   AND to_char(FATURA.DT_COMPETENCIA, 'MM/YYYY')          = :CAMPO_COMPET
GROUP BY REG_FAT.NR_GUIA
       , REG_FAT.CD_PROCEDIMENTO_REALIZADO
       , REG_FAT.DT_INICIO
       , REG_FAT.DT_FINAL
       , PACIENTE.NM_PACIENTE
       , PRESTADOR.NM_PRESTADOR
       , PROCSUS.CD_PROCEDIMENTO


/*----------------------------------------------------------------------------*/
--R_PACIENTES_UTI_P321_RF_UTI1_Q_UTI_ESP_S
SELECT substr(ITREG_FAT.CD_SUS,1,5) UTI
     , PROCSUS.DS_PROCEDIMENTO DS_PROCED_ESP_S
     , REG_FAT.NR_GUIA NR_GUI_ESP_S
     , PACIENTE.CD_PACIENTE          CD_PAC_ESP_S
     , PACIENTE.NM_PACIENTE NM_PAC_ESP_S
     , DECODE('$V{F_EXIBIR_PERIODO_COBRANCA}','S',DIARIA_UTI_SUS.DT_COMPETENCIA,null) DT_COMPETENCIA_ESP_S
     , NVL(SUM(DIARIA_UTI_SUS.QT_LANCADA), 0)     DIAS_ESP_S
     , NVL(SUM((ITREG_FAT.VL_SH/ITREG_FAT.QT_LANCAMENTO)*DIARIA_UTI_SUS.QT_LANCADA),0)  SH_ESP_S
     , NVL(SUM((ITREG_FAT.VL_SP/ITREG_FAT.QT_LANCAMENTO)*DIARIA_UTI_SUS.QT_LANCADA), 0) SP_ESP_S
     , NVL((NVL(SUM((ITREG_FAT.VL_SH/ITREG_FAT.QT_LANCAMENTO)*DIARIA_UTI_SUS.QT_LANCADA),0) +
       NVL(SUM((ITREG_FAT.VL_SP/ITREG_FAT.QT_LANCAMENTO)*DIARIA_UTI_SUS.QT_LANCADA), 0)),0) TOTAL_ESP_S
  FROM DBAMV.REG_FAT
     , DBAMV.ATENDIME
     , DBAMV.REMESSA_FATURA
     , DBAMV.FATURA
     , DBAMV.PACIENTE
     , DBAMV.ITREG_FAT
     , DBAMV.PROCEDIMENTO_SUS PROCSUS
     , DBAMV.PROCEDIMENTO_SUS_DETALHE PROCSD
     , DBAMV.DIARIA_UTI_SUS
 WHERE REG_FAT.Cd_Multi_Empresa  = dbamv.pkg_mv2000.le_empresa()
   AND REG_FAT.CD_REMESSA        = REMESSA_FATURA.CD_REMESSA
   AND REMESSA_FATURA.CD_FATURA  = FATURA.CD_FATURA
   AND ATENDIME.CD_PACIENTE      = PACIENTE.CD_PACIENTE
   AND ITREG_FAT.CD_REG_FAT      = REG_FAT.CD_REG_FAT
   AND ITREG_FAT.cd_procedimento = PROCSD.cd_procedimento
   AND ATENDIME.CD_ATENDIMENTO   = REG_FAT.CD_ATENDIMENTO
   AND PROCSUS.CD_PROCEDIMENTO   = PROCSD.CD_PROCEDIMENTO
   AND PROCSD.sn_diaria_uti      = 'S'
   AND dbamv.pkt_config_ffis.retorna_campo('CD_CONVENIO') = FATURA.CD_CONVENIO
   AND to_char(FATURA.DT_COMPETENCIA, 'MM/YYYY')          = :CAMPO_COMPET
   AND DIARIA_UTI_SUS.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
   AND DIARIA_UTI_SUS.CD_LANCAMENTO = ITREG_FAT.CD_LANCAMENTO
 GROUP BY substr(ITREG_FAT.CD_SUS,1,5)
        , PROCSUS.DS_PROCEDIMENTO
        , REG_FAT.NR_GUIA
        , PACIENTE.CD_PACIENTE
        , PACIENTE.NM_PACIENTE
        , DECODE('$V{F_EXIBIR_PERIODO_COBRANCA}','S',DIARIA_UTI_SUS.DT_COMPETENCIA,null)
