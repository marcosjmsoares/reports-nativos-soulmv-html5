--R_EMITE_ETIQUETA_ACH
SELECT REG_FAT.NR_LAUDO_ACH AS NR_LAUDO_ACH
           ,  substr(REG_FAT.NR_LAUDO_ACH, 1, 4) || '/'
            || substr(REG_FAT.NR_LAUDO_ACH, 5, 2) || '-'
            || substr(REG_FAT.NR_LAUDO_ACH, 7, 6) || '-'
            || substr(REG_FAT.NR_LAUDO_ACH, 13, 1)  AS LAUDO_EDIT
   FROM
              DBAMV.REG_FAT
 WHERE REG_FAT.NR_LAUDO_ACH IS NOT NULL
  ORDER BY NR_LAUDO_ACH
