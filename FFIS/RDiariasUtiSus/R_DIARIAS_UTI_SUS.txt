--R_DIARIAS_UTI_SUS
SELECT r.nr_guia, r.cd_atendimento, r.cd_reg_fat, r.cd_remessa, i.cd_procedimento, d.qt_lancada
         FROM dbamv.ESPECIALIDADE_UTI_SUS e
            , dbamv.diaria_uti_sus d
            , dbamv.itreg_fat      i                                                                        
            , dbamv.reg_fat        r
            , dbamv.remessa_fatura rf
            , dbamv.fatura         f
        WHERE f.cd_fatura       = rf.cd_fatura
          AND rf.cd_remessa     = r.cd_remessa
          AND r.cd_reg_fat      = i.cd_reg_fat
          AND i.cd_reg_fat      = d.cd_reg_fat
          AND i.cd_lancamento   = d.cd_lancamento
          AND i.cd_procedimento = e.cd_procedimento
          AND r.cd_reg_fat NOT IN (select cd_reg_fat from dbamv.glosas_sus)
          AND i.sn_pertence_pacote = 'N'
          AND f.cd_fatura                     = :CAMPO_CD_FATURA
          AND d.dt_competencia        = :CAMPO_DT_COMPETENCIA
          AND e.cd_especialidade_uti = :CAMPO_CD_ESPECIALIDADE
          AND r.cd_multi_empresa     = :CAMPO_CD_MULTI_EMPRESA
     ORDER BY r.NR_GUIA, r.CD_ATENDIMENTO, r.CD_REG_FAT
