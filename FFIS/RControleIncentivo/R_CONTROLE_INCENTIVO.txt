--R_CONTROLE_INCENTIVO
SELECT ir.cd_procedimento, o.ds_procedimento, c.cd_atendimento, a.cd_paciente, p.nm_paciente, c.cd_reg_fat, c.nr_guia, ir.qt_lancamento
  FROM dbamv.fatura         f
     , dbamv.remessa_fatura r
     , dbamv.atendime       a
     , dbamv.reg_fat        c
     , dbamv.itreg_fat     ir
     , dbamv.paciente       p
     , dbamv.incentivo_diferenciado_sus    i
     , dbamv.it_incentivo_diferenciado_sus t
     , dbamv.procedimento_sus o
     , dbamv.reg_fat_incentivo_sus e
 WHERE f.cd_convenio       = dbamv.pkt_config_ffis.retorna_campo('CD_CONVENIO')
   AND f.cd_convenio       = c.cd_convenio
   AND f.cd_convenio       = a.cd_convenio
   AND TRUNC(f.dt_competencia, 'MM') = TRUNC(:CAMPO_COMPET, 'MM')
   AND c.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
   AND c.cd_multi_empresa = i.cd_multi_empresa
   AND f.cd_fatura         = r.cd_fatura
   AND r.cd_remessa        = c.cd_remessa
   AND c.cd_reg_fat       = ir.cd_reg_fat
   AND e.cd_reg_fat       = c.cd_reg_fat
   AND c.cd_atendimento   = a.cd_atendimento
   AND a.cd_paciente       = p.cd_paciente
   AND ir.cd_procedimento = t.cd_procedimento
   AND t.cd_procedimento   = o.cd_procedimento
   AND i.cd_incentivo      = t.cd_incentivo
   AND TRUNC(:CAMPO_COMPET, 'MM') BETWEEN i.dt_vigencia_ini AND Nvl( i.dt_vigencia_fim, TRUNC(:CAMPO_COMPET, 'MM') )
