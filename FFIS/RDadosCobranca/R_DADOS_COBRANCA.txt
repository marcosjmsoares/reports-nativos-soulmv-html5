SELECT esus.CD_ESPEC_SUS                            CD_ESPEC_SUS
     , esus.DS_ESPEC_SUS                            DS_ESPEC_SUS
     , refa.NR_GUIA                                 NR_GUIA
     , LPAD(paci.CD_PACIENTE, 8, '0')               CD_PACIENTE
     , empr.CD_ORGAO_LOCAL                          CD_ORGAO_LOCAL
     , refa.NR_IDENTIFICACAO                        NR_IDENTIFICACAO
     , paci.NM_PACIENTE                             NM_PACIENTE
     , paci.DS_ENDERECO                             DS_ENDERECO
     , paci.NR_ENDERECO                             NR_ENDERECO
     , paci.DS_COMPLEMENTO                          DS_COMPLEMENTO
     , cida.NM_CIDADE                               NM_CIDADE
     , cida.CD_UF                                   CD_UF
     , paci.NR_CEP                                  NR_CEP
     , paci.DT_NASCIMENTO                           DT_NASCIMENTO
     , paci.TP_SEXO                                 TP_SEXO
     , refa.NR_AIH_POSTERIOR                        NR_AIH_POSTERIOR
     , refa.NR_AIH_ANTERIOR                         NR_AIH_ANTERIOR
     , resp.NM_RESPONSAVEL                          NM_RESPONSAVEL
     , LPAD(paci.NR_CPF, 11, '0')                   NR_CPF
     , paci.NR_IDENTIDADE                           NR_IDENTIDADE
     , paci.NR_CNS                                  NR_CNS
     , paci.NR_RG_NASC                              NR_RG_NASC
     , LPAD(pres.NR_CPF_CGC, 14, '0')               CPFSolic
     , prfa.CD_SUS                                  ProcSolic
     , refa.CD_CARATER_INTERNACAO                   CD_CARATER_INTERNACAO
     , refa.DT_EMISSAO_LAUDO                        DT_EMISSAO_LAUDO
     , LPAD(pre1.NR_CPF_CGC, 14, '0')               CPFResp
     , leit.DS_ENFERMARIA                           DS_ENFERMARIA
     , leit.CD_LEITO                                CD_LEITO
     , refa.DT_INICIO                               DT_INICIO
     , empr.SN_AUDITOR_DIR_CLINICO                  SN_AUDITOR_DIR_CLINICO
     , decode(empr.SN_AUDITOR_DIR_CLINICO,'S', LPAD(pre2.NR_CPF_CGC, 14, '0'),LPAD(pre3.nr_cpf_cgc, 14, '0'))         CPFAudit
     , refa.NR_UTI_INICIAL                          NR_UTI_INICIAL
     , refa.NR_UTI_ANTERIOR                         NR_UTI_ANTERIOR
     , refa.NR_UTI_ALTA                             NR_UTI_ALTA
     , refa.NR_DIARIA_ACOMPANHANTE                  NR_DIARIA_ACOMPANHANTE
     , prfa1.CD_SUS                                 CD_SUS_Real
     , refa.DT_FINAL                                DT_FINAL
     , refa.CD_CID_PRINCIPAL                        CD_CID_PRINCIPAL
     , refa.CD_CID_SECUNDARIO                       CD_CID_SECUNDARIO
     , itpa.QT_RN_VIVO                              QT_RN_VIVO
     , itpa.QT_RN_MORTO                             QT_RN_MORTO
     , itpa.QT_RN_ALTA                              QT_RN_ALTA
     , itpa.QT_RN_TRANSFERIDO                       QT_RN_TRANSFERIDO
     , itpa.QT_RN_OBITO                             QT_RN_OBITO
     , LPAD(refa.CD_REG_FAT, 8 , '0')               CD_REG_FAT
     , moal.CD_ALTA_SUS                             CD_ALTA_SUS
     , refa.SN_FATURA_IMPRESSA                      SN_FATURA_IMPRESSA
     , LPAD(refa.CD_ATENDIMENTO, 10, '0')           CD_ATENDIMENTO
     , refa.CD_REMESSA                              CD_REMESSA
     , cida.CD_IBGE                                 CD_IBGE
     , refa.NR_PRE_NATAL                            NR_PRE_NATAL
     , decode(substr(moal.CD_ALTA_SUS,1,1),'4',aten.cd_cid_obito,'5',aten.cd_cid_obito,null) cd_cid
  FROM DBAMV.ATENDIME aten
     , DBAMV.CIDADE cida
     , DBAMV.CONVENIO conv
     , DBAMV.MULTI_EMPRESAS empr
     , DBAMV.FATURA fatu
     , DBAMV.IT_PARTO itpa
     , DBAMV.LEITO leit
     , DBAMV.PACIENTE paci
     , DBAMV.PRESTADOR pre1
     , DBAMV.PRESTADOR pre2
     , DBAMV.PRESTADOR pre3
     , DBAMV.PRESTADOR pres
     , DBAMV.PRO_FAT prfa
     , DBAMV.PRO_FAT prfa1
     , DBAMV.REG_FAT refa
     , DBAMV.REMESSA_FATURA remf
     , DBAMV.RESPONSA resp
     , DBAMV.ESPEC_SUS esus
     , DBAMV.MOT_ALT moal
 WHERE aten.CD_PACIENTE                        = paci.CD_PACIENTE
   AND refa.CD_ATENDIMENTO                     = aten.CD_ATENDIMENTO
   AND refa.CD_REMESSA                         = remf.CD_REMESSA
   AND remf.CD_FATURA                          = fatu.CD_FATURA
   AND to_char(fatu.DT_COMPETENCIA, 'MM/YYYY') = to_char(:CAMPO_COMPET, 'MM/YYYY')
   AND refa.CD_CONVENIO                        = conv.CD_CONVENIO
   AND paci.CD_CIDADE                          = cida.CD_CIDADE
   AND resp.CD_ATENDIMENTO (+)                 = aten.CD_ATENDIMENTO
   AND aten.CD_PRESTADOR                       = pres.CD_PRESTADOR
   AND refa.CD_PRO_FAT_SOLICITADO              = prfa.CD_PRO_FAT
   AND refa.CD_PRESTADOR_RESPONSAVEL           = pre1.CD_PRESTADOR
   AND aten.CD_LEITO                           = leit.CD_LEITO
   AND empr.CD_PRESTADOR_DIR_CLINICO           = pre2.CD_PRESTADOR
   AND refa.CD_PRESTADOR_AUDITOR               = pre3.CD_PRESTADOR(+)
   AND nvl(refa.CD_PRO_FAT_REALIZADO,'X0000000') = nvl(prfa1.CD_PRO_FAT,'X0000000')
   AND itpa.CD_REG_FAT (+)                     = refa.CD_REG_FAT
   AND refa.CD_ESPEC_SUS                       = esus.CD_ESPEC_SUS
   AND refa.CD_MOT_ALT                         = moal.CD_MOT_ALT
   AND conv.TP_CONVENIO                        = 'H'
   AND empr.CD_MULTI_EMPRESA                   = :CAMPO_CD_MULTI_EMPRESA

--R_DADOS_COBRANCA_RF_SERV_PROF2_Q_SERV_PROF
SELECT * FROM (
SELECT itre.CD_REG_FAT CD_REG_FAT1,
       decode(substr(itre.CD_SUS,1,2), '93', 19, decode(itre.CD_ATI_MED, Null, itla.CD_ATI_MED, itre.CD_ATI_MED)) CD_ATI_MED,
       decode(substr(itre.CD_SUS,1,2), '93', Decode(Forn.Sn_Cobranca_Fornec, 'S', 1 , decode( forn.sn_cessao_credito, 'N', 53, decode(hosp.cd_hospital,'339',54,55) ) ) , decode(itre.CD_TIPO_VINCULO, Null, itla.CD_TIPO_VINCULO, itre.CD_TIPO_VINCULO)) CD_TIPO_VINCULO,
       decode(copr.DS_CODIGO_COBRANCA, null, prfa.CD_SUS, copr.DS_CODIGO_COBRANCA) CD_SUS1,
       itre.QT_LANCAMENTO,
       decode(substr(prfa.CD_SUS,1,2), '93', itco.NR_DOCUMENTO, itre.TP_ALTA_UTI_NEO||lpad(to_char(itre.QT_PESO_RN),4,0)||to_char(itre.QT_MES_GESTACAO)) NF, 
       decode(pres2.NR_CPF_CGC, Null, decode(pres1.NR_CPF_CGC, Null, decode(forn.sn_cobranca_fornec, 'S', forn.nr_cgc_cpf, decode( forn.sn_cessao_credito, 'N', forn.nr_cgc_cpf, empr.CD_CGC) ) , pres1.NR_CPF_CGC) , pres2.NR_CPF_CGC) NR_CPF_CGC, 
       decode(pres2.TP_VINCULO, Null, decode(pres1.TP_VINCULO, Null, decode(forn.SN_COBRANCA_FORNEC, 'N','C', decode(forn.nr_cgc_cpf, Null, Null,'J')), pres1.TP_VINCULO), pres2.TP_VINCULO) TP_VINCULO, 
       decode(substr(itre.CD_SUS,1,2), '93',forn.NM_FANTASIA,
              decode(itre.cd_prestador, Null,pres1.nm_prestador, pres2.nm_prestador )) NM_FANTASIA,
       itre.cd_lancamento,
       Decode(Pres1.Cd_Prestador, Null, Pres2.Cd_Prestador_Repasse, Pres1.Cd_Prestador_Repasse) Cd_Prestador_Repasse,
       Decode(Pres1.Cd_Prestador, Null, Pres2.Cd_Prestador, Pres1.Cd_Prestador) Cd_Prestador
  FROM DBAMV.ITCOB_PRE itco,
       DBAMV.ITLAN_MED itla,
       DBAMV.ITREG_FAT itre,
       DBAMV.PRESTADOR pres1,
       DBAMV.PRESTADOR pres2,
       DBAMV.PRO_FAT prfa,
       DBAMV.FORNECEDOR forn,
       DBAMV.MULTI_EMPRESAS empr,
       ( select copr.CD_PRO_FAT,
                 copr.DS_CODIGO_COBRANCA,
                 copr.DS_NOME_COBRANCA   
            from dbamv.cod_pro copr
           where 
                  dbamv.pkt_config_ffis.retorna_campo('cd_convenio') = copr.cd_convenio
              And copr.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA  ) copr
    ,dbamv.hospital hosp 
 WHERE itre.SN_PERTENCE_PACOTE <>'S'
   AND itla.CD_LANCAMENTO (+)            = itre.CD_LANCAMENTO
   AND itla.CD_REG_FAT (+)                    = itre.CD_REG_FAT
   AND itre.CD_PRO_FAT                         = prfa.CD_PRO_FAT
   AND itco.CD_LANCAMENTO (+)           = itre.CD_LANCAMENTO
   AND itco.CD_REG_FAT (+)                   = itre.CD_REG_FAT
   AND copr.CD_PRO_FAT (+)                  = prfa.CD_PRO_FAT
   AND itla.CD_PRESTADOR                    = pres1.CD_PRESTADOR (+)
   AND itre.CD_PRESTADOR                    = pres2.CD_PRESTADOR (+)
   AND NVL(ITRE.SN_PERTENCE_PACOTE, 'N') = 'N' 
   AND ((NVL(itre.SN_PROCED_AUTORIZADO, 'N')  = 'S' AND itre.CD_SUS     <=  '91904013') OR
        (NVL(itre.SN_PROCED_AUTORIZADO, 'N') <> 'S'  AND itre.CD_SUS not in ('00000010'      
                                                                 ,'00000020'
                                                                 ,'00000030'
                                                                 ,'46809015'
                                                                 ,'46810013' )) OR 
        (NVL(itre.SN_PROCED_AUTORIZADO, 'N')  = 'S' AND itre.CD_SUS     in('97004057'
                                                                ,'97001058'
                                                                ,'98303015'
                                                                ,'98303023'
                                                                ,'98003020'
                                                                ,'98503014')))
   AND itco.CD_FORNECEDOR = forn.CD_FORNECEDOR (+)
   AND empr.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
  AND  empr.cd_multi_empresa = hosp.cd_multi_empresa   
ORDER BY 2, 10
)
WHERE CD_REG_FAT1 = :CAMPO_CD_REG_FAT
