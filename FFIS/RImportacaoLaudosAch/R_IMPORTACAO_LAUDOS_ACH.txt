--R_IMPORTACAO_LAUDOS_ACH
SELECT i.nr_linha
     , i.nr_laudo
     , i.tp_laudo
     , i.cd_procedimento
     , decode(i.tp_laudo, 0, ' ', RTrim(DBAMV.PKG_FFIS_PRODUCAO_ACH.F_RECUPERA_VALOR(16, I.DS_LINHA)) ) competencia
     , i.qt_autorizada
     , i.ds_validacao
     , decode(i.tp_laudo, 0,  RTrim(DBAMV.PKG_FFIS_PRODUCAO_ACH.F_RECUPERA_VALOR(3, I.DS_LINHA)), ' ') aih
     , r.cd_atendimento
     , r.cd_reg_fat
  FROM DBAMV.IMPORTACAO_LAUDOS_ACH i
     , DBAMV.REG_FAT r
 WHERE i.nr_laudo   = r.nr_laudo_ach(+)
 ORDER BY i.nr_laudo, i.tp_laudo
