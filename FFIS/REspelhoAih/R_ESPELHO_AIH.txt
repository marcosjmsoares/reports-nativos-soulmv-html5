--R_ESPELHO_AIH_R_2_Q_PREVIA_VALORES
select rownum linha
     , cd_reg_fat_previa
     , ds_organiza_grupo_procedimento
     , vl_sh_principal
     , vl_sh_proprio
     , vl_sh_terceiro
     , vl_sp_proprio
     , vl_sp_terceiro
     , vl_sp_rateio_proprio
     , vl_sp_rateio_terceiro
  from ( select cd_reg_fat_previa
              , ds_organiza_grupo_procedimento
              , vl_sh_principal
              , vl_sh_proprio
              , vl_sh_terceiro
              , vl_sp_proprio
              , vl_sp_terceiro
              , vl_sp_rateio_proprio
              , vl_sp_rateio_terceiro
           from ( select v.cd_reg_fat                           cd_reg_fat_previa
                       , substr(v.cd_procedimento,1,2) ||'.'||
                         substr(v.cd_procedimento,3,2) ||'.'||
                         substr(v.cd_procedimento,5,2) ||'-'||
                         ogp.ds_organiza_grupo_procedimento ds_organiza_grupo_procedimento
                       , sum(nvl(v.vl_sh_principal,0))          vl_sh_principal
                       , sum(nvl(v.vl_sh_proprio,0))            vl_sh_proprio
                       , sum(nvl(v.vl_sh_terceiro,0))           vl_sh_terceiro
                        , sum(nvl(v.vl_sp_direto_proprio,0))     vl_sp_proprio
                       , sum(nvl(v.vl_sp_direto_terceiro,0))    vl_sp_terceiro
                       , sum(nvl(v.vl_sp_rateio_proprio,0))     vl_sp_rateio_proprio
                       , sum(nvl(v.vl_sp_rateio_terceiro,0))    vl_sp_rateio_terceiro
                    from dbamv.organiza_grupo_procedimentos  ogp
                       , dbamv.v_ffis_valor_conta_aih v
                   where ogp.cd_grupo_procedimento          = substr(v.cd_procedimento,1,2)
                     and ogp.cd_sub_grupo_procedimento      = substr(v.cd_procedimento,3,2)
                     and ogp.cd_organiza_grupo_procedimento = substr(v.cd_procedimento,5,2)
                     and v.cd_reg_fat = :CAMPO_CP_CD_REG_FAT
                   Group By v.cd_reg_fat
                          , substr(v.cd_procedimento,1,2) ||'.'||
                            substr(v.cd_procedimento,3,2) ||'.'||
                            substr(v.cd_procedimento,5,2) ||'-'||
                            ogp.ds_organiza_grupo_procedimento
                  )
   )
 order by ds_organiza_grupo_procedimento
