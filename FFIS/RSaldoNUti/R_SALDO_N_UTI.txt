     , v_ffis_producao_uti.qt_uti_3_coronariana
     , (last_day(v_ffis_producao_uti.dt_compet_uti)-trunc(v_ffis_producao_uti.dt_compet_uti,'mm')+1) * nvl(empresa_uti.qt_uci,0) qt_sd_uci
     , (last_day(v_ffis_producao_uti.dt_compet_uti)-trunc(v_ffis_producao_uti.dt_compet_uti,'mm')+1) * nvl(empresa_uti.qt_ucinco,0) qt_sd_ucinco
     , (last_day(v_ffis_producao_uti.dt_compet_uti)-trunc(v_ffis_producao_uti.dt_compet_uti,'mm')+1) * nvl(empresa_uti.qt_ucinca,0) qt_sd_ucinca
     , (last_day(v_ffis_producao_uti.dt_compet_uti)-trunc(v_ffis_producao_uti.dt_compet_uti,'mm')+1) * nvl(empresa_uti.qt_uti_queimado,0) qt_sd_uti_queimado
     , (last_day(v_ffis_producao_uti.dt_compet_uti)-trunc(v_ffis_producao_uti.dt_compet_uti,'mm')+1) * nvl(empresa_uti.qt_uti_transplante,0) qt_sd_uti_transplante
     , (last_day(v_ffis_producao_uti.dt_compet_uti)-trunc(v_ffis_producao_uti.dt_compet_uti,'mm')+1) * nvl(empresa_uti.qt_uti_neonatal,0) qt_sd_uti_neonatal
     , (last_day(v_ffis_producao_uti.dt_compet_uti)-trunc(v_ffis_producao_uti.dt_compet_uti,'mm')+1) * nvl(empresa_uti.qt_uti_pediatrico,0) qt_sd_uti_pediatrico
     , (last_day(v_ffis_producao_uti.dt_compet_uti)-trunc(v_ffis_producao_uti.dt_compet_uti,'mm')+1) * nvl(empresa_uti.qt_uti,0) qt_sd_uti_adulto
     , (last_day(v_ffis_producao_uti.dt_compet_uti)-trunc(v_ffis_producao_uti.dt_compet_uti,'mm')+1) * nvl(empresa_uti.qt_uti_2_neo,0) qt_sd_uti_2_neo
     , (last_day(v_ffis_producao_uti.dt_compet_uti)-trunc(v_ffis_producao_uti.dt_compet_uti,'mm')+1) * nvl(empresa_uti.qt_uti_2_pediatrica,0) qt_sd_uti_2_pediatrica
     , (last_day(v_ffis_producao_uti.dt_compet_uti)-trunc(v_ffis_producao_uti.dt_compet_uti,'mm')+1) * nvl(empresa_uti.qt_uti_2_adulta,0) qt_sd_uti_2_adulta
     , (last_day(v_ffis_producao_uti.dt_compet_uti)-trunc(v_ffis_producao_uti.dt_compet_uti,'mm')+1) * nvl(empresa_uti.qt_uti_2_coronariana,0) qt_sd_uti_2_coronariana
     , (last_day(v_ffis_producao_uti.dt_compet_uti)-trunc(v_ffis_producao_uti.dt_compet_uti,'mm')+1) * nvl(empresa_uti.qt_uti_3_neo,0) qt_sd_uti_3_neo
     , (last_day(v_ffis_producao_uti.dt_compet_uti)-trunc(v_ffis_producao_uti.dt_compet_uti,'mm')+1) * nvl(empresa_uti.qt_uti_3_pediatrica,0) qt_sd_uti_3_pediatrica
     , (last_day(v_ffis_producao_uti.dt_compet_uti)-trunc(v_ffis_producao_uti.dt_compet_uti,'mm')+1) * nvl(empresa_uti.qt_uti_3_adulta,0) qt_sd_uti_3_adulta
     , (last_day(v_ffis_producao_uti.dt_compet_uti)-trunc(v_ffis_producao_uti.dt_compet_uti,'mm')+1) * nvl(empresa_uti.qt_uti_3_coronariana,0) qt_sd_uti_3_coronariana
  from dbamv.v_ffis_producao_uti
     , (select max(dt_vigencia) dt_vigencia
          from dbamv.empresa_uti
          where cd_multi_empresa = dbamv.pkg_mv2000.le_empresa
            and dt_vigencia <= :CAMPO_COMPETENCIA) max_uti
     , dbamv.empresa_uti
 where v_ffis_producao_uti.cd_multi_empresa = empresa_uti.cd_multi_empresa
  and empresa_uti.cd_multi_empresa = dbamv.pkg_mv2000.le_empresa
  and empresa_uti.dt_vigencia = max_uti.dt_vigencia
  and dt_compet_uti between trunc(add_months(:CAMPO_COMPETENCIA,-12),'mm') and last_day(:CAMPO_COMPETENCIA)
  order by v_ffis_producao_uti.dt_compet_uti desc
