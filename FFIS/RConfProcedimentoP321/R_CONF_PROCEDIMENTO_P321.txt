--R_CONF_PROCEDIMENTO_P321
select sysdate from dual


/*----------------------------------------------------------------------------*/
--R_CONF_PROCEDIMENTO_P321_R_1_Q_PROCEDIMENTOS
SELECT
       gru.cd_grupo_procedimento                cd_grupo_procedimento
     , gru.ds_grupo_procedimento                ds_grupo_procedimento
     , sub.CD_SUB_GRUPO_PROCEDIMENTO            CD_SUB_GRUPO_PROCEDIMENTO
     , sub.DS_SUB_GRUPO_PROCEDIMENTO            DS_SUB_GRUPO_PROCEDIMENTO
     , prfa.CD_PROCEDIMENTO                     CD_PROCEDIMENTO
     , prfa.DS_PROCEDIMENTO                     DS_PROCEDIMENTO
     , paci.CD_PACIENTE                         CD_PACIENTE
     , paci.NM_PACIENTE                         NM_PACIENTE
     , refa.CD_ATENDIMENTO                      CD_ATENDIMENTO
     , refa.NR_GUIA                             NR_GUIA
     , refa.CD_PROCEDIMENTO_REALIZADO           CD_PSUS_REA
     , refa.DT_INICIO                           DT_INICIO
     , refa.DT_FINAL                            DT_FINAL
     , itre.QT_LANCAMENTO                       QT_LANCAMENTO
     , refa.cd_mot_alt                          cd_mot_alt
     , refa.cd_espec_sus                        cd_espec_sus
     , espec_sus.ds_espec_sus                   ds_espec_sus
     , dbamv.pkg_ffis_calc_aih.fnc_sus_max_valor_sh(prfa.CD_PROCEDIMENTO,refa.cd_reg_fat) vl_sh
     , dbamv.pkg_ffis_calc_aih.fnc_sus_max_valor_sp(prfa.CD_PROCEDIMENTO,refa.cd_reg_fat)  vl_sp
  FROM DBAMV.ATENDIME aten
     , DBAMV.CONVENIO conv
     , DBAMV.FATURA fatu
     , DBAMV.GRU_PRO grpr
     , DBAMV.ITREG_FAT itre
     , DBAMV.PACIENTE paci
     , DBAMV.PROCEDIMENTO_SUS prfa
     , dbamv.grupo_procedimentos gru
     , DBAMV.SUB_GRUPO_PROCEDIMENTOS sub
     , DBAMV.REG_FAT refa
     , DBAMV.REMESSA_FATURA remf
     , DBAMV.PROCEDIMENTO_SUS_DETALHE PSUSDETALHE
     , dbamv.espec_sus
 WHERE refa.CD_CONVENIO    = conv.CD_CONVENIO
   and sub.cd_grupo_procedimento = gru.cd_grupo_procedimento
   and  prfa.CD_SUB_GRUPO_PROCEDIMENTO = sub.CD_SUB_GRUPO_PROCEDIMENTO
   AND prfa.CD_GRUPO_PROCEDIMENTO     =  sub.CD_GRUPO_PROCEDIMENTO
   AND conv.TP_CONVENIO    ='H'
   AND refa.CD_ATENDIMENTO = aten.CD_ATENDIMENTO
   AND aten.CD_PACIENTE    = paci.CD_PACIENTE
   AND refa.CD_REMESSA     = remf.CD_REMESSA
   AND remf.CD_FATURA      = fatu.CD_FATURA
   AND PSUSDETALHE.CD_GRU_PRO     = grpr.CD_GRU_PRO
   AND itre.CD_PROCEDIMENTO     = prfa.CD_PROCEDIMENTO
   AND prfa.cd_procedimento = PSUSDETALHE.CD_PROCEDIMENTO
   AND itre.CD_REG_FAT     = refa.CD_REG_FAT
   AND itre.SN_PERTENCE_PACOTE = :CAMPO_PACOTE
   AND to_char(fatu.DT_COMPETENCIA, 'MM/YYYY') = to_char(:CAMPO_COMPET, 'MM/YYYY')
   AND refa.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
   AND espec_sus.cd_espec_sus = refa.cd_espec_sus
  Order By gru.CD_GRUPO_PROCEDIMENTO
         , sub.CD_SUB_GRUPO_PROCEDIMENTO
         , itre.CD_PROCEDIMENTO
         , paci.NM_PACIENTE

--R_CONF_PROCEDIMENTO_P321_R_3_Q_GR_ESP
SELECT
       refa.cd_espec_sus                        cd_espec_sus
     , espec_sus.ds_espec_sus                   ds_espec_sus
     , gru.cd_grupo_procedimento                cd_grupo_procedimento
     , gru.ds_grupo_procedimento                ds_grupo_procedimento
     , sub.CD_SUB_GRUPO_PROCEDIMENTO            CD_SUB_GRUPO_PROCEDIMENTO
     , sub.DS_SUB_GRUPO_PROCEDIMENTO            DS_SUB_GRUPO_PROCEDIMENTO
     , prfa.CD_PROCEDIMENTO                     CD_PROCEDIMENTO
     , prfa.DS_PROCEDIMENTO                     DS_PROCEDIMENTO
     , paci.CD_PACIENTE                         CD_PACIENTE
     , paci.NM_PACIENTE                         NM_PACIENTE
     , refa.CD_ATENDIMENTO                      CD_ATENDIMENTO
     , refa.NR_GUIA                             NR_GUIA
     , refa.CD_PROCEDIMENTO_REALIZADO           CD_PSUS_REA
     , refa.DT_INICIO                           DT_INICIO
     , refa.DT_FINAL                            DT_FINAL
     , itre.QT_LANCAMENTO                       QT_LANCAMENTO
     , refa.cd_mot_alt                          cd_mot_alt
     , dbamv.pkg_ffis_calc_aih.fnc_sus_max_valor_sh(prfa.CD_PROCEDIMENTO,refa.cd_reg_fat) vl_sh
     , dbamv.pkg_ffis_calc_aih.fnc_sus_max_valor_sp(prfa.CD_PROCEDIMENTO,refa.cd_reg_fat)  vl_sp
  FROM DBAMV.ATENDIME aten
     , DBAMV.CONVENIO conv
     , DBAMV.FATURA fatu
     , DBAMV.GRU_PRO grpr
     , DBAMV.ITREG_FAT itre
     , DBAMV.PACIENTE paci
     , DBAMV.PROCEDIMENTO_SUS prfa
     , dbamv.grupo_procedimentos gru
     , DBAMV.SUB_GRUPO_PROCEDIMENTOS sub
     , DBAMV.REG_FAT refa
     , DBAMV.REMESSA_FATURA remf
     , DBAMV.PROCEDIMENTO_SUS_DETALHE PSUSDETALHE
     , dbamv.espec_sus
 WHERE refa.CD_CONVENIO    = conv.CD_CONVENIO
   and sub.cd_grupo_procedimento = gru.cd_grupo_procedimento
   and  prfa.CD_SUB_GRUPO_PROCEDIMENTO = sub.CD_SUB_GRUPO_PROCEDIMENTO
   AND prfa.CD_GRUPO_PROCEDIMENTO     =  sub.CD_GRUPO_PROCEDIMENTO
   AND conv.TP_CONVENIO    ='H'
   AND refa.CD_ATENDIMENTO = aten.CD_ATENDIMENTO
   AND aten.CD_PACIENTE    = paci.CD_PACIENTE
   AND refa.CD_REMESSA     = remf.CD_REMESSA
   AND remf.CD_FATURA      = fatu.CD_FATURA
   AND PSUSDETALHE.CD_GRU_PRO     = grpr.CD_GRU_PRO
   AND itre.CD_PROCEDIMENTO     = prfa.CD_PROCEDIMENTO
   AND prfa.cd_procedimento = PSUSDETALHE.CD_PROCEDIMENTO
   AND itre.CD_REG_FAT     = refa.CD_REG_FAT
   AND itre.SN_PERTENCE_PACOTE = :CAMPO_PACOTE
   AND to_char(fatu.DT_COMPETENCIA, 'MM/YYYY') = to_char(:CAMPO_COMPET, 'MM/YYYY')
   AND refa.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
   AND espec_sus.cd_espec_sus = refa.cd_espec_sus
  Order By refa.cd_espec_sus
         , gru.CD_GRUPO_PROCEDIMENTO
         , sub.CD_SUB_GRUPO_PROCEDIMENTO
         , itre.CD_PROCEDIMENTO
         , paci.NM_PACIENTE
