select especialidades.competencia
     , especialidades.cd_espec_sus cd_espec_ind
     , especialidades.ds_espec_sus
     , especialidades.contas
     , trunc((permanencia.permanencia/especialidades.contas),1) media_permanencia
  from ( select to_char(dt_competencia, 'mm/yyyy') competencia
              , cd_espec_sus
              , ds_espec_sus
              , count(cd_reg_fat) contas
           from ( select distinct v_ffis_fat_gerencial_total.dt_competencia
                        , espec_sus.cd_espec_sus
                        , espec_sus.ds_espec_sus
                        , v_ffis_fat_gerencial_total.cd_reg_fat
                     from dbamv.v_ffis_fat_gerencial_total
                        , dbamv.espec_sus
                    where v_ffis_fat_gerencial_total.cd_espec_sus                       = espec_sus.cd_espec_sus
                      and v_ffis_fat_gerencial_total.cd_multi_empresa                   = :CAMPO_CD_MULTI_EMPRESA
                      and to_char(v_ffis_fat_gerencial_total.dt_competencia, 'MM/YYYY') = :CAMPO_COMPET
                    group by v_ffis_fat_gerencial_total.dt_competencia
                           , espec_sus.cd_espec_sus
                           , espec_sus.ds_espec_sus
                           , v_ffis_fat_gerencial_total.cd_reg_fat )
            group by dt_competencia
            , cd_espec_sus
            , ds_espec_sus ) especialidades
     , ( select to_char(fatura.dt_competencia, 'mm/yyyy') competencia
              , reg_fat.cd_espec_sus
              , sum(reg_fat.dt_final-reg_fat.dt_inicio) permanencia
           from dbamv.remessa_fatura
              , dbamv.fatura
              , dbamv.reg_fat
          where remessa_fatura.cd_fatura = fatura.cd_fatura
            and reg_fat.cd_remessa       = remessa_fatura.cd_remessa
            and reg_fat.cd_convenio      = dbamv.pkt_config_ffis.retorna_campo('CD_CONVENIO')
            and reg_fat.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
            and to_char(fatura.dt_competencia, 'MM/YYYY') = :CAMPO_COMPET
          group by to_char(fatura.dt_competencia, 'mm/yyyy')
		         , reg_fat.cd_espec_sus ) permanencia
  where	especialidades.competencia  = permanencia.competencia
    and especialidades.cd_espec_sus = permanencia.cd_espec_sus
