--R_PROC_LANCAMENTOS
select sysdate from dual



/*----------------------------------------------------------------------------*/
--R_PROC_LANCAMENTOS_R_3_Q_LANCAMENTOS
Select Atendime.Cd_Atendimento
     , Paciente.NM_Paciente
     , Reg_Fat.Nr_Guia
     , Reg_Fat.Dt_Inicio
     , Reg_Fat.Dt_Final
     , ItReg_Fat.Cd_Prestador
     , Prestador.Nm_Prestador
     , OGP.Ds_Organiza_Grupo_Procedimento
     , ItReg_Fat.Cd_Procedimento
     , Procedimento_SUS.Ds_Procedimento
     , Sum(ItReg_Fat.Qt_Lancamento) Qt_Lancamento
     , Sum((Nvl(ItReg_Fat.Vl_SH,0) + Nvl(ItReg_Fat.Vl_SP,0))) Total_Ato
  From Dbamv.Paciente
     , Dbamv.Atendime
     , Dbamv.Reg_Fat
     , Dbamv.ItReg_Fat
     , Dbamv.Prestador
     , Dbamv.Remessa_Fatura
     , Dbamv.Fatura
     , Dbamv.Config_FFIS
     , Dbamv.Procedimento_SUS
     , Dbamv.Organiza_Grupo_Procedimentos OGP
     , Dbamv.Procedimento_SUS_Detalhe PDetal
 Where Paciente.Cd_Paciente         = Atendime.Cd_Paciente
   And Atendime.Cd_Atendimento      = Reg_Fat.Cd_Atendimento
   And Reg_Fat.Cd_Reg_Fat           = ItReg_Fat.Cd_Reg_Fat
   And ItReg_Fat.Cd_Prestador       = Prestador.Cd_Prestador(+)
   And Reg_Fat.Cd_Remessa           = Remessa_Fatura.Cd_Remessa
   And Remessa_Fatura.Cd_Fatura     = Fatura.Cd_Fatura
   And Config_FFIS.Cd_Multi_Empresa = Reg_Fat.Cd_Multi_Empresa
   And ItReg_fat.Cd_Procedimento    = Procedimento_SUS.Cd_Procedimento
   And PDetal.Cd_Procedimento       = Procedimento_SUS.Cd_Procedimento
   And OGP.Cd_Grupo_Procedimento           = Substr(ItReg_fat.Cd_Procedimento,1,2)
   And OGP.Cd_Sub_Grupo_Procedimento       = Substr(ItReg_fat.Cd_Procedimento,3,2)
   And OGP.Cd_Organiza_Grupo_Procedimento  = Substr(ItReg_fat.Cd_Procedimento,5,2)
   And To_Char(Fatura.Dt_Competencia,'MM/YYYY') = To_Char(:CAMPO_COMPET,'MM/YYYY')
   And Reg_Fat.Cd_Multi_Empresa     = :CAMPO_CD_MULTI_EMPRESA
 Group By Atendime.Cd_Atendimento
     , Paciente.NM_Paciente
     , Reg_Fat.Nr_Guia
     , Reg_Fat.Dt_Inicio
     , Reg_Fat.Dt_Final
     , ItReg_Fat.Cd_Prestador
     , Prestador.Nm_Prestador
     , OGP.Ds_Organiza_Grupo_Procedimento
     , ItReg_Fat.Cd_Procedimento
     , Procedimento_SUS.Ds_Procedimento
