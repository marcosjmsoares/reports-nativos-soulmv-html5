--R_LAUDO_SUPLEMENTAR
SELECT * FROM (
SELECT rownum
     , reg_fat.NR_GUIA
     , laudos_sih.NR_LAUDO
     , reg_fat.NR_SEQAIH5
     , multi_empresas.DS_MULTI_EMPRESA
     , multi_empresas.CD_CGC
     , paciente.NM_PACIENTE
     , paciente.DT_NASCIMENTO
     , cidade.NM_CIDADE||'/'|| cidade.CD_UF  MUNICIPIO
     , reg_fat.DT_INICIO
     , reg_fat.DT_FINAL
     , paciente.NM_MAE
     , item_laudos_sih.CD_GRU_LAUDO
     , proc.cd_procedimento
     , item_laudos_sih.QT_LANCAMENTO
     , pres_dir_cli.NM_PRESTADOR NM_PREST_DIR_CLI
     , laudos_sih.DT_LAUDO
     , prestador.NM_PRESTADOR
     , gru_laudo.SN_INFORMA_QTD
     , gru_laudo.SN_INFORMA_PROCED
     , hospital.ds_orgao_destino
  FROM DBAMV.PACIENTE paciente
     , DBAMV.CIDADE cidade
     , DBAMV.ATENDIME atendime
     , DBAMV.PRESTADOR pres_dir_cli
     , DBAMV.REG_FAT reg_fat
     , DBAMV.PRESTADOR prestador
     , DBAMV.LAUDOS_SIH laudos_sih
     , DBAMV.MULTI_EMPRESAS multi_empresas
     , DBAMV.ITEM_LAUDOS_SIH item_laudos_sih
     , DBAMV.GRU_LAUDO gru_laudo
     , DBAMV.PROCEDIMENTO_SUS proc
     , dbamv.hospital
 WHERE laudos_sih.CD_REG_FAT                   = reg_fat.CD_REG_FAT
   AND laudos_sih.CD_ATENDIMENTO               = atendime.CD_ATENDIMENTO
   AND atendime.CD_MULTI_EMPRESA               = multi_empresas.CD_MULTI_EMPRESA
   AND atendime.CD_PACIENTE                    = paciente.CD_PACIENTE
   AND paciente.CD_CIDADE                      = cidade.CD_CIDADE
   AND multi_empresas.CD_PRESTADOR_DIR_CLINICO = pres_dir_cli.CD_PRESTADOR
   AND reg_fat.CD_PRESTADOR_RESPONSAVEL        = prestador.CD_PRESTADOR
   and item_laudos_sih.cd_procedimento         = proc.cd_procedimento
   AND item_laudos_sih.CD_laudos_sih           = laudos_sih.CD_laudos_sih
   AND item_laudos_sih.CD_GRU_LAUDO            = gru_laudo.CD_GRU_LAUDO
   AND laudos_sih.TP_LAUDO                     = 'S'
   AND laudos_sih.cd_laudos_sih                  = :CAMPO_CD_LAUDOS_SIH
   and  hospital.cd_multi_empresa = multi_empresas.cd_multi_empresa
)
ORDER BY NR_GUIA,NR_LAUDO,NM_PACIENTE,DT_NASCIMENTO,NM_MAE,DT_INICIO,DT_FINAL,NR_SEQAIH5,DS_MULTI_EMPRESA,CD_CGC,MUNICIPIO,NM_PREST_DIR_CLI,DT_LAUDO,NM_PRESTADOR,DS_ORGAO_DESTINO
