--R_PROTOCOLO_MOV_CONTA
SELECT CD_PROTOCOLO
     , DS_REMESSA_ORIGEM
     , TP_COMPLEXIDADE
     , DS_SETOR_ORIGEM
     , DS_REMESSA_DESTINO
     , DS_SETOR_DESTINO
     , DS_OBSERVACAO
     , TP_TRANSFERENCIA
     , DT_TRANSFERENCIA
     , CD_USUARIO
     , CD_CONTA
     , NR_GUIA
     , DS_HISTORICO
     , TOTAL_GUIAS
 FROM (SELECT PT.CD_PROTOCOLO
             , '('||PT.CD_REMESSA_ORIGEM||') '||RE.DS_REMESSA DS_REMESSA_ORIGEM
             , DECODE(RE.TP_COMPLEXIDADE_PROCEDIMENTO,'AC','ALTA COMPLEXIDADE'
                                                     ,'MC','MÉDIA COMPLEXIDADE'
                                                     ,'CE','CIRÚRGIA ELETIVA'
                                                     ,'CR','CENTRAL DE REGULAÇÃO'
                                                     ,'NÃO DETERMINADO') TP_COMPLEXIDADE
             , '('||DBAMV.PKG_FFIS_REGRA_FATURA.F_SETOR_REMESSA(PT.CD_REMESSA_ORIGEM)
             ||') '|| DBAMV.PKG_SUS_REGRA_SETOR.F_DESCRICAO(DBAMV.PKG_FFIS_REGRA_FATURA.F_SETOR_REMESSA(PT.CD_REMESSA_ORIGEM)) DS_SETOR_ORIGEM
             , '('||PT.CD_REMESSA_DESTINO||') '||RD.DS_REMESSA DS_REMESSA_DESTINO
             , '('||DBAMV.PKG_FFIS_REGRA_FATURA.F_SETOR_REMESSA(PT.CD_REMESSA_DESTINO)
             ||') '|| DBAMV.PKG_SUS_REGRA_SETOR.F_DESCRICAO(DBAMV.PKG_FFIS_REGRA_FATURA.F_SETOR_REMESSA(PT.CD_REMESSA_DESTINO)) DS_SETOR_DESTINO
             , PT.DS_OBSERVACAO
             , DECODE(PT.TP_TRANSFERENCIA,'ENV'
                                         ,'ENVIO DE CONTAS (FATURAMENTO)'
                                         ,'DEV','DEVOLUÇÃO DE CONTAS (PRE-FATURAMENTO)'
                                         ,'NÃO DETERMINADO') TP_TRANSFERENCIA
             , TO_CHAR(PT.DT_TRANSFERENCIA,'DD/MM/YYYY') DT_TRANSFERENCIA
             , PT.CD_USUARIO
             , IT.CD_REG_FAT CD_CONTA
             , DBAMV.PKG_SUS_REGRA_TRANSFERE_LOTE.F_CONTA_GUIA('AIH', IT.CD_REG_FAT) NR_GUIA
             , IT.DS_HISTORICO
             , DBAMV.PKG_SUS_REGRA_TRANSFERE_LOTE.F_CONTAS_PROTOCOLO(PT.CD_PROTOCOLO) TOTAL_GUIAS
          FROM DBAMV.PROTOCOLO_REMESSA_AIH      PT
             , (SELECT I.CD_PROTOCOLO
        	         , I.CD_REG_FAT
        			 , J.DS_MOTIVO DS_HISTORICO
                  FROM DBAMV.ITEM_PROTOCOLO_REMESSA_AIH I
                     , DBAMV.MOTIVO_MOV_CONTA_AIH M
                     , DBAMV.MOT_MOVIMENTACAO_CONTA J
          	     WHERE I.CD_PROTOCOLO = M.CD_PROTOCOLO
          	       AND I.CD_REG_FAT    = M.CD_REG_FAT
          	       AND M.CD_MOTIVO     = J.CD_MOTIVO)IT
             , DBAMV.REMESSA_FATURA             RE
             , DBAMV.REMESSA_FATURA             RD
         WHERE PT.CD_PROTOCOLO  = IT.CD_PROTOCOLO(+)
           AND RE.CD_REMESSA    = PT.CD_REMESSA_ORIGEM
           AND RD.CD_REMESSA    = PT.CD_REMESSA_DESTINO
           AND PT.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
    UNION ALL
        SELECT PT.CD_PROTOCOLO
             , '('||PT.CD_REMESSA_ORIGEM||') '||RE.DS_REMESSA DS_REMESSA_ORIGEM
             , RE.TP_GUIA TP_COMPLEXIDADE
             , '('||DBAMV.PKG_FFIS_REGRA_FATURA.F_SETOR_REMESSA(PT.CD_REMESSA_ORIGEM)
             ||') '|| DBAMV.PKG_SUS_REGRA_SETOR.F_DESCRICAO(DBAMV.PKG_FFIS_REGRA_FATURA.F_SETOR_REMESSA(PT.CD_REMESSA_ORIGEM)) DS_SETOR_ORIGEM
             , '('||PT.CD_REMESSA_DESTINO||') '||RD.DS_REMESSA DS_REMESSA_DESTINO
             , '('||DBAMV.PKG_FFIS_REGRA_FATURA.F_SETOR_REMESSA(PT.CD_REMESSA_DESTINO)
             ||') '|| DBAMV.PKG_SUS_REGRA_SETOR.F_DESCRICAO(DBAMV.PKG_FFIS_REGRA_FATURA.F_SETOR_REMESSA(PT.CD_REMESSA_DESTINO)) DS_SETOR_DESTINO
             , PT.DS_OBSERVACAO
             , DECODE(PT.TP_TRANSFERENCIA,'ENV'
                                         ,'ENVIO DE CONTAS (FATURAMENTO)'
                                         ,'DEV','DEVOLUÇÃO DE CONTAS (PRE-FATURAMENTO)'
                                         ,'NÃO DETERMINADO') TP_TRANSFERENCIA
             , TO_CHAR(PT.DT_TRANSFERENCIA,'DD/MM/YYYY') DT_TRANSFERENCIA
             , PT.CD_USUARIO
             , IT.CD_APAC
             , DBAMV.PKG_SUS_REGRA_TRANSFERE_LOTE.F_CONTA_GUIA('APAC', IT.CD_APAC) NR_GUIA
             , IT.DS_HISTORICO
             , DBAMV.PKG_SUS_REGRA_TRANSFERE_LOTE.F_CONTAS_PROTOCOLO(PT.CD_PROTOCOLO) TOTAL_GUIAS
          FROM DBAMV.PROTOCOLO_REMESSA_APAC      PT
             , (SELECT I.CD_PROTOCOLO
        	         , I.CD_APAC
        			 , J.DS_MOTIVO DS_HISTORICO
                  FROM DBAMV.ITEM_PROTOCOLO_REMESSA_APAC I
                     , DBAMV.MOTIVO_MOV_CONTA_APAC M
                     , DBAMV.MOT_MOVIMENTACAO_CONTA J
          	     WHERE I.CD_PROTOCOLO  = M.CD_PROTOCOLO
          	       AND I.CD_APAC       = M.CD_APAC
          	       AND M.CD_MOTIVO     = J.CD_MOTIVO) IT
             , DBAMV.REMESSA_APAC             RE
             , DBAMV.REMESSA_APAC             RD
         WHERE PT.CD_PROTOCOLO  = IT.CD_PROTOCOLO(+)
           AND RE.CD_REMESSA    = PT.CD_REMESSA_ORIGEM
           AND RD.CD_REMESSA    = PT.CD_REMESSA_DESTINO
           AND PT.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA)
WHERE CD_PROTOCOLO = :CAMPO_PROTOCOLO
