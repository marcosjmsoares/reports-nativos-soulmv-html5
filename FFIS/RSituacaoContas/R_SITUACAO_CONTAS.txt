--R_SITUACAO_CONTAS
Select fatura.dt_competencia
     , reg_fat.cd_reg_fat
     , atendime.cd_atendimento
     , paciente.cd_paciente
     , paciente.nm_paciente
     , reg_fat.cd_procedimento_realizado
     , procedimento_sus.ds_procedimento
     , reg_fat.nr_guia
     , to_char(reg_fat.dt_inicio, 'DD/MM/YYYY') || ' - ' || to_char(reg_fat.dt_final, 'DD/MM/YYYY') periodo
     , nvl(reg_fat.dt_fechamento, remessa_fatura.dt_fechamento) dt_fechamento
     , remessa_fatura.dt_entrega_da_fatura
     , reg_fat.sn_fechada
     , remessa_fatura.sn_paga
     , decode(critica_aih.ds_critica, NULL, 'Não', 'Sim' ) Sn_Existe_Critica
  from Dbamv.paciente
     , dbamv.atendime
     , dbamv.reg_fat
     , dbamv.remessa_fatura
     , dbamv.fatura
     , dbamv.config_ffis
     , dbamv.procedimento_sus
     , dbamv.critica_aih
 where paciente.cd_paciente             = atendime.cd_paciente
   and atendime.cd_atendimento          = reg_fat.cd_atendimento
   and reg_fat.cd_remessa               = remessa_fatura.cd_remessa
   and remessa_fatura.cd_fatura         = fatura.cd_fatura
   and config_ffis.cd_multi_empresa     = reg_fat.cd_multi_empresa
   and procedimento_sus.cd_procedimento = reg_fat.cd_procedimento_realizado
   and To_Char(fatura.dt_competencia,'MM/YYYY') = :CAMPO_COMPET
   and reg_fat.cd_multi_empresa     = dbamv.pkg_mv2000.le_empresa
   and reg_fat.cd_reg_fat = critica_aih.cd_reg_fat (+)
group by fatura.dt_competencia
       , reg_fat.cd_reg_fat
       , atendime.cd_atendimento
       , paciente.cd_paciente
       , paciente.nm_paciente
       , reg_fat.cd_procedimento_realizado
       , procedimento_sus.ds_procedimento
       , reg_fat.nr_guia
       , to_char(reg_fat.dt_inicio, 'DD/MM/YYYY') || ' - ' || to_char(reg_fat.dt_final, 'DD/MM/YYYY')
       , nvl(reg_fat.dt_fechamento, remessa_fatura.dt_fechamento)
       , remessa_fatura.dt_entrega_da_fatura
       , reg_fat.sn_fechada
       , remessa_fatura.sn_paga
       , decode(critica_aih.ds_critica, NULL, 'Não', 'Sim' )
order by atendime.cd_atendimento
