SELECT * FROM (
SELECT gru_pro_sus.DS_GRU_PRO_SUS
     , pro_fat.CD_SUS
     , NVL(pro_fat.NR_AUXILIAR, 0)NR_AUXILIAR
     , pro_fat.NR_DIAS_INTERNACAO
     , pro_fat.NR_IDADE_MINIMA
     , pro_fat.NR_IDADE_MAXIMA
     , pro_fat.DS_SUS
     , pro_fat.CD_POR_ANE
     , pro_fat.CD_PRO_FAT
  FROM DBAMV.GRU_PRO_SUS gru_pro_sus
     , DBAMV.PRO_FAT pro_fat
 WHERE pro_fat.CD_GRU_PRO_SUS       = gru_pro_sus.CD_GRU_PRO_SUS
   AND pro_fat.CD_SUS              is not null
   :CAMPO_FC_ORDEM
)
ORDER BY DS_GRU_PRO_SUS
