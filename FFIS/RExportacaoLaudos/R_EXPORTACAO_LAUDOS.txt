--R_EXPORTACAO_LAUDOS
SELECT l.nr_lote            nr_lote
     , l.nr_seq                    nr_seq
     , l.nr_laudo                nr_laudo
     , l.cd_atendimento  cd_atendimento
     , l.cd_reg_fat             cd_reg_fat
     , r.nr_guia                  nr_guia
     , l.dh_geracao           dh_geracao
     , l.ds_falha                 ds_falha
  FROM DBAMV.LOG_EXPORTACAO_LAUDO_SUS l
            , DBAMV.REG_FAT r
WHERE l.cd_reg_fat = r.cd_reg_fat(+)
  AND l.tp_log  = :CAMPO_TP_RELATORIO
  AND l.nr_lote = :CAMPO_NR_LOTE
 ORDER BY l.nr_laudo, l.cd_atendimento, l.cd_reg_fat
