--R_ORDEM_PROCEDIMENTO
SELECT * FROM (
SELECT PROCEDIMENTO_SUS.CD_PROCEDIMENTO CD_SUS
     , PROCEDIMENTO_SUS.DS_PROCEDIMENTO DS_SUS
     , reg_fat.NR_GUIA
     , paciente.nm_paciente
     , to_char(reg_fat.dt_inicio,'dd/mm/yyyy') dt_inicio
     , to_char(reg_fat.dt_final,'dd/mm/yyyy') dt_final
     , reg_fat.cd_reg_fat
     , motivo_cobranca_sus.cd_motivo
     , motivo_cobranca_sus.ds_motivo
  FROM DBAMV.REG_FAT
     , DBAMV.PROCEDIMENTO_SUS
     , DBAMV.ATENDIME
     , DBAMV.PACIENTE
     , DBAMV.REMESSA_FATURA
     , DBAMV.FATURA
     , DBAMV.CONVENIO
     , DBAMV.motivo_cobranca_sus
 WHERE REG_FAT.CD_PROCEDIMENTO_REALIZADO = PROCEDIMENTO_SUS.CD_PROCEDIMENTO(+)
   AND reg_fat.CD_ATENDIMENTO = atendime.CD_ATENDIMENTO
   AND atendime.CD_PACIENTE = paciente.CD_PACIENTE
   AND PROCEDIMENTO_SUS.CD_PROCEDIMENTO is not null
   AND REG_FAT.CD_CONVENIO = CONVENIO.CD_CONVENIO
   AND REG_FAT.CD_REMESSA = REMESSA_FATURA.CD_REMESSA
   AND REMESSA_FATURA.CD_FATURA = FATURA.CD_FATURA
   AND motivo_cobranca_sus.cd_motivo(+) = REG_FAT.cd_mot_alt
   AND CONVENIO.TP_CONVENIO ='H'
   AND to_char(FATURA.DT_COMPETENCIA, 'MM/YYYY') = to_char(:CAMPO_COMPET, 'MM/YYYY')
   AND reg_fat.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
order by PROCEDIMENTO_SUS.CD_PROCEDIMENTO, paciente.nm_paciente
)
ORDER BY CD_SUS,DS_SUS
