--R_ATENDIMENTOS_UNIFICADOS
select atendime.cd_atendimento
     , reg_fat.cd_reg_fat_pai
     , reg_fat.cd_reg_fat
     , paciente.cd_paciente
     , paciente.nm_paciente
     , reg_fat.dt_inicio
     , reg_fat.dt_final
     , procedimento_sus.cd_procedimento
     , procedimento_sus.ds_procedimento
  from dbamv.reg_fat
     , dbamv.atendime
     , dbamv.paciente
     , dbamv.procedimento_sus
     , dbamv.fatura
     , dbamv.remessa_fatura
 where atendime.cd_atendimento                  = reg_fat.cd_atendimento
   and paciente.cd_paciente                     = atendime.cd_paciente
   and procedimento_sus.cd_procedimento         = nvl(reg_fat.cd_procedimento_realizado, reg_fat.cd_procedimento_solicitado)
   and fatura.cd_fatura                         = remessa_fatura.cd_fatura
   and remessa_fatura.cd_remessa                = reg_fat.cd_remessa
   and reg_fat.cd_multi_empresa                 = :CAMPO_CD_MULTI_EMPRESA
   and to_char(fatura.dt_competencia,'mm/yyyy') = to_char(:CAMPO_COMPET,'mm/yyyy')
   and remessa_fatura.sn_suspensas              = 'S'
 order by paciente.nm_paciente
