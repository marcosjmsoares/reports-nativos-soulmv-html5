--R_FAT_SANGUE_P321
SELECT * FROM (
SELECT
       PRESTADOR.CD_PRESTADOR,
       PRESTADOR.NM_PRESTADOR,
       PROCEDIMENTO_SUS.DS_PROCEDIMENTO,
       PROCEDIMENTO_SUS.CD_PROCEDIMENTO,
       ITREG_FAT.CD_TIPO_VINCULO,
       nvl(SUM(DBAMV.ITREG_FAT.QT_LANCAMENTO),0) PONTOS,
       nvl(SUM(DBAMV.ITREG_FAT.QT_PONTOS),0) TOT_PONTOS,
      (nvl(SUM(DBAMV.ITREG_FAT.QT_PONTOS)/SUM(DBAMV.ITREG_FAT.QT_LANCAMENTO),0)) UNI_PONTOS,
       sum(nvl(NVL(ITREG_FAT.VL_SH,ITREG_FAT.VL_ATO),0)) VL_ATO
  FROM DBAMV.ITREG_FAT,
       DBAMV.PROCEDIMENTO_SUS,
       DBAMV.PROCEDIMENTO_SUS_DETALHE PSUSDETALHE,
       DBAMV.GRU_PRO,
       DBAMV.REG_FAT,
       DBAMV.CONVENIO,
       DBAMV.REMESSA_FATURA,
       DBAMV.FATURA,
       DBAMV.PRESTADOR
 WHERE CONVENIO.TP_CONVENIO     = 'H'
   AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   AND PSUSDETALHE.SN_HEMOTERAPIA  = 'S'
   AND ITREG_FAT.CD_PROCEDIMENTO     = PSUSDETALHE.CD_PROCEDIMENTO
   AND PSUSDETALHE.CD_PROCEDIMENTO = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
   AND PSUSDETALHE.CD_GRU_PRO       = GRU_PRO.CD_GRU_PRO
   AND ITREG_FAT.CD_REG_FAT     = REG_FAT.CD_REG_FAT
   AND REG_FAT.CD_REMESSA       = REMESSA_FATURA.CD_REMESSA
   AND REMESSA_FATURA.CD_FATURA = FATURA.CD_FATURA
   AND REG_FAT.CD_CONVENIO      = CONVENIO.CD_CONVENIO
   AND ITREG_FAT.CD_PRESTADOR   = PRESTADOR.CD_PRESTADOR
   AND to_char(FATURA.DT_COMPETENCIA, 'MM/YYYY') = to_char(to_date(:CAMPO_COMPET, 'MM/YYYY'), 'MM/YYYY')
GROUP BY
         PRESTADOR.NM_PRESTADOR,
         PRESTADOR.CD_PRESTADOR,
         PROCEDIMENTO_SUS.DS_PROCEDIMENTO,
         PROCEDIMENTO_SUS.CD_PROCEDIMENTO,
         ITREG_FAT.CD_TIPO_VINCULO
)
ORDER BY CD_PRESTADOR, NM_PRESTADOR
