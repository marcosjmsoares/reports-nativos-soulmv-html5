--R_COMPATIBILIDADE_PROTESE
select 1 from dual

--R_COMPATIBILIDADE_PROTESE_RF_PROCEDIMENTO_Q_PROCEDIMENTO
select tmp.cd_procedimento cd_principal
     , pSus1.ds_procedimento ds_principal
     , tmp.cd_procedimento_compativel cd_compativel
     , pSus2.ds_procedimento ds_compativel
     , tmp.qt_maxima
     , pdv2.cd_codigo_reduzido cd_reduzido
     , tmp.dt_vigencia
     , psv.vl_total_internacao vl_total
  from dbamv.procedimento_sus pSus1
     , dbamv.procedimento_sus pSus2
     , dbamv.procedimento_detalhe_vigencia pdv1
     , dbamv.procedimento_detalhe_vigencia pdv2
     , dbamv.procedimento_sus_valor psv
     , ( select p_compat_p321.cd_procedimento            cd_procedimento
              , p_compat_p321.cd_procedimento_compativel cd_procedimento_compativel
              , p_compat_p321.qt_maxima
              , p_compat_p321.dt_vigencia
              , p_compat_p321.dt_validade_inicial
              , p_compat_p321.dt_validade_final
           from dbamv.procedimento_sus_compat_p321 p_compat_p321
          where p_compat_p321.tp_compatibilidade = '1'
            and trunc(sysdate,'MM') between trunc(dt_validade_inicial,'MM') and nvl(dt_validade_final, trunc(sysdate))) tmp
where pSus1.cd_procedimento = tmp.cd_procedimento
  and pSus2.cd_procedimento = tmp.cd_procedimento_compativel
  and pSus1.cd_procedimento = pdv1.cd_procedimento
  and pSus2.cd_procedimento = pdv2.cd_procedimento
  and pSus2.cd_procedimento = psv.cd_procedimento
  and psv.dt_vigencia       = dbamv.pkg_ffis_calc_aih.fnc_sus_max_vigencia_valor(pSus2.cd_procedimento)
  and trunc(tmp.dt_vigencia,'MM') between trunc(pdv1.dt_validade_inicial,'MM') and nvl(pdv1.dt_validade_final, trunc(tmp.dt_vigencia))
  and trunc(tmp.dt_vigencia,'MM') between trunc(pdv2.dt_validade_inicial,'MM') and nvl(pdv2.dt_validade_final, trunc(tmp.dt_vigencia))
  and pdv2.sn_ortese_protese = 'S'
  and pdv1.sn_aih_principal = 'S'
  and pSus1.sn_ativo = 'S'
  and pSus2.sn_ativo = 'S'
