--R_LAUDO_AIH_ESPECIAL
SELECT dbamv.pkg_sus_regra_paciente.F_NOME_PACIENTE(PACIENTE.CD_PACIENTE) NM_PACIENTE,
       PACIENTE.DT_NASCIMENTO,
       PACIENTE.TP_SEXO,
       PACIENTE.DS_ENDERECO || ', ' || PACIENTE.NR_ENDERECO || ' - ' || PACIENTE.NM_BAIRRO DS_ENDERECO ,
       PACIENTE.NR_CPF,
       PACIENTE.NR_CEP,
       PACIENTE.NR_CNS,
	   PACIENTE.NM_MAE,
	   PACIENTE.NR_FONE,
       DECODE(CONVENIO.TP_CONVENIO, 'C',  ATENDIME.CD_PRO_INT,
                                    'P',  ATENDIME.CD_PRO_INT,
                                    'A',  PROCEDIMENTO_SUS.CD_PROCEDIMENTO,
                                    'H',  PROCEDIMENTO_SUS.CD_PROCEDIMENTO)     CD_PRO_INT,
       LAUDO_AIH.CD_ESPEC_SUS,
       PRESTADOR.NR_CPF_CGC,
       PRESTADOR.nr_cns NR_CNS_SOLIC,
       PKG_SUS_REGRA_ATENDIMENTO.f_retorna_responsavel(ATENDIME.CD_ATENDIMENTO) NM_RESPONSAVEL,
       RESPONSA.NR_FONE FONE_RESP,
       SUBSTR( CIDADE.NM_CIDADE, 1, 30) NM_CIDADE,
       CIDADE.CD_UF,
       LAUDO_AIH.DT_EMISSAO,
       LAUDO_AIH.CD_CARATER_ATENDIMENTO,
       LAUDO_AIH.DS_SINAIS_SINT_CLINICOS,
       LAUDO_AIH.DS_COND_JUST_INTERNACAO,
       LAUDO_AIH.DS_RESULT_PROV_DIAG,
       LAUDO_AIH.DS_DIAGNOSTICO ,
       LAUDO_AIH.CD_CID_PRINCIPAL,
       LAUDO_AIH.CD_CID_SECUNDARIO,
       LAUDO_AIH.CD_CID_CAUSAS_ASSOCIADAS,
       PROCEDIMENTO_SUS.DS_PROCEDIMENTO,
       DBAMV.FN_IDADE (PACIENTE.DT_NASCIMENTO) IDADE,
       MULTI_EMPRESAS.NR_CNES,
       CIDADE.CD_IBGE,
       PACIENTE.NR_FONE,
       PRESTADOR.NM_PRESTADOR,
       LAUDO_AIH.CD_LAUDO,
       LAUDO_AIH.CD_PRESTADOR_SOL,
       DECODE(PACIENTE.Tp_Cor,'B','01 - Branca','P','02 - Preta','R','03 - Parda','A','04 - Amarela','I','05 - Indigena','99 - Sem Informação') COR_RACA,
       DECODE(ETNIA.CD_ETNIA,NULL,'0000 - Não Se Aplica',LPAD(ETNIA.CD_ETNIA,4,'0')||' - '||ETNIA.DS_ETNIA) DS_ETNIA
  FROM DBAMV.PACIENTE
      ,DBAMV.CIDADE
      ,DBAMV.ATENDIME
      ,DBAMV.PROCEDIMENTO_SUS
      ,DBAMV.CONVENIO
      ,DBAMV.PRESTADOR
      ,DBAMV.RESPONSA
      ,DBAMV.LAUDO_AIH
      ,DBAMV.EMPRESA_CONVENIO
      ,DBAMV.MULTI_EMPRESAS
      ,DBAMV.ETNIA
 WHERE EMPRESA_CONVENIO.CD_CONVENIO      =  CONVENIO.CD_CONVENIO
   AND EMPRESA_CONVENIO.CD_MULTI_EMPRESA =  MULTI_EMPRESAS.CD_MULTI_EMPRESA
   AND ATENDIME.CD_ATENDIMENTO           =  :CAMPO_PCD_ATENDIMENTO
   AND ATENDIME.CD_PACIENTE              = PACIENTE.CD_PACIENTE
   AND LAUDO_AIH.CD_PROCEDIMENTO         = PROCEDIMENTO_SUS.CD_PROCEDIMENTO (+)
   AND ATENDIME.CD_CONVENIO              = CONVENIO.CD_CONVENIO
   AND PACIENTE.CD_CIDADE                = CIDADE.CD_CIDADE(+)
   AND LAUDO_AIH.CD_PRESTADOR_SOL       = PRESTADOR.CD_PRESTADOR(+)
   AND ATENDIME.CD_ATENDIMENTO           = RESPONSA.CD_ATENDIMENTO (+)
   AND ATENDIME.CD_ATENDIMENTO           = LAUDO_AIH.CD_ATENDIMENTO (+)
   AND TRUNC(LAUDO_AIH.DT_EMISSAO(+))           = TRUNC(:CAMPO_PDT_EMISSAO)
   AND MULTI_EMPRESAS.CD_MULTI_EMPRESA   = :CAMPO_CD_MULTI_EMPRESA
   AND PACIENTE.CD_ETNIA                 = ETNIA.CD_ETNIA(+)
