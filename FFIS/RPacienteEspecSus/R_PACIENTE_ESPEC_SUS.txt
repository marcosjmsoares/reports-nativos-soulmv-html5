--R_PACIENTE_ESPEC_SUS
Select espec_sus.cd_espec_sus ||' '||espec_sus.ds_espec_sus ds_espec_sus
     , reg_fat.nr_guia
     , reg_fat.nr_identificacao
     , dbamv.pkg_sus_regra_paciente.F_NOME_PACIENTE(atendime.CD_PACIENTE) NM_PACIENTE
     , lpad(paciente.cd_paciente,8,0) cd_paciente
     , to_char(reg_fat.dt_inicio,'dd/mm/yyyy') dt_inicio
     , to_char(reg_fat.dt_final,'dd/mm/yyyy') dt_final
     , reg_fat.cd_procedimento_realizado||' '|| procedimento_sus.ds_procedimento ds_procedimento
     , reg_fat.cd_mot_alt||' '||motivo_cobranca_sus.ds_motivo ds_motivo
     , paciente.nr_cns
  From dbamv.reg_fat
     , dbamv.remessa_fatura
     , dbamv.fatura
     , dbamv.atendime
     , dbamv.paciente
     , dbamv.procedimento_sus
     , dbamv.espec_sus
     , dbamv.motivo_cobranca_sus
 where reg_fat.cd_remessa = remessa_fatura.cd_remessa
   and remessa_fatura.cd_fatura = fatura.cd_fatura
   and reg_fat.cd_atendimento = atendime.cd_atendimento
   and atendime.cd_paciente = paciente.cd_paciente
   and procedimento_sus.cd_procedimento = reg_fat.cd_procedimento_realizado
   and motivo_cobranca_sus.cd_motivo = reg_fat.cd_mot_alt
   and fatura.cd_convenio = dbamv.pkt_config_ffis.retorna_campo('CD_CONVENIO')
   and fatura.cd_multi_empresa = dbamv.pkg_mv2000.le_empresa
   and reg_fat.cd_multi_empresa = fatura.cd_multi_empresa
   and espec_sus.cd_espec_sus = reg_fat.cd_espec_sus
   and to_char(fatura.dt_competencia, 'mm/yyyy') = to_char(:CAMPO_COMPET, 'mm/yyyy')
