--R_DIARIAS_UTI_SUS_CORTE
SELECT r.nr_guia
            , to_char(c.dt_competencia, 'MM/YYYY') dt_competencia
            , c.cd_procedimento, e.ds_especialidade_uti
            , Nvl(c.qt_lancada_ini, 0) qt_lancada_ini, c.qt_lancada_atu
            , ( Nvl(c.qt_lancada_ini, 0) - c.qt_lancada_atu ) qt_corte
            , ( v.vl_total_internacao * ( Nvl(c.qt_lancada_ini, 0) - c.qt_lancada_atu ) ) vl_corte
         FROM dbamv.remessa_fatura    rf
            , dbamv.reg_fat            r
            , dbamv.corte_diaria_uti_sus c
            , dbamv.especialidade_uti_sus e
            , dbamv.procedimento_sus_valor v
        WHERE rf.cd_remessa      = r.cd_remessa
          AND r.cd_reg_fat       = c.cd_reg_fat
          AND c.cd_procedimento  = e.cd_procedimento
          AND c.cd_procedimento  = v.cd_procedimento
          AND c.dt_competencia   = v.dt_vigencia
          AND r.cd_multi_empresa = v.cd_multi_empresa
          AND rf.cd_fatura       = :CAMPO_CD_FATURA
          AND r.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
        ORDER BY r.NR_GUIA
