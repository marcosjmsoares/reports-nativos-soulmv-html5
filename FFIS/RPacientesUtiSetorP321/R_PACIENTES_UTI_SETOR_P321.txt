--R_PACIENTES_UTI_SETOR_P321
SELECT * FROM (
SELECT decode (SETOR.CD_SETOR, null, '00', SETOR.CD_SETOR ) CD_SET_ESP
     , decode (SETOR.NM_SETOR, null, 'SETOR NÃO ENCONTRADO',SETOR.NM_SETOR) NM_SET_ESP
     , REG_FAT.NR_GUIA NR_GUI_ESP
     , REG_FAT.CD_PROCEDIMENTO_REALIZADO
     , REG_FAT.DT_INICIO
     , REG_FAT.DT_FINAL
     , PACIENTE.NM_PACIENTE NM_PAC_ESP
     , NVL(SUM(ITREG_FAT.QT_LANCAMENTO), 0) DIAS_ESP
     , NVL(SUM(ITREG_FAT.VL_SH), 0) SH_ESP
     , NVL(SUM(ITREG_FAT.VL_SP), 0) SP_ESP
     , NVL((NVL(SUM(ITREG_FAT.VL_SH),0)+
            NVL(SUM(ITREG_FAT.VL_SP),0)),0) TOTAL_ESP
     , PRESTADOR.NM_PRESTADOR
     , PROCSUS.CD_PROCEDIMENTO
  FROM DBAMV.REG_FAT
     , DBAMV.ATENDIME
     , DBAMV.SETOR
     , DBAMV.REMESSA_FATURA
     , DBAMV.FATURA
     , DBAMV.PACIENTE
     , DBAMV.ITREG_FAT
     , DBAMV.PRESTADOR
     , DBAMV.PROCEDIMENTO_SUS PROCSUS
     , DBAMV.PROCEDIMENTO_SUS_DETALHE PROCSD
 WHERE REG_FAT.Cd_Multi_Empresa  = :CAMPO_CD_MULTI_EMPRESA
   AND REG_FAT.CD_REMESSA        = REMESSA_FATURA.CD_REMESSA
   AND REMESSA_FATURA.CD_FATURA  = FATURA.CD_FATURA
   AND ATENDIME.CD_PACIENTE      = PACIENTE.CD_PACIENTE
   AND ITREG_FAT.CD_REG_FAT      = REG_FAT.CD_REG_FAT
   AND ITREG_FAT.CD_SETOR        = SETOR.CD_SETOR(+)
   AND ITREG_FAT.cd_procedimento = PROCSD.cd_procedimento
   AND ITREG_FAT.CD_PRESTADOR    = PRESTADOR.CD_PRESTADOR
   AND ATENDIME.CD_ATENDIMENTO   = REG_FAT.CD_ATENDIMENTO
   AND PROCSUS.CD_PROCEDIMENTO   = PROCSD.CD_PROCEDIMENTO
   AND PROCSD.sn_diaria_uti      = 'S'
   AND dbamv.pkt_config_ffis.retorna_campo('CD_CONVENIO') = FATURA.CD_CONVENIO
   AND to_char(FATURA.DT_COMPETENCIA, 'MM/YYYY')          = to_char(to_date(:CAMPO_COMPET,'MM/YYYY'), 'MM/YYYY')
GROUP BY SETOR.CD_SETOR
       , SETOR.NM_SETOR
       , REG_FAT.NR_GUIA
       , REG_FAT.CD_PROCEDIMENTO_REALIZADO
       , REG_FAT.DT_INICIO
       , REG_FAT.DT_FINAL
       , PACIENTE.NM_PACIENTE
       , PRESTADOR.NM_PRESTADOR
       , PROCSUS.CD_PROCEDIMENTO
:CAMPO_FC_ORDEM
)
ORDER BY NM_SET_ESP,CD_SET_ESP
