SELECT VW.ID                  ID
     , VW.DESCRICAO           DESCRICAO
     , VP.CD_PRESTADOR        CD_PRESTADOR
     , PR.NM_PRESTADOR        NM_PRESTADOR
     , SUM(VP.QT_PONTOS)      QT_PONTOS
     , SUM(DIAS)              DIAS
     , SUM(VP.QT_LANCAMENTO)  QT_LANCAMENTO
     , SUM(VP.VL_SP_PROPRIO)
     + SUM(VP.VL_SP_TERCEIRO) VL_TOTAL
     , SUM(VP.VL_SP_PROPRIO)  VL_SP_PROPRIO
     , SUM(VP.VL_SP_TERCEIRO) VL_SP_TERCEIRO
     , SUM(VP.VL_SP_DIRETO_PROPRIO)  VL_SP_DI_PROPRIO
     , SUM(VP.VL_SP_DIRETO_TERCEIRO) VL_SP_DI_TERCEIRO
     , SUM(VP.VL_SP_RATEIO_PROPRIO)       VL_SP_RT_PROPRIO
     , SUM(VP.VL_SP_RATEIO_TERCEIRO)       VL_SP_RT_TERCEIRO
 FROM DBAMV.V_FFIS_PRODUCAO_AIH VW
    , DBAMV.V_FFIS_VALOR_PRESTADOR_AIH VP
    , DBAMV.PRESTADOR PR
WHERE VP.CD_REG_FAT                        = VW.CD_REG_FAT
  AND VP.CD_LANCAMENTO                     = VW.CD_LANCAMENTO
  AND VP.CD_PRESTADOR                      = PR.CD_PRESTADOR
  AND VP.CD_MULTI_EMPRESA                  = :CAMPO_CD_MULTI_EMPRESA
  AND TO_CHAR(VP.DT_COMPETENCIA,'MM/YYYY') = TO_CHAR(:CAMPO_COMPET,'MM/YYYY')
  AND (VP.VL_SP_PROPRIO + VP.VL_SP_TERCEIRO) > 0
GROUP BY VW.ID
       , VW.DESCRICAO
       , VP.CD_PRESTADOR
       , PR.NM_PRESTADOR
ORDER BY PR.NM_PRESTADOR
       , ID
