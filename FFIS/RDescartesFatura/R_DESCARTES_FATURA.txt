--R_DESCARTES_FATURA
SELECT D.CD_DESCARTE||' - '|| M.DS_DESCARTE DS_DESCARTE
     , R.NR_GUIA                            NR_GUIA
     , A.CD_ATENDIMENTO                     CD_ATENDIMENTO
     , C.CD_PACIENTE                        CD_PACIENTE
     , C.NM_PACIENTE                        NM_PACIENTE
     , I.CD_PROCEDIMENTO                    CD_PROCEDIMENTO
     , I.QT_LANCAMENTO                      QT_LANCAMENTO
     , P.DS_PROCEDIMENTO                    DS_PROCEDIMENTO
     , D.DS_OBSERVACAO                      DS_OBSERVACAO
     , D.NM_USUARIO                         NM_USUARIO
  FROM DBAMV.ITREG_FAT          I
      , DBAMV.ITEM_DESCARTE_AIH D
      , DBAMV.REG_FAT           R
      , DBAMV.REMESSA_FATURA    E
      , DBAMV.FATURA            F
      , DBAMV.PROCEDIMENTO_SUS  P
      , DBAMV.MOTIVO_DESCARTE_SUS M
      , DBAMV.ATENDIME            A
      , DBAMV.PACIENTE            C
  WHERE TO_CHAR(F.DT_COMPETENCIA,'MM/YYYY') = TO_CHAR(:CAMPO_COMPET,'MM/YYYY')
    AND I.CD_REG_FAT      = D.CD_REG_FAT
    AND I.CD_LANCAMENTO   = D.CD_LANCAMENTO
    AND I.CD_REG_FAT      = R.CD_REG_FAT
    AND R.CD_REMESSA      = E.CD_REMESSA
    AND F.CD_FATURA       = E.CD_FATURA
    AND P.CD_PROCEDIMENTO = I.CD_PROCEDIMENTO
    AND M.CD_DESCARTE     = D.CD_DESCARTE
    AND R.CD_ATENDIMENTO  = A.CD_ATENDIMENTO
    AND A.CD_PACIENTE     = C.CD_PACIENTE
    AND I.SN_PERTENCE_PACOTE = 'S'
    AND F.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
ORDER BY DS_DESCARTE
