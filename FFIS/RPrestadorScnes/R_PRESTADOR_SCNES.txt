--R_PRESTADOR_SCNES
SELECT 'PRESTADORES ATUALIZADOS' TP_ACAO
     , L.CD_PRESTADOR
     , P.NM_PRESTADOR
     , DECODE(P.TP_SITUACAO,'A','Sim','Não') SN_ATIVO
     , P.SN_CIRURGIAO||'  '||P.SN_AUXILIAR||'  '||P.SN_ANESTESISTA||'  '||P.SN_OUTROS TP_ATIVIDADES
     , DECODE(S.TP_VINCULO_MV,'U','Funcionário','F','P Física','J','P. Jurídica','O','Outros','C','Cooperado',null) TP_VINCULO_MV
     , DECODE(V.QT_HORAS_SIA,0,'Não','Sim') SIA
     , DECODE(V.QT_HORAS_SIH,0,'Não','Sim') SIH
     , S.CD_VINCULO_CNES
     ||S.CD_TIPO_VINCULO_CNES
     ||S.CD_SUBTIPO_VINCULO_CNES CD_VINCULO_CNES
     , C.CD_CBO
     , B.DS_CBOS
  FROM (SELECT CD_PROCESSA_SCNES
             , CD_PRESTADOR
             , TRUNC(DT_PROCESSAMENTO,'MM') DT_PROCESSAMENTO
          FROM DBAMV.LOG_PROCESSA_SCNES
         WHERE CD_PROCESSA_SCNES = :CAMPO_LOG
           GROUP BY CD_PROCESSA_SCNES
                  , CD_PRESTADOR
                  ,TRUNC(DT_PROCESSAMENTO,'MM')) L
     , DBAMV.PRESTADOR P
     , DBAMV.PRESTADOR_CBO C
     , DBAMV.SUBTIPO_VINCULO_CNES S
     , DBAMV.CBO B
     , (SELECT CD_PRESTADOR
             , SUM(QT_HORAS_SIA) QT_HORAS_SIA
             , SUM(QT_HORAS_SIH) QT_HORAS_SIH
          FROM DBAMV.PRESTADOR_CBO
         WHERE CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
         GROUP BY CD_PRESTADOR) V
 WHERE P.CD_PRESTADOR = L.CD_PRESTADOR
   AND P.CD_PRESTADOR = C.CD_PRESTADOR
   AND C.CD_VINCULO_CNES = S.CD_VINCULO_CNES
   AND C.CD_TIPO_VINCULO_CNES = S.CD_TIPO_VINCULO_CNES
   AND C.CD_SUBTIPO_VINCULO_CNES = S.CD_SUBTIPO_VINCULO_CNES
   AND C.CD_CBO = B.CD_CBOS
   AND V.CD_PRESTADOR = P.CD_PRESTADOR
   AND L.CD_PROCESSA_SCNES = :CAMPO_LOG
   AND C.SN_REGRA_VINCULO = 'S'
   AND L.DT_PROCESSAMENTO BETWEEN C.DT_VALIDADE_INICIAL AND NVL(C.DT_VALIDADE_FINAL,L.DT_PROCESSAMENTO)
   ORDER BY CD_PRESTADOR, CD_CBO
