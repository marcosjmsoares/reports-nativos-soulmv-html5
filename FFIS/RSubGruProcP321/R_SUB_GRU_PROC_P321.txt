--R_SUB_GRU_PROC_P321
SELECT SUB_GRUPO_PROCEDIMENTOS.CD_SUB_GRUPO_PROCEDIMENTO
     , SUB_GRUPO_PROCEDIMENTOS.DS_SUB_GRUPO_PROCEDIMENTO
     , PROCEDIMENTO_SUS.CD_PROCEDIMENTO
     , PROCEDIMENTO_SUS.DS_PROCEDIMENTO
     , Nvl(PROCEDIMENTO_SUS.NR_IDADE_MINIMA,0) NR_IDADE_MINIMA
     , Nvl(PROCEDIMENTO_SUS.NR_IDADE_MAXIMA,0) NR_IDADE_MAXIMA
     , Nvl(PROCEDIMENTO_SUS.NR_DIAS_INTERNACAO,0) NR_DIAS_INTERNACAO
     , NVL(PROCEDIMENTO_SUS.SN_ATO_ANESTESICO,'N') SN_ATO_ANESTESICO
     , PROCEDIMENTO_SUS_COMPLEXIDADE.TP_COMPLEXIDADE_PROCEDIMENTO
     , PROCEDIMENTO_SUS_DETALHE.NR_AUXILIAR
     , PROCEDIMENTO_SUS_DETALHE.SN_URGENCIA
     , PROCEDIMENTO_SUS_DETALHE.SN_PERMITE_PERMANENCIA_MAIOR
  FROM DBAMV.PROCEDIMENTO_SUS,
       DBAMV.SUB_GRUPO_PROCEDIMENTOS,
       DBAMV.PROCEDIMENTO_SUS_COMPLEXIDADE,
       DBAMV.PROCEDIMENTO_SUS_MODALIDADE,
       DBAMV.PROCEDIMENTO_SUS_DETALHE
 WHERE PROCEDIMENTO_SUS.CD_SUB_GRUPO_PROCEDIMENTO = SUB_GRUPO_PROCEDIMENTOS.CD_SUB_GRUPO_PROCEDIMENTO
   AND PROCEDIMENTO_SUS.CD_GRUPO_PROCEDIMENTO     = SUB_GRUPO_PROCEDIMENTOS.CD_GRUPO_PROCEDIMENTO
   AND PROCEDIMENTO_SUS.CD_PROCEDIMENTO           = PROCEDIMENTO_SUS_COMPLEXIDADE.CD_PROCEDIMENTO
   AND PROCEDIMENTO_SUS.CD_PROCEDIMENTO           = PROCEDIMENTO_SUS_MODALIDADE.CD_PROCEDIMENTO
   AND PROCEDIMENTO_SUS_DETALHE.CD_PROCEDIMENTO   = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
 GROUP BY SUB_GRUPO_PROCEDIMENTOS.CD_SUB_GRUPO_PROCEDIMENTO
     , SUB_GRUPO_PROCEDIMENTOS.DS_SUB_GRUPO_PROCEDIMENTO
     , PROCEDIMENTO_SUS.CD_PROCEDIMENTO
     , PROCEDIMENTO_SUS.DS_PROCEDIMENTO
     , Nvl(PROCEDIMENTO_SUS.NR_IDADE_MINIMA,0)
     , Nvl(PROCEDIMENTO_SUS.NR_IDADE_MAXIMA,0)
     , Nvl(PROCEDIMENTO_SUS.NR_DIAS_INTERNACAO,0)
     , NVL(PROCEDIMENTO_SUS.SN_ATO_ANESTESICO,'N')
     , PROCEDIMENTO_SUS_COMPLEXIDADE.TP_COMPLEXIDADE_PROCEDIMENTO
     , PROCEDIMENTO_SUS_DETALHE.NR_AUXILIAR
     , PROCEDIMENTO_SUS_DETALHE.SN_URGENCIA
     , PROCEDIMENTO_SUS_DETALHE.SN_PERMITE_PERMANENCIA_MAIOR
