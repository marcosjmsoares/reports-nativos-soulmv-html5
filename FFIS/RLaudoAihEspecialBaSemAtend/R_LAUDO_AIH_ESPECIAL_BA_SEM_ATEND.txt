SELECT   dbamv.pkg_sus_regra_paciente.F_NOME_PACIENTE(PACIENTE.CD_PACIENTE) NM_PACIENTE,
       PACIENTE.DT_NASCIMENTO,
       PACIENTE.TP_SEXO,
       PACIENTE.DS_ENDERECO DS_ENDERECO, 
       PACIENTE.NR_ENDERECO,
       PACIENTE.NM_BAIRRO,
       PACIENTE.ds_complemento,
       PACIENTE.cd_cidadania,
       PAIS.nm_pais PAIS_NASCIMENTO,
       CIDADANIA.NM_CIDADE DS_MUNICIPIO_NASC,
       PACIENTE.cd_naturalidade,
       CIDADANIA.cd_uf UF_NASCIMENTO,
       PACIENTE.NR_CPF,
       PACIENTE.NR_CEP,
       PACIENTE.NR_CNS,
       PROCEDIMENTO_SUS.CD_PROCEDIMENTO     CD_PRO_INT,
       LAUDO_AIH.CD_ESPEC_SUS,
       PRESTADOR.NR_CPF_CGC,
       PRESTADOR.NR_CNS  NR_CNS_SOLIC,
       NULL  NM_RESPONSAVEL,
       SUBSTR( CIDADE.NM_CIDADE, 1, 30) NM_CIDADE,
       CIDADE.CD_UF,
       LAUDO_AIH.DT_EMISSAO,
       LAUDO_AIH.CD_CARATER_ATENDIMENTO,
       LAUDO_AIH.DS_SINAIS_SINT_CLINICOS,
       LAUDO_AIH.DS_COND_JUST_INTERNACAO,
       LAUDO_AIH.DS_RESULT_PROV_DIAG,
       LAUDO_AIH.DS_DIAGNOSTICO ,
       LAUDO_AIH.CD_CID_PRINCIPAL,
       LAUDO_AIH.CD_CID_SECUNDARIO,
       PROCEDIMENTO_SUS.DS_PROCEDIMENTO,
       DBAMV.FN_IDADE (PACIENTE.DT_NASCIMENTO) IDADE,
       MULTI_EMPRESAS.NR_CNES,
       MULTI_EMPRESAS.NR_CNES NR_CNES_EXE,
       PACIENTE.CD_PACIENTE,
       CIDADE.CD_IBGE,
       PACIENTE.NR_FONE,
       PRESTADOR.NM_PRESTADOR,
       LAUDO_AIH.CD_LAUDO,
       DECODE (paciente.tp_cor,
               'B', '01 - Branca',
               'P', '02 - Preta',
               'R', '03 - Parda',
               'A', '04 - Amarela',
               'I', '05 - Indigena',
               '99 - Sem Informao') cor_raca
  FROM DBAMV.PACIENTE
      ,DBAMV.CIDADE
      ,DBAMV.PROCEDIMENTO_SUS
      ,DBAMV.PRESTADOR
      ,DBAMV.LAUDO_AIH
      ,DBAMV.MULTI_EMPRESAS
      ,DBAMV.CIDADE CIDADANIA
      ,DBAMV.PAIS
 WHERE LAUDO_AIH.CD_PACIENTE = :CAMPO_PCD_PACIENTE
   AND PACIENTE.CD_PACIENTE             = LAUDO_AIH.CD_PACIENTE
   AND LAUDO_AIH.CD_PROCEDIMENTO         = PROCEDIMENTO_SUS.CD_PROCEDIMENTO (+)
   AND PACIENTE.CD_CIDADE                = CIDADE.CD_CIDADE(+)
   AND LAUDO_AIH.CD_PRESTADOR            = PRESTADOR.CD_PRESTADOR(+)
   AND LAUDO_AIH.DT_EMISSAO(+)           = :CAMPO_PDT_EMISSAO
   AND MULTI_EMPRESAS.CD_MULTI_EMPRESA   = DBAMV.PKG_MV2000.LE_EMPRESA
   AND CIDADANIA.CD_CIDADE (+)           = PACIENTE.cd_naturalidade
   AND PAIS.cd_pais(+)                   = PACIENTE.cd_cidadania
   AND LAUDO_AIH.CD_LAUDO                = :CAMPO_PCD_LAUDO
