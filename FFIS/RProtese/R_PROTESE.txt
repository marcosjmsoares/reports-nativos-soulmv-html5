--R_PROTESE
SELECT P_DETAL.CD_CODIGO_REDUZIDO
     , P_SUS.CD_PROCEDIMENTO
     , P_SUS.QT_MAXIMA
     , P_SUS.DS_PROCEDIMENTO
     , DBAMV.PACK_SUS.FNC_SUS_MAX_VALOR_SH(P_SUS.CD_PROCEDIMENTO,SYSDATE) VL_TOTAL
  FROM DBAMV.PROCEDIMENTO_SUS P_SUS
     , DBAMV.PROCEDIMENTO_SUS_DETALHE P_DETAL
 WHERE P_SUS.CD_PROCEDIMENTO = P_DETAL.CD_PROCEDIMENTO
   AND P_DETAL.SN_ORTESE_PROTESE = 'S'
   AND P_DETAL.SN_AIH_ESPECIAL = 'S'
   AND P_SUS.SN_ATIVO = 'S'
 ORDER BY P_SUS.DS_PROCEDIMENTO
