--R_FAT_PROC_SETOR_321_FC_SETOR_ANA1_Q_PROC_ANAL_REAL
SELECT * FROM (
SELECT decode (SETOR.CD_SETOR, null, 0, SETOR.CD_SETOR )  CD_SET_PROC
     , decode (SETOR.NM_SETOR, null, 'SETOR NÃO ENCONTRADO',SETOR.NM_SETOR)  NM_SET_PROC
     , PROCEDIMENTO_SUS.CD_PROCEDIMENTO
     , PROCEDIMENTO_SUS.DS_PROCEDIMENTO
     , REG_FAT.CD_ATENDIMENTO
     , PACIENTE.CD_PACIENTE
     , PACIENTE.NM_PACIENTE
     , SUM(NVL( ITREG_FAT.QT_LANCAMENTO,0))  QTD
     , nvl(PROCEDIMENTO_SUS_VALOR.VL_SERVICO_HOSPITALAR,0)    VL_SH
     , nvl(PROCEDIMENTO_SUS_VALOR.VL_SERVICO_PROFISSIONAL,0)  VL_SP
     , 0                                                      VL_OPM 
  From DBAMV.ITREG_FAT
     , DBAMV.PROCEDIMENTO_SUS
     , DBAMV.PACIENTE
     , DBAMV.REG_FAT
     , DBAMV.ATENDIME
     , DBAMV.REMESSA_FATURA
     , DBAMV.FATURA
     , DBAMV.SETOR
     , DBAMV.PROCEDIMENTO_SUS_VALOR   
     , DBAMV.PROCEDIMENTO_SUS_DETALHE
     , ( SELECT CD_PROCEDIMENTO
              , max(DT_VIGENCIA) MAIOR
           FROM DBAMV.PROCEDIMENTO_SUS_VALOR
          WHERE TRUNC(DT_VIGENCIA,'MM') <= :CAMPO_COMPET
          GROUP BY CD_PROCEDIMENTO ) atual 
 WHERE ITREG_FAT.SN_PERTENCE_PACOTE              = 'N'
   AND ITREG_FAT.CD_PROCEDIMENTO                 = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
   AND ITREG_FAT.CD_REG_FAT                      = REG_FAT.CD_REG_FAT
   AND REG_FAT.CD_ATENDIMENTO                    = ATENDIME.CD_ATENDIMENTO
   AND ATENDIME.CD_PACIENTE                      = PACIENTE.CD_PACIENTE
   AND REG_FAT.CD_REMESSA                        = REMESSA_FATURA.CD_REMESSA
   AND REMESSA_FATURA.CD_FATURA                  = FATURA.CD_FATURA
   AND TO_CHAR(FATURA.DT_COMPETENCIA, 'MM/YYYY') = TO_CHAR(:CAMPO_COMPET, 'MM/YYYY')
   AND ITREG_FAT.CD_PROCEDIMENTO                  IS NOT NULL
   AND REG_FAT.CD_MULTI_EMPRESA                  = :CAMPO_CD_MULTI_EMPRESA
   AND FATURA.CD_MULTI_EMPRESA                   = :CAMPO_CD_MULTI_EMPRESA
   AND REG_FAT.CD_CONVENIO                       = Dbamv.Pkt_CONFIG_FFIS.Retorna_Campo('CD_CONVENIO')
   AND FATURA.CD_CONVENIO                        = Dbamv.Pkt_CONFIG_FFIS.Retorna_Campo('CD_CONVENIO')
   AND ITREG_FAT.SN_PROCED_REALIZADO             = 'S'
   AND PROCEDIMENTO_SUS_VALOR.CD_PROCEDIMENTO = atual.CD_PROCEDIMENTO                    
   AND PROCEDIMENTO_SUS_VALOR.CD_PROCEDIMENTO = ITREG_FAT.CD_PROCEDIMENTO        
   AND PROCEDIMENTO_SUS_VALOR.DT_VIGENCIA = atual.MAIOR                               
   AND PROCEDIMENTO_SUS_DETALHE.CD_PROCEDIMENTO = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
GROUP BY SETOR.CD_SETOR
       , SETOR.NM_SETOR
       , PROCEDIMENTO_SUS.CD_PROCEDIMENTO
       , PROCEDIMENTO_SUS.DS_PROCEDIMENTO
       , REG_FAT.CD_ATENDIMENTO
       , PACIENTE.CD_PACIENTE
       , PACIENTE.NM_PACIENTE
       , PROCEDIMENTO_SUS_VALOR.vl_SERVICO_HOSPITALAR
       , PROCEDIMENTO_SUS_VALOR.VL_SERVICO_PROFISSIONAL
:CAMPO_FC_ORDEM
)
ORDER BY CD_SET_PROC,NM_SET_PROC,
CD_PROCEDIMENTO,DS_PROCEDIMENTO
