--R_GLOSAS_SUS
SELECT * FROM (
select justifica_glosa_sus.cd_justifica_glosa_sus,
       justifica_glosa_sus.ds_justifica_glosa_sus,
       reg_fat.nr_guia,
       reg_fat.cd_atendimento,
       paciente.nm_paciente,
       glosas_sus.cd_motivo_glosa_sus,
       nvl(glosas_sus.vl_glosa, 0) vl_glosa ,
       nvl(glosas_sus.vl_reapresentado, 0) vl_reapresentado,
       reg_fat.cd_espec_sus,
       espec_sus.ds_espec_sus,
       prestador.cd_prestador,
       prestador.nm_prestador
  from DBAMV.glosas_sus,
       DBAMV.reg_fat,
       DBAMV.atendime,
       DBAMV.paciente,
       DBAMV.justifica_glosa_sus,
       DBAMV.remessa_fatura,
       DBAMV.fatura,
       DBAMV.prestador,
       DBAMV.espec_sus
 where glosas_sus.cd_reg_fat                             =  reg_fat.cd_reg_fat
       AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
       and espec_sus.cd_espec_sus(+)                     =  reg_fat.cd_espec_sus
       and atendime.cd_atendimento                     =  reg_fat.cd_atendimento
       and paciente.cd_paciente                           =  atendime.cd_paciente
       and justifica_glosa_sus.cd_justifica_glosa_sus = glosas_sus.cd_justifica_glosa_sus
       and remessa_fatura.cd_remessa                  = reg_fat.cd_remessa
       and fatura.cd_fatura                                     = remessa_fatura.cd_fatura
       and atendime.cd_prestador                          = prestador.cd_prestador
       AND to_char(FATURA.DT_COMPETENCIA, 'MM/YYYY') = TO_CHAR(:CAMPO_COMPET, 'MM/YYYY')
order by justifica_glosa_sus.cd_justifica_glosa_sus, reg_fat.nr_guia
)
ORDER BY DS_JUSTIFICA_GLOSA_SUS,CD_JUSTIFICA_GLOSA_SUS
