--R_DADOS_HOSPITAL
SELECT empr.CD_MULTI_EMPRESA,
       empr.DS_MULTI_EMPRESA,
       empr.DS_ENDERECO,
       cida.NM_CIDADE,
       cida.CD_UF,
       empr.CD_CGC,
       naho.CD_NATUREZA_HOSPITALAR,
       naho.DS_NATUREZA_HOSPITALAR,
       empr.VL_FIDEPS,
       empr.VL_IVHE,
       empr.VL_URGENCIA,
       empr.VL_FIXO_FIDEPS,
       empr.VL_FIXO_IAPI,
       pres.NR_CPF_CGC,
       empr.DS_OBSERVACAO,
       empr.CD_ORGAO_LOCAL,
       empr.NR_CEP,
       empr.nr_cgc_mantenedora,
       empr.nr_cnes
  FROM DBAMV.CIDADE cida,
       DBAMV.MULTI_EMPRESAS empr,
       DBAMV.NATUREZA_HOSPITALAR naho,
       DBAMV.PRESTADOR pres
 WHERE  empr.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
   AND empr.CD_CIDADE                = cida.CD_CIDADE
   AND empr.CD_NATUREZA_HOSPITALAR   = naho.CD_NATUREZA_HOSPITALAR
   AND empr.CD_PRESTADOR_DIR_CLINICO = pres.CD_PRESTADOR
