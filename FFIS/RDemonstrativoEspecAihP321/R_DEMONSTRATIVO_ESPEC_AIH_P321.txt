SELECT ESPEC_SUS.CD_ESPEC_SUS CD_ESPEC_SUS
     , ESPEC_SUS.DS_ESPEC_SUS DS_ESPEC_SUS
     , REG_FAT.NR_GUIA NR_GUIA
     , REG_FAT.NR_IDENTIFICACAO NR_IDENTIFICACAO
     , PROCEDIMENTO_SUS.CD_PROCEDIMENTO CD_PROCEDIMENTO
     , NVL(REG_FAT.NR_UTI_INICIAL , 0)+
       NVL(REG_FAT.NR_UTI_ANTERIOR, 0)+
       NVL(REG_FAT.NR_UTI_ALTA, 0) UTI
     , ATENDIME.CD_PACIENTE CD_PACIENTE
     , REG_FAT.DT_FINAL DT_FINAL
     , REG_FAT.CD_REG_FAT CD_REG_FAT
	 , SUM (NVL (V.VLR_ORTESE_PROTESE, 0)) VLR_ORTESE_PROTESE  
     , SUM (NVL (V.VLR_UTI_ESPEC, 0)) VLR_UTI_ESPEC
     , SUM (NVL (V.VLR_COMPONENTE_I, 0)) VLR_COMPONENTE_I  
     , SUM (NVL (V.VLR_ACOMPANHANTE, 0)) VLR_ACOMPANHANTE
     , SUM (NVL (V.VLR_NEONATOLOGIA, 0)) VLR_NEONATOLOGIA 
     , SUM (NVL (V.VLR_REGISTRO_CIVIL, 0)) VLR_REGISTRO_CIVIL
     , SUM (NVL (V.VLR_NUTRICAO_ENTERAL, 0)) VLR_NUTRICAO_ENTERAL  
     , SUM (NVL (V.VLR_TESTE_HIV, 0)) VLR_TESTE_HIV
     , SUM (NVL (V.VLR_UTI_NORMAL, 0)) VLR_UTI_NORMAL  
     , SUM (NVL (V.VLR_SANGUE, 0)) VLR_SANGUE
     , SUM (NVL (V.VLR_NOTIFICACAO_AGRAVO, 0)) VLR_NOTIFICACAO_AGRAVO  
     , SUM (NVL (V.VLR_CATETERISMO, 0)) VLR_CATETERISMO   
  FROM DBAMV.REG_FAT
     , DBAMV.FATURA
     , DBAMV.REMESSA_FATURA
     , DBAMV.ESPEC_SUS
     , DBAMV.PROCEDIMENTO_SUS
     , DBAMV.ATENDIME
     , DBAMV.V_FFIS_FAT_GERENCIAL_TOTAL V
 WHERE REG_FAT.CD_REMESSA                = REMESSA_FATURA.CD_REMESSA
   AND REG_FAT.Cd_Multi_Empresa          = :CAMPO_CD_MULTI_EMPRESA
   AND REMESSA_FATURA.CD_FATURA          = FATURA.CD_FATURA
   AND REG_FAT.CD_ESPEC_SUS              = ESPEC_SUS.CD_ESPEC_SUS
   AND REG_FAT.CD_PROCEDIMENTO_REALIZADO = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
   AND REG_FAT.CD_ATENDIMENTO            = ATENDIME.CD_ATENDIMENTO
   AND FATURA.CD_CONVENIO                = dbamv.pkt_CONFIG_FFIS.retorna_campo('CD_CONVENIO')
   AND to_char(FATURA.DT_COMPETENCIA, 'MM/YYYY') = to_char(:CAMPO_COMPET, 'MM/YYYY')
   AND V.CD_REG_FAT                      = REG_FAT.CD_REG_FAT
   AND V.CD_ESPEC_SUS                    = REG_FAT.CD_ESPEC_SUS
  GROUP BY  ESPEC_SUS.CD_ESPEC_SUS 
     , ESPEC_SUS.DS_ESPEC_SUS 
     , REG_FAT.NR_GUIA 
     , REG_FAT.NR_IDENTIFICACAO 
     , PROCEDIMENTO_SUS.CD_PROCEDIMENTO 
     , NVL(REG_FAT.NR_UTI_INICIAL , 0)+
       NVL(REG_FAT.NR_UTI_ANTERIOR, 0)+
       NVL(REG_FAT.NR_UTI_ALTA, 0) 
     , ATENDIME.CD_PACIENTE 
     , REG_FAT.DT_FINAL 
     , REG_FAT.CD_REG_FAT 
 ORDER BY ESPEC_SUS.DS_ESPEC_SUS,REG_FAT.NR_GUIA
