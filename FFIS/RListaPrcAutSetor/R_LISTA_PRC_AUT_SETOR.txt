--R_LISTA_PRC_AUT_SETOR
SELECT CD_SETOR,
       DBAMV.PKG_SUS_REGRA_SETOR.F_DESCRICAO(CD_SETOR) NM_SETOR,
       CD_PROCEDIMENTO,
       DBAMV.PKG_SUS_REGRA_PROCEDIMENTO.F_DESCRICAO(CD_PROCEDIMENTO) DS_PROCEDIMENTO,
       TP_INSTRUMENTO,
       QT_LANCAMENTO,
       DECODE(TP_DIA_MES,'D','PACIENTE/DIA','PACIENTE/MES') TP_DIA_MES,
       DBAMV.PKG_SUS_PROCED_AUTORIZADO.F_TRADUZ_TIPO_INSTRUMENTO(tp_instrumento,cd_procedimento,TRUNC(DT_VALIDADE_INICIAL,'MM')) TP_PROCEDIMENTO
  FROM DBAMV.SETOR_PROCEDIMENTO_FATRMTO_SUS
 WHERE TP_INSTRUMENTO = :CAMPO_INSTRUMENTO
AND TO_DATE(:CAMPO_COMPETENCIA,'MM/YYYY') BETWEEN TRUNC(DT_VALIDADE_INICIAL,'MM') AND NVL(TRUNC(DT_VALIDADE_FINAL,'MM'),TO_DATE(:CAMPO_COMPETENCIA,'MM/YYYY'))
   AND CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
ORDER BY CD_SETOR, CD_PROCEDIMENTO
