--R_TAB_EXAMES_P321
SELECT DBAMV.SUS_DETALHE.cd_codigo_reduzido              CD_SUS_REDUZIDO
     , DBAMV.PROCEDIMENTO_SUS.cd_procedimento            CD_SUS
     , DBAMV.PROCEDIMENTO_SUS.ds_procedimento            DS_SUS
     , val_pro.QT_PONTOS                                 QT_PONTOS
  FROM DBAMV.PROCEDIMENTO_SUS         PROCEDIMENTO_SUS
     , DBAMV.PROCEDIMENTO_SUS_VALOR   VAL_PRO
     , DBAMV.PROCEDIMENTO_SUS_DETALHE SUS_DETALHE
 WHERE val_pro.cd_procedimento          = PROCEDIMENTO_SUS.cd_procedimento
   AND PROCEDIMENTO_SUS.cd_procedimento = SUS_DETALHE.cd_procedimento
   AND PROCEDIMENTO_SUS.SN_ATIVO        = 'S'
   and val_pro.dt_vigencia = (select max(val_pro2.dt_vigencia)
                                                 from dbamv.PROCEDIMENTO_SUS_VALOR val_pro2
                                               where val_pro2.cd_procedimento = PROCEDIMENTO_SUS.cd_procedimento)
