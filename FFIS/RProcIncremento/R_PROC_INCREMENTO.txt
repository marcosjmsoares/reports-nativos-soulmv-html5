--R_PROC_INCREMENTO
SELECT DESCRICAO              DS_GRUPO
     , IT.CD_PROCEDIMENTO     CD_PROCEDIMENTO
     , PS.DS_PROCEDIMENTO     DS_PROCEDIMENTO                                               
     , SUM(VW.VL_SH_PROPRIO + VW.VL_SH_TERCEIRO) vl_sh_total
     , SUM(VW.VL_SP_PROPRIO + VW.VL_SP_TERCEIRO) vl_sp_total
     , sum( (Dbamv.Pkg_FFIS_Calc_AIH.fnc_sus_max_valor_sh(it.cd_reg_fat,it.cd_lancamento)
     + Dbamv.Pkg_FFIS_Calc_AIH.fnc_sus_max_valor_sp(it.cd_reg_fat,it.cd_lancamento)) * VW.QT_LANCAMENTO ) vl_procedimento
     , SUM(VW.QT_LANCAMENTO)  QT_LANCAMENTO
     , SUM(PI.VL_SH_FEDERAL)  VL_SH_FEDERAL
     , SUM(PI.VL_SH_LOCAL)    VL_SH_LOCAL
     , SUM(PI.VL_SP_FEDERAL)  VL_SP_FEDERAL
     , SUM(PI.VL_SP_LOCAL)    VL_SP_LOCAL
     , SUM (Nvl(PI.VL_SH_FEDERAL,0) + Nvl(PI.VL_SH_LOCAL,0) + Nvl(PI.VL_SP_FEDERAL,0) + Nvl(PI.VL_SP_LOCAL,0) + Nvl(VW.VL_SH_PROPRIO,0) + Nvl(VW.VL_SH_TERCEIRO,0) + Nvl(VW.VL_SP_PROPRIO,0) + Nvl(VW.VL_SP_TERCEIRO,0)) VL_TOTAL
 FROM DBAMV.V_FFIS_PRODUCAO_AIH VW
    , DBAMV.REMESSA_FATURA RF
    , DBAMV.FATURA FT
    , DBAMV.REG_FAT RG
    , DBAMV.ITREG_FAT IT
    , DBAMV.PROCEDIMENTO_SUS PS
    , dbamv.procedimento_detalhe_vigencia pd
    , dbamv.PROCEDIMENTO_SUS_REGRA_INCRE  PI
WHERE VW.CD_REMESSA      = RF.CD_REMESSA
  AND RF.CD_FATURA       = FT.CD_FATURA
  AND PD.CD_PROCEDIMENTO = PS.CD_PROCEDIMENTO
  AND RG.CD_REG_FAT      = VW.CD_REG_FAT
  AND IT.CD_REG_FAT      = RG.CD_REG_FAT
  AND IT.CD_LANCAMENTO   = VW.CD_LANCAMENTO
  AND PS.CD_PROCEDIMENTO = IT.CD_PROCEDIMENTO
  AND SUBSTR(VW.DESCRICAO,1,5) = 'GRUPO'
  AND TRUNC(IT.DT_LANCAMENTO, 'MM')  BETWEEN PD.DT_VALIDADE_INICIAL  AND  NVL(PD.DT_VALIDADE_FINAL, TRUNC(IT.DT_LANCAMENTO, 'MM'))
  AND FT.CD_FATURA = :CAMPO_CD_FATURA
  AND PD.CD_PROCEDIMENTO = PI.CD_PROCEDIMENTO(+)
  AND TRUNC(IT.DT_LANCAMENTO, 'MM')  BETWEEN PI.DT_VALIDADE_INICIAL(+)  AND  NVL(PI.DT_VALIDADE_FINAL(+), TRUNC(IT.DT_LANCAMENTO, 'MM'))
GROUP BY DESCRICAO
       , IT.CD_PROCEDIMENTO
       , PS.DS_PROCEDIMENTO
 ORDER BY cd_procedimento
