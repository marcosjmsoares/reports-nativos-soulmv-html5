--R_FAT_PROCEDIMENTO_P321
SELECT  PROCEDIMENTO_SUS.CD_PROCEDIMENTO
     , PROCEDIMENTO_SUS.DS_PROCEDIMENTO
     , SUM(NVL( ITREG_FAT.QT_LANCAMENTO,0))  QTD
     , ITREG_FAT.CD_PROCEDIMENTO
     , nvl(decode(substr(PROCEDIMENTO_SUS.CD_PROCEDIMENTO,1,2),'07',0,PROCEDIMENTO_SUS_VALOR.vl_servico_hospitalar),0)  VL_SH
     , nvl(PROCEDIMENTO_SUS_VALOR.vl_servico_profissional,0)  VL_SP
     , nvl(decode(substr(procedimento_sus.cd_procedimento,1,2),'07',procedimento_sus_valor.vl_servico_hospitalar,0),0)   VL_OPM
     , nvl(nvl(procedimento_sus_valor.vl_servico_hospitalar,0) +
       nvl(procedimento_sus_valor.vl_servico_profissional,0),0)  TOT_UNIT
  From DBAMV.ITREG_FAT
     , DBAMV.PROCEDIMENTO_SUS
     , DBAMV.PACIENTE
     , DBAMV.REG_FAT
     , DBAMV.ATENDIME
     , DBAMV.REMESSA_FATURA
     , DBAMV.FATURA
     , DBAMV.SETOR
     , DBAMV.PROCEDIMENTO_SUS_VALOR
     , DBAMV.PROCEDIMENTO_SUS_DETALHE
     , ( SELECT CD_PROCEDIMENTO
              , max(DT_VIGENCIA) MAIOR
           FROM DBAMV.PROCEDIMENTO_SUS_VALOR
          WHERE TRUNC(DT_VIGENCIA,'MM') <= :CAMPO_COMPET
          GROUP BY CD_PROCEDIMENTO ) atual
 WHERE ITREG_FAT.SN_PERTENCE_PACOTE              = 'N'
   AND ITREG_FAT.CD_PROCEDIMENTO                 = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
   AND ITREG_FAT.CD_REG_FAT                      = REG_FAT.CD_REG_FAT
   AND REG_FAT.CD_ATENDIMENTO                    = ATENDIME.CD_ATENDIMENTO
   AND ATENDIME.CD_PACIENTE                      = PACIENTE.CD_PACIENTE
   AND REG_FAT.CD_REMESSA                        = REMESSA_FATURA.CD_REMESSA
   AND REMESSA_FATURA.CD_FATURA                  = FATURA.CD_FATURA
   AND TO_CHAR(FATURA.DT_COMPETENCIA, 'MM/YYYY') = TO_CHAR(:CAMPO_COMPET, 'MM/YYYY')
   AND ITREG_FAT.CD_PROCEDIMENTO                 IS NOT NULL
   AND REG_FAT.CD_MULTI_EMPRESA                  = :CAMPO_CD_MULTI_EMPRESA
   AND FATURA.CD_MULTI_EMPRESA                   = :CAMPO_CD_MULTI_EMPRESA
   AND REG_FAT.CD_CONVENIO                       = Dbamv.Pkt_CONFIG_FFIS.Retorna_Campo('CD_CONVENIO')
   AND FATURA.CD_CONVENIO                        = Dbamv.Pkt_CONFIG_FFIS.Retorna_Campo('CD_CONVENIO')
   AND (NVL(REG_FAT.CD_ESPEC_SUS,0) IN ('04','05','09')
    OR REG_FAT.cd_espec_sus is null   )
   AND PROCEDIMENTO_SUS_VALOR.CD_PROCEDIMENTO = atual.CD_PROCEDIMENTO
   AND PROCEDIMENTO_SUS_VALOR.CD_PROCEDIMENTO = ITREG_FAT.CD_PROCEDIMENTO
   AND PROCEDIMENTO_SUS_VALOR.DT_VIGENCIA = atual.MAIOR
   AND PROCEDIMENTO_SUS.CD_PROCEDIMENTO = PROCEDIMENTO_SUS_DETALHE.CD_PROCEDIMENTO
GROUP BY
        PROCEDIMENTO_SUS.CD_PROCEDIMENTO
       , PROCEDIMENTO_SUS.DS_PROCEDIMENTO
       , ITREG_FAT.CD_PROCEDIMENTO
       , PROCEDIMENTO_SUS_VALOR.vl_servico_hospitalar
       , PROCEDIMENTO_SUS_VALOR.vl_servico_profissional
UNION ALL

--R_FAT_PROCEDIMENTO_P321_RF_CONTAS1_Q_CONTAS
SELECT itreg_fat.CD_PROCEDIMENTO CON_CD_PROCEDIMENTO
     , procedimento_sus.DS_PROCEDIMENTO   CON_DS_PROCEDIMENTO
     , sum(ITREG_FAT.qt_lancamento) CON_QTD
     , nvl(procedimento_sus_valor.vl_servico_hospitalar ,0) CON_SH
     , nvl(procedimento_sus_valor.VL_servico_profissional,0) CON_SP
     , 0 CON_OPM
     , itreg_fat.sn_proced_realizado
     , nvl(PROCEDIMENTO_SUS_VALOR.vl_servico_hospitalar,0) +
       nvl(PROCEDIMENTO_SUS_VALOR.VL_servico_profissional,0)  CON_TOTAL
  FROM DBAMV.REG_FAT
     , DBAMV.ITREG_FAT
     , DBAMV.REMESSA_FATURA
     , DBAMV.FATURA
     , DBAMV.PROCEDIMENTO_SUS
     , DBAMV.PROCEDIMENTO_SUS_VALOR
     , DBAMV.PROCEDIMENTO_SUS_DETALHE
     , ( SELECT CD_PROCEDIMENTO
              , max(DT_VIGENCIA) MAIOR
           FROM DBAMV.PROCEDIMENTO_SUS_VALOR
          WHERE TRUNC(DT_VIGENCIA,'MM') <= :CAMPO_COMPET
          GROUP BY CD_PROCEDIMENTO ) atual
  WHERE ITREG_FAT.CD_REG_FAT                 = REG_FAT.CD_REG_FAT
    AND ITREG_FAT.CD_PROCEDIMENTO            = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
    AND REG_FAT.CD_REMESSA                   = DBAMV.REMESSA_FATURA.CD_REMESSA
    AND REMESSA_FATURA.CD_FATURA             = DBAMV.FATURA.CD_FATURA
    AND REG_FAT.CD_MULTI_EMPRESA             = :CAMPO_CD_MULTI_EMPRESA
    AND FATURA.CD_MULTI_EMPRESA              = :CAMPO_CD_MULTI_EMPRESA
    AND REG_FAT.CD_CONVENIO                  = Dbamv.Pkt_CONFIG_FFIS.Retorna_Campo('CD_CONVENIO')
    AND FATURA.CD_CONVENIO                   = Dbamv.Pkt_CONFIG_FFIS.Retorna_Campo('CD_CONVENIO')
    AND to_char(FATURA.DT_COMPETENCIA, 'MM/YYYY') = to_char(:CAMPO_COMPET, 'MM/YYYY')
    AND ITREG_FAT.SN_PROCED_REALIZADO             = 'S'
    AND PROCEDIMENTO_SUS_VALOR.CD_PROCEDIMENTO = atual.CD_PROCEDIMENTO
    AND PROCEDIMENTO_SUS_VALOR.CD_PROCEDIMENTO = ITREG_FAT.CD_PROCEDIMENTO
    AND PROCEDIMENTO_SUS_VALOR.DT_VIGENCIA = atual.MAIOR
    AND (NVL(REG_FAT.CD_ESPEC_SUS,0) NOT IN ('04','05','09')
    OR REG_fat.cd_espec_sus is not null )
    AND PROCEDIMENTO_SUS_DETALHE.CD_PROCEDIMENTO = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
Group by itreg_fat.cd_procedimento
       , procedimento_sus.ds_procedimento
       , nvl(PROCEDIMENTO_SUS_VALOR.VL_servico_hospitalar,0)
       , nvl(PROCEDIMENTO_SUS_VALOR.VL_servico_profissional,0)
       , 0
       , itreg_fat.sn_proced_realizado
union all
SELECT itreg_fat.CD_PROCEDIMENTO     CON_CD_PROCEDIMENTO
  , PROCEDIMENTO_SUS.DS_PROCEDIMENTO CON_DS_PROCEDIMENTO
  , sum(ITREG_FAT.qt_lancamento) CON_QTD
  , nvl(PROCEDIMENTO_SUS_VALOR.VL_SERVICO_HOSPITALAR,0) CON_SH
  , nvl(PROCEDIMENTO_SUS_VALOR.VL_SERVICO_PROFISSIONAL,0) CON_SP
  , 0 CON_OPM
  , itreg_fat.sn_proced_realizado
  , nvl(PROCEDIMENTO_SUS_VALOR.VL_SERVICO_HOSPITALAR,0) +
    nvl(PROCEDIMENTO_SUS_VALOR.VL_SERVICO_PROFISSIONAL,0) CON_TOTAL
FROM DBAMV.REG_FAT
  , DBAMV.ITREG_FAT
  , DBAMV.REMESSA_FATURA
  , DBAMV.FATURA
  , DBAMV.PROCEDIMENTO_SUS
  , DBAMV.CONFIG_FFIS
  , DBAMV.PROCEDIMENTO_SUS_VALOR
  , DBAMV.PROCEDIMENTO_SUS_DETALHE
  , ( SELECT CD_PROCEDIMENTO
              , max(DT_VIGENCIA) MAIOR
           FROM DBAMV.PROCEDIMENTO_SUS_VALOR
          WHERE TRUNC(DT_VIGENCIA,'MM') <= :CAMPO_COMPET
          GROUP BY CD_PROCEDIMENTO ) atual
WHERE ITREG_FAT.CD_REG_FAT                 = REG_FAT.CD_REG_FAT
AND ITREG_FAT.CD_PROCEDIMENTO                 = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
AND REG_FAT.CD_REMESSA                   = DBAMV.REMESSA_FATURA.CD_REMESSA
AND REMESSA_FATURA.CD_FATURA             = DBAMV.FATURA.CD_FATURA
AND REG_FAT.CD_MULTI_EMPRESA             = :CAMPO_CD_MULTI_EMPRESA
AND FATURA.CD_MULTI_EMPRESA              = :CAMPO_CD_MULTI_EMPRESA
AND REG_FAT.CD_CONVENIO                  = CONFIG_FFIS.CD_CONVENIO
AND FATURA.CD_CONVENIO                   = CONFIG_FFIS.CD_CONVENIO
AND to_char(FATURA.DT_COMPETENCIA, 'MM/YYYY') = to_char(:CAMPO_COMPET, 'MM/YYYY')
AND ITREG_FAT.SN_PROCED_REALIZADO             = 'S'
AND PROCEDIMENTO_SUS_VALOR.CD_PROCEDIMENTO = atual.CD_PROCEDIMENTO
AND PROCEDIMENTO_SUS_VALOR.CD_PROCEDIMENTO = ITREG_FAT.CD_PROCEDIMENTO
AND PROCEDIMENTO_SUS_VALOR.DT_VIGENCIA = atual.MAIOR
AND  NVL(REG_FAT.CD_ESPEC_SUS,0) IN ('04','05','09')
 and reg_fat.cd_espec_sus is null
AND PROCEDIMENTO_SUS_DETALHE.CD_PROCEDIMENTO = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
Group by itreg_fat.CD_PROCEDIMENTO
     , PROCEDIMENTO_SUS.DS_PROCEDIMENTO
     , nvl(PROCEDIMENTO_SUS_VALOR.VL_SERVICO_HOSPITALAR,0)
     , nvl(PROCEDIMENTO_SUS_VALOR.VL_SERVICO_PROFISSIONAL,0)
     , 0
     , itreg_fat.sn_proced_realizado

   AND REG_FAT.CD_REMESSA                        = REMESSA_FATURA.CD_REMESSA
   AND REMESSA_FATURA.CD_FATURA                  = FATURA.CD_FATURA
   AND TO_CHAR(FATURA.DT_COMPETENCIA, 'DD/MM/YYYY') = :CAMPO_COMPET
   AND ITREG_FAT.CD_PROCEDIMENTO                 IS NOT NULL
   AND REG_FAT.CD_MULTI_EMPRESA                  = :CAMPO_CD_MULTI_EMPRESA
   AND FATURA.CD_MULTI_EMPRESA                   = :CAMPO_CD_MULTI_EMPRESA
   AND REG_FAT.CD_CONVENIO                       = Dbamv.Pkt_CONFIG_FFIS.Retorna_Campo('CD_CONVENIO')
   AND FATURA.CD_CONVENIO                        = Dbamv.Pkt_CONFIG_FFIS.Retorna_Campo('CD_CONVENIO')
   AND ITREG_FAT.SN_PROCED_REALIZADO             = 'S'
   AND PROCEDIMENTO_SUS_VALOR.CD_PROCEDIMENTO = atual.CD_PROCEDIMENTO
   AND PROCEDIMENTO_SUS_VALOR.CD_PROCEDIMENTO = ITREG_FAT.CD_PROCEDIMENTO
   AND PROCEDIMENTO_SUS_VALOR.DT_VIGENCIA = atual.MAIOR
   AND (NVL(REG_FAT.CD_ESPEC_SUS,0) NOT IN ('04','05','09')
   OR reg_fat.cd_espec_sus is not null )
   AND PROCEDIMENTO_SUS_DETALHE.CD_PROCEDIMENTO = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
GROUP BY PROCEDIMENTO_SUS.CD_PROCEDIMENTO
       , PROCEDIMENTO_SUS.DS_PROCEDIMENTO
       , REG_FAT.CD_ATENDIMENTO
       , PACIENTE.CD_PACIENTE
       , PACIENTE.NM_PACIENTE
       , PROCEDIMENTO_SUS_VALOR.VL_SERVICO_HOSPITALAR
       , PROCEDIMENTO_SUS_VALOR.VL_SERVICO_PROFISSIONAL
union all
SELECT PROCEDIMENTO_SUS.CD_PROCEDIMENTO
     , PROCEDIMENTO_SUS.DS_PROCEDIMENTO
     , REG_FAT.CD_ATENDIMENTO
     , PACIENTE.CD_PACIENTE
     , PACIENTE.NM_PACIENTE
     , SUM(NVL( ITREG_FAT.QT_LANCAMENTO,0))  QTD
     , nvl(decode(substr(PROCEDIMENTO_SUS.cd_procedimento,1,2),'07',0,PROCEDIMENTO_SUS_VALOR.VL_SERVICO_HOSPITALAR),0)  VL_SH                                                                  
     , nvl(PROCEDIMENTO_SUS_VALOR.VL_SERVICO_PROFISSIONAL,0)  VL_SP
     , nvl(decode(substr(PROCEDIMENTO_SUS.cd_procedimento,1,2),'07',PROCEDIMENTO_SUS_VALOR.VL_SERVICO_HOSPITALAR,0),0)   VL_OPM
     , nvl(nvl(PROCEDIMENTO_SUS_VALOR.VL_SERVICO_HOSPITALAR,0) +
       nvl(PROCEDIMENTO_SUS_VALOR.VL_SERVICO_PROFISSIONAL,0),0)  TOT_UNIT
  From DBAMV.ITREG_FAT
     , DBAMV.PROCEDIMENTO_SUS
     , DBAMV.PACIENTE
     , DBAMV.REG_FAT
     , DBAMV.ATENDIME
     , DBAMV.REMESSA_FATURA
     , DBAMV.FATURA
     , DBAMV.CONFIG_FFIS
     , DBAMV.SETOR
     , DBAMV.PROCEDIMENTO_SUS_VALOR
     , DBAMV.PROCEDIMENTO_SUS_DETALHE
     , ( SELECT CD_PROCEDIMENTO
              , max(DT_VIGENCIA) MAIOR
           FROM DBAMV.PROCEDIMENTO_SUS_VALOR
          GROUP BY CD_PROCEDIMENTO ) atual
 WHERE ITREG_FAT.SN_PERTENCE_PACOTE              = 'N'
   AND ITREG_FAT.CD_PROCEDIMENTO                      = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
   AND ITREG_FAT.CD_REG_FAT                      = REG_FAT.CD_REG_FAT
   AND REG_FAT.CD_ATENDIMENTO                    = ATENDIME.CD_ATENDIMENTO
   AND ATENDIME.CD_PACIENTE                      = PACIENTE.CD_PACIENTE
   AND REG_FAT.CD_REMESSA                        = REMESSA_FATURA.CD_REMESSA
   AND REMESSA_FATURA.CD_FATURA                  = FATURA.CD_FATURA
   AND TO_CHAR(FATURA.DT_COMPETENCIA, 'DD/MM/YYYY') = :CAMPO_COMPET
   AND ITREG_FAT.CD_PROCEDIMENTO                 IS NOT NULL
   AND REG_FAT.CD_MULTI_EMPRESA                  = :CAMPO_CD_MULTI_EMPRESA
   AND FATURA.CD_MULTI_EMPRESA                   = :CAMPO_CD_MULTI_EMPRESA
   AND REG_FAT.CD_CONVENIO                       = CONFIG_FFIS.CD_CONVENIO
   AND FATURA.CD_CONVENIO                        = CONFIG_FFIS.CD_CONVENIO
   AND ITREG_FAT.SN_PROCED_REALIZADO             = 'S'
   AND PROCEDIMENTO_SUS_VALOR.CD_PROCEDIMENTO = atual.CD_PROCEDIMENTO
   AND PROCEDIMENTO_SUS_VALOR.CD_PROCEDIMENTO = ITREG_FAT.CD_PROCEDIMENTO
   AND PROCEDIMENTO_SUS_VALOR.DT_VIGENCIA = atual.MAIOR
   AND NVL(REG_FAT.CD_ESPEC_SUS,0) IN ('04','05','09')
    AND reg_fat.cd_espec_sus is null
   AND PROCEDIMENTO_SUS_DETALHE.CD_PROCEDIMENTO = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
GROUP BY PROCEDIMENTO_SUS.CD_PROCEDIMENTO
       , PROCEDIMENTO_SUS.DS_PROCEDIMENTO
       , REG_FAT.CD_ATENDIMENTO
       , PACIENTE.CD_PACIENTE
       , PACIENTE.NM_PACIENTE
       , PROCEDIMENTO_SUS_VALOR.VL_SERVICO_HOSPITALAR
       , PROCEDIMENTO_SUS_VALOR.VL_SERVICO_PROFISSIONAL
)
ORDER BY CD_PROCEDIMENTO,DS_PROCEDIMENTO
