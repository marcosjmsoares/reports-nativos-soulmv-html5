--R_LAUDO_AIH_SUS_SEM_ATEND
SELECT         LAUDO_AIH.cd_laudo, LAUDO_AIH.DT_EMISSAO DT_EMISSAO,
               dbamv.pkg_sus_regra_paciente.F_NOME_PACIENTE(PACIENTE.CD_PACIENTE) NM_PACIENTE ,
               PACIENTE.DT_NASCIMENTO,
               PACIENTE.TP_SEXO,
               PACIENTE.DS_ENDERECO || ', ' || PACIENTE.NR_ENDERECO || ' - ' || PACIENTE.NM_BAIRRO DS_ENDERECO ,
               PACIENTE.NR_CPF,
               PACIENTE.NR_CEP,
               PACIENTE.NR_CNS,
               procedimento_sus.cd_procedimento,
               procedimento_suS.ds_procedimento,
               LAUDO_AIH.CD_ESPEC_SUS,
               ESPEC_SUS.DS_ESPEC_SUS,
               PRESTADOR.NR_CPF_CGC,
               PRESTADOR.NR_CNS    NR_CNS_SOLIC,
               PRESTADOR.DS_CODIGO_CONSELHO,
               PACIENTE.NM_MAE NM_RESPONSAVEL,
               PACIENTE.NM_MAE,
               SUBSTR( CIDADE.NM_CIDADE, 1, 30) NM_CIDADE,
               CIDADE.CD_UF,
               LAUDO_AIH.CD_CARATER_ATENDIMENTO,
               LAUDO_AIH.DS_SINAIS_SINT_CLINICOS,
               LAUDO_AIH.DS_COND_JUST_INTERNACAO,
               LAUDO_AIH.DS_RESULT_PROV_DIAG,
               substr(replace(replace(LAUDO_AIH.DS_DIAGNOSTICO, '  ', ' '), CHR(10), ' '), 1, 190) DS_DIAGNOSTICO,
               LAUDO_AIH.CD_CID_PRINCIPAL,
               LAUDO_AIH.CD_CID_SECUNDARIO,
			   LAUDO_AIH.CD_CID_CAUSAS_ASSOCIADAS,
               DBAMV.FN_IDADE (PACIENTE.DT_NASCIMENTO) IDADE,
               MULTI_EMPRESAS.NR_CNES,
               DBAMV.PKG_FFIS_REGRA_LAUDO_AIH.f_pac_same_prontuario('PD',PACIENTE.CD_PACIENTE) CD_PACIENTE,
               cidade.cd_ibge,
               paciente.nr_fone,
               prestador.nm_prestador,
               MULTI_EMPRESAS.CD_ORGAO_EMISSOR_AIH,
               Decode(Paciente.Tp_Cor,'B','01 - Branca','P','02 - Preta','R','03 - Parda','A','04 - Amarela','I','05 - Indigena','99 - Sem Informação') Cor_Raca,
               DECODE(ETNIA.CD_ETNIA,NULL,'0000 - Não Se Aplica',lPad(ETNIA.CD_ETNIA,4,'0')||' - '||ETNIA.DS_ETNIA) DS_ETNIA,
               laudo_aih.cd_estabelecimento,  
               nvl(pkg_mv2000.Le_Configuracao('FFIS', 'SN_IMPRIME_CPF_MEDICO_SOLICITANTE_LAUDO_AIH'), 'N') V_CHAVE_CPF
FROM     DBAMV.PACIENTE
               ,DBAMV.CIDADE
               ,DBAMV.PROCEDIMENTO_SUS
               ,DBAMV.PRESTADOR
               ,DBAMV.LAUDO_AIH
               ,DBAMV.MULTI_EMPRESAS
               ,DBAMV.ESPEC_SUS
               ,DBAMV.ETNIA
WHERE     PACIENTE.CD_ETNIA = ETNIA.CD_ETNIA(+)
AND       LAUDO_AIH.CD_PROCEDIMENTO  =  PROCEDIMENTO_SUS.CD_PROCEDIMENTO(+)
AND       PACIENTE.CD_CIDADE      = CIDADE.CD_CIDADE(+)
AND       LAUDO_AIH.CD_PRESTADOR_SOL   = PRESTADOR.CD_PRESTADOR(+)
AND       MULTI_EMPRESAS.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
AND       LAUDO_AIH.CD_ESPEC_SUS = ESPEC_SUS.CD_ESPEC_SUS(+)
AND       PACIENTE.CD_PACIENTE = :CAMPO_PCD_PACIENTE
AND       LAUDO_AIH.cd_paciente = PACIENTE.CD_PACIENTE
AND       LAUDO_AIH.CD_LAUDO = :CAMPO_PCD_LAUDO
