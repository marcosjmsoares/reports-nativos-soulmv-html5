SELECT SYSDATE
     , atendime.dt_atendimento
     , multi_empresas.ds_multi_empresa
     , multi_empresas.nr_cnes
     , paciente.cd_paciente
     , dbamv.pkg_sus_regra_paciente.F_NOME_PACIENTE(PACIENTE.CD_PACIENTE)  nm_paciente
     , INSTR (dbamv.pkg_sus_regra_paciente.F_NOME_PACIENTE(PACIENTE.CD_PACIENTE), ' ') primeiro_nome
     , paciente.nm_mnemonico
     , paciente.tp_sexo
     , paciente.nr_cns
     , paciente.nr_cpf
     , paciente.nr_rg_nasc
     , paciente.dt_nascimento
     , paciente.nm_mae
     , paciente.ds_endereco
     , paciente.ds_complemento
     , paciente.nr_cep
     , paciente.nm_bairro
     , paciente.nr_fone
     , paciente.nr_endereco
     , NVL (NVL (paciente.nr_cpf, paciente.nr_identidade),paciente.nr_rg_nasc) nr_identidade
     , cidadanias.ds_cidadania
     , prestador.nr_cpf_cgc
     , prestador.nr_cns    NR_CNS_SOLIC
     , paciente.nr_identidade nr_id
     , prestador.ds_codigo_conselho
     , responsa.nm_responsavel
     , SUBSTR (cidade.nm_cidade, 1, 30) nm_cidade
     , cidade.cd_uf
     , laudo_aih.ds_sinais_sint_clinicos
     , laudo_aih.ds_cond_just_internacao
     , laudo_aih.ds_result_prov_diag
     , laudo_aih.cd_procedimento
     , atendime.cd_procedimento cd_procedimento_atend
     , procedimento_sus.ds_procedimento
     , laudo_aih.ds_diagnostico
     , laudo_aih.cd_cid_principal
     , laudo_aih.cd_cid_secundario
     , laudo_aih.cd_espec_sus
     , laudo_aih.dt_emissao
     , cidade.cd_ibge
     , LPAD (laudo_aih.cd_carater_atendimento, 2, '0') cd_carater_atendimento
     , prestador.nm_prestador
     , DECODE (paciente.tp_cor,
               'B', '01 - Branca',
               'P', '02 - Preta',
               'R', '03 - Parda',
               'A', '04 - Amarela',
               'I', '05 - Indigena',
               '99 - Sem Informao') cor_raca
  FROM dbamv.paciente,
       dbamv.cidade,
       dbamv.atendime,
       dbamv.convenio,
       dbamv.cidadanias,
       dbamv.prestador,
       dbamv.responsa,
       dbamv.procedimento_sus,
       dbamv.laudo_aih,
       dbamv.multi_empresas,
       DBAMV.EMPRESA_CONVENIO
 WHERE atendime.cd_multi_empresa    = dbamv.multi_empresas.cd_multi_empresa
   AND atendime.cd_multi_empresa    = dbamv.pkg_mv2000.le_empresa
   AND EMPRESA_CONVENIO.CD_CONVENIO =  CONVENIO.CD_CONVENIO
   AND paciente.cd_cidade           = cidade.cd_cidade(+)
   AND paciente.cd_cidadania        = cidadanias.cd_cidadania(+)
   AND atendime.cd_paciente         = paciente.cd_paciente
   AND atendime.cd_convenio         = convenio.cd_convenio
   AND prestador.cd_prestador(+)       = laudo_aih.cd_prestador_sol
   AND atendime.cd_atendimento      = responsa.cd_atendimento(+)
   AND atendime.cd_atendimento      = :CAMPO_PCD_ATENDIMENTO
   AND laudo_aih.cd_procedimento    = procedimento_sus.cd_procedimento(+)
   AND atendime.cd_atendimento  = laudo_aih.cd_atendimento(+)
   AND laudo_aih.DT_EMISSAO (+)    = :CAMPO_PDT_EMISSAO
   AND EMPRESA_CONVENIO.CD_MULTI_EMPRESA  =  :CAMPO_CD_MULTI_EMPRESA
