--R_ETIQUETA_LAUDO_R_1_Q_LAUDO_ALTERACAO
SELECT laudos_sih.NR_LAUDO_ALT_CUS
     , laudos_sih.NR_LAUDO
     , laudos_sih.CD_GRU_LAUDO
     , reg_fat.NR_GUIA
     , reg_fat.NR_SEQAIH5
     , reg_fat.DT_INICIO
     , reg_fat.DT_FINAL
     , reg_fat.CD_CARATER_INTERNACAO
     , reg_fat.CD_CID_PRINCIPAL
     , multi_empresas.DS_MULTI_EMPRESA
     , multi_empresas.CD_CGC
     , paciente.cd_paciente
     , paciente.nm_paciente
     , paciente.dt_nascimento
     , paciente.cd_cidade
     , paciente.nm_mae
     , paciente.tp_sexo
     , paciente.nr_identidade
     , paciente.ds_endereco||' - '||paciente.nm_bairro||', '||paciente.nr_endereco  endereco
     , paciente.nr_cep
     , pro_fat.CD_SUS cd_sus_solicitado
     , pro_fat.DS_SUS ds_pro_fat_solicitado
     , laudos_sih.DS_JUSTIFICATIVA
     , laudos_sih.DT_LAUDO
     , cidade.NM_CIDADE||'/'||cidade.CD_UF municipio
     , cidade1.NM_CIDADE||'-'||cidade1.CD_UF naturalidade
     , leito.ds_enfermaria
     , atendime.cd_leito
     , atendime.cd_atendimento
     , atendime.dt_atendimento
     , pres_dir_cli.nr_cpf_cgc
  FROM DBAMV.PACIENTE paciente
     , DBAMV.CIDADE cidade
     , dbamv.cidade cidade1
     , DBAMV.REG_FAT reg_fat
     , DBAMV.ATENDIME atendime
     , DBAMV.PRESTADOR pres_dir_cli
     , DBAMV.laudos_sih laudos_sih
     , DBAMV.PRO_FAT pro_fat
     , DBAMV.MULTI_EMPRESAS multi_empresas
     , DBAMV.PRESTADOR prestador
     , dbamv.leito leito
 WHERE laudos_sih.CD_REG_FAT                   = reg_fat.CD_REG_FAT
   AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
    AND laudos_sih.CD_ATENDIMENTO               = atendime.CD_ATENDIMENTO
   AND atendime.CD_MULTI_EMPRESA               = multi_empresas.CD_MULTI_EMPRESA
   AND atendime.CD_PACIENTE                    = paciente.CD_PACIENTE
   AND paciente.CD_CIDADE                      = cidade.CD_CIDADE
   and cidade1.cd_cidade(+) = paciente.cd_naturalidade
   AND laudos_sih.CD_PRO_FAT                   = pro_fat.CD_PRO_FAT
   AND multi_empresas.CD_PRESTADOR_DIR_CLINICO = pres_dir_cli.CD_PRESTADOR
   AND reg_fat.CD_PRESTADOR_RESPONSAVEL        = prestador.CD_PRESTADOR
   AND laudos_sih.TP_LAUDO                     = 'M'
   and atendime.cd_leito                       = leito.cd_leito
   AND laudos_sih.cd_laudos_sih                =  :CAMPO_CD_LAUDOS_SIH
