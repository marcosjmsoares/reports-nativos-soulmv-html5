--R_PREVISAO_GLOBAL_SETOR
SELECT CD_SETOR_PRODUZIU   CD_SETOR_PRODUZIU
     , S.NM_SETOR||' ('||CD_SETOR_PRODUZIU||')' NM_SETOR
     , TP_INSTRUMENTO      TP_INSTRUMENTO
     , DS_INSTRUMENTO      DS_INSTRUMENTO
     , ID                  ID
     , DESCRICAO           DESCRICAO
     , SUM(DIAS)           DIAS
     , SUM(QT_LANCAMENTO)  QT_LANCAMENTO
     , SUM(VL_TOTAL)       VL_TOTAL
     , SUM(VL_SH_PROPRIO)  VL_SH_PROPRIO
     , SUM(VL_SH_TERCEIRO) VL_SH_TERCEIRO
     , SUM(VL_SP_PROPRIO)  VL_SP_PROPRIO
     , SUM(VL_SP_TERCEIRO) VL_SP_TERCEIRO
 FROM DBAMV.V_FFIS_PRODUCAO_AIH VW
    , DBAMV.SETOR S
WHERE S.CD_SETOR = VW.CD_SETOR_PRODUZIU
GROUP BY CD_SETOR_PRODUZIU
       , S.NM_SETOR
       , TP_INSTRUMENTO
       , DS_INSTRUMENTO
       , ID
       , DESCRICAO
ORDER BY NM_SETOR
       , TP_INSTRUMENTO
       , DS_INSTRUMENTO
       , ID
