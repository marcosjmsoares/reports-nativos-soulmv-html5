SELECT TO_CHAR(PACIENTE.CD_PACIENTE)||'-'|| dbamv.pkg_sus_regra_paciente.F_NOME_PACIENTE(atendime.CD_PACIENTE) NM_PACIENTE
     , REG_FAT.CD_ATENDIMENTO
     , REG_FAT.NR_GUIA
     , REG_FAT.DT_INICIO
     , REG_FAT.DT_FINAL
     , PROC.CD_PROCEDIMENTO CD_SUS
     , PACIENTE.NR_CNS
  FROM DBAMV.REG_FAT REG_FAT
     , DBAMV.CONVENIO CONVENIO
     , DBAMV.REMESSA_FATURA REMESSA_FATURA
     , DBAMV.FATURA FATURA
     , DBAMV.ATENDIME ATENDIME
     , DBAMV.PACIENTE PACIENTE
     , DBAMV.PROCEDIMENTO_SUS   PROC
 WHERE REG_FAT.CD_CONVENIO               = CONVENIO.CD_CONVENIO
   AND REG_FAT.CD_REMESSA                = REMESSA_FATURA.CD_REMESSA
   AND REMESSA_FATURA.CD_FATURA          = FATURA.CD_FATURA
   AND REG_FAT.CD_ATENDIMENTO            = ATENDIME.CD_ATENDIMENTO
   AND ATENDIME.CD_PACIENTE              = PACIENTE.CD_PACIENTE
   AND REG_FAT.CD_PROCEDIMENTO_REALIZADO = PROC.CD_PROCEDIMENTO
   AND CONVENIO.TP_CONVENIO              = 'H'
   AND REG_FAT.CD_MULTI_EMPRESA          = :CAMPO_CD_MULTI_EMPRESA
   AND to_char(DBAMV.FATURA.DT_COMPETENCIA, 'MM/YYYY') = to_char(:CAMPO_COMPET, 'MM/YYYY')
   AND Nvl(remessa_fatura.sn_suspensas, 'N') = 'N'
