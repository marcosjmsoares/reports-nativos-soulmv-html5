--R_AIH_AUT_GESTOR_P321_RF_BITMAP1_Q_PENDENTE
SELECT * FROM (
Select temp.CD_GESTOR_AUTORIZOU
     , temp.DS_AUTORIZACAO
     , temp.NR_GUIA
     , temp.CD_PACIENTE
     , temp.NM_PACIENTE
     , temp.DT_NASCIMENTO
     , temp.DT_ATENDIMENTO
     , temp.DT_ALTA
     , temp.CD_PROCEDIMENTO
     , temp.DS_PROCEDIMENTO
     , temp.CD_REG_FAT
     , temp.CD_ATENDIMENTO
  from (SELECT NVL(REG_FAT.CD_GESTOR_AUTORIZOU, dbamv.fnc_ffis_autorizacao_gestor(cd_reg_fat))CD_GESTOR_AUTORIZOU
             , 'PENDENTE DE AUTORIZACAO'  DS_AUTORIZACAO
             , REG_FAT.NR_GUIA
             , ATENDIME.CD_PACIENTE
             , PACIENTE.NM_PACIENTE
             , PACIENTE.DT_NASCIMENTO
             , reg_fat.dt_inicio DT_ATENDIMENTO
             , reg_fat.dt_final  DT_ALTA
             , PROCEDIMENTO_SUS.CD_PROCEDIMENTO
             , PROCEDIMENTO_SUS.DS_PROCEDIMENTO
             , REG_FAT.CD_REG_FAT
             , REG_FAT.CD_ATENDIMENTO
          FROM DBAMV.REG_FAT
             , DBAMV.PACIENTE
             , DBAMV.ATENDIME
             , DBAMV.REMESSA_FATURA
             , DBAMV.FATURA
             , DBAMV.PROCEDIMENTO_SUS
             , dbamv.mot_alt
             , dbamv.config_ffis
         WHERE REG_FAT.CD_REMESSA         = REMESSA_FATURA.CD_REMESSA
           AND nvl(dbamv.fnc_ffis_autorizacao_gestor(cd_reg_fat),'X') <> nvl(cd_gestor_autorizou,'X')
           AND REG_FAT.Cd_Multi_Empresa   = :CAMPO_CD_MULTI_EMPRESA
           AND ATENDIME.CD_ATENDIMENTO    = REG_FAT.CD_ATENDIMENTO
           AND fatura.cd_multi_empresa    = :CAMPO_CD_MULTI_EMPRESA
           AND PACIENTE.CD_PACIENTE       = ATENDIME.CD_PACIENTE
           AND REMESSA_FATURA.CD_FATURA   = FATURA.CD_FATURA
           AND PROCEDIMENTO_SUS.CD_PROCEDIMENTO         = REG_FAT.CD_PROCEDIMENTO_REALIZADO
           and TO_CHAR( FATURA.DT_COMPETENCIA,'MM/YYYY') = TO_CHAR(:CAMPO_COMPET,'MM/YYYY')
           AND mot_alt.cd_mot_alt = reg_fat.cd_mot_alt
           AND reg_fat.cd_convenio = config_ffis.cd_convenio
           AND fatura.cd_convenio = config_ffis.cd_convenio
           :CAMPO_FC_AUTORIZACAO
           ) temp
 GROUP BY temp.CD_GESTOR_AUTORIZOU
        , temp.DS_AUTORIZACAO
        , temp.NR_GUIA
        , temp.CD_PACIENTE
        , temp.NM_PACIENTE
        , temp.DT_NASCIMENTO
        , temp.DT_ATENDIMENTO
        , temp.DT_ALTA
        , temp.CD_PROCEDIMENTO
        , temp.DS_PROCEDIMENTO
        , temp.CD_REG_FAT
        , temp.CD_ATENDIMENTO
ORDER BY temp.CD_GESTOR_AUTORIZOU
)
ORDER BY CD_GESTOR_AUTORIZOU,DS_AUTORIZACAO,CD_REG_FAT

--R_AIH_AUT_GESTOR_P321_RF_BITMAP_Q_AIH_AUTORIZADA
SELECT * FROM (
Select temp.CD_GESTOR_AUTORIZOU
     , temp.DS_AUTORIZACAO
     , temp.NR_GUIA
     , temp.CD_PACIENTE
     , temp.NM_PACIENTE
     , temp.DT_NASCIMENTO
     , temp.DT_ATENDIMENTO
     , temp.DT_ALTA
     , temp.CD_PROCEDIMENTO
     , temp.DS_PROCEDIMENTO
     , temp.CD_REG_FAT
     , temp.CD_ATENDIMENTO
  from (SELECT REG_FAT.CD_GESTOR_AUTORIZOU
             , autorizacao_gestor.ds_resumida  DS_AUTORIZACAO
             , REG_FAT.NR_GUIA
             , ATENDIME.CD_PACIENTE
             , PACIENTE.NM_PACIENTE
             , PACIENTE.DT_NASCIMENTO
             , reg_fat.dt_inicio DT_ATENDIMENTO
             , reg_fat.dt_final DT_ALTA
             , PROCEDIMENTO_SUS.CD_PROCEDIMENTO
             , PROCEDIMENTO_SUS.DS_PROCEDIMENTO
             , REG_FAT.CD_REG_FAT
             , REG_FAT.CD_ATENDIMENTO
          FROM DBAMV.REG_FAT
             , DBAMV.PACIENTE
             , DBAMV.ATENDIME
             , DBAMV.REMESSA_FATURA
             , DBAMV.FATURA
             , DBAMV.PROCEDIMENTO_SUS
             , dbamv.config_ffis
             , dbamv.autorizacao_gestor
         WHERE REG_FAT.CD_REMESSA         = REMESSA_FATURA.CD_REMESSA
           AND REG_FAT.Cd_Multi_Empresa   = :CAMPO_CD_MULTI_EMPRESA
           AND ATENDIME.CD_ATENDIMENTO    = REG_FAT.CD_ATENDIMENTO
           AND PACIENTE.CD_PACIENTE       = ATENDIME.CD_PACIENTE
           AND REMESSA_FATURA.CD_FATURA   = FATURA.CD_FATURA
           AND PROCEDIMENTO_SUS.CD_PROCEDIMENTO         = REG_FAT.CD_PROCEDIMENTO_REALIZADO
           and TO_CHAR( FATURA.DT_COMPETENCIA,'MM/YYYY') = TO_CHAR(:CAMPO_COMPET,'MM/YYYY')
           AND REG_FAT.CD_GESTOR_AUTORIZOU IS NOT NULL
           AND reg_fat.cd_convenio = config_ffis.cd_convenio
           AND fatura.cd_convenio = config_ffis.cd_convenio
           AND reg_fat.cd_gestor_autorizou = autorizacao_gestor.cd_gestor
           :CAMPO_FC_AUTORIZACAO) temp
 GROUP BY temp.CD_GESTOR_AUTORIZOU
        , temp.DS_AUTORIZACAO
        , temp.NR_GUIA
        , temp.CD_PACIENTE
        , temp.NM_PACIENTE
        , temp.DT_NASCIMENTO
        , temp.DT_ATENDIMENTO
        , temp.DT_ALTA
        , temp.CD_PROCEDIMENTO
        , temp.DS_PROCEDIMENTO
        , temp.CD_REG_FAT
        , temp.CD_ATENDIMENTO
ORDER BY temp.CD_GESTOR_AUTORIZOU
)
ORDER BY CD_GESTOR_AUTORIZOU,DS_AUTORIZACAO,CD_REG_FAT
