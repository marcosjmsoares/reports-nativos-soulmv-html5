--R_CONTAS_APRESENTACAO
select * from dual


/*----------------------------------------------------------------------------*/
--R_CONTAS_APRESENTACAO_R_Q_NORMAIS
SELECT A.CD_ATENDIMENTO
			 , A.CD_PACIENTE
     		 , DBAMV.PKG_SUS_REGRA_PACIENTE.F_NOME_PACIENTE(A.CD_PACIENTE) NM_PACIENTE
     		 , R.CD_REG_FAT
     		 , R.NR_GUIA
     		 , R.CD_PROCEDIMENTO_SOLICITADO
     		 , S.DS_PROCEDIMENTO
		 , TO_CHAR(R.DT_INICIO, 'DD/MM') || ' a ' ||  TO_CHAR(R.DT_FINAL, 'DD/MM')  DT_PERIODO
     		 , R.CD_MOT_ALT
             	 , TO_CHAR(ADD_MONTHS (DT_FINAL, 4), 'MM/YYYY') DT_EXPIRA
			 FROM DBAMV.REG_FAT R
  			 , DBAMV.CONFIG_FFIS F
     		 , DBAMV.ATENDIME A
     		 , DBAMV.PROCEDIMENTO_SUS S
     		 , DBAMV.MOTIVO_COBRANCA_SUS M
     		 WHERE A.CD_ATENDIMENTO    = R.CD_ATENDIMENTO
		     AND R.CD_CONVENIO       = F.CD_CONVENIO
		     AND S.CD_PROCEDIMENTO   = R.CD_PROCEDIMENTO_SOLICITADO
		     AND A.CD_MULTI_EMPRESA  = F.CD_MULTI_EMPRESA
		     AND R.CD_MOT_ALT        = M.CD_MOTIVO
		     AND R.CD_MULTI_EMPRESA  = PKG_MV2000.LE_EMPRESA
		     AND R.SN_FECHADA        = 'N'
		     AND R.CD_REG_FAT_GLOSA  IS NULL
   			 AND R.CD_REG_FAT_PAI    IS NULL
   			 AND R.CD_REG_FAT        NOT IN (SELECT CD_REG_FAT_GLOSA
		     FROM REG_FAT
             WHERE CD_REG_FAT_GLOSA IS NOT NULL)
			 AND R.CD_REG_FAT NOT IN (SELECT CD_REG_FAT
		     FROM GLOSAS_SUS)
     AND ADD_MONTHS (TRUNC (DT_FINAL, 'MM'), 4) BETWEEN TRUNC ( to_date(:CAMPO_COMPET1,'DD/MM/YYYY'),'MM')
                                                    AND TRUNC (NVL ( to_date(:CAMPO_COMPET2,'DD/MM/YYYY'),to_date(:CAMPO_COMPET1_1,'DD/MM/YYYY')),'MM')
			 ORDER BY R.NR_GUIA,R.DT_FINAL


/*----------------------------------------------------------------------------*/
--R_CONTAS_APRESENTACAO_R_Q_REAPRESENTADAS
SELECT    A.CD_ATENDIMENTO
	, A.CD_PACIENTE
	, P.NM_PACIENTE
	, R.CD_REG_FAT
	, R.NR_GUIA
	, R.CD_PROCEDIMENTO_REALIZADO
	, S.DS_PROCEDIMENTO
	, TO_CHAR(R.DT_INICIO, 'DD/MM') || ' a ' ||  TO_CHAR(R.DT_FINAL, 'DD/MM')  DT_PERIODO
	, R.CD_MOT_ALT
	, TO_CHAR (ADD_MONTHS (DT_FINAL, 6), 'MM/YYYY') DT_EXPIRA
    FROM REG_FAT R,
         CONFIG_FFIS F,
         ATENDIME A,
         PACIENTE P,
         PROCEDIMENTO_SUS S,
         MOTIVO_COBRANCA_SUS M
   WHERE A.CD_PACIENTE = P.CD_PACIENTE
     AND A.CD_ATENDIMENTO = R.CD_ATENDIMENTO
     AND R.CD_CONVENIO = F.CD_CONVENIO
     AND S.CD_PROCEDIMENTO = R.CD_PROCEDIMENTO_REALIZADO
     AND A.CD_MULTI_EMPRESA = F.CD_MULTI_EMPRESA
     AND R.CD_MOT_ALT = M.CD_MOTIVO
     AND R.CD_MULTI_EMPRESA = PKG_MV2000.LE_EMPRESA
     AND R.SN_FECHADA = 'N'
     AND R.CD_REG_FAT_PAI IS NULL
     AND (R.CD_REG_FAT) IN ((SELECT CD_REG_FAT_GLOSA
                               FROM REG_FAT
                              WHERE CD_REG_FAT_GLOSA IS NOT NULL))
     AND (R.CD_REG_FAT) NOT IN ((SELECT CD_REG_FAT
                                   FROM GLOSAS_SUS))
     AND ADD_MONTHS (TRUNC (DT_FINAL, 'MM'), 6) BETWEEN TRUNC ( to_date(:CAMPO_COMPET1,'DD/MM/YYYY'),'MM')
                                                    AND TRUNC (NVL ( to_date(:CAMPO_COMPET2,'DD/MM/YYYY'),to_date(:CAMPO_COMPET1_1,'DD/MM/YYYY')),'MM')
ORDER BY R.NR_GUIA,R.DT_FINAL
