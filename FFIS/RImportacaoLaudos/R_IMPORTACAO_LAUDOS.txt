--R_IMPORTACAO_LAUDOS
SELECT nm_arquivo
     , nr_seq
     , dh_geracao
     , nr_laudo
     , nr_guia
     , cd_atendimento
     , cd_reg_fat
     , ds_acao
  FROM DBAMV.LOG_IMPORTACAO_LAUDO_SUS
WHERE nm_arquivo = :CAMPO_NM_ARQUIVO
 ORDER BY nr_laudo, cd_atendimento, cd_reg_fat
