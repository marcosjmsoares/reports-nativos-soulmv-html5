SELECT  DBAMV.REG_FAT.CD_ATENDIMENTO
            , DBAMV.REG_FAT.NR_GUIA
            , DBAMV.REG_FAT.CD_REG_FAT
            ,Nvl(DBAMV.PKG_SUS_UTIL.F_RETORNA_VALOR_PREVIO(REG_FAT.CD_REG_FAT), 0) VL_TOTAL_CONTA
            , PACIENTE.NM_PACIENTE
FROM DBAMV.FATURA
             ,DBAMV.REMESSA_FATURA
            ,DBAMV.REG_FAT
            , DBAMV.PACIENTE
            , DBAMV.ATENDIME
            , DBAMV.CONFIG_FFIS
WHERE (DBAMV.FATURA.DT_COMPETENCIA = TO_DATE(:CAMPO_DT_COMP,'MM/YYYY'))
       AND (REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA )
       AND (DBAMV.REMESSA_FATURA.CD_FATURA = DBAMV.FATURA.CD_FATURA)
       AND (DBAMV.REG_FAT.CD_REMESSA = DBAMV.REMESSA_FATURA.CD_REMESSA)
       AND (DBAMV.REG_FAT.CD_ATENDIMENTO = DBAMV.ATENDIME.CD_ATENDIMENTO)
       AND (DBAMV.ATENDIME.CD_PACIENTE = DBAMV.PACIENTE.CD_PACIENTE)
       AND (DBAMV.FATURA.CD_CONVENIO = DBAMV.CONFIG_FFIS.CD_CONVENIO)
       AND (DBAMV.CONFIG_FFIS.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA )
