--R_ACOMPANHA_FPO
SELECT DBAMV.PKG_SUS_REGRA_FPO.F_RETORNA_DESC_TP_FINANC(F.TP_FINANCIAMENTO) TP_FINANCIAMENTO,
       F.CD_PROCEDIMENTO CD_PROCEDIMENTO,
       DBAMV.PKG_SUS_REGRA_FPO.F_DS_PROCEDIMENTO(F.CD_PROCEDIMENTO) DS_PROCEDIMENTO,
       F.QT_ORCADA QT_ORCADA,
       F.VL_UNITARIO_ORCADO VL_UNITARIO_ORCADO,
       F.VL_TOTAL_ORCADO VL_TOTAL_ORCADO,
       F.VL_ORCADO_COM_INCREMENTO VL_ORCADO_COM_INCREMENTO,
       V.QTD_TOTAL QTD_TOTAL,
       V.VL_TOTAL VL_TOTAL,
       V.PERC_VLR PERC_VLR
  FROM DBAMV.FPO F
     , DBAMV.V_PRODUCAO_FPO V
 WHERE V.DT_COMPET = F.DT_COMPETENCIA
   AND V.CD_PROCEDIMENTO = F.CD_PROCEDIMENTO
   AND F.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
   AND TO_CHAR(F.DT_COMPETENCIA,'MM/YYYY') = '03/2013'
ORDER BY DBAMV.PKG_SUS_REGRA_FPO.F_RETORNA_DESC_TP_FINANC(F.TP_FINANCIAMENTO)
