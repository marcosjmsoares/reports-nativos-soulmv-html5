--R_LAUDOS_STATUS
SELECT l.cd_atendimento  cd_atendimento
     , l.cd_reg_fat      cd_reg_fat
     , l.nr_laudo        nr_laudo
     , l.nr_guia         nr_guia
     , l.nm_paciente     nm_paciente
     , To_Char(l.dt_laudo, 'dd/mm/yyyy') dt_laudo
     , l.nr_lote         nr_lote
     , To_Char(a.dt_atendimento, 'dd/mm/yyyy') dt_atendimento
     , l.tp_status       tp_status
     , To_Char(l.dt_status, 'dd/mm/yyyy') dt_status
     , DBAMV.PKG_FFIS_LAUDO_INTEGRACAO.F_DIFERENCA_DATAS(l.dt_status, SYSDATE) duracao
  FROM DBAMV.LAUDOS_SIH l
     , DBAMV.ATENDIME   a
WHERE l.cd_atendimento = a.cd_atendimento
 ORDER BY l.dt_laudo DESC
