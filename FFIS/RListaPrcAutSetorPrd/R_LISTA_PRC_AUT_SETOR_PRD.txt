--R_LISTA_PRC_AUT_SETOR_PRD
SELECT PRD.TIPO
       , PRD.CD_PACIENTE
       , PRD.NM_PACIENTE
       , TO_CHAR(PRD.DT_LANCAMENTO,'DD/MM/YYYY') DT_LANCAMENTO
       , PRD.CD_SETOR
       , DBAMV.PKG_SUS_REGRA_SETOR.F_DESCRICAO(PRD.CD_SETOR) NM_SETOR
       , PRD.CD_PROCEDIMENTO
       , DBAMV.PKG_SUS_REGRA_PROCEDIMENTO.F_DESCRICAO(PRD.CD_PROCEDIMENTO) DS_PROCEDIMENTO
       , PRD.QTD QTD_LAN
       , NVL(PRG.QT_LANCAMENTO,0) QTD_PRG
    FROM (
          SELECT 'BPA' TIPO
                , EVE.CD_PACIENTE
                , DBAMV.PKG_SUS_REGRA_PACIENTE.F_NOME_PACIENTE(EVE.CD_PACIENTE) NM_PACIENTE
                , TRUNC(EVE.DT_EVE_SIASUS) DT_LANCAMENTO
                , EVE.CD_SETOR_PRODUZIU CD_SETOR
                , EVE.CD_PROCEDIMENTO CD_PROCEDIMENTO
                , SUM(EVE.QT_LANCADA) QTD
           FROM DBAMV.REMESSA_BPA  REMBPA
              , DBAMV.EVE_SIASUS EVE
          WHERE EVE.CD_REMESSA = REMBPA.CD_REMESSA
            AND EVE.CD_APAC IS NULL
            AND EVE.CD_PROCEDIMENTO IS NOT NULL
            AND EVE.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
       GROUP BY EVE.CD_PACIENTE
              , DBAMV.PKG_SUS_REGRA_PACIENTE.F_NOME_PACIENTE(EVE.CD_PACIENTE)
              , EVE.CD_SETOR_PRODUZIU
              , TRUNC(EVE.DT_EVE_SIASUS)
              , EVE.CD_PROCEDIMENTO
          UNION
         SELECT 'APAC' TIPO
                , EVE.CD_PACIENTE
                , DBAMV.PKG_SUS_REGRA_PACIENTE.F_NOME_PACIENTE(EVE.CD_PACIENTE) NM_PACIENTE
                , TRUNC(EVE.DT_EVE_SIASUS) DT_LANCAMENTO
                , EVE.CD_SETOR_PRODUZIU CD_SETOR
                , EVE.CD_PROCEDIMENTO CD_PROCEDIMENTO
                , SUM(EVE.QT_LANCADA) QTD
           FROM DBAMV.REMESSA_APAC  REMAPAC
              , DBAMV.APAC APAC
              , DBAMV.EVE_SIASUS EVE
          WHERE EVE.CD_APAC = APAC.CD_APAC
            AND APAC.CD_REMESSA = REMAPAC.CD_REMESSA
            AND EVE.CD_APAC IS NOT NULL
            AND EVE.CD_PROCEDIMENTO IS NOT NULL
            AND EVE.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
       GROUP BY EVE.CD_PACIENTE,TRUNC(EVE.DT_EVE_SIASUS)
              , EVE.CD_PROCEDIMENTO, EVE.CD_SETOR_PRODUZIU
          UNION
         SELECT 'AIH' TIPO
                , AD.CD_PACIENTE
                , DBAMV.PKG_SUS_REGRA_PACIENTE.F_NOME_PACIENTE(AD.CD_PACIENTE) NM_PACIENTE
                , TRUNC(DT_LANCAMENTO) DT_LANCAMENTO
                , ITF.CD_SETOR_PRODUZIU
                , ITF.CD_PROCEDIMENTO
                , SUM(ITF.QT_LANCAMENTO) QTD
           FROM DBAMV.ITREG_FAT ITF,
                DBAMV.REG_FAT RF,
                DBAMV.ATENDIME AD
          WHERE ITF.CD_PROCEDIMENTO IS NOT NULL
            AND AD.CD_ATENDIMENTO = RF.CD_ATENDIMENTO
            AND ITF.CD_REG_FAT NOT IN (SELECT CD_REG_FAT FROM DBAMV.GLOSAS_SUS)
            AND ITF.CD_REG_FAT = RF.CD_REG_FAT
            AND RF.CD_MULTI_EMPRESA = ITF.CD_MULTI_EMPRESA
            AND RF.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
       GROUP BY AD.CD_PACIENTE,TRUNC(DT_LANCAMENTO)
              , ITF.CD_PROCEDIMENTO,ITF.CD_SETOR_PRODUZIU) PRD
              , DBAMV.SETOR_PROCEDIMENTO_FATRMTO_SUS PRG
  WHERE PRD.TIPO            = PRG.TP_INSTRUMENTO(+)
    AND PRD.CD_SETOR        = PRG.CD_SETOR(+)
    AND PRD.CD_PROCEDIMENTO = PRG.CD_PROCEDIMENTO(+)
    AND PRD.QTD       > NVL(PRG.QT_LANCAMENTO,0)
    AND TO_DATE(:CAMPO_COMPETENCIA,'MM/YYYY') BETWEEN TRUNC(NVL(PRG.DT_VALIDADE_INICIAL,TO_DATE(:CAMPO_COMPETENCIA,'MM/YYYY')),'MM') AND TRUNC(NVL(PRG.DT_VALIDADE_FINAL,TO_DATE(:CAMPO_COMPETENCIA,'MM/YYYY')),'MM')
    AND TO_CHAR(PRD.DT_LANCAMENTO,'MM/YYYY') = :CAMPO_COMPETENCIA
    AND PRD.TIPO = :CAMPO_INSTRUMENTO
ORDER BY CD_SETOR,CD_PROCEDIMENTO
