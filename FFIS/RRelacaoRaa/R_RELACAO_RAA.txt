--R_RELACAO_RAA
select r.cd_espec_sus,
           r.nr_guia
  from dbamv.reg_fat r,
       dbamv.remessa_fatura e,
       dbamv.fatura f
 where r.cd_remessa = e.cd_remessa
   and r.cd_convenio = dbamv.pkt_config_ffis.retorna_campo('CD_CONVENIO')
   and e.cd_fatura = f.cd_fatura
   and r.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
   and to_char(f.dt_competencia,'mm/yyyy') = :CAMPO_COMPET
   and r.nr_guia is not null
group by r.cd_espec_sus,
         r.nr_guia
ORDER BY CD_ESPEC_SUS
