SELECT DISTINCT FATURA.CD_FATURA
            , REG_FAT.CD_REMESSA
            , REG_FAT.CD_ATENDIMENTO
            , REG_FAT.NR_GUIA
            , REG_FAT.CD_REG_FAT
            , FATURA.DT_COMPETENCIA
            , PACIENTE.CD_PACIENTE
            , PACIENTE.NM_PACIENTE
            , REG_FAT.CD_PROCEDIMENTO_REALIZADO
            , ATENDIME.CD_PACIENTE
            , ATENDIME.CD_ATENDIMENTO
            , REMESSA_FATURA.CD_REMESSA
  FROM DBAMV.FATURA
           , DBAMV.REMESSA_FATURA
           , DBAMV.REG_FAT
           , DBAMV.PACIENTE
           , DBAMV.ATENDIME
WHERE TO_CHAR(FATURA.DT_COMPETENCIA, 'MM/YYYY') = TO_CHAR(:CAMPO_DT_COMP, 'MM/YYYY')
      AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
      AND REMESSA_FATURA.CD_FATURA = FATURA.CD_FATURA
      AND REG_FAT.CD_REMESSA = REMESSA_FATURA.CD_REMESSA
      AND REG_FAT.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
      AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
ORDER BY DBAMV.PACIENTE.NM_PACIENTE
