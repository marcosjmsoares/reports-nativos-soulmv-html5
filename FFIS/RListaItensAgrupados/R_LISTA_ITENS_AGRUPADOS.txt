--R_LISTA_ITENS_AGRUPADOS
SELECT * FROM (
select reg_fat.cd_atendimento
     , reg_fat.cd_reg_fat
     , itreg_fat.cd_procedimento || ' - ' || procedimento_sus.ds_procedimento cd_procedimento
     , itreg_fat.cd_lancamento
     , itreg_fat.qt_lancamento
     , itreg_fat.cd_setor
     , itreg_fat.cd_setor_produziu
     , itreg_fat.cd_prestador
     , itreg_fat.cd_cbo_prestador
     , itreg_fat.cd_tipo_vinculo
     , itreg_fat.dt_lancamento
     , agrupamento_item_conta.cd_lancamento      cd_lancamento_itagrp
     , agrupamento_item_conta.qt_lancamento      qt_lancamento_itagrp
     , agrupamento_item_conta.cd_setor          cd_setor_itagrp
     , agrupamento_item_conta.cd_setor_produziu cd_setor_produziu_itagrp
     , agrupamento_item_conta.cd_prestador       cd_prestador_itagrp
     , agrupamento_item_conta.cd_cbo_prestador   cd_cbo_prest_itagrp
     , agrupamento_item_conta.cd_tipo_vinculo    cd_tipo_vinculo_itagrp
     , agrupamento_item_conta.dt_lancamento     dt_lancamento_itagrp
     , agrupamento_item_conta.nm_usuario         nm_usuario_itagrp
     , agrupamento_item_conta.dt_agrupamento     dt_agrupamento_itagrp
     , decode(nvl(agrupamento_item_conta.sn_pertence_pacote,'N'),'N','Não','Sim') sn_pacote_itagrp
     , decode(nvl(agrupamento_item_conta.sn_pertence_pacote,'N'),'N',1,0) qt_pacote_n_itagrp
     , decode(nvl(agrupamento_item_conta.sn_pertence_pacote,'N'),'S',1,0) qt_pacote_s_itagrp
  from dbamv.reg_fat
     , dbamv.itreg_fat
     , dbamv.procedimento_sus
     , dbamv.agrupamento_item_conta
 where itreg_fat.cd_reg_fat = reg_fat.cd_reg_fat
   and procedimento_sus.cd_procedimento = itreg_fat.cd_procedimento
   and agrupamento_item_conta.cd_reg_fat = itreg_fat.cd_reg_fat
   and agrupamento_item_conta.cd_lancamento_pai = itreg_fat.cd_lancamento
   :CAMPO_CF_REMESSA
)
ORDER BY CD_ATENDIMENTO,CD_REG_FAT,
CD_PROCEDIMENTO,CD_LANCAMENTO,QT_LANCAMENTO,CD_SETOR,CD_SETOR_PRODUZIU,CD_PRESTADOR,CD_CBO_PRESTADOR,DT_LANCAMENTO,CD_TIPO_VINCULO,NM_USUARIO_ITAGRP,DT_AGRUPAMENTO_ITAGRP
