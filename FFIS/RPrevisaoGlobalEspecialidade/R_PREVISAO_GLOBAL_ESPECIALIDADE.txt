--R_PREVISAO_GLOBAL_ESPECIALIDADE
SELECT ESPEC_SUS.CD_ESPEC_SUS
     ,  DS_ESPEC_SUS||' ('||ESPEC_SUS.CD_ESPEC_SUS||')' DS_ESPEC_SUS
     , TP_INSTRUMENTO      TP_INSTRUMENTO
     , DS_INSTRUMENTO      DS_INSTRUMENTO
     , ID                  ID
     , DESCRICAO           DESCRICAO
     , SUM(DIAS)           DIAS
     , SUM(QT_LANCAMENTO)  QT_LANCAMENTO
     , SUM(VL_TOTAL)       VL_TOTAL
     , SUM(VL_SH_PROPRIO)  VL_SH_PROPRIO
     , SUM(VL_SH_TERCEIRO) VL_SH_TERCEIRO
     , SUM(VL_SP_PROPRIO)  VL_SP_PROPRIO
     , SUM(VL_SP_TERCEIRO) VL_SP_TERCEIRO
 FROM DBAMV.V_FFIS_PRODUCAO_AIH VW
    , DBAMV.REMESSA_FATURA RF
    , DBAMV.FATURA FT
    , DBAMV.ESPEC_SUS
WHERE VW.CD_REMESSA = RF.CD_REMESSA
  AND RF.CD_FATURA = FT.CD_FATURA
  AND TO_CHAR(VW.DT_COMPETENCIA,'MM/YYYY') = TO_CHAR(:CAMPO_COMPET,'MM/YYYY')
  AND ESPEC_SUS.CD_ESPEC_SUS = VW.CD_ESPEC_SUS
GROUP BY ESPEC_SUS.CD_ESPEC_SUS
       , DS_ESPEC_SUS
       , TP_INSTRUMENTO
       , DS_INSTRUMENTO
       , ID
       , DESCRICAO
ORDER BY DS_ESPEC_SUS
       , TP_INSTRUMENTO
       , DS_INSTRUMENTO
       , ID
