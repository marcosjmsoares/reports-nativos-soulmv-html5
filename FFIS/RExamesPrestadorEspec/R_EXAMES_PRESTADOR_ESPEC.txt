--R_EXAMES_PRESTADOR_ESPEC
SELECT * FROM (
SELECT ESPEC_SUS.CD_ESPEC_SUS
     , ESPEC_SUS.DS_ESPEC_SUS
     , PROCEDIMENTO_SUS.CD_PROCEDIMENTO
     , PROCEDIMENTO_SUS.DS_PROCEDIMENTO
     , PRESTADOR.CD_PRESTADOR
     , PRESTADOR.NM_PRESTADOR
     ,  SUM(nvl(ITREG_FAT.QT_LANCAMENTO,0))  QT_LANCAMENTO
     , (SUM(nvl(ITREG_FAT.QT_PONTOS    ,0)) /
        SUM(nvl(ITREG_FAT.QT_LANCAMENTO,0))) QT_UNIT_PONTOS
     ,  SUM(nvl(ITREG_FAT.QT_PONTOS    ,0))  QT_PONTOS
     ,  SUM(nvl(DECODE(PROCEDIMENTO_SUS_DETALHE.SN_SANGUE,'S',ITREG_FAT.VL_SH,ITREG_FAT.VL_TOTAL_CONTA),0)) VL_TOTAL
     ,  SUM(NVL(ITENS_SD_FFIS.VL_SD,0))   VL_SD
     ,  SUM(NVL(ITENS_SD_FFIS.VL_SD_GLOSA,0))  VL_SD_GLOSA
     ,  SUM(NVL(ITENS_SD_FFIS.VL_SD_SEM_GLOSA,0))  VL_SD_SEM_GLOSA
     ,  SUM(NVL(ITENS_SD_FFIS.VL_SD_GLOSA_TOTAL,0))  VL_SD_GLOSA_TOTAL
     ,  SUM(NVL(ITENS_SD_FFIS.VL_SD_GLOSA_REAP,0))   VL_SD_GLOSA_REAP
  FROM DBAMV.ITREG_FAT
     , DBAMV.GRU_PRO
     , DBAMV.PROCEDIMENTO_SUS
     , DBAMV.PROCEDIMENTO_SUS_DETALHE
     , DBAMV.PRESTADOR
     , DBAMV.REG_FAT
     , DBAMV.CONVENIO
     , DBAMV.REMESSA_FATURA
     , DBAMV.FATURA
     , DBAMV.ESPEC_SUS
     , DBAMV.ITENS_SD_FFIS
 WHERE CONVENIO.TP_CONVENIO            = 'H'
   AND ITREG_FAT.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
   AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   AND PROCEDIMENTO_SUS_DETALHE.CD_PROCEDIMENTO  = REG_FAT.CD_PROCEDIMENTO_REALIZADO
   AND ITREG_FAT.CD_PROCEDIMENTO                 = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
   AND PROCEDIMENTO_SUS_DETALHE.CD_GRU_PRO       = GRU_PRO.CD_GRU_PRO
   AND ITREG_FAT.CD_REG_FAT                      = REG_FAT.CD_REG_FAT
   AND ITREG_FAT.CD_REG_FAT                      = ITENS_SD_FFIS.CD_REG_FAT
   AND ITREG_FAT.CD_LANCAMENTO                   = ITENS_SD_FFIS.CD_LANCAMENTO
   AND REG_FAT.CD_REMESSA                        = REMESSA_FATURA.CD_REMESSA
   AND REMESSA_FATURA.CD_FATURA                  = FATURA.CD_FATURA
   AND REG_FAT.CD_CONVENIO                       = CONVENIO.CD_CONVENIO
   AND REG_FAT.CD_ESPEC_SUS                      = ESPEC_SUS.CD_ESPEC_SUS
   AND to_char(FATURA.DT_COMPETENCIA, 'MM/YYYY') = TO_CHAR(:CAMPO_COMPET, 'MM/YYYY')
GROUP BY PRESTADOR.CD_PRESTADOR
       , PRESTADOR.NM_PRESTADOR
       , ESPEC_SUS.DS_ESPEC_SUS
       , ESPEC_SUS.CD_ESPEC_SUS
       , PROCEDIMENTO_SUS.CD_PROCEDIMENTO
       , PROCEDIMENTO_SUS.DS_PROCEDIMENTO
ORDER BY CD_PRESTADOR,NM_PRESTADOR,
DS_ESPEC_SUS,CD_ESPEC_SUS
)
ORDER BY CD_PRESTADOR,NM_PRESTADOR,
DS_ESPEC_SUS,CD_ESPEC_SUS
