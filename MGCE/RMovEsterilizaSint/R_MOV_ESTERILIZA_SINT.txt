--R_MOV_ESTERILIZA_SINT
Select distinct
    Maq_esterilizador.cd_esterilizador Esterilizadora,
    Bens.Ds_Bem Descricao,
    Cx_Kit.Cd_Kit CodBarras,
    Prestador.nm_prestador Identificacao,
    it_mvto_cme.cd_usuario_inc NaoIdentifica,
    Cx_kit.ds_kit KitouCaixa,
    Esterilizacao.cd_carga_lote Carga,
    Esterilizacao.cd_esteriliza cd_esteriliza,
    To_char(esterilizacao.dt_inic_esteriliza,'DD/MM/YYYY HH24')InicioData,
    To_char(esterilizacao.dt_fim_esteriliza,'DD/MM/YYYY HH24')FimData,
    Config_Cme.Sn_Identi_Geral
From
    Dbamv.Maq_Esterilizador,
    Dbamv.Bens,
    Dbamv.prestador,
    Dbamv.cx_kit,
    dbamv.lote_ampola,
    Dbamv.esterilizacao,
    Dbamv.it_mvto_cme,
    Dbamv.config_cme,
    Dbamv.Mvto_Cme
Where Maq_Esterilizador.cd_esterilizador = esterilizacao.cd_esterilizador
  and Maq_Esterilizador.cd_bem = Bens.Cd_bem
  and Prestador.cd_prestador(+) = It_mvto_cme.cd_prestador_identifica
  and Cx_kit.cd_kit = it_mvto_cme.cd_kit
  and Lote_ampola.cd_esteriliza(+) = Esterilizacao.cd_esteriliza
  and Esterilizacao.cd_esteriliza = It_mvto_cme.cd_esteriliza
  and Config_cme.cd_setor_esterilizacao = It_mvto_cme.cd_setor
  and Esterilizacao.Sn_Est_Completa = 'S'
  and (Esterilizacao.Sn_Biologico <> 'N' or Esterilizacao.Sn_Quimico <> 'N' or Esterilizacao.Sn_Fisico = 'S')
  and Trunc(Esterilizacao.dt_inic_esteriliza) between :CAMPO_DATA_INI AND 
  and Mvto_Cme.Cd_Mvto_Cme = It_Mvto_Cme.Cd_Mvto_Cme
  and (config_cme.cd_setor_principal in (select cd_setor
                                                                  from dbamv.setor
                                                               where cd_multi_empresa = Pkg_Mv2000.Le_Empresa)
       or config_cme.cd_setor_principal is null)
order by Maq_esterilizador.cd_esterilizador, substr('InicioData',0), Prestador.nm_prestador
