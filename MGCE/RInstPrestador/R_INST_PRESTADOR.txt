--R_INST_PRESTADOR
select cx_kit.cd_kit CodCaixaKit
      ,cx_kit.ds_kit DescCaixaKit
      ,cx_kit.ds_res_kit DescResumida
      ,tp_embalagem.ds_embalagem TpEmbalagem
      ,prestador.cd_prestador CodPrestador
      ,prestador.nm_prestador DesPrestador
      ,cx_kit.qt_itens QtdItens
      ,tp_instrumental.ds_instrumental TpInstrumental
      ,cx_kit.dt_val_esterilizacao Validade
      ,cx_kit.qt_x_esteriliza QtdEsterilizacao
from dbamv.cx_kit
    ,dbamv.prestador
    ,dbamv.tp_embalagem
    ,dbamv.tp_instrumental
where cx_kit.cd_prestador = prestador.cd_prestador
and tp_embalagem.cd_embalagem = cx_kit.cd_embalagem
and tp_instrumental.cd_tp_instrumental = cx_kit.cd_tp_instrumental
order by cx_kit.cd_kit, prestador.nm_prestador
