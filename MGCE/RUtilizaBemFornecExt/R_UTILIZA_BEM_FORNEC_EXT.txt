--R_UTILIZA_BEM_FORNEC_EXT
SELECT BEM_TERCEIRO_CME.DS_BEM_TERCEIRO AS DS_BEM_TERCEIRO
     , FORNECEDOR.NM_FORNECEDOR AS NM_FORNECEDOR
     , ITEM_AVULSO_SOLICITACAO_CME.CD_SOLICITACAO_CME AS CD_SOLICITACAO_CME
     , ITEM_AVULSO_SOLICITACAO_CME.QT_ITEM_AVULSO AS QT_ITEM_AVULSO
     , SOLICITACAO_CME.CD_AVISO_CIRURGIA AS CD_AVISO_CIRURGIA
     , DECODE(AVISO_CIRURGIA.TP_SITUACAO
            , 'A', :CAMPO_TRAD_EM_AVISO
            , 'R', :CAMPO_TRAD_REALIZADA
            , 'C', :CAMPO_TRAD_CANCELADA
            , 'G', :CAMPO_TRAD_AGENDADA
            , 'T', :CAMPO_TRAD_CONTR_CHEC
            , 'P', :CAMPO_TRAD_PRE_AGEND
            , AVISO_CIRURGIA.TP_SITUACAO) AS TP_SITUACAO
     , NVL((SELECT PRESTADOR_AVISO.NM_PRESTADOR AS NM_PRESTADOR
              FROM DBAMV.CIRURGIA_AVISO
                 , DBAMV.PRESTADOR_AVISO
             WHERE CIRURGIA_AVISO.CD_AVISO_CIRURGIA  = AVISO_CIRURGIA.CD_AVISO_CIRURGIA
               AND PRESTADOR_AVISO.CD_AVISO_CIRURGIA = AVISO_CIRURGIA.CD_AVISO_CIRURGIA
               AND CIRURGIA_AVISO.SN_PRINCIPAL       = 'S'
               AND PRESTADOR_AVISO.SN_PRINCIPAL      = 'S'
               AND ROWNUM = 1
           ), 'CIRURGIÃO NÃO INFORMADO') AS NM_CIRURGIAO
     ,NVL((SELECT ESPECIALID.DS_ESPECIALID
             FROM DBAMV.CIRURGIA_AVISO
                , DBAMV.PRESTADOR_AVISO
                , DBAMV.ESPECIALID
                , DBAMV.PRESTADOR
            WHERE CIRURGIA_AVISO.CD_AVISO_CIRURGIA  = AVISO_CIRURGIA.CD_AVISO_CIRURGIA
              AND PRESTADOR_AVISO.CD_AVISO_CIRURGIA = AVISO_CIRURGIA.CD_AVISO_CIRURGIA
              AND PRESTADOR_AVISO.CD_PRESTADOR      = PRESTADOR.CD_PRESTADOR
              AND ESPECIALID.CD_ESPECIALID          = PRESTADOR.CD_ESPECIALID_RESIDENTE
              AND CIRURGIA_AVISO.SN_PRINCIPAL       = 'S'
              AND PRESTADOR_AVISO.SN_PRINCIPAL      = 'S'
              AND ROWNUM = 1), ' ') AS DS_ESPECIALIDADE
     , NVL((SELECT SUM(QT_UTILIZADA)
              FROM DBAMV.IT_AV_BEM_FORN_EXT_UTIL
             WHERE IT_AV_BEM_FORN_EXT_UTIL.CD_SOLICITACAO_CME = SOLICITACAO_CME.CD_SOLICITACAO_CME
               AND IT_AV_BEM_FORN_EXT_UTIL.CD_AVISO_CIRURGIA  = SOLICITACAO_CME.CD_AVISO_CIRURGIA
               AND IT_AV_BEM_FORN_EXT_UTIL.CD_BEM_TERCEIRO    = ITEM_AVULSO_SOLICITACAO_CME.CD_BEM_TERCEIRO
               AND IT_AV_BEM_FORN_EXT_UTIL.CD_FORNECEDOR      = ITEM_AVULSO_SOLICITACAO_CME.CD_FORNECEDOR
               AND IT_AV_BEM_FORN_EXT_UTIL.CD_SOLICITACAO_CME = ITEM_AVULSO_SOLICITACAO_CME.CD_SOLICITACAO_CME
            ), 0) AS QT_UTILIZADA
  FROM DBAMV.SOLICITACAO_CME
     , DBAMV.ITEM_AVULSO_SOLICITACAO_CME
     , DBAMV.BEM_TERCEIRO_CME
     , DBAMV.FORNECEDOR
     , DBAMV.AVISO_CIRURGIA
 WHERE ITEM_AVULSO_SOLICITACAO_CME.CD_SOLICITACAO_CME = SOLICITACAO_CME.CD_SOLICITACAO_CME
   AND BEM_TERCEIRO_CME.CD_BEM_TERCEIRO               = ITEM_AVULSO_SOLICITACAO_CME.CD_BEM_TERCEIRO
   AND FORNECEDOR.CD_FORNECEDOR                       = ITEM_AVULSO_SOLICITACAO_CME.CD_FORNECEDOR
   AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA               = SOLICITACAO_CME.CD_AVISO_CIRURGIA
   AND SOLICITACAO_CME.CD_MULTI_EMPRESA               = :CAMPO_CD_MULTI_EMPRESA
   AND SOLICITACAO_CME.TP_SITUACAO NOT IN (98,99)
   AND SOLICITACAO_CME.CD_AVISO_CIRURGIA IS NOT NULL
   AND ITEM_AVULSO_SOLICITACAO_CME.CD_BEM_TERCEIRO IS NOT NULL
   AND ITEM_AVULSO_SOLICITACAO_CME.CD_FORNECEDOR IS NOT NULL
ORDER BY SOLICITACAO_CME.CD_AVISO_CIRURGIA
       , ITEM_AVULSO_SOLICITACAO_CME.CD_SOLICITACAO_CME
       , BEM_TERCEIRO_CME.DS_BEM_TERCEIRO
       , FORNECEDOR.NM_FORNECEDOR
