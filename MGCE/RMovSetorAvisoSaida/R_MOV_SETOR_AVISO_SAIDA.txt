--R_MOV_SETOR_AVISO_SAIDA
SELECT
						MVTO_CME.CD_MVTO_CME,
						SETOR_PRINCIPAL.CD_SETOR,
						SETOR_PRINCIPAL.NM_SETOR,
						MVTO_CME.CD_AVISO_CIRURGIA,
						AVISO_CIRURGIA.NM_PACIENTE,
						PACIENTE.NM_SOCIAL_PACIENTE NM_SOCIAL_PACIENTE ,
						PACIENTE.DT_NASCIMENTO DT_NASCIMENTO ,
						IT_MVTO_CME.CD_KIT,
						CX_KIT.DS_KIT
					FROM
						DBAMV.MVTO_CME ,
						DBAMV.IT_MVTO_CME ,
						DBAMV.CX_KIT ,
						DBAMV.SETOR SETOR_PRINCIPAL ,
						DBAMV.AVISO_CIRURGIA ,
						DBAMV.PACIENTE
					WHERE
						MVTO_CME.CD_MVTO_CME = IT_MVTO_CME.CD_MVTO_CME
						AND IT_MVTO_CME.CD_KIT = CX_KIT.CD_KIT
						AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA(+) = MVTO_CME.CD_AVISO_CIRURGIA
						AND AVISO_CIRURGIA.CD_PACIENTE = PACIENTE.CD_PACIENTE(+)
						AND TRUNC(MVTO_CME.DT_INC_USUARIO) BETWEEN :CAMPO_DATA_INI AND 
						AND MVTO_CME.CD_SETOR_PRINCIPAL = SETOR_PRINCIPAL.CD_SETOR
						AND SETOR_PRINCIPAL.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
						AND MVTO_CME.TP_MVTO_CME = 'S' 
					ORDER BY
						SETOR_PRINCIPAL.CD_SETOR,
						MVTO_CME.CD_AVISO_CIRURGIA,
						MVTO_CME.CD_MVTO_CME DESC
