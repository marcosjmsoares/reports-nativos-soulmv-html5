--R_MOV_CICLO
SELECT DISTINCT CICLO.* FROM ( 
SELECT ESOC.CD_ESTERILIZA CD_CICLO,
       NULL COD_MOV,
       SOL.CD_SETOR_PRINCIPAL,
       ITAV.CD_BEM AS CD_KIT,
       BENS.DS_BEM  DS_KIT,
       'INSTR. AVULSO' AS TABELA,
       'ESTERILIZAÇÃO' AS TP_SITUACAO,
        E.DT_FIM_ESTERILIZA AS DT_MOVIMENTO
        ,1 AS CD_FASE    
  FROM DBAMV.ESTERILIZA_SOLICITACAO_CME ESOC,
       DBAMV.SOLICITACAO_CME SOL,
       DBAMV.ESTERILIZACAO E,
       DBAMV.ITEM_AVULSO_SOLICITACAO_CME ITAV,
       DBAMV.BENS 
WHERE ESOC.CD_SOLICITACAO_CME = SOL.CD_SOLICITACAO_CME
AND E.CD_ESTERILIZA = ESOC.CD_ESTERILIZA
AND ESOC.CD_SOLICITACAO_CME = ITAV.CD_SOLICITACAO_CME
AND BENS.CD_BEM = ITAV.CD_BEM
AND E.DT_FIM_ESTERILIZA IS NOT NULL
UNION ALL 
SELECT ESOC.CD_ESTERILIZA CD_CICLO,
       NULL COD_MOV,
       SOL.CD_SETOR_PRINCIPAL,
       ITAV.CD_BEM_TERCEIRO AS CD_KIT,                                                                   
       BEM_TERCEIRO_CME.DS_BEM_TERCEIRO DS_KIT,
       'BEM TERCEIRO' AS TABELA ,
       'ESTERILIZAÇÃO' AS TP_SITUACAO,
        E.DT_FIM_ESTERILIZA AS DT_MOVIMENTO
        ,1 AS CD_FASE  
  FROM DBAMV.ESTERILIZA_SOLICITACAO_CME ESOC,
       DBAMV.SOLICITACAO_CME SOL,
       DBAMV.ESTERILIZACAO E,
       DBAMV.ITEM_AVULSO_SOLICITACAO_CME ITAV,
       DBAMV.BEM_TERCEIRO_CME 
WHERE ESOC.CD_SOLICITACAO_CME = SOL.CD_SOLICITACAO_CME 
AND ESOC.CD_SOLICITACAO_CME = ITAV.CD_SOLICITACAO_CME
AND E.CD_ESTERILIZA = ESOC.CD_ESTERILIZA
AND BEM_TERCEIRO_CME.CD_BEM_TERCEIRO = ITAV.CD_BEM_TERCEIRO
AND E.DT_FIM_ESTERILIZA IS NOT NULL
UNION ALL 
SELECT ESOC.CD_ESTERILIZA CD_CICLO,
       NULL COD_MOV,
       SOL.CD_SETOR_PRINCIPAL,
       CX_KIT.CD_KIT AS CD_KIT,                                                                   
       CX_KIT.DS_KIT DS_KIT,
       'KIT OU CAIXA' AS TABELA ,
       'ESTERILIZAÇÃO' AS TP_SITUACAO,
        E.DT_FIM_ESTERILIZA AS DT_MOVIMENTO
        ,1 AS CD_FASE  
  FROM DBAMV.ESTERILIZA_SOLICITACAO_CME ESOC,
       DBAMV.SOLICITACAO_CME SOL,
       DBAMV.ESTERILIZACAO E,
       DBAMV.COMPOSICAO_SOLICITACAO_CME CSC,
       DBAMV.CX_KIT 
WHERE ESOC.CD_SOLICITACAO_CME = SOL.CD_SOLICITACAO_CME 
AND ESOC.CD_SOLICITACAO_CME = CSC.CD_SOLICITACAO_CME
AND E.CD_ESTERILIZA = ESOC.CD_ESTERILIZA
AND CX_KIT.CD_KIT = CSC.CD_KIT
AND E.DT_FIM_ESTERILIZA IS NOT NULL
UNION ALL 
SELECT Nvl(I.CD_ESTERILIZA,DBAMV.FNC_MGCE_ESTERILIZA_ANTERIOR(M.CD_MVTO_CME,I.CD_KIT,'K')) AS  CD_CICLO,
       I.CD_MVTO_CME COD_MOV,
       M.CD_SETOR_PRINCIPAL,
       CX_KIT.CD_KIT AS CD_KIT,                                                                   
       CX_KIT.DS_KIT DS_KIT,
        'KIT OU CAIXA' AS TABELA ,
        DECODE(M.TP_MVTO_CME,'E','ENTRADA ARSENAL','S','SAÍDA ARSENAL','D','DEVOLUÇÃO ARSENAL') AS TP_SITUACAO,
        M.DT_INC_USUARIO AS DT_MOVIMENTO
        ,DECODE(M.TP_MVTO_CME,'E',2,'S',3,'D',4) AS CD_FASE
   FROM DBAMV.MVTO_CME M,
        DBAMV.IT_MVTO_CME I,
        DBAMV.CX_KIT
  WHERE M.CD_MVTO_CME = I.CD_MVTO_CME 
  AND I.CD_KIT = CX_KIT.CD_KIT
  AND M.TP_MVTO_CME IN ('S','E','D')
UNION ALL 
SELECT Nvl(I.CD_ESTERILIZA,DBAMV.FNC_MGCE_ESTERILIZA_ANTERIOR(M.CD_MVTO_CME,I.CD_BEM,'B')) AS  CD_CICLO,
       I.CD_MVTO_CME COD_MOV,
         M.CD_SETOR_PRINCIPAL,
       BENS.CD_BEM AS CD_KIT,                                                                   
       BENS.DS_BEM DS_KIT,
        'INSTR. AVULSO' AS TABELA ,
        DECODE(M.TP_MVTO_CME,'E','ENTRADA ARSENAL','S','SAÍDA ARSENAL','D','DEVOLUÇÃO ARSENAL') AS TP_SITUACAO,
        M.DT_INC_USUARIO AS DT_MOVIMENTO
        ,DECODE(M.TP_MVTO_CME,'E',2,'S',3,'D',4) AS CD_FASE
   FROM DBAMV.MVTO_CME M,
        DBAMV.IT_MVTO_CME I,
        DBAMV.BENS
  WHERE M.CD_MVTO_CME = I.CD_MVTO_CME 
  AND I.CD_BEM = BENS.CD_BEM
  AND I.CD_KIT IS NULL
  AND M.TP_MVTO_CME IN ('S','E','D')
UNION ALL 
SELECT Nvl(I.CD_ESTERILIZA,DBAMV.FNC_MGCE_ESTERILIZA_ANTERIOR(M.CD_MVTO_CME,I.CD_BEM_TERCEIRO,'BT')) AS  CD_CICLO,
       I.CD_MVTO_CME COD_MOV,
         M.CD_SETOR_PRINCIPAL,
       BEM_TERCEIRO_CME.CD_BEM_TERCEIRO AS CD_KIT,                                                                   
       BEM_TERCEIRO_CME.DS_BEM_TERCEIRO DS_KIT,
        'BEM TERCEIRO' AS TABELA ,
        DECODE(M.TP_MVTO_CME,'E','ENTRADA ARSENAL','S','SAÍDA ARSENAL','D','DEVOLUÇÃO ARSENAL') AS TP_SITUACAO,
        M.DT_INC_USUARIO AS DT_MOVIMENTO
        ,DECODE(M.TP_MVTO_CME,'E',2,'S',3,'D',4) AS CD_FASE
   FROM DBAMV.MVTO_CME M,
        DBAMV.IT_MVTO_CME I,
        DBAMV.BEM_TERCEIRO_CME
  WHERE M.CD_MVTO_CME = I.CD_MVTO_CME 
  AND I.CD_BEM_TERCEIRO = BEM_TERCEIRO_CME.CD_BEM_TERCEIRO
  AND I.CD_KIT IS NULL
  AND M.TP_MVTO_CME IN ('S','E','D')
) CICLO, DBAMV.ESTERILIZACAO , DBAMV.SETOR                                                
WHERE CICLO.CD_CICLO IS NOT NULL 
AND CICLO.CD_CICLO = ESTERILIZACAO.CD_ESTERILIZA
AND  CICLO.CD_SETOR_PRINCIPAL = SETOR.CD_SETOR
AND TRUNC(CICLO.DT_MOVIMENTO)  Between :CAMPO_DATA_INI  ORDER BY CD_CICLO,DT_MOVIMENTO,CD_FASE
