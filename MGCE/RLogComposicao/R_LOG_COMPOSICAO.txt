--R_LOG_COMPOSICAO
Select Distinct ds_cd_barras
     , dsp_ds_plaqueta
     , cd_bem
     , dsp_ds_bem
     , qt_instrumentais
     , cd_kit
  From (
Select it_cx_kit.ds_cd_barras
     , NVL(plaqueta_secundaria_bem.ds_plaqueta, bens.Ds_Plaqueta) Dsp_Ds_Plaqueta
     , bens.cd_bem      Cd_Bem
     , bens.ds_bem      Dsp_Ds_Bem
     , Count(Sub_Clas.Cd_Sub_Cla) Qt_Instrumentais
     , Cx_Kit.Cd_Kit              Cd_Kit
  From Dbamv.Cx_Kit
     , Dbamv.It_Cx_Kit
     , Dbamv.Bens
     , Dbamv.plaqueta_secundaria_bem
     , Dbamv.Sub_Clas
 Where Bens.Cd_Sub_Cla     = Sub_Clas.Cd_Sub_Cla
   And Bens.cd_bem         = plaqueta_secundaria_bem.cd_bem(+)
   And Sub_Clas.Cd_Especie = Bens.Cd_Especie
   And Sub_Clas.Cd_Classe  = Bens.Cd_Classe
   And Cx_Kit.Cd_Kit       = It_Cx_Kit.Cd_Kit
   And It_Cx_Kit.Cd_Bem    = Bens.Cd_Bem
   And Cx_Kit.Sn_Ativo     = 'S'
   And It_Cx_Kit.Sn_Ativo  = 'S'
 Group By it_cx_kit.ds_cd_barras
     , Sub_Clas.Ds_Sub_Cla
     , Sub_Clas.Cd_Especie
     , Sub_Clas.Cd_Classe
     , Sub_Clas.Cd_Sub_Cla
     , Cx_Kit.Cd_Kit
     , NVL(plaqueta_secundaria_bem.ds_plaqueta, bens.Ds_Plaqueta)
     , bens.cd_bem
     , bens.ds_bem
Union all
Select it_cx_kit.ds_cd_barras
     , NVL(plaqueta_secundaria_bem.ds_plaqueta, bens.Ds_Plaqueta) Dsp_Ds_Plaqueta
	 , Bens.Cd_Bem         Cd_Bem
     , Bens.Ds_Bem         Dsp_Ds_Bem
     , 1                   Qt_Instrumentais
     , Cx_Kit.Cd_Kit              Cd_Kit
  From Dbamv.Cx_Kit
     , Dbamv.It_Cx_kit
     , Dbamv.Bens
     , Dbamv.plaqueta_secundaria_bem
 Where Cx_Kit.Cd_Kit 	  = It_Cx_Kit.Cd_Kit
   And It_Cx_Kit.Cd_Bem   = Bens.Cd_Bem
   And Bens.cd_bem        = plaqueta_secundaria_bem.cd_bem(+)
   And Cx_Kit.Sn_Ativo    = 'S'
   And It_Cx_Kit.Sn_Ativo = 'S')
WHERE cd_kit = :CAMPO_CD_KIT
