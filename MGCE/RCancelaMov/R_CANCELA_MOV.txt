--R_CANCELA_MOV
select distinct
       Prestador.Nm_Mnemonico
      ,Cx_kit.ds_res_kit DescResumida
      ,Cx_Kit.Cd_Kit CaixaKit
      ,It_Mvto_Cme.Cd_Localidade CodLocalidade
      ,Localidade.Ds_localidade DesLocalidade
      ,to_char(It_Mvto_Cme.Dt_Inc_Usuario,'dd/mm/yyyy') DataEntrada
      ,to_char(It_Mvto_Cme.Dt_Inc_Usuario,'hh24') HoraEntrada
      ,It_Mvto_Cme.Cd_Usuario_Inc UsuarioIncNao
      ,Prestador.Cd_Identificacao UsuarioSim
      ,to_char(It_Mvto_Cme.Dt_Alt_Usuario,'dd/mm/yyyy') DataCancela
      ,to_char(It_Mvto_Cme.Dt_Alt_Usuario,'hh24') HoraCancela
      ,It_Mvto_Cme.Cd_Usuario_Alt UsuarioCancelaNao
      ,Usuarios.Cd_Usuario CodUsu
      ,Usuarios.Nm_Usuario NomeUsu
      ,It_Mvto_Cme.Cd_Ciclo Ciclo
from Dbamv.Cx_Kit
    ,Dbamv.Bens
    ,Dbamv.It_Mvto_Cme
    ,Dbamv.Localidade
    ,Dbamv.Prestador
    ,Dbamv.It_Cx_Kit
    ,Dbasgu.Usuarios
Where Cx_Kit.Cd_Kit = It_Mvto_Cme.Cd_Kit
  and it_cx_kit.cd_bem = Bens.cd_bem
  and Localidade.cd_Localidade = It_Mvto_Cme.Cd_Localidade
  and Prestador.Cd_Prestador(+) = It_mvto_Cme.Cd_Prestador_Identifica
  and It_Mvto_Cme.Cd_Cancelamento is not null
  and it_cx_kit.sn_mestre = 'S'
  and usuarios.cd_usuario = it_mvto_cme.cd_usuario_alt
Order by to_date(DataCancela,'dd/mm/yyyy'), HoraCancela
