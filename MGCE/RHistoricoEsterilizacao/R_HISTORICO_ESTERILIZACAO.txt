--R_HISTORICO_ESTERILIZACAO
SELECT  To_Char(hsc.dh_cadastro,'dd/mm/yyyy hh24') dh_cadastro_formatado,
                  hsc.dh_cadastro,
        us.nm_usuario usuario_alteracao,                                                                          us.cd_usuario cd_usuario_alteracao,
        lo.ds_localidade,
        :CAMPO_CF_TP_SITUACAO
        setor.cd_setor,
        setor.nm_setor,
        usuarios.cd_usuario,
        usuarios.nm_usuario,
       hsc.CD_SOLICITACAO_CME
FROM dbamv.historico_solicitacao_cme hsc
 join dbamv.solicitacao_cme cme on (hsc.cd_solicitacao_cme = cme.cd_solicitacao_cme)
 join dbamv.localidade lo on (hsc.cd_localidade = lo.cd_localidade)
 join dbasgu.usuarios us on (hsc.cd_usuario_cadastro = us.cd_usuario)
 JOIN dbamv.setor ON (cme.cd_setor_entrega=setor.cd_setor)
 JOIN dbasgu.usuarios ON (cme.cd_usuario_entrada= usuarios.cd_usuario)
order by dh_cadastro desc
