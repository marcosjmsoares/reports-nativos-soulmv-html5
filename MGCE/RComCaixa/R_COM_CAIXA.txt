--R_COM_CAIXA
select distinct (cx_kit.cd_kit)
, cx_kit.ds_kit
, cx_kit.ds_res_kit
, cx_kit.dt_val_esterilizacao
, cx_kit.qt_x_esteriliza
, cx_kit.qt_itens
, cx_kit.ds_cd_barras
, prestador.cd_prestador
, prestador.nm_prestador
, tp_embalagem.ds_embalagem
, tp_instrumental.ds_instrumental
FROM dbamv.cx_kit
   , dbamv.prestador
   , dbamv.tp_embalagem
   , dbamv.tp_instrumental
   , dbamv.it_cx_kit
   , dbamv.setor
WHERE cx_kit.cd_prestador = prestador.cd_prestador(+)
AND setor.cd_setor (+) = cx_kit.cd_setor
AND setor.cd_multi_empresa = pkg_mv2000.le_empresa()
AND tp_embalagem.cd_embalagem = cx_kit.cd_embalagem
AND tp_instrumental.cd_tp_instrumental = cx_kit.cd_tp_instrumental
AND cx_kit.cd_kit = it_cx_kit.cd_kit(+)
AND cx_kit.sn_caixa = 'S'
AND cx_kit.sn_ativo in ('N', 'S')
