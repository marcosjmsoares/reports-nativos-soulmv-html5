--R_MOV_DESINFECCAO_ANA_OLD
SELECT * FROM (
select distinct
           It_Mvto_Cme.Cd_Mvto_Cme
          ,It_Mvto_cme.Cd_Cancelamento
         ,cx_kit.cd_kit
         ,cx_kit.ds_res_kit
         ,prestador.cd_identificacao RecebidoPorSim
         ,it_mvto_cme.cd_usuario_inc RecebidoPorNao
         ,TO_CHAR(it_mvto_cme.dt_inc_usuario,'DD/MM/YYYY') DataRecebimento
         ,bens.cd_bem
         ,bens.ds_bem
         ,bens.ds_plaqueta
         ,produto.ds_produto_resumido
         ,Config_Cme.Sn_Identi_Geral
         ,mvto_cme.cd_aviso_cirurgia
         ,mvto_cme.cd_atendimento
         ,nvl(aviso_cirurgia.nm_paciente, paciente.nm_paciente) nm_paciente
   from  dbamv.it_mvto_cme
        ,dbamv.mvto_cme
        ,dbamv.cx_kit
        ,dbamv.prestador
        ,dbamv.bens
        ,dbamv.config_cme
        ,dbamv.tp_instrumental
        ,dbamv.produto
        ,Dbamv.aviso_cirurgia
        ,Dbamv.paciente
  where  tp_instrumental.cd_tp_instrumental = cx_kit.cd_tp_instrumental
    and  It_mvto_Cme.Cd_Mvto_Cme = Mvto_Cme.cd_mvto_cme
    and  IT_MVTO_CME.CD_KIT = CX_KIT.CD_KIT
    And  aviso_cirurgia.cd_paciente          = paciente.cd_paciente(+)
    And  aviso_cirurgia.cd_aviso_cirurgia (+)= mvto_cme.cd_aviso_cirurgia
    and  Prestador.Cd_Prestador(+) = It_mvto_Cme.Cd_Prestador_Identifica
    and  Bens.Cd_Bem (+)= It_Mvto_Cme.Cd_Bem
    and  Config_Cme.Cd_Setor_Desinfeccao = It_Mvto_Cme.Cd_Setor
    and  Config_Cme.Cd_Localidade_Desinfeccao = It_Mvto_Cme.Cd_Localidade
    and  Produto.Cd_Produto(+) = Cx_Kit.Cd_Produto
    and trunc(it_mvto_cme.dt_inc_usuario) BETWEEN :CAMPO_DATA_INI AND 
    and (config_cme.cd_setor_principal in (select cd_setor
                                                                   from dbamv.setor
                                                                where cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA)
 or config_cme.cd_setor_principal is null)
order by cx_kit.cd_kit
)
ORDER BY CD_KIT,DS_RES_KIT,RECEBIDOPORSIM,RECEBIDOPORNAO,DATARECEBIMENTO,DS_PRODUTO_RESUMIDO,CD_MVTO_CME,SN_IDENTI_GERAL
