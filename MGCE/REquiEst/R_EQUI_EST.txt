SELECT ALL MAQ_ESTERILIZADOR.CD_ESTERILIZADOR,
BENS.DS_BEM,
decode (MAQ_ESTERILIZADOR.SN_ATIVO, 'S', :CAMPO_PARAMETRO_SIM, 'N', :CAMPO_PARAMETRO_NAO) SN_ATIVO
FROM DBAMV.MAQ_ESTERILIZADOR, DBAMV.BENS
WHERE (MAQ_ESTERILIZADOR.CD_BEM = BENS.CD_BEM)
