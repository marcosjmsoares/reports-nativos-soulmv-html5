      ,Cx_Kit.Ds_Res_Kit
      ,It_Mvto_Cme.Dt_Inc_Usuario
      ,Prestador.Nm_Prestador
      ,Produto.Cd_Produto
      ,Prestador.Cd_Identificacao
union all
Select '5' Sequencia
      ,'Arsenal' Fase
      ,nvl(It_Mvto_Cme.Cd_Ciclo,0) cd_ciclo
      ,Cx_Kit.Ds_Res_Kit
      ,It_Mvto_Cme.Dt_Inc_Usuario
      ,Prestador.Nm_Prestador
      ,Produto.Cd_Produto
      ,Prestador.Cd_Identificacao
From Dbamv.It_Mvto_Cme
    ,Dbamv.Cx_kit
    ,Dbamv.Config_Cme
    ,Dbamv.Prestador
    ,Dbamv.Produto
    ,dbamv.setor
Where It_Mvto_Cme.Cd_Kit = Cx_kit.Cd_kit
  and It_Mvto_Cme.Cd_Localidade = Config_Cme.Cd_Localidade_Arsenal
  and It_Mvto_Cme.Cd_Setor = Config_Cme.Cd_Setor_Arsenal
  and setor.cd_setor = config_cme.cd_setor_principal
  and setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
  and It_Mvto_Cme.Cd_Prestador_Identifica = Prestador.Cd_Prestador(+)
  and Cx_Kit.Cd_Produto = Produto.Cd_Produto(+)
  and It_Mvto_Cme.Cd_Cancelamento is null
  and Prestador.Nm_Prestador = :CAMPO_NM_PRESTADOR
group by
     It_Mvto_Cme.Cd_Ciclo
      ,Cx_Kit.Ds_Res_Kit
      ,It_Mvto_Cme.Dt_Inc_Usuario
      ,Prestador.Nm_Prestador
      ,Produto.Cd_Produto
      ,Prestador.Cd_Identificacao
union all
Select '6' Sequencia
      ,'Externo' Fase
      ,nvl(It_Mvto_Cme.Cd_Ciclo,0) cd_ciclo
      ,Cx_Kit.Ds_Res_Kit
      ,It_Mvto_Cme.Dt_Inc_Usuario
      ,Prestador.Nm_Prestador
      ,Produto.Cd_Produto
      ,Prestador.Cd_Identificacao
From Dbamv.It_Mvto_Cme
    ,Dbamv.Cx_kit
    ,Dbamv.Config_Cme
    ,Dbamv.Prestador
    ,Dbamv.Produto
    ,dbamv.setor
Where It_Mvto_Cme.Cd_Kit = Cx_kit.Cd_kit
  and It_Mvto_Cme.Cd_Localidade = Config_Cme.Cd_Localidade_Externo
  and It_Mvto_Cme.Cd_Setor = Config_Cme.Cd_Setor_Externo
  and setor.cd_setor = config_cme.cd_setor_principal
  and setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
  and It_Mvto_Cme.Cd_Prestador_Identifica = Prestador.Cd_Prestador(+)
  and Cx_Kit.Cd_Produto = Produto.Cd_Produto(+)
  and It_Mvto_Cme.Cd_Cancelamento is null
  and Prestador.Nm_Prestador = :CAMPO_NM_PRESTADOR
group by
     It_Mvto_Cme.Cd_Ciclo
      ,Cx_Kit.Ds_Res_Kit
      ,It_Mvto_Cme.Dt_Inc_Usuario
      ,Prestador.Nm_Prestador
     ,Produto.Cd_Produto
      ,Prestador.Cd_Identificacao
)
group by Sequencia
      ,Fase
      ,Cd_Ciclo
      ,Ds_Res_Kit
      ,Dt_Inc_Usuario
      ,Nm_Prestador
      ,Cd_Produto
      ,Cd_Identificacao
Order by Sequencia
)
ORDER BY NM_PRESTADOR,CD_IDENTIFICACAO,
CD_CICLO,DS_RES_KIT,
SEQUENCIA,FASE
