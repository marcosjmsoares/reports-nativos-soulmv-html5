--R_COM_KIT
SELECT * FROM (
select cx_kit.cd_kit, cx_kit.ds_cd_barras, cx_kit.ds_kit, cx_kit.ds_res_kit,
cx_kit.dt_val_esterilizacao, cx_kit.qt_x_esteriliza, cx_kit.qt_itens,
prestador.cd_prestador, prestador.nm_prestador,
tp_embalagem.ds_embalagem, tp_instrumental.ds_instrumental,
bens.cd_bem, bens.ds_bem, bens.ds_plaqueta,
decode(it_cx_kit.sn_ativo, 'S',  :CAMPO_PARAMETRO_SIM, 'N', :CAMPO_PARAMETRO_NAO) sn_ativo,
decode(it_cx_kit.sn_suspenso, 'S',  'Sim', 'N', 'Não') sn_suspenso,
it_cx_kit.qt_item it_qt_item,
it_cx_kit.ds_cd_barras it_ds_cd_barras
from dbamv.cx_kit, dbamv.prestador, dbamv.tp_embalagem, dbamv.tp_instrumental,
dbamv.bens, dbamv.it_cx_kit
where cx_kit.cd_prestador = prestador.cd_prestador(+)
and tp_embalagem.cd_embalagem = cx_kit.cd_embalagem
and tp_instrumental.cd_tp_instrumental = cx_kit.cd_tp_instrumental
and bens.cd_bem(+) = it_cx_kit.cd_bem
and cx_kit.cd_kit = it_cx_kit.cd_kit(+)
and cx_kit.sn_caixa = 'N'
AND cx_kit.cd_setor IN (SELECT cd_setor FROM dbamv.setor WHERE cd_multi_empresa = dbamv.pkg_mv2000.le_empresa())
:CAMPO_CF_CX_KIT
)
ORDER BY :CAMPO_CF_ORDEM,CD_KIT,DS_KIT,DS_RES_KIT,CD_PRESTADOR,DS_EMBALAGEM,NM_PRESTADOR,QT_ITENS,DS_INSTRUMENTAL,DT_VAL_ESTERILIZACAO,QT_X_ESTERILIZA
