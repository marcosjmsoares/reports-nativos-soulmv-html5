--R_ESTERILIZACAO_CLIENTE_EXTERNO
select   item_avulso_solicitacao_cme.cd_fornecedor
       , fornecedor.nm_fornecedor
       , item_avulso_solicitacao_cme.cd_bem_terceiro
       , bem_terceiro_cme.ds_bem_terceiro
       , item_avulso_solicitacao_cme.qt_item_avulso
       , esterilizacao.cd_esterilizador
       , bens.ds_bem
       , esterilizacao.dt_inic_esteriliza
       , esterilizacao.dt_fim_esteriliza
  FROM   dbamv.item_avulso_solicitacao_cme
       , dbamv.esterilizacao
       , dbamv.bem_terceiro_cme
       , dbamv.fornecedor
       , dbamv.maq_esterilizador
       , dbamv.bens
 WHERE item_avulso_solicitacao_cme.cd_esteriliza = esterilizacao.cd_esteriliza
   AND item_avulso_solicitacao_cme.cd_bem_terceiro = bem_terceiro_cme.cd_bem_terceiro
   AND item_avulso_solicitacao_cme.cd_fornecedor = fornecedor.cd_fornecedor
   AND esterilizacao.cd_esterilizador = maq_esterilizador.cd_esterilizador
   AND maq_esterilizador.cd_bem = bens.cd_bem
   AND item_avulso_solicitacao_cme.cd_esteriliza IS NOT NULL
   AND item_avulso_solicitacao_cme.cd_bem_terceiro IS NOT NULL
   AND item_avulso_solicitacao_cme.cd_fornecedor IS NOT NULL
   AND item_avulso_solicitacao_cme.tp_situacao IN (7, 71, 8)
   AND trunc(esterilizacao.dt_fim_esteriliza) BETWEEN :CAMPO_DATA_INI order by fornecedor.nm_fornecedor, esterilizacao.dt_fim_esteriliza
