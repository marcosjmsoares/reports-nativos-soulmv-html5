--R_VALIDADE_MATERIAL
Select CD_KIT,
       DS_RES_KIT,
       DT_VALIDADE_ESTERILIZACAO,
       DT_INIC_ESTERILIZA
  From (Select CX_KIT.CD_KIT,
               CX_KIT.DS_RES_KIT,
               To_Char(COMPOSICAO_SOLICITACAO_CME.DT_VALIDADE_ESTERILIZACAO,
                       'DD/MM/YYYY') As DT_VALIDADE_ESTERILIZACAO,
               To_Char(ESTERILIZACAO.DT_INIC_ESTERILIZA,
                       'DD/MM/YYYY') As DT_INIC_ESTERILIZA
          From DBAMV.SOLICITACAO_CME
    Inner Join DBAMV.COMPOSICAO_SOLICITACAO_CME On
               COMPOSICAO_SOLICITACAO_CME.CD_SOLICITACAO_CME = SOLICITACAO_CME.CD_SOLICITACAO_CME
    Inner Join DBAMV.ESTERILIZACAO On
               ESTERILIZACAO.CD_ESTERILIZA = COMPOSICAO_SOLICITACAO_CME.CD_ESTERILIZA
    Inner Join DBAMV.CX_KIT On
               CX_KIT.CD_KIT = COMPOSICAO_SOLICITACAO_CME.CD_KIT
         Where SOLICITACAO_CME.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.Le_Empresa()
           And SOLICITACAO_CME.TP_SITUACAO In (61,
                                               7)
     Union All
        Select Distinct CX_KIT.CD_KIT,
                        CX_KIT.DS_RES_KIT,
                        To_Char(IT_MVTO_CME.DT_VALID_ESTERILIZA,
                                'DD/MM/YYYY') As DT_VALIDADE_ESTERILIZACAO,
                        To_Char(ESTERILIZACAO.DT_INIC_ESTERILIZA,
                                'DD/MM/YYYY') As DT_INIC_ESTERILIZA
          From DBAMV.MVTO_CME
    Inner Join DBAMV.IT_MVTO_CME On
               IT_MVTO_CME.CD_MVTO_CME = MVTO_CME.CD_MVTO_CME
    Inner Join DBAMV.ESTERILIZACAO On
               ESTERILIZACAO.CD_ESTERILIZA = IT_MVTO_CME.CD_ESTERILIZA
    Inner Join DBAMV.CX_KIT On
               IT_MVTO_CME.CD_KIT = CX_KIT.CD_KIT)
 Where 1 = 1
Order By 3
