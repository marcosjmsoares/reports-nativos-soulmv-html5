--R_MVTO_ARSENAL_SAI
SELECT * FROM (
Select distinct
       it_mvto_cme.cd_mvto_cme,
       it_mvto_cme.cd_setor,
       it_mvto_cme.cd_kit,
       it_mvto_cme.cd_localidade,
       localidade.ds_localidade,
       setor.nm_setor,
       prestador.cd_identificacao,
       prestador.nm_prestador,
       cx_kit.ds_kit,
       'NÃO' avulso
       ,mvto_cme.cd_atendimento
       ,mvto_cme.cd_aviso_cirurgia
       ,nvl(aviso_cirurgia.nm_paciente, paciente.nm_paciente) nm_paciente
  From dbamv.it_mvto_cme,
       dbamv.localidade,
       dbamv.setor,
       dbamv.prestador,
       dbamv.cx_kit,
       dbamv.mvto_cme,
       Dbamv.aviso_cirurgia,
       Dbamv.paciente
 Where it_mvto_cme.cd_localidade = localidade.cd_localidade
   and it_mvto_cme.cd_setor = setor.cd_setor
   and it_mvto_cme.cd_prestador_identifica = prestador.cd_prestador(+)
   and it_mvto_cme.cd_kit = cx_kit.cd_kit
   and cx_kit.sn_ativo = 'S'
   and it_mvto_cme.sn_final_fase = 'S'
   AND it_mvto_cme.cd_kit IS NOT null
   and It_mvto_Cme.Cd_Mvto_Cme = Mvto_Cme.cd_mvto_cme
   And aviso_cirurgia.cd_paciente          = paciente.cd_paciente(+)
   And aviso_cirurgia.cd_aviso_cirurgia (+)= mvto_cme.cd_aviso_cirurgia
 UNION
 Select distinct
       it_mvto_cme.cd_mvto_cme,
       it_mvto_cme.cd_setor,
       it_mvto_cme.cd_bem cd_kit,
       it_mvto_cme.cd_localidade,
       localidade.ds_localidade,
       setor.nm_setor,
       prestador.cd_identificacao,
       prestador.nm_prestador,
       bens.ds_bem ds_kit,
	'SIM' avulso,
       mvto_cme.cd_atendimento,
       mvto_cme.cd_aviso_cirurgia,
       nvl(aviso_cirurgia.nm_paciente, paciente.nm_paciente) nm_paciente
  From dbamv.it_mvto_cme,
       dbamv.localidade,
       dbamv.setor,
       dbamv.prestador,
       dbamv.bens,
       dbamv.mvto_cme,
       Dbamv.aviso_cirurgia,
       Dbamv.paciente
 Where it_mvto_cme.cd_localidade = localidade.cd_localidade
   and it_mvto_cme.cd_setor = setor.cd_setor
   and it_mvto_cme.cd_prestador_identifica = prestador.cd_prestador(+)
   and it_mvto_cme.cd_bem = bens.cd_bem
   and it_mvto_cme.sn_final_fase = 'S'
   AND it_mvto_cme.cd_kit IS NULL
   and It_mvto_Cme.Cd_Mvto_Cme = Mvto_Cme.cd_mvto_cme
   And aviso_cirurgia.cd_paciente          = paciente.cd_paciente(+)
   And aviso_cirurgia.cd_aviso_cirurgia (+)= mvto_cme.cd_aviso_cirurgia
   :CAMPO_CF_MOVIMENTACAO
)
ORDER BY CD_MVTO_CME,CD_SETOR,NM_SETOR,CD_LOCALIDADE,DS_LOCALIDADE,NM_PRESTADOR,CD_IDENTIFICACAO
