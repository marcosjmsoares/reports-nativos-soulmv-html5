--R_PONTOS_ESTERILIZACAO
SELECT  est.cd_esterilizador AS CD_MAQ
      , bem.DS_BEM    AS DS_BEM
      , Sum(Nvl(est.NR_PONTOS,0)) AS NR_PONTOS
  FROM  Dbamv.Maq_Esterilizador maq
      , Dbamv.Bens              bem
      , Dbamv.esterilizacao     est
   WHERE maq.Cd_Bem = bem.Cd_Bem
   AND est.cd_esterilizador = maq.Cd_esterilizador
   AND   trunc(est.dt_fim_esteriliza) BETWEEN :CAMPO_DATA_INI GROUP BY  est.cd_esterilizador
           , bem.ds_bem
