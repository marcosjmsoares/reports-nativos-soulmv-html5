SELECT * FROM (
select distinct
         solicitacao_cme.cd_solicitacao_cme
        ,solicitacao_cme.cd_cancelamento
        ,solicitacao_cme.cd_usuario_entrada RecebidoPorSim
        ,solicitacao_cme.cd_usuario_cadastro RecebidoPorNao
        ,to_char(solicitacao_cme.dh_cadastro,'DD/MM/YYYY') DataRecebimento
        ,solicitacao_cme.cd_aviso_cirurgia
        ,solicitacao_cme.cd_atendimento
        ,nvl(aviso_cirurgia.nm_paciente, paciente.nm_paciente) nm_paciente
        ,(SELECT Config_Cme.Sn_Identi_Geral FROM dbamv.Config_Cme WHERE cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA AND cd_setor_preparo = setor.cd_setor) Sn_Identi_Geral
from dbamv.solicitacao_cme
	 ,dbamv.composicao_solicitacao_cme
	 ,dbamv.item_avulso_solicitacao_cme
    ,dbamv.aviso_cirurgia
    ,dbamv.paciente
    ,dbamv.setor
WHERE 
   solicitacao_cme.cd_setor_principal = setor.cd_setor
   AND solicitacao_cme.cd_aviso_cirurgia = aviso_cirurgia.cd_aviso_cirurgia (+)
   AND aviso_cirurgia.cd_paciente = paciente.cd_paciente(+)
   AND solicitacao_cme.cd_solicitacao_cme = item_avulso_solicitacao_cme.cd_solicitacao_cme(+)
   AND solicitacao_cme.cd_solicitacao_cme = composicao_solicitacao_cme.cd_solicitacao_cme(+)
   AND trunc(solicitacao_cme.dh_cadastro) BETWEEN  :CAMPO_DATA_INI AND 
   AND setor.cd_multi_empresa =  :CAMPO_CD_MULTI_EMPRESA
   AND solicitacao_cme.tp_situacao >= 4
Order by solicitacao_cme.cd_solicitacao_cme
)
