--R_MOV_SETOR_AVISO
SELECT
						*
					FROM
						(
						SELECT
							MVTO_CME.CD_MVTO_CME CD_MVTO_CME ,
							ESTERILIZACAO.CD_ESTERILIZADOR CD_ESTERILIZADOR ,
							DECODE(Esterilizacao.SN_BIOLOGICO, 'S', 'Conforme', 'N', 'Não Conforme', ' ') sn_biologico ,
							DECODE(Esterilizacao.SN_QUIMICO, 'S', 'Conforme', 'N', 'Não Conforme', ' ') sn_quimico ,
							MVTO_CME.CD_SETOR_PRINCIPAL CD_SETOR_PRINCIPAL ,
							MVTO_CME.CD_AVISO_CIRURGIA CD_AVISO_CIRURGIA ,
							AVISO_CIRURGIA.NM_PACIENTE NM_PACIENTE ,
							PACIENTE.NM_SOCIAL_PACIENTE NM_SOCIAL_PACIENTE ,
							PACIENTE.DT_NASCIMENTO DT_NASCIMENTO ,
							BENS.DS_BEM DS_BEM ,
							Nvl(IT_MVTO_CME.CD_KIT, Nvl(IT_MVTO_CME.CD_BEM, Nvl(IT_MVTO_CME.CD_BEM_TERCEIRO, Nvl(IT_MVTO_CME.CD_PRODUTO_ROUPARIA, IT_MVTO_CME.CD_PRODUTO_REPROCESSADO))))   CD_KIT ,
							CASE 
								WHEN IT_MVTO_CME.CD_KIT IS NOT NULL THEN
									(SELECT CX_KIT.DS_KIT FROM dbamv.CX_KIT WHERE CX_KIT.CD_KIT = IT_MVTO_CME.CD_KIT)
								WHEN IT_MVTO_CME.CD_BEM IS NOT NULL THEN
									(SELECT BENS.DS_BEM FROM dbamv.BENS WHERE BENS.CD_BEM = IT_MVTO_CME.CD_BEM)
								WHEN IT_MVTO_CME.CD_BEM_TERCEIRO IS NOT NULL THEN
									(SELECT BEM_TERCEIRO_CME.DS_BEM_TERCEIRO FROM dbamv.BEM_TERCEIRO_CME WHERE BEM_TERCEIRO_CME.CD_BEM_TERCEIRO = IT_MVTO_CME.CD_BEM_TERCEIRO)
								WHEN IT_MVTO_CME.CD_PRODUTO_ROUPARIA IS NOT NULL THEN
									(SELECT Nvl(PRODUTO.DS_PRODUTO_RESUMIDO, PRODUTO.DS_PRODUTO) FROM dbamv.PRODUTO WHERE PRODUTO.CD_PRODUTO = IT_MVTO_CME.CD_PRODUTO_ROUPARIA)
								WHEN IT_MVTO_CME.CD_PRODUTO_REPROCESSADO IS NOT NULL THEN
									(SELECT Nvl(PRODUTO.DS_PRODUTO_RESUMIDO, PRODUTO.DS_PRODUTO) FROM dbamv.PRODUTO WHERE PRODUTO.CD_PRODUTO = IT_MVTO_CME.CD_PRODUTO_REPROCESSADO)
							END  DSP_DS_KIT,
							IT_MVTO_CME.Dt_Valid_Esteriliza DT_VALID_ESTERILIZA,
							IT_MVTO_CME.CD_ESTERILIZA CD_CICLO ,
							IT_MVTO_CME.CD_CANCELAMENTO CD_CANCELAMENTO ,
							CANCELAMENTO.DS_CANCELAMENTO DS_CANCELAMENTO ,
							MVTO_CME.CD_SETOR_ORIGEM CD_SETOR_ORIGEM
						FROM
							DBAMV.MVTO_CME ,
							DBAMV.IT_MVTO_CME ,
							DBAMV.SETOR ,
							DBAMV.ESTERILIZACAO ,
							DBAMV.MAQ_ESTERILIZADOR ,
							DBAMV.BENS ,
							DBAMV.CANCELAMENTO ,
							DBAMV.AVISO_CIRURGIA ,
							DBAMV.PACIENTE
						WHERE
							MVTO_CME.CD_MVTO_CME = IT_MVTO_CME.CD_MVTO_CME
							AND SETOR.CD_SETOR(+) = MVTO_CME.CD_SETOR_PRINCIPAL
							AND IT_MVTO_CME.CD_ESTERILIZA = ESTERILIZACAO.CD_ESTERILIZA(+)
							AND IT_MVTO_CME.CD_CANCELAMENTO = CANCELAMENTO.CD_CANCELAMENTO(+)
							AND ESTERILIZACAO.CD_ESTERILIZADOR = MAQ_ESTERILIZADOR.CD_ESTERILIZADOR(+)
							AND MAQ_ESTERILIZADOR.CD_BEM = BENS.CD_BEM(+)
							AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA(+) = MVTO_CME.CD_AVISO_CIRURGIA
							AND AVISO_CIRURGIA.CD_PACIENTE = PACIENTE.CD_PACIENTE(+)
							AND TRUNC(MVTO_CME.DT_INC_USUARIO) BETWEEN :CAMPO_DATA_INI AND 
							AND ( ('S' = :CAMPO_SN_CANCELADO
							AND IT_MVTO_CME.CD_CANCELAMENTO IS NOT NULL )
							OR ('N' = :CAMPO_SN_CANCELADO
							AND IT_MVTO_CME.CD_CANCELAMENTO IS NULL ) )) A
					GROUP BY
						CD_MVTO_CME,
						CD_ESTERILIZADOR,
						SN_BIOLOGICO,
						SN_QUIMICO,
						CD_SETOR_PRINCIPAL,
						CD_AVISO_CIRURGIA,
						NM_PACIENTE,
						NM_SOCIAL_PACIENTE,
						DT_NASCIMENTO,
						DS_BEM,
						CD_KIT,
						DSP_DS_KIT,
						DT_VALID_ESTERILIZA,
						CD_CICLO,
						CD_CANCELAMENTO,
						DS_CANCELAMENTO,
						CD_SETOR_ORIGEM
					ORDER BY
						CD_SETOR_PRINCIPAL,
						CD_ESTERILIZADOR,
						CD_SETOR_ORIGEM,
						CD_AVISO_CIRURGIA,
						NM_PACIENTE,
						NM_SOCIAL_PACIENTE,
						DT_NASCIMENTO,
						CD_MVTO_CME,
						CD_KIT
