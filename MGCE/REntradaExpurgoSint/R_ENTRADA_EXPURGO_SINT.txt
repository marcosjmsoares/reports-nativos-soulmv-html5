--R_ENTRADA_EXPURGO_SINT
Select Distinct SOLICITACAO_CME.CD_SOLICITACAO_CME,
                SOLICITACAO_CME.CD_CANCELAMENTO,
                SOLICITACAO_CME.CD_USUARIO_ENTRADA As RECEBIDOPORSIM,
                SOLICITACAO_CME.CD_USUARIO_CADASTRO As RECEBIDOPORNAO,
                To_Char(SOLICITACAO_CME.DH_CADASTRO,
                        'DD/MM/YYYY') As DATARECEBIMENTO,
                SOLICITACAO_CME.CD_AVISO_CIRURGIA,
                SOLICITACAO_CME.CD_ATENDIMENTO,
                Nvl(AVISO_CIRURGIA.NM_PACIENTE,
                    PACIENTE.NM_PACIENTE) As NM_PACIENTE,
                PACIENTE.NM_SOCIAL_PACIENTE,
                PACIENTE.DT_NASCIMENTO,
                (Select CONFIG_CME.SN_IDENTI_GERAL
                   From DBAMV.CONFIG_CME
                  Where CONFIG_CME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
                    And CONFIG_CME.CD_SETOR_PRINCIPAL = SOLICITACAO_CME.CD_SETOR_PRINCIPAL) As SN_IDENTI_GERAL,
                Nvl((Select Sum(CSCME.QT_KIT_CAIXA)
                       From DBAMV.COMPOSICAO_SOLICITACAO_CME CSCME
                      Where CSCME.CD_SOLICITACAO_CME = SOLICITACAO_CME.CD_SOLICITACAO_CME),
                    0) As QT_KIT,
                Nvl((Select Sum(IASCME.QT_ITEM_AVULSO)
                       From DBAMV.ITEM_AVULSO_SOLICITACAO_CME IASCME
                      Where IASCME.CD_SOLICITACAO_CME = SOLICITACAO_CME.CD_SOLICITACAO_CME
                        And IASCME.CD_BEM Is Not Null),
                    0) As QT_BEM,
                Nvl((Select Sum(IASCME.QT_ITEM_AVULSO)
                       From DBAMV.ITEM_AVULSO_SOLICITACAO_CME IASCME
                      Where IASCME.CD_SOLICITACAO_CME = SOLICITACAO_CME.CD_SOLICITACAO_CME
                        And IASCME.CD_BEM_TERCEIRO Is Not Null),
                    0) As QT_BEM_TERCEIRO,
                Nvl((Select Sum(IASCME.QT_ITEM_AVULSO)
                       From DBAMV.ITEM_AVULSO_SOLICITACAO_CME IASCME
                      Where IASCME.CD_SOLICITACAO_CME = SOLICITACAO_CME.CD_SOLICITACAO_CME
                        And IASCME.CD_PRODUTO_REPROCESSADO Is Not Null),
                    0) As QT_PROD_REPRO
  From DBAMV.SOLICITACAO_CME
Inner Join DBAMV.SETOR On
       SETOR.CD_SETOR = SOLICITACAO_CME.CD_SETOR_PRINCIPAL
   And SETOR.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
Left Join DBAMV.COMPOSICAO_SOLICITACAO_CME On
       COMPOSICAO_SOLICITACAO_CME.CD_SOLICITACAO_CME = SOLICITACAO_CME.CD_SOLICITACAO_CME
Left Join DBAMV.ITEM_AVULSO_SOLICITACAO_CME On
       ITEM_AVULSO_SOLICITACAO_CME.CD_SOLICITACAO_CME = SOLICITACAO_CME.CD_SOLICITACAO_CME
Left Join DBAMV.AVISO_CIRURGIA On
       AVISO_CIRURGIA.CD_AVISO_CIRURGIA = SOLICITACAO_CME.CD_AVISO_CIRURGIA
Left Join DBAMV.PACIENTE On
       PACIENTE.CD_PACIENTE = AVISO_CIRURGIA.CD_PACIENTE
 Where SOLICITACAO_CME.DH_CADASTRO Between :CAMPO_DATA_INI
                                       And :CAMPO_DATA_FIM + 0.99999
   And SOLICITACAO_CME.TP_SITUACAO >= 1
Order By SOLICITACAO_CME.CD_SOLICITACAO_CME
