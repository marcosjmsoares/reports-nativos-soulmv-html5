--R_LISTA_SOLIC_ESTERILIZACAO
select    solic.cd_solicitacao_cme
        , decode(solic.tp_situacao
	      	,'0', :CAMPO_SOLICITADO
                	,'1', :CAMPO_REC_EXPURGO
	      	,'2', :CAMPO_LAVAGEM
		,'3', :CAMPO_DESINFECTACAO
		,'4', :CAMPO_PREPARO
		,'5', :CAMPO_AGUARDANDO_EST
		,'51', :CAMPO_REC_EST
		,'6', :CAMPO_TESTE
		,'61', :CAMPO_ESTERELIZADO
		,'7', :CAMPO_DISPONIVEL
		,'71', :CAMPO_REC_ARSENAL
		,'8', :CAMPO_LIBERADO
		,'99', :CAMPO_CANCELADA) tp_situacao
        , solic.dh_cadastro
        , (SELECT cd_setor || ' - ' || nm_setor FROM dbamv.setor WHERE  cd_setor = solic.cd_setor_responsavel_entrada) AS  cd_setor_responsavel_entrada
        , (SELECT usu.nm_usuario FROM dbasgu.usuarios usu WHERE cd_usuario = solic.cd_usuario_entrada) AS cd_usuario_entrada
        , (SELECT cd_setor || ' - ' || nm_setor FROM dbamv.setor WHERE  cd_setor = solic.cd_setor_entrega) AS cd_setor_entrega
        , (SELECT usu.nm_usuario FROM dbasgu.usuarios usu WHERE cd_usuario = solic.cd_usuario_entrega) AS cd_usuario_entrega
        , solic.ds_observacao
        , (SELECT distinct est.cd_estoque || ' - ' || est.ds_estoque FROM dbamv.estoque est WHERE est.cd_estoque = solic.cd_estoque_rouparia) AS CD_ESTOQUE_ROUPARIA
        , solic.dh_cadastro
    from    dbamv.solicitacao_cme solic
      where  solic.cd_solicitacao_cme =  :CAMPO_CD_SOLICITACAO_CME
