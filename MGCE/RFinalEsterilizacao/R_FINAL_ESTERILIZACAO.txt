--R_FINAL_ESTERILIZACAO
Select Distinct It_Mvto_Cme.Cd_Kit 				 	 		 Cd_Kit
                       ,Cx_Kit.Ds_Kit     				 	 		 Dsp_Ds_Kit
	       ,It_Mvto_Cme.Cd_Esteriliza 	 		                                                 Cd_Esteriliza
	       ,Cx_Kit.Cd_Prestador				 		                 Cd_Prestador
	       ,Cx_Kit.Dt_Val_Esterilizacao 		                                                                 Dt_Val_Esterilizacao
	       ,NVL(It_Mvto_Cme.Qt_Kit_Caixa,1)                  Qt_Itens
  From Dbamv.Mvto_Cme
         ,Dbamv.It_Mvto_Cme
         ,Dbamv.Cx_Kit
         ,Dbamv.Setor
         ,dbamv.esterilizacao
         ,(Select Cd_Kit,COUNT(*) Qtd
             From Dbamv.It_Cx_Kit
           group by Cd_Kit)                   Qt_Itens_Kit
Where Mvto_Cme.Cd_Mvto_Cme   = It_Mvto_Cme.Cd_Mvto_Cme
    And It_Mvto_Cme.Cd_Kit            = Cx_Kit.Cd_Kit
    And Qt_Itens_Kit.Cd_kit(+)          = Cx_Kit.Cd_Kit
    And Setor.Cd_Setor(+)                = Mvto_Cme.Cd_Setor_Principal
    And  esterilizacao.cd_esteriliza = it_mvto_cme.cd_esteriliza
    And  esterilizacao.Sn_Est_Completa = 'S'
order by It_Mvto_Cme.Cd_Esteriliza
