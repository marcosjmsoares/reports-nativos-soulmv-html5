--R_INST_INATIVO
Select CD_BEM,
       DS_BEM,
       DS_PLAQUETA,
       CD_KIT,
       DS_KIT,
       SN_ATIVO,
       SN_ATIVO1
  From (Select BENS.CD_BEM,
               BENS.DS_BEM,
               BENS.DS_PLAQUETA,
               CX_KIT.CD_KIT,
               CX_KIT.DS_KIT,
               Decode(CX_KIT.SN_ATIVO,
                      'S',
                      :CAMPO_PARAMETRO_SIM,
                      'N',
                      :CAMPO_PARAMETRO_NAO) As SN_ATIVO,
               Decode(IT_CX_KIT.SN_ATIVO,
                      'S',
                      :CAMPO_PARAMETRO_SIM,
                      'N',
                      :CAMPO_PARAMETRO_NAO) As SN_ATIVO1
          From DBAMV.BENS
    Inner Join DBAMV.IT_CX_KIT On
               IT_CX_KIT.CD_BEM = BENS.CD_BEM
    Inner Join DBAMV.CX_KIT On
               CX_KIT.CD_KIT = IT_CX_KIT.CD_KIT
         Where 1 = 1
           :CAMPO_CF_BENS)
