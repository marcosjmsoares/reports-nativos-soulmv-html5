--R_PROD_FASE
SELECT * FROM (
Select Sequencia
      ,Fase
      ,Cd_Ciclo
      ,Ds_Res_Kit
      ,Dt_Inc_Usuario
      ,Cd_Produto
      ,Sn_Caixa
From (
Select It_Mvto_Cme.Dt_Inc_Usuario
      ,'1' Sequencia
      ,'Expurgo' Fase
      ,It_Mvto_Cme.Cd_Ciclo
      ,Cx_Kit.Ds_Res_Kit
      ,Produto.Cd_Produto
      ,Cx_Kit.Sn_Caixa
From Dbamv.It_Mvto_Cme
    ,Dbamv.Cx_kit
    ,Dbamv.Config_Cme
    ,Dbamv.Produto
    ,dbamv.setor
Where It_Mvto_Cme.Cd_Kit = Cx_kit.Cd_kit
  and It_Mvto_Cme.Cd_Localidade = Config_Cme.Cd_Localidade_Expurgo
  and It_Mvto_Cme.Cd_Setor = Config_Cme.Cd_Setor_Expurgo
  and setor.cd_setor = config_cme.cd_setor_principal
  and setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
  and Cx_Kit.Cd_Produto = Produto.Cd_Produto(+)
  and It_Mvto_Cme.Cd_Ciclo is not null
  and It_Mvto_Cme.Cd_Cancelamento is null
group by
     It_Mvto_Cme.Cd_Ciclo
      ,Cx_Kit.Ds_Res_Kit
      ,It_Mvto_Cme.Dt_Inc_Usuario
      ,Produto.Cd_Produto
      ,Cx_Kit.Sn_Caixa
union all
Select It_Mvto_Cme.Dt_Inc_Usuario
      ,'2' Sequencia
      ,'Desinfeccao' Fase
      ,It_Mvto_Cme.Cd_Ciclo
      ,Cx_Kit.Ds_Res_Kit
      ,Produto.Cd_Produto
      ,Cx_Kit.Sn_Caixa
From Dbamv.It_Mvto_Cme
    ,Dbamv.Cx_kit
    ,Dbamv.Config_Cme
    ,Dbamv.Produto
    ,dbamv.setor
Where It_Mvto_Cme.Cd_Kit = Cx_kit.Cd_kit
  and It_Mvto_Cme.Cd_Localidade = Config_Cme.Cd_Localidade_Desinfeccao
  and It_Mvto_Cme.Cd_Setor = Config_Cme.Cd_Setor_Desinfeccao
  and setor.cd_setor = config_cme.cd_setor_principal
  and setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
  and Cx_Kit.Cd_Produto = Produto.Cd_Produto(+)
  and It_Mvto_Cme.Cd_Ciclo is not null
  and It_Mvto_Cme.Cd_Cancelamento is null
group by
     It_Mvto_Cme.Cd_Ciclo
      ,Cx_Kit.Ds_Res_Kit
      ,It_Mvto_Cme.Dt_Inc_Usuario
      ,Produto.Cd_Produto
      ,Cx_Kit.Sn_Caixa
union all
Select It_Mvto_Cme.Dt_Inc_Usuario
      ,'3' Sequencia
      ,'Preparo' Fase
      ,It_Mvto_Cme.Cd_Ciclo
      ,Cx_Kit.Ds_Res_Kit
      ,Produto.Cd_Produto
      ,Cx_Kit.Sn_Caixa
From Dbamv.It_Mvto_Cme
    ,Dbamv.Cx_kit
    ,Dbamv.Config_Cme
    ,Dbamv.Produto
    ,dbamv.setor
Where It_Mvto_Cme.Cd_Kit = Cx_kit.Cd_kit
  and It_Mvto_Cme.Cd_Localidade = Config_Cme.Cd_Localidade_Preparo
  and It_Mvto_Cme.Cd_Setor = Config_Cme.Cd_Setor_Preparo
  and setor.cd_setor = config_cme.cd_setor_principal
  and setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
  and Cx_Kit.Cd_Produto = Produto.Cd_Produto(+)
  and It_Mvto_Cme.Cd_Ciclo is not null
  and It_Mvto_Cme.Cd_Cancelamento is null
group by
     It_Mvto_Cme.Cd_Ciclo
      ,Cx_Kit.Ds_Res_Kit
      ,It_Mvto_Cme.Dt_Inc_Usuario
      ,Produto.Cd_Produto
      ,Cx_Kit.Sn_Caixa
union all
Select It_Mvto_Cme.Dt_Inc_Usuario
      ,'4' Sequencia
      ,'Esterilizacao' Fase
      ,It_Mvto_Cme.Cd_Ciclo
      ,Cx_Kit.Ds_Res_Kit
      ,Produto.Cd_Produto
      ,Cx_Kit.Sn_Caixa
From Dbamv.It_Mvto_Cme
    ,Dbamv.Cx_kit
    ,Dbamv.Config_Cme
    ,Dbamv.Produto
    ,dbamv.setor
Where It_Mvto_Cme.Cd_Kit = Cx_kit.Cd_kit
  and It_Mvto_Cme.Cd_Localidade = Config_Cme.Cd_Localidade_Esterilizacao
  and It_Mvto_Cme.Cd_Setor = Config_Cme.Cd_Setor_Esterilizacao
  and setor.cd_setor = config_cme.cd_setor_principal
  and setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
  and Cx_Kit.Cd_Produto = Produto.Cd_Produto(+)
  and It_Mvto_Cme.Cd_Ciclo is not null
  and It_Mvto_Cme.Cd_Cancelamento is null
group by
     It_Mvto_Cme.Cd_Ciclo
      ,Cx_Kit.Ds_Res_Kit
      ,It_Mvto_Cme.Dt_Inc_Usuario
      ,Produto.Cd_Produto
      ,Cx_Kit.Sn_Caixa
union all
Select It_Mvto_Cme.Dt_Inc_Usuario
      ,'5' Sequencia
      ,'Arsenal' Fase
      ,It_Mvto_Cme.Cd_Ciclo
      ,Cx_Kit.Ds_Res_Kit
      ,Produto.Cd_Produto
      ,Cx_Kit.Sn_Caixa
From Dbamv.It_Mvto_Cme
    ,Dbamv.Cx_kit
    ,Dbamv.Config_Cme
    ,Dbamv.Produto
    ,dbamv.setor
Where It_Mvto_Cme.Cd_Kit = Cx_kit.Cd_kit
  and It_Mvto_Cme.Cd_Localidade = Config_Cme.Cd_Localidade_Arsenal
  and It_Mvto_Cme.Cd_Setor = Config_Cme.Cd_Setor_Arsenal
  and setor.cd_setor = config_cme.cd_setor_principal
  and setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
  and Cx_Kit.Cd_Produto = Produto.Cd_Produto(+)
  and It_Mvto_Cme.Cd_Ciclo is not null
  and It_Mvto_Cme.Cd_Cancelamento is null
group by
     It_Mvto_Cme.Cd_Ciclo
      ,Cx_Kit.Ds_Res_Kit
      ,It_Mvto_Cme.Dt_Inc_Usuario
      ,Produto.Cd_Produto
      ,Cx_Kit.Sn_Caixa
union all
Select It_Mvto_Cme.Dt_Inc_Usuario
      ,'6' Sequencia
      ,'Externo' Fase
      ,It_Mvto_Cme.Cd_Ciclo
      ,Cx_Kit.Ds_Res_Kit
      ,Produto.Cd_Produto
      ,Cx_Kit.Sn_Caixa
From Dbamv.It_Mvto_Cme
    ,Dbamv.Cx_kit
    ,Dbamv.Config_Cme
    ,Dbamv.Produto
    ,dbamv.setor
Where It_Mvto_Cme.Cd_Kit = Cx_kit.Cd_kit
  and It_Mvto_Cme.Cd_Localidade = Config_Cme.Cd_Localidade_Externo
  and It_Mvto_Cme.Cd_Setor = Config_Cme.Cd_Setor_Externo
  and setor.cd_setor = config_cme.cd_setor_principal
  and setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
  and Cx_Kit.Cd_Produto = Produto.Cd_Produto(+)
  and It_Mvto_Cme.Cd_Ciclo is not null
  and It_Mvto_Cme.Cd_Cancelamento is null
  and It_Mvto_Cme.Sn_Mestre = 'S'
group by
     It_Mvto_Cme.Cd_Ciclo
      ,Cx_Kit.Ds_Res_Kit
      ,It_Mvto_Cme.Dt_Inc_Usuario
      ,Produto.Cd_Produto
      ,Cx_Kit.Sn_Caixa
)
group by Sequencia
      ,Fase
      ,Cd_Ciclo
      ,Ds_Res_Kit
      ,Dt_Inc_Usuario
      ,Cd_Produto
      ,Sn_Caixa
Order by  Cd_Ciclo,Sequencia,Cd_Produto,Dt_Inc_Usuario desc
)
ORDER BY CD_CICLO,SN_CAIXA,DS_RES_KIT,
SEQUENCIA,FASE
