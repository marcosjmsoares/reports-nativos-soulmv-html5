Select distinct
           It_Mvto_cme.Cd_Cancelamento Cancelamento
          ,cx_kit.cd_kit Caixa1
          ,cx_kit.ds_res_kit descresumida
          ,produto.ds_produto_resumido tipodecaixa
          ,prestador.cd_identificacao RecebidoPorSim
          ,it_mvto_cme.cd_usuario_inc RecebidoPorNao
          ,TO_CHAR(it_mvto_cme.dt_inc_usuario, 'HH24')  HoraRecebimento
          ,TO_CHAR(it_mvto_cme.dt_inc_usuario,'DD/MM/YYYY')DataRecebimento
          ,Config_Cme.Sn_Identi_Geral Sn_Identi_Geral
          ,Mvto_Cme.cd_setor_rsp_ent SetorRspEntrega
          ,Mvto_Cme.cd_colaborador_rsp_ent ColaboradorRspEntrega
From Dbamv.It_mvto_Cme
        ,Dbamv.Mvto_Cme
        ,Dbamv.Cx_Kit
        ,Dbamv.Prestador
        ,Dbamv.Produto
        ,Dbamv.Config_Cme
Where It_mvto_Cme.Cd_Mvto_Cme = Mvto_Cme.cd_mvto_cme
    And Cx_Kit.Cd_Kit = It_MVto_Cme.Cd_Kit
    And Prestador.Cd_Prestador(+) = It_mvto_Cme.Cd_Prestador_Identifica
    And Produto.Cd_Produto(+) = Cx_Kit.Cd_Produto
    And Config_Cme.Cd_Setor_Expurgo = It_Mvto_Cme.Cd_Setor
    And Config_Cme.Cd_Localidade_expurgo = It_Mvto_Cme.Cd_Localidade
    And Trunc(it_mvto_cme.dt_inc_usuario) BETWEEN :CAMPO_DATA_INI And 
   and (config_cme.cd_setor_principal in (select cd_setor
                                                                  from dbamv.setor
                                                               where cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA)
       or config_cme.cd_setor_principal is null)
order by cx_kit.cd_kit
