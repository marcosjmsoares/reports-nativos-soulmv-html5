--R_LIST_FUNCIONARIO
SELECT * FROM (
SELECT F.CD_FORNECEDOR AS CODIGO,
       F.NM_FORNECEDOR AS FUNCIONARIO,
       U.DS_UNIDADE AS UNIDADE,
       P.DS_PRODUTO,
       SUM(I.QT_MOVIMENTACAO) AS QTDE,
       SUM(I.QT_MOVIMENTACAO*I.VL_UNITARIO) AS VALOR
FROM   DBAMV.FORNECEDOR 	F,
       DBAMV.MVTO_ESTOQUE 	M,
       DBAMV.ITMVTO_ESTOQUE 	I,
       DBAMV.UNI_PRO		U,
       DBAMV.PRODUTO 		P,
       DBAMV.ESTOQUE 		E
WHERE   M.CD_MVTO_ESTOQUE = I.CD_MVTO_ESTOQUE
    AND M.CD_ESTOQUE  = E.CD_ESTOQUE
    AND E.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
    AND I.CD_UNI_PRO = U.CD_UNI_PRO
    AND F.CD_FORNECEDOR = M.CD_OPERADOR
    AND P.CD_PRODUTO  = I.CD_PRODUTO
    AND F.SN_OPERADOR = 'S'
    AND M.DT_MVTO_ESTOQUE BETWEEN :CAMPO_DT_INICIAL GROUP BY F.CD_FORNECEDOR,
 	 F.NM_FORNECEDOR,
   	 U.DS_UNIDADE,
         P.DS_PRODUTO
ORDER BY F.NM_FORNECEDOR, SUM(I.QT_MOVIMENTACAO*I.VL_UNITARIO)  DESC
)
