--R_MOV_FUNC
select f.NR_MATRICULA,
	   f.NM_FORNECEDOR funcionario,
sum(reccon_rec.vl_recebido) vl_bruto,
       sum(reccon_rec.vl_recebido - (reccon_rec.vl_recebido * (nvl(m.VL_percentual_DESCONTO,0)/100))) total_consumo
from dbamv.fornecedor f ,
     dbamv.mvto_estoque m,
     dbamv.itcon_rec ,
     dbamv.reccon_rec ,
     dbamv.estoque e  
where m.TP_MVTO_ESTOQUE = 'V'
      and f.CD_FORNECEDOR = m.CD_FORNECEDOR
      and m.DT_MVTO_ESTOQUE between :CAMPO_DT_INI and 
      AND M.CD_ESTOQUE  = E.CD_ESTOQUE
      AND E.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
      and m.cd_con_rec = itcon_rec.cd_con_rec
      and itcon_rec.cd_itcon_rec = reccon_rec.cd_itcon_rec
      and reccon_rec.tp_recebimento = '9'
group by f.NR_MATRICULA,
	 f.NM_FORNECEDOR
order by f.nm_FORNECEDOR
