--R_EXAME_PRESTADOR_R_SETORES1_Q_EXAME_PRESTADOR
SELECT * FROM (
SELECT PRESTADOR.CD_PRESTADOR
      ,PRESTADOR.NM_PRESTADOR
      ,EXAME_PRESTADOR.DS_ORI_ATE
      ,EXAME_PRESTADOR.CD_EXA_RX
      ,EXAME_PRESTADOR.DS_EXA_RX
      ,COUNT(*) QTDE
FROM    DBAMV.PRESTADOR,
       (SELECT  NVL(LAUDO_RX.CD_PRESTADOR,NVL(ITPED_RX.CD_PRESTADOR,SET_EXA.CD_PRESTADOR))
CD_PRESTADOR
               ,ORI_ATE.DS_ORI_ATE
               ,ITPED_RX.CD_EXA_RX
               ,EXA_RX.DS_EXA_RX
         FROM   DBAMV.PED_RX
               ,DBAMV.ATENDIME
               ,DBAMV.ORI_ATE
               ,DBAMV.ITPED_RX
               ,DBAMV.EXA_RX
               ,DBAMV.LAUDO_RX
               ,DBAMV.SET_EXA
               ,DBAMV.PRESTADOR
        WHERE  TRUNC(ITPED_RX.DT_REALIZADO) BETWEEN  :CAMPO_DT_INICIAL  AND  
          AND  PED_RX.CD_ATENDIMENTO  =  ATENDIME.CD_ATENDIMENTO
          AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
          AND  ATENDIME.CD_ORI_ATE    =  ORI_ATE.CD_ORI_ATE(+)
          AND  ITPED_RX.SN_REALIZADO = 'S'
          AND  PED_RX.CD_PED_RX       =  ITPED_RX.CD_PED_RX
          AND  ITPED_RX.CD_EXA_RX     =  EXA_RX.CD_EXA_RX
          AND  ITPED_RX.CD_LAUDO      =  LAUDO_RX.CD_LAUDO(+)
          AND  ITPED_RX.CD_PED_RX     =  LAUDO_RX.CD_PED_RX(+)
          AND  PED_RX.CD_SET_EXA      =  SET_EXA.CD_SET_EXA
          AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
          AND NVL(LAUDO_RX.CD_PRESTADOR,NVL(ITPED_RX.CD_PRESTADOR,SET_EXA.CD_PRESTADOR)) =
PRESTADOR.CD_PRESTADOR
          :CAMPO_CF_TP_ATENDE ) EXAME_PRESTADOR
WHERE  EXAME_PRESTADOR.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
GROUP BY PRESTADOR.CD_PRESTADOR
        ,PRESTADOR.NM_PRESTADOR
       ,EXAME_PRESTADOR.DS_ORI_ATE
       ,EXAME_PRESTADOR.CD_EXA_RX
       ,EXAME_PRESTADOR.DS_EXA_RX
)
ORDER BY CD_PRESTADOR,NM_PRESTADOR,
DS_ORI_ATE, CD_EXA_RX
