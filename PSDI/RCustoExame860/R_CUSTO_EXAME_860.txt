--R_CUSTO_EXAME_860
SELECT * FROM (
SELECT   ATENDIME.CD_CONVENIO
               ,CONVENIO.NM_CONVENIO
               ,decode( convenio.tp_convenio, 'A', exa_rx.CD_PROCEDIMENTO_SIA
                                                                 , 'H', exa_rx.CD_PROCEDIMENTO_SIH
                                                                 , 'C', exa_rx.exa_rx_cd_pro_fat
                                                                 , 'P', exa_rx.exa_rx_cd_pro_fat, '' )     CD_PRO_FAT
               ,EXA_RX.DS_EXA_RX                                                                        DS_PRO_FAT
               ,COUNT(*)  QTD
               ,dbamv.calc_vl_proc_unit( convenio.tp_convenio,
                                                          decode( convenio.tp_convenio, 'A', exa_rx.CD_PROCEDIMENTO_SIA
                                                                                                           , 'H', exa_rx.CD_PROCEDIMENTO_SIH
                                                                                                           , 'C', exa_rx.exa_rx_cd_pro_fat
                                                                                                           , 'P', exa_rx.exa_rx_cd_pro_fat, '' ),
                                                           ped_rx.dt_pedido,
                                                           ped_rx.hr_pedido,
                                                           convenio.cd_convenio,
                                                           con_pla.cd_con_pla,
                                                           atendime.tp_atendimento,
                                                           atendime.cd_tip_acom, 'U' )     VL_UNITARIO
                ,dbamv.calc_vl_filme_incid( atendime.cd_convenio,
                                                            con_pla.cd_con_pla,
                                                            decode( convenio.tp_convenio, 'A', exa_rx.CD_PROCEDIMENTO_SIA
                                                                                                             , 'H', exa_rx.CD_PROCEDIMENTO_SIH
                                                                                                             , 'C', exa_rx.exa_rx_cd_pro_fat
                                                                                                             , 'P', exa_rx.exa_rx_cd_pro_fat, '' ),
                                                            ped_rx.dt_pedido,
                                                            'F' )          vl_qtd_m2
FROM     DBAMV.ATENDIME
              ,DBAMV.CONVENIO
              ,DBAMV.PED_RX
              ,DBAMV.ITPED_RX
              ,DBAMV.EXA_RX
              ,DBAMV.PRO_FAT
              ,DBAMV.CON_PLA
,Dbamv.Empresa_Con_Pla
Where Empresa_Con_Pla.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And Empresa_Con_Pla.Cd_Convenio = Con_Pla.Cd_Convenio
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
And Empresa_Con_Pla.Cd_Con_Pla = Con_Pla.Cd_Con_Pla
And PED_RX.DT_PEDIDO  >= TO_DATE( TO_CHAR( :CAMPO_DT_INICIAL, 'DD/MM/YYYY' ) || ' 00', 'DD/MM/YYYY HH24' )
AND         PED_RX.DT_PEDIDO  <= TO_DATE( TO_CHAR( :CAMPO_DT_FINAL  , 'DD/MM/YYYY' ) || ' 23', 'DD/MM/YYYY HH24' )
AND        PED_RX.CD_SET_EXA IN (SELECT PSDI_SET_EXA.CD_SET_EXA
                                                              FROM DBAMV.PSDI_SET_EXA
                                                              WHERE NM_USUARIO = USER)
AND         ATENDIME.CD_CONVENIO  =  CONVENIO.CD_CONVENIO
AND         ATENDIME.CD_ATENDIMENTO  =  PED_RX.CD_ATENDIMENTO
AND         PED_RX.CD_PED_RX  =  ITPED_RX.CD_PED_RX
AND         EXA_RX.CD_EXA_RX  =  ITPED_RX.CD_EXA_RX
AND         EXA_RX.EXA_RX_CD_PRO_FAT  =  PRO_FAT.CD_PRO_FAT(+)
AND         NVL(PRO_FAT.SN_ATIVO,'S')  =  'S'
AND         CON_PLA.CD_CONVENIO  =  CONVENIO.CD_CONVENIO
AND         CON_PLA.CD_CON_PLA  =  PED_RX.CD_CON_PLA
GROUP BY   ATENDIME.CD_CONVENIO
                    ,CONVENIO.NM_CONVENIO
                    ,decode( convenio.tp_convenio, 'A', exa_rx.CD_PROCEDIMENTO_SIA
                                                                      , 'H', exa_rx.CD_PROCEDIMENTO_SIH
                                                                      , 'C', exa_rx.exa_rx_cd_pro_fat
                                                                      , 'P', exa_rx.exa_rx_cd_pro_fat, '' )
                     ,EXA_RX.DS_EXA_RX
                     ,dbamv.calc_vl_proc_unit( convenio.tp_convenio,
                                                                decode( convenio.tp_convenio, 'A', exa_rx.CD_PROCEDIMENTO_SIA
                                                                                                                 , 'H', exa_rx.CD_PROCEDIMENTO_SIH
                                                                                                                 , 'C', exa_rx.exa_rx_cd_pro_fat
                                                                                                                 , 'P', exa_rx.exa_rx_cd_pro_fat, '' ),
                                                                 ped_rx.dt_pedido,
                                                                 ped_rx.hr_pedido,
                                                                 convenio.cd_convenio,
                                                                 con_pla.cd_con_pla,
                                                                 atendime.tp_atendimento,
                                                                 atendime.cd_tip_acom, 'U' )
                    ,dbamv.calc_vl_filme_incid( atendime.cd_convenio,
                                                                 con_pla.cd_con_pla,
                                                                 decode( convenio.tp_convenio, 'A', exa_rx.CD_PROCEDIMENTO_SIA
                                                                                                                  , 'H', exa_rx.CD_PROCEDIMENTO_SIH
                                                                                                                  , 'C', exa_rx.exa_rx_cd_pro_fat
                                                                                                                  , 'P', exa_rx.exa_rx_cd_pro_fat, '' ),
                                                                  ped_rx.dt_pedido,
                                                                  'F' )
ORDER BY  CONVENIO.NM_CONVENIO
                    ,EXA_RX.DS_EXA_RX
)
ORDER BY NM_CONVENIO,CD_CONVENIO, DS_PRO_FAT
