--R_GRAF_EXA_POR_GRUPO
SELECT * FROM (
SELECT  CD_GRU_PRO
               ,DS_GRU_PRO
               ,PERIODO
               ,ORDEM
               ,COUNT(*)   QT_EXAME
FROM (  SELECT  GRU_PRO.CD_GRU_PRO
                             ,GRU_PRO.DS_GRU_PRO
                             ,'Manhã'      PERIODO
                             ,1                ORDEM
               FROM    DBAMV.PED_RX
                             ,DBAMV.ITPED_RX
                             ,DBAMV.EXA_RX
                             ,DBAMV.PRO_FAT
                             ,DBAMV.GRU_PRO
              WHERE TRUNC(PED_RX.DT_PEDIDO)  BETWEEN  :CAMPO_DT_INICIAL  AND  
              AND       TO_NUMBER(TO_CHAR(PED_RX.HR_PEDIDO,'HH24') )  BETWEEN :CAMPO_PER01_I  AND 
              AND         PED_RX.CD_SET_EXA IN ( SELECT PSDI_SET_EXA.CD_SET_EXA
                                                                             FROM DBAMV.PSDI_SET_EXA
                                                                             WHERE PSDI_SET_EXA.NM_USUARIO = USER )
              AND       PED_RX.CD_PED_RX  =  ITPED_RX.CD_PED_RX
              AND       ITPED_RX.CD_EXA_RX  =  EXA_RX.CD_EXA_RX
              AND       EXA_RX.EXA_RX_CD_PRO_FAT  =  PRO_FAT.CD_PRO_FAT
              AND       NVL( PRO_FAT.SN_ATIVO, 'S' )  =  'S'
              AND       PRO_FAT.CD_GRU_PRO  =  GRU_PRO.CD_GRU_PRO
             UNION ALL
              SELECT  GRU_PRO.CD_GRU_PRO
                             ,GRU_PRO.DS_GRU_PRO
                             ,'Tarde'        PERIODO
                             ,2                 ORDEM
              FROM    DBAMV.PED_RX
                           ,DBAMV.ITPED_RX
                           ,DBAMV.EXA_RX
                           ,DBAMV.PRO_FAT
                           ,DBAMV.GRU_PRO
              WHERE TRUNC(PED_RX.DT_PEDIDO)  BETWEEN  :CAMPO_DT_INICIAL  AND  
              AND       TO_NUMBER(TO_CHAR(PED_RX.HR_PEDIDO,'HH24') )  BETWEEN :CAMPO_PER02_I  AND 
              AND         PED_RX.CD_SET_EXA IN ( SELECT PSDI_SET_EXA.CD_SET_EXA
                                                                             FROM DBAMV.PSDI_SET_EXA
                                                                             WHERE PSDI_SET_EXA.NM_USUARIO = USER )
              AND        PED_RX.CD_PED_RX  =  ITPED_RX.CD_PED_RX
              AND        ITPED_RX.CD_EXA_RX  =  EXA_RX.CD_EXA_RX
              AND        EXA_RX.EXA_RX_CD_PRO_FAT  =  PRO_FAT.CD_PRO_FAT
              AND        NVL( PRO_FAT.SN_ATIVO, 'S' )  =  'S'
              AND        PRO_FAT.CD_GRU_PRO  =  GRU_PRO.CD_GRU_PRO
              UNION ALL
              SELECT  GRU_PRO.CD_GRU_PRO
                             ,GRU_PRO.DS_GRU_PRO
                             ,'Noite'         PERIODO
                             ,3                 ORDEM
               FROM    DBAMV.PED_RX
                             ,DBAMV.ITPED_RX
                             ,DBAMV.EXA_RX
                             ,DBAMV.PRO_FAT
                             ,DBAMV.GRU_PRO
               WHERE TRUNC(PED_RX.DT_PEDIDO)  BETWEEN  :CAMPO_DT_INICIAL  AND  
               AND       TO_NUMBER(TO_CHAR(PED_RX.HR_PEDIDO,'HH24') )  BETWEEN :CAMPO_PER03_I  AND 
              AND         PED_RX.CD_SET_EXA IN ( SELECT PSDI_SET_EXA.CD_SET_EXA
                                                                             FROM DBAMV.PSDI_SET_EXA
                                                                             WHERE PSDI_SET_EXA.NM_USUARIO = USER )
               AND       PED_RX.CD_PED_RX  =  ITPED_RX.CD_PED_RX
               AND       ITPED_RX.CD_EXA_RX  =  EXA_RX.CD_EXA_RX
               AND       EXA_RX.EXA_RX_CD_PRO_FAT  =  PRO_FAT.CD_PRO_FAT
               AND       NVL( PRO_FAT.SN_ATIVO, 'S' )  =  'S'
               AND       PRO_FAT.CD_GRU_PRO  =  GRU_PRO.CD_GRU_PRO
               UNION ALL
               SELECT  GRU_PRO.CD_GRU_PRO
                              ,GRU_PRO.DS_GRU_PRO
                              ,'Madrugada'     PERIODO
                              ,4                      ORDEM
                FROM    DBAMV.PED_RX
                              ,DBAMV.ITPED_RX
                              ,DBAMV.EXA_RX
                              ,DBAMV.PRO_FAT
                              ,DBAMV.GRU_PRO
                WHERE TRUNC(PED_RX.DT_PEDIDO)  BETWEEN  :CAMPO_DT_INICIAL  AND  
              AND         PED_RX.CD_SET_EXA IN ( SELECT PSDI_SET_EXA.CD_SET_EXA
                                                                             FROM DBAMV.PSDI_SET_EXA
                                                                             WHERE PSDI_SET_EXA.NM_USUARIO = USER )
                 AND       PED_RX.CD_PED_RX  =  ITPED_RX.CD_PED_RX
                 AND       ITPED_RX.CD_EXA_RX  =  EXA_RX.CD_EXA_RX
                 AND       EXA_RX.EXA_RX_CD_PRO_FAT  =  PRO_FAT.CD_PRO_FAT
                 AND       NVL( PRO_FAT.SN_ATIVO, 'S' )  =  'S'
                 AND       PRO_FAT.CD_GRU_PRO  =  GRU_PRO.CD_GRU_PRO  )
GROUP BY   CD_GRU_PRO
                     ,DS_GRU_PRO
                     ,PERIODO
                     ,ORDEM
)
ORDER BY CD_GRU_PRO,DS_GRU_PRO,ORDEM
