--R_EST_EXAME_PRESTADOR
SELECT * FROM (
SELECT SET_EXA.CD_SET_EXA
              ,SET_EXA.NM_SET_EXA
              ,PRESTADOR.CD_PRESTADOR
              ,PRESTADOR.NM_PRESTADOR
              ,ITPED_RX.CD_EXA_RX
              ,EXA_RX.DS_EXA_RX
              ,COUNT(*) QTDE
FROM   DBAMV.ATENDIME
            ,DBAMV.PED_RX
            ,DBAMV.PRESTADOR
            ,DBAMV.SET_EXA
            ,DBAMV.ITPED_RX
            ,DBAMV.EXA_RX
WHERE  ATENDIME.CD_ATENDIMENTO  =  PED_RX.CD_ATENDIMENTO
AND        PED_RX.CD_PRESTADOR IS NOT NULL
AND        PED_RX.CD_PRESTADOR  =  PRESTADOR.CD_PRESTADOR
AND        PED_RX.CD_SET_EXA  =  SET_EXA.CD_SET_EXA
AND        PED_RX.CD_PED_RX  =  ITPED_RX.CD_PED_RX
AND        ITPED_RX.CD_EXA_RX  =  EXA_RX.CD_EXA_RX
AND        ITPED_RX.DT_REALIZADO BETWEEN  :CAMPO_DT_INICIAL  AND  
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND        SET_EXA.CD_SET_EXA IN (SELECT PSDI_SET_EXA.CD_SET_EXA
                                                               FROM DBAMV.PSDI_SET_EXA
                                                               WHERE NM_USUARIO = USER)
GROUP BY  SET_EXA.CD_SET_EXA
                    ,SET_EXA.NM_SET_EXA
                    ,PRESTADOR.CD_PRESTADOR
                    ,PRESTADOR.NM_PRESTADOR
                    ,ITPED_RX.CD_EXA_RX
                    ,EXA_RX.DS_EXA_RX
)
ORDER BY CD_SET_EXA,NM_SET_EXA,
CD_PRESTADOR,NM_PRESTADOR
