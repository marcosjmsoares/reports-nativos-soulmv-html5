--R_PLANTOES
SELECT * FROM (
SELECT  ESCALA_DIA.CD_PRESTADOR     CODIGO
               ,PRESTADOR.NM_PRESTADOR   DESCRICAO
               ,ESCALA_DIA.CD_ESCALA_DIA
               ,ESCALA_DIA.DT_ESCALA_DIA
               ,ESCALA_DIA.CD_RECURSO
               ,RECURSO.NM_RECURSO
               ,ESCALA_DIA.HR_INIC
               ,ESCALA_DIA.HR_FIM
               ,ESCALA_DIA.SN_ATIVO
               ,COUNT(*)  QT_AGENDA
FROM    dbamv.ESCALA_DIA
             ,dbamv.PRESTADOR
             ,dbamv.RECURSO
             ,dbamv.IT_ESCALA_DIA
WHERE  ESCALA_DIA.DT_ESCALA_DIA BETWEEN :CAMPO_DT_INIC AND 
AND        ESCALA_DIA.SN_ATIVO = 'S'
AND        ESCALA_DIA.CD_PRESTADOR  =  PRESTADOR.CD_PRESTADOR(+)
AND        ESCALA_DIA.CD_RECURSO  =  RECURSO.CD_RECURSO
AND        RECURSO.CD_SET_EXA IN ( SELECT PSDI_SET_EXA.CD_SET_EXA
                                                                FROM DBAMV.PSDI_SET_EXA
                                                                WHERE PSDI_SET_EXA.NM_USUARIO = USER )
AND        ESCALA_DIA.CD_ESCALA_DIA  =  IT_ESCALA_DIA.CD_ESCALA_DIA(+)
GROUP BY ESCALA_DIA.CD_PRESTADOR
                   ,PRESTADOR.NM_PRESTADOR
                   ,ESCALA_DIA.CD_ESCALA_DIA
                   ,ESCALA_DIA.DT_ESCALA_DIA
                   ,ESCALA_DIA.CD_RECURSO
                   ,RECURSO.NM_RECURSO
                   ,ESCALA_DIA.HR_INIC
                   ,ESCALA_DIA.HR_FIM
                   ,ESCALA_DIA.SN_ATIVO
ORDER BY :CAMPO_ORDEM, ESCALA_DIA.DT_ESCALA_DIA, CD_RECURSO
)
