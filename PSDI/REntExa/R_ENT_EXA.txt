--R_ENT_EXA
SELECT  PED_RX.CD_PED_RX CD_PED_RX
       ,PED_RX.CD_ATENDIMENTO CD_ATENDIMENTO
       ,PACIENTE.NM_PACIENTE NM_PACIENTE
       ,PRESTADOR.NM_PRESTADOR NM_PRESTADOR
       ,SETOR.NM_SETOR NM_SETOR
       ,SET_EXA.NM_SET_EXA NM_SET_EXA
       ,PED_RX.NR_CONTROLE NR_CONTROLE
       ,PED_RX.DT_ENTREGA DT_ENTREGA
       ,PED_RX.HR_ENTREGA HR_ENTREGA
FROM    dbamv.PED_RX PED_RX
       ,dbamv.ATENDIME ATENDIME
       ,dbamv.PACIENTE PACIENTE
       ,dbamv.PRESTADOR PRESTADOR
       ,dbamv.SETOR SETOR
       ,dbamv.SET_EXA SET_EXA
WHERE PED_RX.CD_PED_RX = :CAMPO_PPEDIDO
AND   PRESTADOR.CD_PRESTADOR (+) = PED_RX.CD_PRESTADOR
AND   SETOR.CD_SETOR = PED_RX.CD_SETOR
AND   ATENDIME.CD_ATENDIMENTO = PED_RX.CD_ATENDIMENTO
AND   PACIENTE.CD_PACIENTE = ATENDIME.CD_PACIENTE
AND   SET_EXA.CD_SET_EXA = PED_RX.CD_SET_EXA
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND   SET_EXA.CD_SET_EXA IN (SELECT PSDI_SET_EXA.CD_SET_EXA
                                                               FROM DBAMV.PSDI_SET_EXA
                                                               WHERE NM_USUARIO = USER)
AND   PED_RX.LG_ENTREGA IS NULL
