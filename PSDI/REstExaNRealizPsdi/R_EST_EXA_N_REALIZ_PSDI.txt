--R_EST_EXA_N_REALIZ_PSDI
SELECT     CODIGO
              ,:CAMPO_DS_COLUNA    DESCRICAO
              ,COUNT(*)                QT_EXAMES
FROM   DBAMV.V_EXAME_N_REALIZADO_PSDI  V
WHERE  TRUNC(V.DT_PEDIDO)  BETWEEN  :CAMPO_DT_INI  GROUP BY  :CAMPO_CD_COLUNA
                    ,:CAMPO_DS_COLUNA
ORDER BY QT_EXAMES DESC
