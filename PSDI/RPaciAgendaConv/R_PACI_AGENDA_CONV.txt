--R_PACI_AGENDA_CONV
select * from (
SELECT CONVENIO.CD_CONVENIO                                              CD_CONVENIO
              ,CONVENIO.NM_CONVENIO                                             NM_CONVENIO
              ,TRUNC(ESCALA_DIA.DT_ESCALA_DIA)                          DT_ESCALA_DIA
              ,TO_CHAR(IT_ESCALA_DIA.HR_AGENDA, 'HH24')    HR_AGENDA
              ,RECURSO.NM_RECURSO                                               NM_RECURSO
              ,PRESTADOR.NM_PRESTADOR                                      NM_PRESTADOR
              ,PACIENTE.CD_PACIENTE                                                CD_PACIENTE
              ,DECODE(IT_ESCALA_DIA.CD_PACIENTE, NULL, IT_ESCALA_DIA.NM_PACIENTE, PACIENTE.NM_PACIENTE)   NM_PACIENTE
              ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND paciente.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE
              ,IT_ESCALA_DIA.NR_TELEFONE                                     NR_TELEFONE
              ,IT_ESCALA_DIA.SN_ENCAIXE                                         SN_ENCAIXE
              ,IT_ESCALA_DIA.NR_IDADE                                             NR_IDADE
FROM  DBAMV.ESCALA_DIA
            ,DBAMV.PRESTADOR
            ,DBAMV.RECURSO
            ,DBAMV.IT_ESCALA_DIA
            ,DBAMV.PACIENTE
            ,DBAMV.CONVENIO
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And TRUNC(ESCALA_DIA.DT_ESCALA_DIA)  BETWEEN  :CAMPO_DT_INI  AND  
AND       ESCALA_DIA.CD_RECURSO = RECURSO.CD_RECURSO
AND       ESCALA_DIA.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
AND       ESCALA_DIA.CD_ESCALA_DIA = IT_ESCALA_DIA.CD_ESCALA_DIA
AND       (IT_ESCALA_DIA.NM_PACIENTE IS NOT NULL OR IT_ESCALA_DIA.CD_PACIENTE IS NOT NULL)
AND        IT_ESCALA_DIA.CD_PACIENTE = PACIENTE.CD_PACIENTE(+)
AND        IT_ESCALA_DIA.CD_CONVENIO = CONVENIO.CD_CONVENIO(+)
:CAMPO_COND_ATENDE
)
ORDER BY NM_CONVENIO, DT_ESCALA_DIA, HR_AGENDA
