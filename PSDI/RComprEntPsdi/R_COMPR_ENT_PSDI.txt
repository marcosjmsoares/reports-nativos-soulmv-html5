--R_COMPR_ENT_PSDI
SELECT * FROM (
SELECT PROT.CD_ENT_PSDI
             ,PROT.DT_ENTREGA
             ,PROT.HR_ENTREGA
             ,PROT.NM_RESPONSAVEL
             ,PROT.NR_CPF
             ,PROT.CD_USUARIO
             ,PROT.DT_PROTOCOLO
             ,ATENDIME.CD_ATENDIMENTO
             ,PED_RX.CD_PED_RX
             ,PACIENTE.CD_PACIENTE
             ,PACIENTE.NM_PACIENTE
             ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND paciente.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE
             ,PACIENTE.CD_PACIENTE||' '||PACIENTE.NM_PACIENTE PACIENTE
             ,EXA_RX.CD_EXA_RX
             ,EXA_RX.DS_EXA_RX
             ,EXA_RX.CD_EXA_RX||' '||EXA_RX.DS_EXA_RX EXA_RX
             ,ITPED_RX.CD_LAUDO CD_LAUDO
  FROM DBAMV.ENT_PSDI PROT
            ,DBAMV.PED_RX
            ,DBAMV.ATENDIME
            ,DBAMV.PACIENTE
            ,DBAMV.ITPED_RX
            ,DBAMV.EXA_RX
 WHERE PED_RX.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
      AND  ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
      AND  PED_RX.CD_PED_RX = ITPED_RX.CD_PED_RX
      AND  ITPED_RX.CD_ENT_PSDI = PROT.CD_ENT_PSDI
      AND  ITPED_RX.CD_EXA_RX = EXA_RX.CD_EXA_RX
      AND  ITPED_RX.CD_LAUDO IS NOT NULL
       AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
      :CAMPO_CF_PROTOCOLO
)
ORDER BY NM_RESPONSAVEL,NR_CPF,CD_USUARIO,DT_PROTOCOLO,DT_ENTREGA,HR_ENTREGA,CD_ENT_PSDI,
CD_ATENDIMENTO,CD_PACIENTE,NM_PACIENTE,PACIENTE,CD_PED_RX,
CD_LAUDO, CD_EXA_RX, DS_EXA_RX, EXA_RX
