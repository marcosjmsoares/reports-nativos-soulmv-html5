--R_RECURSO
SELECT  RECURSO.CD_RECURSO CODIGO
       ,RECURSO.NM_RECURSO DESCRICAO
       ,SET_EXA.CD_SET_EXA CD_SET_EXA
       ,SET_EXA.NM_SET_EXA NM_SET_EXA
FROM    dbamv.RECURSO RECURSO
       ,dbamv.SET_EXA SET_EXA
WHERE SET_EXA.CD_SET_EXA = RECURSO.CD_SET_EXA
AND        SET_EXA.CD_SET_EXA IN ( SELECT PSDI_SET_EXA.CD_SET_EXA
                                                                FROM DBAMV.PSDI_SET_EXA
                                                                WHERE PSDI_SET_EXA.NM_USUARIO = USER )
ORDER BY :CAMPO_ORDEM, NM_SET_EXA
