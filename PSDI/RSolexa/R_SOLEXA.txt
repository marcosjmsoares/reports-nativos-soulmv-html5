--R_SOLEXA
SELECT * FROM (
SELECT   PED_RX.CD_PRESTADOR  CD_PRESTADOR
        ,DECODE(PED_RX.CD_PRESTADOR, null, PED_RX.NM_PRESTADOR, PRESTADOR.NM_PRESTADOR)  NM_PRESTADOR
        ,PED_RX.DT_PEDIDO              DT_PEDIDO
        ,ATENDIME.CD_ORI_ATE        CD_ORI_ATE
        ,ORI_ATE.DS_ORI_ATE           DS_ORI_ATE
        ,ATENDIME.CD_PACIENTE     CD_PACIENTE
        ,PACIENTE.NM_PACIENTE     NM_PACIENTE
        ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND paciente.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE
        ,ATENDIME.CD_CONVENIO    CD_CONVENIO
        ,CONVENIO.NM_CONVENIO    NM_CONVENIO
        ,CONVENIO.TP_CONVENIO    TP_CONVENIO
        ,SETOR.NM_SETOR                NM_SETOR
        ,PED_RX.CD_PED_RX             CD_PED_RX
        ,PED_RX.CD_TECNICO            CD_TECNICO
        ,ITPED_RX.CD_EXA_RX           CD_EXA_RX
        ,EXA_RX.DS_EXA_RX
FROM     DBAMV.ITPED_RX
        ,DBAMV.EXA_RX
        ,DBAMV.PED_RX
        ,DBAMV.ATENDIME
        ,DBAMV.ORI_ATE
        ,DBAMV.PACIENTE
        ,DBAMV.CONVENIO
        ,DBAMV.SETOR
        ,DBAMV.PRESTADOR
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And PED_RX.DT_PEDIDO  >= TO_DATE( TO_CHAR( :CAMPO_DT_INICIAL, 'DD/MM/YYYY' ) || ' 00', 'DD/MM/YYYY HH24' )
AND      PED_RX.DT_PEDIDO  <= TO_DATE( TO_CHAR( :CAMPO_DT_FINAL , 'DD/MM/YYYY' ) || ' 23', 'DD/MM/YYYY HH24' )
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND      PED_RX.CD_PRESTADOR      = PRESTADOR.CD_PRESTADOR(+)
AND      EXA_RX.CD_EXA_RX         = ITPED_RX.CD_EXA_RX
AND      ITPED_RX.CD_PED_RX       = PED_RX.CD_PED_RX
AND PED_RX.CD_SET_EXA IN ( SELECT CD_SET_EXA
                           FROM   DBAMV.PSDI_SET_EXA
                           WHERE  NM_USUARIO = USER )
AND ATENDIME.CD_ATENDIMENTO = PED_RX.CD_ATENDIMENTO
AND ORI_ATE.CD_ORI_ATE      = ATENDIME.CD_ORI_ATE
AND ATENDIME.CD_PACIENTE    = PACIENTE.CD_PACIENTE
AND CONVENIO.CD_CONVENIO    = ATENDIME.CD_CONVENIO
AND SETOR.CD_SETOR          = PED_RX.CD_SETOR
)
ORDER BY NM_PRESTADOR,CD_PRESTADOR,
DT_PEDIDO,CD_ORI_ATE,DS_ORI_ATE,CD_PACIENTE,NM_PACIENTE,CD_CONVENIO,NM_CONVENIO,TP_CONVENIO,NM_SETOR,CD_TECNICO,
CD_PED_RX
