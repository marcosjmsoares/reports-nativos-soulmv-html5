--R_PEDIDO_SEM_LAUDO
SELECT * FROM (
SELECT SETOR.CD_SETOR
              ,SETOR.NM_SETOR
              ,PED_RX.CD_PED_RX
              ,PED_RX.DT_PEDIDO
              ,ORI_ATE.DS_ORI_ATE
              ,PACIENTE.CD_PACIENTE
              ,PACIENTE.NM_PACIENTE
              ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND paciente.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE
              ,CONVENIO.NM_CONVENIO
              ,EXA_RX.CD_EXA_RX
              ,EXA_RX.DS_EXA_RX
FROM   DBAMV.ATENDIME
            ,DBAMV.ORI_ATE
            ,DBAMV.PACIENTE
            ,DBAMV.PED_RX
            ,DBAMV.CONVENIO
            ,DBAMV.ITPED_RX
            ,DBAMV.EXA_RX
            ,DBAMV.SETOR
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And TRUNC(PED_RX.DT_PEDIDO)  BETWEEN :CAMPO_DT_INI AND 
AND         PED_RX.CD_SET_EXA IN ( SELECT PSDI_SET_EXA.CD_SET_EXA
                                                                FROM DBAMV.PSDI_SET_EXA
                                                                WHERE PSDI_SET_EXA.NM_USUARIO = USER )
AND       ATENDIME.CD_ORI_ATE  =  ORI_ATE.CD_ORI_ATE
AND       ATENDIME.CD_PACIENTE  =  PACIENTE.CD_PACIENTE
AND       ATENDIME.CD_ATENDIMENTO  =  PED_RX.CD_ATENDIMENTO
AND       PED_RX.CD_CONVENIO  =  CONVENIO.CD_CONVENIO
AND       PED_RX.CD_PED_RX  =  ITPED_RX.CD_PED_RX
AND       ITPED_RX.CD_EXA_RX  =  EXA_RX.CD_EXA_RX
AND       SETOR.CD_SETOR  =  PED_RX.CD_SETOR
AND       ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND ITPED_RX.CD_LAUDO IS NULL
)
ORDER BY CD_SETOR,NM_SETOR,
CD_PED_RX,DT_PEDIDO,CD_PACIENTE,NM_PACIENTE,DS_ORI_ATE,NM_CONVENIO,CD_EXA_RX
