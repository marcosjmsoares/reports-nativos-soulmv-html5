--R_EXA_REALIZ_CONV_GRUPO_R_CONVENIO_Q_CONVENIO
SELECT  CONVENIO.CD_CONVENIO     CD_CONVENIO
               ,CONVENIO.NM_CONVENIO    NM_CONVENIO
               ,COUNT(*)                                  QT_CONVENIO,
(SELECT COUNT(CONVENIO.CD_CONVENIO)
FROM      DBAMV.PED_RX
               ,DBAMV.CONVENIO
               ,DBAMV.ITPED_RX
               ,DBAMV.EXA_RX
               ,DBAMV.PRO_FAT
               ,DBAMV.GRU_PRO
               ,DBAMV.ATENDIME
WHERE PED_RX.CD_CONVENIO  =  CONVENIO.CD_CONVENIO
AND       PED_RX.CD_PED_RX  =  ITPED_RX.CD_PED_RX
AND       ITPED_RX.CD_EXA_RX  =  EXA_RX.CD_EXA_RX
AND       EXA_RX.EXA_RX_CD_PRO_FAT  =  PRO_FAT.CD_PRO_FAT(+)
AND       PRO_FAT.CD_GRU_PRO  =  GRU_PRO.CD_GRU_PRO(+)
AND       ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND       ATENDIME.CD_ATENDIMENTO = PED_RX.CD_ATENDIMENTO
:CAMPO_COND_CD_EXA_RX
) QTD_CONV
FROM      DBAMV.PED_RX
               ,DBAMV.CONVENIO
               ,DBAMV.ITPED_RX
               ,DBAMV.EXA_RX
               ,DBAMV.PRO_FAT
               ,DBAMV.GRU_PRO
               ,DBAMV.ATENDIME
WHERE PED_RX.CD_CONVENIO  =  CONVENIO.CD_CONVENIO
AND       PED_RX.CD_PED_RX  =  ITPED_RX.CD_PED_RX
AND       ITPED_RX.CD_EXA_RX  =  EXA_RX.CD_EXA_RX
AND       EXA_RX.EXA_RX_CD_PRO_FAT  =  PRO_FAT.CD_PRO_FAT(+)
AND       PRO_FAT.CD_GRU_PRO  =  GRU_PRO.CD_GRU_PRO(+)
AND       ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND       ATENDIME.CD_ATENDIMENTO = PED_RX.CD_ATENDIMENTO
GROUP BY  CONVENIO.CD_CONVENIO
                    ,CONVENIO.NM_CONVENIO
ORDER BY  QT_CONVENIO  DESC
