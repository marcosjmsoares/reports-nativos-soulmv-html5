--R_EXAME_EXCLUIDO
SELECT * FROM (
SELECT LOG_EXC_PED_RX.CD_PED_RX
      ,LOG_EXC_PED_RX.CD_ATENDIMENTO
      ,LOG_EXC_PED_RX.CD_PACIENTE
      ,PACIENTE.NM_PACIENTE
      ,LOG_EXC_PED_RX.DT_LOG_EXC_PED_RX
      ,LOG_EXC_PED_RX.HR_LOG_EXC_PED_RX
      ,LOG_EXC_PED_RX.DS_MOTIVO
      ,LOG_EXC_PED_RX.NM_USUARIO
FROM   DBAMV.LOG_EXC_PED_RX
            ,DBAMV.PACIENTE
WHERE  TRUNC(LOG_EXC_PED_RX.DT_LOG_EXC_PED_RX) BETWEEN :CAMPO_DT_INICIAL AND 
AND         LOG_EXC_PED_RX.CD_PACIENTE = PACIENTE.CD_PACIENTE
ORDER BY LOG_EXC_PED_RX.HR_LOG_EXC_PED_RX
)
ORDER BY DT_LOG_EXC_PED_RX, cd_ped_rx desc
