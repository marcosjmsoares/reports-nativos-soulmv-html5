--R_ESTAT_DIA_CONV_PAC
SELECT * FROM (
SELECT DATA
          ,CD_CONVENIO
          ,NM_CONVENIO
          ,CD_PROCEDIMENTO
          ,DS_PROCEDIMENTO
          ,CD_ATENDIMENTO
          ,CD_PACIENTE
          ,NM_PACIENTE
          ,NM_SOCIAL_PACIENTE
          ,COUNT(*) QT_EXAME
FROM
(
SELECT DECODE(:CAMPO_SITU_EXAME, 'S' , TRUNC(PED_RX.DT_PEDIDO), 'N', TRUNC(PED_RX.DT_PEDIDO), TRUNC(ITPED_RX.DT_REALIZADO))  DATA
              ,CONVENIO.CD_CONVENIO    CD_CONVENIO
              ,CONVENIO.NM_CONVENIO   NM_CONVENIO
              ,DECODE(CONVENIO.TP_CONVENIO,'C',EXA_RX.EXA_RX_CD_PRO_FAT
                                          ,'P',EXA_RX.EXA_RX_CD_PRO_FAT
                                          ,'H',EXA_RX.CD_PROCEDIMENTO_SIH)  CD_PROCEDIMENTO
              ,PRO_FAT.DS_PRO_FAT     DS_PROCEDIMENTO
              ,PED_RX.CD_ATENDIMENTO  CD_ATENDIMENTO
              ,PACIENTE.CD_PACIENTE   CD_PACIENTE
              ,PACIENTE.NM_PACIENTE   NM_PACIENTE
              ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND paciente.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE
FROM  DBAMV.ATENDIME
      ,DBAMV.PACIENTE
      ,DBAMV.PED_RX
      ,DBAMV.CONVENIO
      ,DBAMV.PRO_FAT
      ,DBAMV.ITPED_RX
      ,DBAMV.EXA_RX
      ,DBAMV.LAUDO_RX
      ,DBAMV.PRESTADOR
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND    ATENDIME.CD_ATENDIMENTO  =  PED_RX.CD_ATENDIMENTO
AND    ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND    PED_RX.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND    EXA_RX.EXA_RX_CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
AND    PED_RX.CD_PED_RX  =  ITPED_RX.CD_PED_RX
AND    ITPED_RX.CD_EXA_RX  =  EXA_RX.CD_EXA_RX
AND    ITPED_RX.CD_PED_RX = LAUDO_RX.CD_PED_RX(+)
AND    ITPED_RX.CD_LAUDO  =  LAUDO_RX.CD_LAUDO(+)
AND    LAUDO_RX.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
AND    CONVENIO.TP_CONVENIO <> 'A'
AND    :CAMPO_TP_PROCED IN ('A','P')
UNION ALL
SELECT DECODE(:CAMPO_SITU_EXAME, 'S' , TRUNC(PED_RX.DT_PEDIDO), 'N', TRUNC(PED_RX.DT_PEDIDO), TRUNC(ITPED_RX.DT_REALIZADO))  DATA
              ,CONVENIO.CD_CONVENIO           CD_CONVENIO
              ,CONVENIO.NM_CONVENIO          NM_CONVENIO
              ,EXA_RX.CD_PROCEDIMENTO_SIA          CD_PROCEDIMENTO
              ,PROCEDIMENTO_SUS.DS_PROCEDIMENTO    DS_PROCEDIMENTO
              ,PED_RX.CD_ATENDIMENTO       CD_ATENDIMENTO
              ,PACIENTE.CD_PACIENTE            CD_PACIENTE
              ,PACIENTE.NM_PACIENTE            NM_PACIENTE
              ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND paciente.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE
FROM  DBAMV.ATENDIME
      ,DBAMV.PACIENTE
      ,DBAMV.PED_RX
      ,DBAMV.CONVENIO
      ,DBAMV.PROCEDIMENTO_SUS
      ,DBAMV.ITPED_RX
      ,DBAMV.EXA_RX
      ,DBAMV.LAUDO_RX
      ,DBAMV.PRESTADOR
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And     ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND    ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND    ATENDIME.CD_ATENDIMENTO  =  PED_RX.CD_ATENDIMENTO
AND    PED_RX.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND    EXA_RX.CD_PROCEDIMENTO_SIA = PROCEDIMENTO_SUS.CD_PROCEDIMENTO(+)
AND    PED_RX.CD_PED_RX  =  ITPED_RX.CD_PED_RX
AND    ITPED_RX.CD_EXA_RX  =  EXA_RX.CD_EXA_RX
AND    ITPED_RX.CD_PED_RX = LAUDO_RX.CD_PED_RX(+)
AND    ITPED_RX.CD_LAUDO  =  LAUDO_RX.CD_LAUDO(+)
AND    LAUDO_RX.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
AND    CONVENIO.TP_CONVENIO = 'A'
AND    :CAMPO_TP_PROCED IN ('A','S')
:CAMPO_CF_SITU_EXAME
)
GROUP BY  DATA
          ,CD_CONVENIO
          ,NM_CONVENIO
          ,CD_PROCEDIMENTO
          ,DS_PROCEDIMENTO
          ,CD_ATENDIMENTO
          ,CD_PACIENTE
          ,NM_PACIENTE
          ,NM_SOCIAL_PACIENTE
)
ORDER BY DATA,
CD_CONVENIO,NM_CONVENIO,
CD_PROCEDIMENTO,DS_PROCEDIMENTO
