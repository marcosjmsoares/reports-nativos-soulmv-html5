--R_ENTR_POR_TIPO_PSDI
SELECT
    LOCAL_PADRAO.CD_LOCAL_PADRAO CD_LOCAL_PADRAO ,
    LOCAL_PADRAO.DS_LOCAL_PADRAO    DS_LOCAL_PADRAO  ,
    DECODE(LOCAL_PADRAO.CD_LOCAL_PADRAO, NULL, 'OUTROS LOCAIS', LOCAL_PADRAO.CD_LOCAL_PADRAO
    ||' - '
    ||LOCAL_PADRAO.DS_LOCAL_PADRAO) LOCAL_PADRAO ,
    PED_RX.CD_ATENDIMENTO CD_ATENDIMENTO         ,
    PED_RX.CD_PED_RX      CD_PED_RX                       ,
    PACIENTE.CD_PACIENTE  CD_PACIENTE                       ,
    PACIENTE.NM_PACIENTE  NM_PACIENTE                       ,
    (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND paciente.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE          ,
    PACIENTE.CD_PACIENTE
    ||' - '
    ||PACIENTE.NM_PACIENTE PACIENTE      ,
    ORI_ATE.DS_ORI_ATE  DS_ORI_ATE                 ,
    ITPED_RX.CD_EXA_RX  CD_EXA_RX                 ,
    ENDER_COLETA_ENTREGA.CD_TIPO_ENTREGA CD_TIPO_ENTREGA ,
    TIPO_ENTREGA.ds_tipo_entrega      ds_tipo_entrega    ,
    ENDER_COLETA_ENTREGA.ds_logradouro  ds_logradouro  ,
    ENDER_COLETA_ENTREGA.nr_logradouro  nr_logradouro ,
    ENDER_COLETA_ENTREGA.ds_complemento ds_complemento ,
    ENDER_COLETA_ENTREGA.nm_bairro   nm_bairro    ,
    ENDER_COLETA_ENTREGA.nr_cep     nr_cep     ,
    ENDER_COLETA_ENTREGA.nr_telefone  nr_telefone   ,
    ENDER_COLETA_ENTREGA.nr_celular   nr_celular   ,
    ENDER_COLETA_ENTREGA.nr_fax   nr_fax       ,
    ENDER_COLETA_ENTREGA.ds_email  ds_email      ,
    ENDER_COLETA_ENTREGA.cd_cidade  cd_cidade     ,
    Cidade.Nm_Cidade  Nm_Cidade
     FROM
    DBAMV.PED_RX                        ,
    DBAMV.ATENDIME                            ,
    DBAMV.ITPED_RX                            ,
    DBAMV.PACIENTE                            ,
    DBAMV.ORI_ATE                             ,
    DBAMV.ENDER_COLETA_ENTREGA                ,
    DBAMV.TIPO_ENTREGA_RESULTADO TIPO_ENTREGA ,
    DBAMV.LOCAL_PADRAO_ENTREGA LOCAL_PADRAO   ,
    dbamv.cidade
    WHERE ATENDIME.CD_ATENDIMENTO          = PED_RX.CD_ATENDIMENTO
  AND ATENDIME.CD_PACIENTE                 = PACIENTE.CD_PACIENTE
  AND ATENDIME.CD_ORI_ATE                  = ORI_ATE.CD_ORI_ATE
  AND PED_RX.CD_PED_RX                     = ENDER_COLETA_ENTREGA.CD_PED_RX
  AND PED_RX.CD_PED_RX                     = ITPED_RX.CD_PED_RX
  AND ENDER_COLETA_ENTREGA.CD_TIPO_ENTREGA = TIPO_ENTREGA.CD_TIPO_ENTREGA
  AND ENDER_COLETA_ENTREGA.CD_LOCAL_PADRAO = LOCAL_PADRAO.CD_LOCAL_PADRAO(+)
  AND ENDER_COLETA_ENTREGA.CD_Cidade       = Cidade.Cd_Cidade
  AND ENDER_COLETA_ENTREGA.DT_SERVICO BETWEEN :CAMPO_DTINIC AND 
  AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY
ENDER_COLETA_ENTREGA.CD_TIPO_ENTREGA ,
TIPO_ENTREGA.ds_tipo_entrega ,
LOCAL_PADRAO.CD_LOCAL_PADRAO ,
LOCAL_PADRAO.DS_LOCAL_PADRAO ,
PED_RX.CD_ATENDIMENTO,
PED_RX.CD_PED_RX ,
PACIENTE.CD_PACIENTE ,
PACIENTE.NM_PACIENTE ,
NM_SOCIAL_PACIENTE ,
ORI_ATE.DS_ORI_ATE ,
ITPED_RX.CD_EXA_RX ,
ENDER_COLETA_ENTREGA.ds_logradouro ,
ENDER_COLETA_ENTREGA.nr_logradouro ,
ENDER_COLETA_ENTREGA.ds_complemento ,
ENDER_COLETA_ENTREGA.nm_bairro ,
ENDER_COLETA_ENTREGA.nr_cep ,
ENDER_COLETA_ENTREGA.nr_telefone ,
ENDER_COLETA_ENTREGA.nr_celular ,
ENDER_COLETA_ENTREGA.nr_fax ,
ENDER_COLETA_ENTREGA.ds_email ,
ENDER_COLETA_ENTREGA.cd_cidade ,
Cidade.Nm_Cidade
ORDER BY ENDER_COLETA_ENTREGA.CD_TIPO_ENTREGA ,
TIPO_ENTREGA.ds_tipo_entrega ,
LOCAL_PADRAO.CD_LOCAL_PADRAO ,
LOCAL_PADRAO.DS_LOCAL_PADRAO ,
PED_RX.CD_ATENDIMENTO,
ITPED_RX.CD_EXA_RX ,
PED_RX.CD_PED_RX ,
PACIENTE.CD_PACIENTE ,
PACIENTE.NM_PACIENTE
