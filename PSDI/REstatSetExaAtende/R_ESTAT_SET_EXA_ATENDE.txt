--R_ESTAT_SET_EXA_ATENDE
SELECT * FROM (
SELECT  SET_EXA.NM_SET_EXA                NM_SET_EXA
               ,DECODE(ATENDIME.TP_ATENDIMENTO, 'A', 'AMBULATORIAL'
                                                                                     , 'E', 'EXTERNO'
                                                                                     , 'H', 'HOME CARE'
                                                                                     , 'I',  'INTERNAÇÃO'
                                                                                     , 'U', 'URGÊNCIA/EMERGÊNCIA'
                                                                                     , 'S', 'SUS', '' )     TP_ATENDIMENTO
               ,ITPED_RX.CD_EXA_RX                CD_EXA_RX
               ,EXA_RX.DS_EXA_RX                    DS_EXA_RX
               ,COUNT(*)                                        QTDE
FROM   DBAMV.ATENDIME
            ,DBAMV.PED_RX
            ,DBAMV.SET_EXA
            ,DBAMV.ITPED_RX
            ,DBAMV.EXA_RX
WHERE ATENDIME.CD_ATENDIMENTO  =  PED_RX.CD_ATENDIMENTO
AND       PED_RX.CD_SET_EXA  =  SET_EXA.CD_SET_EXA
AND       ITPED_RX.CD_PED_RX  =  PED_RX.CD_PED_RX
AND       EXA_RX.CD_EXA_RX  =  ITPED_RX.CD_EXA_RX
AND      ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY  SET_EXA.NM_SET_EXA
                    ,DECODE(ATENDIME.TP_ATENDIMENTO, 'A', 'AMBULATORIAL'
                                                                                          , 'E', 'EXTERNO'
                                                                                          , 'H', 'HOME CARE'
                                                                                          , 'I',  'INTERNAÇÃO'
                                                                                          , 'U', 'URGÊNCIA/EMERGÊNCIA'
                                                                                          , 'S', 'SUS', '' )
                    ,ITPED_RX.CD_EXA_RX
                    ,EXA_RX.DS_EXA_RX
)
ORDER BY NM_SET_EXA,
TP_ATENDIMENTO ,CD_EXA_RX,DS_EXA_RX
