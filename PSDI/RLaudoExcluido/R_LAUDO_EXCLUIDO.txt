--R_LAUDO_EXCLUIDO
SELECT * FROM
  (SELECT log_exc_laudo_rx.dt_log_exc_laudo ,
          to_char(log_exc_laudo_rx.dt_log_exc_laudo, 'dd/mm/yyyy') dt_exclusao ,
          log_exc_laudo_rx.cd_atendimento ,
          paciente.cd_paciente ,
          paciente.nm_paciente ,
       (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
          to_char(log_exc_laudo_rx.hr_log_exc_laudo, 'hh24') hr_log_exc_laudo ,
          log_exc_laudo_rx.cd_ped_rx ,
          log_exc_laudo_rx.cd_laudo ,
          log_exc_laudo_rx.ds_motivo ,
          log_exc_laudo_rx.nm_usuario
   FROM dbamv.log_exc_laudo_rx ,
        dbamv.atendime ,
        dbamv.paciente
   WHERE atendime.cd_atendimento = log_exc_laudo_rx.cd_atendimento
     AND log_exc_laudo_rx.dt_log_exc_laudo BETWEEN to_date(:CAMPO_DT_INICIAL, 'dd/mm/yyyy') AND to_date(:CAMPO_DT_FINAL, 'dd/mm/yyyy')
     AND paciente.cd_paciente = atendime.cd_paciente
     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
   ORDER BY 1)
ORDER BY DT_LOG_EXC_LAUDO,
         DT_EXCLUSAO,
         CD_ATENDIMENTO,
         CD_PACIENTE,
         NM_PACIENTE,
         HR_LOG_EXC_LAUDO
