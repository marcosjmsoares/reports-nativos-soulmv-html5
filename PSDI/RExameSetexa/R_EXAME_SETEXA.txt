--R_EXAME_SETEXA
SELECT * FROM (
SELECT SET_EXA.CD_SET_EXA
              ,SET_EXA.NM_SET_EXA
              ,EXA_RX.CD_EXA_RX
              ,EXA_RX.DS_EXA_RX
  ,decode(exa_set.sn_ativo,'S','SIM','N','NÃO') SITUACAO
FROM   DBAMV.EXA_SET
            ,DBAMV.SET_EXA
            ,DBAMV.EXA_RX
WHERE  EXA_SET.CD_SET_EXA = SET_EXA.CD_SET_EXA
AND        EXA_SET.CD_EXA_RX  = EXA_RX.CD_EXA_RX
AND        SET_EXA.CD_SET_EXA IN (SELECT PSDI_SET_EXA.CD_SET_EXA
                                                             FROM    DBAMV.PSDI_SET_EXA
                                                             WHERE NM_USUARIO = USER)
ORDER BY SET_EXA.CD_SET_EXA
)
ORDER BY CD_SET_EXA,NM_SET_EXA
