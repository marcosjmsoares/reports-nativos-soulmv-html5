--R_COMAGE
SELECT ESCALA_DIA.CD_ESCALA_DIA                                         CD_ESCALA_DIA
              ,ESCALA_DIA.DT_ESCALA_DIA                                        DT_ESCALA_DIA
              ,TO_CHAR(ESCALA_DIA.DT_ESCALA_DIA, 'D' )              NM_DIA
              ,TO_CHAR(IT_ESCALA_DIA.HR_AGENDA, 'HH24')    HR_AGENDA
              ,RECURSO.CD_RECURSO                                                CD_RECURSO
              ,RECURSO.NM_RECURSO                                               NM_RECURSO
              ,RECURSO.CD_SET_EXA                                                 CD_SET_EXA
              ,SET_EXA.NM_SET_EXA                                                  NM_SET_EXA
              ,ESCALA_DIA.CD_PRESTADOR                                       CD_PRESTADOR
              ,PRESTADOR.NM_PRESTADOR                                     NM_PRESTADOR
              ,IT_ESCALA_DIA.CD_PACIENTE                                      CD_PACIENTE
              ,IT_ESCALA_DIA.CD_ATENDIMENTO                             CD_ATENDIMENTO
              ,IT_ESCALA_DIA.NM_PACIENTE                                     NM_PACIENTE
              ,IT_ESCALA_DIA.CD_CONVENIO                                     CD_CONVENIO
              ,CONVENIO.NM_CONVENIO                                             NM_CONVENIO
              ,IT_ESCALA_DIA.CD_PRESTADOR                                 CD_PRESTADOR_IT
              ,MEDIC_SOLIC.NM_PRESTADOR                                    MEDIC_SOLIC
              ,IT_ESCALA_DIA.NR_TELEFONE                                     NR_TELEFONE
              ,IT_ESCALA_DIA.NM_USUARIO                                       NM_USUARIO
FROM  dbamv.IT_ESCALA_DIA
           ,dbamv.ESCALA_DIA
           ,dbamv.RECURSO
           ,dbamv.PRESTADOR     MEDIC_SOLIC
           ,dbamv.PRESTADOR
           ,dbamv.SET_EXA
           ,dbamv.CONVENIO
Where TO_CHAR(ESCALA_DIA.CD_ESCALA_DIA)  =  :CAMPO_ESCALA_DIA
AND       TO_CHAR(IT_ESCALA_DIA.HR_AGENDA, 'HH24')  =  :CAMPO_HR_AGENDA
AND       ESCALA_DIA.CD_ESCALA_DIA  =  IT_ESCALA_DIA.CD_ESCALA_DIA
AND       PRESTADOR.CD_PRESTADOR(+)  =  ESCALA_DIA.CD_PRESTADOR
AND       MEDIC_SOLIC.CD_PRESTADOR(+)  =  IT_ESCALA_DIA.CD_PRESTADOR
AND       CONVENIO.CD_CONVENIO(+)  =  IT_ESCALA_DIA.CD_CONVENIO
AND       RECURSO.CD_RECURSO(+)  =  ESCALA_DIA.CD_RECURSO
AND       SET_EXA.CD_SET_EXA(+)  =  RECURSO.CD_SET_EXA
AND       SET_EXA.CD_SET_EXA IN (SELECT PSDI_SET_EXA.CD_SET_EXA
                                                              FROM DBAMV.PSDI_SET_EXA
                                                              WHERE NM_USUARIO = USER)
ORDER BY  ESCALA_DIA.CD_ESCALA_DIA
                    ,IT_ESCALA_DIA.HR_AGENDA
