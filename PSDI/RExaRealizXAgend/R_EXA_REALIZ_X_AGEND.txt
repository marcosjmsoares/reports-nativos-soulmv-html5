--R_EXA_REALIZ_X_AGEND
SELECT                     DESCRICAO
             ,SUM(TOT_AGENDADO)        TOT_AGENDADO
             ,SUM(QT_AGENDADO_01)     QT_AGENDADO_01
             ,SUM(QT_AGENDADO_02)     QT_AGENDADO_02
             ,SUM(QT_AGENDADO_03)     QT_AGENDADO_03
             ,SUM(QT_AGENDADO_04)     QT_AGENDADO_04
             ,SUM(QT_AGENDADO_05)     QT_AGENDADO_05
             ,SUM(QT_AGENDADO_06)     QT_AGENDADO_06
             ,SUM(QT_AGENDADO_07)     QT_AGENDADO_07
             ,SUM(QT_AGENDADO_08)     QT_AGENDADO_08
             ,SUM(QT_AGENDADO_09)     QT_AGENDADO_09
             ,SUM(QT_AGENDADO_10)     QT_AGENDADO_10
             ,SUM(QT_AGENDADO_11)     QT_AGENDADO_11
             ,SUM(QT_AGENDADO_12)     QT_AGENDADO_12
             ,SUM(TOT_REALIZADO)        TOT_REALIZADO
             ,SUM(QT_REALIZADO_01)     QT_REALIZADO_01
             ,SUM(QT_REALIZADO_02)     QT_REALIZADO_02
             ,SUM(QT_REALIZADO_03)     QT_REALIZADO_03
             ,SUM(QT_REALIZADO_04)     QT_REALIZADO_04
             ,SUM(QT_REALIZADO_05)     QT_REALIZADO_05
             ,SUM(QT_REALIZADO_06)     QT_REALIZADO_06
             ,SUM(QT_REALIZADO_07)     QT_REALIZADO_07
             ,SUM(QT_REALIZADO_08)     QT_REALIZADO_08
             ,SUM(QT_REALIZADO_09)     QT_REALIZADO_09
             ,SUM(QT_REALIZADO_10)     QT_REALIZADO_10
             ,SUM(QT_REALIZADO_11)     QT_REALIZADO_11
             ,SUM(QT_REALIZADO_12)     QT_REALIZADO_12
FROM  (
SELECT PRESTADOR.NM_PRESTADOR
             ,CONVENIO.NM_CONVENIO
             ,EXA_RX.DS_EXA_RX
             ,COUNT(*)   TOT_AGENDADO
             ,SUM( DECODE( TO_CHAR(ESCALA_DIA.DT_ESCALA_DIA,'MM'),'01',1,0 ) )  QT_AGENDADO_01
             ,SUM( DECODE( TO_CHAR(ESCALA_DIA.DT_ESCALA_DIA,'MM'),'02',1,0 ) )  QT_AGENDADO_02
             ,SUM( DECODE( TO_CHAR(ESCALA_DIA.DT_ESCALA_DIA,'MM'),'03',1,0 ) )  QT_AGENDADO_03
             ,SUM( DECODE( TO_CHAR(ESCALA_DIA.DT_ESCALA_DIA,'MM'),'04',1,0 ) )  QT_AGENDADO_04
             ,SUM( DECODE( TO_CHAR(ESCALA_DIA.DT_ESCALA_DIA,'MM'),'05',1,0 ) )  QT_AGENDADO_05
             ,SUM( DECODE( TO_CHAR(ESCALA_DIA.DT_ESCALA_DIA,'MM'),'06',1,0 ) )  QT_AGENDADO_06
             ,SUM( DECODE( TO_CHAR(ESCALA_DIA.DT_ESCALA_DIA,'MM'),'07',1,0 ) )  QT_AGENDADO_07
             ,SUM( DECODE( TO_CHAR(ESCALA_DIA.DT_ESCALA_DIA,'MM'),'08',1,0 ) )  QT_AGENDADO_08
             ,SUM( DECODE( TO_CHAR(ESCALA_DIA.DT_ESCALA_DIA,'MM'),'09',1,0 ) )  QT_AGENDADO_09
             ,SUM( DECODE( TO_CHAR(ESCALA_DIA.DT_ESCALA_DIA,'MM'),'10',1,0 ) )  QT_AGENDADO_10
             ,SUM( DECODE( TO_CHAR(ESCALA_DIA.DT_ESCALA_DIA,'MM'),'11',1,0 ) )  QT_AGENDADO_11
             ,SUM( DECODE( TO_CHAR(ESCALA_DIA.DT_ESCALA_DIA,'MM'),'12',1,0 ) )  QT_AGENDADO_12
             ,0          TOT_REALIZADO
             ,0          QT_REALIZADO_01
             ,0          QT_REALIZADO_02
             ,0          QT_REALIZADO_03
             ,0          QT_REALIZADO_04
             ,0          QT_REALIZADO_05
             ,0          QT_REALIZADO_06
             ,0          QT_REALIZADO_07
             ,0          QT_REALIZADO_08
             ,0          QT_REALIZADO_09
             ,0          QT_REALIZADO_10
             ,0          QT_REALIZADO_11
             ,0          QT_REALIZADO_12
FROM  DBAMV.ESCALA_DIA
           ,DBAMV.PRESTADOR
           ,DBAMV.IT_ESCALA_DIA
           ,DBAMV.CONVENIO
           ,DBAMV.EXA_RX_IT_ESCALA_DIA
           ,DBAMV.EXA_RX
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And TO_CHAR(ESCALA_DIA.DT_ESCALA_DIA,'YYYY')  =  :CAMPO_ANO
AND       ESCALA_DIA.CD_PRESTADOR  =  PRESTADOR.CD_PRESTADOR(+)
AND       ESCALA_DIA.CD_ESCALA_DIA  =  IT_ESCALA_DIA.CD_ESCALA_DIA
AND       IT_ESCALA_DIA.CD_CONVENIO  =  CONVENIO.CD_CONVENIO(+)
AND       IT_ESCALA_DIA.CD_ESCALA_DIA  =  EXA_RX_IT_ESCALA_DIA.CD_ESCALA_DIA
AND       IT_ESCALA_DIA.HR_AGENDA  =  EXA_RX_IT_ESCALA_DIA.HR_AGENDA
AND       EXA_RX_IT_ESCALA_DIA.CD_EXA_RX  =  EXA_RX.CD_EXA_RX(+)
AND       nvl(IT_ESCALA_DIA.TP_SITUACAO,'S') in ('A','S') 
GROUP BY PRESTADOR.NM_PRESTADOR
                  ,CONVENIO.NM_CONVENIO
                  ,EXA_RX.DS_EXA_RX
UNION ALL
SELECT PRESTADOR.NM_PRESTADOR
              ,CONVENIO.NM_CONVENIO
              ,EXA_RX.DS_EXA_RX
              ,0  TOT_AGENDADO
              ,0  QT_AGENDADO_01
              ,0  QT_AGENDADO_02
              ,0  QT_AGENDADO_03
              ,0  QT_AGENDADO_04
              ,0  QT_AGENDADO_05
              ,0  QT_AGENDADO_06
              ,0  QT_AGENDADO_07
              ,0  QT_AGENDADO_08
              ,0  QT_AGENDADO_09
              ,0  QT_AGENDADO_10
              ,0  QT_AGENDADO_11
              ,0  QT_AGENDADO_12
              ,COUNT(*)   TOT_REALIZADO
              ,SUM( DECODE( TO_CHAR(ITPED_RX.DT_REALIZADO,'MM'),'01',1,0 ) )  QT_AGENDADO_01
              ,SUM( DECODE( TO_CHAR(ITPED_RX.DT_REALIZADO,'MM'),'02',1,0 ) )  QT_AGENDADO_02
              ,SUM( DECODE( TO_CHAR(ITPED_RX.DT_REALIZADO,'MM'),'03',1,0 ) )  QT_AGENDADO_03
              ,SUM( DECODE( TO_CHAR(ITPED_RX.DT_REALIZADO,'MM'),'04',1,0 ) )  QT_AGENDADO_04
              ,SUM( DECODE( TO_CHAR(ITPED_RX.DT_REALIZADO,'MM'),'05',1,0 ) )  QT_AGENDADO_05
              ,SUM( DECODE( TO_CHAR(ITPED_RX.DT_REALIZADO,'MM'),'06',1,0 ) )  QT_AGENDADO_06
              ,SUM( DECODE( TO_CHAR(ITPED_RX.DT_REALIZADO,'MM'),'07',1,0 ) )  QT_AGENDADO_07
              ,SUM( DECODE( TO_CHAR(ITPED_RX.DT_REALIZADO,'MM'),'08',1,0 ) )  QT_AGENDADO_08
              ,SUM( DECODE( TO_CHAR(ITPED_RX.DT_REALIZADO,'MM'),'09',1,0 ) )  QT_AGENDADO_09
              ,SUM( DECODE( TO_CHAR(ITPED_RX.DT_REALIZADO,'MM'),'10',1,0 ) )  QT_AGENDADO_10
              ,SUM( DECODE( TO_CHAR(ITPED_RX.DT_REALIZADO,'MM'),'11',1,0 ) )  QT_AGENDADO_11
              ,SUM( DECODE( TO_CHAR(ITPED_RX.DT_REALIZADO,'MM'),'12',1,0 ) )  QT_AGENDADO_12
FROM  DBAMV.PED_RX
          , (select cd_prestador,nm_prestador from  DBAMV.PRESTADOR
              union
              select -1, 'Prestador não informado' FROM dual
             )  PRESTADOR
           ,DBAMV.CONVENIO
           ,DBAMV.ITPED_RX
           ,DBAMV.EXA_RX
           ,DBAMV.LAUDO_RX 
           ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And Laudo_RX.CD_laudo(+) = Itped_rx.Cd_laudo 
AND PRESTADOR.CD_PRESTADOR = Decode ( Nvl(itped_rx.cd_prestador,0),0,      
                              Decode(Nvl(laudo_rx.cd_prestador,0),0, -1,laudo_rx.cd_prestador)  
                                    ,itped_rx.cd_prestador)
AND       PED_RX.CD_CONVENIO  =  CONVENIO.CD_CONVENIO
AND       ITPED_RX.CD_PED_RX  =  PED_RX.CD_PED_RX
AND       NVL(ITPED_RX.SN_REALIZADO,'N') = 'S'
AND       TO_CHAR(ITPED_RX.DT_REALIZADO,'YYYY')  =  :CAMPO_ANO
AND       EXA_RX.CD_EXA_RX  =  ITPED_RX.CD_EXA_RX
GROUP BY PRESTADOR.NM_PRESTADOR
                   ,CONVENIO.NM_CONVENIO
                   ,EXA_RX.DS_EXA_RX       )
GROUP BY  
ORDER BY TOT_REALIZADO   DESC
