--R_EXAME_REALIZ_CONV_XY_R_1_Q_EXA_REALIZ
SELECT * FROM (
SELECT  CONVENIO.CD_CONVENIO      CD_CONVENIO,
                CONVENIO.NM_CONVENIO     NM_CONVENIO,
                GRU_PRO.CD_GRU_PRO        CD_GRU_PRO,
                GRU_PRO.DS_GRU_PRO        DS_GRU_PRO,
                COUNT( DECODE( ATENDIME.TP_ATENDIMENTO, 'I', 1, NULL ) )    QTD_I,
                COUNT( DECODE( ATENDIME.TP_ATENDIMENTO, 'A', 1, NULL ) )   QTD_A,
                COUNT( DECODE( ATENDIME.TP_ATENDIMENTO, 'U', 1, NULL ) )   QTD_U,
                COUNT( DECODE( ATENDIME.TP_ATENDIMENTO, 'E', 1, NULL ) )   QTD_E,
                COUNT( DECODE( ATENDIME.TP_ATENDIMENTO, 'S', 1, NULL ) )   QTD_S,
                COUNT( DECODE( ATENDIME.TP_ATENDIMENTO, 'S', null, 'I', null,'A', null,'U', null,'E', null, 1 ) )   QTD_outros,
                COUNT(*) QTD_TOTAL
FROM     DBAMV.CONVENIO,
               DBAMV.PED_RX,
               DBAMV.ITPED_RX,
               DBAMV.EXA_RX,
               DBAMV.PRO_FAT,
               DBAMV.GRU_PRO,
               DBAMV.ATENDIME
WHERE CONVENIO.CD_CONVENIO  =  PED_RX.CD_CONVENIO
AND       ITPED_RX.CD_PED_RX  =  PED_RX.CD_PED_RX
AND       ITPED_RX.CD_EXA_RX  =  EXA_RX.CD_EXA_RX
AND       NVL( PRO_FAT.SN_ATIVO, 'S' )  =  'S'
AND       EXA_RX.EXA_RX_CD_PRO_FAT  =  PRO_FAT.CD_PRO_FAT(+)
AND       PRO_FAT.CD_GRU_PRO  =  GRU_PRO.CD_GRU_PRO(+)
AND       PED_RX.CD_ATENDIMENTO  =  ATENDIME.CD_ATENDIMENTO
AND       ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY  CONVENIO.CD_CONVENIO,
                    CONVENIO.NM_CONVENIO,
                    GRU_PRO.CD_GRU_PRO,
                    GRU_PRO.DS_GRU_PRO
)
ORDER BY CD_CONVENIO, DS_GRU_PRO
