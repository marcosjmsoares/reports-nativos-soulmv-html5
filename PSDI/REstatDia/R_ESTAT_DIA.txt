--R_ESTAT_DIA
SELECT DATA, CD_EXA_RX, DS_EXA_RX, COUNT(*)    QTDE
 FROM (
SELECT DECODE(:CAMPO_SITU_EXAME, 'S' , TRUNC(PED_RX.DT_PEDIDO), 'N', TRUNC(PED_RX.DT_PEDIDO), TRUNC(ITPED_RX.DT_REALIZADO))  DATA
              ,ITPED_RX.CD_EXA_RX
              ,EXA_RX.DS_EXA_RX
FROM  DBAMV.ITPED_RX
           ,DBAMV.EXA_RX
           ,DBAMV.CONVENIO
           ,DBAMV.PED_RX
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And EXA_RX.CD_EXA_RX   = ITPED_RX.CD_EXA_RX
AND       ITPED_RX.CD_PED_RX = PED_RX.CD_PED_RX
AND       PED_RX.CD_CONVENIO = CONVENIO.CD_CONVENIO
:CAMPO_CF_COND_PERIODO
)
GROUP BY  DATA
         ,CD_EXA_RX
         ,DS_EXA_RX
ORDER BY DATA
