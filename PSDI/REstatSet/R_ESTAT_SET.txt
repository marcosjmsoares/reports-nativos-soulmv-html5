--R_ESTAT_SET
SELECT * FROM (
SELECT  PED_RX.CD_SETOR
              ,SETOR.NM_SETOR
              ,ITPED_RX.CD_EXA_RX
              ,ITPED_RX.SN_REALIZADO
              ,EXA_RX.DS_EXA_RX
              ,COUNT(*)            QTDE
FROM    DBAMV.ITPED_RX
             ,DBAMV.EXA_RX
             ,DBAMV.SETOR
             ,DBAMV.PED_RX
WHERE  EXA_RX.CD_EXA_RX  =  ITPED_RX.CD_EXA_RX
AND        ITPED_RX.CD_PED_RX  =  PED_RX.CD_PED_RX
AND        SETOR.CD_SETOR  =  PED_RX.CD_SETOR
AND        SETOR.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
GROUP BY  PED_RX.CD_SETOR
                   ,SETOR.NM_SETOR
                   ,ITPED_RX.CD_EXA_RX
                   ,ITPED_RX.SN_REALIZADO
                   ,EXA_RX.DS_EXA_RX
)ORDER BY NM_SETOR,CD_SETOR,CD_EXA_RX,DS_EXA_RX,SN_REALIZADO,QTDE
