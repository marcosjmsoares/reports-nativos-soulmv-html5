--R_EXA_REALIZ_DIA
SELECT * FROM (
SELECT EXAMES_DIA.DT_REALIZADO
              ,EXAMES_DIA.CD_EXA_RX
              ,EXAMES_DIA.DS_EXA_RX
              ,COUNT(*)   QT_EXAME
FROM DBAMV.PRESTADOR,
            (SELECT trunc(ITPED_RX.DT_REALIZADO) DT_REALIZADO
                           ,EXA_RX.CD_EXA_RX
                           ,EXA_RX.DS_EXA_RX
                           ,DECODE(LAUDO_RX.CD_PRESTADOR, NULL, ITPED_RX.CD_PRESTADOR,
                                                                                                           LAUDO_RX.CD_PRESTADOR)  CD_PRESTADOR
            FROM      DBAMV.PED_RX
                           ,DBAMV.ITPED_RX
                           ,DBAMV.EXA_RX
                           ,DBAMV.ATENDIME
                           ,DBAMV.LAUDO_RX
          WHERE    PED_RX.CD_PED_RX                         = ITPED_RX.CD_PED_RX
                AND    PED_RX.CD_ATENDIMENTO             = ATENDIME.CD_ATENDIMENTO
                AND    ATENDIME.CD_MULTI_EMPRESA    = :CAMPO_CD_MULTI_EMPRESA
                AND    ITPED_RX.SN_REALIZADO               = 'S'
                AND   TRUNC(ITPED_RX.DT_REALIZADO)  BETWEEN  :CAMPO_DT_INI  AND  
                AND   ITPED_RX.CD_EXA_RX                      = EXA_RX.CD_EXA_RX
                AND   ITPED_RX.CD_LAUDO                        = LAUDO_RX.CD_LAUDO(+)
                 AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
                                                                   ) EXAMES_DIA
WHERE  EXAMES_DIA.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
GROUP BY  EXAMES_DIA.DT_REALIZADO
                   ,EXAMES_DIA.CD_EXA_RX
                   ,EXAMES_DIA.DS_EXA_RX
)
ORDER BY DT_REALIZADO
