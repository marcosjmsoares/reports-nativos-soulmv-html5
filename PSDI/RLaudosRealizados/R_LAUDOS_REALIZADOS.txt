--R_LAUDOS_REALIZADOS
SELECT * FROM (
SELECT LAUDO_RX.CD_LAUDO
              ,EXA_RX.DS_EXA_RX
              ,ITPED_RX.DT_REALIZADO
              ,PACIENTE.CD_PACIENTE
              ,PACIENTE.NM_PACIENTE
              ,PRESTADOR.NM_PRESTADOR  NM_PRESTA_SOLIC
              ,PRESTA.NM_PRESTADOR     NM_PRESTA_EXEC
FROM DBAMV.ATENDIME
          ,DBAMV.PACIENTE
          ,DBAMV.PED_RX
          ,DBAMV.PRESTADOR
          ,DBAMV.ITPED_RX
          ,DBAMV.EXA_RX
          ,DBAMV.LAUDO_RX
          ,DBAMV.PRESTADOR  PRESTA
WHERE ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND       ATENDIME.CD_ATENDIMENTO = PED_RX.CD_ATENDIMENTO
AND       PED_RX.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
AND       PED_RX.CD_PED_RX = ITPED_RX.CD_PED_RX
AND       ITPED_RX.SN_REALIZADO = 'S'
AND       TRUNC(ITPED_RX.DT_REALIZADO) BETWEEN :CAMPO_DT_INI AND 
AND       ITPED_RX.CD_EXA_RX = EXA_RX.CD_EXA_RX
AND       ITPED_RX.CD_PED_RX = LAUDO_RX.CD_PED_RX
AND       ITPED_RX.CD_LAUDO = LAUDO_RX.CD_LAUDO
AND       LAUDO_RX.CD_PRESTADOR = PRESTA.CD_PRESTADOR
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
ORDER BY ITPED_RX.DT_REALIZADO
)
ORDER BY DT_REALIZADO
