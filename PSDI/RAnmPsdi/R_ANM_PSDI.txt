       ,PACIENTE.NM_PACIENTE                         NM_PACIENTE
       ,PACIENTE.DT_NASCIMENTO                    DT_NASCIMENTO
       ,PACIENTE.TP_SEXO                                  TP_SEXO
       ,PACIENTE.TP_COR                                    TP_COR
       ,DBAMV.FNC_ESTADO_CIVIL(PACIENTE.TP_ESTADO_CIVIL) TP_ESTADO_CIVIL1
       ,PACIENTE.NR_DOCUMENTO                    NR_DOCUMENTO
       ,PACIENTE.NM_MAE                                   NM_MAE
       ,PACIENTE.NM_PAI                                     NM_PAI
       ,PACIENTE.DS_ENDERECO                        DS_ENDERECO
       ,PACIENTE.NR_ENDERECO                       NR_ENDERECO
       ,PACIENTE.DS_COMPLEMENTO                DS_COMPLEMENTO
       ,PACIENTE.NM_BAIRRO                             NM_BAIRRO
       ,PACIENTE.NR_CEP                                    NR_CEP
       ,PACIENTE.NR_FONE                                  NR_FONE
       ,PACIENTE.DT_CADASTRO                        DT_CADASTRO
       ,PACIENTE.NR_CPF                                    NR_CPF
       ,PACIENTE.NR_IDENTIDADE                     NR_IDENTIDADE
       ,PACIENTE.DS_OM_IDENTIDADE             DS_OM_IDENTIDADE
       ,ATENDIME.CD_ATENDIMENTO                CD_ATENDIMENTO
       ,ATENDIME.CD_PACIENTE                         CD_PACIENTE
       ,ATENDIME.DT_ATENDIMENTO                DT_ATENDIMENTO
       ,ATENDIME.HR_ATENDIMENTO                HR_ATENDIMENTO
       ,ATENDIME.CD_SERVICO                          CD_SERVICO
       ,ATENDIME.CD_SER_DIS                          CD_SER_DIS
       ,ATENDIME.TP_ATENDIMENTO               TP_ATENDIMENTO
       ,ATENDIME.CD_DES_ATE                         CD_DES_ATE
       ,ATENDIME.NM_USUARIO                         NM_USUARIO
       ,ATENDIME.CD_CONVENIO                       CD_CONVENIO
       ,ATENDIME.NR_CHAMADA_PAINEL          NR_CHAMADA_PAINEL
       ,ATENDIME_INFO.DS_INFO_ATENDIMENTO                                  DS_INFO_ATENDIMENTO
       ,DECODE( NVL(ATENDIME.SN_RETORNO, 'N'), 'N', 'NÃO', 'SIM')   SN_RETORNO
       ,ORI_ATE.DS_ORI_ATE                              DS_ORI_ATE
       ,ATENDIME.CD_PRO_INT                          CD_PRO_INT
       ,DBAMV.PROCEDIAS(ATENDIME.CD_CONVENIO, ATENDIME.CD_PRO_INT, 'D')    DS_PRO_INT
       ,PRESTADOR.CD_PRESTADOR                CD_PRESTADOR
       ,PRESTADOR.NM_PRESTADOR               NM_PRESTADOR
       ,PRESTADOR.DS_CODIGO_CONSELHO   DS_CODIGO_CONSELHO
       ,LEITO.DS_ENFERMARIA                            DS_ENFERMARIA
       ,LEITO.DS_LEITO                                         DS_LEITO
       ,CONVENIO.NM_CONVENIO                       NM_CONVENIO
       ,CID.DS_CID                                                 DS_CID
       ,SERVICO.DS_SERVICO                              DS_SERVICO
       ,Nvl(DES_ATE.TP_ANAMNESE,0)               TP_ANAM_SERVICO
       ,SER_DIS.DS_SER_DIS                               DS_SER_DIS
       ,Nvl(DES_ATE.TP_ANAMNESE,0)               TP_ANAM_SER_DIS
       ,DES_ATE.DS_DES_ATE                             DS_DES_ATE
       ,TIP_ACOM.DS_TIP_ACOM                          DS_TIP_ACOM
       ,RESPONSA.NM_RESPONSAVEL                NM_RESPONSAVEL
       ,RESPONSA.NR_FONE                                 NR_FONE_RESPONSAVEL
       ,RESPONSA.DS_DOCUMENTO                    DS_DOCUMENTO_RESPONSAVEL
       ,RESPONSA.DS_ENDERECO                       DS_ENDERECO_RESPONSAVEL
       ,RESPONSA.NR_CPF                                    NR_CPF_RESP
       ,RESPONSA.DS_CONTATO_FAMILIA          DS_CONTATO_FAMILIA
       ,CARTEIRA.NR_CARTEIRA                           NR_CARTEIRA
       ,CARTEIRA.NM_EMPRESA                           NM_EMPRESA
       ,CARTEIRA.NM_TITULAR                             NM_TITULAR
       ,CARTEIRA.DT_VALIDADE                           DT_VALIDADE
       ,CARTEIRA.DT_ULT_PGTO                          DT_ULT_PGTO
       ,CON_PLA.CD_CON_PLA                              CD_CON_PLA
       ,CON_PLA.DS_CON_PLA                              DS_CON_PLA
       ,CIDADE.NM_CIDADE                                   NM_CIDADE
       ,CIDADE.CD_UF                                            CD_UF
       ,TIPO_INTERNACAO.CD_TIPO_INTERNACAO     CD_TIPO_INTERNACAO
       ,TIPO_INTERNACAO.DS_TIPO_INTERNACAO      DS_TIPO_INTERNACAO
       ,PED_RX.CD_PED_RX                                  CD_PED_RX
       ,SET_EXA.NM_SET_EXA                              NM_SET_EXA
       ,SETOR.NM_SETOR                                     NM_SETOR
FROM   DBAMV.PED_RX
      ,DBAMV.PACIENTE
      ,DBAMV.CIDADE
      ,DBAMV.ATENDIME
      ,DBAMV.ORI_ATE
      ,DBAMV.PRESTADOR
      ,DBAMV.LEITO
      ,DBAMV.CONVENIO
      ,DBAMV.CID
      ,DBAMV.SERVICO
      ,DBAMV.SER_DIS
      ,DBAMV.TIP_ACOM
      ,DBAMV.RESPONSA
      ,DBAMV.CARTEIRA
      ,DBAMV.CON_PLA
      ,DBAMV.DES_ATE
      ,DBAMV.TIPO_INTERNACAO
      ,DBAMV.SET_EXA
      ,DBAMV.SETOR
      ,DBAMV.ATENDIME_INFO
WHERE PED_RX.CD_SET_EXA = SET_EXA.CD_SET_EXA(+)
  AND       PED_RX.CD_SETOR = SETOR.CD_SETOR(+)
  AND       ATENDIME.CD_ATENDIMENTO  =  PED_RX.CD_ATENDIMENTO (+)
  AND       ATENDIME.CD_PACIENTE  =  PACIENTE.CD_PACIENTE
  AND       PACIENTE.CD_CIDADE  =  CIDADE.CD_CIDADE(+)
  AND       DES_ATE.CD_DES_ATE(+)  =  ATENDIME.CD_DES_ATE
  AND       ATENDIME.CD_ORI_ATE =  ORI_ATE.CD_ORI_ATE (+)
  AND       ATENDIME.CD_PRESTADOR  =  PRESTADOR.CD_PRESTADOR(+)
  AND       ATENDIME.CD_LEITO  =  LEITO.CD_LEITO(+)
  AND       ATENDIME.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
  AND       ATENDIME.CD_CONVENIO = CON_PLA.CD_CONVENIO(+)
  AND       ATENDIME.CD_CON_PLA = CON_PLA.CD_CON_PLA(+)
  AND       ATENDIME.CD_CID  =  CID.CD_CID (+)
  AND       ATENDIME.CD_SER_DIS  =  SERVICO.CD_SERVICO (+)
  AND       ATENDIME.CD_SER_DIS  =  SER_DIS.CD_SER_DIS (+)
  AND       ATENDIME.CD_TIP_ACOM  =  TIP_ACOM.CD_TIP_ACOM (+)
  AND       ATENDIME.CD_ATENDIMENTO  =  RESPONSA.CD_ATENDIMENTO (+)
  AND       ATENDIME.CD_PACIENTE  =  CARTEIRA.CD_PACIENTE (+)
  AND       ATENDIME.CD_CON_PLA  =  CARTEIRA.CD_CON_PLA (+)
  AND       ATENDIME.NR_CARTEIRA          =  CARTEIRA.NR_CARTEIRA(+)
  AND       ATENDIME.CD_TIPO_INTERNACAO  =  TIPO_INTERNACAO.CD_TIPO_INTERNACAO(+)
  AND       ATENDIME.CD_ATENDIMENTO  =  ATENDIME_INFO.CD_ATENDIMENTO(+)
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
