--R_EXANENT
SELECT PED_RX.CD_PED_RX CD_PED_RX ,
       PED_RX.DT_PEDIDO DT_PEDIDO ,
       ATENDIME.CD_ORI_ATE CD_ORI_ATE ,
       ORI_ATE.DS_ORI_ATE DS_ORI_ATE ,
       ATENDIME.CD_PACIENTE CD_PACIENTE ,
       PACIENTE.NM_PACIENTE NM_PACIENTE ,
       (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
       ATENDIME.CD_CONVENIO CD_CONVENIO ,
       CONVENIO.NM_CONVENIO NM_CONVENIO ,
       SETOR.NM_SETOR
FROM dbamv.LAUDO_RX ,
     dbamv.PED_RX ,
     dbamv.ITPED_RX ,
     dbamv.ATENDIME ,
     dbamv.ORI_ATE ,
     dbamv.PACIENTE ,
     dbamv.CONVENIO ,
     dbamv.SETOR,
     Dbamv.Empresa_Convenio
WHERE Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
  AND Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
  AND LAUDO_RX.CD_LAUDO NOT IN
    (SELECT ITENT_PSDI.CD_LAUDO
     FROM DBAMV.ITENT_PSDI
     WHERE ITENT_PSDI.CD_LAUDO = LAUDO_RX.CD_LAUDO )
  AND TRUNC(PED_RX.DT_PEDIDO) BETWEEN :CAMPO_DT_INIC AND 
  AND PED_RX.CD_SET_EXA IN
    (SELECT PSDI_SET_EXA.CD_SET_EXA
     FROM DBAMV.PSDI_SET_EXA
     WHERE PSDI_SET_EXA.NM_USUARIO = USER
       AND PSDI_SET_EXA.CD_SET_EXA = PED_RX.CD_SET_EXA)
  AND ATENDIME.CD_ATENDIMENTO = PED_RX.CD_ATENDIMENTO
  AND ORI_ATE.CD_ORI_ATE = ATENDIME.CD_ORI_ATE
  AND PACIENTE.CD_PACIENTE = ATENDIME.CD_PACIENTE
  AND CONVENIO.CD_CONVENIO = ATENDIME.CD_CONVENIO
  AND SETOR.CD_SETOR = PED_RX.CD_SETOR
  AND ITPED_RX.CD_PED_RX = PED_RX.CD_PED_RX
  AND LAUDO_RX.CD_LAUDO = ITPED_RX.CD_LAUDO
  AND LAUDO_RX.SN_ENTREGUE <> 'S'
  AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY PED_RX.CD_PED_RX ,
         PED_RX.DT_PEDIDO ,
         ATENDIME.CD_ORI_ATE ,
         ORI_ATE.DS_ORI_ATE ,
         ATENDIME.CD_PACIENTE ,
         PACIENTE.NM_PACIENTE ,
         ATENDIME.CD_CONVENIO ,
         CONVENIO.NM_CONVENIO ,
         SETOR.NM_SETOR
ORDER BY PED_RX.DT_PEDIDO ,
         ATENDIME.CD_ORI_ATE ,
         ORI_ATE.DS_ORI_ATE ,
         ATENDIME.CD_PACIENTE ,
         PACIENTE.NM_PACIENTE ,
         ATENDIME.CD_CONVENIO ,
         CONVENIO.NM_CONVENIO
