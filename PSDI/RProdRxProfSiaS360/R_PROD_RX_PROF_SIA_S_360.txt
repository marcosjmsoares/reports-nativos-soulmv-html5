--R_PROD_RX_PROF_SIA_S_360
SELECT *
  FROM (SELECT PRODUCAO.CD_SET_EXA
              ,PRODUCAO.NM_SET_EXA
              ,PRODUCAO.CD_PRESTADOR
              ,PRESTADOR.NM_PRESTADOR
              ,PRODUCAO.CD_CONVENIO
              ,PRODUCAO.NM_CONVENIO
              ,PRODUCAO.CONTADOR
              ,PRODUCAO.VALOR
          FROM DBAMV.PRESTADOR
             ,(
              SELECT SET_EXA.CD_SET_EXA                          CD_SET_EXA
                    ,SET_EXA.NM_SET_EXA                          NM_SET_EXA
                    ,DECODE(LAUDO_RX.CD_PRESTADOR, NULL, NVL(ITPED_RX.CD_PRESTADOR,PED_RX.CD_PRESTADOR),
                                                         LAUDO_RX.CD_PRESTADOR)  CD_PRESTADOR
                    ,CONVENIO.CD_CONVENIO                        CD_CONVENIO
                    ,CONVENIO.NM_CONVENIO                        NM_CONVENIO
                    ,SUM(1)                                      CONTADOR
                    ,SUM(DBAMV.CALC_VL_PROC_UNIT( CONVENIO.TP_CONVENIO , DECODE( convenio.tp_convenio, 'A', EXA_RX.cd_procedimento_sia ,
                                                                                                      'H', EXA_RX.cd_procedimento_sih ,
                                                                                                      'C', EXA_RX.exa_rx_cd_pro_fat ,
                                                                                                      'P', EXA_RX.exa_rx_cd_pro_fat ) ,
                        PED_RX.DT_PEDIDO ,
                        PED_RX.HR_PEDIDO ,
                        CONVENIO.CD_CONVENIO ,
                        NVL(PED_RX.CD_CON_PLA,1) ,
                        ATENDIME.TP_ATENDIMENTO ,
                        ATENDIME.CD_TIP_ACOM ,'U' )
                       ) VALOR
                  FROM   DBAMV.ATENDIME
                        ,DBAMV.PED_RX
                        ,DBAMV.CONVENIO
                        ,DBAMV.SET_EXA
                        ,DBAMV.ITPED_RX
                        ,DBAMV.LAUDO_RX
                        ,DBAMV.EXA_RX
                        ,DBAMV.EMPRESA_CONVENIO
                 Where  Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
                   And  Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
                   And  ATENDIME.CD_ATENDIMENTO  =  PED_RX.CD_ATENDIMENTO
                   AND  CONVENIO.TP_CONVENIO IN  ('A', 'H')
                   AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
                   AND  PED_RX.CD_CONVENIO  =  CONVENIO.CD_CONVENIO
                   AND  PED_RX.CD_SET_EXA  =  SET_EXA.CD_SET_EXA
                   AND  PED_RX.CD_SET_EXA IN ( SELECT PSDI_SET_EXA.CD_SET_EXA
                                                 FROM DBAMV.PSDI_SET_EXA
                                                WHERE PSDI_SET_EXA.NM_USUARIO = USER )
                                                  AND PED_RX.CD_PED_RX  =  ITPED_RX.CD_PED_RX
                                                  AND ITPED_RX.SN_REALIZADO = 'S'
                                                  AND TRUNC(ITPED_RX.DT_REALIZADO)  BETWEEN  :CAMPO_DT_INICIAL  AND  
                                                  AND ITPED_RX.CD_PED_RX  =  LAUDO_RX.CD_PED_RX(+)
                                                  AND ITPED_RX.CD_LAUDO  =  LAUDO_RX.CD_LAUDO(+)
                                                  AND ITPED_RX.CD_EXA_RX  =  EXA_RX.CD_EXA_RX
                                  GROUP BY SET_EXA.CD_SET_EXA
                                          ,SET_EXA.NM_SET_EXA
                                          ,DECODE(LAUDO_RX.CD_PRESTADOR, NULL, NVL(ITPED_RX.CD_PRESTADOR,PED_RX.CD_PRESTADOR)
                                                                             , LAUDO_RX.CD_PRESTADOR)
                                          ,CONVENIO.CD_CONVENIO
                                          ,CONVENIO.NM_CONVENIO
               )   PRODUCAO
          WHERE   PRODUCAO.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
            :CAMPO_SQL_CD_PRESTADOR
        )
ORDER BY CD_SET_EXA,
         NM_SET_EXA,
         CD_PRESTADOR,
         NM_PRESTADOR
