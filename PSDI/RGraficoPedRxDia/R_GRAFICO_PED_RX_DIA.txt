--R_GRAFICO_PED_RX_DIA
SELECT CD_CIDADE,
               '1' A
FROM DBAMV.CIDADE WHERE ROWNUM <= 31

              ,SUM(DECODE(TO_CHAR(PED_RX.DT_PEDIDO,'DD'), '04',1,0))  DIA4
              ,SUM(DECODE(TO_CHAR(PED_RX.DT_PEDIDO,'DD'), '05',1,0))  DIA5
              ,SUM(DECODE(TO_CHAR(PED_RX.DT_PEDIDO,'DD'), '06',1,0))  DIA6
              ,SUM(DECODE(TO_CHAR(PED_RX.DT_PEDIDO,'DD'), '07',1,0))  DIA7
              ,SUM(DECODE(TO_CHAR(PED_RX.DT_PEDIDO,'DD'), '08',1,0))  DIA8
              ,SUM(DECODE(TO_CHAR(PED_RX.DT_PEDIDO,'DD'), '09',1,0))  DIA9
              ,SUM(DECODE(TO_CHAR(PED_RX.DT_PEDIDO,'DD'), '10',1,0))  DIA10
              ,SUM(DECODE(TO_CHAR(PED_RX.DT_PEDIDO,'DD'), '11',1,0))  DIA11
              ,SUM(DECODE(TO_CHAR(PED_RX.DT_PEDIDO,'DD'), '12',1,0))  DIA12
              ,SUM(DECODE(TO_CHAR(PED_RX.DT_PEDIDO,'DD'), '13',1,0))  DIA13
              ,SUM(DECODE(TO_CHAR(PED_RX.DT_PEDIDO,'DD'), '14',1,0))  DIA14
              ,SUM(DECODE(TO_CHAR(PED_RX.DT_PEDIDO,'DD'), '15',1,0))  DIA15
              ,SUM(DECODE(TO_CHAR(PED_RX.DT_PEDIDO,'DD'), '16',1,0))  DIA16
              ,SUM(DECODE(TO_CHAR(PED_RX.DT_PEDIDO,'DD'), '17',1,0))  DIA17
              ,SUM(DECODE(TO_CHAR(PED_RX.DT_PEDIDO,'DD'), '18',1,0))  DIA18
              ,SUM(DECODE(TO_CHAR(PED_RX.DT_PEDIDO,'DD'), '19',1,0))  DIA19

              ,SUM(1)  TOT
FROM     DBAMV.PED_RX
              ,DBAMV.ATENDIME
WHERE  PED_RX.DT_PEDIDO BETWEEN  TO_DATE( :CAMPO_DT_INICIAL, 'DD/MM/YYYY' ) AND  TO_DATE( :CAMPO_DT_FINAL, 'DD/MM/YYYY')
      AND ATENDIME.CD_ATENDIMENTO = PED_RX.CD_ATENDIMENTO
    AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
