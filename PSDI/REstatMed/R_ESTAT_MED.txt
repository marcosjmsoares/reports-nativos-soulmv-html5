--R_ESTAT_MED
SELECT  PRESTADOR.NM_PRESTADOR   NM_PRESTADOR
               ,ITPED_RX.CD_EXA_RX                CD_EXA_RX
               ,EXA_RX.DS_EXA_RX                    DS_EXA_RX
               ,COUNT(*)                                        QTDE
FROM  DBAMV.PRESTADOR
            ,DBAMV.PED_RX
            ,DBAMV.ITPED_RX
            ,DBAMV.EXA_RX
WHERE PED_RX.CD_PRESTADOR  =  PRESTADOR.CD_PRESTADOR(+)
AND       ITPED_RX.CD_PED_RX  =  PED_RX.CD_PED_RX
AND       EXA_RX.CD_EXA_RX  =  ITPED_RX.CD_EXA_RX
GROUP BY  PRESTADOR.NM_PRESTADOR
         ,ITPED_RX.CD_EXA_RX
         ,EXA_RX.DS_EXA_RX
ORDER BY NM_PRESTADOR,CD_EXA_RX
