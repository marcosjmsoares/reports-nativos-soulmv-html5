--R_PEDIDO_EXCLUIDO
SELECT * FROM (
Select log_exc_ped_rx.dt_log_exc_ped_rx
      ,to_char( log_exc_ped_rx.dt_log_exc_ped_rx, 'dd/mm/yyyy') dt_exclusao
      ,to_char( log_exc_ped_rx.hr_log_exc_ped_rx, 'HH24') hr_log_exc_ped_rx
      ,log_exc_ped_rx.cd_atendimento
      ,paciente.cd_paciente
      ,paciente.nm_paciente
	  ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND paciente.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE
      ,log_exc_ped_rx.cd_ped_rx
      ,log_exc_ped_rx.ds_motivo
      ,log_exc_ped_rx.nm_usuario
      ,log_exc_ped_rx.cd_mot_exc
	  ,mot_exc_rx.ds_mot_exc
 From dbamv.log_exc_ped_rx
	 ,dbamv.atendime
	 ,dbamv.paciente
	 ,dbamv.mot_exc_rx 
Where log_exc_ped_rx.dt_log_exc_ped_rx between to_date( :CAMPO_DT_INICIAL, 'DD/MM/YYYY') And to_Date( :CAMPO_DT_FINAL, 'DD/MM/YYYY')
  And atendime.cd_atendimento = log_exc_ped_rx.cd_atendimento
  And paciente.cd_paciente = atendime.cd_paciente
  AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
  AND log_exc_ped_rx.cd_mot_exc = mot_exc_rx.cd_mot_exc(+) 
Order by 1
)
ORDER BY DT_LOG_EXC_PED_RX,DT_EXCLUSAO,
HR_LOG_EXC_PED_RX,CD_ATENDIMENTO,CD_PACIENTE,NM_PACIENTE
