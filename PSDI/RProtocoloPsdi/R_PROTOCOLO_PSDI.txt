--R_PROTOCOLO_PSDI
SELECT * FROM (
SELECT  ENT_PSDI.CD_ENT_PSDI
       ,ENT_PSDI.NM_RESPONSAVEL
       ,ENT_PSDI.DT_ENTREGA
       ,ENT_PSDI.HR_ENTREGA
       ,ITENT_PSDI.CD_LAUDO
       ,LAUDO_RX.DT_LAUDO
       ,ITPED_RX.CD_EXA_RX
       ,PED_RX.CD_PED_RX
       ,PED_RX.CD_ATENDIMENTO
       ,PACIENTE.CD_PACIENTE
       ,PACIENTE.NM_PACIENTE
       ,EXA_RX.DS_EXA_RX
FROM    DBAMV.ENT_PSDI
       ,DBAMV.ITENT_PSDI
       ,DBAMV.LAUDO_RX
       ,DBAMV.ITPED_RX
       ,DBAMV.PED_RX
       ,DBAMV.EXA_RX
       ,DBAMV.ATENDIME
       ,DBAMV.PACIENTE
WHERE  ENT_PSDI.CD_ENT_PSDI    = :CAMPO_ENTREGA                  AND
                ENT_PSDI.CD_SET_EXA IN ( SELECT PSDI_SET_EXA.CD_SET_EXA
                                                                FROM DBAMV.PSDI_SET_EXA
                                                                WHERE PSDI_SET_EXA.NM_USUARIO = USER ) AND
       ITENT_PSDI.CD_ENT_PSDI  = ENT_PSDI.CD_ENT_PSDI  AND
       LAUDO_RX.CD_LAUDO       = ITENT_PSDI.CD_LAUDO   AND
       ITPED_RX.CD_LAUDO       = LAUDO_RX.CD_LAUDO     AND
       PED_RX.CD_PED_RX        = ITPED_RX.CD_PED_RX    AND
       ATENDIME.CD_ATENDIMENTO = PED_RX.CD_ATENDIMENTO AND
       PACIENTE.CD_PACIENTE    = ATENDIME.CD_PACIENTE  AND
       ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  AND
       EXA_RX.CD_EXA_RX        = ITPED_RX.CD_EXA_RX
ORDER BY  CD_PED_RX
         ,CD_LAUDO
)
ORDER BY CD_ENT_PSDI,NM_RESPONSAVEL,DT_ENTREGA,HR_ENTREGA,
CD_PED_RX,CD_ATENDIMENTO,CD_PACIENTE,NM_PACIENTE
