--R_EXAME_CONVENIO
SELECT * FROM
  (SELECT NVL(PRESTADOR.CD_PRESTADOR, 0) CD_PRESTADOR ,
          NVL(PRESTADOR.NM_PRESTADOR, 'NÃO INFORMADO') NM_PRESTADOR ,
          CONVENIO.CD_CONVENIO CD_CONVENIO ,
          CONVENIO.NM_CONVENIO NM_CONVENIO ,
          PACIENTE.CD_PACIENTE CD_PACIENTE ,
          PACIENTE.NM_PACIENTE NM_PACIENTE ,
          (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE,
          DECODE(CONVENIO.TP_CONVENIO, 
                  'A', EXA_RX.EXA_RX_CD_SSM, 
                  'C', EXA_RX.EXA_RX_CD_PRO_FAT, 
                  'P', EXA_RX.EXA_RX_CD_PRO_FAT, 
                  'H', EXA_RX.CD_PRO_FAT_SUS) CD_EXA_RX ,
          EXA_RX.DS_EXA_RX DS_EXA_RX ,
          COUNT(*) QT_EXAME ,
          AVG(DBAMV.CALC_VL_PROC_UNIT(CONVENIO.TP_CONVENIO, 
              DECODE(TP_CONVENIO, 
                     'A', EXA_RX.EXA_RX_CD_SSM, 
                     'C', EXA_RX.EXA_RX_CD_PRO_FAT, 
                     'P', EXA_RX.EXA_RX_CD_PRO_FAT, 
                     'H', EXA_RX.CD_PRO_FAT_SUS), 
                     PED_RX.DT_PEDIDO, 
                     PED_RX.HR_PEDIDO, 
                     ATENDIME.CD_CONVENIO, 
                     ATENDIME.CD_CON_PLA, 
                     ATENDIME.TP_ATENDIMENTO, 
                     ATENDIME.CD_TIP_ACOM, 
                     'U')) VL_UNITARIO ,
          AVG(DBAMV.CALC_VL_PROC_UNIT(CONVENIO.TP_CONVENIO, 
              DECODE(TP_CONVENIO, 
                     'A', EXA_RX.EXA_RX_CD_SSM, 
                     'C', EXA_RX.EXA_RX_CD_PRO_FAT, 
                     'P', EXA_RX.EXA_RX_CD_PRO_FAT, 
                     'H', EXA_RX.CD_PRO_FAT_SUS), 
                     PED_RX.DT_PEDIDO, 
                     PED_RX.HR_PEDIDO, 
                     ATENDIME.CD_CONVENIO,
                     ATENDIME.CD_CON_PLA, 
                     ATENDIME.TP_ATENDIMENTO, 
                     ATENDIME.CD_TIP_ACOM, 
                     'F')) VL_FILME ,
          AVG(DBAMV.CALC_VL_PROC_UNIT(CONVENIO.TP_CONVENIO, 
          DECODE(TP_CONVENIO, 'A', EXA_RX.EXA_RX_CD_SSM, 
                 'C', EXA_RX.EXA_RX_CD_PRO_FAT, 
                 'P', EXA_RX.EXA_RX_CD_PRO_FAT, 
                 'H', EXA_RX.CD_PRO_FAT_SUS), 
                 PED_RX.DT_PEDIDO, 
                 PED_RX.HR_PEDIDO,
                 ATENDIME.CD_CONVENIO, 
                 ATENDIME.CD_CON_PLA, 
                 ATENDIME.TP_ATENDIMENTO, 
                 ATENDIME.CD_TIP_ACOM, 'P')) VL_PORTE
   FROM DBAMV.ATENDIME ,
        DBAMV.PACIENTE ,
        DBAMV.PED_RX ,
        DBAMV.PRESTADOR ,
        DBAMV.CONVENIO ,
        DBAMV.ITPED_RX ,
        DBAMV.EXA_RX ,
        DBAMV.SET_EXA,
        Dbamv.Empresa_Convenio
   WHERE Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
     AND Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
     AND ATENDIME.CD_ATENDIMENTO = PED_RX.CD_ATENDIMENTO
     AND PED_RX.CD_CONVENIO = CONVENIO.CD_CONVENIO
     AND PED_RX.CD_PED_RX = ITPED_RX.CD_PED_RX
     AND PED_RX.CD_SET_EXA = SET_EXA.CD_SET_EXA
     AND ITPED_RX.CD_EXA_RX = EXA_RX.CD_EXA_RX
     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA 
   GROUP BY PRESTADOR.CD_PRESTADOR ,
            PRESTADOR.NM_PRESTADOR ,
            CONVENIO.CD_CONVENIO ,
            CONVENIO.NM_CONVENIO ,
            PACIENTE.CD_PACIENTE ,
            PACIENTE.NM_PACIENTE ,
            DECODE(CONVENIO.TP_CONVENIO, 
                    'A', EXA_RX.EXA_RX_CD_SSM, 
                    'C', EXA_RX.EXA_RX_CD_PRO_FAT, 
                    'P', EXA_RX.EXA_RX_CD_PRO_FAT, 
                    'H', EXA_RX.CD_PRO_FAT_SUS) ,
            EXA_RX.DS_EXA_RX)
ORDER BY CD_PRESTADOR,
         NM_PRESTADOR,
         CD_CONVENIO,
         NM_CONVENIO,
         CD_PACIENTE,
         NM_PACIENTE
