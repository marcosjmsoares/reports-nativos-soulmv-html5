--R_EXAME_SOLIC_CONV_XY
SELECT * FROM (
SELECT  CONVENIO.CD_CONVENIO,
                CONVENIO.NM_CONVENIO,
                GRU_PRO.CD_GRU_PRO,
                GRU_PRO.DS_GRU_PRO,
                COUNT( DECODE( ATENDIME.TP_ATENDIMENTO, 'I', 1, NULL ) )    QTD_I,
                COUNT( DECODE( ATENDIME.TP_ATENDIMENTO, 'A', 1, NULL ) )   QTD_A,
                COUNT( DECODE( ATENDIME.TP_ATENDIMENTO, 'U', 1, NULL ) )   QTD_U,
                COUNT( DECODE( ATENDIME.TP_ATENDIMENTO, 'E', 1, NULL ) )   QTD_E
FROM     DBAMV.CONVENIO,
               DBAMV.PED_RX,
               DBAMV.ITPED_RX,
               DBAMV.EXA_RX,
               DBAMV.PRO_FAT,
               DBAMV.GRU_PRO,
               DBAMV.ATENDIME
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And CONVENIO.CD_CONVENIO  =  PED_RX.CD_CONVENIO
AND       ITPED_RX.CD_PED_RX  =  PED_RX.CD_PED_RX
AND       TRUNC(PED_RX.DT_PEDIDO)  BETWEEN :CAMPO_DT_INICIAL  AND  
AND        PED_RX.CD_SET_EXA IN (SELECT PSDI_SET_EXA.CD_SET_EXA
                                                             FROM    DBAMV.PSDI_SET_EXA
                                                             WHERE NM_USUARIO = USER)
AND       ITPED_RX.CD_EXA_RX  =  EXA_RX.CD_EXA_RX
AND       NVL( PRO_FAT.SN_ATIVO, 'S' )  =  'S'
AND       EXA_RX.EXA_RX_CD_PRO_FAT  =  PRO_FAT.CD_PRO_FAT
AND       PRO_FAT.CD_GRU_PRO  =  GRU_PRO.CD_GRU_PRO
AND       PED_RX.CD_ATENDIMENTO  =  ATENDIME.CD_ATENDIMENTO
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY  CONVENIO.CD_CONVENIO,
                    CONVENIO.NM_CONVENIO,
                    GRU_PRO.CD_GRU_PRO,
                    GRU_PRO.DS_GRU_PRO
ORDER BY  CONVENIO.NM_CONVENIO,
                     GRU_PRO.DS_GRU_PRO
)
ORDER BY CD_CONVENIO,NM_CONVENIO
