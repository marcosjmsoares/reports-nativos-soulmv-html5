--R_ESTAT_SET_EXA
SELECT * FROM (
SELECT SET_EXA.NM_SET_EXA
               ,ITPED_RX.CD_EXA_RX
               ,EXA_RX.DS_EXA_RX
               ,COUNT(*) QTDE
FROM  DBAMV.SET_EXA
            ,DBAMV.PED_RX
            ,DBAMV.ITPED_RX
            ,DBAMV.EXA_RX
WHERE PED_RX.CD_SET_EXA  =  SET_EXA.CD_SET_EXA
AND       ITPED_RX.CD_PED_RX  =  PED_RX.CD_PED_RX
AND       EXA_RX.CD_EXA_RX  =  ITPED_RX.CD_EXA_RX
GROUP BY  SET_EXA.NM_SET_EXA
                    ,ITPED_RX.CD_EXA_RX
                    ,EXA_RX.DS_EXA_RX
)
ORDER BY NM_SET_EXA, CD_EXA_RX
