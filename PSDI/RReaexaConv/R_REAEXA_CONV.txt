--R_REAEXA_CONV
SELECT CONVENIO.CD_CONVENIO
              ,CONVENIO.NM_CONVENIO
              ,COUNT(*)   QTD_CONV
,(
SELECT COUNT(*)   QTD_CONV
FROM  DBAMV.ITPED_RX
           ,DBAMV.EXA_RX
           ,DBAMV.PED_RX
           ,DBAMV.CONVENIO
           ,DBAMV.PRO_FAT
           ,DBAMV.GRU_PRO
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And ITPED_RX.SN_REALIZADO  =  'S'
AND        EXA_RX.CD_EXA_RX  =  ITPED_RX.CD_EXA_RX
AND        ITPED_RX.CD_PED_RX  =  PED_RX.CD_PED_RX
AND         PED_RX.CD_SET_EXA IN ( SELECT PSDI_SET_EXA.CD_SET_EXA
                                                                FROM DBAMV.PSDI_SET_EXA
                                                                WHERE PSDI_SET_EXA.NM_USUARIO = USER )
AND        PED_RX.CD_CONVENIO  =  CONVENIO.CD_CONVENIO
AND        DECODE(CONVENIO.TP_CONVENIO, 'A', EXA_RX.CD_PROCEDIMENTO_SIA
                                                                            , 'H', EXA_RX.CD_PROCEDIMENTO_SIH
                                                                            ,       EXA_RX.EXA_RX_CD_PRO_FAT)  =  PRO_FAT.CD_PRO_FAT
AND   PRO_FAT.CD_GRU_PRO  =  GRU_PRO.CD_GRU_PRO
AND        ITPED_RX.DT_REALIZADO  BETWEEN :CAMPO_DT_INICIAL AND :CAMPO_DT_FINAL
)QTD_TOTAL
FROM  DBAMV.ITPED_RX
           ,DBAMV.EXA_RX
           ,DBAMV.PED_RX
           ,DBAMV.CONVENIO
           ,DBAMV.PRO_FAT
           ,DBAMV.GRU_PRO
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And ITPED_RX.SN_REALIZADO  =  'S'
AND        EXA_RX.CD_EXA_RX  =  ITPED_RX.CD_EXA_RX
AND        ITPED_RX.CD_PED_RX  =  PED_RX.CD_PED_RX
AND         PED_RX.CD_SET_EXA IN ( SELECT PSDI_SET_EXA.CD_SET_EXA
                                                                FROM DBAMV.PSDI_SET_EXA
                                                                WHERE PSDI_SET_EXA.NM_USUARIO = USER )
AND        PED_RX.CD_CONVENIO  =  CONVENIO.CD_CONVENIO
AND        DECODE(CONVENIO.TP_CONVENIO, 'A', EXA_RX.CD_PROCEDIMENTO_SIA
                                                                            , 'H', EXA_RX.CD_PROCEDIMENTO_SIH
                                                                            ,       EXA_RX.EXA_RX_CD_PRO_FAT)  =  PRO_FAT.CD_PRO_FAT
AND   PRO_FAT.CD_GRU_PRO  =  GRU_PRO.CD_GRU_PRO
AND        ITPED_RX.DT_REALIZADO  BETWEEN :CAMPO_DT_INICIAL GROUP BY CONVENIO.CD_CONVENIO
                   ,CONVENIO.NM_CONVENIO
ORDER BY QTD_CONV DESC

--R_REAEXA_CONV_R_EXAME_Q_EXAME
SELECT * FROM (
SELECT CONVENIO.CD_CONVENIO   CD_CONV_EXA
              ,GRU_PRO.CD_GRU_PRO     CD_GRU_EXA
              ,ITPED_RX.CD_EXA_RX
              ,EXA_RX.DS_EXA_RX
              ,COUNT(*)   QTD_EXA
FROM  DBAMV.ITPED_RX
            ,DBAMV.EXA_RX
            ,DBAMV.PED_RX
            ,DBAMV.CONVENIO
            ,DBAMV.PRO_FAT
            ,DBAMV.GRU_PRO
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And ITPED_RX.SN_REALIZADO = 'S'
AND   EXA_RX.CD_EXA_RX = ITPED_RX.CD_EXA_RX
AND   ITPED_RX.CD_PED_RX = PED_RX.CD_PED_RX
AND         PED_RX.CD_SET_EXA IN ( SELECT PSDI_SET_EXA.CD_SET_EXA
                                                                FROM DBAMV.PSDI_SET_EXA
                                                                WHERE PSDI_SET_EXA.NM_USUARIO = USER )
AND   PED_RX.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND   DECODE(CONVENIO.TP_CONVENIO, 'A', EXA_RX.CD_PROCEDIMENTO_SIA
                                                                        , 'H', EXA_RX.CD_PROCEDIMENTO_SIH
                                                                        ,       EXA_RX.EXA_RX_CD_PRO_FAT) = PRO_FAT.CD_PRO_FAT
AND   PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
AND   ITPED_RX.DT_REALIZADO  BETWEEN :CAMPO_DT_INICIAL GROUP BY CONVENIO.CD_CONVENIO
                   ,GRU_PRO.CD_GRU_PRO
                   ,ITPED_RX.CD_EXA_RX
                   ,EXA_RX.DS_EXA_RX
ORDER BY QTD_EXA DESC
)
WHERE CD_CONV_EXA = :CAMPO_CD_CONV_GRU
AND CD_GRU_EXA = :CAMPO_CD_GRU_PRO
