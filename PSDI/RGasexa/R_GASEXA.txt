--R_GASEXA
SELECT  EXA_RX.CD_EXA_RX CODIGO
       ,EXA_RX.DS_EXA_RX DESCRICAO
       ,GAS_EXA.CD_PRODUTO CD_PRODUTO
       ,PRODUTO.DS_PRODUTO DS_PRODUTO
       ,UNI_PRO.CD_UNIDADE CD_UNIDADE
       ,UNI_PRO.DS_UNIDADE DS_UNIDADE
       ,GAS_EXA.QT_USADA QT_USADA
FROM    dbamv.EXA_RX EXA_RX
       ,dbamv.GAS_EXA GAS_EXA
       ,dbamv.PRODUTO PRODUTO
       ,dbamv.UNI_PRO UNI_PRO
WHERE GAS_EXA.CD_EXA_RX = EXA_RX.CD_EXA_RX
AND   PRODUTO.CD_PRODUTO = GAS_EXA.CD_PRODUTO
AND   UNI_PRO.CD_UNI_PRO = GAS_EXA.CD_UNI_PRO
AND   UNI_PRO.CD_PRODUTO = gas_exa.cd_produto
And Exists ( Select Empresa_Produto.Cd_Produto
                     From Dbamv.Empresa_Produto
                   Where Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
                        and Empresa_Produto.Cd_Produto = Produto.Cd_Produto )
ORDER BY
