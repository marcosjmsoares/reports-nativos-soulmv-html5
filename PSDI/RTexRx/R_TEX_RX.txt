SELECT PRESTADOR.CD_PRESTADOR
      ,PRESTADOR.NM_PRESTADOR
      ,TEX_RX.CD_TEX_RX
      ,TEX_RX.DS_TEX_RX
      ,TEX_RX.DS_TEXTO
FROM DBAMV.TEX_RX
   , DBAMV.PRESTADOR
WHERE PRESTADOR.CD_PRESTADOR(+) = TEX_RX.CD_PRESTADOR
