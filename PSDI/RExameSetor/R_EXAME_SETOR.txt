--R_EXAME_SETOR
SELECT * FROM (
SELECT SET_EXA.CD_SET_EXA      CD_SET_EXA
              ,SET_EXA.NM_SET_EXA     NM_SET_EXA
              ,ORI_ATE.DS_ORI_ATE       DS_ORI_ATE
              ,EXA_RX.CD_EXA_RX         CD_EXA_RX
              ,EXA_RX.DS_EXA_RX         DS_EXA_RX
              ,COUNT(*)                             QTDE
FROM    DBAMV.ATENDIME
             ,DBAMV.ORI_ATE
             ,DBAMV.PED_RX
             ,DBAMV.SET_EXA
             ,DBAMV.ITPED_RX
             ,DBAMV.EXA_RX
WHERE  ATENDIME.CD_ORI_ATE  =  ORI_ATE.CD_ORI_ATE
AND        TRUNC(PED_RX.DT_PEDIDO)  BETWEEN  TRUNC(:CAMPO_DT_INICIAL)  AND  TRUNC(:CAMPO_DT_FINAL)
AND        ATENDIME.CD_ATENDIMENTO = PED_RX.CD_ATENDIMENTO
AND        PED_RX.CD_SET_EXA  =  SET_EXA.CD_SET_EXA
AND        PED_RX.CD_PED_RX  =  ITPED_RX.CD_PED_RX
AND        ITPED_RX.CD_EXA_RX  =  EXA_RX.CD_EXA_RX
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY SET_EXA.CD_SET_EXA
                   ,SET_EXA.NM_SET_EXA
                   ,ORI_ATE.DS_ORI_ATE
                   ,EXA_RX.CD_EXA_RX
                   ,EXA_RX.DS_EXA_RX
)
ORDER BY CD_SET_EXA,NM_SET_EXA,
DS_ORI_ATE, CD_EXA_RX
