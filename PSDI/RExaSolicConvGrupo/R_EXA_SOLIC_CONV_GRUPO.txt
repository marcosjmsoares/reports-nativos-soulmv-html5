--R_EXA_SOLIC_CONV_GRUPO
SELECT  CONVENIO.CD_CONVENIO
               ,CONVENIO.NM_CONVENIO
               ,COUNT(*)    QT_CONVENIO
               , (SELECT COUNT(*)
                  FROM      DBAMV.PED_RX
                                ,DBAMV.CONVENIO
                                ,DBAMV.ITPED_RX
                                ,DBAMV.EXA_RX
                                ,DBAMV.PRO_FAT
                  ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And TRUNC(PED_RX.DT_PEDIDO)  BETWEEN  :CAMPO_DT_INICIAL  AND  
AND        PED_RX.CD_SET_EXA IN (SELECT PSDI_SET_EXA.CD_SET_EXA
                                                             FROM    DBAMV.PSDI_SET_EXA
                                                             WHERE NM_USUARIO = USER)
AND       PED_RX.CD_CONVENIO  =  CONVENIO.CD_CONVENIO
AND       PED_RX.CD_PED_RX  =  ITPED_RX.CD_PED_RX
AND       ITPED_RX.CD_EXA_RX  =  EXA_RX.CD_EXA_RX
AND       NVL( PRO_FAT.SN_ATIVO, 'S' )  =  'S'
AND       EXA_RX.EXA_RX_CD_PRO_FAT  =  PRO_FAT.CD_PRO_FAT) QT_TOTAL
FROM      DBAMV.PED_RX
               ,DBAMV.CONVENIO
               ,DBAMV.ITPED_RX
               ,DBAMV.EXA_RX
               ,DBAMV.PRO_FAT
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And TRUNC(PED_RX.DT_PEDIDO)  BETWEEN  :CAMPO_DT_INICIAL  AND  
AND        PED_RX.CD_SET_EXA IN (SELECT PSDI_SET_EXA.CD_SET_EXA
                                                             FROM    DBAMV.PSDI_SET_EXA
                                                             WHERE NM_USUARIO = USER)
AND       PED_RX.CD_CONVENIO  =  CONVENIO.CD_CONVENIO
AND       PED_RX.CD_PED_RX  =  ITPED_RX.CD_PED_RX
AND       ITPED_RX.CD_EXA_RX  =  EXA_RX.CD_EXA_RX
AND       NVL( PRO_FAT.SN_ATIVO, 'S' )  =  'S'
AND       EXA_RX.EXA_RX_CD_PRO_FAT  =  PRO_FAT.CD_PRO_FAT
GROUP BY  CONVENIO.CD_CONVENIO
                    ,CONVENIO.NM_CONVENIO
ORDER BY  QT_CONVENIO  DESC, NM_CONVENIO
