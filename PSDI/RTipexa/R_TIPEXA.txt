--R_TIPEXA
SELECT  EXA_RX.CD_EXA_RX CODIGO
       ,EXA_RX.DS_EXA_RX DESCRICAO
       ,EXA_RX.TP_SEXO TP_SEXO2
       ,EXA_RX.VL_TEMPO_MEDIO VL_TEMPO_MEDIO3
       ,EXA_RX.EXA_RX_CD_PRO_FAT
       ,EXA_RX.CD_PROCEDIMENTO_SIH EXA_RX_CD_SIH_SUS
       ,EXA_RX.CD_PROCEDIMENTO_SIA EXA_RX_CD_SSM
       ,empresa_orientacoes_exa_rx.DS_ORIENTACAO DS_ORIENTACAO4
FROM    dbamv.EXA_RX EXA_RX
             ,dbamv.empresa_orientacoes_exa_rx
WHERE empresa_orientacoes_exa_rx.cd_exa_rx(+) = exa_rx.cd_exa_rx
    and Nvl(empresa_orientacoes_exa_rx.cd_multi_empresa,dbamv.pkg_mv2000.le_empresa) = dbamv.pkg_mv2000.le_empresa
ORDER BY
