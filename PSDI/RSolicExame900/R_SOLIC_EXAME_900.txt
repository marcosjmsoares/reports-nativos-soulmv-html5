--R_SOLIC_EXAME_900
SELECT * FROM
  (SELECT ATENDIME.CD_ATENDIMENTO CD_ATENDIMENTO ,
          LEITO.DS_RESUMO DS_LEITO ,
          UNID_INT.CD_UNID_INT CD_UNID_INT ,
          UNID_INT.DS_UNID_INT DS_UNID_INT ,
          PACIENTE.CD_PACIENTE CD_PACIENTE ,
          PACIENTE.NM_PACIENTE NM_PACIENTE ,
          (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE,
          PACIENTE.DT_NASCIMENTO DT_NASCIMENTO ,
          DBAMV.PACK_INTERNAMENTO.RETORNA_IDADE(PACIENTE.DT_NASCIMENTO, TRUNC(SYSDATE)) NR_IDADE ,
          DECODE(PACIENTE.TP_SEXO, 
                  'M', 'MASCULINO', 
                  'F', 'FEMININO', 
                  'A', 'AMBOS') TP_SEXO ,
          CONVENIO.CD_CONVENIO CD_CONVENIO ,
          CONVENIO.NM_CONVENIO NM_CONVENIO ,
          PRESTADOR.CD_PRESTADOR CD_PRESTADOR ,
          PRESTADOR.NM_PRESTADOR NM_PRESTADOR ,
          SETOR.CD_SETOR CD_SETOR ,
          SETOR.NM_SETOR NM_SETOR ,
          SET_EXA.CD_SET_EXA CD_SET_EXA ,
          SET_EXA.NM_SET_EXA NM_SET_EXA ,
          PED_RX.CD_PED_RX CD_PED_RX ,
          PED_RX.DT_PEDIDO DT_PEDIDO ,
          PED_RX.HR_PEDIDO HR_PEDIDO ,
          PED_RX.NM_USUARIO NM_USUARIO ,
          PED_RX.DT_ENTREGA DT_ENTREGA ,
          PED_RX.HR_ENTREGA HR_ENTREGA ,
          EXA_RX.CD_EXA_RX CD_EXA_RX ,
          EXA_RX.DS_EXA_RX DS_EXA_RX ,
          ITPED_RX.DS_OBSERVACAO DS_OBSERVACAO
   FROM DBAMV.ATENDIME ,
        DBAMV.LEITO ,
        DBAMV.UNID_INT ,
        DBAMV.PACIENTE ,
        DBAMV.CID ,
        DBAMV.PED_RX ,
        DBAMV.CONVENIO ,
        DBAMV.PRESTADOR ,
        DBAMV.SETOR ,
        DBAMV.SET_EXA ,
        DBAMV.ITPED_RX ,
        DBAMV.EXA_RX,
        Dbamv.Empresa_Convenio
   WHERE Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
     AND Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
     AND ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
     AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)
     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
     AND ATENDIME.CD_CID = CID.CD_CID(+)
     AND ATENDIME.CD_ATENDIMENTO = PED_RX.CD_ATENDIMENTO
     AND PED_RX.CD_PED_RX = :CAMPO_NR_PEDIDO
     AND PED_RX.CD_CONVENIO = CONVENIO.CD_CONVENIO
     AND PED_RX.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
     AND PED_RX.CD_SETOR = SETOR.CD_SETOR
     AND PED_RX.CD_SET_EXA = SET_EXA.CD_SET_EXA
     AND PED_RX.CD_PED_RX = ITPED_RX.CD_PED_RX
     AND ITPED_RX.CD_EXA_RX = EXA_RX.CD_EXA_RX
     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  )
ORDER BY CD_SETOR,
         NM_SETOR,
         CD_SET_EXA,
         NM_SET_EXA,
         CD_PED_RX,
         DT_PEDIDO,
         HR_PEDIDO,
         DT_NASCIMENTO,
         NR_IDADE,
         TP_SEXO,
         DT_ENTREGA,
         HR_ENTREGA,
         CD_ATENDIMENTO,
         CD_UNID_INT,
         DS_UNID_INT,
         CD_PACIENTE,
         NM_PACIENTE,
         CD_CONVENIO,
         NM_CONVENIO,
         CD_PRESTADOR,
         NM_PRESTADOR,
         NM_USUARIO,
         DS_LEITO
