--R_ESCALAS
SELECT * FROM (
SELECT  ESCALA.CD_ESCALA
               ,ESCALA.CD_RECURSO  CODIGO
               ,RECURSO.NM_RECURSO DESCRICAO
               ,ESCALA.TP_DIA_SEMANA
               ,ESCALA.HR_INIC HR_INIC
               ,ESCALA.HR_FIM HR_FIM
               ,ESCALA.QT_ATENDIMENTO
               ,PRESTADOR.CD_PRESTADOR
               ,PRESTADOR.NM_PRESTADOR
FROM    dbamv.ESCALA
             ,dbamv.RECURSO
             ,dbamv.PRESTADOR
WHERE ESCALA.CD_RECURSO LIKE :CAMPO_RECURSO
AND       PRESTADOR.CD_PRESTADOR(+)  =  ESCALA.CD_PRESTADOR
AND       RECURSO.CD_RECURSO  =  ESCALA.CD_RECURSO
AND        RECURSO.CD_SET_EXA IN (SELECT PSDI_SET_EXA.CD_SET_EXA
                                                               FROM DBAMV.PSDI_SET_EXA
                                                               WHERE NM_USUARIO = USER)
ORDER BY :CAMPO_ORDEM, CD_ESCALA
)
ORDER BY CODIGO,DESCRICAO
