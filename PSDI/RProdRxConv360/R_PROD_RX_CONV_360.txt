--R_PROD_RX_CONV_360
SELECT * FROM (
SELECT SET_EXA.CD_SET_EXA
              ,SET_EXA.NM_SET_EXA
              ,CONVENIO.CD_CONVENIO
              ,CONVENIO.NM_CONVENIO
              ,EXA_RX.CD_EXA_RX
              ,EXA_RX.DS_EXA_RX
              ,COUNT(1)      CONTADOR
              ,SUM(  DBAMV.CALC_VL_PROC_UNIT( CONVENIO.TP_CONVENIO
                                             ,EXA_RX.EXA_RX_CD_PRO_FAT
                                             ,PED_RX.DT_PEDIDO
                                             ,PED_RX.HR_PEDIDO
                                             ,CONVENIO.CD_CONVENIO
                                             ,NVL(PED_RX.CD_CON_PLA,1)
                                             ,ATENDIME.TP_ATENDIMENTO
                                             ,ATENDIME.CD_TIP_ACOM
                                             ,:CAMPO_TP_VALOR
                                             ,null
                                             )     )       VALOR_UNIT
              ,SUM(  DBAMV.CALC_VL_PROC_UNIT( CONVENIO.TP_CONVENIO
                                             ,EXA_RX.EXA_RX_CD_PRO_FAT
                                             ,PED_RX.DT_PEDIDO
                                             ,PED_RX.HR_PEDIDO
                                             ,CONVENIO.CD_CONVENIO
                                             ,NVL(PED_RX.CD_CON_PLA,1)
                                             ,ATENDIME.TP_ATENDIMENTO
                                             ,ATENDIME.CD_TIP_ACOM
                                             ,'F'
                                             ,null
                                             )     )       VALOR_UNIT_FILME
FROM   DBAMV.ATENDIME
            ,DBAMV.PED_RX
            ,DBAMV.CONVENIO
            ,DBAMV.SET_EXA
            ,DBAMV.ITPED_RX
            ,DBAMV.EXA_RX
            ,DBAMV.LAUDO_RX
            ,DBAMV.PRESTADOR
Where
                ATENDIME.CD_ATENDIMENTO            =  PED_RX.CD_ATENDIMENTO
AND        ATENDIME.CD_MULTI_EMPRESA  =  dbamv.pkg_mv2000.le_empresa
AND        ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND        PED_RX.CD_CONVENIO                =  CONVENIO.CD_CONVENIO
AND        PED_RX.CD_SET_EXA                  =  SET_EXA.CD_SET_EXA
AND        PED_RX.CD_PED_RX                   =  ITPED_RX.CD_PED_RX
AND        NVL(ITPED_RX.SN_REALIZADO,'N')  =  'S'
AND        TRUNC(ITPED_RX.DT_REALIZADO)  BETWEEN  :CAMPO_DT_INICIAL  AND  
AND        ITPED_RX.CD_EXA_RX                =  EXA_RX.CD_EXA_RX
AND        ITPED_RX.CD_PED_RX                =  LAUDO_RX.CD_PED_RX(+)
AND        ITPED_RX.CD_LAUDO                  =  LAUDO_RX.CD_LAUDO(+)
AND        LAUDO_RX.CD_PRESTADOR       = PRESTADOR.CD_PRESTADOR(+)
AND  PED_RX.CD_SET_EXA IN  (SELECT  PSDI_SET_EXA.CD_SET_EXA
                                  FROM DBAMV.PSDI_SET_EXA
                               WHERE PSDI_SET_EXA.CD_SET_EXA  = PED_RX.CD_SET_EXA
                                     AND PSDI_SET_EXA.NM_USUARIO = USER)
GROUP BY  SET_EXA.CD_SET_EXA
                    ,SET_EXA.NM_SET_EXA
                   ,CONVENIO.CD_CONVENIO
                   ,CONVENIO.NM_CONVENIO
                   ,EXA_RX.CD_EXA_RX
                   ,EXA_RX.DS_EXA_RX
)
ORDER BY CD_SET_EXA,NM_SET_EXA,
CD_CONVENIO,NM_CONVENIO,CD_EXA_RX,DS_EXA_RX
