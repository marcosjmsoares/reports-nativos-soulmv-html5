--R_ESTATISTICA_DIA
SELECT  EXA_RX.CD_EXA_RX
              , EXA_RX.DS_EXA_RX
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '01',1))  DIA1
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '02',1))  DIA2
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '03',1))  DIA3
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '04',1))  DIA4
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '05',1))  DIA5
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '06',1))  DIA6
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '07',1))  DIA7
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '08',1))  DIA8
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '09',1))  DIA9
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '10',1))  DIA10
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '11',1))  DIA11
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '12',1))  DIA12
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '13',1))  DIA13
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '14',1))  DIA14
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '15',1))  DIA15
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '16',1))  DIA16
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '17',1))  DIA17
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '18',1))  DIA18
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '19',1))  DIA19
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '20',1))  DIA20
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '21',1))  DIA21
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '22',1))  DIA22
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '23',1))  DIA23
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '24',1))  DIA24
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '25',1))  DIA25
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '26',1))  DIA26
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '27',1))  DIA27
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '28',1))  DIA28
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '29',1))  DIA29
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '30',1))  DIA30
              ,SUM(DECODE(TO_CHAR(ITPED_RX.DT_REALIZADO,'DD'), '31',1))  DIA31
              ,SUM(1)  TOT
FROM     DBAMV. PED_RX
              ,DBAMV.ITPED_RX
              ,DBAMV.EXA_RX
WHERE  TO_CHAR(ITPED_RX.DT_REALIZADO,'MM/YYYY')  =  :CAMPO_MES
AND        ITPED_RX.CD_PED_RX  =  PED_RX.CD_PED_RX
AND        ITPED_RX.CD_EXA_RX  =  EXA_RX.CD_EXA_RX
AND        NVL(ITPED_RX.SN_REALIZADO,'N')  =  'S'
GROUP BY  EXA_RX.CD_EXA_RX
                    ,EXA_RX.DS_EXA_RX
