--R_EXA_SET
SELECT * FROM (
SELECT  SETOR.CD_SETOR
       ,SETOR.NM_SETOR
       ,0 CD_PED_RX
       ,'' CD_EXA_RX
       ,'' DS_EXA_RX
FROM    DBAMV.SETOR
       , DBAMV.PED_RX
       , DBAMV.ITPED_RX
       , DBAMV.EXA_RX
WHERE  SETOR.CD_SETOR LIKE '%'
    AND   PED_RX.CD_SETOR    =  SETOR.CD_SETOR
    AND   PED_RX.CD_SET_EXA IN (SELECT   PSDI_SET_EXA.CD_SET_EXA
                                                                 FROM    DBAMV.PSDI_SET_EXA
                                                                 WHERE NM_USUARIO = USER)
   AND   ITPED_RX.CD_PED_RX = PED_RX.CD_PED_RX
   AND   EXA_RX.CD_EXA_RX   = ITPED_RX.CD_EXA_RX
group by SETOR.CD_SETOR
       ,SETOR.NM_SETOR
 ORDER BY SETOR.CD_SETOR,
          CD_EXA_RX
)
ORDER BY CD_SETOR,NM_SETOR
