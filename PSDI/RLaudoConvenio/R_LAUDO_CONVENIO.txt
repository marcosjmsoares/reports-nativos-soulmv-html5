--R_LAUDO_CONVENIO
SELECT * FROM (
SELECT NVL(PRESTADOR.CD_PRESTADOR,NVL(PREST.CD_PRESTADOR,0))                  CD_PRESTADOR
       , NVL(PRESTADOR.NM_PRESTADOR,NVL(PREST.NM_PRESTADOR, 'NÃO INFORMADO')) NM_PRESTADOR
       ,CONVENIO.CD_CONVENIO                                                  CD_CONVENIO
       ,CONVENIO.NM_CONVENIO                                                  NM_CONVENIO
       ,PACIENTE.CD_PACIENTE                                                  CD_PACIENTE
       ,PACIENTE.NM_PACIENTE                                                  NM_PACIENTE
       ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE
       ,DECODE(CONVENIO.TP_CONVENIO,'A', EXA_RX.CD_PROCEDIMENTO_SIA
                                   , 'H', EXA_RX.CD_PROCEDIMENTO_SIH
                                   , EXA_RX.EXA_RX_CD_PRO_FAT)            CD_EXA_RX
               ,EXA_RX.DS_EXA_RX                                              DS_EXA_RX
               ,COUNT(*)                                                      QT_EXAME
               ,DBAMV.CALC_VL_PROC_UNIT( CONVENIO.TP_CONVENIO,
                                          DECODE(TP_CONVENIO, 'A', EXA_RX.CD_PROCEDIMENTO_SIA
                                                             , 'H', EXA_RX.CD_PROCEDIMENTO_SIH
                                                             , EXA_RX.EXA_RX_CD_PRO_FAT),
                                         PED_RX.DT_PEDIDO,
                                         PED_RX.HR_PEDIDO,
                                         ATENDIME.CD_CONVENIO,
                                         ATENDIME.CD_CON_PLA,
                                         ATENDIME.TP_ATENDIMENTO,
                                         ATENDIME.CD_TIP_ACOM,
                                         'U')             VL_UNITARIO
               ,DBAMV.CALC_VL_PROC_UNIT( CONVENIO.TP_CONVENIO,
                                          DECODE(TP_CONVENIO, 'A', EXA_RX.CD_PROCEDIMENTO_SIA
                                                             , 'H', EXA_RX.CD_PROCEDIMENTO_SIH
                                                             , EXA_RX.EXA_RX_CD_PRO_FAT),
                                           PED_RX.DT_PEDIDO,
                                           PED_RX.HR_PEDIDO,
                                           ATENDIME.CD_CONVENIO,
                                           ATENDIME.CD_CON_PLA,
                                           ATENDIME.TP_ATENDIMENTO,
                                           ATENDIME.CD_TIP_ACOM,
                                           'F')             VL_FILME
               ,DBAMV.CALC_VL_PROC_UNIT( CONVENIO.TP_CONVENIO,
                                          DECODE(TP_CONVENIO, 'A', EXA_RX.CD_PROCEDIMENTO_SIA
                                                             , 'H', EXA_RX.CD_PROCEDIMENTO_SIH
                                                             , EXA_RX.EXA_RX_CD_PRO_FAT),
                                           PED_RX.DT_PEDIDO,
                                           PED_RX.HR_PEDIDO,
                                           ATENDIME.CD_CONVENIO,
                                           ATENDIME.CD_CON_PLA,
                                           ATENDIME.TP_ATENDIMENTO,
                                           ATENDIME.CD_TIP_ACOM,
                                           'P')             VL_PORTE
  FROM DBAMV.ATENDIME
       , DBAMV.PACIENTE
       , DBAMV.PED_RX
       , DBAMV.PRESTADOR
       , DBAMV.PRESTADOR PREST
       , DBAMV.LAUDO_RX
       , DBAMV.CONVENIO
       , DBAMV.ITPED_RX
       , DBAMV.EXA_RX
       , DBAMV.SET_EXA
       ,Dbamv.Empresa_Convenio
 Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
   And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   And ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
   AND ATENDIME.CD_ATENDIMENTO = PED_RX.CD_ATENDIMENTO
   AND PED_RX.CD_CONVENIO = CONVENIO.CD_CONVENIO
   AND PED_RX.CD_PED_RX = ITPED_RX.CD_PED_RX
   AND PED_RX.CD_SET_EXA = SET_EXA.CD_SET_EXA
   AND ITPED_RX.CD_EXA_RX = EXA_RX.CD_EXA_RX
   AND ITPED_RX.CD_LAUDO = LAUDO_RX.CD_LAUDO
   AND CONVENIO.TP_CONVENIO  <> 'H'
 GROUP BY NVL(PRESTADOR.CD_PRESTADOR,NVL(PREST.CD_PRESTADOR,0))
       , NVL(PRESTADOR.NM_PRESTADOR,NVL(PREST.NM_PRESTADOR, 'NÃO INFORMADO'))
       ,CONVENIO.CD_CONVENIO
       ,CONVENIO.NM_CONVENIO
       ,PACIENTE.CD_PACIENTE
       ,PACIENTE.NM_PACIENTE
       ,DECODE(TP_CONVENIO, 'A', EXA_RX.CD_PROCEDIMENTO_SIA
                                                             , 'H', EXA_RX.CD_PROCEDIMENTO_SIH
                                                             , EXA_RX.EXA_RX_CD_PRO_FAT)
      ,EXA_RX.DS_EXA_RX
      ,DBAMV.CALC_VL_PROC_UNIT( CONVENIO.TP_CONVENIO,
                               DECODE(TP_CONVENIO, 'A', EXA_RX.CD_PROCEDIMENTO_SIA
                                                  , 'H', EXA_RX.CD_PROCEDIMENTO_SIH
                                                  , EXA_RX.EXA_RX_CD_PRO_FAT),
                               PED_RX.DT_PEDIDO,
                               PED_RX.HR_PEDIDO,
                               ATENDIME.CD_CONVENIO,
                               ATENDIME.CD_CON_PLA,
                               ATENDIME.TP_ATENDIMENTO,
                               ATENDIME.CD_TIP_ACOM,
                               'U')
      ,DBAMV.CALC_VL_PROC_UNIT( CONVENIO.TP_CONVENIO,
                               DECODE(TP_CONVENIO, 'A', EXA_RX.CD_PROCEDIMENTO_SIA
                                                 , 'H', EXA_RX.CD_PROCEDIMENTO_SIH
                                                 , EXA_RX.EXA_RX_CD_PRO_FAT),
                               PED_RX.DT_PEDIDO,
                               PED_RX.HR_PEDIDO,
                               ATENDIME.CD_CONVENIO,
                               ATENDIME.CD_CON_PLA,
                               ATENDIME.TP_ATENDIMENTO,
                               ATENDIME.CD_TIP_ACOM,
                               'F')
      ,DBAMV.CALC_VL_PROC_UNIT( CONVENIO.TP_CONVENIO,
                               DECODE(TP_CONVENIO, 'A', EXA_RX.CD_PROCEDIMENTO_SIA
                                                 , 'H', EXA_RX.CD_PROCEDIMENTO_SIH
                                                 , EXA_RX.EXA_RX_CD_PRO_FAT),
                               PED_RX.DT_PEDIDO,
                               PED_RX.HR_PEDIDO,
                               ATENDIME.CD_CONVENIO,
                               ATENDIME.CD_CON_PLA,
                               ATENDIME.TP_ATENDIMENTO,
                               ATENDIME.CD_TIP_ACOM,
                               'P')
UNION ALL
SELECT NVL(PRESTADOR.CD_PRESTADOR,NVL(PREST.CD_PRESTADOR,0))                  CD_PRESTADOR
       , NVL(PRESTADOR.NM_PRESTADOR,NVL(PREST.NM_PRESTADOR, 'NÃO INFORMADO')) NM_PRESTADOR
       , CONVENIO.CD_CONVENIO                             CD_CONVENIO
       , CONVENIO.NM_CONVENIO                             NM_CONVENIO
       , PACIENTE.CD_PACIENTE                             CD_PACIENTE
       , PACIENTE.NM_PACIENTE                             NM_PACIENTE
       ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE
       , NVL(ITREG_FAT.CD_SUS_AUXILIAR, ITREG_FAT.CD_SUS) CD_EXA_RX
       , EXA_RX.DS_EXA_RX                                 DS_EXA_RX
       , COUNT(*)                                         QT_EXAME
       , Nvl(ITENS_SD_FFIS.VL_SD, ITREG_FAT.VL_SH)        VL_UNITARIO
       , 0                                                VL_FILME
       , 0                                                VL_PORTE
  FROM DBAMV.ITREG_FAT
       , DBAMV.ITPED_RX
       , DBAMV.ITENS_SD_FFIS
       , DBAMV.REG_FAT
       , DBAMV.PED_RX
       , DBAMV.ATENDIME
       , DBAMV.PACIENTE
       , DBAMV.PRESTADOR
       , DBAMV.PRESTADOR PREST
       , DBAMV.LAUDO_RX
       , DBAMV.CONVENIO
       , DBAMV.EMPRESA_CONVENIO
       , DBAMV.EXA_RX
 WHERE EMPRESA_CONVENIO.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
   AND EMPRESA_CONVENIO.CD_CONVENIO      = CONVENIO.CD_CONVENIO
   AND ATENDIME.CD_PACIENTE       = PACIENTE.CD_PACIENTE
   AND ATENDIME.CD_CONVENIO       = CONVENIO.CD_CONVENIO
   AND REG_FAT.CD_REG_FAT         = ITREG_FAT.CD_REG_FAT
   AND REG_FAT.CD_ATENDIMENTO     = PED_RX.CD_ATENDIMENTO
   AND REG_FAT.CD_MULTI_EMPRESA   =  :CAMPO_CD_MULTI_EMPRESA
   AND REG_FAT.CD_CONVENIO        = convenio.CD_CONVENIO
   AND ITREG_FAT.CD_MVTO          = ITPED_RX.CD_ITPED_RX
   AND ITREG_FAT.CD_LANCAMENTO    = ITENS_SD_FFIS.CD_LANCAMENTO (+)
   AND ITREG_FAT.CD_REG_FAT       = ITENS_SD_FFIS.CD_REG_FAT(+)
   AND UPPER( ITREG_FAT.TP_MVTO ) = 'IMAGEM'
   AND PED_RX.CD_CONVENIO         = CONVENIO.CD_CONVENIO
   AND PED_RX.CD_ATENDIMENTO      = ATENDIME.CD_ATENDIMENTO
   AND PED_RX.CD_PED_RX           = ITPED_RX.CD_PED_RX
   AND ITPED_RX.CD_EXA_RX         = EXA_RX.CD_EXA_RX
   AND ITPED_RX.CD_LAUDO          = LAUDO_RX.CD_LAUDO (+)
   AND CONVENIO.TP_CONVENIO      = 'H'
   AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
 GROUP BY NVL(PRESTADOR.CD_PRESTADOR,NVL(PREST.CD_PRESTADOR,0))
       , NVL(PRESTADOR.NM_PRESTADOR,NVL(PREST.NM_PRESTADOR, 'NÃO INFORMADO'))
       , CONVENIO.CD_CONVENIO
       , CONVENIO.NM_CONVENIO
       , PACIENTE.CD_PACIENTE
       , PACIENTE.NM_PACIENTE
       , NVL(ITREG_FAT.CD_SUS_AUXILIAR, ITREG_FAT.CD_SUS)
       , EXA_RX.DS_EXA_RX
       , Nvl(ITENS_SD_FFIS.VL_SD, ITREG_FAT.VL_SH)
)
ORDER BY CD_PRESTADOR,NM_PRESTADOR,
CD_CONVENIO,NM_CONVENIO,
CD_PACIENTE,NM_PACIENTE
