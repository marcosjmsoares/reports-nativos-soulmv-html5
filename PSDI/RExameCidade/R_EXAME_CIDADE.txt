--R_EXAME_CIDADE_R_TIPO_ATENDIMENTO2_Q_EXAMES_X_CIDADES_A
SELECT * FROM (
SELECT   CIDADE.CD_UF                                                          CD_UF_A
               ,NVL(CIDADE.NM_CIDADE,'NÃO ESPECIFICADA')    NM_CIDADE_A
               ,ATENDIME.CD_ORI_ATE                                          CD_ORI_ATE_A
               ,ORI_ATE.DS_ORI_ATE                                              DS_ORI_ATE_A
               ,DECODE(TP_CONVENIO, 'A', EXA_RX.CD_PROCEDIMENTO_SIA,
                                                           'C', EXA_RX.EXA_RX_CD_PRO_FAT,
                                                           'P', EXA_RX.EXA_RX_CD_PRO_FAT,
                                                           'H', EXA_RX.CD_PROCEDIMENTO_SIH)        CD_EXA_RX_A
               ,EXA_RX.DS_EXA_RX        DS_EXA_RX_A
               ,COUNT(*)                            QTD_A
               ,DBAMV.CALC_VL_PROC_UNIT( CONVENIO.TP_CONVENIO,
                                                                      DECODE(TP_CONVENIO, 'A', EXA_RX.CD_PROCEDIMENTO_SIA,
                                                                                                                 'C', EXA_RX.EXA_RX_CD_PRO_FAT,
                                                                                                                 'P', EXA_RX.EXA_RX_CD_PRO_FAT,
                                                                                                                  'H', EXA_RX.CD_PROCEDIMENTO_SIH),
                                                                       PED_RX.DT_PEDIDO,
                                                                       PED_RX.HR_PEDIDO,
                                                                       ATENDIME.CD_CONVENIO,
                                                                       ATENDIME.CD_CON_PLA,
                                                                       ATENDIME.TP_ATENDIMENTO,
                                                                       ATENDIME.CD_TIP_ACOM, 'U'
                                                                       ,null
                                                                       )             VL_EXAME_A
FROM      DBAMV.ATENDIME
               ,DBAMV.PACIENTE
               ,DBAMV.CIDADE
               ,DBAMV.CONVENIO
               ,DBAMV.PED_RX
               ,DBAMV.ITPED_RX
               ,DBAMV.EXA_RX
               ,DBAMV.ORI_ATE
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
And :CAMPO_TP_APRESENTACAO   =  'A'
AND         PACIENTE.CD_CIDADE       =  CIDADE.CD_CIDADE(+)
AND         ATENDIME.CD_PACIENTE  =  PACIENTE.CD_PACIENTE
AND         ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND         ATENDIME.CD_ATENDIMENTO   =  PED_RX.CD_ATENDIMENTO
AND         PED_RX.CD_PED_RX          =  ITPED_RX.CD_PED_RX
AND         ITPED_RX.CD_EXA_RX       =  EXA_RX.CD_EXA_RX
AND         ATENDIME.CD_ORI_ATE     =  ORI_ATE.CD_ORI_ATE
GROUP BY CIDADE.CD_UF,
                   CIDADE.NM_CIDADE,
                   ATENDIME.CD_ORI_ATE,
                   ORI_ATE.DS_ORI_ATE,
                   DECODE(TP_CONVENIO, 'A', EXA_RX.CD_PROCEDIMENTO_SIA,
                                                              'C', EXA_RX.EXA_RX_CD_PRO_FAT,
                                                              'P', EXA_RX.EXA_RX_CD_PRO_FAT,
                                                              'H', EXA_RX.CD_PROCEDIMENTO_SIH),
                    EXA_RX.DS_EXA_RX
                   ,DBAMV.CALC_VL_PROC_UNIT( CONVENIO.TP_CONVENIO,
                                                                          DECODE(TP_CONVENIO, 'A', EXA_RX.CD_PROCEDIMENTO_SIA,
                                                                                                                     'C', EXA_RX.EXA_RX_CD_PRO_FAT,
                                                                                                                     'P', EXA_RX.EXA_RX_CD_PRO_FAT,
                                                                                                                     'H', EXA_RX.CD_PROCEDIMENTO_SIH),
                                                                           PED_RX.DT_PEDIDO,
                                                                           PED_RX.HR_PEDIDO,
                                                                           ATENDIME.CD_CONVENIO,
                                                                           ATENDIME.CD_CON_PLA,
                                                                           ATENDIME.TP_ATENDIMENTO,
                                                                           ATENDIME.CD_TIP_ACOM, 'U'
                                                                           ,null
                                                                           )
ORDER BY CIDADE.CD_UF
                   ,CIDADE.NM_CIDADE
)
ORDER BY NM_CIDADE_A,CD_UF_A,
CD_ORI_ATE_A,DS_ORI_ATE_A,CD_EXA_RX_A,DS_EXA_RX_A,QTD_A,VL_EXAME_A
