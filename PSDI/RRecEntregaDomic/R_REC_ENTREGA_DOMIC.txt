--R_REC_ENTREGA_DOMIC
Select
    Paciente.Nm_Paciente,
    Ped_Rx.Cd_Ped_Rx Codigo_Pedido,
    Ender_Coleta_Entrega.Vl_Servico,
    dbamv.pack_fncp.extensos(Ender_Coleta_Entrega.Vl_Servico) Vl_Servico_extenso,
    To_Char(Sysdate,'dd/mm/yyyy hh24') Data_Hora,
    USER USUARIO
  From
    Dbamv.Ped_Rx,
    Dbamv.Atendime,
    Dbamv.Paciente,
    Dbamv.Ender_Coleta_Entrega
 Where
    Ped_Rx.Cd_Atendimento = Atendime.Cd_Atendimento And
    Atendime.Cd_Paciente   = Paciente.Cd_Paciente    And
    Ped_Rx.Cd_Ped_Rx     = Ender_Coleta_Entrega.Cd_Ped_Rx And
    Ender_Coleta_Entrega.Tp_Servico = 'E' And
    Ped_Rx.Cd_Ped_Rx     =  :CAMPO_CD_PED_RX
   AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
