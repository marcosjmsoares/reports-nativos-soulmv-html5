--R_GERENCIA_ALTA
select
       aten.cd_atendimento
     , aten.cd_leito
     , leit.ds_resumo
     , paca.nm_paciente
     , (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND paca.CD_PACIENTE =  CD_PACIENTE) NM_SOCIAL_PACIENTE
     , trunc(aten.dt_alta_medica) +  Abs(trunc(aten.hr_alta_medica)-aten.hr_alta_medica)         dt_alta_medica
     , trunc(aten.dt_alta) +  Abs(trunc(aten.hr_alta)-aten.hr_alta)                                                 dt_alta
     , trunc(soli.dt_inicio_higieniza) +  Abs(trunc(soli.hr_inicio_higieniza)-soli.hr_inicio_higieniza) dt_inicio_higieniza
     , trunc(soli.dt_realizado) +  Abs(trunc(soli.hr_realizado)-soli.hr_realizado)                              dt_realizado
  from dbamv.atendime      aten
     , dbamv.paciente      paca
     , dbamv.solic_limpeza soli
     , dbamv.leito         leit
 where aten.dt_alta  between  :CAMPO_PDT_INICIAL and 
   and paca.cd_paciente    = aten.cd_paciente
   and  aten.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   and soli.cd_atendimento = aten.cd_atendimento
   and soli.cd_leito = aten.cd_leito
   and soli.TP_SOLICITACAO='A'
   and soli.dt_cancelamento is null
   and leit.cd_leito       = aten.cd_leito
   and aten.cd_atendimento_pai is null
   and DT_ALTA <= DT_INICIO_HIGIENIZA
order  by 4
