--R_HISTORICO_QUARTOS
SELECT * FROM (
SELECT DISTINCT SOLIC_LIMPEZA.CD_SOLIC_LIMPEZA,
               SOLIC_LIMPEZA.CD_ATENDIMENTO,
               PACIENTE.NM_PACIENTE,
               (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE,
               DECODE(:CAMPO_SITUACAO,'R',Trunc(SOLIC_LIMPEZA.DT_REALIZADO),
                                                   'C',Trunc(SOLIC_LIMPEZA.DT_CANCELAMENTO),Trunc(DT_SOLIC_LIMPEZA)) DATA,
               UNID_INT.DS_UNID_INT,
               DECODE(:CAMPO_SITUACAO,'R',SOLIC_LIMPEZA.HR_REALIZADO,
                                                   'C',SOLIC_LIMPEZA.HR_CANCELAMENTO,HR_SOLIC_LIMPEZA) HORA,
               LEITO.DS_RESUMO,
               DECODE ( :CAMPO_SITUACAO,'R',TIPO_LIMPEZA.DS_TIPO_LIMPEZA,SETOR.NM_SETOR) TIPO,
               DECODE(:CAMPO_SITUACAO,'R',FUNCIONARIO.NM_FUNC,
                                                   'C',NM_USUARIO_CANCEL,SOLIC_LIMPEZA.NM_USUARIO) USUARIO
FROM DBAMV.LEITO,
           DBAMV.SOLIC_LIMPEZA,
           DBAMV.TIPO_LIMPEZA,
           DBAMV.FUNCIONARIO,
           DBAMV.UNID_INT,
           DBAMV.SETOR,
           DBAMV.ATENDIME,
           DBAMV.PACIENTE
WHERE (SOLIC_LIMPEZA.CD_LEITO = LEITO.CD_LEITO)
               AND (LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT)
               AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
               AND (SETOR.CD_SETOR = SOLIC_LIMPEZA.CD_SETOR)
               AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
               AND ATENDIME.CD_ATENDIMENTO = SOLIC_LIMPEZA.CD_ATENDIMENTO
               AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
ORDER BY  UNID_INT.DS_UNID_INT,
                    DATA DESC
)
ORDER BY DS_UNID_INT,DATA DESC
