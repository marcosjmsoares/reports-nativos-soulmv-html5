--R_EST_LIB_LEITO
SELECT * FROM (
SELECT leito.cd_leito cd_leito
      ,leito.ds_resumo ds_resumo
      ,solic_limpeza.cd_solic_limpeza cd_solic_limpeza
      ,solic_limpeza.cd_tipo_limpeza cd_tipo_limpeza
      ,tipo_limpeza.ds_tipo_limpeza ds_tipo_limpeza
      ,trunc(solic_limpeza.dt_solic_limpeza) dt_solic_limpeza
      ,TO_CHAR(solic_limpeza.hr_solic_limpeza, 'HH24') hr_solic_limpeza
      ,solic_limpeza.dt_hr_fim_ag_higieniza dt_hr_fim_ag_higieniza
      ,solic_limpeza.dt_hr_ini_rouparia dt_hr_ini_rouparia
      ,solic_limpeza.dt_hr_fim_rouparia dt_hr_fim_rouparia
      ,solic_limpeza.dt_inicio_higieniza dt_inicio_higieniza
      ,solic_limpeza.hr_inicio_higieniza hr_inicio_higieniza
      ,solic_limpeza.dt_hr_fim_higieniza dt_hr_fim_higieniza
      ,solic_limpeza.dt_hr_ini_pos_higieniza dt_hr_ini_pos_higieniza
      ,solic_limpeza.dt_realizado dt_realizado
      ,TO_CHAR(solic_limpeza.hr_realizado, 'HH24') hr_realizado
      ,dbamv.fn_diferenca_data_minutos(To_Date(To_Char(Trunc(solic_limpeza.dt_solic_limpeza), 'dd/mm/yyyy') || to_char(solic_limpeza.hr_solic_limpeza,'hh24'), 'dd/mm/yyyy hh24') ,solic_limpeza.dt_hr_fim_ag_higieniza) TEMPO_AG
      ,dbamv.fn_diferenca_data_minutos(solic_limpeza.dt_hr_ini_rouparia , solic_limpeza.dt_hr_fim_rouparia) TEMPO_RO
      ,dbamv.fn_diferenca_data_minutos(To_Date(To_Char(Trunc(solic_limpeza.dt_inicio_higieniza), 'dd/mm/yyyy') || to_char(solic_limpeza.hr_inicio_higieniza,'hh24'), 'dd/mm/yyyy hh24') , solic_limpeza.dt_hr_fim_higieniza) TEMPO_HIG
      ,dbamv.fn_diferenca_data_minutos(solic_limpeza.dt_hr_ini_pos_higieniza , To_Date(To_Char(Trunc(solic_limpeza.dt_realizado), 'dd/mm/yyyy') || to_char(solic_limpeza.hr_realizado,'hh24'), 'dd/mm/yyyy hh24')  ) TEMPO_POS
  FROM dbamv.leito
      ,dbamv.solic_limpeza
      ,dbamv.tipo_limpeza
      ,dbamv.unid_int
 WHERE leito.cd_leito = solic_limpeza.cd_leito(+)
   AND leito.cd_unid_int = unid_int.cd_unid_int
   AND solic_limpeza.cd_tipo_limpeza = tipo_limpeza.cd_tipo_limpeza(+)
   AND solic_limpeza.sn_realizado = 'S'
   AND Trunc(solic_limpeza.dt_solic_limpeza) between :CAMPO_DT_INICIAL and 
   AND solic_limpeza.cd_multi_empresa = DBAMV.PKG_MV2000.LE_EMPRESA
ORDER BY Trunc(solic_limpeza.dt_solic_limpeza)
)
ORDER BY DT_SOLIC_LIMPEZA, HR_SOLIC_LIMPEZA
