--R_CONTA_RUBRICA
SELECT * FROM (
SELECT ALL CLASSIFICACAO_RUBRICA.CD_CLASSIFICACAO_RUBRICA,
CLASSIFICACAO_RUBRICA.DS_CLASSIFICACAO_RUBRICA, CONTA.CD_CONTA, RUBRICA.CD_RUBRICA, RUBRICA.DS_RUBRICA, SUB_ITEM_RUBRICA.DS_SUB_ITEM,
CONTA.DS_CONTA, SUB_ITEM_RUBRICA.CD_SUB_ITEM
FROM dbamv.CONTA, dbamv.CLASSIFICACAO_RUBRICA, dbamv.RUBRICA, dbamv.SUB_ITEM_RUBRICA
WHERE ((CONTA.CD_CONTA = RUBRICA.CD_CONTA)
 AND (CLASSIFICACAO_RUBRICA.CD_CLASSIFICACAO_RUBRICA = RUBRICA.CD_CLASSIFICACAO_RUBRICA)
 AND (RUBRICA.CD_RUBRICA = SUB_ITEM_RUBRICA.CD_RUBRICA)
 AND (RUBRICA.CD_CONTA = SUB_ITEM_RUBRICA.CD_CONTA))
:CAMPO_CF_SUB_ITEM
)
ORDER BY CD_CONTA,DS_CONTA,
CD_RUBRICA,DS_RUBRICA,
CD_CLASSIFICACAO_RUBRICA,DS_CLASSIFICACAO_RUBRICA
