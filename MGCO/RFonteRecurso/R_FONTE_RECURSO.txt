--R_FONTE_RECURSO
SELECT ALL FONTE_RECURSO.CD_FONTE_RECURSO, FONTE_RECURSO.DS_FONTE_RECURSO,
VINCULACAO.CD_VINCULACAO, VINCULACAO.DS_VINCULACAO
FROM dbamv.FONTE_RECURSO, dbamv.VINCULACAO
WHERE (FONTE_RECURSO.CD_VINCULACAO = VINCULACAO.CD_VINCULACAO)
