--R_SOL_COM_S_FORNEC
SELECT a.cd_sol_com
      ,a.sn_urgente
      ,a.ds_obs_cotacao
      ,a.dt_sol_com
      ,a.dt_maxima
      ,a.cd_cotador
      ,a.ds_observacao
      ,a.ds_cotador
      ,a.cd_sequencial
      ,a.cd_produto
      ,a.qt_solic
      ,a.ds_dica_da_compra
      ,a.ds_produto
      ,a.ds_unidade
      ,a.ds_especificacao
      ,( SELECT  nm_laborator
        FROM     laborator c
                ,lab_pro b
        WHERE    c.cd_laborator = b.cd_laborator
        AND      b.cd_produto = a.cd_produto
        AND      ROWNUM <= 1 ) nm_laborator
 FROM (
Select   sol_com.cd_sol_com		cd_sol_com,
          sol_com.sn_urgente		sn_urgente,
          sol_com.ds_obs_cotacao	ds_obs_cotacao ,
         sol_com.dt_sol_com		dt_sol_com,
         sol_com.dt_maxima		dt_maxima,
         sol_com.cd_cotador		cd_cotador,
         sol_com.ds_observacao                ds_observacao,
         cotador.ds_cotador		ds_cotador,
         itsol_com.cd_sequencial,
         itsol_com.cd_produto		cd_produto,
         itsol_com.qt_solic		qt_solic,
         itsol_com.ds_dica_da_compra       ds_dica_da_compra,
         PRODUTO.ds_produto || DECODE(PRODUTO.ds_complemento, null,null,' ') || PRODUTO.ds_complemento || DECODE(PRODUTO.nr_referencia, null,null,' ') || PRODUTO.nr_referencia	ds_produto,
         uni_pro.ds_unidade		ds_unidade,
         produto.ds_especificacao               ds_especificacao
From   dbamv.sol_com              sol_com,
       dbamv.itsol_com            itsol_com,
       dbamv.cotador              cotador,
       dbamv.produto              produto,
       dbamv.uni_pro              uni_pro,
       dbamv.estoque             estoque
Where  sol_com.tp_situacao      not in ( 'C', 'F' )
and    sol_com.cd_sol_com       = itsol_com.cd_sol_com
and    sol_com.cd_cotador       = cotador.cd_cotador(+)
and    itsol_com.cd_produto     = produto.cd_produto
and    itsol_com.cd_uni_pro     = uni_pro.cd_uni_pro
and    sol_com.cd_sol_com       = :CAMPO_SOLIC
and    itsol_com.cd_mot_cancel is null
and    Estoque.Cd_Estoque         =  Sol_Com.Cd_Estoque
and    Estoque.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
Union All
Select  sol_com.cd_sol_com             cd_sol_com,
          sol_com.sn_urgente		sn_urgente,
        sol_com.ds_obs_cotacao              ds_obs_cotacao ,
        sol_com.dt_sol_com             dt_sol_com,
        sol_com.dt_maxima              dt_maxima,
        sol_com.cd_cotador             cd_cotador,
        sol_com.ds_observacao	  ds_observacao,
        cotador.ds_cotador           ds_cotador,
        itsol_ser.cd_sequencial,
        itsol_ser.cd_ser_sol           cd_produto,
        0                            qt_solic,
        null                         ds_dica_da_compra,
        manu_serv.nm_servico||' - '|| itsol_ser.ds_servico           ds_produto,
        'Serviço'                      ds_unidade,
        null                                        ds_especificacao
From   dbamv.sol_com                 sol_com,
       dbamv.itsol_ser               itsol_ser,
       dbamv.cotador                 cotador,
       dbamv.manu_serv,
       dbamv.estoque                 estoque
Where  sol_com.tp_situacao       not in ( 'C', 'F' )
and    sol_com.cd_sol_com        = itsol_ser.cd_sol_com
and    sol_com.cd_cotador        = cotador.cd_cotador(+)
and    sol_com.cd_sol_com        = :CAMPO_SOLIC
and    manu_serv.cd_servico           = itsol_ser.cd_ser_sol
and Estoque.Cd_Estoque         =  Sol_Com.Cd_Estoque
and Estoque.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
) a
:CAMPO_CF_ORDEM, DS_OBS_COTACAO,CD_COTADOR,DS_COTADOR,CD_SOL_COM,DT_SOL_COM,DT_MAXIMA,DS_OBSERVACAO,SN_URGENTE
