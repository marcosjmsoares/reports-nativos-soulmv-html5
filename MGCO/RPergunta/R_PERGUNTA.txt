--R_PERGUNTA
SELECT * FROM(
Select Iso_Pergunta.Cd_Pergunta
	,Iso_Pergunta.Ds_Pergunta
	,Iso_Pergunta.Nm_Identificador
	,Iso_ItPergunta.Cd_Itpergunta
	,Iso_ItPergunta.Ds_Itpergunta
	,Iso_ItPergunta.Vl_Itpergunta
  From Dbamv.Iso_ItPergunta
	,Dbamv.Iso_Pergunta
 Where Iso_ItPergunta.Cd_Pergunta = Iso_Pergunta.Cd_Pergunta
)
