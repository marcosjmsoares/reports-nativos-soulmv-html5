--R_PEDCOMP
select sysdate from dual

--R_PEDCOMP_R_PED_COM_Q_PED_COM
SELECT  ORD_COM.CD_ORD_COM                          CD_OR_COM
                  ,FORNECEDOR.CD_FORNECEDOR      CD_FORNECEDOR
                  ,FORNECEDOR.NM_FORNECEDOR      NM_FORNECEDOR
                  ,FORNECEDOR.NR_CGC_CPF                  NR_CGC_CPF
                  ,FORNECEDOR.NR_INSC_EST                NR_INSC_EST
                  ,FORNECEDOR.DS_ENDERECO             DS_ENDERECO
                  ,FORNECEDOR.DS_BAIRRO                     DS_BAIRRO
                  ,FORNECEDOR.NR_CEP                             NR_CEP  
                 ,CIDADE.CD_CIDADE                                      CD_CIDADE
                 ,CIDADE.NM_CIDADE                                     NM_CIDADE
                 ,CIDADE.CD_UF                                                                   CD_UF
                 ,FORNECEDOR.NM_CONTATO                                    NM_CONTATO
                 ,COND_PAGTO.DS_CONDICAO_PAGAMENTO    DS_CONDICAO_PAGAMENTO
                ,ORD_COM.VL_PERC_DESCONTO                              VL_PERC_DESCONTO_ORD_COM
                ,ORD_COM.VL_DESCONTO                                            VL_DESCONTO_ORD_COM 
                ,ORD_COM.DT_PREV_ENTREGA                                 DT_PREV_ENTREGA
                ,ORD_COM.TP_FRETE                                                      TP_FRETE
                ,ORD_COM.VL_FRETE                                                       VL_FRETE
               ,ORD_COM.DS_ORDEM_DE_COMPRA                       DS_ORDEM_DE_COMPRA
               ,ORD_COM.VL_TOTAL                                                        VL_TOTAL_ORD_COM
FROM   DBAMV.ORD_COM            ORD_COM
               ,DBAMV.FORNECEDOR         FORNECEDOR
               ,DBAMV.CIDADE             CIDADE
               ,DBAMV.CONDICOES_PAGAMENTO COND_PAGTO
	       , DBAMV.ESTOQUE 
WHERE ORD_COM.CD_FORNECEDOR = FORNECEDOR.CD_FORNECEDOR
AND   FORNECEDOR.TP_CLIENTE_FORN IN ( 'A', 'F' )
AND   FORNECEDOR.CD_CIDADE = CIDADE.CD_CIDADE
AND   ORD_COM.CD_ORD_COM = :CAMPO_CD_ORD_COM
AND        ESTOQUE.CD_ESTOQUE = ORD_COM.CD_ESTOQUE 
AND        ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA 
AND   ORD_COM.CD_CONDICAO_PAGAMENTO = COND_PAGTO.CD_CONDICAO_PAGAMENTO
