--R_ATIFOR
SELECT * FROM (
SELECT ALL DBAMV.TIP_ATIV.CD_TIP_ATIV,
                      DBAMV.TIP_ATIV.DS_TIP_ATIV,
                      DBAMV.FORNECEDOR.CD_FORNECEDOR,
                      DBAMV.FORNECEDOR.NM_FANTASIA,
                      DBAMV.FORNECEDOR.NM_FORNECEDOR
FROM DBAMV.ATI_FOR,
           DBAMV.FORNECEDOR,
           DBAMV.TIP_ATIV
WHERE DBAMV.FORNECEDOR.TP_CLIENTE_FORN IN ('F', 'A', 'R', 'T')
      AND DBAMV.ATI_FOR.CD_FORNECEDOR=DBAMV.FORNECEDOR.CD_FORNECEDOR
      AND DBAMV.ATI_FOR.CD_TIP_ATIV=DBAMV.TIP_ATIV.CD_TIP_ATIV
:CAMPO_CF_ATIVIDADE
)
ORDER BY :CAMPO_FC_ORDEM,CD_TIP_ATIV,DS_TIP_ATIV
