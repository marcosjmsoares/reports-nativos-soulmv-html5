--R_ACOMPANHAMENTO_ORD_COM
SELECT * FROM (
Select acompanhamento_ordem_compra.cd_acompanhamento_ordem_compra,
           acompanhamento_ordem_compra.cd_ord_com,
           acompanhamento_ordem_compra.dh_registro,
           acompanhamento_ordem_compra.dt_contato,
           acompanhamento_ordem_compra.hr_contato,
           acompanhamento_ordem_compra.nm_contato,
           acompanhamento_ordem_compra.nr_fone_contato,
           acompanhamento_ordem_compra.ds_email_contato,
           acompanhamento_ordem_compra.ds_andamento,
           acompanhamento_ordem_compra.tp_status,
           acompanhamento_ordem_compra.cd_Acompanhamento_Ord_Cpr_pai,
           usuarios.nm_usuario,
           ord_com.dt_ord_com,
           ord_com.cd_sol_com,
           fornecedor.nm_fornecedor,
           estoque.ds_estoque
   from dbamv.ord_com,
           dbamv.fornecedor,
           dbamv.estoque,
           dbasgu.usuarios,
           (Select acompanhamento_ordem_compra.cd_acompanhamento_ordem_compra,
                       acompanhamento_ordem_compra.cd_ord_com,
                       acompanhamento_ordem_compra.dh_registro,
                       acompanhamento_ordem_compra.dt_contato,
                       acompanhamento_ordem_compra.hr_contato,
                       acompanhamento_ordem_compra.nm_contato,
                       acompanhamento_ordem_compra.nr_fone_contato,
                       acompanhamento_ordem_compra.ds_email_contato,
                       acompanhamento_ordem_compra.ds_andamento,
                       acompanhamento_ordem_compra.cd_usuario,
                       acompanhamento_ordem_compra.tp_status,
                       acompanhamento_ordem_compra.cd_Acompanhamento_Ord_Cpr_pai
               from dbamv.acompanhamento_ordem_compra
             where acompanhamento_ordem_compra.cd_ord_com    = :CAMPO_ORD_COM
                       Connect By prior Cd_Acompanhamento_Ordem_Compra = Cd_Acompanhamento_Ord_Cpr_pai
                       Start With Cd_Acompanhamento_Ord_Cpr_pai Is Null) acompanhamento_ordem_compra
where ord_com.cd_ord_com     =  acompanhamento_ordem_compra.cd_ord_com
    and ord_com.cd_fornecedor = fornecedor.cd_fornecedor
    and ord_com.cd_estoque     = estoque.cd_estoque
    and usuarios.cd_usuario       = acompanhamento_ordem_compra.cd_usuario
)
ORDER BY CD_ORD_COM,DT_ORD_COM,NM_FORNECEDOR,CD_SOL_COM,DS_ESTOQUE
