--R_SOL_ALTERADA
Select coleta.cd_coleta,
        (itcol_pro.qt_cotada - itsol_com.qt_solic) qtd_compl,
       Uni_Pro.ds_unidade,
       itcol_pro.cd_produto,
       PRODUTO.ds_produto || DECODE(PRODUTO.ds_complemento, null,null,' ') || PRODUTO.ds_complemento || DECODE(PRODUTO.nr_referencia, null,null,' ') || PRODUTO.nr_referencia	ds_produto,
       itcol_pro.ds_just_qtd_alt,
       coleta.cd_fornecedor ||'-'||Fornecedor.nm_fornecedor Fornecedor
 From  Dbamv.Coleta,
       Dbamv.Itcol_pro,
       Dbamv.Sol_com,
       Dbamv.Itsol_com,
       Dbamv.Ord_com,
       Dbamv.Itord_pro,
       Dbamv.Produto,
       Dbamv.Uni_Pro,
       Dbamv.Fornecedor,
       Dbamv.estoque 
 Where  Coleta.Cd_coleta     = Itcol_pro.Cd_coleta
   And Coleta.Cd_sol_com    = Sol_com.Cd_sol_com
   And Sol_com.Cd_sol_com   = Itsol_com.Cd_sol_com
   And Itcol_pro.Cd_produto = Itsol_com.Cd_produto
   And Coleta.Cd_fornecedor = Ord_com.Cd_fornecedor
   And Coleta.Cd_sol_com    = Ord_com.Cd_sol_com
   And Ord_com.Cd_ord_com   = Itord_pro.Cd_ord_com
   And Itcol_pro.Cd_produto = Itord_pro.Cd_produto
   And Itcol_Pro.Cd_Produto = Produto.Cd_Produto
   And Itcol_Pro.Cd_Uni_pro = Uni_Pro.cd_Uni_Pro
   And Uni_Pro.Cd_Produto   = Produto.Cd_Produto
   And Ord_Com.cd_fornecedor = Fornecedor.cd_fornecedor
   and ORD_COM.Cd_estoque = estoque.Cd_estoque 
   and Cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA 
  And itcol_pro.qt_cotada <> itsol_com.qt_solic
