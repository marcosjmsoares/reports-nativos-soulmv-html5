--R_PTRES
SELECT ALL PTRES.CD_PTRES, PTRES.DS_OBSERVACAO, TIPO_ORCAMENTO.DS_TIPO_ORCAMENTO
FROM dbamv.PROGRAMA_TRABALHO_RESUMIDO PTRES, dbamv.TIPO_ORCAMENTO
WHERE (PTRES.CD_TIPO_ORCAMENTO = TIPO_ORCAMENTO.CD_TIPO_ORCAMENTO)
