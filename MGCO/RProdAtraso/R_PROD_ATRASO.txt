--R_PROD_ATRASO
SELECT * FROM (
SELECT ORD_COM.DT_PREV_ENTREGA
              ,ORD_COM.CD_ORD_COM
              ,PRODUTO.DS_PRODUTO
              ,UNI_PRO.DS_UNIDADE
              ,(ENT_PRO.NR_DOCUMENTO || ' - ' || ENT_PRO.NR_SERIE) NR_DOCUMENTO
              ,ENT_PRO.DT_ENTRADA
              ,FORNECEDOR.NM_FORNECEDOR
              ,ITORD_PRO.QT_COMPRADA
              ,ITENT_PRO.QT_ENTRADA
FROM   DBAMV.ORD_COM
             ,DBAMV.FORNECEDOR
             ,DBAMV.ESTOQUE
             ,DBAMV.ITORD_PRO
             ,DBAMV.PRODUTO
             ,DBAMV.UNI_PRO
             ,DBAMV.ENT_PRO
             ,DBAMV.ITENT_PRO
WHERE  ORD_COM.CD_FORNECEDOR = FORNECEDOR.CD_FORNECEDOR
AND        FORNECEDOR.TP_CLIENTE_FORN IN ( 'F', 'A' )
AND        ORD_COM.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
AND        ORD_COM.CD_ORD_COM = ITORD_PRO.CD_ORD_COM
AND        ITORD_PRO.CD_PRODUTO = PRODUTO.CD_PRODUTO
AND        ITORD_PRO.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO
AND        ORD_COM.CD_ORD_COM = ENT_PRO.CD_ORD_COM
AND        ENT_PRO.CD_ENT_PRO = ITENT_PRO.CD_ENT_PRO
AND        ITENT_PRO.CD_PRODUTO = ITORD_PRO.CD_PRODUTO
AND        ORD_COM.DT_PREV_ENTREGA < ENT_PRO.DT_ENTRADA
AND        ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND        ORD_COM.DT_PREV_ENTREGA BETWEEN :CAMPO_DATA_INI AND 
and          ITORD_PRO.cd_mot_cancel is null
ORDER BY ORD_COM.DT_PREV_ENTREGA, ORD_COM.CD_ORD_COM, PRODUTO.DS_PRODUTO
)
ORDER BY DT_PREV_ENTREGA,CD_ORD_COM, DS_PRODUTO
