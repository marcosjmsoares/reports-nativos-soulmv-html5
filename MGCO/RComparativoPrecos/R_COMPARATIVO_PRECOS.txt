--R_COMPARATIVO_PRECOS
SELECT * FROM (
select produto.cd_especie,
          especie.ds_especie,
          produto.cd_produto,
          produto.ds_produto,
          produto.cd_pro_fat,
          produto.vl_fator_pro_fat,
         (sum(itent_pro.qt_entrada    * uni_pro.vl_fator ) / dbamv.verif_vl_fator_prod (produto.cd_produto) )         qt_entrada,
         sum(itent_pro.vl_custo_real * itent_pro.qt_entrada)  vl_total_custo_real
  from dbamv.produto,
       dbamv.especie,
       dbamv.itent_pro,
       dbamv.uni_pro,
       dbamv.ent_pro,
       dbamv.estoque
 where produto.cd_produto   = itent_pro.cd_produto
   and produto.cd_especie   = especie.cd_especie
   and itent_pro.cd_uni_pro = uni_pro.cd_uni_pro
   and itent_pro.cd_ent_pro = ent_pro.cd_ent_pro
   and ent_pro.cd_fornecedor is not null
   and ent_pro.dt_entrada between :CAMPO_DT_INICIAL and 
   and estoque.Cd_Estoque       = Ent_Pro.Cd_Estoque
   and estoque.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
group by produto.cd_especie,
         especie.ds_especie,
         produto.cd_produto,
         produto.ds_produto,
         produto.cd_pro_fat,
         produto.vl_fator_pro_fat
Order By  Produto.ds_produto
)
ORDER BY cd_produto, CD_ESPECIE,DS_ESPECIE
