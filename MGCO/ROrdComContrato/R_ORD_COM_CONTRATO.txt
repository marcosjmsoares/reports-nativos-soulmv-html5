--R_ORD_COM_CONTRATO
SELECT  ORD_COM.CD_ORD_COM,
               ORD_COM.CD_SOL_COM,
        FORNECEDOR.CD_FORNECEDOR,
        FORNECEDOR.NM_FORNECEDOR,
        FORNECEDOR.NM_FANTASIA,
        FORNECEDOR.NR_CGC_CPF,
        FORNECEDOR.NR_INSC_EST,
        FORNECEDOR.DS_ENDERECO,
        FORNECEDOR.NR_ENDERECO,
        FORNECEDOR.DS_COMPLEMENTO,
        FORNECEDOR.DS_BAIRRO,
        FORNECEDOR.NR_CEP,
        CIDADE.NM_CIDADE,
        CIDADE.CD_UF,
        FORNECEDOR.NR_CONTA,
	ORD_COM.TP_ORD_COM,
        TO_DATE(TO_CHAR(ORD_COM.DT_PREV_ENTREGA,'DD'),'DD')         DT_PREV_ENTREGA1,
        ORD_COM.DT_INICIO_CONTRATO,
        ORD_COM.DT_FINAL_CONTRATO,
        ORD_COM.VL_PERC_DESCONTO,
        ORD_COM.VL_DESCONTO,
        ORD_COM.DT_PREV_ENTREGA,
        ORD_COM.TP_FRETE,
        ORD_COM.VL_FRETE,
        ORD_COM.DS_ORDEM_DE_COMPRA ,
        ORD_COM.VL_TOTAL    VL_TOTAL_ORD_COM,
        CONDICOES_PAGAMENTO.CD_CONDICAO_PAGAMENTO,
        CONDICOES_PAGAMENTO.DS_CONDICAO_PAGAMENTO
        ,ORD_COM.DT_ORD_COM DT_LANCAMENTO
FROM    DBAMV.ORD_COM
       ,DBAMV.ESTOQUE
       ,DBAMV.FORNECEDOR
       ,DBAMV.SOL_COM
       ,DBAMV.CIDADE
       ,DBAMV.CONDICOES_PAGAMENTO
WHERE   ORD_COM.CD_ORD_COM         = :CAMPO_CD_ORD_COM
AND     ORD_COM.CD_ESTOQUE             = ESTOQUE.CD_ESTOQUE
AND     ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND     ORD_COM.CD_FORNECEDOR      = FORNECEDOR.CD_FORNECEDOR
AND     FORNECEDOR.TP_CLIENTE_FORN IN ( 'F', 'A', 'R', 'T' )
AND     FORNECEDOR.CD_CIDADE        = CIDADE.CD_CIDADE
AND     ORD_COM.CD_SOL_COM           = SOL_COM.CD_SOL_COM(+)
AND     ORD_COM.CD_CONDICAO_PAGAMENTO = CONDICOES_PAGAMENTO.CD_CONDICAO_PAGAMENTO
order by ORD_COM.CD_ORD_COM
