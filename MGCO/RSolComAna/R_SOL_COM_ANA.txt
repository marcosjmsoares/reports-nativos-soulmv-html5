--R_SOL_COM_ANA
SELECT * FROM (
  SELECT
        sol_com.dt_impressao ,
        to_date(to_char(PRODUTO.DT_ULTIMA_ENTRADA, 'dd/mm/yyyy'), 'dd/mm/yyyy') ULT_ENT
       ,PRODUTO.QT_ULTIMA_ENTRADA / UNI_PRO.VL_FATOR QT_ULTIMA_ENT
       ,SOL_COM.CD_SOL_COM		CD_SOL_COM
       ,SOL_COM.SN_URGENTE		SN_URGENTE
       ,SOL_COM.DT_SOL_COM		DT_SOL_COM
       ,TRUNC(SOL_COM.DT_MAXIMA)		DT_MAXIMA
       ,SOL_COM.CD_SETOR			CD_SETOR
       ,SETOR.NM_SETOR			NM_SETOR
       ,SOL_COM.TP_SITUACAO		TP_SITUACAO
       ,SOL_COM.NM_SOLICITANTE		NM_SOLICITANTE
       ,SOL_COM.CD_MOT_PED		CD_MOT_PED
       ,MOT_PED.DS_MOT_PED		DS_MOT_PED
       ,SOL_COM.CD_COTADOR		CD_COTADOR
       ,COTADOR.DS_COTADOR		DS_COTADOR
       ,SOL_COM.CD_ESTOQUE		CD_ESTOQUE
       ,ESTOQUE.DS_ESTOQUE		DS_ESTOQUE
       ,SOL_COM.DS_OBSERVACAO		DS_OBSERVACAO
       ,PRODUTO.CD_SICAN
       ,PRODUTO.SN_MESTRE                               SN_MESTRE
       ,PRODUTO.VL_ULTIMA_ENTRADA  *   UNI_PRO.VL_FATOR  VL_ULTIMA_CUSTO_REAL
       ,(PRODUTO.VL_ULTIMA_CUSTO_REAL  *   UNI_PRO.VL_FATOR) VL_ULTIMA_COMPRA
      ,(PRODUTO.VL_ULTIMA_CUSTO_REAL  *   UNI_PRO.VL_FATOR) * ITSOL_COM.QT_SOLIC  VL_TOTAL
        ,'PRODUTO'				TP_DO_ITEM
       ,ITSOL_COM.CD_SEQUENCIAL
       ,ITSOL_COM.CD_PRODUTO		CD_PRODUTO
       ,PRODUTO.ds_produto || DECODE(PRODUTO.ds_complemento, null,null,' ') || PRODUTO.ds_complemento || DECODE(PRODUTO.nr_referencia, null,null,' ') || PRODUTO.nr_referencia	DS_PRODUTO
       ,ITSOL_COM.QT_SOLIC			QT_SOLIC_PRO
       ,UNI_PRO.DS_UNIDADE		DS_UNIDADE
       ,ITSOL_COM.DS_DICA_DA_COMPRA
       ,PRODUTO.DS_ESPECIFICACAO
       ,FORNECEDOR.NM_FANTASIA		ULT_FORNECEDOR
       ,UNI_PRO.VL_FATOR                                     VL_FATOR
       ,EST_PRO.QT_ESTOQUE_ATUAL / UNI_PRO.VL_FATOR QT_ESTOQUE_ATUAL
       ,' '                                                       DS_SERVICO
       ,SOL_COM.CD_CONTA ||'-'|| CONTA.DS_CONTA   DS_CONTA
       ,SOL_COM.CD_RUBRICA ||'-'|| RUBRICA.DS_RUBRICA  DS_RUBRICA
       ,SOL_COM.CD_FONTE_RECURSO ||'-'|| FONTE_RECURSO.DS_FONTE_RECURSO  DS_FONTE_RECURSO
       ,SOL_COM.TP_SOL_COM
       ,Decode(Sol_com.sn_opme,'N','','TIPO : OPME')  DS_OPME
       ,EMPRESA_PRODUTO.TP_CLASSIFICACAO_ABC CLASSIFICACAO_ABC,
       SOL_COM.VL_TOTAL VL_TOTAL_SOLIC
FROM    DBAMV.SOL_COM		SOL_COM
       ,DBAMV.COTADOR		COTADOR
       ,DBAMV.SETOR		SETOR
       ,DBAMV.ESTOQUE		ESTOQUE
       ,DBAMV.MOT_PED		MOT_PED
       ,DBAMV.EST_PRO		EST_PRO
       ,DBAMV.ITSOL_COM		ITSOL_COM
       ,DBAMV.PRODUTO		PRODUTO
       ,DBAMV.UNI_PRO		UNI_PRO
       ,DBAMV.FORNECEDOR		FORNECEDOR
       ,DBAMV.CONTA
       ,DBAMV.RUBRICA
       ,DBAMV.FONTE_RECURSO
       ,DBAMV.EMPRESA_PRODUTO     EMPRESA_PRODUTO
WHERE  SOL_COM.CD_COTADOR		= COTADOR.CD_COTADOR(+)
AND    SOL_COM.CD_SETOR		= SETOR.CD_SETOR
AND    SOL_COM.CD_ESTOQUE		= NVL( EST_PRO.CD_ESTOQUE, SOL_COM.CD_ESTOQUE )
AND    SOL_COM.CD_MOT_PED		= MOT_PED.CD_MOT_PED
AND    ITSOL_COM.CD_SOL_COM		= SOL_COM.CD_SOL_COM
AND    ITSOL_COM.CD_PRODUTO		= PRODUTO.CD_PRODUTO
AND    ITSOL_COM.CD_UNI_PRO		= UNI_PRO.CD_UNI_PRO
AND    PRODUTO.CD_PRODUTO		= EST_PRO.CD_PRODUTO(+)
AND    ESTOQUE.CD_ESTOQUE		(+)= EST_PRO.CD_ESTOQUE
AND    FORNECEDOR.CD_FORNECEDOR	(+)= PRODUTO.CD_ULTIMO_FORNECEDOR
AND   ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND    SOL_COM.CD_CONTA = CONTA.CD_CONTA(+)
AND    SOL_COM.CD_RUBRICA = RUBRICA.CD_RUBRICA(+)
AND    SOL_COM.CD_FONTE_RECURSO = FONTE_RECURSO.CD_FONTE_RECURSO(+)
AND    EMPRESA_PRODUTO.CD_MULTI_EMPRESA(+)   = :CAMPO_CD_MULTI_EMPRESA
AND    EMPRESA_PRODUTO.CD_PRODUTO(+)         =  PRODUTO.CD_PRODUTO
UNION ALL
SELECT
        sol_com.dt_impressao
       ,sysdate		                                ULT_ENT
       ,0					QT_ULTIMA_ENT
       ,SOL_COM.CD_SOL_COM		CD_SOL_COM
       ,SOL_COM.SN_URGENTE		SN_URGENTE
       ,SOL_COM.DT_SOL_COM		DT_SOL_COM
       ,TRUNC(SOL_COM.DT_MAXIMA)		DT_MAXIMA
       ,SOL_COM.CD_SETOR			CD_SETOR
       ,SETOR.NM_SETOR                   		NM_SETOR
       ,SOL_COM.TP_SITUACAO		TP_SITUACAO
       ,SOL_COM.NM_SOLICITANTE		NM_SOLICITANTE
       ,SOL_COM.CD_MOT_PED		CD_MOT_PED
       ,MOT_PED.DS_MOT_PED		DS_MOT_PED
       ,SOL_COM.CD_COTADOR		CD_COTADOR
       ,COTADOR.DS_COTADOR		DS_COTADOR
       ,SOL_COM.CD_ESTOQUE		CD_ESTOQUE
       ,ESTOQUE.DS_ESTOQUE		DS_ESTOQUE
       ,SOL_COM.DS_OBSERVACAO		DS_OBSERVACAO
       ,''					cd_sican
       ,'N'                                                                    sn_mestre
       ,0					vl_ultima_custo_real
       ,0                                                                      VL_ULTIMA_COMPRA
      ,SOL_COM.VL_TOTAL
       ,'SERVIÃ‡O'			    TP_DO_ITEM
       ,ITSOL_SER.CD_SEQUENCIAL
       ,ITSOL_SER.CD_SER_SOL	    CD_PRODUTO
       ,MANU_SERV.NM_SERVICO	    DS_PRODUTO
       ,0				    QT_SOLIC_PRO
       ,''				     DS_UNIDADE
       ,null                                                         DS_DICA_DA_COMPRA
       ,null                                                         DS_ESPECIFICACAO
       ,Null				     ULT_FORNECEDOR
       ,0                                                           VL_FATOR
       ,0                                                           qt_estoque_atual
    ,ITSOL_SER.DS_SERVICO	     DS_SERVICO
       ,SOL_COM.CD_CONTA ||'-'|| CONTA.DS_CONTA   DS_CONTA
       ,SOL_COM.CD_RUBRICA ||'-'|| RUBRICA.DS_RUBRICA  DS_RUBRICA
       ,SOL_COM.CD_FONTE_RECURSO ||'-'|| FONTE_RECURSO.DS_FONTE_RECURSO  DS_FONTE_RECURSO
       ,SOL_COM.TP_SOL_COM
       ,Decode(Sol_com.sn_opme,'N','','TIPO : OPME')  DS_OPME
      , NULL  CLASSIFICACAO_ABC,
      SOL_COM.VL_TOTAL VL_TOTAL_SOLIC
FROM    DBAMV.SOL_COM		SOL_COM
       ,DBAMV.COTADOR		COTADOR
       ,DBAMV.SETOR		SETOR
       ,DBAMV.ESTOQUE		ESTOQUE
       ,DBAMV.MOT_PED		MOT_PED
       ,DBAMV.ITSOL_SER		ITSOL_SER
       ,DBAMV.MANU_SERV                    MANU_SERV
       ,DBAMV.CONTA
       ,DBAMV.RUBRICA
       ,DBAMV.FONTE_RECURSO
WHERE  SOL_COM.CD_COTADOR	= COTADOR.CD_COTADOR(+)
AND    SOL_COM.CD_SETOR	= SETOR.CD_SETOR
AND    SOL_COM.CD_ESTOQUE	= ESTOQUE.CD_ESTOQUE
AND    SOL_COM.CD_MOT_PED	= MOT_PED.CD_MOT_PED
AND    ITSOL_SER.CD_SOL_COM	= SOL_COM.CD_SOL_COM
AND  ITSOL_SER.CD_SER_SOL	= MANU_SERV.CD_SERVICO
AND ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND    SOL_COM.CD_CONTA = CONTA.CD_CONTA(+)
AND    SOL_COM.CD_RUBRICA = RUBRICA.CD_RUBRICA(+)
AND    SOL_COM.CD_FONTE_RECURSO = FONTE_RECURSO.CD_FONTE_RECURSO(+)
:CAMPO_CF_PRODUTO
)
:CAMPO_CF_ORDEM , DT_SOL_COM,DT_MAXIMA,NM_SOLICITANTE,CD_SETOR,NM_SETOR,CD_MOT_PED,DS_MOT_PED,TP_SITUACAO,DS_OBSERVACAO,SN_URGENTE,CD_SOL_COM,DT_IMPRESSAO,DS_CONTA,DS_RUBRICA,DS_FONTE_RECURSO,DS_OPME
