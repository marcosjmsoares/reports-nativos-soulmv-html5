--R_DOC_FORNECEDOR
SELECT * FROM (
SELECT distinct DOC_COMPRAS.CD_DOC_COMPRAS
               ,DOC_COMPRAS.NM_DOC_COMPRAS
               ,FORNECEDOR.CD_FORNECEDOR
               ,FORNECEDOR.NM_FORNECEDOR
               ,FORNEC_DOC_COMPRAS.DT_EFETIVACAO
               ,to_char(FORNEC_DOC_COMPRAS.DT_VALIDADE,'dd/mm/yyyy') DT_VALIDADE
               ,DECODE(DOC_COMPRAS.SN_BLOQUEIA,'S','Sim','N','Nao',Null) SN_BLOQUEIA
  FROM DBAMV.DOC_COMPRAS
       ,DBAMV.TIP_ATIV
       ,DBAMV.FORNECEDOR
       ,DBAMV.FORNEC_DOC_COMPRAS
       ,DBAMV.ATI_FOR
 WHERE DOC_COMPRAS.CD_DOC_COMPRAS 	= FORNEC_DOC_COMPRAS.CD_DOC_COMPRAS
   AND FORNECEDOR.CD_FORNECEDOR   	= FORNEC_DOC_COMPRAS.CD_FORNECEDOR
   AND FORNECEDOR.CD_FORNECEDOR   	= ATI_FOR.CD_FORNECEDOR
   AND TIP_ATIV.CD_TIP_ATIV       		= ATI_FOR.CD_TIP_ATIV
:CAMPO_CF_CRITERIO
)
ORDER BY CD_FORNECEDOR,NM_FORNECEDOR
