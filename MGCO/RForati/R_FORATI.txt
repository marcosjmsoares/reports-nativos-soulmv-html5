--R_FORATI
Select  ati_for.cd_fornecedor             cd_fornecedor,
            fornecedor.nm_fornecedor     nm_fornecedor,
            fornecedor.nm_fantasia          nm_fantasia,
            ati_for.cd_tip_ativ                   cd_tip_ativ,
            tip_ativ.ds_tip_ativ                 ds_tip_ativ
  From  dbamv.ati_for                          ati_for,
           dbamv.fornecedor                   fornecedor,
           dbamv.tip_ativ                         tip_ativ
 Where  fornecedor.cd_fornecedor     =    ati_for.cd_fornecedor
and        fornecedor.tp_cliente_forn    in ( 'F', 'A', 'R', 'T' )
and        tip_ativ.cd_tip_ativ                =    ati_for.cd_tip_ativ
