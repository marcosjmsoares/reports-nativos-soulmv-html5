--R_NOTIFICACAO_FORNECEDOR
SELECT * FROM (
SELECT         NOTIFICACAO_FORNECEDOR.CD_NOTIFICA_FORNECEDOR         CD_NOTIFICA_FORNECEDOR
              ,NOTIFICACAO_FORNECEDOR.DT_NOTIFICA_FORNECEDOR         DT_NOTIFICA_FORNECEDOR
              ,NOTIFICACAO_FORNECEDOR.HR_NOTIFICA_FORNECEDOR         HR_NOTIFICA_FORNECEDOR
              ,NOTIFICACAO_FORNECEDOR.DT_ENTRADA                     DT_ENTRADA
              ,NOTIFICACAO_FORNECEDOR.NR_DOCUMENTO                   NR_DOCUMENTO
              ,NOTIFICACAO_FORNECEDOR.NR_SERIE                       NR_SERIE
              ,NOTIFICACAO_FORNECEDOR.DS_MOTIVO_NOTIFICA             DS_MOTIVO_NOTIFICA
              ,NOTIFICACAO_FORNECEDOR.CD_FORNECEDOR                  CD_FORNECEDOR
	      ,FORNECEDOR.TP_FORNECEDOR                              TP_FORNECEDOR
              ,FORNECEDOR.NM_FORNECEDOR                              NM_FORNECEDOR
              ,FORNECEDOR.NR_CGC_CPF                                 NR_CGC_CPF
              ,nvl(NOTIFICACAO_FORNECEDOR.DS_JUSTIFICATIVA,'')       DS_JUSTIFICATIVA
              ,nvl(NOTIFICACAO_FORNECEDOR.DS_EMAIL_FORNECEDOR,'')    DS_EMAIL_FORNECEDOR
              ,nvl(NOTIFICACAO_FORNECEDOR.DS_EMAIL_ADICIONAL_FORNECEDOR, '') DS_EMAIL_ADICIONAL_FORNECEDOR
              ,NOTIFICACAO_FORNECEDOR.CD_ENT_PRO                     CD_ENT_PRO
              ,NOTIFICACAO_FORNECEDOR.VL_NOTA                        VL_NOTA
              ,NOTIFICACAO_FORNECEDOR.CD_USUARIO                     CD_USUARIO
              ,ITNOTIFICACAO_FORNECEDOR.CD_PRODUTO                   CD_PRODUTO
             ,PRODUTO.DS_PRODUTO                                    DS_PRODUTO
              ,ITLOT_ENT.CD_LOTE                                     CD_LOTE
              ,ITLOT_ENT.DT_VALIDADE                                 DT_VALIDADE
              ,UNI_PRO.DS_UNIDADE                                    DS_UNIDADE
              ,ITNOTIFICACAO_FORNECEDOR.QT_NOTIFICADA                QT_NOTIFICADA
              ,ITNOTIFICACAO_FORNECEDOR.VL_UNITARIO                  VL_UNITARIO
              ,ITNOTIFICACAO_FORNECEDOR.VL_TOTAL                     VL_TOTAL
	      ,'P'                                                   TP_ENT
FROM          DBAMV.NOTIFICACAO_FORNECEDOR
             ,DBAMV.ITNOTIFICACAO_FORNECEDOR
             ,DBAMV.FORNECEDOR
             ,DBAMV.ITLOT_ENT
             ,DBAMV.PRODUTO
             ,DBAMV.UNI_PRO
WHERE       NOTIFICACAO_FORNECEDOR.CD_NOTIFICA_FORNECEDOR = :CAMPO_NOTIFICACAO
AND         NOTIFICACAO_FORNECEDOR.CD_FORNECEDOR = FORNECEDOR.CD_FORNECEDOR
AND         NOTIFICACAO_FORNECEDOR.CD_NOTIFICA_FORNECEDOR = ITNOTIFICACAO_FORNECEDOR.CD_NOTIFICA_FORNECEDOR
AND         ITNOTIFICACAO_FORNECEDOR.CD_PRODUTO IS NOT NULL
AND         ITNOTIFICACAO_FORNECEDOR.CD_PRODUTO = PRODUTO.CD_PRODUTO
AND         ITNOTIFICACAO_FORNECEDOR.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO
AND         ITNOTIFICACAO_FORNECEDOR.CD_ITLOT_ENT = ITLOT_ENT.CD_ITLOT_ENT (+)
UNION
SELECT NOTIFICACAO_FORNECEDOR.CD_NOTIFICA_FORNECEDOR         CD_NOTIFICA_FORNECEDOR
              ,NOTIFICACAO_FORNECEDOR.DT_NOTIFICA_FORNECEDOR         DT_NOTIFICA_FORNECEDOR
              ,NOTIFICACAO_FORNECEDOR.HR_NOTIFICA_FORNECEDOR         HR_NOTIFICA_FORNECEDOR
              ,NOTIFICACAO_FORNECEDOR.DT_ENTRADA                     DT_ENTRADA
              ,NOTIFICACAO_FORNECEDOR.NR_DOCUMENTO                   NR_DOCUMENTO
              ,NOTIFICACAO_FORNECEDOR.NR_SERIE                       NR_SERIE
              ,NOTIFICACAO_FORNECEDOR.DS_MOTIVO_NOTIFICA             DS_MOTIVO_NOTIFICA
              ,NOTIFICACAO_FORNECEDOR.CD_FORNECEDOR                  CD_FORNECEDOR
	      ,FORNECEDOR.TP_FORNECEDOR                              TP_FORNECEDOR
              ,FORNECEDOR.NM_FORNECEDOR                              NM_FORNECEDOR
              ,FORNECEDOR.NR_CGC_CPF                                 NR_CGC_CPF
              ,nvl(NOTIFICACAO_FORNECEDOR.DS_JUSTIFICATIVA,'')       DS_JUSTIFICATIVA
              ,nvl(NOTIFICACAO_FORNECEDOR.DS_EMAIL_FORNECEDOR,'')    DS_EMAIL_FORNECEDOR
              ,nvl(NOTIFICACAO_FORNECEDOR.DS_EMAIL_ADICIONAL_FORNECEDOR, '') DS_EMAIL_ADICIONAL_FORNECEDOR
              ,NOTIFICACAO_FORNECEDOR.CD_ENT_SERV                     CD_ENT_PRO
              ,NOTIFICACAO_FORNECEDOR.VL_NOTA                        VL_NOTA
              ,NOTIFICACAO_FORNECEDOR.CD_USUARIO                     CD_USUARIO
              ,ITNOTIFICACAO_FORNECEDOR.CD_SERVICO                   CD_PRODUTO
              ,MANU_SERV.NM_SERVICO                                  DS_PRODUTO
              ,NULL                                     CD_LOTE
              ,NULL                                 DT_VALIDADE
              ,NULL                                    DS_UNIDADE
              ,ITNOTIFICACAO_FORNECEDOR.QT_NOTIFICADA                QT_NOTIFICADA
              ,ITNOTIFICACAO_FORNECEDOR.VL_UNITARIO                  VL_UNITARIO
              ,ITNOTIFICACAO_FORNECEDOR.VL_TOTAL                     VL_TOTAL
	      ,null                                                   TP_ENT
FROM          DBAMV.NOTIFICACAO_FORNECEDOR
             ,DBAMV.ITNOTIFICACAO_FORNECEDOR
             ,DBAMV.FORNECEDOR
             ,DBAMV.MANU_SERV
WHERE   NOTIFICACAO_FORNECEDOR.CD_NOTIFICA_FORNECEDOR = :CAMPO_NOTIFICACAO
AND         NOTIFICACAO_FORNECEDOR.CD_FORNECEDOR = FORNECEDOR.CD_FORNECEDOR
AND         NOTIFICACAO_FORNECEDOR.CD_NOTIFICA_FORNECEDOR = ITNOTIFICACAO_FORNECEDOR.CD_NOTIFICA_FORNECEDOR
AND       ITNOTIFICACAO_FORNECEDOR.CD_SERVICO IS NOT NULL
AND       ITNOTIFICACAO_FORNECEDOR.CD_SERVICO = MANU_SERV.CD_SERVICO
)
ORDER BY  DS_PRODUTO
