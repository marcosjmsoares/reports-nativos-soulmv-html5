SELECT * FROM (
SELECT sol_com.cd_sol_com     cd_sol_com
      ,sol_com.nm_solicitante  nm_solicitante
      ,sol_com.ds_observacao ds_observacao
      ,sol_com.dt_sol_com dt_sol_com
      ,sol_com.dt_maxima        dt_maxima
      ,sol_com.tp_situacao tp_situacao
      ,Setor.cd_setor cd_setor
      ,setor.nm_setor nm_setor
      ,mot_ped.cd_mot_ped  cd_mot_ped
      ,mot_ped.ds_mot_ped         ds_mot_ped
      ,ITSOL_COM.CD_PRODUTO      CD_PRODUTO
      ,Produto.ds_produto || DECODE(Produto.ds_complemento, null,null,' ') || Produto.ds_complemento || DECODE(Produto.nr_referencia, null,null,' ') || Produto.nr_referencia DS_PRODUTO
      ,UNI_PRO.DS_UNIDADE        DS_UNIDADE
      ,FORNECEDOR.NM_FORNECEDOR NM_FORNECEDOR
      ,FORNECEDOR.NM_FANTASIA  NM_FANTASIA
      ,(PRODUTO.VL_ULTIMA_ENTRADA*UNI_PRO.VL_FATOR)  VL_UNITARIO
FROM     DBAMV.ITSOL_COM           ITSOL_COM
        ,DBAMV.UNI_PRO             UNI_PRO
        ,DBAMV.PRODUTO             PRODUTO
        ,DBAMV.FORNECEDOR      FORNECEDOR
        ,dbamv.sol_com
        ,dbamv.setor
        ,dbamv.mot_ped
WHERE   itSOL_COM.CD_SOL_COM         = :CAMPO_SOLIC
AND Exists ( SELECT Empresa_Produto.Cd_Produto FROM Dbamv.Empresa_Produto WHERE Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA AND Empresa_Produto.Cd_Produto = Produto.Cd_Produto )
AND     ITSOL_COM.CD_UNI_PRO         = UNI_PRO.CD_UNI_PRO
AND     ITSOL_COM.CD_PRODUTO         = PRODUTO.CD_PRODUTO
AND     PRODUTO.CD_ULTIMO_FORNECEDOR = FORNECEDOR.CD_FORNECEDOR(+)
AND     sol_com.cd_setor             = setor.cd_setor
AND     sol_com.cd_mot_ped           = mot_ped.cd_mot_ped
AND     SOL_COM.CD_SOL_COM           = itSOL_COM.CD_SOL_COM
AND     ITSOL_COM.CD_MOT_CANCEL IS NULL
ORDER BY  DS_PRODUTO
)
ORDER BY DS_UNIDADE,NM_SOLICITANTE,DS_OBSERVACAO,DT_SOL_COM,DT_MAXIMA,TP_SITUACAO,CD_SETOR,NM_SETOR,DS_MOT_PED,CD_SOL_COM,CD_MOT_PED
