--R_PRODFORN
SELECT * FROM (
SELECT DISTINCT PROD.CD_PRODUTO,
               PROD.DS_PRODUTO,
               FORN.CD_FORNECEDOR,
               FORN.NM_FORNECEDOR,
               FORN.NM_FANTASIA,
               FORN.NM_CONTATO
FROM
               dbamv.PRODUTO PROD,
               dbamv.ATI_FOR    ATI,
               dbamv.FORNECEDOR FORN
WHERE PROD.CD_TIP_ATIV = ATI.CD_TIP_ATIV
      AND Exists ( SELECT Empresa_Produto.Cd_Produto FROM Dbamv.Empresa_Produto WHERE Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA AND Empresa_Produto.Cd_Produto = prod.Cd_Produto )
    AND ATI.CD_FORNECEDOR = FORN.CD_FORNECEDOR
:CAMPO_CF_SN_MESTRE
)
ORDER BY CD_PRODUTO,DS_PRODUTO, CD_FORNECEDOR
