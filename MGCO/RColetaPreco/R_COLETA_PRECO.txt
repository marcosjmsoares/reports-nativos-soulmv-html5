--R_COLETA_PRECO
SELECT * FROM (
Select   sol_com.cd_sol_com		cd_sol_com,
         sol_com.tp_sol_com		tp_sol_com,
          sol_com.ds_obs_cotacao	ds_obs_cotacao ,
         sol_com.dt_sol_com		dt_sol_com,
         sol_com.dt_maxima		dt_maxima,
         sol_com.cd_cotador		cd_cotador,
         sol_com.ds_observacao		ds_observacao,
         cotador.ds_cotador		ds_cotador,
         itsol_com.cd_produto		cd_produto,
         itcol_pro.qt_cotada                         qt_solic,
         itcol_pro.ds_dica_da_compra       ds_dica_da_compra,
         produto.ds_produto		ds_produto,
         ''                     ds_servico,
         uni_pro.ds_unidade		ds_unidade,
         fornecedor.cd_fornecedor	cd_fornecedor,
         fornecedor.nm_fornecedor	nm_fornecedor,
         fornecedor.ds_endereco		ds_end_for,
         fornecedor.ds_bairro		ds_bai_for,
         fornecedor.nr_cep		nr_cep_for,
         cidade.nm_cidade		nm_cid_for,
         cidade.cd_uf			cd_uf_for,
         fornecedor.nr_cgc_cpf		nr_cgc_cpf,
         fornecedor.nm_contato		nm_contato,
         coleta.cd_coleta		cd_coleta,
         Nvl(Decode(multi_empresas.sn_utiliza_moeda,'N', coleta.vl_frete, 'S', coleta_moeda.vl_frete_moeda), 0)   vl_frete,
         nvl(coleta.nr_dias_entrega,0)         nr_dias_entrega,
	     nvl(Decode(multi_empresas.sn_utiliza_moeda,'N', coleta.vl_desconto, 'S', coleta_moeda.vl_desconto_moeda), 0) vl_total_desconto,
         dt_validade,
         nvl(coleta.cd_condicao_pagamento, 0)   cd_cond_pgto,
         itcol_pro.ds_marca                           ds_marca,
         Nvl(Decode(multi_empresas.sn_utiliza_moeda,'N', itcol_pro.vl_unitario, 'S', itcol_pro.vl_unitario_moeda), 0) vl_unitario,
         Nvl(Decode(multi_empresas.sn_utiliza_moeda,'N', itcol_pro.vl_total, 'S', itcol_pro.vl_total_moeda), 0)  vl_total,
         Nvl(Decode(multi_empresas.sn_utiliza_moeda,'N', itcol_pro.vl_desconto, 'S', itcol_pro.vl_desconto_moeda), 0) vl_desconto,
         Nvl(Decode(multi_empresas.sn_utiliza_moeda,'N', itcol_pro.vl_imposto, 'S', itcol_pro.vl_imposto_moeda), 0) vl_imposto
        ,coleta.dt_coleta DT_LANCAMENTO
        ,nvl(multi_empresas.sn_utiliza_moeda,'N') sn_utiliza_moeda
From   dbamv.sol_com              sol_com,
       dbamv.coleta               coleta,
       dbamv.itcol_pro            itcol_pro,
       dbamv.itsol_com            itsol_com,
       dbamv.cotador              cotador,
       dbamv.produto              produto,
       dbamv.uni_pro              uni_pro,
       dbamv.fornecedor           fornecedor,
       dbamv.cidade               cidade,
       dbamv.multi_empresas       multi_empresas,
       dbamv.estoque estoque,
       dbamv.coleta_moeda coleta_moeda
Where  sol_com.cd_sol_com       = itsol_com.cd_sol_com
and    coleta.cd_coleta         = itcol_pro.cd_coleta
and    itcol_pro.cd_produto     = itsol_com.cd_produto
and    coleta.cd_fornecedor     = fornecedor.cd_fornecedor
and    coleta.cd_sol_com        = sol_com.cd_sol_com
and    sol_com.cd_cotador       = cotador.cd_cotador(+)
and    itsol_com.cd_produto     = produto.cd_produto
and    itsol_com.cd_uni_pro     = uni_pro.cd_uni_pro
and    fornecedor.cd_cidade     = cidade.cd_cidade
and    coleta.cd_coleta         = coleta_moeda.cd_coleta(+)
AND COLETA.CD_COLETA = :CAMPO_COL
And   estoque.Cd_Estoque = Sol_Com.Cd_Estoque
And   estoque.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND  multi_empresas.Cd_Multi_Empresa = estoque.Cd_Multi_Empresa
Union All
Select  sol_com.cd_sol_com             cd_sol_com,
         sol_com.tp_sol_com		tp_sol_com,
        sol_com.ds_obs_cotacao              ds_obs_cotacao ,
        sol_com.dt_sol_com             dt_sol_com,
        sol_com.dt_maxima              dt_maxima,
        sol_com.cd_cotador             cd_cotador,
        sol_com.ds_observacao	  ds_observacao,
        cotador.ds_cotador           ds_cotador,
        itcol_ser.cd_ser_sol           cd_produto,
        0                            qt_solic,
        ''                            ds_dica_da_compra,
        manu_serv.nm_servico           ds_produto,
        itcol_ser.ds_servico           ds_servico,
        'Serviço'                      ds_unidade,
        fornecedor.cd_fornecedor       cd_fornecedor,
        fornecedor.nm_fornecedor       nm_fornecedor,
        fornecedor.ds_endereco         ds_end_for,
        fornecedor.ds_bairro           ds_bai_for,
        fornecedor.nr_cep              nr_cep_for,
        cidade.nm_cidade               nm_cid_for,
        cidade.cd_uf                   cd_uf_for,
        fornecedor.nr_cgc_cpf          nr_cgc_cpf,
        fornecedor.nm_contato          nm_contato,
        coleta.cd_coleta			      cd_coleta,
        0			      vl_frete,
        nvl(coleta.nr_dias_entrega,0)       nr_dias_entrega,
        Nvl(Decode(multi_empresas.sn_utiliza_moeda,'N', coleta.vl_desconto, 'S', coleta_moeda.vl_desconto_moeda), 0) vl_total_desconto,
        dt_validade,
        nvl(coleta.cd_condicao_pagamento, 0)   cd_cond_pgto,
        ''                                            ds_marca,
        0                                           vl_unitario,
       Nvl(Decode(multi_empresas.sn_utiliza_moeda,'N', itcol_ser.vl_total, 'S', itcol_ser.vl_total_moeda), 0)      vl_total,
        0                                           vl_desconto,
        0                                           vl_imposto
        ,coleta.dt_coleta DT_LANCAMENTO
        ,nvl(multi_empresas.sn_utiliza_moeda,'N') sn_utiliza_moeda
From dbamv.sol_com                 sol_com,
     dbamv.cotador                 cotador,
     dbamv.coleta               coleta,
     dbamv.itcol_ser               itcol_ser,
     dbamv.manu_serv ,
     dbamv.fornecedor              fornecedor,
     dbamv.cidade                  cidade,
     dbamv.multi_empresas       multi_empresas,
     dbamv.estoque estoque,
     dbamv.coleta_moeda coleta_moeda
where  sol_com.cd_cotador        = cotador.cd_cotador
and    coleta.cd_sol_com        = sol_com.cd_sol_com
and    coleta.cd_coleta        = itcol_ser.cd_coleta
and    manu_serv.cd_servico        = itcol_ser.cd_ser_sol
and    coleta.cd_fornecedor     = fornecedor.cd_fornecedor
and    fornecedor.cd_cidade      = cidade.cd_cidade
and    coleta.cd_coleta         = coleta_moeda.cd_coleta(+)
AND COLETA.CD_COLETA = :CAMPO_COL
And Estoque.Cd_Estoque = Sol_Com.Cd_Estoque
And  Estoque.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND  multi_empresas.Cd_Multi_Empresa = estoque.Cd_Multi_Empresa
:CAMPO_CF_FORNECEDOR
)
ORDER BY cd_produto
