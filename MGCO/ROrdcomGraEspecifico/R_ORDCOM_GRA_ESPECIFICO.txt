--R_ORDCOM_GRA_ESPECIFICO
SELECT
	ORD_COM.CD_ORD_COM,
	ORD_COM.CD_SOL_COM,
	ORD_COM.DT_ORD_COM,
	FORNECEDOR.CD_FORNECEDOR,
	FORNECEDOR.NM_FORNECEDOR,
	FORNECEDOR.NM_FANTASIA,
	FORNECEDOR.NR_CGC_CPF,
	FORNECEDOR.NR_INSC_EST,
	FORNECEDOR.DS_ENDERECO,
	FORNECEDOR.DS_BAIRRO,
	FORNECEDOR.NR_CEP,
	CIDADE.NM_CIDADE,
	CIDADE.CD_UF,
	FORNECEDOR.NR_CONTA,
	ESTOQUE.CD_ESTOQUE || ' - ' || ESTOQUE.DS_ESTOQUE || ' / ' || SOL_COM.NM_SOLICITANTE REQUISITANTE,
	ORD_COM.VL_PERC_DESCONTO,
	ORD_COM.VL_DESCONTO,
	ORD_COM.DT_PREV_ENTREGA_INTERVALO,
	ORD_COM.DT_PREV_ENTREGA,
	ORD_COM.TP_FRETE,
	ORD_COM.VL_FRETE,
	ORD_COM.DS_ORDEM_DE_COMPRA ,
	ORD_COM.VL_TOTAL VL_TOTAL_ORD_COM,
	CONDICOES_PAGAMENTO.DS_CONDICAO_PAGAMENTO,
	ORD_COM.CD_ID_USUARIO_AUTORIZOU CD_ID_USUARIO_AUTORIZOU,
	COTADOR.DS_COTADOR DS_COTADOR,
	COTADOR.DS_ENDERECO_EMAIL E_MAIL,
	HOSPITAL.LOGOTIPO,
	ORD_COM.VL_TOTAL
FROM
	DBAMV.ORD_COM ,
	DBAMV.ESTOQUE ,
	DBAMV.FORNECEDOR ,
	DBAMV.SOL_COM ,
	DBAMV.CIDADE ,
	DBAMV.CONDICOES_PAGAMENTO ,
	DBAMV.COTADOR ,
	DBAMV.HOSPITAL
WHERE
	ORD_COM.CD_ORD_COM = :CAMPO_CD_ORD_COM
	AND ORD_COM.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
	AND ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
	AND ORD_COM.CD_FORNECEDOR = FORNECEDOR.CD_FORNECEDOR
	AND FORNECEDOR.TP_CLIENTE_FORN IN ( 'F',
	'A' )
	AND FORNECEDOR.CD_CIDADE = CIDADE.CD_CIDADE
	AND ORD_COM.CD_SOL_COM = SOL_COM.CD_SOL_COM(+)
	AND ORD_COM.CD_CONDICAO_PAGAMENTO = CONDICOES_PAGAMENTO.CD_CONDICAO_PAGAMENTO
	AND COTADOR.CD_COTADOR(+) = SOL_COM.CD_COTADOR
	AND ESTOQUE.CD_MULTI_EMPRESA = HOSPITAL.CD_MULTI_EMPRESA
ORDER BY
	ORD_COM.CD_ORD_COM

--R_ORDCOM_GRA_ESPECIFICO_R_IT_ORD_PRO1_Q_IT_ORD_PRO
SELECT ITORD_PRO.CD_ORD_COM        CD_ORD_COM_PROD
      ,PRODUTO.CD_PRODUTO
      ,PRODUTO.DS_PRODUTO
      ,UNI_PRO.DS_UNIDADE
      ,ITORD_PRO.QT_COMPRADA
      ,ITORD_PRO.VL_UNITARIO
      ,ITORD_PRO.VL_TOTAL + Nvl( ItOrd_Pro.Vl_Imposto, 0 ) - Nvl( ItOrd_Pro.Vl_Desconto, 0 )     VL_TOTAL_PROD
      ,ITORD_PRO.VL_DESCONTO
      ,ITORD_PRO.VL_IMPOSTO
      ,ITORD_PRO.VL_PERC_DESCONTO
      ,ITORD_PRO.VL_PERC_IMPOSTO
      ,ITORD_PRO.DS_MARCA
      ,ITORD_PRO.CD_MOT_CANCEL
FROM     DBAMV.ITORD_PRO
        ,DBAMV.PRODUTO
        ,DBAMV.UNI_PRO
WHERE   ITORD_PRO.CD_ORD_COM        =  :CAMPO_CD_ORD_COM
AND     ITORD_PRO.CD_PRODUTO        = PRODUTO.CD_PRODUTO
AND     ITORD_PRO.CD_UNI_PRO        = UNI_PRO.CD_UNI_PRO
AND     ITORD_PRO.CD_MOT_CANCEL IS NULL
AND     'P' = (SELECT Nvl(tp_ord_com,'P') FROM dbamv.ord_com WHERE cd_ord_com = :CAMPO_CD_ORD_COM ) 
UNION 
SELECT ITORD_SER.CD_ORD_COM         CD_ORD_COM_PROD
       ,MANU_SERV.CD_SERVICO        cd_produto
       ,MANU_SERV.nm_servico        ds_produto
       ,'SERV.'                     DS_UNIDADE
       ,1                           QT_COMPRADA
       ,ITORD_SER.VL_TOTAL          VL_UNITARIO
       ,ITORD_SER.VL_TOTAL          VL_TOTAL_PROD                
       ,0                        VL_DESCONTO
       ,0                        VL_IMPOSTO
       ,0                        VL_PERC_DESCONTO
       ,0                        VL_PERC_IMPOSTO
       ,''                        DS_MARCA
       ,0                           CD_MOT_CANCEL
FROM DBAMV.ITORD_SER, dbamv.MANU_SERV 
WHERE ITORD_SER.CD_ORD_COM= :CAMPO_CD_ORD_COM
  AND ITORD_SER.cd_ser_sol = MANU_SERV.cd_servico(+)
  AND 'S' = (SELECT Nvl(tp_ord_com,'P') FROM dbamv.ord_com WHERE cd_ord_com = :CAMPO_CD_ORD_COM )  
order by CD_ORD_COM_PROD
