--R_ORD_COM_PROD
select
  autorizador_ordem_compra.cd_ord_com,
  autorizador_ordem_compra.cd_usuario,
  autorizador_ordem_compra.dt_autorizacao,
  autorizador_ordem_compra.cd_nivel_autorizador,
  nivel_autorizador.ds_nivel_autorizador
from
  dbamv.autorizador_ordem_compra,
  dbamv.nivel_autorizador
where autorizador_ordem_compra.cd_nivel_autorizador = nivel_autorizador.cd_nivel_autorizador
and autorizador_ordem_compra.cd_ord_com = :CAMPO_CD_ORD_COM
order by autorizador_ordem_compra.dt_autorizacao
