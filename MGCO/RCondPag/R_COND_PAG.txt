--R_COND_PAG
SELECT * FROM (
select condicoes_pagamento.cd_condicao_pagamento codigo
      ,ds_condicao_pagamento descricao
      ,nr_ordem_preferencia ordem
      ,PARCELA_COND_PAGAMENTO.NR_PARCELA
      ,PARCELA_COND_PAGAMENTO.QT_DIAS_PARCELA
      ,PARCELA_COND_PAGAMENTO.VL_PERCENTUAL_PARCELA
  from dbamv.condicoes_pagamento
      ,dbamv.PARCELA_COND_PAGAMENTO
 WHERE condicoes_pagamento.cd_condicao_pagamento =  PARCELA_COND_PAGAMENTO.CD_CONDICAO_PAGAMENTO
)
ORDER BY
