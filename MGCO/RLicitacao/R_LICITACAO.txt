--R_LICITACAO
SELECT * FROM (
select licitacao.cd_licitacao,
          licitacao.cd_sol_com cd_sol_com,
         sol_com.nm_solicitante,
         licitacao.cd_tip_emp,
         tip_emp.ds_tip_emp,
         licitacao.cd_tip_ativ,
         tip_ativ.ds_tip_ativ,
         licitacao.ds_detalhes,
         licitacao.dt_licitacao,
         to_char(licitacao.hr_licitacao, 'hh24') hr_licitacao,
        decode( licitacao.tp_licitacao , 'M', 'Menor Preço', 'Quantidade' ) tp_licitacao,
        decode( licitacao.tp_modalidade, 'C', 'Convite'    ,
	                                    'T', 'Tomada de Preço',
			    'D', 'Dispensa de Licitação',
			    'P', 'Concorrencia Publica',
		                    'i', 'Inegibilidade' ) tp_modalidade,
         licitacao.nr_processo,
         licitacao.dt_abertura,
         licitacao.hr_abertura,
         licitacao.vl_Estimado_compra,
         licitacao.nr_licitacao,
         itsol_com.cd_produto,
         produto.cd_sican,
         produto.ds_produto,
         uni_pro.ds_unidade,
         itsol_com.qt_solic,
         itsol_com.ds_dica_da_compra ds_dica_da_compra
 from dbamv.licitacao,
       dbamv.sol_com,
       dbamv.itsol_com,
       dbamv.produto,
       dbamv.uni_pro,
       dbamv.tip_ativ,
       dbamv.tip_emp,
       dbamv.estoque
where licitacao.cd_sol_com = :CAMPO_CD_SOL_COM
and licitacao.cd_sol_com = sol_com.cd_sol_com
and itsol_com.cd_sol_com = sol_com.cd_sol_com
and itsol_com.cd_produto = produto.cd_produto
and itsol_com.cd_uni_pro = uni_pro.cd_uni_pro
and licitacao.cd_tip_ativ = tip_ativ.cd_tip_ativ
and licitacao.cd_tip_emp = tip_emp.cd_tip_emp
and Estoque.Cd_Estoque = Sol_Com.Cd_Estoque
and  Estoque.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
order by licitacao.cd_sol_com, produto.ds_produto
)
order by cd_sol_com, ds_produto
