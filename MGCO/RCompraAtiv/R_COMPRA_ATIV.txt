--R_COMPRA_ATIV
select RETORNO.DATA                             DATA
                  ,RETORNO.COMPETENCIA          COMPETENCIA
                  ,RETORNO.CD_TIP_ATIV          CD_TIP_ATIV
                  ,RETORNO.DS_TIP_ATIV          DS_TIP_ATIV
                  ,SUM(RETORNO.QTD_CD_TIP_ATIV) QTD_CD_TIP_ATIV
                  ,SUM(VL_TOTAL)                VL_TOTAL
from(
SELECT  decode('', 'C', DATA, 'Período de ' || :CAMPO_COMP_INI || ' até ' || :CAMPO_COMP_FIM) DATA
              ,decode('', 'C', COMPETENCIA, 'Período de ' || :CAMPO_COMP_INI || ' até ' || :CAMPO_COMP_FIM) COMPETENCIA
              ,CD_TIP_ATIV
              ,DS_TIP_ATIV
              ,NVL(SUM(QTD_CD_TIP_ATIV),0) QTD_CD_TIP_ATIV
              ,NVL(SUM(VL_TOTAL),0) VL_TOTAL
FROM     (SELECT   TO_CHAR(ORD_COM.DT_ORD_COM,'MM/YYYY')  DATA
                            ,TO_CHAR(ORD_COM.DT_ORD_COM,'YYYYMM')        COMPETENCIA
                            ,DECODE(TIP_ATIV.CD_TIP_ATIV, '', 0, TIP_ATIV.CD_TIP_ATIV)  CD_TIP_ATIV
                            ,DECODE(TIP_ATIV.DS_TIP_ATIV, '', 'NÃO INFORMADA',TIP_ATIV.DS_TIP_ATIV)  DS_TIP_ATIV
                            ,NVL(COUNT (NVL( PRODUTO.CD_PRODUTO, 0) ), 0)  QTD_CD_TIP_ATIV
                           ,SUM(NVL( ITORD_PRO.VL_TOTAL,0) + NVL(ITORD_PRO.VL_IMPOSTO,0)-NVL(ITORD_PRO.VL_DESCONTO,0))  VL_TOTAL
             FROM     DBAMV.ORD_COM
                           ,DBAMV.ITORD_PRO
                           ,DBAMV.PRODUTO
                           ,DBAMV.TIP_ATIV
                           ,DBAMV.estoque
              WHERE    ORD_COM.CD_ORD_COM = ITORD_PRO.CD_ORD_COM
              AND      ITORD_PRO.CD_PRODUTO = PRODUTO.CD_PRODUTO
              AND      PRODUTO.CD_TIP_ATIV = TIP_ATIV.CD_TIP_ATIV(+)
              AND      ORD_COM.TP_SITUACAO NOT IN ('N','C')
              AND      NVL(ORD_COM.TP_CONTRATO, 'O') = 'O'
              AND      ORD_COM.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
              AND      ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
              AND      TO_DATE(TO_CHAR(ORD_COM.DT_ORD_COM,'DD/MM/YYYY'), 'DD/MM/YYYY')
                            BETWEEN TO_DATE(:CAMPO_COMP_INI, 'DD/MM/YYYY')
                            AND  TO_DATE(:CAMPO_COMP_FIM, 'DD/MM/YYYY')
              GROUP BY TO_CHAR(ORD_COM.DT_ORD_COM,'MM/YYYY')
                                 ,TO_CHAR(ORD_COM.DT_ORD_COM,'YYYYMM')
                                 ,DECODE(TIP_ATIV.CD_TIP_ATIV, '', 0, TIP_ATIV.CD_TIP_ATIV)
                                 ,DECODE(TIP_ATIV.DS_TIP_ATIV, '', 'NÃO INFORMADA',TIP_ATIV.DS_TIP_ATIV)
              UNION ALL
              SELECT   TO_CHAR(ENT_PRO.DT_ENTRADA,'MM/YYYY')  DATA
                              ,TO_CHAR(ENT_PRO.DT_ENTRADA,'YYYYMM')   COMPETENCIA
                              ,DECODE(TIP_ATIV.CD_TIP_ATIV, '', 0, TIP_ATIV.CD_TIP_ATIV)  CD_TIP_ATIV
                              ,DECODE(TIP_ATIV.DS_TIP_ATIV, '', 'NÃO INFORMADA',TIP_ATIV.DS_TIP_ATIV)  DS_TIP_ATIV
                              ,NVL(COUNT (NVL( PRODUTO.CD_PRODUTO, 0) ), 0)  QTD_CD_TIP_ATIV
                              ,SUM(NVL( ITENT_PRO.VL_TOTAL,0) + NVL(ITENT_PRO.VL_IMPOSTO,0)-NVL(ITENT_PRO.VL_DESCONTO,0))  VL_TOTAL
              FROM     DBAMV.ORD_COM
                            ,DBAMV.ENT_PRO
                            ,DBAMV.ITENT_PRO
                            ,DBAMV.PRODUTO
                            ,DBAMV.TIP_ATIV
                           ,DBAMV.estoque
              WHERE    ENT_PRO.CD_ENT_PRO   = ITENT_PRO.CD_ENT_PRO
              AND      ITENT_PRO.CD_PRODUTO = PRODUTO.CD_PRODUTO
              AND      PRODUTO.CD_TIP_ATIV  = TIP_ATIV.CD_TIP_ATIV(+)
              AND      ENT_PRO.CD_ORD_COM   = ORD_COM.CD_ORD_COM
              AND      ORD_COM.TP_SITUACAO NOT IN ('N','C')
              AND      NVL(ORD_COM.TP_CONTRATO, 'O') <> 'O'
              AND      ORD_COM.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
              AND      ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
              AND      TO_DATE(TO_CHAR(ENT_PRO.DT_ENTRADA,'DD/MM/YYYY'), 'DD/MM/YYYY')
                            BETWEEN TO_DATE(:CAMPO_COMP_INI, 'DD/MM/YYYY')
                            AND     TO_DATE(:CAMPO_COMP_FIM, 'DD/MM/YYYY')
              GROUP BY TO_CHAR(ENT_PRO.DT_ENTRADA,'MM/YYYY')
                                ,TO_CHAR(ENT_PRO.DT_ENTRADA,'YYYYMM')
                                 ,DECODE(TIP_ATIV.CD_TIP_ATIV, '', 0, TIP_ATIV.CD_TIP_ATIV)
                                 ,DECODE(TIP_ATIV.DS_TIP_ATIV, '', 'NÃO INFORMADA',TIP_ATIV.DS_TIP_ATIV))
GROUP BY           DATA
                  ,COMPETENCIA
                  ,CD_TIP_ATIV
                  ,DS_TIP_ATIV
ORDER BY CD_TIP_ATIV,DS_TIP_ATIV,COMPETENCIA,DATA, VL_TOTAL
) RETORNO
GROUP BY  RETORNO.DATA
                  ,RETORNO.COMPETENCIA
                  ,RETORNO.CD_TIP_ATIV
                  ,RETORNO.DS_TIP_ATIV
ORDER BY CD_TIP_ATIV,DS_TIP_ATIV,COMPETENCIA,DATA
