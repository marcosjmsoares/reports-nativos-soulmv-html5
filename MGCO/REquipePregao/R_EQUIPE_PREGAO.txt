--R_EQUIPE_PREGAO
SELECT ALL DECODE(EQUIPE_PREGAO.TP_DIA_SEMANA,
'1','Domingo',
                       '2','Segunda-feira',
                       '3','Terca-feira',
                       '4','Quarta-feira',
                       '5','Quinta-feira',
                       '6','Sexta-feira',
                       '7','Sabado') AS TP_DIA_SEMANA,
DECODE(EQUIPE_PREGAO.TP_TURNO,
                       '1','1 Turno','2','2 Turno') AS TP_TURNO,
EQUIPE_PREGAO.NM_GUERRA, EQUIPE_PREGAO.CD_USUARIO,
USUARIOS.NM_USUARIO, DECODE(EQUIPE_PREGAO.TP_SITUACAO,
                       'A','Ativo',
                       'I','Inativo') AS TP_SITUACAO
FROM dbasgu.USUARIOS, dbamv.EQUIPE_PREGAO
WHERE (EQUIPE_PREGAO.CD_USUARIO = USUARIOS.CD_USUARIO)
