--R_PACIENTES_HEMODIALISE_P321
SELECT * FROM (
Select fat_sia.cd_fat_sia       CD_FAT_SIA
     , fat_sia.ds_fat_sia       DS_FAT_SIA
     , remessa.cd_remessa       CD_REMESSA
     , remessa.ds_remessa       DS_REMESSA
     , apac.cd_apac             CD_APAC
     , apac.cd_motivo_cobranca  CD_MOTIVO
     , apac.dt_ocorrencia       DT_OCORRENCIA
     , apac.qt_creatinina       QT_CREATININA
     , apac.qt_ureia_pre        QT_UREIA_PRE
     , apac.qt_ureia_pos        QT_UREIA_POS
     , paciente.cd_paciente     CD_PACIENTE
     , paciente.nr_cpf          NR_CPF
     , paciente.nr_cns          NR_CNS
     ,proc_sus.cd_procedimento  CD_PROCEDIMENTO
     , apac.cd_cid_principal    CD_CID_PRINCIPAL
     , apac.cd_cid_secundario   CD_CID_SECUNDARIO
     , to_char( apac.dt_inicio_tratamento,'MM/YYYY') DT_INICIO_TRATAMENTO
     ,proc_sus.ds_procedimento
     , eve_siasus.qt_lancada    QT_LANCADA
     , ori_ate.cd_ori_ate       CD_ORI_ATE
     , eve_siasus.sn_apac_principal SN_PRINCIPAL
  From dbamv.fat_sia      FAT_SIA
     , dbamv.remessa_apac REMESSA
     , dbamv.apac         APAC
     , dbamv.paciente     PACIENTE
     , dbamv.eve_siasus   EVE_SIASUS
     , dbamv.ori_ate      ORI_ATE
     ,dbamv.procedimento_sus  PROC_SUS
 Where fat_sia.cd_fat_sia = apac.cd_fat_sia
   And apac.cd_apac is not null
   And apac.cd_tip_ate    = 27
   And apac.cd_remessa    = remessa.cd_remessa
   And apac.cd_ori_ate    = ori_ate.cd_ori_ate
   And apac.cd_paciente   = paciente.cd_paciente
   And apac.cd_apac       = eve_siasus.cd_apac
   And proc_sus .cd_procedimento = eve_siasus.cd_procedimento
 Order by CD_REMESSA
     , CD_APAC
)
ORDER BY CD_FAT_SIA,DS_FAT_SIA,CD_ORI_ATE,
CD_REMESSA,DS_REMESSA,
CD_APAC,CD_PACIENTE,NR_CPF,NR_CNS
