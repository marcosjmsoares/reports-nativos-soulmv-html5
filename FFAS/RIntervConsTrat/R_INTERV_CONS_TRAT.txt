--R_INTERV_CONS_TRAT
SELECT * FROM (
select prestador.nm_prestador,
laudo_apac.cd_tip_ate,
       paciente.cd_paciente,
       paciente.nm_paciente,
       laudo_apac.dt_consulta,
       decode(laudo_apac.cd_tip_ate,29,laudo_apac.dt_inicio_trat_aut,28,laudo_apac.DT_INICIO_IRRADIADA) dt_inicio,
       (decode(laudo_apac.cd_tip_ate,29,laudo_apac.dt_inicio_trat_aut,28,laudo_apac.DT_INICIO_IRRADIADA) - laudo_apac.DT_CONSULTA) tempo 
  from dbamv.prestador,
       dbamv.paciente,
       dbamv.laudo_apac
 where laudo_apac.dt_consulta is not null
   and prestador.cd_prestador = laudo_apac.cd_prestador
   and paciente.cd_paciente = laudo_apac.cd_paciente
   and laudo_apac.cd_tip_ate = :CAMPO_TP_ATENDIMENTO 
   and laudo_apac.tp_laudo = 'N'
UNION ALL
select prestador.nm_prestador,
       laudo_sia_apac.cd_tip_ate,
       paciente.cd_paciente,
       paciente.nm_paciente,
       laudo_sia_apac.dt_consulta,
       laudo_sia_apac.dt_inicio_trat_solicitado dt_inicio,
       (laudo_sia_apac.dt_inicio_trat_solicitado - laudo_sia_apac.DT_CONSULTA) tempo
  from dbamv.prestador,
       dbamv.paciente,
       dbamv.laudo_sia_apac
 where laudo_sia_apac.dt_consulta is not null
   and prestador.cd_prestador = laudo_sia_apac.cd_prestador
   and paciente.cd_paciente = laudo_sia_apac.cd_paciente
   and laudo_sia_apac.cd_tip_ate = :CAMPO_TP_ATENDIMENTO
   and laudo_sia_apac.tp_laudo = 'N'
   :CAMPO_CF_PRESTADOR_SIA
)
ORDER BY NM_PRESTADOR
