--R_FINANC_COMPLX_PROC_R_HOSPITAL4_Q_DETALHAMENTO_NIVEL3
SELECT * FROM (
Select TF.Ds_Financiamento                   Financiamento_Nv3
     , CP.Ds_Complexidade_Procedimento       Complexidade_Nv3
     , Eve_Siasus.Cd_Procedimento            Procedimento_Nv3
     , Sum(Eve_Siasus.Qt_Lancada)            Qt_Lancada_Nv3
     , Sum(Eve_Siasus.Vl_Total_Ambulatorial) Valor_Total_Nv3
  From Dbamv.Procedimento_Detalhe_Vigencia   PD
     , Dbamv.Procedimento_SUS_Complexidade   PSC
     , Dbamv.Tipo_Financiamento              TF
     , Dbamv.Eve_Siasus                      Eve_Siasus
     , Dbamv.Complexidade_Procedimento       CP
     , Dbamv.Remessa_Apac                    RA
     , Dbamv.Apac                            AP
 Where PD.Cd_Procedimento                = PSC.Cd_Procedimento
   And PSC.Tp_Complexidade_Procedimento  = CP.Tp_Complexidade_Procedimento
   And PD.Tp_Financiamento               = TF.Tp_Financiamento
   And AP.Cd_Apac                        = Eve_Siasus.Cd_Apac
   And RA.Cd_Remessa                     = AP.Cd_Remessa
   And PD.Cd_Procedimento                = Eve_Siasus.Cd_Procedimento
   And TRUNC(RA.dt_competencia_apresentacao, 'mm') BETWEEN PD.Dt_Validade_Inicial AND NVL(PD.Dt_Validade_Final, TRUNC(RA.dt_competencia_apresentacao, 'mm'))
   AND EVE_SIASUS.SN_PACOTE		 = 'N'
 Group by TF.Ds_Financiamento
     , CP.Ds_Complexidade_Procedimento
     , Eve_Siasus.Cd_Procedimento
UNION ALL
Select TF.Ds_Financiamento                   Financiamento_Nv3
     , CP.Ds_Complexidade_Procedimento       Complexidade_Nv3
     , Eve_Siasus.Cd_Procedimento            Procedimento_Nv3
     , Sum(Eve_Siasus.Qt_Lancada)            Qt_Lancada_Nv3
     , Sum(Eve_Siasus.Vl_Total_Ambulatorial) Valor_Total_Nv3
  From Dbamv.Procedimento_Detalhe_Vigencia   PD
     , Dbamv.Procedimento_SUS_Complexidade   PSC
     , Dbamv.Tipo_Financiamento              TF
     , Dbamv.Eve_Siasus                      Eve_Siasus
     , Dbamv.Complexidade_Procedimento       CP
     , Dbamv.Remessa_BPA                     RA
 Where PD.Cd_Procedimento                = PSC.Cd_Procedimento
   And PSC.Tp_Complexidade_Procedimento  = CP.Tp_Complexidade_Procedimento
   And PD.Tp_Financiamento               = TF.Tp_Financiamento
   And RA.Cd_Remessa                     = Eve_Siasus.Cd_Remessa
   And PD.Cd_Procedimento                = Eve_Siasus.Cd_Procedimento
   And TRUNC(RA.dt_competencia_apresentacao, 'mm') BETWEEN PD.Dt_Validade_Inicial AND NVL(PD.Dt_Validade_Final, TRUNC(RA.dt_competencia_apresentacao, 'mm'))
   AND EVE_SIASUS.SN_PACOTE		 = 'N'
   AND EVE_SIASUS.CD_APAC IS NULL 
 Group by TF.Ds_Financiamento
     , CP.Ds_Complexidade_Procedimento
     , Eve_Siasus.Cd_Procedimento
UNION ALL
Select TF.Ds_Financiamento                   Financiamento_Nv3
     , CP.Ds_Complexidade_Procedimento       Complexidade_Nv3
     , Eve_Siasus.Cd_Procedimento            Procedimento_Nv3
     , Sum(Eve_Siasus.Qt_Lancada)            Qt_Lancada_Nv3
     , Sum(Eve_Siasus.Vl_Total_Ambulatorial) Valor_Total_Nv3
  From Dbamv.Procedimento_Detalhe_Vigencia   PD
     , Dbamv.Procedimento_SUS_Complexidade   PSC
     , Dbamv.Tipo_Financiamento              TF
     , Dbamv.Eve_Siasus                      Eve_Siasus
     , Dbamv.Complexidade_Procedimento       CP
     , dbamv.fat_sia                         ft   
 Where PD.Cd_Procedimento                = PSC.Cd_Procedimento
   And PSC.Tp_Complexidade_Procedimento  = CP.Tp_Complexidade_Procedimento
   And PD.Tp_Financiamento               = TF.Tp_Financiamento
   And PD.Cd_Procedimento                = Eve_Siasus.Cd_Procedimento
   AND ft.cd_fat_sia                     = eve_siasus.cd_fat_sia
   AND ft.tipo_fatura  = 'SISC'  
   AND EVE_SIASUS.SN_PACOTE		 = 'N'
 Group by TF.Ds_Financiamento
     , CP.Ds_Complexidade_Procedimento
     , Eve_Siasus.Cd_Procedimento    
)
ORDER BY FINANCIAMENTO_NV3,
COMPLEXIDADE_NV3,
PROCEDIMENTO_NV3
