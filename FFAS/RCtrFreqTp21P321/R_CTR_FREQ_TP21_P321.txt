--R_CTR_FREQ_TP21_P321
SELECT CTR_FREQUENCIA.CD_FREQUENCIA
       ,PACIENTE.CD_PACIENTE
       , dbamv.PKG_SUS_REGRA_PACIENTE.F_NOME_PACIENTE(PACIENTE.CD_PACIENTE) NM_PACIENTE
       ,PACIENTE.NM_MAE
       ,TO_CHAR(PACIENTE.NR_CPF) NR_CPF
       ,PACIENTE.DS_ENDERECO
       ,PACIENTE.NR_CEP
       ,PACIENTE.NR_FONE
       ,PACIENTE.DT_NASCIMENTO
       ,PACIENTE.TP_SEXO
       ,PACIENTE.NM_BAIRRO
       ,PACIENTE.NR_CNS
       ,CIDADE.NM_CIDADE
       ,CIDADE.CD_UF
       ,PRESTADOR.NM_PRESTADOR
       ,CTR_FREQUENCIA.DT_REFERENCIA
       ,CTR_FREQUENCIA.CD_PROCEDIMENTO
       ,CTR_FREQUENCIA.NR_COMPARECIMENTO
  FROM DBAMV.CTR_FREQUENCIA
       ,DBAMV.PACIENTE
       ,DBAMV.CIDADE
       ,DBAMV.PRESTADOR
       ,DBAMV.PROCEDIMENTO_SUS
  WHERE CTR_FREQUENCIA.CD_PACIENTE     = PACIENTE.CD_PACIENTE
    AND CTR_FREQUENCIA.CD_FREQUENCIA   = :CAMPO_FREQ
    AND CTR_FREQUENCIA.CD_TIP_ATE      = :CAMPO_TIPO
    AND CTR_FREQUENCIA.CD_PRESTADOR    = PRESTADOR.CD_PRESTADOR
    AND CTR_FREQUENCIA.CD_PROCEDIMENTO = PROCEDIMENTO_SUS.CD_PROCEDIMENTO(+)
    AND PACIENTE.CD_CIDADE             = CIDADE.CD_CIDADE(+)
    AND ROWNUM = 1
