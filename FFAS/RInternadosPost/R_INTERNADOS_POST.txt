--R_INTERNADOS_POST
SELECT * FROM (
Select eve_siasus.cd_fat_sia,
       atendime.cd_atendimento,
       atendime.cd_paciente,
       paciente.nm_paciente,
       proc.ds_procedimento,            
       proc.cd_procedimento,            
       eve_siasus.dt_eve_siasus,
       atendime.dt_atendimento,
       atendime.dt_alta,
       eve_siasus.qt_lancada
  From dbamv.atendime,
       dbamv.paciente,
       dbamv.eve_siasus,
       dbamv.procedimento_sus proc,     
       dbamv.convenio convambu,
       dbamv.convenio convinte,
       dbamv.empresa_convenio
 Where Empresa_Convenio.Cd_Convenio         = ConvInte.Cd_Convenio
   And Empresa_Convenio.Cd_Multi_Empresa    = :CAMPO_CD_MULTI_EMPRESA
   and atendime.cd_multi_empresa            = dbamv.Pkg_Mv2000.Le_Empresa 
   and atendime.tp_atendimento in ( 'I', 'H' )
   and paciente.cd_paciente                 = atendime.cd_paciente
   and eve_siasus.cd_paciente               = atendime.cd_paciente
   and proc.cd_procedimento                 = eve_siasus.cd_procedimento    
   and eve_siasus.cd_convenio               = convambu.cd_convenio
   and atendime.cd_convenio                 = convinte.cd_convenio
   and convinte.tp_convenio                 = 'H'
   and convambu.tp_convenio                 = 'A'
   and eve_siasus.cd_fat_sia = :CAMPO_CD_FATURA
   and eve_siasus.cd_atendimento is not null
   and atendime.dt_atendimento between eve_siasus.dt_eve_siasus
                                   and eve_siasus.dt_eve_siasus + 1)
ORDER BY DT_ALTA,DT_ATENDIMENTO,NM_PACIENTE,CD_PACIENTE,CD_ATENDIMENTO
