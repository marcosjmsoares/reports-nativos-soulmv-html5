      ,CD_CBO_PRESTADOR
      ,NR_PAGINA
      ,NR_LINHAS_DA_PAGINA
      ,CD_PROCEDIMENTO
      ,NR_IDADE

       AND CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
