SELECT DT_COMPT
      ,CD_CBO_PRESTADOR
      ,NR_PAGINA
      ,NR_LINHAS_DA_PAGINA
      ,CD_PROCEDIMENTO
      ,NR_IDADE
      ,QT_LANCADA
      ,'SEM ERROS' SEMERROS
      ,NM_DIS_SAN
  FROM DBAMV.TMPMV_FFAS_GERAR_BPA
 WHERE TP_FORMULARIO = 'C'
       AND CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
 ORDER BY DT_COMPT,NM_DIS_SAN,NR_PAGINA, NR_LINHAS_DA_PAGINA
