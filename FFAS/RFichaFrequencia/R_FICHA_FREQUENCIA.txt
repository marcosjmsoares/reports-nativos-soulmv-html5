--R_FICHA_FREQUENCIA
SELECT * FROM (
SELECT CTR_FREQUENCIA.DT_REFERENCIA
       ,CTR_FREQUENCIA.CD_PACIENTE
       ,PACIENTE.NM_PACIENTE
       ,CTR_FREQUENCIA.CD_PRESTADOR
       ,NVL(PRESTADOR.NM_PRESTADOR,'NÃO INFORMADO') NM_PRESTADOR
       ,CTR_FREQUENCIA.CD_TIP_ATE
       ,TIP_ATE_APAC.DS_TIP_ATE
FROM dbamv.CTR_FREQUENCIA
     ,dbamv.PACIENTE
     ,dbamv.PRESTADOR
     ,dbamv.TIP_ATE_APAC
WHERE CTR_FREQUENCIA.CD_PACIENTE  = PACIENTE.CD_PACIENTE
  AND CTR_FREQUENCIA.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
  AND CTR_FREQUENCIA.CD_TIP_ATE   = TIP_ATE_APAC.CD_TIP_ATE
ORDER BY DT_REFERENCIA
)
ORDER BY DT_REFERENCIA,
DS_TIP_ATE,CD_TIP_ATE,
NM_PRESTADOR,CD_PRESTADOR
