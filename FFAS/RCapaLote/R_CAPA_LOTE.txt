--R_CAPA_LOTE
SELECT DECENDIOS_FAT_SIA.DS_DECENDIO
             , DECENDIOS_FAT_SIA.DT_INICIAL
             , DECENDIOS_FAT_SIA.DT_FINAL
             , FAT_SIA.DT_PERIODO_INICIAL
             , FAT_SIA.DT_PERIODO_FINAL
             , MULTI_EMPRESAS.DS_MULTI_EMPRESA
             , MULTI_EMPRESAS.NR_CNES
             , to_char(decode(CONFIG_FFAS.TP_ENTREGA_COMP, 'I', fat_sia.dt_periodo_inicial, 'F', fat_sia.dt_periodo_final, null),'MM/YYYY') TP_ENTREGA_COMP
  FROM dbamv.DECENDIOS_FAT_SIA
            , dbamv.FAT_SIA
            , dbamv.MULTI_EMPRESAS
            , dbamv.CONFIG_FFAS
 WHERE DECENDIOS_FAT_SIA.CD_FAT_SIA  = :CAMPO_FAT_SIA
       AND DECENDIOS_FAT_SIA.CD_FAT_SIA  = FAT_SIA.CD_FAT_SIA
       AND MULTI_EMPRESAS.CD_MULTI_EMPRESA = dbamv.pkg_mv2000.le_empresa
       AND MULTI_EMPRESAS.CD_MULTI_EMPRESA = CONFIG_FFAS.CD_MULTI_EMPRESA
