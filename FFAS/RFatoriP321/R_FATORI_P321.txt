--R_FATORI_P321
Select CD_FAT_SIA
     , DS_FAT_SIA
     , DT_PERIODO_FINAL
     , CD_SERVICO
     , DS_SERVICO
     , PROCEDIMENTO
     , DESCRICAO
     , CD_GRUPO_PROCEDIMENTO
     , DS_GRUPO_PROCEDIMENTO
     , nvl(sum(QT_LANCADA),0) QT_LANCADA
     , CD_ORI_ATE
     , DS_ORI_ATE
     , VL_SERVICO_AMBULATORIAL
     , SUM(vl_geral) vl_geral
     , SUM(TESTE) TESTE
     , SUM(vl_TOTAL_ambulatoriaL) vl_TOTAL_ambulatoriaL
  From (
  SELECT FAT_SIA.CD_FAT_SIA          CD_FAT_SIA
       , FAT_SIA.DS_FAT_SIA          DS_FAT_SIA
       , fat_sia.dt_periodo_final    DT_PERIODO_FINAL
       , EVE_SIASUS.CD_SERVICO       CD_SERVICO
       , Decode(EVE_SIASUS.CD_SERVICO,NULL,'Serviço não informado.',dbamv.pkg_sus_regra_servico.f_descricao_padrao(EVE_SIASUS.CD_ATENDIMENTO,EVE_SIASUS.CD_SERVICO)) DS_SERVICO
       , Decode(EVE_SIASUS.TP_ATENDIMENTO, NULL, 'A', 'E', 'U', EVE_SIASUS.TP_ATENDIMENTO) TP_ATENDIMENTO
       , EVE_SIASUS.CD_ATENDIMENTO
       , EVE_SIASUS.CD_PROCEDIMENTO  PROCEDIMENTO
       , PROCEDIMENTO_SUS.DS_PROCEDIMENTO           DESCRICAO
       , GRUPO_PROCEDIMENTOS.CD_GRUPO_PROCEDIMENTO  CD_GRUPO_PROCEDIMENTO
       , GRUPO_PROCEDIMENTOS.DS_GRUPO_PROCEDIMENTO  DS_GRUPO_PROCEDIMENTO
       , SUM(EVE_SIASUS.QT_LANCADA)  QT_LANCADA
       , ORI_ATE.CD_ORI_ATE          CD_ORI_ATE
       , ORI_ATE.DS_ORI_ATE          DS_ORI_ATE
       , VL_SERVICO_AMBULATORIAL
       , (Eve_SiaSUS.VL_TOTAL_AMBULATORIAL / Decode(Eve_SiaSUS.QT_LANCADA,0,1,Eve_SiaSUS.QT_LANCADA)) vl_geral
       , SUM(VL_TOTAL_AMBULATORIAL)  TESTE
       ,SUM(VL_TOTAL_AMBULATORIAL) VL_TOTAL_AMBULATORIAL
    FROM dbamv.FAT_SIA                  FAT_SIA
       , Dbamv.V_FFAS_Fat_Global_p321   EVE_SIASUS
       , DBAMV.PROCEDIMENTO_SUS_DETALHE PROCEDIMENTO_SUS_DETALHE
       , dbamv.PROCEDIMENTO_SUS         PROCEDIMENTO_SUS
       , dbamv.GRUPO_PROCEDIMENTOS      GRUPO_PROCEDIMENTOS
       , dbamv.ORI_ATE                  ORI_ATE
       , DBAMV.PROCEDIMENTO_SUS_COMPLEXIDADE
       , dbamv.Config_FFAS
   WHERE dbamv.FAT_SIA.CD_FAT_SIA IS NOT NULL
     AND EVE_SIASUS.CD_FAT_SIA                  = FAT_SIA.CD_FAT_SIA
     AND EVE_SIASUS.CD_ORI_ATE                  = ORI_ATE.CD_ORI_ATE(+)
     AND EVE_SIASUS.CD_PROCEDIMENTO             = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
     AND PROCEDIMENTO_SUS_DETALHE.CD_PROCEDIMENTO = EVE_SIASUS.CD_PROCEDIMENTO
     AND EVE_SIASUS.CD_PROCEDIMENTO       =  PROCEDIMENTO_SUS_COMPLEXIDADE.CD_PROCEDIMENTO
     AND PROCEDIMENTO_SUS.CD_PROCEDIMENTO = PROCEDIMENTO_SUS_COMPLEXIDADE.CD_PROCEDIMENTO
     AND PROCEDIMENTO_SUS.CD_GRUPO_PROCEDIMENTO = GRUPO_PROCEDIMENTOS.CD_GRUPO_PROCEDIMENTO
     And Config_FFAS.Cd_Multi_Empresa = EVE_SIASUS.Cd_Multi_Empresa
     AND EVE_SIASUS.CD_MULTI_EMPRESA  =   :CAMPO_CD_MULTI_EMPRESA
   GROUP BY FAT_SIA.CD_FAT_SIA
       , FAT_SIA.DS_FAT_SIA
       , fat_sia.dt_periodo_final
       , EVE_SIASUS.CD_SERVICO
       , Decode(EVE_SIASUS.CD_SERVICO,NULL,'Serviço não informado.',dbamv.pkg_sus_regra_servico.f_descricao_padrao(EVE_SIASUS.CD_ATENDIMENTO,EVE_SIASUS.CD_SERVICO))
       , Decode(EVE_SIASUS.TP_ATENDIMENTO, NULL, 'A', 'E', 'U', EVE_SIASUS.TP_ATENDIMENTO)
       , EVE_SIASUS.CD_ATENDIMENTO
       , EVE_SIASUS.CD_PROCEDIMENTO
       , PROCEDIMENTO_SUS.DS_PROCEDIMENTO
       , GRUPO_PROCEDIMENTOS.CD_GRUPO_PROCEDIMENTO
       , GRUPO_PROCEDIMENTOS.DS_GRUPO_PROCEDIMENTO
       , ORI_ATE.CD_ORI_ATE
       , ORI_ATE.DS_ORI_ATE
       , VL_SERVICO_AMBULATORIAL
       , (Eve_SiaSUS.VL_TOTAL_AMBULATORIAL / Decode(Eve_SiaSUS.QT_LANCADA,0,1,Eve_SiaSUS.QT_LANCADA))
)
group by CD_FAT_SIA
    , DS_FAT_SIA
    , DT_PERIODO_FINAL
    , CD_SERVICO
    , DS_SERVICO
    , PROCEDIMENTO
    , DESCRICAO
    , CD_GRUPO_PROCEDIMENTO
    , DS_GRUPO_PROCEDIMENTO
    , CD_ORI_ATE
    , DS_ORI_ATE
    , VL_SERVICO_AMBULATORIAL
ORDER BY CD_FAT_SIA
    , DS_FAT_SIA
	, CD_ORI_ATE
    , CD_GRUPO_PROCEDIMENTO
    , :CAMPO_ORDEM
