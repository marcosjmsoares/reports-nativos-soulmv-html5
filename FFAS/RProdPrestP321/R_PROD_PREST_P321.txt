--R_PROD_PREST_P321
SELECT * FROM (
SELECT PRESTADOR.DS_CODIGO_CONSELHO
     , PRESTADOR.NM_PRESTADOR
     , PROCEDIMENTO_SUS.CD_PROCEDIMENTO CD_SSM
     , PROCEDIMENTO_SUS.DS_PROCEDIMENTO DS_TITULO
     , SUM(EVE_SIASUS.QT_LANCADA) QT_LANCADA
     , EVE_SIASUS.VL_SERVICO_AMBULATORIAL VL_GERAL2
     , EVE_SIASUS.VL_SERVICO_PROFISSIONAL VL_AUXILIAR2
     , EVE_SIASUS.VL_SERVICO_ANESTESIA VL_ANESTESISTA2
     , SUM(EVE_SIASUS.VL_TOTAL_AMBULATORIAL) VL_TOTAL_AMBULATORIAL
     , DECODE('','E',EVE_SIASUS.CD_PRESTADOR_EXECUTANTE
                                 ,EVE_SIASUS.CD_PRESTADOR) CD_PRESTADOR
     , 'M' TP_MEDICO
     , CONSELHO.TP_CONSELHO
  FROM DBAMV.PRESTADOR             PRESTADOR
     , DBAMV.PROCEDIMENTO_SUS   PROCEDIMENTO_SUS
     , dbamv.v_ffas_fat_global_p321      EVE_SIASUS
     , DBAMV.PROCEDIMENTO_SUS_COMPLEXIDADE   PROCEDIMENTO_SUS_COMPLEXIDADE
     , DBAMV.CONSELHO
 WHERE EVE_SIASUS.CD_PROCEDIMENTO IS NOT NULL
      AND procedimento_sus_complexidade.cd_procedimento = eve_siasus.cd_procedimento
      AND procedimento_sus_complexidade.cd_procedimento = procedimento_sus.cd_procedimento
      AND EVE_SIASUS.CD_PROCEDIMENTO = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
      AND EVE_SIASUS.cd_multi_empresa =  DBAMV.PKG_MV2000.le_empresa
      AND PROCEDIMENTO_SUS.sn_ativo = 'S'
      AND PRESTADOR.CD_CONSELHO = CONSELHO.CD_CONSELHO
 GROUP BY PRESTADOR.DS_CODIGO_CONSELHO
        , PRESTADOR.NM_PRESTADOR
        , PROCEDIMENTO_SUS.CD_PROCEDIMENTO
        , PROCEDIMENTO_SUS.DS_PROCEDIMENTO
        , EVE_SIASUS.VL_SERVICO_PROFISSIONAL
        , EVE_SIASUS.VL_SERVICO_ANESTESIA
        , DECODE('','E',EVE_SIASUS.CD_PRESTADOR_EXECUTANTE
                                    ,EVE_SIASUS.CD_PRESTADOR)
        , EVE_SIASUS.VL_SERVICO_AMBULATORIAL
        , 'M'
        , CONSELHO.TP_CONSELHO
        :CAMPO_CF_ORDEM
)

--R_PROD_PREST_P321_R_PRESTADOR2_Q_ANESTESISTA
SELECT * FROM (
SELECT PRESTADOR.DS_CODIGO_CONSELHO,
              PRESTADOR.NM_PRESTADOR,
              PROCEDIMENTO_SUS.CD_PROCEDIMENTO CD_SSM,
              PROCEDIMENTO_SUS.DS_PROCEDIMENTO DS_TITULO,
              SUM(EVE_SIASUS.QT_LANCADA) QT_LANCADA,
              EVE_SIASUS.VL_TOTAL_AMBULATORIAL VL_GERAL,
              EVE_SIASUS.VL_SERVICO_PROFISSIONAL VL_AUXILIAR,
              EVE_SIASUS.VL_SERVICO_ANESTESIA VL_ANESTESISTA,
              SUM(EVE_SIASUS.VL_TOTAL_AMBULATORIAL) VL_TOTAL_AMBULATORIAL,
              NVL(EVE_SIASUS.CD_PRESTADOR_PODE_TER, EVE_SIASUS.CD_PRESTADOR)       CD_PRESTADOR,
              'A'     TP_MEDICO
            , CONSELHO.TP_CONSELHO
FROM    DBAMV.PRESTADOR           PRESTADOR,
        DBAMV.PROCEDIMENTO_SUS,
        dbamv.v_ffas_fat_global_p321            EVE_SIASUS,
        DBAMV.FAT_SIA                    FAT_SIA
      , DBAMV.CONSELHO
WHERE EVE_SIASUS.CD_PROCEDIMENTO IS NOT NULL
AND      EVE_SIASUS.CD_PRESTADOR_PODE_TER = PRESTADOR.CD_PRESTADOR
AND       EVE_SIASUS.CD_FAT_SIA = FAT_SIA.CD_FAT_SIA
AND       EVE_SIASUS.CD_PROCEDIMENTO  =  PROCEDIMENTO_SUS.CD_PROCEDIMENTO
AND 	  EVE_SIASUS.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
and        PROCEDIMENTO_SUS.sn_ativo = 'S'
AND       PRESTADOR.CD_CONSELHO = CONSELHO.CD_CONSELHO
GROUP BY PRESTADOR.DS_CODIGO_CONSELHO,
                    PRESTADOR.NM_PRESTADOR,
                    PROCEDIMENTO_SUS.CD_PROCEDIMENTO,
                    PROCEDIMENTO_SUS.DS_PROCEDIMENTO,
                    EVE_SIASUS.VL_TOTAL_AMBULATORIAL,
                    EVE_SIASUS.VL_SERVICO_PROFISSIONAL,
                    EVE_SIASUS.VL_SERVICO_ANESTESIA,
                    NVL(EVE_SIASUS.CD_PRESTADOR_PODE_TER, EVE_SIASUS.CD_PRESTADOR) ,
                     'A'
                  , CONSELHO.TP_CONSELHO
:CAMPO_CF_ORDEM
)
