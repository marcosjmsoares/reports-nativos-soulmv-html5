SELECT * FROM (
SELECT CIDADE.NM_CIDADE                                         NM_CIDADE
      ,GRU_PRO.CD_GRUPO_PROCEDIMENTO                            GRUPO
      ,SUB_PRO.CD_SUB_GRUPO_PROCEDIMENTO                        SUB_GRUPO
      ,SUM( EVE_SUS.QT_LANCADA )                                QT_LANCADA
      ,SUM( EVE_SUS.vl_total_ambulatorial )                     TOTAL
  FROM DBAMV.EVE_SIASUS                 EVE_SUS
      ,DBAMV.PROCEDIMENTO_SUS           PROC_SUS    
      ,DBAMV.CIDADE                     CIDADE
      ,DBAMV.GRUPO_PROCEDIMENTOS        GRU_PRO     
      ,DBAMV.SUB_GRUPO_PROCEDIMENTOS    SUB_PRO     
WHERE EVE_SUS.cd_procedimento            = PROC_SUS.cd_procedimento
  AND EVE_SUS.CD_CIDADE                  = CIDADE.CD_CIDADE
  AND EVE_SUS.CD_FAT_SIA                 = :CAMPO_FAT_SIA
  AND GRU_PRO.CD_GRUPO_PROCEDIMENTO      = PROC_SUS.CD_GRUPO_PROCEDIMENTO
  AND SUB_PRO.CD_GRUPO_PROCEDIMENTO      = PROC_SUS.CD_GRUPO_PROCEDIMENTO
  AND SUB_PRO.CD_SUB_GRUPO_PROCEDIMENTO  = PROC_SUS.CD_SUB_GRUPO_PROCEDIMENTO
GROUP BY CIDADE.NM_CIDADE
        ,GRU_PRO.CD_GRUPO_PROCEDIMENTO          
        ,SUB_PRO.CD_SUB_GRUPO_PROCEDIMENTO      
)
ORDER BY NM_CIDADE,
GRUPO
