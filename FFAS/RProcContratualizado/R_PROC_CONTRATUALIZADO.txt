Select Ups_Val_Contrat.Cd_Multi_Empresa 			Cd_Multi_Empresa
      ,Ups_Val_Contrat.Cd_Ups									Cd_Ups
      ,Unid_Prest.Ds_Ups											Ds_Ups
      ,Ups_Val_Contrat.Cd_Procedimento				Cd_Procedimento
      ,Proc_Sus.Ds_procedimento       				Ds_procedimento
      ,Nvl(Ups_Val_Contrat.Vl_Servico_Ambulatorial,0) Vl_Servico_Ambulatorial
      ,Nvl(Ups_Val_Contrat.Vl_Servico_Profissional_amb,0) Vl_Servico_Profissional_amb
      ,Nvl(Ups_Val_Contrat.Vl_Servico_Anestesia_Amb,0) Vl_Servico_Anestesia_Amb
      ,Nvl(Ups_Val_Contrat.Vl_Total_Ambulatorial,0) Vl_Total_Ambulatorial
      ,Ups_Val_Contrat.Dt_Vigencia						Dt_Vigencia
  From dbamv.Procedimento_Sus_Valor_Contrat  Ups_Val_Contrat
      ,dbamv.Unidade_Prestadora_Servico      Unid_Prest
      ,dbamv.Procedimento_Sus                Proc_Sus
 where Ups_Val_Contrat.cd_ups 					= Unid_Prest.cd_ups
  and  Ups_Val_Contrat.cd_procedimento 	= Proc_Sus.cd_procedimento
Order by Ups_Val_Contrat.Cd_Ups ,Ups_Val_Contrat.Cd_Procedimento
