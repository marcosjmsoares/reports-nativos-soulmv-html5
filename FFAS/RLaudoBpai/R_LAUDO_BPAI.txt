--R_LAUDO_BPAI
SELECT LAUDO_BPAI.CD_LAUDO
     , DBAMV.PKG_SUS_REGRA_SETOR.F_DESCRICAO(DBAMV.pkg_ffas_regra_origem_atend.F_SETOR_PADRAO(LAUDO_BPAI.CD_ORI_ATE)) SETOR_DESCRICAO
     , LAUDO_BPAI.DT_COMPETENCIA
     , DBAMV.PKG_FFAS_REGRA_LAUDO_BPAI.f_pac_same_prontuario('PD',PACIENTE.CD_PACIENTE) CD_PACIENTE
     , TO_CHAR(LAUDO_BPAI.dt_emissao_laudo,'DD/MM/YYYY') DT_LAUDO
     , TO_CHAR(LAUDO_BPAI.dt_realizacao,'DD/MM/YYYY') DT_REALIZACAO
     , LAUDO_BPAI.DS_DIAGNOSTICO
     , LAUDO_BPAI.CD_CID_PRINCIPAL
     , LAUDO_BPAI.CD_CID_SECUNDARIO
     , LAUDO_BPAI.CD_CID_CAUSAS_ASSOCIADAS
     , LAUDO_BPAI.DS_OBSERVACAO
     , LAUDO_BPAI.DS_JUSTIFICATIVA
     , LAUDO_BPAI.SN_IMPRESSO
     , LAUDO_BPAI.SN_RETORNO_SECRETARIA
     , TO_NUMBER( CIDADE.CD_IBGE  || CIDADE.NR_DIGITO_IBGE)  CD_IBGE
     , dbamv.pkg_sus_regra_paciente.F_NOME_PACIENTE(PACIENTE.CD_PACIENTE) NM_PACIENTE
     , PACIENTE.NR_FONE
     , PACIENTE.NR_CPF
     , PACIENTE.DT_NASCIMENTO
     , PACIENTE.TP_SEXO
     , PACIENTE.NM_MAE
     , PACIENTE.DS_ENDERECO||', '||PACIENTE.NR_ENDERECO||', '||PACIENTE.NM_BAIRRO||', '||PACIENTE.DS_COMPLEMENTO  DS_ENDERECO
     , PACIENTE.DS_COMPLEMENTO
     , PACIENTE.NR_CEP
     , PACIENTE.CD_CIDADE
     , PACIENTE.NR_CNS
     , CIDADE.NM_CIDADE
     , CIDADE.CD_UF
     , NVL(LAUDO_BPAI.CD_PRESTADOR_SOLICITANTE , PREST_SOL.CD_PRESTADOR ) CD_PRESTADOR
     , DECODE(LAUDO_BPAI.DS_NOME_PRESTADOR_SOLICITANTE
             ,NULL
             ,DECODE(LAUDO_BPAI.CD_PRESTADOR_SOLICITANTE
                     ,NULL
                     ,(SELECT C.DS_CONSELHO FROM DBAMV.PRESTADOR P, DBAMV.CONSELHO C WHERE P.CD_CONSELHO = P.CD_CONSELHO
                                AND P.CD_PRESTADOR = PREST_RES.CD_PRESTADOR AND C.CD_CONSELHO = PREST_RES.CD_CONSELHO)
		    ||' '||PREST_RES.DS_CODIGO_CONSELHO||' - '||PREST_RES.NM_PRESTADOR
             ,(SELECT C.DS_CONSELHO FROM DBAMV.PRESTADOR P, DBAMV.CONSELHO C WHERE P.CD_CONSELHO = P.CD_CONSELHO
                                AND P.CD_PRESTADOR = PREST_SOL.CD_PRESTADOR AND C.CD_CONSELHO = PREST_SOL.CD_CONSELHO)
	     ||' '||PREST_SOL.DS_CODIGO_CONSELHO||' - '||PREST_SOL.NM_PRESTADOR)
             , PRESTADOR_EXTERNO.NR_CONSELHO||' '|| LAUDO_BPAI.DS_CONSELHO_SOLICITANTE||' - '||LAUDO_BPAI.DS_NOME_PRESTADOR_SOLICITANTE) NM_PRESTADOR
     , PREST_RES.NR_CPF_CGC
     , Nvl(PREST_SOL.NR_CNS, PREST_RES.NR_CNS)    NR_CNS_SOLIC
     , PREST_RES.NR_CNS    NR_CNS_RESP
     , PREST_RES.NM_PRESTADOR NM_PREST_RESP
     , PREST_RES.DS_CODIGO_CONSELHO DS_CRM_RESP
     , ITEM_LAUDO_BPAI.QT_PROCEDIMENTO
     , ITEM_LAUDO_BPAI.CD_PROCEDIMENTO  CD_PROCEDIMENTO_IT
     , ITEM_LAUDO_BPAI.CD_CBO_PRESTADOR
     , SUBSTR(PROCEDIMENTO_SUS.DS_PROCEDIMENTO,0,90)
     , nvl(RESPONSA.NM_RESPONSAVEL,PACIENTE.NM_PACIENTE) NM_RESPONSAVEL
     , nvl(RESPONSA.NR_FONE,PACIENTE.NR_FONE) FONE_RESP
     , Decode(Paciente.Tp_Cor,'B','01 - Branca','P','02 - Preta','R','03 - Parda','A','04 - Amarela','I','05 - Indigena','99 - Sem InformaÃƒÂ§ÃƒÂ£o') Cor_Raca         
     , Decode(Etnia.Cd_Etnia,NULL,'',Etnia.Cd_Etnia || ' - ' || Etnia.Ds_Etnia) Ds_Etnia
     , LAUDO_BPAI.CD_ESTABELECIMENTO
     , ME.CD_ORGAO_EMISSOR_AIH
     , CIDADANIAS.DS_CIDADANIA Nacionalidade
     , CIDADANIAS.CD_CIDADANIA Cd_Nacionalidade
	 , AUT.NM_AUTORIZADOR Nm_Autorizador
	 , AUT.NR_CNS  NR_CNS_AUT
     , AUT.NR_CPF  NR_CPF_AUT
  FROM DBAMV.LAUDO_BPAI
     , DBAMV.PACIENTE PACIENTE
     , DBAMV.CIDADE CIDADE
     , DBAMV.PRESTADOR PREST_SOL
     , DBAMV.PRESTADOR PREST_RES
     , DBAMV.ITEM_LAUDO_BPAI
     , DBAMV.PROCEDIMENTO_SUS
     , DBAMV.RESPONSA
     , DBAMV.ATENDIME
     , DBAMV.MULTI_EMPRESAS ME
     , DBAMV.ETNIA
     , DBAMV.CIDADANIAS
     , DBAMV.PRESTADOR_EXTERNO
	 , DBAMV.AUTORIZADOR AUT
 WHERE LAUDO_BPAI.CD_LAUDO                 = :CAMPO_LAUDO
   AND PROCEDIMENTO_SUS.CD_PROCEDIMENTO    = ITEM_LAUDO_BPAI.CD_PROCEDIMENTO
   AND LAUDO_BPAI.CD_PACIENTE              = PACIENTE.CD_PACIENTE
   AND ME.CD_MULTI_EMPRESA                 = DBAMV.PKG_MV2000.LE_EMPRESA
   AND LAUDO_BPAI.CD_PRESTADOR             = PREST_RES.CD_PRESTADOR(+)
   AND LAUDO_BPAI.CD_PRESTADOR_SOLICITANTE = PREST_SOL.CD_PRESTADOR(+)
   AND RESPONSA.CD_ATENDIMENTO(+)          = ATENDIME.CD_ATENDIMENTO
   AND PACIENTE.CD_CIDADE                  = CIDADE.CD_CIDADE(+)
   AND PACIENTE.CD_ETNIA                   = ETNIA.CD_ETNIA(+)
   AND LAUDO_BPAI.CD_LAUDO                 = ITEM_LAUDO_BPAI.CD_LAUDO(+)
   AND LAUDO_BPAI.CD_MULTI_EMPRESA         = DBAMV.PKG_MV2000.LE_EMPRESA
   AND LAUDO_BPAI.CD_MULTI_EMPRESA         = ME.CD_MULTI_EMPRESA
   AND LAUDO_BPAI.CD_ATENDIMENTO           = ATENDIME.CD_ATENDIMENTO(+)
   AND PACIENTE.CD_CIDADANIA               = CIDADANIAS.CD_CIDADANIA
   AND PRESTADOR_EXTERNO.CD_PRES_EXT(+)    = LAUDO_BPAI.CD_PRES_EXT
   AND LAUDO_BPAI.CD_AUTORIZADOR           = AUT.CD_AUTORIZADOR(+)
   AND ROWNUM <= 3
   ORDER BY ITEM_LAUDO_BPAI.CD_ITEM_LAUDO
