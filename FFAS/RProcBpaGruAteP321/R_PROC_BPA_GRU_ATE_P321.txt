--R_PROC_BPA_GRU_ATE_P321
SELECT EVE_SIASUS.cd_procedimento CD_PROCEDIMENTO
       ,PROC.ds_procedimento DS_TITULO
       ,SUM(NVL(EVE_SIASUS.QT_LANCADA,0)) QT_LANCADA
       ,EVE_SIASUS.vl_servico_ambulatorial VL_GERAL
       ,SUM(EVE_SIASUS.vl_total_ambulatorial ) VL_TOTAL
 FROM dbamv.EVE_SIASUS ,
      DBAMV.PROCEDIMENTO_SUS        PROC,
      dbamv.PROCEDIMENTO_SUS_COMPLEXIDADE             COMP,
      dbamv.FAT_SIA
WHERE EVE_SIASUS.CD_FAT_SIA IS NOT NULL
  AND EVE_SIASUS.CD_FAT_SIA       = FAT_SIA.CD_FAT_SIA
  AND COMP.CD_PROCEDIMENTO   = PROC.CD_PROCEDIMENTO
  AND EVE_SIASUS.cd_procedimento    = PROC.CD_PROCEDIMENTO
 and comp.sn_ativo = 'S'
GROUP BY  EVE_SIASUS.CD_PROCEDIMENTO
        ,PROC.DS_PROCEDIMENTO
        ,EVE_SIASUS.vl_servico_ambulatorial
ORDER BY PROC.DS_PROCEDIMENTO
