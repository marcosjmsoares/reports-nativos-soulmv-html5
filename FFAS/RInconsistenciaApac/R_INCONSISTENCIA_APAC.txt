--R_INCONSISTENCIA_APAC
SELECT CRITICA_SIA.CD_EVENTOS CD_EVENTOS
       ,CRITICA_SIA.CD_APAC CD_APAC
       ,CRITICA_SIA.NR_APAC NR_APAC
       ,APAC.CD_PACIENTE CD_PACIENTE
       ,PACIENTE.NM_PACIENTE NM_PACIENTE
       ,LTrim(CRITICA_SIA.DS_CRITICA,Chr(10)) DS_CRITICA
       ,CRITICA_SIA.TP_CRITICA TP_CRITICA
       ,APAC.CD_ORI_ATE
       ,ORI_ATE.DS_ORI_ATE
       ,APAC.CD_REMESSA
       ,REMESSA_APAC.DS_REMESSA
  FROM DBAMV.CRITICA_SIA
       ,DBAMV.APAC
       ,DBAMV.PACIENTE
       ,DBAMV.ORI_ATE
       ,DBAMV.REMESSA_APAC
WHERE CRITICA_SIA.CD_FAT_SIA = :CAMPO_FATURA
  AND CRITICA_SIA.CD_FAT_SIA = APAC.CD_FAT_SIA
  AND APAC.CD_REMESSA = REMESSA_APAC.CD_REMESSA
  AND CRITICA_SIA.CD_APAC = APAC.CD_APAC
  AND APAC.CD_PACIENTE = PACIENTE.CD_PACIENTE
  AND APAC.CD_ORI_ATE = ORI_ATE.CD_ORI_ATE
