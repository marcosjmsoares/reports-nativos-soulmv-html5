--R_RELACAO_NOMINAL_ORIGEM
SELECT * FROM (
SELECT DECODE(EVE_SIASUS.CD_PACIENTE,null,EVE_SIASUS.NM_PACIENTE, PACIENTE.NM_PACIENTE)    NM_PACIENTE
     , EVE_SIASUS.DT_EVE_SIASUS
     , EVE_SIASUS.CD_PACIENTE
     , EVE_SIASUS.CD_SSM
     , eve_siasus.cd_eventos
     , EVE_SIASUS.CD_PROCEDIMENTO
     , EVE_SIASUS.cd_cbo_prestador
     , EVE_SIASUS.qt_lancada
     , prestador.cd_prestador
     , prestador.nm_prestador
     , cidade.nm_cidade || ' - ' || cidade.cd_uf nm_cidade
     , eve_siasus.cd_ori_ate
     , ori_ate.ds_ori_ate
  FROM dbamv.EVE_SIASUS
     , DBAMV.PACIENTE
     , dbamv.prestador
     , dbamv.cidade
     , dbamv.ori_ate
 WHERE EVE_SIASUS.CD_PACIENTE          = PACIENTE.CD_PACIENTE(+)
   AND trunc(eve_siasus.dt_eve_siasus) between :CAMPO_DT_INICIAL and 
   AND EVE_SIASUS.CD_MULTI_EMPRESA     = :CAMPO_CD_MULTI_EMPRESA
   AND EVE_SIASUS.CD_CONVENIO          = Dbamv.Pkt_Config_ffas.Retorna_Campo('cd_convenio')
   and cidade.cd_cidade(+)             = paciente.cd_cidade
   and ori_ate.cd_ori_ate(+)           = eve_siasus.cd_ori_ate
   :CAMPO_CF_PRESTADORES
)
