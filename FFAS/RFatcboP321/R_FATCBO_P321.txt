--R_FATCBO_P321_R_FAT_SIA2_Q_FAT_SIA_TP_FINANCIAMENTO_AMB
SELECT * FROM (
SELECT  FAT_SIA.CD_FAT_SIA                          CD_FAT_SIA
       ,FAT_SIA.DS_FAT_SIA                          DS_FAT_SIA
       ,EVE_SIASUS.CD_SERVICO                       CD_SERVICO
       ,SER_DIS.DS_SER_DIS                          DS_SER_DIS
       ,CBO.DS_CBOS                                 DS_ESPECIALID
       ,EVE_SIASUS.CD_CBO_PRESTADOR                 CD_ESPECIALID
       ,EVE_SIASUS.CD_PROCEDIMENTO                  PROCEDIMENTO
       ,EVE_SIASUS.CD_ORI_ATE                       CD_ORI_ATE
       ,ORI_ATE.DS_ORI_ATE                          DS_ORI_ATE
       ,PROCEDIMENTO_SUS.ds_procedimento            DESCRIÇÃO
       ,EVE_SIASUS.vl_servico_ambulatorial          VL_GERAL
       ,GRUPO_PROCEDIMENTOS.cd_grupo_procedimento   cd_grupo_procedimento
       ,GRUPO_PROCEDIMENTOS.ds_grupo_procedimento   ds_grupo_procedimento
       ,SUM(EVE_SIASUS.QT_LANCADA)                  QT_LANCADA
       ,SUM(EVE_SIASUS.VL_SERVICO_AMBULATORIAL)     VL_SERVICO_AMBULATORIAL
       ,SUM(EVE_SIASUS.VL_TOTAL_AMBULATORIAL)       VL_TOTAL_AMBULATORIAL
       ,PROCEDIMENTO_SUS.TP_FINANCIAMENTO           TP_FINANCIAMENTO
FROM   dbamv.FAT_SIA                                       FAT_SIA
      ,dbamv.EVE_SIASUS                                    EVE_SIASUS
      ,dbamv.ATENDIME                                      ATENDIME
      ,dbamv.SER_DIS                                       SER_DIS
      ,dbamv.CBO                                           CBO
      ,dbamv.PROCEDIMENTO_SUS                              PROCEDIMENTO_SUS
      ,dbamv.GRUPO_PROCEDIMENTOS                           GRUPO_PROCEDIMENTOS
      ,dbamv.ORI_ATE                                       ORI_ATE
      ,dbamv.PROCEDIMENTO_SUS_DETALHE
      ,dbamv.PROCEDIMENTO_SUS_COMPLEXIDADE
  WHERE FAT_SIA.CD_FAT_SIA      = :CAMPO_FAT_SIA
  AND EVE_SIASUS.CD_FAT_SIA     =  FAT_SIA.CD_FAT_SIA
  AND EVE_SIASUS.CD_SERVICO = SER_DIS.CD_SER_DIS (+)
  AND EVE_SIASUS.CD_CBO_PRESTADOR = CBO.CD_CBOS
  AND EVE_SIASUS.CD_ORI_ATE  =  ORI_ATE.CD_ORI_ATE
  AND EVE_SIASUS.CD_PROCEDIMENTO  =  PROCEDIMENTO_SUS.CD_PROCEDIMENTO
  AND EVE_SIASUS.CD_ATENDIMENTO  = ATENDIME.CD_ATENDIMENTO(+)
  AND PROCEDIMENTO_SUS.cd_grupo_procedimento  =  GRUPO_PROCEDIMENTOS.cd_grupo_procedimento
  AND PROCEDIMENTO_SUS_DETALHE.CD_PROCEDIMENTO = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
  AND PROCEDIMENTO_SUS_COMPLEXIDADE.CD_PROCEDIMENTO =  EVE_SIASUS.CD_PROCEDIMENTO
and PROCEDIMENTO_SUS_COMPLEXIDADE.sn_ativo = 'S'
GROUP BY FAT_SIA.CD_FAT_SIA
        ,FAT_SIA.DS_FAT_SIA
        ,EVE_SIASUS.CD_SERVICO
        ,SER_DIS.DS_SER_DIS
        ,CBO.DS_CBOS
        ,EVE_SIASUS.CD_CBO_PRESTADOR
        ,EVE_SIASUS.CD_PROCEDIMENTO
        ,EVE_SIASUS.CD_ORI_ATE
        ,ORI_ATE.DS_ORI_ATE
        ,PROCEDIMENTO_SUS.ds_procedimento
        ,GRUPO_PROCEDIMENTOS.cd_grupo_procedimento
        ,GRUPO_PROCEDIMENTOS.ds_grupo_procedimento
        ,EVE_SIASUS.vl_servico_ambulatorial
        ,PROCEDIMENTO_SUS.TP_FINANCIAMENTO
ORDER BY FAT_SIA.CD_FAT_SIA
        ,FAT_SIA.DS_FAT_SIA
        ,CBO.DS_CBOS
        ,EVE_SIASUS.CD_CBO_PRESTADOR
        ,GRUPO_PROCEDIMENTOS.cd_grupo_procedimento
        ,:CAMPO_ORDEM
)
ORDER BY CD_FAT_SIA,DS_FAT_SIA,
TP_FINANCIAMENTO,
CD_SERVICO,DS_SER_DIS,
CD_ORI_ATE,DS_ORI_ATE,
CD_ESPECIALID,DS_ESPECIALID
