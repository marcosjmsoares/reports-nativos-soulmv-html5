--R_REL_NOM_AN_PG_321
SELECT * FROM (
SELECT *
  FROM (
  SELECT dbamv.pkg_sus_regra_paciente.F_NOME_PACIENTE(EVE_SIASUS.CD_PACIENTE) NM_PACIENTE
       , EVE_SIASUS.DT_EVE_SIASUS
       , EVE_SIASUS.CD_PACIENTE
       , EVE_SIASUS.CD_SSM
       , EVE_SIASUS.QT_LANCADA
       , EVE_SIASUS.CD_PROCEDIMENTO
        ,EVE_SIASUS.CD_CBO_PRESTADOR
       , EVE_SIASUS.CD_ATENDIMENTO
       , EVE_SIASUS.vl_total_ambulatorial VL_GERAL
        ,CBO.DS_CBOS
       , TO_DATE(NULL) DT_NASCIMENTO
       , CD_EVENTOS
    FROM
         dbamv.v_ffas_fat_global_p321 EVE_SIASUS
        ,DBAMV.CBO
       , DBAMV.MULTI_EMPRESAS
   WHERE EVE_SIASUS.CD_CBO_PRESTADOR        = CBO.CD_CBOS
     AND EVE_SIASUS.CD_PACIENTE          IS NULL
     AND EVE_SIASUS.NM_PACIENTE          IS NOT NULL
     AND MULTI_EMPRESAS.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.le_empresa
     AND EVE_SIASUS.CD_MULTI_EMPRESA     = MULTI_EMPRESAS.CD_MULTI_EMPRESA
     AND EVE_SIASUS.CD_CONVENIO          =  Dbamv.Pkt_Config_ffas.Retorna_Campo('cd_convenio')
UNION ALL
 SELECT dbamv.pkg_sus_regra_paciente.F_NOME_PACIENTE(EVE_SIASUS.CD_PACIENTE) NM_PACIENTE
      , EVE_SIASUS.DT_EVE_SIASUS
      , EVE_SIASUS.CD_PACIENTE
      , EVE_SIASUS.CD_SSM
       , EVE_SIASUS.QT_LANCADA
      , EVE_SIASUS.CD_PROCEDIMENTO
       ,EVE_SIASUS.CD_CBO_PRESTADOR
      , EVE_SIASUS.CD_ATENDIMENTO
        , EVE_SIASUS.vl_total_ambulatorial VL_GERAL
        ,CBO.DS_CBOS
      , PACIENTE.DT_NASCIMENTO
     , CD_EVENTOS
   FROM dbamv.v_ffas_fat_global_p321 EVE_SIASUS
       ,DBAMV.CBO
      , DBAMV.procedimento_sus_valor VTS
      , DBAMV.PACIENTE
      , DBAMV.MULTI_EMPRESAS
  WHERE EVE_SIASUS.CD_CBO_PRESTADOR        = CBO.CD_CBOS
    AND EVE_SIASUS.CD_PACIENTE          = PACIENTE.CD_PACIENTE(+)
    AND EVE_SIASUS.CD_PROCEDIMENTO      = VTS.CD_PROCEDIMENTO
    AND MULTI_EMPRESAS.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.le_empresa
    AND EVE_SIASUS.CD_MULTI_EMPRESA     = MULTI_EMPRESAS.CD_MULTI_EMPRESA
    AND EVE_SIASUS.CD_CONVENIO          =  Dbamv.Pkt_Config_ffas.Retorna_Campo('cd_convenio')  
    AND VTS.DT_VIGENCIA                 = (SELECT MAX(psv.DT_VIGENCIA) dt_vigencia
                                             FROM DBAMV.procedimento_sus_valor psv
                                            WHERE psv.dt_vigencia <= nvl(EVE_SIASUS.dt_periodo_final, sysdate)
                                                  AND psv.CD_PROCEDIMENTO = EVE_SIASUS.CD_PROCEDIMENTO)
  group by EVE_SIASUS.CD_PACIENTE
      , EVE_SIASUS.DT_EVE_SIASUS
      , EVE_SIASUS.CD_PACIENTE
      , EVE_SIASUS.CD_SSM
       , EVE_SIASUS.QT_LANCADA
      , EVE_SIASUS.CD_PROCEDIMENTO
       ,EVE_SIASUS.CD_CBO_PRESTADOR
      , EVE_SIASUS.CD_ATENDIMENTO
      , EVE_SIASUS.QT_LANCADA
       , CD_EVENTOS
, EVE_SIASUS.vl_total_ambulatorial
       ,CBO.DS_CBOS
      , PACIENTE.DT_NASCIMENTO) VISAO
 :CAMPO_CF_ORDEM
)
ORDER BY DS_CBOS,CD_CBO_PRESTADOR, NM_PACIENTE ASC
