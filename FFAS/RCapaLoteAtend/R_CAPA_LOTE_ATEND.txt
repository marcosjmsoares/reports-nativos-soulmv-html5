--R_CAPA_LOTE_ATEND
SELECT * FROM (
SELECT       EVE_SIASUS.CD_PROCEDIMENTO
                   , EVE_SIASUS.CD_ATENDIMENTO
                   , EVE_SIASUS.CD_CBO_PRESTADOR
                   , EVE_SIASUS.CD_CARATER_ATENDIMENTO
                   , EVE_SIASUS.NR_IDADE
                   , sum(nvl(QT_LANCADA,0)) APRESENTADAS
                   , FAT_SIA.DS_FAT_SIA
                   , DECENDIOS_FAT_SIA.DS_DECENDIO
                   , PACIENTE.NM_PACIENTE
FROM dbamv.EVE_SIASUS
                   , dbamv.ATENDIME
                   , dbamv.TIPO_INTERNACAO
                   , dbamv.FAT_SIA
                   , dbamv.DECENDIOS_FAT_SIA
                   , DBAMV.PACIENTE
WHERE EVE_SIASUS.CD_MULTI_EMPRESA    = dbamv.pkg_mv2000.le_empresa
                 AND EVE_SIASUS.CD_FAT_SIA          = :CAMPO_FAT_SIA
                 AND FAT_SIA.CD_FAT_SIA             = EVE_SIASUS.CD_FAT_SIA
                 AND DECENDIOS_FAT_SIA.CD_FAT_SIA   = EVE_SIASUS.CD_FAT_SIA
                 AND DECENDIOS_FAT_SIA.CD_DECENDIO  = EVE_SIASUS.CD_DECENDIO
                 AND PACIENTE.CD_PACIENTE(+)        = EVE_SIASUS.CD_PACIENTE
                 AND ATENDIME.CD_CONVENIO(+)        = EVE_SIASUS.CD_CONVENIO
                 AND ATENDIME.CD_ATENDIMENTO(+)     = EVE_SIASUS.CD_ATENDIMENTO
                 AND ATENDIME.CD_TIPO_INTERNACAO    = TIPO_INTERNACAO.CD_TIPO_INTERNACAO(+)
                 AND TRUNC(EVE_SIASUS.dt_decendio) Between :CAMPO_DT_INICIAL group by
                     EVE_SIASUS.CD_PROCEDIMENTO
                   , EVE_SIASUS.CD_ATENDIMENTO
                   , FAT_SIA.DS_FAT_SIA
                   , DECENDIOS_FAT_SIA.DS_DECENDIO
                   , PACIENTE.NM_PACIENTE
                   , EVE_SIASUS.CD_CBO_PRESTADOR
                   , EVE_SIASUS.CD_CARATER_ATENDIMENTO
                   , EVE_SIASUS.NR_IDADE
order by EVE_SIASUS.CD_ATENDIMENTO
)
ORDER BY DS_FAT_SIA,DS_DECENDIO
