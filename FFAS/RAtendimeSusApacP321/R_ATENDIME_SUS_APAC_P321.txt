--R_ATENDIME_SUS_APAC_P321
SELECT  APAC.CD_PACIENTE                                    CD_PACIENTE
       ,PACIENTE.NM_PACIENTE                                NM_PACIENTE
       ,PACIENTE.NR_CPF                                     NR_CPF
       ,APAC.NR_APAC                                        NR_APAC
       ,EVE_SIASUS.cd_procedimento                          CD_PROCEDIMENTO
       ,EVE_SIASUS.SN_APAC_PRINCIPAL                        SN_APAC_PRINCIPAL
       ,APAC.CD_CID_PRINCIPAL                               CD_CID_PRINCIPAL
       ,sum(EVE_SIASUS.QT_LANCADA)                               QT_LANCADA
       ,sum(EVE_SIASUS.vl_servico_ambulatorial)                  VL_GERAL
       ,sum(EVE_SIASUS.vl_total_ambulatorial)                    VL_TOTAL
       ,APAC.CD_MOTIVO_COBRANCA                             CD_MOTIVO_COBRANCA
       ,NVL(CIDADE.NM_CIDADE,'NAO INFORMADO')               NM_CIDADE
       ,APAC.DS_MEDICAMENTOS                                DS_MEDICAMENTOS
 FROM dbamv.APAC       ,
      dbamv.FAT_SIA   ,
      dbamv.PACIENTE   ,
      dbamv.EVE_SIASUS ,
      dbamv.procedimento_sus,
      dbamv.CIDADE
WHERE APAC.CD_FAT_SIA      = :CAMPO_FAT_SIA
  AND FAT_SIA.CD_FAT_SIA   = EVE_SIASUS.CD_FAT_SIA
  AND APAC.CD_FAT_SIA      = EVE_SIASUS.CD_FAT_SIA
  AND APAC.CD_PACIENTE     = PACIENTE.CD_PACIENTE
  AND PACIENTE.CD_CIDADE   = CIDADE.CD_CIDADE (+)
  AND APAC.CD_APAC         = EVE_SIASUS.CD_APAC
  AND EVE_SIASUS.SN_PACOTE = 'N'
  AND EVE_SIASUS.cd_procedimento    =    procedimento_sus.cd_procedimento
GROUP BY APAC.CD_PACIENTE
       ,PACIENTE.NM_PACIENTE
       ,PACIENTE.NR_CPF
       ,APAC.NR_APAC
       ,EVE_SIASUS.cd_procedimento
       ,EVE_SIASUS.SN_APAC_PRINCIPAL
       ,APAC.CD_CID_PRINCIPAL
       ,APAC.CD_MOTIVO_COBRANCA
       ,NVL(CIDADE.NM_CIDADE,'NAO INFORMADO')
       ,APAC.DS_MEDICAMENTOS
