SELECT * FROM (
SELECT * FROM (
SELECT FAT_SIA.DS_FAT_SIA
      ,GRUPO_PROCEDIMENTOS.CD_GRUPO_PROCEDIMENTO                             GRUPO
      ,GRUPO_PROCEDIMENTOS.DS_GRUPO_PROCEDIMENTO                             DESCRICAO
      ,Sum(Nvl(EVE_SIASUS.VL_TOTAL_AMBULATORIAL,0))                          VL_REALIZADO
      ,SUM(EVE_SIASUS.QT_LANCADA)                                            QT_LANCADA
FROM   DBAMV.FAT_SIA
      ,DBAMV.EVE_SIASUS
      ,DBAMV.PROCEDIMENTO_SUS
      ,DBAMV.GRUPO_PROCEDIMENTOS
WHERE FAT_SIA.CD_FAT_SIA                     = :CAMPO_FAT_SIA
AND   FAT_SIA.CD_FAT_SIA                     = EVE_SIASUS.CD_FAT_SIA
AND   EVE_SIASUS.CD_PROCEDIMENTO             = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
AND   PROCEDIMENTO_SUS.CD_GRUPO_PROCEDIMENTO = GRUPO_PROCEDIMENTOS.CD_GRUPO_PROCEDIMENTO
GROUP BY FAT_SIA.DS_FAT_SIA
	 ,GRUPO_PROCEDIMENTOS.DS_GRUPO_PROCEDIMENTO
         ,GRUPO_PROCEDIMENTOS.CD_GRUPO_PROCEDIMENTO ) VISAO
:CAMPO_CF_ORD
)
ORDER BY DS_FAT_SIA
