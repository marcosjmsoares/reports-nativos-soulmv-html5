--R_PAC_FALTOSOS
select max(cd_laudo) cd_laudo
         ,laudo_apac.cd_paciente
         ,paciente.nm_paciente
         ,dt_laudo
         ,cd_tip_ate
         ,cd_procedimento cd_ssm_principal
from dbamv.laudo_sia_apac laudo_apac
      ,dbamv.paciente
where laudo_apac.cd_paciente     = paciente.cd_paciente
and to_char(dt_laudo,'mm/yyyy') = to_char(:CAMPO_DT_LAUDO,'mm/yyyy')
group by laudo_apac.cd_paciente
         ,paciente.nm_paciente
         ,dt_laudo
         ,cd_tip_ate
         ,cd_procedimento
order by paciente.nm_paciente
