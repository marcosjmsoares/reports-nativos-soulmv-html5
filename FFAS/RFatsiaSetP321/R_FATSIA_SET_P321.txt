--R_FATSIA_SET_P321
SELECT * FROM (
 Select NVL(V_FFAS_Fat_Global_p321.cd_setor_produziu,V_FFAS_Fat_Global_p321.cd_setor) CD_SETOR
           , setor.NM_SETOR
           , CD_GRUPO_PROCEDIMENTO
           , DS_GRUPO_PROCEDIMENTO
           , TP_UNIDADE
           , CD_SUB_GRUPO_PROCEDIMENTO
           , DS_SUB_GRUPO_PROCEDIMENTO
           , CD_ORGANIZA_GRUPO_PROCEDIMENTO
           , DS_ORGANIZA_GRUPO_PROCEDIMENTO
           , CD_PROCEDIMENTO
           , DS_PROCEDIMENTO
           , Sum(QT_LANCADA)                                          QT_LANCADA
           , Sum(VL_SERVICO_AMBULATORIAL)              VL_SERVICO_AMBULATORIAL
           , Sum(VL_SERVICO_ANESTESIA)                      VL_SERVICO_ANESTESIA
           , Sum(P_VL_SERVICO_PROFISSIONAL_AMB)     VL_SERVICO_PROFISSIONAL
           , Sum(VL_TOTAL_AMBULATORIAL)                   VL_TOTAL_AMBULATORIAL
           , VL_UNITARIO
    From Dbamv.V_FFAS_Fat_Global_p321
          , dbamv.setor
 Where 1 = 1
     And nvl(V_FFAS_Fat_Global_p321.cd_setor_produziu,V_FFAS_Fat_Global_p321.cd_setor) = setor.cd_setor(+)
Group By
            NVL(V_FFAS_Fat_Global_p321.cd_setor_produziu,V_FFAS_Fat_Global_p321.cd_setor)
           , setor.NM_SETOR
           , CD_GRUPO_PROCEDIMENTO
           , DS_GRUPO_PROCEDIMENTO
           , CD_SUB_GRUPO_PROCEDIMENTO
           , TP_UNIDADE
           , DS_SUB_GRUPO_PROCEDIMENTO
           , CD_ORGANIZA_GRUPO_PROCEDIMENTO
           , DS_ORGANIZA_GRUPO_PROCEDIMENTO
           , CD_PROCEDIMENTO
           , DS_PROCEDIMENTO
          , VL_UNITARIO
Order By
            NVL(V_FFAS_Fat_Global_p321.cd_setor_produziu,V_FFAS_Fat_Global_p321.cd_setor)
          , setor.NM_SETOR
          , CD_GRUPO_PROCEDIMENTO
          , CD_SUB_GRUPO_PROCEDIMENTO
          , CD_ORGANIZA_GRUPO_PROCEDIMENTO
          , TP_UNIDADE
          :CAMPO_CF_ORDER_BY
)
ORDER BY CD_SETOR,NM_SETOR,
CD_GRUPO_PROCEDIMENTO,DS_GRUPO_PROCEDIMENTO,
TP_UNIDADE,
CD_SUB_GRUPO_PROCEDIMENTO,DS_SUB_GRUPO_PROCEDIMENTO,
CD_ORGANIZA_GRUPO_PROCEDIMENTO,DS_ORGANIZA_GRUPO_PROCEDIMENTO
