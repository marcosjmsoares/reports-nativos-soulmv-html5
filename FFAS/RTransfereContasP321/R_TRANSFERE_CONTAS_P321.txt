--R_TRANSFERE_CONTAS_P321
SELECT * FROM (
select  TRANSFERE_CONTAS.CD_FAT_SIA_ANTERIOR
     , TRANSFERE_CONTAS.CD_FAT_SIA_NOVA
     , TRANSFERE_CONTAS.DT_TRANSFERENCIA
     , TRANSFERE_CONTAS.NM_USUARIO
     , TRANSFERE_CONTAS.DS_OBSERVACAO
     , EVE_SIASUS.cd_atendimento
    , EVE_SIASUS.cd_procedimento
     , PACIENTE.nm_paciente
     , FAT_SIA_NOVA.DS_FAT_SIA DS_FAT_SIA_NOVA
     , FAT_SIA_ANTIGA.DS_FAT_SIA DS_FAT_SIA_ANTIGA
from dbamv.TRANSFERE_CONTAS TRANSFERE_CONTAS
     , dbamv.EVE_SIASUS EVE_SIASUS
     , dbamv.PACIENTE PACIENTE
     , dbamv.FAT_SIA FAT_SIA_NOVA
     , dbamv.FAT_SIA FAT_SIA_ANTIGA
where TRANSFERE_CONTAS.cd_eventos = EVE_SIASUS.cd_eventos
   and PACIENTE.CD_PACIENTE(+) = EVE_SIASUS.CD_PACIENTE
   and FAT_SIA_NOVA.CD_FAT_SIA    = TRANSFERE_CONTAS.CD_FAT_SIA_NOVA
   and FAT_SIA_ANTIGA.CD_FAT_SIA    = TRANSFERE_CONTAS.CD_FAT_SIA_ANTERIOR
   and EVE_SIASUS.CD_MULTI_EMPRESA = dbamv.pkg_mv2000.le_empresa
order by PACIENTE.nm_paciente
)
ORDER BY CD_FAT_SIA_NOVA,CD_FAT_SIA_ANTERIOR,DS_FAT_SIA_NOVA,DS_FAT_SIA_ANTIGA
