--R_LANC_PLAN
SELECT * FROM (
SELECT  IT_MARCACAO.CD_MARCACAO
     ,  PRESTADOR.CD_PRESTADOR
     ,  PRESTADOR.NM_PRESTADOR
     ,  EVE_SIASUS.DT_EVE_SIASUS
     ,  PACIENTE.CD_PACIENTE
     ,  PACIENTE.NM_PACIENTE
     ,  PROC.CD_PROCEDIMENTO                                                 
     ,  PROC.DS_PROCEDIMENTO                                                 
     ,  EVE_SIASUS.QT_LANCADA
FROM    
        DBAMV.PROCEDIMENTO_SUS  PROC                                
     ,  DBAMV.EVE_SIASUS
     ,  DBAMV.IT_MARCACAO
     ,  DBAMV.PRESTADOR
     ,  DBAMV.PACIENTE
WHERE   PRESTADOR.CD_PRESTADOR  LIKE        :CAMPO_PRESTADOR
  AND   EVE_SIASUS.DT_EVE_SIASUS BETWEEN    :CAMPO_DTINIC  AND  
  AND   EVE_SIASUS.CD_IT_MARCACAO                   = IT_MARCACAO.CD_IT_MARCACAO
  AND   EVE_SIASUS.CD_PRESTADOR                      = PRESTADOR.CD_PRESTADOR
  AND   EVE_SIASUS.CD_PACIENTE                           = PACIENTE.CD_PACIENTE(+)
  AND   EVE_SIASUS.CD_PROCEDIMENTO                = PROC.CD_PROCEDIMENTO                   
UNION ALL
SELECT  IT_MARCACAO.CD_MARCACAO
     ,  PRESTADOR.CD_PRESTADOR
     ,  PRESTADOR.NM_PRESTADOR
     ,  EVE_SIASUS.DT_EVE_SIASUS
     ,  PACIENTE.CD_PACIENTE
     ,  PACIENTE.NM_PACIENTE
     ,  PROC.CD_PROCEDIMENTO                                                    
     ,  PROC.DS_PROCEDIMENTO                                                    
     ,  EVE_SIASUS.QT_LANCADA
FROM    
        DBAMV.PROCEDIMENTO_SUS  PROC                                  
     ,  DBAMV.EVE_SIASUS
     ,  DBAMV.IT_MARCACAO
     ,  DBAMV.PRESTADOR
     ,  DBAMV.PACIENTE
WHERE   PRESTADOR.CD_PRESTADOR  LIKE             :CAMPO_PRESTADOR
  AND   EVE_SIASUS.DT_EVE_SIASUS  BETWEEN       :CAMPO_DTINIC AND 
  AND   EVE_SIASUS.CD_IT_MARCACAO                       = IT_MARCACAO.CD_IT_MARCACAO
  AND   EVE_SIASUS.CD_PRESTADOR_PODE_TER     = PRESTADOR.CD_PRESTADOR
  AND   EVE_SIASUS.CD_PACIENTE                               = PACIENTE.CD_PACIENTE(+)
  AND   EVE_SIASUS.CD_PROCEDIMENTO                    = PROC.CD_PROCEDIMENTO                        
ORDER BY CD_MARCACAO,
         CD_PACIENTE,
         DT_EVE_SIASUS
)
ORDER BY CD_MARCACAO,CD_PRESTADOR,NM_PRESTADOR,DT_EVE_SIASUS,
CD_PACIENTE,NM_PACIENTE
