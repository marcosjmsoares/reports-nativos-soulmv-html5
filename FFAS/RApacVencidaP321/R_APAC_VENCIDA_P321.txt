--R_APAC_VENCIDA_P321
SELECT * FROM (
SELECT APAC.CD_APAC CD_APAC
        ,APAC.NR_APAC NR_APAC
       ,APAC.CD_PACIENTE CD_PACIENTE
       ,PACIENTE.NM_PACIENTE NM_PACIENTE
       ,APAC.DT_INICIAL DT_INICIAL
       ,APAC.DT_FINAL DT_FINAL
       ,EVE_SIASUS.CD_PROCEDIMENTO CD_SSM
       ,PROCEDIMENTO.DS_PROCEDIMENTO DS_TITULO
       ,APAC.CD_MOTIVO_COBRANCA_P321 CD_MOTIVO_COBRANCA
       ,MOTIVO_COBRANCA.DS_MOTIVO DS_MOTIVO_COBRANCA
       ,EVE_SIASUS.QT_LANCADA QT_LANCADA
  FROM DBAMV.APAC APAC
       ,DBAMV.PACIENTE PACIENTE
       ,DBAMV.EVE_SIASUS EVE_SIASUS
       ,DBAMV.PROCEDIMENTO_SUS PROCEDIMENTO
       ,DBAMV.MOTIVO_COBRANCA_SUS  MOTIVO_COBRANCA
       ,DBAMV.FAT_SIA FAT_SIA
WHERE APAC.CD_FAT_SIA = :CAMPO_FAT_SIA
  AND APAC.CD_FAT_SIA = FAT_SIA.CD_FAT_SIA
  AND TO_CHAR(NVL(APAC.DT_OCORRENCIA,APAC.DT_FINAL),'MM/YYYY') = TO_CHAR(FAT_SIA.DT_PERIODO_FINAL,'MM/YYYY')
  AND APAC.CD_PACIENTE = PACIENTE.CD_PACIENTE
  AND EVE_SIASUS.CD_PROCEDIMENTO = PROCEDIMENTO.CD_PROCEDIMENTO
  AND EVE_SIASUS.SN_APAC_PRINCIPAL = 'S'
  AND EVE_SIASUS.CD_APAC = APAC.CD_APAC
  AND APAC.CD_MOTIVO_COBRANCA_P321 = MOTIVO_COBRANCA.CD_MOTIVO ) VENCIDA
