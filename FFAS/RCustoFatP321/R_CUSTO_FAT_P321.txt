--R_CUSTO_FAT_P321
select atendime.cd_atendimento atend_sint
     , apac.nr_apac nr_apac_sint
     , paciente.cd_paciente cd_paciente_sint
     , paciente.nm_paciente nm_paciente_sint
     , prestador.nm_prestador nm_prestador_sint
     , sum(nvl(eve_siasus.vl_total_ambulatorial,0)) receita_sint
	 , Nvl(Decode( custo.tp_mvto_estoque, 'P', custo.Custo_Sintetico, custo.Custo_Sintetico * -1),0) Custo_Sintetico
 from  dbamv.atendime
     , dbamv.paciente
     , dbamv.eve_siasus
     , dbamv.apac
     , dbamv.fat_sia
     , dbamv.prestador
     , dbamv.procedimento_sus proc_sus
     , dbamv.procedimento_sus_detalhe proc_sus_detalhe
     , (Select mvto_estoque.tp_mvto_estoque
         , mvto_estoque.cd_atendimento
         , sum(trunc(nvl(dbamv.verif_vl_custo_medio( itmvto_estoque.cd_produto 
				                                             , mvto_estoque.dt_mvto_estoque 
				                                             , 'H' 
				                                             , produto.vl_custo_medio 
				                                             , Mvto_Estoque.Hr_Mvto_Estoque ) * uni_pro.vl_fator,0),5) * 
				    nvl(itmvto_estoque.qt_movimentacao,0)) Custo_Sintetico   
          From dbamv.mvto_estoque 
		     , dbamv.itmvto_estoque       
             , dbamv.produto       
             , dbamv.uni_pro 
             , dbamv.eve_siasus
		 Where mvto_estoque.tp_mvto_estoque in ('P', 'C') 
		   and mvto_estoque.cd_mvto_estoque = itmvto_estoque.cd_mvto_estoque 
		   and itmvto_estoque.cd_produto    = produto.cd_produto 
		   and itmvto_estoque.cd_uni_pro    = uni_pro.cd_uni_pro 
		   AND eve_siasus.cd_atendimento    =  mvto_estoque.cd_atendimento
         group by mvto_estoque.tp_mvto_estoque ,
                  mvto_estoque.cd_atendimento ) custo 
 where eve_siasus.cd_fat_sia       = fat_sia.cd_fat_sia
   and eve_siasus.cd_atendimento   = atendime.cd_atendimento
   and atendime.cd_atendimento_pai is null
   and eve_siasus.cd_paciente      = paciente.cd_paciente
   and eve_siasus.cd_prestador     = prestador.cd_prestador
   and eve_siasus.cd_apac          = apac.cd_apac
   AND eve_siasus.cd_atendimento   = custo.cd_atendimento(+)
   and apac.CD_MOTIVO_COBRANCA_p321 not in( select CD_MOTIVO
                                              from dbamv.MOTIVO_COBRANCA_sus
                                             where tp_modalidade in('A','O'))
  and eve_siasus.cd_procedimento = proc_sus.cd_procedimento
  and proc_sus.cd_procedimento = proc_sus_detalhe.cd_procedimento
group by atendime.cd_atendimento
     , apac.nr_apac
     , paciente.cd_paciente
     , paciente.nm_paciente
     , prestador.nm_prestador
     , eve_siasus.cd_fat_sia
	 , Decode( custo.tp_mvto_estoque, 'P', custo.Custo_Sintetico, custo.Custo_Sintetico * -1)
order by 4

--R_CUSTO_FAT_P321_R_4_Q_ATEND_MOV_ESTOQUE
SELECT * FROM (
Select mvto_estoque.CD_ATENDIMENTO atend_estoque_anal
     , mvto_estoque.HR_MVTO_ESTOQUE
     , itmvto_estoque.CD_PRODUTO
     , produto.DS_PRODUTO
     , uni_pro.DS_UNIDADE
     , sum(nvl(itmvto_estoque.QT_MOVIMENTACAO,0)) Qtd_Anal
     , trunc(sum(nvl(dbamv.verif_vl_custo_medio( itmvto_estoque.CD_PRODUTO
                                               , mvto_estoque.DT_MVTO_ESTOQUE
                                               , 'H'
                                               , produto.VL_CUSTO_MEDIO
                                               , Mvto_Estoque.HR_MVTO_ESTOQUE ) * uni_pro.VL_FATOR,0)),5)  Preco_medio
  From dbamv.MVTO_ESTOQUE
  	 , dbamv.ITMVTO_ESTOQUE
     , dbamv.PRODUTO
     , dbamv.UNI_PRO
 Where mvto_estoque.TP_MVTO_ESTOQUE   = 'P'
   and mvto_estoque.CD_MVTO_ESTOQUE   = itmvto_estoque.CD_MVTO_ESTOQUE
   and itmvto_estoque.CD_PRODUTO      = produto.CD_PRODUTO
   and itmvto_estoque.CD_UNI_PRO      = uni_pro.CD_UNI_PRO
 group by mvto_estoque.CD_ATENDIMENTO
        , mvto_estoque.HR_MVTO_ESTOQUE
        , itmvto_estoque.CD_PRODUTO
        , produto.DS_PRODUTO
        , uni_pro.DS_UNIDADE
        , itmvto_estoque.QT_MOVIMENTACAO
 union all
Select mvto_estoque.CD_ATENDIMENTO atend_estoque_anal
     , mvto_estoque.HR_MVTO_ESTOQUE
     , itmvto_estoque.CD_PRODUTO
     , produto.DS_PRODUTO
     , uni_pro.DS_UNIDADE
     , sum(nvl(itmvto_estoque.QT_MOVIMENTACAO,0) * -1) Qtd_Anal
     , trunc(sum(nvl(dbamv.verif_vl_custo_medio( itmvto_estoque.CD_PRODUTO
                                               , mvto_estoque.DT_MVTO_ESTOQUE
                                               , 'H'
                                               , produto.VL_CUSTO_MEDIO
                                               , Mvto_Estoque.HR_MVTO_ESTOQUE ) * uni_pro.VL_FATOR,0)),5)  Preco_medio
  From dbamv.MVTO_ESTOQUE
  	 , dbamv.ITMVTO_ESTOQUE
     , dbamv.PRODUTO
     , dbamv.UNI_PRO
 Where mvto_estoque.TP_MVTO_ESTOQUE   = 'C'
   and mvto_estoque.CD_MVTO_ESTOQUE   = itmvto_estoque.CD_MVTO_ESTOQUE
   and itmvto_estoque.CD_PRODUTO      = produto.CD_PRODUTO
   and itmvto_estoque.CD_UNI_PRO      = uni_pro.CD_UNI_PRO
 group by mvto_estoque.CD_ATENDIMENTO
        , mvto_estoque.HR_MVTO_ESTOQUE
        , itmvto_estoque.CD_PRODUTO
        , produto.DS_PRODUTO
        , uni_pro.DS_UNIDADE
        , itmvto_estoque.QT_MOVIMENTACAO
)
WHERE ATEND_ESTOQUE_ANAL = :CAMPO_ATEND_FIL_ANAL
