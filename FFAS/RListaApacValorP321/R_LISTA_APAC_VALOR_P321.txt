--R_LISTA_APAC_VALOR_P321
SELECT * FROM (
SELECT APAC.NR_APAC
        , PACIENTE.NM_PACIENTE
        , APAC.DT_INICIAL
        , APAC.DT_FINAL
        , EVE_SIASUS.CD_PROCEDIMENTO
        , PROCEDIMENTO_SUS.DS_PROCEDIMENTO
        , APAC.CD_MOTIVO_COBRANCA_P321
        , MOTIVO_COBRANCA_SUS.DS_MOTIVO
        , EVE_SIASUS.QT_LANCADA
        ,  sum(EVE_SIASUS.VL_TOTAL_AMBULATORIAL) VL_TOTAL_AMBULATORIAL
        , APAC.CD_APAC
        , 1 QTD_APAC
        , EVE_SIASUS.DT_EVE_SIASUS
  FROM DBAMV.APAC
        , DBAMV.PACIENTE
        , DBAMV.EVE_SIASUS
        , DBAMV.MOTIVO_COBRANCA_SUS
        , DBAMV.FAT_SIA
        , DBAMV.PROCEDIMENTO_SUS
        , DBAMV.PROCEDIMENTO_SUS_COMPLEXIDADE
 WHERE EVE_SIASUS.CD_APAC = APAC.CD_APAC
        AND EVE_SIASUS.CD_PACIENTE = PACIENTE.CD_PACIENTE
        AND PROCEDIMENTO_SUS_COMPLEXIDADE.CD_PROCEDIMENTO = EVE_SIASUS.CD_PROCEDIMENTO
        AND PROCEDIMENTO_SUS_COMPLEXIDADE.CD_PROCEDIMENTO = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
        AND EVE_SIASUS.CD_PROCEDIMENTO = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
       AND EVE_SIASUS.SN_APAC_PRINCIPAL = 'S'
        AND APAC.CD_MOTIVO_COBRANCA_P321 = MOTIVO_COBRANCA_SUS.CD_MOTIVO
        AND APAC.CD_FAT_SIA = FAT_SIA.CD_FAT_SIA
group by
APAC.NR_APAC
        , PACIENTE.NM_PACIENTE
        , APAC.DT_INICIAL
        , APAC.DT_FINAL
        , EVE_SIASUS.CD_PROCEDIMENTO
        , PROCEDIMENTO_SUS.DS_PROCEDIMENTO
        , APAC.CD_MOTIVO_COBRANCA_P321
        , MOTIVO_COBRANCA_SUS.DS_MOTIVO
        , EVE_SIASUS.QT_LANCADA
        , APAC.CD_APAC
        , 1
        , EVE_SIASUS.DT_EVE_SIASUS
)
