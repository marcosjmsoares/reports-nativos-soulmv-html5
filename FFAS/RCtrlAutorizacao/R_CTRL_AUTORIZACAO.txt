--R_CTRL_AUTORIZACAO
SELECT EVE_SIASUS.CD_FAT_SIA        CD_FAT_SIA
      ,FAT_SIA.DT_PERIODO_FINAL			DT_PERIODO_FINAL
      ,EVE_SIASUS.CD_ATENDIMENTO		CD_ATENDIMENTO
			,EVE_SIASUS.CD_PACIENTE       CD_PACIENTE
			,PACIENTE.NM_PACIENTE         NM_PACIENTE
			,EVE_SIASUS.CD_PROCEDIMENTO   CD_PROCEDIMENTO
			,PROCEDIMENTO_SUS.DS_PROCEDIMENTO DS_PROCEDIMENTO
			,EVE_SIASUS.QT_LANCADA				QT_LANCADA
			,EVE_SIASUS.NR_AUTORIZACAO_GESTOR NR_AUTORIZACAO_GESTOR
			,EVE_SIASUS.DT_EVE_SIASUS  DT_EVE_SIASUS
			,EVE_SIASUS.CD_UPS      	 CD_UPS
			,(SELECT VL_TOTAL_AMBULATORIAL  FROM DBAMV.PROCEDIMENTO_SUS_VALOR_CONTRAT P_SUS_VALOR
            WHERE TO_CHAR(P_SUS_VALOR.DT_VIGENCIA,'MM/YYYY') IN (SELECT TO_CHAR(MAX(DT_VIGENCIA),'MM/YYYY') DT_VIGENCIA
                                                									 FROM DBAMV.PROCEDIMENTO_SUS_VALOR_CONTRAT
                                               										WHERE DT_VIGENCIA                           <= NVL(FAT_SIA.DT_PERIODO_FINAL, SYSDATE)
                                                 										AND DBAMV.PROCEDIMENTO_SUS_VALOR_CONTRAT.CD_PROCEDIMENTO = EVE_SIASUS.CD_PROCEDIMENTO )
              AND P_SUS_VALOR.CD_PROCEDIMENTO  = EVE_SIASUS.CD_PROCEDIMENTO
							AND P_SUS_VALOR.CD_MULTI_EMPRESA = NVL(DBAMV.PKG_MV2000.LE_EMPRESA,1)
							AND P_SUS_VALOR.CD_UPS           = EVE_SIASUS.CD_UPS ) VL_TOTAL_AMBULATORIAL_CONTRAT
			,(SELECT VL_TOTAL_AMBULATORIAL  FROM DBAMV.PROCEDIMENTO_SUS_VALOR P_SUS_VALOR
         WHERE P_SUS_VALOR.DT_VIGENCIA IN (SELECT MAX (DT_VIGENCIA) DT_VIGENCIA
                                             FROM DBAMV.PROCEDIMENTO_SUS_VALOR
                                            WHERE DT_VIGENCIA                           <= NVL(FAT_SIA.DT_PERIODO_FINAL, SYSDATE)
                                              AND PROCEDIMENTO_SUS_VALOR.CD_PROCEDIMENTO = EVE_SIASUS.CD_PROCEDIMENTO )
          AND P_SUS_VALOR.CD_PROCEDIMENTO = EVE_SIASUS.CD_PROCEDIMENTO ) VL_TOTAL_AMBULATORIAL_SUS
FROM DBAMV.EVE_SIASUS
    ,DBAMV.PACIENTE
		,DBAMV.PROCEDIMENTO_SUS
		,DBAMV.PROCEDIMENTO_DETALHE_VIGENCIA   DETALHE
    ,DBAMV.FAT_SIA
WHERE EVE_SIASUS.CD_PACIENTE 								= PACIENTE.CD_PACIENTE	 (+)
  AND EVE_SIASUS.CD_PROCEDIMENTO 							= PROCEDIMENTO_SUS.CD_PROCEDIMENTO
  AND EVE_SIASUS.CD_PROCEDIMENTO 							= DETALHE.CD_PROCEDIMENTO
  AND TRUNC(EVE_SIASUS.DT_EVE_SIASUS) BETWEEN DETALHE.DT_VALIDADE_INICIAL AND NVL(DETALHE.DT_VALIDADE_FINAL, TRUNC(EVE_SIASUS.DT_EVE_SIASUS))
  AND DETALHE.SN_OBRIGA_BPAI = 'S'
  AND EVE_SIASUS.CD_FAT_SIA  = FAT_SIA.CD_FAT_SIA
ORDER BY EVE_SIASUS.CD_FAT_SIA,EVE_SIASUS.CD_ATENDIMENTO,EVE_SIASUS.CD_PACIENTE
