SELECT * FROM (
SELECT  VISAO.SN_APAC_PRINCIPAL
       ,VISAO.CD_APAC
       ,VISAO.CD_ORI_ATE
       ,VISAO.DS_ORI_ATE
       ,VISAO.NR_APAC
       ,VISAO.CD_PACIENTE
       ,VISAO.NM_PACIENTE
       ,VISAO.DT_INICIAL
       ,VISAO.DT_FINAL
       ,VISAO.NR_CPF
       ,VISAO.TP_APAC
       ,VISAO.NM_PRESTADOR
       ,VISAO.cd_procedimento
       ,VISAO.QT_LANCADA
       ,VISAO.VL_GERAL
       ,VISAO.VL_TOTAL
FROM (
SELECT  EVE_SIASUS.SN_APAC_PRINCIPAL
       ,APAC.CD_APAC
       ,APAC.CD_ORI_ATE
       ,ORI_ATE.DS_ORI_ATE
       ,APAC.NR_APAC
       ,APAC.CD_PACIENTE
       ,PACIENTE.NM_PACIENTE
       ,APAC.DT_INICIAL
       ,APAC.DT_FINAL
       ,PACIENTE.NR_CPF
       ,DECODE(APAC.TP_APAC,'1','Inicial','2','Continuação','3','Única','Única') TP_APAC
       ,PRESTADOR.NM_PRESTADOR
       ,EVE_SIASUS.cd_procedimento
       ,EVE_SIASUS.QT_LANCADA
       ,EVE_SIASUS.vl_servico_ambulatorial                          VL_GERAL
       ,EVE_SIASUS.vl_total_ambulatorial                            VL_TOTAL
 FROM dbamv.APAC       ,
      dbamv.ORI_ATE    ,
      dbamv.PACIENTE   ,
      dbamv.EVE_SIASUS ,
      dbamv.PRESTADOR ,
      dbamv.FAT_SIA
WHERE APAC.CD_FAT_SIA               =  :CAMPO_FAT_SIA
  AND APAC.CD_FAT_SIA               = EVE_SIASUS.CD_FAT_SIA
  AND APAC.CD_ORI_ATE               = ORI_ATE.CD_ORI_ATE
  AND APAC.CD_ORI_ATE               = EVE_SIASUS.CD_ORI_ATE
  AND APAC.CD_PACIENTE              = PACIENTE.CD_PACIENTE
  AND APAC.CD_APAC                  = EVE_SIASUS.CD_APAC
  AND APAC.CD_PRESTADOR             = PRESTADOR.CD_PRESTADOR
  AND EVE_SIASUS.CD_FAT_SIA         =  FAT_SIA.CD_FAT_SIA
  AND EVE_SIASUS.SN_APAC_PRINCIPAL  = 'S'  AND EVE_SIASUS.QT_LANCADA = 0
GROUP BY EVE_SIASUS.SN_APAC_PRINCIPAL
        ,APAC.CD_APAC
        ,APAC.CD_ORI_ATE
        ,ORI_ATE.DS_ORI_ATE
        ,APAC.NR_APAC
        ,APAC.CD_PACIENTE
        ,PACIENTE.NM_PACIENTE
        ,APAC.DT_INICIAL
        ,APAC.DT_FINAL
        ,PACIENTE.NR_CPF
        ,DECODE(APAC.TP_APAC,'1','Inicial','2','Continuação','3','Única','Única')
        ,PRESTADOR.NM_PRESTADOR
        ,EVE_SIASUS.cd_procedimento
        ,EVE_SIASUS.QT_LANCADA
        ,EVE_SIASUS.vl_servico_ambulatorial
        ,EVE_SIASUS.vl_total_ambulatorial
UNION ALL
SELECT  EVE_SIASUS.SN_APAC_PRINCIPAL
       ,APAC.CD_APAC
       ,APAC.CD_ORI_ATE
       ,ORI_ATE.DS_ORI_ATE
       ,APAC.NR_APAC
       ,APAC.CD_PACIENTE
       ,PACIENTE.NM_PACIENTE
       ,APAC.DT_INICIAL
       ,APAC.DT_FINAL
       ,PACIENTE.NR_CPF
       ,DECODE(APAC.TP_APAC,'1','Inicial','2','Continuação','3','Única','Única') TP_APAC
       ,PRESTADOR.NM_PRESTADOR
       ,EVE_SIASUS.cd_procedimento
       ,EVE_SIASUS.QT_LANCADA
       ,EVE_SIASUS.vl_servico_ambulatorial                          VL_GERAL
       ,EVE_SIASUS.vl_total_ambulatorial                            VL_TOTAL
 FROM dbamv.APAC       ,
      dbamv.ORI_ATE    ,
      dbamv.PACIENTE   ,
      dbamv.EVE_SIASUS ,
      dbamv.PRESTADOR ,
      dbamv.FAT_SIA
WHERE APAC.CD_FAT_SIA           =  :CAMPO_FAT_SIA
  AND APAC.CD_FAT_SIA           = EVE_SIASUS.CD_FAT_SIA
  AND APAC.CD_ORI_ATE           = ORI_ATE.CD_ORI_ATE
  AND APAC.CD_ORI_ATE           = EVE_SIASUS.CD_ORI_ATE
  AND APAC.CD_PACIENTE          = PACIENTE.CD_PACIENTE
  AND APAC.CD_APAC              = EVE_SIASUS.CD_APAC
  AND APAC.CD_PRESTADOR         = PRESTADOR.CD_PRESTADOR
  AND EVE_SIASUS.CD_FAT_SIA     =  FAT_SIA.CD_FAT_SIA
  AND EVE_SIASUS.QT_LANCADA     > 0
  AND EVE_SIASUS.CD_APAC  NOT IN(SELECT CD_APAC
                                   FROM DBAMV.EVE_SIASUS
                                  WHERE CD_FAT_SIA  = :CAMPO_FAT_SIA
                                    AND SN_APAC_PRINCIPAL = 'S'
                                    AND QT_LANCADA = 0 )
GROUP BY EVE_SIASUS.SN_APAC_PRINCIPAL
        ,APAC.CD_APAC
        ,APAC.CD_ORI_ATE
        ,ORI_ATE.DS_ORI_ATE
        ,APAC.NR_APAC
        ,APAC.CD_PACIENTE
        ,PACIENTE.NM_PACIENTE
        ,APAC.DT_INICIAL
        ,APAC.DT_FINAL
        ,PACIENTE.NR_CPF
        ,DECODE(APAC.TP_APAC,'1','Inicial','2','Continuação','3','Única','Única')
        ,PRESTADOR.NM_PRESTADOR
       ,EVE_SIASUS.cd_procedimento
       ,EVE_SIASUS.QT_LANCADA
       ,EVE_SIASUS.vl_servico_ambulatorial
       ,EVE_SIASUS.vl_total_ambulatorial
) VISAO
ORDER BY 7,1 DESC
)
ORDER BY DS_ORI_ATE,CD_ORI_ATE,
NM_PACIENTE,NR_APAC,DT_INICIAL,DT_FINAL,NR_CPF,TP_APAC,NM_PRESTADOR,CD_PROCEDIMENTO
