--R_CONSULTA_SEM_DECENDIO
select a.cd_atendimento  cd_atendimento,
       e.cd_procedimento cd_procedimento,
       p.ds_procedimento ds_procedimento,
       c.cd_paciente     cd_paciente,
       c.nm_paciente     nm_paciente,
       e.dt_eve_siasus  dt_eve_siasus
  from dbamv.atendime a,
       dbamv.eve_siasus e,
       dbamv.procedimento_sus p,
       dbamv.paciente c
 where a.cd_atendimento  = e.cd_atendimento
   and e.cd_procedimento = p.cd_procedimento
   and e.cd_paciente     = c.cd_paciente
   and substr(e.cd_procedimento,1,4) = '0301'
   and e.cd_decendio is null
   and e.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
   and exists  (  select 'x'
                    from dbamv.atendime a,
                         dbamv.eve_siasus v
                   where v.cd_itped_rx is not null
                     and v.cd_decendio is not null
                     and a.cd_atendimento = v.cd_atendimento
                     union all
                   select 'x'
                    from dbamv.atendime a,
                         dbamv.eve_siasus v
                   where v.cd_itped_lab is not null
                     and v.cd_decendio is not null
                     and a.cd_atendimento = v.cd_atendimento)
order by a.cd_atendimento
