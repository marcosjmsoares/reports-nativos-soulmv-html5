--R_LISTA_COPIA_SEM_ATEND
SELECT * FROM (
SELECT APAC.NR_APAC NR_APAC
       ,EVE_SIASUS.CD_PACIENTE CD_PACIENTE
       ,dbamv.pkg_sus_regra_paciente.F_NOME_PACIENTE(PACIENTE.CD_PACIENTE) NM_PACIENTE
       ,EVE_SIASUS.CD_SSM CD_SSM
       ,EVE_SIASUS.CD_PROCEDIMENTO CD_PROCEDIMENTO 
       ,PROCEDIMENTO_SUS.DS_PROCEDIMENTO 
       ,TAB_SIASUS.DS_TITULO
       ,EVE_SIASUS.QT_LANCADA
       ,EVE_SIASUS.SN_APAC_PRINCIPAL
  FROM DBAMV.APAC
       ,DBAMV.PACIENTE
       ,DBAMV.EVE_SIASUS
       ,DBAMV.TAB_SIASUS
       ,DBAMV.PROCEDIMENTO_SUS 
 WHERE EVE_SIASUS.CD_APAC IS NOT NULL
   AND EVE_SIASUS.CD_APAC = APAC.CD_APAC
   AND EVE_SIASUS.CD_FAT_SIA = APAC.CD_FAT_SIA
   AND EVE_SIASUS.CD_PACIENTE = PACIENTE.CD_PACIENTE
   AND EVE_SIASUS.CD_SSM = TAB_SIASUS.CD_SSM
   AND EVE_SIASUS.CD_PROCEDIMENTO = PROCEDIMENTO_SUS.CD_PROCEDIMENTO(+)  
   AND EVE_SIASUS.CD_ATENDIMENTO IS NULL
    GROUP BY  APAC.NR_APAC
       ,EVE_SIASUS.CD_PACIENTE
       ,PACIENTE.CD_PACIENTE
       ,EVE_SIASUS.CD_SSM
       ,EVE_SIASUS.CD_PROCEDIMENTO
       ,PROCEDIMENTO_SUS.DS_PROCEDIMENTO
       ,TAB_SIASUS.DS_TITULO
       ,EVE_SIASUS.QT_LANCADA
       ,EVE_SIASUS.SN_APAC_PRINCIPAL
   ORDER BY NR_APAC,SN_APAC_PRINCIPAL DESC
)
ORDER BY NR_APAC,CD_PACIENTE,NM_PACIENTE
