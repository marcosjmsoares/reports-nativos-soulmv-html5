--R_PROC_CA_ATENDIME_P321
SELECT * FROM (
SELECT V_FFAS_FAT_GLOBAL_P321.CD_CARATER_ATENDIMENTO
      ,CARATER_ATENDIMENTO.DS_CARATER_ATENDIMENTO
      ,V_FFAS_FAT_GLOBAL_P321.CD_PROCEDIMENTO
      ,PROCEDIMENTO_SUS.DS_PROCEDIMENTO
      ,SUM(V_FFAS_FAT_GLOBAL_P321.QT_LANCADA) QT_LANCADA
 FROM  DBAMV.V_FFAS_FAT_GLOBAL_P321
      ,DBAMV.PROCEDIMENTO_SUS  PROCEDIMENTO_SUS
      ,DBAMV.CARATER_ATENDIMENTO CARATER_ATENDIMENTO
      ,DBAMV.PROCEDIMENTO_SUS_COMPLEXIDADE PROCEDIMENTO_SUS_COMPLEXIDADE
WHERE V_FFAS_FAT_GLOBAL_P321.CD_FAT_SIA IS NOT NULL
  AND PROCEDIMENTO_SUS_COMPLEXIDADE.CD_PROCEDIMENTO = V_FFAS_FAT_GLOBAL_P321.CD_PROCEDIMENTO
  AND PROCEDIMENTO_SUS_COMPLEXIDADE.CD_PROCEDIMENTO = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
  AND V_FFAS_FAT_GLOBAL_P321.CD_CARATER_ATENDIMENTO = CARATER_ATENDIMENTO.CD_CARATER_ATENDIMENTO (+)
  AND V_FFAS_FAT_GLOBAL_P321.CD_PROCEDIMENTO = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
  AND V_FFAS_FAT_GLOBAL_P321.CD_CONVENIO LIKE :CAMPO_CONVENIO
  AND PROCEDIMENTO_SUS_COMPLEXIDADE.SN_ATIVO = 'S'
GROUP BY V_FFAS_FAT_GLOBAL_P321.CD_CARATER_ATENDIMENTO
        ,CARATER_ATENDIMENTO.DS_CARATER_ATENDIMENTO
        ,V_FFAS_FAT_GLOBAL_P321.CD_PROCEDIMENTO
        ,PROCEDIMENTO_SUS.DS_PROCEDIMENTO
)
ORDER BY CD_CARATER_ATENDIMENTO,DS_CARATER_ATENDIMENTO
