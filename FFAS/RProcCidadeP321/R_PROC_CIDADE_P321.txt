--R_PROC_CIDADE_P321
SELECT * FROM (
SELECT  EVE_SIASUS.CD_CIDADE
       ,NVL(CIDADE.NM_CIDADE,'NAO INFORMADA')   NM_CIDADE
       ,EVE_SIASUS.cd_procedimento              CD_PROCEDIMENTO
       ,PROCEDIMENTO_SUS.ds_procedimento        DS_TITULO
       ,EVE_SIASUS.QT_LANCADA
 FROM dbamv.APAC       ,
      dbamv.CIDADE     ,
      dbamv.EVE_SIASUS ,
      dbamv.PROCEDIMENTO_SUS
WHERE APAC.CD_FAT_SIA             = :CAMPO_FAT_SIA
  AND APAC.CD_FAT_SIA             = EVE_SIASUS.CD_FAT_SIA
  AND EVE_SIASUS.CD_CIDADE          = CIDADE.CD_CIDADE(+)
  AND EVE_SIASUS.cd_procedimento    = PROCEDIMENTO_SUS.cd_procedimento
  AND APAC.CD_APAC                  = EVE_SIASUS.CD_APAC
  AND EVE_SIASUS.SN_PACOTE	    = 'N'
GROUP BY EVE_SIASUS.CD_CIDADE
       ,CIDADE.NM_CIDADE
       ,EVE_SIASUS.cd_procedimento
       ,PROCEDIMENTO_SUS.ds_procedimento
       ,EVE_SIASUS.QT_LANCADA
ORDER BY EVE_SIASUS.CD_CIDADE)
ORDER BY NM_CIDADE,CD_CIDADE
