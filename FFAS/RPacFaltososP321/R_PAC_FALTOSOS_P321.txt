--R_PAC_FALTOSOS_P321
select max(cd_laudo) cd_laudo
         ,paciente.cd_paciente
         ,paciente.nm_paciente
         ,dt_laudo
         ,cd_tip_ate
         ,cd_procedimento
from   dbamv.laudo_sia_apac
      ,dbamv.paciente
where laudo_sia_apac.cd_paciente     = paciente.cd_paciente
and to_char(dt_laudo,'mm/yyyy') = to_char(:CAMPO_DT_LAUDO,'mm/yyyy')
group by paciente.cd_paciente
         ,paciente.nm_paciente
         ,dt_laudo
         ,cd_tip_ate
         ,cd_procedimento
order by paciente.nm_paciente
