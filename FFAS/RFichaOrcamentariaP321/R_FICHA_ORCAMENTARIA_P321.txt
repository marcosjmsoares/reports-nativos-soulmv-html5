SELECT ROWNUM LINHA
       ,DECODE( mod(ROWNUM, 2), 0, ROWNUM-1, ROWNUM) GRUPO_LINHA
       ,CD_FAT_SIA
       ,CD_PROCEDIMENTO
       ,QT_FISICO
       ,VL_ORCAMENTO
  FROM DBAMV.TETO_ORCAMENTARIO_PROCED_SUS
  WHERE CD_FAT_SIA = :CAMPO_FATURA
