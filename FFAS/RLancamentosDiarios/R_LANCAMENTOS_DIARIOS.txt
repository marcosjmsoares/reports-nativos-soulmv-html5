--R_LANCAMENTOS_DIARIOS
Select eve.nm_usuario       Codigo_Usuario
     , usu.nm_usuario       Nm_Usuario
     , eve.cd_atendimento
     , eve.cd_paciente
     , pac.nm_paciente
     , eve.cd_procedimento
     , proc.ds_procedimento
     , eve.qt_lancada
     , eve.cd_ups
     , eve.cd_decendio
  From dbamv.eve_siasus         eve
     , dbamv.procedimento_sus   proc
     , dbamv.paciente           pac
     , dbasgu.usuarios          usu
 Where eve.cd_procedimento = proc.cd_procedimento
   And eve.cd_paciente     = pac.cd_paciente(+)
   And eve.nm_usuario      = usu.cd_usuario
   And eve.cd_convenio = 2
   AND eve.cd_apac IS  NULL
   And eve.cd_multi_empresa = Dbamv.Pkg_Mv2000.Le_Empresa
 Order by 2, 5, 6
