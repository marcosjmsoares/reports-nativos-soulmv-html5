--R_REL_NOM_ANALIT_P321
SELECT * FROM (
 SELECT dbamv.pkg_sus_regra_paciente.F_NOME_PACIENTE(EVE_SIASUS.CD_PACIENTE) NM_PACIENTE
      , EVE_SIASUS.DT_EVE_SIASUS
      , EVE_SIASUS.CD_PACIENTE
      , EVE_SIASUS.CD_SSM
      , EVE_SIASUS.QT_LANCADA
      , EVE_SIASUS.CD_PROCEDIMENTO
      , EVE_SIASUS.CD_CBO_PRESTADOR
      , EVE_SIASUS.CD_ATENDIMENTO
      , EVE_SIASUS.vl_total_ambulatorial VL_GERAL
      , CBO.DS_CBOS
      , PACIENTE.DT_NASCIMENTO
      , CD_EVENTOS
   FROM dbamv.v_ffas_fat_global_p321 EVE_SIASUS     
      , DBAMV.CBO      
      , DBAMV.PACIENTE      
  WHERE EVE_SIASUS.CD_CBO_PRESTADOR        = CBO.CD_CBOS
    AND EVE_SIASUS.CD_PACIENTE          = PACIENTE.CD_PACIENTE(+)        
    AND EVE_SIASUS.CD_MULTI_EMPRESA     = :CAMPO_CD_MULTI_EMPRESA  
  group by EVE_SIASUS.CD_PACIENTE
      , EVE_SIASUS.DT_EVE_SIASUS
      , EVE_SIASUS.CD_PACIENTE
      , EVE_SIASUS.CD_SSM
      , EVE_SIASUS.QT_LANCADA
      , EVE_SIASUS.CD_PROCEDIMENTO      
      , EVE_SIASUS.CD_CBO_PRESTADOR
      , EVE_SIASUS.CD_ATENDIMENTO
      , EVE_SIASUS.QT_LANCADA
      , CD_EVENTOS
	  , EVE_SIASUS.vl_total_ambulatorial
      , CBO.DS_CBOS
      , PACIENTE.DT_NASCIMENTO) VISAO
