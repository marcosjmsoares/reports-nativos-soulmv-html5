--R_PREV_FAT_APAC_P321
SELECT * FROM (
SELECT SETOR.CD_SETOR
     , SETOR.NM_SETOR
     , grupo_proc.cd_grupo_procedimento
     , grupo_proc.ds_grupo_procedimento
     , sub_grupo_proc.cd_sub_grupo_procedimento
     , sub_grupo_proc.ds_sub_grupo_procedimento
     , PACIENTE.NM_PACIENTE
     , PACIENTE.CD_PACIENTE
     , APAC.NR_APAC
     , proc_sus.cd_procedimento
     , EVE_SIASUS.DT_EVE_SIASUS
     , sum(EVE_SIASUS.QT_LANCADA) qt_lancada
     , sum(EVE_SIASUS.vl_total_ambulatorial) VALOR
  FROM DBAMV.EVE_SIASUS
     , DBAMV.APAC
     , DBAMV.PACIENTE
     , DBAMV.SETOR
     , dbamv.procedimento_sus           proc_sus
     , dbamv.grupo_procedimentos        grupo_proc
     , dbamv.sub_grupo_procedimentos    sub_grupo_proc
 WHERE APAC.CD_APAC          = EVE_SIASUS.CD_APAC
   AND PACIENTE.CD_PACIENTE  = APAC.CD_PACIENTE
   AND SETOR.CD_SETOR        = EVE_SIASUS.CD_SETOR_PRODUZIU
   AND EVE_SIASUS.cd_procedimento = proc_sus.cd_procedimento
   AND proc_sus.cd_grupo_procedimento = grupo_proc.cd_grupo_procedimento
   and grupo_proc.cd_grupo_procedimento = sub_grupo_proc.cd_grupo_procedimento
   and proc_sus.cd_sub_grupo_procedimento = sub_grupo_proc.cd_sub_grupo_procedimento
   and eve_siasus.sn_pacote = 'N'
group by
        SETOR.CD_SETOR
     , SETOR.NM_SETOR
     , grupo_proc.cd_grupo_procedimento
     , grupo_proc.ds_grupo_procedimento
     , sub_grupo_proc.cd_sub_grupo_procedimento
     , sub_grupo_proc.ds_sub_grupo_procedimento
     , PACIENTE.NM_PACIENTE
     , PACIENTE.CD_PACIENTE
     , APAC.NR_APAC
     , proc_sus.cd_procedimento
     , EVE_SIASUS.DT_EVE_SIASUS
 ORDER BY SETOR.NM_SETOR ASC
     , PACIENTE.NM_PACIENTE ASC
)
ORDER BY NM_SETOR,CD_SETOR,
CD_GRUPO_PROCEDIMENTO,DS_GRUPO_PROCEDIMENTO,
DS_SUB_GRUPO_PROCEDIMENTO,CD_SUB_GRUPO_PROCEDIMENTO
