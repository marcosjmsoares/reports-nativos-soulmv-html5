--Nome módulo: FFAS - SISTEMA FATURAMENTO AMBULATORIO SUS

/*----------------------------------------------------------------------------*/
--R_PROC_BPA_ORIGEM_P321
select sysdate from dual


/*----------------------------------------------------------------------------*/
--R_PROC_BPA_ORIGEM_P321_R_1_Q_BPA_ORIGEM
SELECT EVE_SIASUS.CD_ORI_ATE       CD_ORI_ATE
             ,ORI_ATE.DS_ORI_ATE               DS_ORI_ATE
             ,EVE_SIASUS.CD_PRESTADOR CD_PRESTADOR
             ,PRESTADOR.NM_PRESTADOR  NM_PRESTADOR
             ,EVE_SIASUS.CD_PACIENTE  CD_PACIENTE
             ,DECODE(EVE_SIASUS.CD_PACIENTE, NULL, 'NÃO INFORMADO', PACIENTE.NM_PACIENTE) NM_PACIENTE
             ,NVL(ATENDIME.DT_ATENDIMENTO,EVE_SIASUS.DT_EVE_SIASUS ) DT_ATENDIMENTO
             ,EVE_SIASUS.CD_PROCEDIMENTO  CD_PROCEDIMENTO
            ,PROCEDIMENTO_SUS.DS_PROCEDIMENTO  DS_TITULO
           ,sum(EVE_SIASUS.QT_LANCADA) QT_LANCADA
 FROM dbamv.EVE_SIASUS ,
      dbamv.ORI_ATE    ,
      dbamv.PRESTADOR  ,
      dbamv.PACIENTE   ,
      dbamv.ATENDIME   ,
      dbamv.PROCEDIMENTO_SUS ,
      dbamv.procedimento_sus_complexidade
WHERE EVE_SIASUS.CD_FAT_SIA IS NOT NULL
  AND EVE_SIASUS.CD_ORI_ATE     = ORI_ATE.CD_ORI_ATE
  AND EVE_SIASUS.CD_PRESTADOR   = PRESTADOR.CD_PRESTADOR
  AND EVE_SIASUS.CD_PACIENTE                 = PACIENTE.CD_PACIENTE (+)
  AND EVE_SIASUS.CD_ATENDIMENTO        =  ATENDIME.CD_ATENDIMENTO (+)
  AND EVE_SIASUS.CD_PROCEDIMENTO     =  PROCEDIMENTO_SUS.CD_PROCEDIMENTO
AND EVE_SIASUS.CD_PROCEDIMENTO = procedimento_sus_complexidade.CD_PROCEDIMENTO
GROUP BY EVE_SIASUS.CD_ORI_ATE
       ,ORI_ATE.DS_ORI_ATE
       ,EVE_SIASUS.CD_PRESTADOR
       ,PRESTADOR.NM_PRESTADOR
       ,EVE_SIASUS.CD_PACIENTE
       ,PACIENTE.NM_PACIENTE
       ,NVL(ATENDIME.DT_ATENDIMENTO,EVE_SIASUS.DT_EVE_SIASUS )
       ,EVE_SIASUS.CD_PROCEDIMENTO
       ,PROCEDIMENTO_SUS.DS_PROCEDIMENTO
ORDER BY ORI_ATE.DS_ORI_ATE, PRESTADOR.NM_PRESTADOR, PACIENTE.NM_PACIENTE


/*----------------------------------------------------------------------------*/
--R_PROC_BPA_ORIGEM_P321_R_9_Q_BPA_PACIENTE
SELECT * FROM (
SELECT EVE_SIASUS.CD_ORI_ATE    CD_ORI_ATE
       ,ORI_ATE.DS_ORI_ATE      DS_ORI_ATE
       ,EVE_SIASUS.CD_PACIENTE  CD_PACIENTE
       ,DECODE(EVE_SIASUS.CD_PACIENTE, NULL, 'NÃO INFORMADO', PACIENTE.NM_PACIENTE) NM_PACIENTE  
       ,EVE_SIASUS.CD_PRESTADOR CD_PRESTADOR
       ,PRESTADOR.NM_PRESTADOR  NM_PRESTADOR
       ,NVL(ATENDIME.DT_ATENDIMENTO,EVE_SIASUS.DT_EVE_SIASUS ) DT_ATENDIMENTO
       ,EVE_SIASUS.CD_PROCEDIMENTO  CD_PROCEDIMENTO
       ,PROCEDIMENTO_SUS.DS_PROCEDIMENTO  DS_TITULO
       ,sum(EVE_SIASUS.QT_LANCADA) QT_LANCADA
 FROM dbamv.EVE_SIASUS ,
      dbamv.ORI_ATE    ,
      dbamv.PRESTADOR  ,
      dbamv.PACIENTE   ,
      dbamv.ATENDIME   ,
      dbamv.PROCEDIMENTO_SUS,
      dbamv.procedimento_sus_complexidade
WHERE EVE_SIASUS.CD_FAT_SIA IS NOT NULL
  AND EVE_SIASUS.CD_ORI_ATE     = ORI_ATE.CD_ORI_ATE
  AND EVE_SIASUS.CD_PRESTADOR   = PRESTADOR.CD_PRESTADOR
  AND EVE_SIASUS.CD_PACIENTE    = PACIENTE.CD_PACIENTE (+)
  AND EVE_SIASUS.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO (+)
  AND EVE_SIASUS.CD_PROCEDIMENTO         = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
AND EVE_SIASUS.CD_PROCEDIMENTO = procedimento_sus_complexidade.CD_PROCEDIMENTO
GROUP BY EVE_SIASUS.CD_PACIENTE
       ,EVE_SIASUS.CD_ORI_ATE
       ,EVE_SIASUS.CD_PRESTADOR
       ,PRESTADOR.NM_PRESTADOR
       ,ORI_ATE.DS_ORI_ATE
       ,PACIENTE.NM_PACIENTE
       ,NVL(ATENDIME.DT_ATENDIMENTO,EVE_SIASUS.DT_EVE_SIASUS )
       ,EVE_SIASUS.CD_PROCEDIMENTO
       ,PROCEDIMENTO_SUS.DS_PROCEDIMENTO
ORDER BY ORI_ATE.DS_ORI_ATE
                   ,PRESTADOR.NM_PRESTADOR
                   ,PACIENTE.NM_PACIENTE
)
ORDER BY CD_ORI_ATE,DS_ORI_ATE,
NM_PACIENTE,CD_PACIENTE
