--R_CRITICA_BPA_ANALITICO
SELECT EVE.CD_PROCEDIMENTO
     , DBAMV.PKG_SUS_REGRA_PROCEDIMENTO.F_DESCRICAO(EVE.CD_PROCEDIMENTO) DS_PROCEDIMENTO
     , EVE.CD_EVENTOS
     , EVE.CD_PACIENTE
     , DBAMV.PKG_SUS_REGRA_PACIENTE.F_NOME_PACIENTE(EVE.CD_PACIENTE) NM_PACIENTE
     , EVE.NR_IDADE
     , EVE.CD_CIDADE
     , DECODE(TP_SEXO,'M','MASCULINO','F','FEMININO','DESCONHECIDO') TP_SEXO
     , TO_CHAR(PAC.DT_NASCIMENTO,'DD/MM/YYYY') DT_NASCIMENTO
     , TO_CHAR(EVE.DT_EVE_SIASUS,'DD/MM/YYYY') DT_EVE_SIASUS
     , PAC.NR_CNS NR_CNS_PACIENTE
     , EVE.QT_LANCADA
     , LPAD(EVE.CD_CARATER_ATENDIMENTO,2,'0') ||' '||DBAMV.PKG_SUS_REGRA_CARATER_ATEND.F_DESCRICAO(EVE.CD_CARATER_ATENDIMENTO) DS_CARATER_ATENDIMENTO
     , EVE.CD_SERVICO_SUS||' / '||EVE.CD_CLASSIFICACAO_SUS CD_SERVICO_SUS
     , EVE.CD_CID_PRINCIPAL
     , DBAMV.PKG_SUS_REGRA_CID.F_DESCRICAO_CID(EVE.CD_CID_PRINCIPAL) DS_CID_PRINCIPAL
     , EVE.CD_SETOR||' '||DBAMV.PKG_SUS_REGRA_SETOR.F_DESCRICAO(EVE.CD_SETOR)DS_SETOR
     , EVE.CD_SETOR_PRODUZIU||' '|| DBAMV.PKG_SUS_REGRA_SETOR.F_DESCRICAO(EVE.CD_SETOR_PRODUZIU) DS_SETOR_PRODUZIU
     , EVE.CD_PRESTADOR||' '||DBAMV.PKG_SUS_REGRA_PRESTADOR.F_DESCRICAO(EVE.CD_PRESTADOR)NM_PRESTADOR
     , DBAMV.PKG_SUS_REGRA_PRESTADOR.F_CNS(EVE.CD_PRESTADOR) NR_CNS_PRESTADOR
     , EVE.CD_CBO_PRESTADOR
     , DBAMV.PKG_SUS_REGRA_CBO.F_DESCRICAO(EVE.CD_CBO_PRESTADOR) DS_CBO_PRESTADOR
     , TO_CHAR(EVE.DT_LANCAMENTO,'DD/MM/YYYY') DT_LANCAMENTO
     , EVE.NM_USUARIO
     , DECODE(PAC.TP_COR,'B','01 BRANCA','P','02 NEGRA','A','04 AMARELA','I','05 INDÍGENA','R','03 PARDA','99 SEM INFORMAÇÃO') TP_COR
     , DBAMV.PKG_SUS_REGRA_PACIENTE.F_DS_ETNIA(PAC.CD_ETNIA) DS_ETNIA
     , DBAMV.PKG_SUS_REGRA_PACIENTE.F_DS_CIDADANIA(PAC.CD_CIDADANIA) DS_CIDADANIA
     , DBAMV.PKG_SUS_REGRA_PACIENTE.F_DS_CIDADE(PAC.CD_CIDADE) DS_CIDADE
     , DBAMV.PKG_SUS_REGRA_PACIENTE.F_RETORNA_TELEFONE(PAC.CD_PACIENTE,'N') DS_TELEFONE
     , DBAMV.PKG_SUS_REGRA_PACIENTE.F_DS_LOGRADOURO(PAC.CD_PACIENTE) DS_LOGRADOURO
     , DBAMV.PKG_SUS_REGRA_PACIENTE.F_NR_CEP(PAC.CD_PACIENTE) NR_CEP
     , DBAMV.PKG_SUS_REGRA_PACIENTE.F_NR_ENDERECO(PAC.CD_PACIENTE) NR_ENDERECO
     , DBAMV.PKG_SUS_REGRA_PACIENTE.F_NM_BAIRRO(PAC.CD_PACIENTE) NM_BAIRRO
     , EVE.VL_TOTAL_AMBULATORIAL VL_TOTAL
     , NVL(TO_CHAR(EVE.CD_LAUDO_BPAI),'NÃO') CD_LAUDO_BPAI
     , DBAMV.PKG_FFAS_REGRA_CONTA_BPA.F_DS_ERRO_BPA(EVE.CD_EVENTOS) DS_CRITICA
     , DBAMV.PKG_FFAS_REGRA_CONTA_BPA.F_DS_TITULO_REL_CONS_01(EVE.CD_REMESSA) DS_TITULO_01
     , DBAMV.PKG_FFAS_REGRA_CONTA_BPA.F_ORIGEM_LANCAMENTO(EVE.CD_EVENTOS) DS_ORIGEM_LANCAMENTO
     , DBAMV.PKG_FFAS_REGRA_CONTA_BPA.F_DS_COLA_PROCEDIMENTO(EVE.CD_FAT_SIA,EVE.CD_PROCEDIMENTO,EVE.CD_PACIENTE,EVE.CD_PRESTADOR,EVE.DT_EVE_SIASUS) DS_DETAL_PROCEDIMENTO
  FROM DBAMV.EVE_SIASUS  EVE
     , DBAMV.CIDADE      CDA
     , DBAMV.PACIENTE    PAC
  WHERE CDA.CD_CIDADE(+)   = EVE.CD_CIDADE
    AND PAC.CD_PACIENTE = EVE.CD_PACIENTE
    AND EVE.CD_EVENTOS IN (SELECT CD_EVENTOS
                              FROM DBAMV.CRITICA_SIA)
ORDER BY EVE.CD_PROCEDIMENTO
