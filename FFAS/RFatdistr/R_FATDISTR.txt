--R_FATDISTR
SELECT  EVE_SIASUS.CD_FAT_SIA CD_FAT_SIA2
               ,FAT_SIA.DS_FAT_SIA DS_FAT_SIA
               ,FAT_SIA.CD_FAT_SIA CD_FAT_SIA
               ,fat_sia.dt_periodo_final
               ,PACIENTE.CD_DIS_SAN CD_DIS_SAN2
               ,DIS_SAN.CD_DIS_SAN CD_DIS_SAN
               ,DIS_SAN.NM_DIS_SAN NM_DIS_SAN
               ,EVE_SIASUS.cd_procedimento SSM
               ,PROCEDIMENTO_SUS.ds_procedimento DESCRIÇÃO
               ,PROCEDIMENTO_SUS.cd_procedimento CD_SSM2
               ,EVE_SIASUS.vl_servico_ambulatorial VL_GERAL
               ,TAB_SIASUS.TP_SSM
               ,SUM(EVE_SIASUS.QT_LANCADA) QT_LANCADA1
FROM dbamv.EVE_SIASUS EVE_SIASUS
              ,dbamv.FAT_SIA                     FAT_SIA
              ,dbamv.PACIENTE                  PACIENTE
              ,dbamv.DIS_SAN                    DIS_SAN
              ,dbamv.TAB_SIASUS             TAB_SIASUS
              ,dbamv.PROCEDIMENTO_SUS       PROCEDIMENTO_SUS
              ,dbamv.PROCEDIMENTO_SUS_COMPLEXIDADE       PROCEDIMENTO_SUS_COMPLEXIDADE
WHERE EVE_SIASUS.CD_FAT_SIA IS NOT NULL
AND       EVE_SIASUS.CD_FAT_SIA  =  FAT_SIA.CD_FAT_SIA
AND       EVE_SIASUS.CD_PROCEDIMENTO  =  PROCEDIMENTO_SUS.CD_PROCEDIMENTO
AND       PROCEDIMENTO_SUS.CD_PROCEDIMENTO  =  PROCEDIMENTO_SUS_COMPLEXIDADE.CD_PROCEDIMENTO
AND       EVE_SIASUS.CD_PACIENTE  =  PACIENTE.CD_PACIENTE
AND       PACIENTE.CD_DIS_SAN  =  DIS_SAN.CD_DIS_SAN
GROUP BY EVE_SIASUS.CD_FAT_SIA
                   ,FAT_SIA.DS_FAT_SIA
                   ,FAT_SIA.CD_FAT_SIA
                   ,fat_sia.dt_periodo_final
                   ,PACIENTE.CD_DIS_SAN
                   ,DIS_SAN.CD_DIS_SAN
                   ,DIS_SAN.NM_DIS_SAN
                   ,EVE_SIASUS.cd_procedimento
                   ,PROCEDIMENTO_SUS.ds_procedimento
                   ,PROCEDIMENTO_SUS.cd_procedimento
                   ,EVE_SIASUS.QT_LANCADA
                   ,EVE_SIASUS.vl_servico_ambulatorial
                   ,TAB_SIASUS.TP_SSM
ORDER BY EVE_SIASUS.CD_FAT_SIA
                  ,FAT_SIA.DS_FAT_SIA
                  ,FAT_SIA.CD_FAT_SIA
                  ,DIS_SAN.NM_DIS_SAN
                  ,:CAMPO_ORDEM
