--R_DESCARTES_APAC
SELECT  D.CD_DESCARTE||' - '|| M.DS_DESCARTE DS_DESCARTE
     , R.NR_APAC                            NR_APAC
     , A.CD_ATENDIMENTO                     CD_ATENDIMENTO
     , R.CD_PACIENTE                        CD_PACIENTE
     , C.NM_PACIENTE                        NM_PACIENTE
     , I.CD_PROCEDIMENTO                    CD_PROCEDIMENTO
     , I.QT_LANCADA                         QT_LANCAMENTO
     , P.DS_PROCEDIMENTO                    DS_PROCEDIMENTO
     , D.DS_OBSERVACAO                      DS_OBSERVACAO
     , D.NM_USUARIO                         NM_USUARIO
  FROM DBAMV.EVE_SIASUS              I
      , DBAMV.ITEM_DESCARTE_APAC     D
      , DBAMV.APAC                   R
      , DBAMV.REMESSA_APAC           E
      , DBAMV.FAT_SIA                F
      , DBAMV.PROCEDIMENTO_SUS       P
      , DBAMV.MOTIVO_DESCARTE_SUS    M
      , DBAMV.ATENDIME               A
      , DBAMV.PACIENTE               C
  WHERE TO_CHAR(e.dt_competencia_apresentacao,'MM/YYYY') = TO_CHAR(:CAMPO_COMPET,'MM/YYYY')
    AND I.CD_EVENTOS       = D.CD_EVENTOS
    AND I.CD_APAC          = R.CD_APAC
    AND R.CD_REMESSA       = E.CD_REMESSA
    AND F.CD_FAT_SIA       = E.CD_FAT_SIA
    AND P.CD_PROCEDIMENTO  = I.CD_PROCEDIMENTO
    AND M.CD_DESCARTE      = D.CD_DESCARTE
    AND I.CD_ATENDIMENTO   = A.CD_ATENDIMENTO(+)
    AND R.CD_PACIENTE      = C.CD_PACIENTE
    AND I.SN_PACOTE        = 'S'
ORDER BY DS_DESCARTE
