--R_CTR_FREQUENCIA_TIPO27
SELECT CTR_FREQUENCIA.CD_FREQUENCIA
       ,CTR_FREQUENCIA.CD_PACIENTE
       , dbamv.PKG_SUS_REGRA_PACIENTE.F_NOME_PACIENTE(CTR_FREQUENCIA.CD_PACIENTE) NM_PACIENTE
       ,TO_CHAR(PACIENTE.NR_CPF) NR_CPF
       ,PRESTADOR.NM_PRESTADOR
       ,CTR_FREQUENCIA.CD_TIP_ATE
       ,CTR_FREQUENCIA.DT_REFERENCIA
       ,CTR_FREQUENCIA.DT_INICIO_TRATAMENTO
       ,CTR_FREQUENCIA.DT_FINAL_TRATAMENTO
       ,CTR_FREQUENCIA.CD_PROG_DIALI_ATUAL
       ,CTR_FREQUENCIA.SN_HEMODIALISE
       ,CTR_FREQUENCIA.SN_CAPD
       ,CTR_FREQUENCIA.SN_DPA
       ,CTR_FREQUENCIA.SN_DPI
       ,CTR_FREQUENCIA.QT_KIT_CAPD
       ,CTR_FREQUENCIA.DT_FREQ_1
       ,CTR_FREQUENCIA.DT_FREQ_2
       ,CTR_FREQUENCIA.DT_FREQ_3
       ,CTR_FREQUENCIA.DT_FREQ_4
       ,CTR_FREQUENCIA.DT_FREQ_5
       ,CTR_FREQUENCIA.DT_FREQ_6
       ,CTR_FREQUENCIA.DT_FREQ_7
       ,CTR_FREQUENCIA.DT_FREQ_8
       ,CTR_FREQUENCIA.DT_FREQ_9
       ,CTR_FREQUENCIA.DT_FREQ_10
       ,CTR_FREQUENCIA.DT_FREQ_11
       ,CTR_FREQUENCIA.DT_FREQ_12
       ,CTR_FREQUENCIA.DT_FREQ_13
       ,CTR_FREQUENCIA.DT_FREQ_14
  FROM DBAMV.CTR_FREQUENCIA CTR_FREQUENCIA
       ,DBAMV.PACIENTE PACIENTE
       ,DBAMV.PRESTADOR PRESTADOR
  WHERE CTR_FREQUENCIA.CD_PACIENTE = PACIENTE.CD_PACIENTE
    AND CTR_FREQUENCIA.CD_FREQUENCIA = :CAMPO_FREQ
    AND CTR_FREQUENCIA.CD_TIP_ATE = :CAMPO_TIPO
    AND CTR_FREQUENCIA.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
