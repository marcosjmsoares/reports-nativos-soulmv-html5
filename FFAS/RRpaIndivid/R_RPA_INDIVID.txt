      ,NR_PAGINA
      ,NR_LINHAS_DA_PAGINA
      ,CD_PROCEDIMENTO
      ,NR_CNSPACIENTE
      ,TP_SEXO
      ,CD_IBGE
      ,CD_CID_PRINCIPAL

      ,TP_COR

 WHERE TP_FORMULARIO = 'I'
        AND CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA

:CAMPO_CF_ORD
