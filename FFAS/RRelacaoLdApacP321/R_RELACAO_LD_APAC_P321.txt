--R_RELACAO_LD_APAC_P321
SELECT * FROM (
SELECT LAUDO_SIA_APAC.CD_PRESTADOR
     , PRESTADOR.NM_PRESTADOR
     , PRESTADOR.DS_CODIGO_CONSELHO
     , LAUDO_SIA_APAC.CD_PACIENTE
     , PACIENTE.NM_PACIENTE
     , PACIENTE.NR_CPF
     , LAUDO_SIA_APAC.CD_PROCEDIMENTO
     , LAUDO_SIA_APAC.DT_LAUDO
     , LAUDO_SIA_APAC.CD_LAUDO
 FROM dbamv.LAUDO_SIA_APAC
    , dbamv.PRESTADOR
    , dbamv.PACIENTE
WHERE LAUDO_SIA_APAC.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
  AND LAUDO_SIA_APAC.CD_PACIENTE  = PACIENTE.CD_PACIENTE
 ORDER BY NM_PACIENTE
     , DT_LAUDO
)
ORDER BY NM_PRESTADOR,NM_PACIENTE,DT_LAUDO
