--R_CTR_FREQUENCIA_TIPO29
SELECT * FROM (
SELECT CTR_FREQUENCIA.CD_FREQUENCIA
       ,CTR_FREQUENCIA.CD_PACIENTE
      , dbamv.PKG_SUS_REGRA_PACIENTE.F_NOME_PACIENTE(CTR_FREQUENCIA.CD_PACIENTE) NM_PACIENTE
       ,PACIENTE.DT_NASCIMENTO
       ,TO_CHAR(PACIENTE.NR_CPF) NR_CPF
       ,TO_CHAR(PACIENTE.NR_CNS) NR_CNS
       ,PRESTADOR.NM_PRESTADOR
       ,CTR_FREQUENCIA.CD_TIP_ATE
       ,CTR_FREQUENCIA.DT_REFERENCIA
       ,CTR_FREQUENCIA.DT_INICIO_TRATAMENTO
       ,CTR_FREQUENCIA.DT_FINAL_TRATAMENTO
       ,CTR_FREQUENCIA.TP_COMPARECIMENTO
       ,CTR_FREQUENCIA.CD_SSM_PRINCIPAL
       ,CTR_FREQUENCIA.CD_SSM_SECUNDARIO
       ,CTR_FREQUENCIA.CD_SSM_SECUNDARIO2
       ,CTR_FREQUENCIA.CD_SSM_SECUNDARIO3
       ,CTR_FREQUENCIA.CD_procedimento 
       ,CFPS.cd_procedimento cd_proc_secundario 
  FROM DBAMV.CTR_FREQUENCIA CTR_FREQUENCIA
       ,DBAMV.PACIENTE PACIENTE
       ,DBAMV.PRESTADOR PRESTADOR
       ,dbamv.CTR_FREQUENCIA_PROC_SECUNDARIO CFPS 
  WHERE CTR_FREQUENCIA.CD_PACIENTE = PACIENTE.CD_PACIENTE
    AND CTR_FREQUENCIA.CD_FREQUENCIA = :CAMPO_FREQ
    AND CTR_FREQUENCIA.CD_TIP_ATE = :CAMPO_TIPO
    AND CTR_FREQUENCIA.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
     AND CTR_FREQUENCIA.cd_frequencia = CFPS.cd_frequencia(+) 
)
ORDER BY CD_PACIENTE,CD_FREQUENCIA,NM_PACIENTE,NR_CPF,CD_TIP_ATE,DT_REFERENCIA,NM_PRESTADOR,DT_INICIO_TRATAMENTO,DT_FINAL_TRATAMENTO,TP_COMPARECIMENTO,CD_SSM_PRINCIPAL,CD_SSM_SECUNDARIO,CD_SSM_SECUNDARIO2,CD_SSM_SECUNDARIO3,CD_PROCEDIMENTO
