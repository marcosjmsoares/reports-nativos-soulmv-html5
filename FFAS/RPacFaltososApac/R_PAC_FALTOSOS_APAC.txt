--R_PAC_FALTOSOS_APAC
SELECT DISTINCT p.cd_paciente, p.nm_paciente, a.cd_tip_ate, a.nr_apac
  FROM DBAMV.APAC     a
     , DBAMV.PACIENTE p
 WHERE a.cd_paciente = p.cd_paciente
   AND a.tp_apac <> '3'
   AND TRUNC(:CAMPO_DT_LAUDO, 'MM') BETWEEN a.DT_INICIAL AND a.DT_FINAL
   AND a.nr_apac NOT IN (SELECT DISTINCT a.nr_apac
                           FROM dbamv.remessa_apac r
                              , dbamv.apac a
                          WHERE a.cd_remessa = r.cd_remessa
                            AND a.cd_tip_ate IN (28, 29)
                            AND r.dt_competencia_apresentacao = TRUNC(:CAMPO_DT_LAUDO, 'MM')
						  )
order by p.nm_paciente, a.nr_apac
