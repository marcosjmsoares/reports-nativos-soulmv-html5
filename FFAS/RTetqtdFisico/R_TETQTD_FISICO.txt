--R_TETQTD_FISICO
SELECT * FROM (
Select Tet_Ups.Cd_Multi_Empresa 			Cd_Multi_Empresa
     ,Tet_Ups.Cd_Ups				Cd_Ups
     ,Unid_Prest.Ds_Ups				Ds_Ups
     ,Tet_Ups.Cd_Procedimento			Cd_Procedimento
     ,Proc_Sus.Ds_procedimento                                              Ds_procedimento
     ,Tet_Ups.Qt_Fisico				Qt_Fisico
    ,Tet_Ups.Vl_Orcamentario				Vl_Orcamentario
   ,Tet_Ups.Dt_Vigencia				Dt_Vigencia
  ,Nvl(Tet_Ups.Qt_Fisico_Atual,0) 			Qt_Fisico_Atual
,Nvl(Tet_Ups.Vl_Orcamentario_Atual,0) Vl_Orcamentario_Atual
  From dbamv.Procedimento_Sus_Teto_Ups    Tet_Ups
          ,dbamv.Unidade_Prestadora_Servico   Unid_Prest
          ,dbamv.Procedimento_Sus                     Proc_Sus
 where Tet_Ups.cd_ups = Unid_Prest.cd_ups
  and  Tet_Ups.cd_procedimento = Proc_Sus.cd_procedimento
Order by Tet_Ups.Cd_Ups ,Tet_Ups.Cd_Procedimento
)
ORDER BY CD_MULTI_EMPRESA,CD_UPS,DS_UPS
