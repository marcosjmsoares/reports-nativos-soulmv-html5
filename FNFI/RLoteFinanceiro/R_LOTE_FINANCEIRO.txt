--R_LOTE_FINANCEIRO
SELECT LF.CD_LOTE_FINANCEIRO || ' - ' || DS_LOTE AS LOTE_FINANCEIRO,
       (SELECT CD_MULTI_EMPRESA || ' - ' || DS_MULTI_EMPRESA FROM DBAMV.MULTI_EMPRESAS WHERE CD_MULTI_EMPRESA = LF.CD_MULTI_EMPRESA) AS EMPRESA,
       LF.CD_USUARIO_ABER AS USUARIO_ABERTURA,
       LF.CD_USUARIO_PROC AS USUARIO_PROCESSAMENTO,
       LF.CD_USUARIO_CANC AS USUARIO_CANCELAMENTO,
       (SELECT CD_CAMARA_FINA || ' - ' || DS_CAMARA FROM DBAMV.CAMARA_FINA WHERE CD_CAMARA_FINA = LF.CD_CAMARA_FINA) AS CAMARA,
      (SELECT CD_FORNECEDOR || ' - ' || NM_FORNECEDOR FROM DBAMV.FORNECEDOR WHERE CD_FORNECEDOR = LF.CD_FORNECEDOR) AS RESPONSAVEL,
       LF.CD_CON_REC,
       LF.CD_CON_PAG,
       nvl(LF.CD_CON_REC, LF.CD_CON_PAG)  as CONTA,
       (SELECT CD_PROCESSO || ' - ' || DS_PROCESSO FROM DBAMV.PROCESSO WHERE CD_PROCESSO = LF.CD_PROCESSO) AS PROCESSO,
       (SELECT CD_CON_COR || ' - ' || DS_CON_COR FROM DBAMV.CON_COR WHERE CD_CON_COR = LF.CD_CON_COR) AS CONTA_CORRENTE,
       (SELECT CD_CAIXA || ' - ' || DS_CAIXA FROM DBAMV.CAIXA WHERE CD_CAIXA = LF.CD_CAIXA) AS CAIXA,
       LF.CD_DOCUMENTO as DOCUMENTO,
        LF.CD_MOEDA,
       (SELECT CD_FIN_CAR || ' - ' || NM_FIN_CAR FROM DBAMV.FIN_CAR WHERE CD_FIN_CAR = LF.CD_FIN_CAR) AS FIN_CAR,
       LF.TP_PERIODO,
       LF.TP_LOTE,
       LF.VL_RECEBER,
       LF.VL_PAGAR,
       TO_CHAR(LF.DT_PERIODO_INI, 'DD/MM/YYYY') as DT_PERIODO_INI,
       TO_CHAR(LF.DT_PERIODO_FIN, 'DD/MM/YYYY') as DT_PERIODO_FIN,
       LF.DT_MOVIMENTO,
       LF.DT_ABERTURA,
       LF.DT_PROCESSADO,
       LF.DT_CANCELADO,
       LF.TP_STATUS
  FROM DBAMV.LOTE_FINANCEIRO LF, DBAMV.LOTE_FINANCEIRO_ITEM LFI
 WHERE LF.CD_LOTE_FINANCEIRO = LFI.CD_LOTE_FINANCEIRO
 AND LF.CD_LOTE_FINANCEIRO = :CAMPO_CD_LOTE_FINANCEIRO

--R_LOTE_FINANCEIRO_ITEM
SELECT CP.CD_CON_PAG  CD_CONTA,
		 CONCAT(CONCAT(FO.CD_FORNECEDOR, '-'), FO.NM_FORNECEDOR)  NM_RESPONSAVEL,
		 TPD.DS_TIP_DOC  TP_DOCUMENTO,
		 CASE WHEN CP.NR_SERIE IS NOT NULL
		 THEN CONCAT(CONCAT(CP.NR_DOCUMENTO, '-'), CP.NR_SERIE)
		 ELSE CP.NR_DOCUMENTO END  NR_DOCUMENTO,
		 PR.CD_ESTRUTURAL ||'-'||PR.DS_PROCESSO DS_PROCESSO,
		 TO_CHAR(CP.DT_LANCAMENTO, 'DD/MM/YYYY')  DT_LANCAMENTO,
		 TO_CHAR(CP.DT_EMISSAO, 'DD/MM/YYYY')  DT_EMISSAO,
		 TO_CHAR(ITCP.DT_VENCIMENTO, 'DD/MM/YYYY')  DT_VENCIMENTO,
		 ITCP.CD_MOEDA  CD_MOEDA,
		 CASE WHEN ITCP.TP_QUITACAO = 'V' THEN 'PREVISTO'
		      WHEN ITCP.TP_QUITACAO = 'C' THEN 'COMPROMETIDO'
		      WHEN ITCP.TP_QUITACAO = 'P' THEN 'PARCIAMENTE PAGO'
		      WHEN ITCP.TP_QUITACAO = 'Q' THEN 'QUITADO'
		      WHEN ITCP.TP_QUITACAO = 'T' THEN 'LIQ. TOTAL DEVOLUÇÃO'
		      WHEN ITCP.TP_QUITACAO = 'L' THEN 'LIQ. PARCIAL DEVOLUÇÃO'
		 ELSE '' END  TP_QUITACAO,
		 LFI.VL_TITULO   VL_DUPLICATA,
		 LFI.VL_PROCESSO,
		 ITCP.CD_ITCON_PAG AS CD_CONTA_ITEM,
         LFI.TP_FINANCEIRO
		 FROM DBAMV.CON_PAG CP, DBAMV.ITCON_PAG ITCP, DBAMV.FORNECEDOR FO,
		 DBAMV.TIP_DOC TPD, DBAMV.PROCESSO PR, DBAMV.CAMARA_FINA_PESS PESS,
		 DBAMV.LOTE_FINANCEIRO_ITEM LFI
		 WHERE CP.CD_CON_PAG = ITCP.CD_CON_PAG
		 AND CP.CD_FORNECEDOR = FO.CD_FORNECEDOR
		 AND CP.CD_FORNECEDOR = PESS.CD_FORNECEDOR
		 AND CP.CD_TIP_DOC = TPD.CD_TIP_DOC
		 AND CP.CD_PROCESSO = PR.CD_PROCESSO
		 AND CP.CD_MULTI_EMPRESA = PR.CD_MULTI_EMPRESA
		 AND LFI.CD_ITCON_PAG = ITCP.CD_ITCON_PAG
		 AND LFI.CD_LOTE_FINANCEIRO = :CAMPO_CD_LOTE_FINANCEIRO
		 UNION
		 SELECT CR.CD_CON_REC  CD_CONTA,
		 CONCAT(CONCAT(FO.CD_FORNECEDOR, '-'), FO.NM_FORNECEDOR)  NM_RESPONSAVEL,
		 TPD.DS_TIP_DOC  TP_DOCUMENTO,
		 CASE WHEN CR.NR_SERIE IS NOT NULL
		 THEN CONCAT(CONCAT(CR.NR_DOCUMENTO, '-'), CR.NR_SERIE)
		 ELSE CR.NR_DOCUMENTO END NR_DOCUMENTO,
		 PR.CD_ESTRUTURAL ||'-'||PR.DS_PROCESSO DS_PROCESSO,
		 TO_CHAR(CR.DT_LANCAMENTO, 'DD/MM/YYYY')  DT_LANCAMENTO,
		 TO_CHAR(CR.DT_EMISSAO, 'DD/MM/YYYY')  DT_EMISSAO,
		 TO_CHAR(ITCR.DT_VENCIMENTO, 'DD/MM/YYYY')  DT_VENCIMENTO,
		 ITCR.CD_MOEDA  CD_MOEDA,
		 CASE WHEN ITCR.TP_QUITACAO = 'V' THEN 'PREVISTO'
		      WHEN ITCR.TP_QUITACAO = 'C' THEN 'COMPROMETIDO'
		      WHEN ITCR.TP_QUITACAO = 'P' THEN 'PARCIAMENTE QUITADO'
		      WHEN ITCR.TP_QUITACAO = 'Q' THEN 'QUITADO'
		      WHEN ITCR.TP_QUITACAO = 'L' THEN 'CANCELADO'
		      WHEN ITCR.TP_QUITACAO = 'R' THEN 'PROTESTADO'
		      WHEN ITCR.TP_QUITACAO = 'G' THEN 'QUITADO COM GLOSA'
		 ELSE '' END  TP_QUITACAO,
		 LFI.VL_TITULO   VL_DUPLICATA,
		 LFI.VL_PROCESSO,
		 ITCR.CD_ITCON_REC AS CD_CONTA_ITEM,
         LFI.TP_FINANCEIRO
		 FROM DBAMV.CON_REC CR, DBAMV.ITCON_REC ITCR, DBAMV.FORNECEDOR FO,
		 DBAMV.TIP_DOC TPD, DBAMV.PROCESSO PR, DBAMV.CAMARA_FINA_PESS PESS,
		 DBAMV.LOTE_FINANCEIRO_ITEM LFI
		 WHERE CR.CD_CON_REC = ITCR.CD_CON_REC
		 AND CR.CD_FORNECEDOR = FO.CD_FORNECEDOR
		 AND CR.CD_FORNECEDOR = PESS.CD_FORNECEDOR
		 AND CR.CD_TIP_DOC = TPD.CD_TIP_DOC
		 AND CR.CD_PROCESSO = PR.CD_PROCESSO
		 AND CR.CD_MULTI_EMPRESA = PR.CD_MULTI_EMPRESA
		 AND LFI.CD_ITCON_REC = ITCR.CD_ITCON_REC
		 AND LFI.CD_LOTE_FINANCEIRO = :CAMPO_CD_LOTE_FINANCEIRO
