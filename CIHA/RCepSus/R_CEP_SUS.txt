--R_CEP_SUS
SELECT * FROM (
select uf.cd_uf
     , uf.nm_uf
     , cidade.cd_ibge
     , cidade.nm_cidade
     , cep_sus.nr_cep
  from dbamv.uf
     , dbamv.cidade
     , dbamv.cep_sus
 where uf.cd_uf               = cep_sus.cd_uf
   and cep_sus.cd_ibge        = cidade.cd_ibge
   and cep_sus.cd_uf          = cidade.cd_uf
order by cd_uf, cd_ibge, nr_cep
)
ORDER BY CD_UF,NM_UF,
CD_IBGE,NM_CIDADE
