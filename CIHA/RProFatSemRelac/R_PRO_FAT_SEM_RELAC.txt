--R_PRO_FAT_SEM_RELAC
select tp_relacionamento
     , cd_pro_fat
     , ds_pro_fat
     , count('x') qt_ocorrencias
  from ( select decode(v.tp_registro_ciha,'2','I','A') tp_relacionamento
              , v.cd_pro_fat
              , p.ds_pro_fat
           from dbamv.v_ciha_producao_geral v
              , dbamv.pro_fat p
              , dbamv.pro_fat_sus ps
          where v.cd_pro_fat = p.cd_pro_fat
            and v.cd_pro_fat = ps.cd_pro_fat (+)   
            and v.tp_registro_ciha in ('2','4')
            and nvl(v.sn_pacote_ciha,'N') = 'N'
            and 'N' = Decode(v.tp_registro_ciha, '4', Nvl(v.cd_procedimento,'N'), Nvl(ps.cd_procedimento, 'N') )
            and ( dbamv.pkg_ciha_producao.F_CHECA_REGRA_GERACAO ( v.tp_atendimento
                                                                , v.cd_convenio
                                                                , v.cd_ori_ate
                                                                , v.tp_gru_pro
                                                                , v.cd_gru_pro
                                                                , v.cd_pro_fat
                                                                , v.sn_atendimento_sem_conta ) = 'S' ) )
 group by tp_relacionamento
        , cd_pro_fat
        , ds_pro_fat
