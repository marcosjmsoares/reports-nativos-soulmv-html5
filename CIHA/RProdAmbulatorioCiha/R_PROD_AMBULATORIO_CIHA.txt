--R_PROD_AMBULATORIO_CIHA_R_ORIGEM_A_Q_PROD_ANALITCO
SELECT * FROM (
select dt_competencia_a
     , cd_ori_ate_a
     , ds_ori_ate_a
     , cd_convenio_a
     , nm_convenio_a
     , cd_atendimento_a
     , dt_atendimento_a
     , dt_alta_a
     , qt_ambulatorial_a
     , cd_paciente_a
     , nm_paciente_a
     , cd_conta_a
     , cd_procedimento_a
     , tp_registro_ciha_a
     , tp_instrumento_registro
 from (
        select trunc(geracao_ciha.dt_competencia,'MM')                dt_competencia_a
             , producao_convenio_ciha.cd_ori_ate                      cd_ori_ate_a
             , ori_ate.ds_ori_ate                                     ds_ori_ate_a
             , producao_convenio_ciha.cd_convenio                     cd_convenio_a
             , convenio.nm_convenio                                   nm_convenio_a
             , producao_convenio_ciha.cd_atendimento                  cd_atendimento_a
             , producao_convenio_ciha.dt_atendimento                  dt_atendimento_a
             , producao_convenio_ciha.dt_alta                         dt_alta_a
             , 1                                                      qt_ambulatorial_a
             , producao_convenio_ciha.cd_paciente                     cd_paciente_a
             , paciente.nm_paciente                                   nm_paciente_a
             , null                                                   cd_conta_a
             , null                                                   cd_procedimento_a
             , null                                                   tp_registro_ciha_a
             , null                                                   tp_instrumento_registro
          from dbamv.geracao_ciha
             , dbamv.producao_convenio_ciha
             , dbamv.ori_ate
             , dbamv.convenio
             , dbamv.paciente
         where paciente.cd_paciente                      = producao_convenio_ciha.cd_paciente
           and producao_convenio_ciha.cd_convenio        = convenio.cd_convenio
           and producao_convenio_ciha.cd_ori_ate         = ori_ate.cd_ori_ate
           and producao_convenio_ciha.cd_modalidade_ciha = '01'
           and producao_convenio_ciha.cd_geracao         = geracao_ciha.cd_geracao
           and to_char(geracao_ciha.dt_competencia,'MMYYYY') = to_char(:CAMPO_COMPET,'MMYYYY')
           and geracao_ciha.cd_multi_empresa          = dbamv.pkg_mv2000.le_empresa
           and ( :CAMPO_TP_ORIGEM_CALCULO = 'A' and :CAMPO_TP_PRODUCAO = 'CNV-A' )
        group by trunc(geracao_ciha.dt_competencia,'MM')
             , producao_convenio_ciha.cd_ori_ate
             , ori_ate.ds_ori_ate
             , producao_convenio_ciha.cd_convenio
             , convenio.nm_convenio
             , producao_convenio_ciha.cd_atendimento
             , producao_convenio_ciha.dt_atendimento
             , producao_convenio_ciha.dt_alta
             , producao_convenio_ciha.cd_paciente
             , paciente.nm_paciente
        union all
        select trunc(geracao_ciha.dt_competencia,'MM')                    dt_competencia_a
             , producao_convenio_ciha.cd_ori_ate                          cd_ori_ate_a
             , ori_ate.ds_ori_ate                                         ds_ori_ate_a
             , producao_convenio_ciha.cd_convenio                         cd_convenio_a
             , convenio.nm_convenio                                       nm_convenio_a
             , producao_convenio_ciha.cd_atendimento                      cd_atendimento_a
             , producao_convenio_ciha.dt_atendimento                      dt_atendimento_a
             , producao_convenio_ciha.dt_alta                             dt_alta_a
             , producao_convenio_ciha.qt_lancamento                       qt_ambulatorial_a
             , producao_convenio_ciha.cd_paciente                         cd_paciente_a
             , paciente.nm_paciente                                       nm_paciente_a
             , nvl(producao_convenio_ciha.cd_reg_amb,producao_convenio_ciha.cd_aviso_cirurgia) cd_conta_a
             , producao_convenio_ciha.cd_pro_fat||'  '                    cd_procedimento_a
             , producao_convenio_ciha.tp_registro_ciha                    tp_registro_ciha_a
             , producao_convenio_ciha.tp_instrumento_registro             tp_instrumento_registro
          from dbamv.geracao_ciha
             , dbamv.producao_convenio_ciha
             , dbamv.ori_ate
             , dbamv.convenio
             , dbamv.paciente
         where paciente.cd_paciente                   = producao_convenio_ciha.cd_paciente
           and producao_convenio_ciha.cd_convenio        = convenio.cd_convenio
           and producao_convenio_ciha.cd_ori_ate         = ori_ate.cd_ori_ate
           and producao_convenio_ciha.cd_modalidade_ciha = '01'
           and producao_convenio_ciha.cd_geracao         = geracao_ciha.cd_geracao
           and to_char(geracao_ciha.dt_competencia,'MMYYYY') = to_char(:CAMPO_COMPET,'MMYYYY')
           and geracao_ciha.cd_multi_empresa          = dbamv.pkg_mv2000.le_empresa
           and ( :CAMPO_TP_ORIGEM_CALCULO = 'P' and :CAMPO_TP_PRODUCAO = 'CNV-A' )
        union all
        select trunc(atendimento_sus_ciha.dt_atendimento,'MM')          dt_competencia_s
             , atendimento_sus_ciha.cd_ori_ate       cd_ori_ate_a
             , ori_ate.ds_ori_ate                    ds_ori_ate_a
             , atendimento_sus_ciha.cd_convenio      cd_convenio_a
             , convenio.nm_convenio                  nm_convenio_a
             , atendimento_sus_ciha.cd_atendimento   cd_atendimento_a
             , atendimento_sus_ciha.dt_atendimento   dt_atendimento_a
             , atendimento_sus_ciha.dt_alta          dt_alta_a
             , 1                                     qt_ambulatorial_a
             , atendimento_sus_ciha.cd_paciente      cd_paciente_a
             , paciente.nm_paciente                  nm_paciente_a
             , null                                  cd_conta_a
             , null                                  cd_procedimento_a
             , null                                  tp_registro_ciha_a
             , null                                  tp_instrumento_registro
          from dbamv.atendimento_sus_ciha
             , dbamv.ori_ate
             , dbamv.convenio
             , dbamv.paciente
         where paciente.cd_paciente                  = atendimento_sus_ciha.cd_paciente
           and atendimento_sus_ciha.cd_convenio      = convenio.cd_convenio
           and atendimento_sus_ciha.cd_ori_ate       = ori_ate.cd_ori_ate
           and to_char(atendimento_sus_ciha.dt_atendimento,'MMYYYY')    = to_char(:CAMPO_COMPET,'MMYYYY')
           and atendimento_sus_ciha.cd_multi_empresa = dbamv.pkg_mv2000.le_empresa
           and ( :CAMPO_TP_ORIGEM_CALCULO = 'A' and :CAMPO_TP_PRODUCAO = 'SUS-A' and atendimento_sus_ciha.tp_producao = :CAMPO_TP_PRODUCAO )
        union all
        select trunc(producao_sus_ciha.dt_competencia,'MM') dt_competencia_s
             , producao_sus_ciha.cd_ori_ate                 cd_ori_ate_a
             , ori_ate.ds_ori_ate                           ds_ori_ate_a
             , producao_sus_ciha.cd_convenio                cd_convenio_a
             , convenio.nm_convenio                         nm_convenio_a
             , producao_sus_ciha.cd_atendimento             cd_atendimento_a
             , atendimento_sus_ciha.dt_atendimento          dt_atendimento_a
             , atendimento_sus_ciha.dt_alta                 dt_alta_a
             , sum(nvl(producao_sus_ciha.qt_lancamento,0))  qt_ambulatorial_a
             , producao_sus_ciha.cd_paciente                cd_paciente_a
             , nvl(paciente.nm_paciente,'NÃO INFORMADO')                         nm_paciente_a
             , decode(producao_sus_ciha.cd_apac, null, producao_sus_ciha.cd_eventos, producao_sus_ciha.cd_apac ) cd_conta_a
             , producao_sus_ciha.cd_procedimento            cd_procedimento_a
             , null                                         tp_registro_ciha_a
             , producao_sus_ciha.tp_instrumento_registro    tp_instrumento_registro
          from dbamv.producao_sus_ciha
             , dbamv.ori_ate
             , dbamv.convenio
             , dbamv.atendimento_sus_ciha
             , dbamv.paciente
         where producao_sus_ciha.cd_paciente    = paciente.cd_paciente(+)
           and producao_sus_ciha.cd_atendimento   = atendimento_sus_ciha.cd_atendimento(+)
           and producao_sus_ciha.cd_convenio      = convenio.cd_convenio
           and producao_sus_ciha.cd_ori_ate       = ori_ate.cd_ori_ate
           and to_char(producao_sus_ciha.dt_competencia,'MMYYYY') = to_char(:CAMPO_COMPET,'MMYYYY')
           and producao_sus_ciha.cd_multi_empresa = dbamv.pkg_mv2000.le_empresa
           and ( producao_sus_ciha.tp_producao = :CAMPO_TP_PRODUCAO and :CAMPO_TP_PRODUCAO = 'SUS-A' and :CAMPO_TP_ORIGEM_CALCULO = 'P' )
         group by trunc(producao_sus_ciha.dt_competencia,'MM')
                , producao_sus_ciha.cd_ori_ate
                , ori_ate.ds_ori_ate
                , producao_sus_ciha.cd_convenio
                , convenio.nm_convenio
                , producao_sus_ciha.cd_atendimento
                , atendimento_sus_ciha.dt_atendimento
                , atendimento_sus_ciha.dt_alta
                , producao_sus_ciha.cd_paciente
                , paciente.nm_paciente
                , decode(producao_sus_ciha.cd_apac, null, producao_sus_ciha.cd_eventos, producao_sus_ciha.cd_apac )
                , producao_sus_ciha.cd_procedimento
                , producao_sus_ciha.tp_instrumento_registro
) order by Trunc(dt_competencia_a,'MM')
         , cd_ori_ate_a
         , cd_convenio_a
         , cd_atendimento_a
)
ORDER BY CD_ORI_ATE_A,DS_ORI_ATE_A,DT_COMPETENCIA_A,
CD_CONVENIO_A,NM_CONVENIO_A

--R_PROD_AMBULATORIO_CIHA_R_PROC_MEDIA_PERM_SUS1_Q_PROD_SINTETICO
SELECT * FROM (
select cd_ori_ate_s
     , ds_ori_ate_s
     , cd_convenio_s
     , nm_convenio_s
     , count('x') qt_ambulatorial_s
  from ( select producao_convenio_ciha.cd_ori_ate          cd_ori_ate_s
              , ori_ate.ds_ori_ate                      ds_ori_ate_s
              , producao_convenio_ciha.cd_convenio         cd_convenio_s
              , convenio.nm_convenio                    nm_convenio_s
              , producao_convenio_ciha.cd_atendimento      cd_atendimento_s
           from dbamv.geracao_ciha
              , dbamv.producao_convenio_ciha
              , dbamv.ori_ate
              , dbamv.convenio
          where producao_convenio_ciha.cd_convenio               = convenio.cd_convenio
            and producao_convenio_ciha.cd_ori_ate                = ori_ate.cd_ori_ate
            and producao_convenio_ciha.cd_modalidade_ciha        = '01'
            and producao_convenio_ciha.cd_geracao                = geracao_ciha.cd_geracao
            and to_char(geracao_ciha.dt_competencia,'MMYYYY') = to_char(:CAMPO_COMPET,'MMYYYY')
            and geracao_ciha.cd_multi_empresa                 = dbamv.pkg_mv2000.le_empresa
            and (:CAMPO_TP_PRODUCAO = 'CNV-A' and :CAMPO_TP_ORIGEM_CALCULO = 'A' )
          group by producao_convenio_ciha.cd_ori_ate
                 , ori_ate.ds_ori_ate
                 , producao_convenio_ciha.cd_convenio
                 , convenio.nm_convenio
                 , producao_convenio_ciha.cd_atendimento
) group by cd_ori_ate_s
         , ds_ori_ate_s
         , cd_convenio_s
         , nm_convenio_s
union all
select producao_convenio_ciha.cd_ori_ate cd_ori_ate_s
     , ori_ate.ds_ori_ate ds_ori_ate_s
     , producao_convenio_ciha.cd_convenio cd_convenio_s
     , convenio.nm_convenio nm_convenio_s
     , sum(nvl(producao_convenio_ciha.qt_lancamento,0)) qt_ambulatorial_s
  from dbamv.geracao_ciha
     , dbamv.producao_convenio_ciha
     , dbamv.ori_ate
     , dbamv.convenio
 where producao_convenio_ciha.cd_convenio               = convenio.cd_convenio
   and producao_convenio_ciha.cd_ori_ate                = ori_ate.cd_ori_ate
   and producao_convenio_ciha.cd_modalidade_ciha        = '01'
   and producao_convenio_ciha.cd_geracao                = geracao_ciha.cd_geracao
   and to_char(geracao_ciha.dt_competencia,'MMYYYY') = to_char(:CAMPO_COMPET,'MMYYYY')
   and geracao_ciha.cd_multi_empresa                 = dbamv.pkg_mv2000.le_empresa
   and (:CAMPO_TP_PRODUCAO = 'CNV-A' and :CAMPO_TP_ORIGEM_CALCULO = 'P' )
 group by producao_convenio_ciha.cd_ori_ate
        , ori_ate.ds_ori_ate
        , producao_convenio_ciha.cd_convenio
        , convenio.nm_convenio
union all
select atendimento_sus_ciha.cd_ori_ate                 cd_ori_ate_s
     , ori_ate.ds_ori_ate                              ds_ori_ate_s
     , atendimento_sus_ciha.cd_convenio                cd_convenio_s
     , convenio.nm_convenio                            nm_convenio_s
     , count('x')                                      qt_ambulatorial_s
  from dbamv.atendimento_sus_ciha
     , dbamv.ori_ate
     , dbamv.convenio
 where atendimento_sus_ciha.cd_convenio        = convenio.cd_convenio
   and atendimento_sus_ciha.cd_ori_ate         = ori_ate.cd_ori_ate
   and to_char(atendimento_sus_ciha.dt_atendimento,'MMYYYY') = to_char(:CAMPO_COMPET,'MMYYYY')
   and atendimento_sus_ciha.cd_multi_empresa = dbamv.pkg_mv2000.le_empresa
   and ( atendimento_sus_ciha.tp_producao = :CAMPO_TP_PRODUCAO and :CAMPO_TP_PRODUCAO = 'SUS-A' and :CAMPO_TP_ORIGEM_CALCULO = 'A' )
 group by atendimento_sus_ciha.cd_ori_ate
        , ori_ate.ds_ori_ate
        , atendimento_sus_ciha.cd_convenio
        , convenio.nm_convenio
union all
select producao_sus_ciha.cd_ori_ate                 cd_ori_ate_s
     , ori_ate.ds_ori_ate                           ds_ori_ate_s
     , producao_sus_ciha.cd_convenio                cd_convenio_s
     , convenio.nm_convenio                         nm_convenio_s
     , sum(nvl(producao_sus_ciha.qt_lancamento,0))  qt_ambulatorial_s
  from dbamv.producao_sus_ciha
     , dbamv.ori_ate
     , dbamv.convenio
 where producao_sus_ciha.cd_convenio        = convenio.cd_convenio
   and producao_sus_ciha.cd_ori_ate         = ori_ate.cd_ori_ate
   and to_char(producao_sus_ciha.dt_competencia,'MMYYYY') = to_char(:CAMPO_COMPET,'MMYYYY')
   and producao_sus_ciha.cd_multi_empresa = dbamv.pkg_mv2000.le_empresa
   and ( producao_sus_ciha.tp_producao = :CAMPO_TP_PRODUCAO and :CAMPO_TP_PRODUCAO = 'SUS-A' and :CAMPO_TP_ORIGEM_CALCULO = 'P' )
 group by producao_sus_ciha.cd_ori_ate
        , ori_ate.ds_ori_ate
        , producao_sus_ciha.cd_convenio
        , convenio.nm_convenio
order by cd_ori_ate_s, cd_convenio_s
)
ORDER BY CD_ORI_ATE_S,DS_ORI_ATE_S
