--R_CONSIST_CIHA
SELECT * FROM (
select tp_registro
     , cd_atendimento
     , dt_atendimento
     , dt_alta
     , cd_paciente
     , nm_paciente
     , cd_critica
     , ds_critica
  from ( select tmpmv_ciha_critica.tp_registro             tp_registro
	              , tmpmv_ciha_critica.cd_atendimento          cd_atendimento
	              , atendime.dt_atendimento                    dt_atendimento
	              , atendime.dt_alta                           dt_alta
	              , tmpmv_ciha_critica.cd_paciente             cd_paciente
	              , paciente.nm_paciente                       nm_paciente
	              , tmpmv_ciha_critica.cd_critica              cd_critica
	              , tmpmv_ciha_critica.ds_critica              ds_critica
	           from dbamv.tmpmv_ciha_critica
	              , dbamv.atendime
	              , dbamv.paciente
	          where tmpmv_ciha_critica.cd_geracao = :CAMPO_CD_GERACAO
	            and tmpmv_ciha_critica.cd_atendimento = atendime.cd_atendimento(+)
	            and atendime.cd_paciente              = paciente.cd_paciente(+)         
	      group by tmpmv_ciha_critica.tp_registro
	                 , tmpmv_ciha_critica.cd_atendimento
	                 , atendime.dt_atendimento
	                 , atendime.dt_alta
	                 , tmpmv_ciha_critica.cd_paciente
	                 , paciente.nm_paciente
	                 , tmpmv_ciha_critica.cd_critica
	                 , tmpmv_ciha_critica.ds_critica 	                 
)  order by tp_registro
         , cd_atendimento
         , cd_critica
)
ORDER BY DT_ATENDIMENTO,DT_ALTA,NM_PACIENTE,CD_CRITICA,TP_REGISTRO,CD_ATENDIMENTO,CD_PACIENTE
