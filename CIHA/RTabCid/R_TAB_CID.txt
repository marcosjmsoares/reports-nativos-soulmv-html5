--R_TAB_CID
SELECT * FROM (
select tp_cid
     , cd_cid
     , ds_cid
     , tp_sexo
  from ( select '1-Categoria' tp_cid
              , cid.cd_cid
              , cid.ds_cid
              , decode(cid.tp_sexo, 'A', 'Ambos', 'F', 'Feminino', 'M', 'Masculino') tp_sexo
           from dbamv.cid
          where nvl(cid.cat,'N') = 'S' and nvl(cid.subcat,'N') = 'N'
            and length(cid.cd_cid) < 4
          union all
         select '2-Subcategoria' tp_cid
              , cid.cd_cid
              , cid.ds_cid
              , decode(cid.tp_sexo, 'A', 'Ambos', 'F', 'Feminino', 'M', 'Masculino') tp_sexo
           from dbamv.cid
          where nvl(cid.subcat,'N') = 'S'
            and length(cid.cd_cid) in (3,4)
         union all
         select '3-Morfológico' tp_cid
              , cid.cd_cid
              , cid.ds_cid
              , decode(cid.tp_sexo, 'A', 'Ambos', 'F', 'Feminino', 'M', 'Masculino') tp_sexo
           from dbamv.cid
          where length(cid.cd_cid) = 6
         union all
         select '4-Indefinido' tp_cid
              , cid.cd_cid
              , cid.ds_cid
              , decode(cid.tp_sexo, 'A', 'Ambos', 'F', 'Feminino', 'M', 'Masculino') tp_sexo
           from dbamv.cid
          where ( length(cid.cd_cid) <> 6 and ( nvl(cid.cat,'X') = 'N' and nvl(cid.subcat,'N') = 'N' ) )
       )
:CAMPO_CF_WHERE
)
