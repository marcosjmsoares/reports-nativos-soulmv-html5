--R_PRO_FAT_PRO_SUS
SELECT * FROM (
select tp_relacionamento
     , cd_pro_fat
     , ds_pro_fat
     , cd_procedimento_sus
     , ds_procedimento_sus
  from ( select 'Internação' tp_relacionamento
              , pro_fat.cd_pro_fat
              , pro_fat.ds_pro_fat
              , procedimento_sus.cd_procedimento cd_procedimento_sus
              , procedimento_sus.ds_procedimento ds_procedimento_sus
           from dbamv.pro_fat_sus
              , dbamv.pro_fat
              , dbamv.procedimento_sus
          where procedimento_sus.cd_procedimento = pro_fat_sus.cd_procedimento
            and pro_fat.cd_pro_fat               = pro_fat_sus.cd_pro_fat
         union all
         select 'Ambulatorial' tp_relacionamento
              , pro_fat.cd_pro_fat
              , pro_fat.ds_pro_fat
              , procedimento_sus.cd_procedimento cd_procedimento_sus
              , procedimento_sus.ds_procedimento ds_procedimento_sus
           from dbamv.pro_fat_sus_amb
              , dbamv.pro_fat
              , dbamv.procedimento_sus
          where procedimento_sus.cd_procedimento = pro_fat_sus_amb.cd_procedimento_sus
            and pro_fat.cd_pro_fat               = pro_fat_sus_amb.cd_pro_fat
  )
 group by tp_relacionamento
        , cd_pro_fat
        , ds_pro_fat
        , cd_procedimento_sus
        , ds_procedimento_sus
)
ORDER BY TP_RELACIONAMENTO
