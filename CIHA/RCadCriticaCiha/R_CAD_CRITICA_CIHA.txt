--R_CAD_CRITICA_CIHA
select cd_critica
     , replace(replace(ds_critica, '#PARAM#', ''),'  ', ' ' ) ds_critica
     , decode(sn_internacao,'S','Sim','Não')      sn_internacao
     , decode(sn_individualizado,'S','Sim','Não') sn_individualizado
     , decode(sn_consolidado,'S','Sim','Não')     sn_consolidado
     , dt_validade_inicial
     , dt_validade_final
     , ds_orientacao
  from dbamv.critica_ciha
