--R_ORC_REALIZ
SELECT * FROM (
select cota.cd_cota,
       cota.ds_cota,
       cota_competencia.dt_competencia,
       to_char(cota_competencia.dt_competencia, 'mm/yyyy') dt_competencia_string,
       cota_competencia.vl_orcado ,
       cota_competencia.vl_realizado
  from dbamv.cota,
       dbamv.cota_competencia
 where cota.cd_cota = cota_competencia.cd_cota
    and  Cota.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
    and cota_competencia.dt_competencia between :CAMPO_COMPET_INI order by cota.cd_cota, cota_competencia.dt_competencia
)
ORDER BY DS_COTA,CD_COTA, DT_COMPETENCIA
