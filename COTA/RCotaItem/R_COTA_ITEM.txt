--R_COTA_ITEM
SELECT * FROM (
 select a.cd_cota,
        a.ds_cota,
        a.cd_item_res,
        a.ds_item_res,
        sum(a.vl_itmvto_cotas) vl_itmvto_cotas
 from(
 select cota.cd_cota,
        cota.ds_cota,
        item_res.cd_item_res,
        item_res.ds_item_res,
        itmvto_cotas.vl_itmvto_cotas
   from dbamv.cota,
        dbamv.cota_competencia,
        dbamv.itmvto_cotas,
        dbamv.ratcon_pag,
        dbamv.con_pag,
        dbamv.item_res
  where  to_char(cota_competencia.dt_competencia,'YYYYMM') = to_char( :CAMPO_COMPET ,'YYYYMM')
    and cota.cd_cota = cota_competencia.cd_cota
    and cota_competencia.cd_competencia = itmvto_cotas.cd_competencia
    and itmvto_cotas.cd_ratcon_pag = ratcon_pag.cd_ratcon_pag
    and con_pag.cd_con_pag = ratcon_pag.cd_con_pag
    and con_pag.cd_previsao is null
    and ratcon_pag.cd_item_res = item_res.cd_item_res
   and  Cota.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
union all
 select cota.cd_cota,
        cota.ds_cota,
        item_res.cd_item_res,
        item_res.ds_item_res,
        itmvto_cotas.vl_itmvto_cotas
   from dbamv.cota,
        dbamv.cota_competencia,
        dbamv.itmvto_cotas,
        dbamv.itent_pro,
        dbamv.produto,
        dbamv.especie,
        dbamv.item_res
  where to_char(cota_competencia.dt_competencia,'MMYYYY') = to_char( :CAMPO_COMPET ,'MMYYYY')
    and cota.cd_cota = cota_competencia.cd_cota
    and cota_competencia.cd_competencia = itmvto_cotas.cd_competencia
    and itmvto_cotas.cd_itent_pro = itent_pro.cd_itent_pro
    and itent_pro.cd_produto = produto.cd_produto
    and produto.cd_especie = especie.cd_especie
    and especie.cd_item_res = item_res.cd_item_res
    and Cota.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
    :CAMPO_CF_COTA
)a
  group by a.cd_cota,
           a.ds_cota,
           a.cd_item_res,
           a.ds_item_res
   order by  a.ds_cota,
                 a.ds_item_res
)
ORDER BY DS_COTA, DS_ITEM_RES
