--R_COTA_COMPETENCIA_ANA
select cota.cd_cota,
          cota.ds_cota,
          cota.cd_usuario,
          cota_competencia.cd_competencia,
          cota_competencia.vl_orcado
  from dbamv.cota,
         dbamv.cota_competencia
where cota.cd_cota = cota_competencia.cd_cota
   and  to_char(cota_competencia.dt_competencia,'MM/YYYY') =  to_char( :CAMPO_COMPET ,'MM/YYYY')
   and  cota.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
   and Cota.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA


/*----------------------------------------------------------------------------*/
--R_COTA_COMPETENCIA_ANA_R_DESCRICAO_Q_ITMVTO_COTAS
SELECT * FROM (
SELECT * FROM (
select TRUNC(itmvto_cotas.dh_itmvto_cotas)  data,
       'ENTRADA DE PRODUTO Nº '||ent_pro.cd_ent_pro ||' '||produto.ds_produto  descricao,
       (nvl(vl_ultima_entrada,0) *  itent_pro.qt_entrada) vl_debito,
       itmvto_cotas.cd_competencia   cd_competencia,
       0    vl_credito,
       itmvto_cotas.cd_itmvto_cotas  valor
  from dbamv.cota_competencia,
       dbamv.itmvto_cotas,
       dbamv.itent_pro,
       dbamv.ent_pro,
       dbamv.produto,
       dbamv.estoque
 where itmvto_cotas.cd_itent_pro is not null
   and cota_competencia.cd_competencia = itmvto_cotas.cd_competencia
   and itmvto_cotas.cd_produto   = produto.cd_produto
   and itmvto_cotas.cd_itent_pro = itent_pro.cd_itent_pro
   and itent_pro.cd_ent_pro      = ent_pro.cd_ent_pro
   and Estoque.Cd_Estoque       = Ent_Pro.Cd_Estoque
   and Estoque.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
union
select TRUNC(con_pag.dt_emissao)  data,
       ltrim(decode(con_pag.ds_con_pag,'',con_pag.ds_fornecedor ||' Nº '||con_pag.cd_con_pag,con_pag.ds_con_pag)) descricao,
       itmvto_cotas.vl_itmvto_cotas vl_debito,
       itmvto_cotas.cd_competencia   cd_competencia,
       0    vl_credito,
       itmvto_cotas.cd_itmvto_cotas  valor
  from dbamv.cota_competencia,
       dbamv.itmvto_cotas,
       dbamv.con_pag,
       dbamv.ratcon_pag,
        dbamv.cota 
 where itmvto_cotas.cd_ratcon_pag is not null
   and cota_competencia.cd_competencia = itmvto_cotas.cd_competencia
   and itmvto_cotas.cd_ratcon_pag   = ratcon_pag.cd_ratcon_pag
   and ratcon_pag.cd_con_pag        = con_pag.cd_con_pag
   and con_pag.cd_previsao is null
   and Cota.cd_cota = Cota_Competencia.cd_cota
   and to_char(cota_competencia.dt_competencia,'YYYYMM') = to_char( :CAMPO_COMPET ,'YYYYMM')
   and Cota.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
union
SELECT  trunc(ord_com.dt_ord_com) data,
       'PREVISÃO REF.ORDEM DE COMPRA Nº '||previsao.cd_ord_com descricao,
       ord_dupl.vl_parcela vl_debito,
       ord_dupl_cota.cd_competencia   cd_competencia,
       0    vl_credito,
       previsao.cd_ord_com  valor
   FROM dbamv.ord_dupl_cota
,dbamv.ord_com
,dbamv.previsao
,dbamv.cota_competencia
,dbamv.cota
,dbamv.ord_dupl
WHERE ord_com.cd_ord_com = previsao.cd_ord_com
and ord_com.cd_ord_com = ord_dupl_cota.cd_ord_com
and cota.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and cota_competencia.cd_competencia = ord_dupl_cota.cd_competencia
and cota.cd_cota = cota_competencia.cd_cota
and  to_char(ord_com.dt_ord_com,'YYYYMM') = to_char( :CAMPO_COMPET  ,'YYYYMM')
and previsao.vl_previsto > 0 
and cota.cd_cota IN (select cota.cd_cota  
	                                               from dbamv.cota
	                                              , dbamv.cota_competencia 	cotaComp
	                                              , dbamv.cota_resp_autoriza	cotaResp
	                                                where cota.cd_cota   = cotaComp.cd_cota
	                                                and cota.cd_cota    = cotaResp.cd_cota
	                                                and cotaResp.cd_usuario_autoriza = user)
and previsao.cd_ord_com not in (select cd_ord_com from dbamv.ent_pro where ent_pro.cd_ord_com = ord_com.cd_ord_com) 
and ord_dupl.cd_ord_com = ord_com.cd_ord_com
and to_char(cota_competencia.dt_competencia ,'yyyymm')= to_char(ord_dupl.dt_vencimento ,'yyyymm')
union
select TRUNC(dt_movimentacao)  data,
       ds_movimentacao  descricao,
       0                vl_credito,
       cd_competencia_transf   cd_competencia,
       vl_movimentacao  vl_debito,
       mov_cota_competencia.cd_mov_cota  valor
  from dbamv.mov_cota_competencia
union
select TRUNC(dt_movimentacao)  data,
       ds_movimentacao  descricao,
       vl_movimentacao  vl_credito,
       cd_competencia   cd_competencia,
       0                vl_debito ,
       mov_cota_competencia.cd_mov_cota  valor
  from dbamv.mov_cota_competencia
)
WHERE CD_COMPETENCIA = :CAMPO_CD_COMPETENCIA
)
ORDER BY DATA,CD_COMPETENCIA
