--R_TRANSFERENCIA
select mcc.cd_competencia,
          mcc.cd_competencia_transf,
          mcc.ds_movimentacao,
          mcc.vl_movimentacao,
          mcc.dt_movimentacao
  from dbamv.mov_cota_competencia mcc,
         dbamv.cota_competencia,
         dbamv.cota
 where cota.cd_cota = cota_competencia.cd_cota
   and   cota_competencia.cd_competencia = mcc.cd_competencia
   and   mcc.cd_competencia_transf is not null
   and   to_char(cota_competencia.dt_competencia,'MMYYYY') = to_char( :CAMPO_COMPET,'MMYYYY')
   and Cota.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
