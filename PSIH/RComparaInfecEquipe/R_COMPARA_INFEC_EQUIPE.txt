--R_COMPARA_INFEC_EQUIPE
SELECT CODIGO, EQUIPE_MEDICA, SUM(N_INFECCOES), SUM(N_CIRURGIAS), (SUM(N_INFECCOES) * 100 / DECODE(SUM(N_CIRURGIAS),0,1,SUM(N_CIRURGIAS))) TAXA_INFEC
FROM (
SELECT EQUIPE_MEDICA.CD_EQUIPE_MEDICA   CODIGO,
       DS_EQUIPE_MEDICA                 EQUIPE_MEDICA,
       COUNT(CD_REG_INF)                N_INFECCOES,
       0 N_CIRURGIAS
  FROM DBAMV.REG_INF,
       DBAMV.AVISO_CIRURGIA,
       DBAMV.CIRURGIA_AVISO,
       DBAMV.EQUIPE_MEDICA,
       DBAMV.CIRURGIA
WHERE  REG_INF.CD_AVISO_CIRURGIA            = AVISO_CIRURGIA.CD_AVISO_CIRURGIA
  AND  AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
  AND  AVISO_CIRURGIA.CD_EQUIPE_MEDICA  = EQUIPE_MEDICA.CD_EQUIPE_MEDICA
  AND  CIRURGIA_AVISO.CD_CIRURGIA              = CIRURGIA.CD_CIRURGIA
  AND  AVISO_CIRURGIA.TP_SITUACAO              = 'R'
  AND  CIRURGIA.TP_CLASSIF_PSIH                    = 'C'
  AND REG_INF.TP_SIT IN ('C','T') 
      AND trunc(REG_INF.DT_REG_INF) between :CAMPO_DT_INICIAL group by EQUIPE_MEDICA.CD_EQUIPE_MEDICA,
               DS_EQUIPE_MEDICA
UNION
SELECT EQUIPE_MEDICA.CD_EQUIPE_MEDICA   CODIGO,
             DS_EQUIPE_MEDICA                 EQUIPE_MEDICA,
             0 N_INFECCOES,
             COUNT(AVISO_CIRURGIA.CD_AVISO_CIRURGIA) N_CIRURGIAS
  FROM
       DBAMV.AVISO_CIRURGIA,
       DBAMV.CIRURGIA_AVISO,
       DBAMV.EQUIPE_MEDICA,
       DBAMV.CIRURGIA
WHERE
       AVISO_CIRURGIA.CD_AVISO_CIRURGIA     = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
  AND  AVISO_CIRURGIA.CD_EQUIPE_MEDICA = EQUIPE_MEDICA.CD_EQUIPE_MEDICA
  AND  CIRURGIA_AVISO.CD_CIRURGIA             = CIRURGIA.CD_CIRURGIA
  AND  AVISO_CIRURGIA.TP_SITUACAO             = 'R'
  AND  CIRURGIA.TP_CLASSIF_PSIH                   = 'C'
  AND trunc(AVISO_CIRURGIA.DT_REALIZACAO) between :CAMPO_DT_INICIAL group by EQUIPE_MEDICA.CD_EQUIPE_MEDICA,
               DS_EQUIPE_MEDICA
)
GROUP BY CODIGO, EQUIPE_MEDICA
