--R_PACIENTE_TIPO_MICRORGANISMO
SELECT *
FROM
  (SELECT
     (SELECT PACIENTE.CD_PACIENTE ||' - ' || PACIENTE.NM_PACIENTE 
      FROM DBAMV.PACIENTE,
           DBAMV.ATENDIME
      WHERE PACIENTE.CD_PACIENTE = ATENDIME.CD_PACIENTE
        AND ATENDIME.CD_ATENDIMENTO = PL.CD_ATENDIMENTO) PACIENTE,
      (SELECT PACIENTE.NM_SOCIAL_PACIENTE 
      FROM DBAMV.PACIENTE,
           DBAMV.ATENDIME
      WHERE PACIENTE.CD_PACIENTE = ATENDIME.CD_PACIENTE
        AND ATENDIME.CD_ATENDIMENTO = PL.CD_ATENDIMENTO
        AND  NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE,
     (SELECT PACIENTE.NM_PACIENTE
      FROM DBAMV.PACIENTE,
           DBAMV.ATENDIME
      WHERE PACIENTE.CD_PACIENTE= ATENDIME.CD_PACIENTE
        AND ATENDIME.CD_ATENDIMENTO = PL.CD_ATENDIMENTO) DS_PACIENTE,
          G.CD_GERMES CD_GERME,
          G.DS_GERME GERME,
          '1' QUANTIDADE
   FROM PED_LAB PL,
        GERMES_CULTURA GC,
        GERMES G
   WHERE PL.CD_PED_LAB = GC.CD_PED_LAB
     AND GC.CD_GERMES = G.CD_GERMES
     AND PL.DT_EXAME BETWEEN :CAMPO_DT_INICIAL AND 
     AND PL.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
UNION SELECT
     (SELECT PACIENTE.CD_PACIENTE ||' - ' || PACIENTE.NM_PACIENTE
      FROM DBAMV.PACIENTE,
           DBAMV.ATENDIME
      WHERE PACIENTE.CD_PACIENTE= ATENDIME.CD_PACIENTE
        AND ATENDIME.CD_ATENDIMENTO = CM.CD_ATENDIMENTO) PACIENTE,
      (SELECT PACIENTE.NM_SOCIAL_PACIENTE
      FROM DBAMV.PACIENTE,
           DBAMV.ATENDIME
      WHERE PACIENTE.CD_PACIENTE = ATENDIME.CD_PACIENTE
        AND ATENDIME.CD_ATENDIMENTO = CM.CD_ATENDIMENTO
        AND  NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE,
     (SELECT PACIENTE.NM_PACIENTE
      FROM DBAMV.PACIENTE,
           DBAMV.ATENDIME
      WHERE PACIENTE.CD_PACIENTE = ATENDIME.CD_PACIENTE
        AND ATENDIME.CD_ATENDIMENTO = CM.CD_ATENDIMENTO) DS_PACIENTE,
                G.CD_GERMES CD_GERME,
                G.DS_GERME GERME,
                '1' QUANTIDADE
   FROM DBAMV.ATENDIME ATE,
        DBAMV.CULTURA_MANUAL CM,
        DBAMV.GERMES_CULT_MANUAL GCM,
        DBAMV.GERMES G
   WHERE CM.CD_CULTURA_MANUAL = GCM.CD_CULTURA_MANUAL
     AND GCM.CD_GERMES = G.CD_GERMES
     AND CM.TP_RESULTADO = 'P'
     AND ATE.CD_ATENDIMENTO = CM.CD_ATENDIMENTO 
     AND CM.DT_EXAME BETWEEN :CAMPO_DT_INICIAL AND 
     AND ATE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
     :CAMPO_CF_GERMES
  )
ORDER BY GERME,
         DS_PACIENTE
