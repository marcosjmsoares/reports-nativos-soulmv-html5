--R_QT_PRO_INV
SELECT CD_PRO_INVASIVO
      ,DS_PRO_INVASIVO
      ,SUM(QT_ANTES)  QT_ANTES
      ,SUM(QT_DEPOIS) QT_DEPOIS
      ,SUM(QT_ANTES) + SUM(QT_DEPOIS) QT_TOTAL
      ,SUM(TOTAL_GERAL) TOTAL_GERAL
      ,DECODE( SUM(TOTAL_GERAL), 0,0, SUM(TOTAL_REG_INF)/SUM(TOTAL_GERAL))*100 TX_INFEC_PORTARIA_MS_2616_98
      ,SUM(PRO_INV_DIA) PRO_INV_DIA
      ,(DECODE( SUM(PRO_INV_DIA), 0,0, SUM(QT_REG_INF)/SUM(PRO_INV_DIA))*1000) TX_INFEC_NNISS
      ,SUM(QT_REG_INF) QT_REG_INF
FROM(
  SELECT PRO_INVASIVO.CD_PRO_INVASIVO
        ,PRO_INVASIVO.DS_PRO_INVASIVO
        ,COUNT(REG_INF.CD_REG_INF)  QT_ANTES
        ,0 QT_DEPOIS
        ,0 QT_REG_INF
        ,0 QT_ALTA
        ,0 TOTAL_GERAL
        ,0 TOTAL_REG_INF
        ,0 PRO_INV_DIA
   FROM DBAMV.MOV_INV
       ,DBAMV.REG_INF
       ,DBAMV.PRO_INVASIVO
       ,DBAMV.ATENDIME
       ,DBAMV.LEITO 
  WHERE MOV_INV.CD_ATENDIMENTO       = REG_INF.CD_ATENDIMENTO
    AND MOV_INV.CD_PRO_INVASIVO      = PRO_INVASIVO.CD_PRO_INVASIVO
    AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
    AND ATENDIME.CD_ATENDIMENTO      = REG_INF.CD_ATENDIMENTO
    AND REG_INF.CD_LEITO             = LEITO.CD_LEITO(+) 
    AND atendime.tp_atendimento      IN  ('I', 'A', 'E', 'U')
    AND REG_INF.DT_REG_INF BETWEEN :CAMPO_DT_INICIAL AND 
    AND REG_INF.DT_REG_INF           < MOV_INV.DT_MOV_INV
GROUP BY PRO_INVASIVO.CD_PRO_INVASIVO
        ,PRO_INVASIVO.DS_PRO_INVASIVO
UNION ALL
 SELECT PRO_INVASIVO.CD_PRO_INVASIVO
       ,PRO_INVASIVO.DS_PRO_INVASIVO
       ,0 QT_ANTES
       ,COUNT(REG_INF.CD_REG_INF)  QT_DEPOIS
       ,0 QT_REG_INF
       ,0 QT_ALTA
       ,0 TOTAL_GERAL
       ,0 TOTAL_REG_INF
       ,0 PRO_INV_DIA
   FROM DBAMV.MOV_INV
       ,DBAMV.REG_INF
       ,DBAMV.PRO_INVASIVO
       ,DBAMV.ATENDIME
       ,DBAMV.LEITO 
  WHERE MOV_INV.CD_ATENDIMENTO      = REG_INF.CD_ATENDIMENTO
    AND MOV_INV.CD_PRO_INVASIVO     = PRO_INVASIVO.CD_PRO_INVASIVO
    AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
    AND ATENDIME.CD_ATENDIMENTO     = REG_INF.CD_ATENDIMENTO
    AND REG_INF.CD_LEITO            = LEITO.CD_LEITO(+) 
    AND atendime.tp_atendimento     IN  ('I', 'A', 'E', 'U')
    AND REG_INF.DT_REG_INF BETWEEN :CAMPO_DT_INICIAL AND 
    AND REG_INF.DT_REG_INF          >= MOV_INV.DT_MOV_INV
  GROUP BY PRO_INVASIVO.CD_PRO_INVASIVO
          ,PRO_INVASIVO.DS_PRO_INVASIVO
UNION ALL
 SELECT PRO_INVASIVO.CD_PRO_INVASIVO
       ,PRO_INVASIVO.DS_PRO_INVASIVO
       ,0 QT_ANTES
       ,0 QT_DEPOIS
       ,COUNT(DISTINCT REG_INF.CD_REG_INF) QT_REG_INF
       ,0 QT_ALTA
       ,0 TOTAL_GERAL
       ,0 TOTAL_REG_INF
       ,0 PRO_INV_DIA
   FROM DBAMV.MOV_INV
       ,DBAMV.REG_INF
       ,DBAMV.PRO_INVASIVO
       ,DBAMV.ATENDIME
       ,DBAMV.LEITO 
  WHERE MOV_INV.CD_REG_INF              = REG_INF.CD_REG_INF
    AND MOV_INV.CD_PRO_INVASIVO         = PRO_INVASIVO.CD_PRO_INVASIVO
    AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
    AND ATENDIME.CD_ATENDIMENTO         = REG_INF.CD_ATENDIMENTO
    AND REG_INF.CD_LEITO                = LEITO.CD_LEITO(+) 
    AND ATENDIME.TP_ATENDIMENTO         IN  ('I', 'A', 'E', 'U')
    AND REG_INF.DT_REG_INF BETWEEN :CAMPO_DT_INICIAL GROUP BY PRO_INVASIVO.CD_PRO_INVASIVO
          ,PRO_INVASIVO.DS_PRO_INVASIVO
UNION ALL
 SELECT PRO_INVASIVO.CD_PRO_INVASIVO
       ,PRO_INVASIVO.DS_PRO_INVASIVO
       ,0 QT_ANTES
       ,0 QT_DEPOIS
       ,0 QT_REG_INF
       ,COUNT(ATENDIME.CD_ATENDIMENTO) QT_ALTA
       ,0 TOTAL_GERAL
       ,0 TOTAL_REG_INF
       ,0 PRO_INV_DIA
   FROM DBAMV.MOV_INV
       ,DBAMV.REG_INF
       ,DBAMV.PRO_INVASIVO
       ,DBAMV.ATENDIME
       ,DBAMV.LEITO 
  WHERE MOV_INV.CD_REG_INF          = REG_INF.CD_REG_INF
    AND MOV_INV.CD_PRO_INVASIVO     = PRO_INVASIVO.CD_PRO_INVASIVO
    AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
    AND ATENDIME.CD_ATENDIMENTO     = REG_INF.CD_ATENDIMENTO
    AND REG_INF.CD_LEITO            = LEITO.CD_LEITO(+)  
    AND ATENDIME.TP_ATENDIMENTO     IN  ('I', 'A', 'E', 'U')
    AND ATENDIME.DT_ALTA BETWEEN :CAMPO_DT_INICIAL GROUP BY PRO_INVASIVO.CD_PRO_INVASIVO
          ,PRO_INVASIVO.DS_PRO_INVASIVO
UNION ALL
 SELECT PRO_INVASIVO.CD_PRO_INVASIVO
       ,PRO_INVASIVO.DS_PRO_INVASIVO
       ,0 QT_ANTES
       ,0 QT_DEPOIS
       ,0 QT_REG_INF
       ,0 QT_ALTA
       ,COUNT(DISTINCT REG_INF.CD_REG_INF) TOTAL_GERAL
       ,0 TOTAL_REG_INF
       ,0 PRO_INV_DIA
   FROM DBAMV.MOV_INV
       ,DBAMV.PRO_INVASIVO
       ,DBAMV.ATENDIME
       ,DBAMV.REG_INF
       ,DBAMV.LEITO 
  WHERE REG_INF.CD_ATENDIMENTO      = ATENDIME.CD_ATENDIMENTO
    AND MOV_INV.CD_ATENDIMENTO      = ATENDIME.CD_ATENDIMENTO
    AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
    AND MOV_INV.CD_PRO_INVASIVO     = PRO_INVASIVO.CD_PRO_INVASIVO
    AND REG_INF.CD_LEITO            = LEITO.CD_LEITO(+) 
    AND ATENDIME.TP_ATENDIMENTO     IN  ('I', 'A', 'E', 'U')
    AND REG_INF.DT_REG_INF BETWEEN :CAMPO_DT_INICIAL GROUP BY PRO_INVASIVO.CD_PRO_INVASIVO
          ,PRO_INVASIVO.DS_PRO_INVASIVO
UNION ALL
 SELECT PRO_INVASIVO.CD_PRO_INVASIVO
       ,PRO_INVASIVO.DS_PRO_INVASIVO
       ,0 QT_ANTES
       ,0 QT_DEPOIS
       ,0 QT_REG_INF
       ,0 QT_ALTA
       ,0 TOTAL_GERAL
       ,COUNT(DISTINCT CD_MOV_INV) TOTAL_REG_INF
       ,0 PRO_INV_DIA
   FROM DBAMV.MOV_INV
       ,DBAMV.REG_INF
       ,DBAMV.PRO_INVASIVO
       ,DBAMV.ATENDIME
       ,DBAMV.LEITO 
  WHERE MOV_INV.CD_REG_INF           = REG_INF.CD_REG_INF
    AND MOV_INV.CD_PRO_INVASIVO      = PRO_INVASIVO.CD_PRO_INVASIVO
    AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
    AND MOV_INV.CD_ATENDIMENTO       = ATENDIME.CD_ATENDIMENTO
    AND REG_INF.CD_LEITO             = LEITO.CD_LEITO(+) 
    AND ATENDIME.TP_ATENDIMENTO      IN  ('I', 'A', 'E', 'U')
    AND REG_INF.DT_REG_INF BETWEEN :CAMPO_DT_INICIAL GROUP BY PRO_INVASIVO.CD_PRO_INVASIVO
          ,PRO_INVASIVO.DS_PRO_INVASIVO
UNION ALL
 SELECT PRO_INVASIVO.CD_PRO_INVASIVO
       ,PRO_INVASIVO.DS_PRO_INVASIVO
       ,0 QT_ANTES
       ,0 QT_DEPOIS
       ,0 QT_REG_INF
       ,0 QT_ALTA
       ,0 TOTAL_GERAL
       ,0 TOTAL_REG_INF
       ,DBAMV.FNC_PSIH_CONTA_DIAS_PROC_INV(PRO_INVASIVO.CD_PRO_INVASIVO,:CAMPO_DT_INICIAL,:CAMPO_DT_FINAL) PRO_INV_DIA
   FROM DBAMV.MOV_INV
       ,DBAMV.PRO_INVASIVO
       ,DBAMV.ATENDIME
       ,DBAMV.REG_INF
       ,DBAMV.LEITO 
  WHERE MOV_INV.CD_ATENDIMENTO      = ATENDIME.CD_ATENDIMENTO
    AND MOV_INV.CD_PRO_INVASIVO     = PRO_INVASIVO.CD_PRO_INVASIVO
    AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
    AND MOV_INV.CD_REG_INF          = REG_INF.CD_REG_INF(+)  
    AND REG_INF.CD_LEITO            = LEITO.CD_LEITO(+) 
    AND ATENDIME.TP_ATENDIMENTO     IN  ('I', 'A', 'E', 'U')
  GROUP BY PRO_INVASIVO.CD_PRO_INVASIVO
          ,PRO_INVASIVO.DS_PRO_INVASIVO
 )
GROUP BY  CD_PRO_INVASIVO
         ,DS_PRO_INVASIVO
