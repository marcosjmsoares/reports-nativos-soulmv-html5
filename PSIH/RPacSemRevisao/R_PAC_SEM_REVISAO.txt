--R_PAC_SEM_REVISAO
SELECT     ATENDIME.CD_ATENDIMENTO
          ,ATENDIME.DT_ALTA
          ,PACIENTE.CD_PACIENTE
          ,INITCAP(PACIENTE.NM_PACIENTE) NM_PACIENTE
          ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  CD_PACIENTE) NM_SOCIAL_PACIENTE
          ,ATENDIME.DT_ATENDIMENTO
          ,ATENDIME.CD_LEITO
          ,INITCAP(LEITO.DS_LEITO) DS_LEITO
          ,ATENDIME.CD_PRESTADOR
          ,INITCAP(PRESTADOR.NM_PRESTADOR) NM_PRESTADOR
          ,1 conta
FROM DBAMV.ATENDIME
          ,DBAMV.PACIENTE
          ,DBAMV.LEITO
          ,DBAMV.PRESTADOR
          ,(SELECT RI.CD_ATENDIMENTO
            FROM DBAMV.REG_INF  RI
                ,DBAMV.ATENDIME  ATEN
           WHERE RI.TP_SIT IN ('T','C')
            AND RI.CD_ATENDIMENTO = ATEN.CD_ATENDIMENTO
            AND  RI.DT_REG_INF BETWEEN  :CAMPO_DT_INICIAL AND 
            AND  ATEN.CD_MULTI_EMPRESA    = :CAMPO_CD_MULTI_EMPRESA )  REG_INF
WHERE ATENDIME.CD_PACIENTE              = PACIENTE.CD_PACIENTE
      AND  ATENDIME.CD_MULTI_EMPRESA    = :CAMPO_CD_MULTI_EMPRESA
      AND ATENDIME.CD_LEITO             = LEITO.CD_LEITO
      AND ATENDIME.CD_ATENDIMENTO       = REG_INF.CD_ATENDIMENTO(+)
      AND ATENDIME.CD_PRESTADOR         = PRESTADOR.CD_PRESTADOR(+)
      AND ATENDIME.TP_ATENDIMENTO       = ('I')
      AND ATENDIME.DT_ALTA              IS NOT NULL
      AND ATENDIME.DT_REVISAO           IS NULL
      AND ATENDIME.dt_alta     BETWEEN  :CAMPO_DT_INICIAL ORDER BY DT_ATENDIMENTO
