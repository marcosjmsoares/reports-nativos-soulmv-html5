--R_AVALIACAO_PRESCRICAO
SELECT * FROM (
     SELECT REG.CD_UNID_INT,
            REG.DS_UNID_INT,
            REG.CD_PRESTADOR,
            REG.NM_PRESTADOR,
            REG.CD_TIP_PRESC,
            REG.DS_TIP_PRESC,
            ROUND((REG.QTD_TOTAL_DROGA_ADEQUADA/REG.QTD_TOTAL)*100, 2) QTD_TOTAL_DROGA_ADEQUADA,
            ROUND((REG.QTD_TOTAL_DROGA_INADEQUADA/REG.QTD_TOTAL)*100, 2) QTD_TOTAL_DROGA_INADEQUADA,
            ROUND((REG.QTD_TOTAL_DOSE_ADEQUADA/REG.QTD_TOTAL)*100, 2) QTD_TOTAL_DOSE_ADEQUADA,
            ROUND((REG.QTD_TOTAL_DOSE_INADEQUADA/REG.QTD_TOTAL)*100, 2) QTD_TOTAL_DOSE_INADEQUADA,
            ROUND((REG.QTD_TOTAL_TEMPO_ADEQUADO/REG.QTD_TOTAL)*100, 2) QTD_TOTAL_TEMPO_ADEQUADO,
            ROUND((REG.QTD_TOTAL_TEMPO_INADEQUADO/REG.QTD_TOTAL)*100, 2) QTD_TOTAL_TEMPO_INADEQUADO,
            ROUND((REG.QTD_TOTAL_VIA_ADEQUADA/REG.QTD_TOTAL)*100, 2) QTD_TOTAL_VIA_ADEQUADA,
            ROUND((REG.QTD_TOTAL_VIA_INADEQUADA/REG.QTD_TOTAL)*100, 2) QTD_TOTAL_VIA_INADEQUADA
       FROM ( SELECT PRE_MED.CD_UNID_INT    ,
             UNID_INT.DS_UNID_INT   ,
             PRE_MED.CD_PRESTADOR   ,
             PRESTADOR.NM_PRESTADOR ,
             ITPRE_MED.CD_TIP_PRESC  ,
             TIP_PRESC.DS_TIP_PRESC ,
             COUNT(ITPRE_MED.CD_TIP_PRESC)  QTD_TOTAL ,
             SUM(DECODE(AAP.TP_DROGA, 'A', 1, 0)) QTD_TOTAL_DROGA_ADEQUADA,
             SUM(DECODE(AAP.TP_DROGA, 'I', 1, 0)) QTD_TOTAL_DROGA_INADEQUADA,
             SUM(DECODE(AAP.TP_DOSE , 'A', 1, 0)) QTD_TOTAL_DOSE_ADEQUADA,
             SUM(DECODE(AAP.TP_DOSE , 'I', 1, 0)) QTD_TOTAL_DOSE_INADEQUADA,
             SUM(DECODE(AAP.TP_TEMPO, 'A', 1, 0)) QTD_TOTAL_TEMPO_ADEQUADO,
             SUM(DECODE(AAP.TP_TEMPO, 'I', 1, 0)) QTD_TOTAL_TEMPO_INADEQUADO,
             SUM(DECODE(AAP.TP_VIA  , 'A', 1, 0)) QTD_TOTAL_VIA_ADEQUADA,
             SUM(DECODE(AAP.TP_VIA  , 'I', 1, 0)) QTD_TOTAL_VIA_INADEQUADA
     FROM DBAMV.ITPRE_MED                      ITPRE_MED ,
                DBAMV.PRE_MED                        PRE_MED   ,
	DBAMV.TIP_PRESC                      TIP_PRESC ,
	DBAMV.PRESTADOR                      PRESTADOR ,
	DBAMV.UNID_INT                       UNID_INT  ,
	DBAMV.SETOR							 SETOR	   ,
	DBAMV.AVALIACAO_ANTIMICROBIANO_PRSCT AAP       ,
        DBAMV.ANT_PRO
 WHERE AAP.SN_AVALIAVEL = 'S'
       AND PRE_MED.CD_UNID_INT    = UNID_INT.CD_UNID_INT
       AND PRE_MED.CD_PRESTADOR   = PRESTADOR.CD_PRESTADOR
       AND ITPRE_MED.CD_TIP_PRESC = TIP_PRESC.CD_TIP_PRESC
       AND ITPRE_MED.CD_ITPRE_MED = AAP.CD_ITPRE_MED
       AND ITPRE_MED.CD_PRE_MED   = PRE_MED.CD_PRE_MED
       AND ANT_PRO.CD_PRODUTO     = TIP_PRESC.CD_PRODUTO
       AND 	  UNID_INT.CD_SETOR = SETOR.CD_SETOR
	AND       SETOR.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
  GROUP BY  PRE_MED.CD_UNID_INT    ,
                    UNID_INT.DS_UNID_INT   ,
                    PRE_MED.CD_PRESTADOR   ,
                    PRESTADOR.NM_PRESTADOR ,
                   ITPRE_MED.CD_TIP_PRESC ,
                   TIP_PRESC.DS_TIP_PRESC  ) REG
)
ORDER BY CD_UNID_INT,DS_UNID_INT,
CD_PRESTADOR,NM_PRESTADOR
