--R_PAC_REGRESSO
SELECT DISTINCT ATENDIME.DT_RETORNO
      ,ATENDIME.CD_ATENDIMENTO
      ,PACIENTE.CD_PACIENTE
      ,INITCAP (PACIENTE.NM_PACIENTE) NM_PACIENTE
      ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND P.CD_PACIENTE = paciente.CD_PACIENTE) NM_SOCIAL_PACIENTE
      ,ATENDIME.DT_ATENDIMENTO
      ,ATENDIME.DT_ALTA
      ,CONVENIO.CD_CONVENIO
      ,CONVENIO.NM_CONVENIO
      ,INITCAP(PRESTADOR.NM_PRESTADOR) NM_PRESTADOR
      ,INITCAP(UNID_INT.DS_UNID_INT) DS_UNID_INT
      ,INITCAP(LEITO.DS_LEITO) DS_LEITO
  FROM DBAMV.ATENDIME
      ,DBAMV.PACIENTE
      ,DBAMV.CONVENIO
      ,DBAMV.PRESTADOR
      ,DBAMV.UNID_INT
      ,DBAMV.LEITO
      ,(SELECT CD_ATENDIMENTO
              ,DT_ATENDIMENTO
              ,DT_ALTA
              ,CD_PACIENTE
          FROM DBAMV.ATENDIME
          WHERE ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA ) NR_PERIODO
 WHERE ATENDIME.CD_PACIENTE  = PACIENTE.CD_PACIENTE
   AND ATENDIME.CD_CONVENIO     = CONVENIO.CD_CONVENIO
   AND ATENDIME.CD_PRESTADOR  = PRESTADOR.CD_PRESTADOR
   AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
   AND ATENDIME.CD_LEITO              = LEITO.CD_LEITO
   AND LEITO.CD_UNID_INT               = UNID_INT.CD_UNID_INT
   AND ATENDIME.CD_ATENDIMENTO <> NR_PERIODO.CD_ATENDIMENTO
   AND ATENDIME.CD_PACIENTE      = NR_PERIODO.CD_PACIENTE
   AND ATENDIME.DT_ATENDIMENTO  > NR_PERIODO.DT_ALTA
  AND (ATENDIME.DT_ATENDIMENTO - NR_PERIODO.DT_ALTA) <= 30
   AND  ATENDIME.DT_ATENDIMENTO BETWEEN :CAMPO_DT_INICIAL ORDER BY DT_RETORNO
