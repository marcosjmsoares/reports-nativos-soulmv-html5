--R_EST_CULTURA_MATERIAL
SELECT * FROM (
Select  DISTINCT
      itcultura.CD_EXA_LAB
     ,itcultura.CD_MATERIAL
     ,itcultura.Nm_Exa_Lab
     ,itcultura.Ds_Material
From dbamv.v_psih_cultura_cultura_manual itcultura,
     DBAMV.REG_INF
Where (itcultura.Tp_Resultado   = 'P'
OR itcultura.Tp_Resultado     = 'C')
AND REG_INF.CD_REG_INF(+) = itcultura.CD_REG_INF
:CAMPO_CF_TP_INFEC
)
ORDER BY CD_MATERIAL
