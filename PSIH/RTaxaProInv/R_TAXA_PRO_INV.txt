--R_TAXA_PRO_INV_GRAFICO
SELECT * FROM (
SELECT CD_PRO_INVASIVO,
        DS_PRO_INVASIVO,
              DESCRICAO,
              SUM(DIAS)
  from(  Select pro_inv_psih.cd_pro_invasivo,
                      pro_invasivo.ds_pro_invasivo,
                      tip_presc.ds_tip_presc descricao,
                      NVL((DT_TERMINO - DT_INICIO +1),0) DIAS
              from dbamv.tip_presc,
                      dbamv.pro_inv_psih,
                      dbamv.pro_invasivo,
                      DBAMV.UNID_INT,
                      DBAMV.MOV_INV,
                      dbamv.leito
            where tip_presc.cd_tip_presc                 = pro_inv_psih.cd_tip_presc
               and pro_invasivo.cd_pro_invasivo      = pro_inv_psih.cd_pro_invasivo
              AND MOV_INV.CD_PRO_INV_PSIH = PRO_INV_PSIH.CD_PRO_INV_PSIH
              and mov_inv.cd_leito                          = leito.cd_leito
              and leito.cd_unid_int                           = unid_int.cd_unid_int
              And Trunc(MOV_INV.DT_MOV_INV) between :CAMPO_DT_INICIAL and 
              AND NVL((DT_TERMINO - DT_INICIO +1),0)  > 0
         UNION
         Select pro_inv_psih.CD_PRO_INVASIVO,
                   cirurgia.ds_cirurgia descricao,
                   pro_invasivo.ds_pro_invasivo,
                   NVL((DT_TERMINO - DT_INICIO +1),0) DIAS
           from dbamv.cirurgia,
                  dbamv.pro_inv_psih,
                  dbamv.pro_invasivo,
                  DBAMV.UNID_INT,
                  DBAMV.MOV_INV,
                  dbamv.leito
       where cirurgia.cd_cirurgia                        = pro_inv_psih.cd_cirurgia
           and pro_invasivo.cd_pro_invasivo      = pro_inv_psih.cd_pro_invasivo
           AND MOV_INV.CD_PRO_INV_PSIH = PRO_INV_PSIH.CD_PRO_INV_PSIH
          and mov_inv.cd_leito                           = leito.cd_leito
          and leito.cd_unid_int                            = unid_int.cd_unid_int
           And Trunc(MOV_INV.DT_MOV_INV) between :CAMPO_DT_INICIAL and 
           AND  NVL((DT_TERMINO - DT_INICIO +1),0)  > 0
           and tp_classif_psih = 'P'
         :CAMPO_CF_GRAFICO_PRO_INVASIVO)
            GROUP BY cd_pro_invasivo, descricao,ds_pro_invasivo
)order by ds_pro_invasivo
