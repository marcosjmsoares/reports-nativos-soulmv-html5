--R_FICHA_INFEC
SELECT * FROM (
SELECT     ATENDIME.CD_ATENDIMENTO
         , ATENDIME.DT_ATENDIMENTO
         , ATENDIME.HR_ATENDIMENTO
         , ATENDIME.DT_ALTA
         , ATENDIME.HR_ALTA
         , PACIENTE.CD_PACIENTE
         , PACIENTE.NM_PACIENTE
         ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  CD_PACIENTE) NM_SOCIAL_PACIENTE
         , PRESTADOR.CD_PRESTADOR
         , PRESTADOR.NM_PRESTADOR
         , LEITO.CD_LEITO
         , LEITO.DS_RESUMO
         , UNID_INT.CD_UNID_INT
         , UNID_INT.DS_UNID_INT
         , SERVICO.CD_SERVICO
         , SERVICO.DS_SERVICO
         , REG_INF.DT_REG_INF
         , REG_INF.CD_CID
         , REG_INF.TP_INFEC
         , REG_INF.TP_MODO
         , REG_INF.TP_SIT
         , TOPOGRAFIA.CD_TOPOGRAFIA
         , TOPOGRAFIA.DS_TOPOGRAFIA
         , Nvl(V_CATEGORIA_TOPOGRAFIA.TP_CATEGORIA, TOPOGRAFIA.TP_CATEGORIA) TP_CATEGORIA 
         , LOC_PROCED.DS_LOC_PROCED
      FROM DBAMV.ATENDIME
         , DBAMV.PACIENTE
         , DBAMV.PRESTADOR
         , DBAMV.LEITO
         , DBAMV.UNID_INT
         , DBAMV.SERVICO
         , DBAMV.CID
         , DBAMV.REG_INF
         , DBAMV.TOPOGRAFIA
         , DBAMV.V_CATEGORIA_TOPOGRAFIA
         , DBAMV.LOC_PROCED
WHERE ATENDIME.CD_ATENDIMENTO      LIKE :CAMPO_ATENDIMENTO
   AND ATENDIME.CD_PACIENTE         = PACIENTE.CD_PACIENTE
   AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
   AND ATENDIME.CD_PRESTADOR        = PRESTADOR.CD_PRESTADOR
   AND ATENDIME.CD_LEITO            = LEITO.CD_LEITO
   AND LEITO.CD_UNID_INT            = UNID_INT.CD_UNID_INT
   AND ATENDIME.CD_SERVICO          = SERVICO.CD_SERVICO
   AND ATENDIME.CD_CID              = CID.CD_CID  (+)
   AND REG_INF.CD_ATENDIMENTO(+)       = ATENDIME.CD_ATENDIMENTO
   AND REG_INF.CD_TOPOGRAFIA        = TOPOGRAFIA.CD_TOPOGRAFIA(+)
    AND REG_INF.CD_LOC_PROCED        = LOC_PROCED.CD_LOC_PROCED(+)
  AND TOPOGRAFIA.TP_CATEGORIA = DBAMV.V_CATEGORIA_TOPOGRAFIA.TP_CATEGORIA(+) 
   :CAMPO_CF_MULT_EMPRESA
)
ORDER BY CD_ATENDIMENTO,DT_ATENDIMENTO,HR_ATENDIMENTO,DT_ALTA,HR_ALTA,CD_PACIENTE,NM_PACIENTE,CD_PRESTADOR,NM_PRESTADOR,CD_LEITO,CD_UNID_INT,DS_UNID_INT,CD_SERVICO,DS_SERVICO,DS_RESUMO

--R_FICHA_INFEC_R_1_Q_2
SELECT * FROM (
SELECT * FROM (
SELECT      DISTINCT
                   PED_LAB.CD_ATENDIMENTO
                  ,PED_LAB.CD_PED_LAB
                  ,PRESTADOR.NM_PRESTADOR
                  ,GERMES.CD_GERMES
                  ,GERMES.DS_GERME
                  ,ITCULTURA.CD_EXA_LAB
                  ,EXA_LAB.NM_EXA_LAB
                  ,CULTURA.CD_MATERIAL
                  ,MATERIAL.DS_MATERIAL
                  ,ITCULTURA.CD_ANTIMICROB
                  ,ANTIMICROB.NM_ANTIMICROB
                  ,ITCULTURA.TP_RESULTADO || ' - ' || DBAMV.FNC_PSSD_DS_TP_RES_ANTIMIC_DS(ITCULTURA.TP_RESULTADO) TP_RESULTADO
FROM
                   DBAMV.ITCULTURA
                  ,DBAMV.GERMES
                  ,DBAMV.PED_LAB
                  ,DBAMV.PRESTADOR
                  ,DBAMV.EXA_LAB
                  ,DBAMV.CULTURA
                  ,DBAMV.MATERIAL
                  ,DBAMV.ANTIMICROB
 WHERE    PED_LAB.CD_PED_LAB                 =            ITCULTURA.CD_PED_LAB
      AND     GERMES.CD_GERMES                     =            ITCULTURA.CD_GERMES
      AND     PED_LAB.CD_PRESTADOR           =            PRESTADOR.CD_PRESTADOR
      AND     ITCULTURA.CD_EXA_LAB                =            EXA_LAB.CD_EXA_LAB
      AND     ITCULTURA.CD_EXA_LAB                =            CULTURA.CD_EXA_LAB
      AND     ITCULTURA.CD_PED_LAB                =            CULTURA.CD_PED_LAB
      AND     CULTURA.CD_MATERIAL                 =            MATERIAL.CD_MATERIAL
      AND     ITCULTURA.CD_ANTIMICROB         =             ANTIMICROB.CD_ANTIMICROB
UNION
SELECT DISTINCT CULTURA_MANUAL.CD_ATENDIMENTO
      ,TO_NUMBER(NULL)
      ,PRESTADOR.NM_PRESTADOR
      ,GERMES.CD_GERMES
      ,GERMES.DS_GERME
      ,CULTURA_MANUAL.CD_EXA_LAB
      ,EXA_LAB.NM_EXA_LAB
      ,CULTURA_MANUAL.CD_MATERIAL
      ,MATERIAL.DS_MATERIAL
      ,ITCULTURA_MANUAL.CD_ANTIMICROB
      ,ANTIMICROB.NM_ANTIMICROB
      ,ITCULTURA_MANUAL.TP_RESULTADO || ' - ' || DBAMV.FNC_PSSD_DS_TP_RES_ANTIMIC_DS(ITCULTURA_MANUAL.TP_RESULTADO) TP_RESULTADO
FROM DBAMV.CULTURA_MANUAL
    ,DBAMV.GERMES_CULT_MANUAL
    ,DBAMV.GERMES
    ,DBAMV.ITCULTURA_MANUAL
    ,DBAMV.ATENDIME
    ,DBAMV.PRESTADOR
    ,DBAMV.EXA_LAB
    ,DBAMV.MATERIAL
    ,DBAMV.ANTIMICROB
WHERE CULTURA_MANUAL.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
  AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
  AND CULTURA_MANUAL.CD_EXA_LAB = EXA_LAB.CD_EXA_LAB
  AND CULTURA_MANUAL.CD_MATERIAL = MATERIAL.CD_MATERIAL
  AND cultura_manual.cd_cultura_manual = GERMES_CULT_MANUAL.CD_CULTURA_MANUAL(+)
  AND GERMES_CULT_MANUAL.CD_GERMES = GERMES.CD_GERMES(+)
  AND GERMES_CULT_MANUAL.CD_CULTURA_MANUAL = ITCULTURA_MANUAL.CD_CULTURA_MANUAL(+)
  AND GERMES_CULT_MANUAL.CD_GERMES = ITCULTURA_MANUAL.CD_GERMES(+)
  AND GERMES_CULT_MANUAL.CD_INCIDENCIA_CULTURA = ITCULTURA_MANUAL.CD_INCIDENCIA_CULTURA(+)
  AND ITCULTURA_MANUAL.CD_ANTIMICROB = ANTIMICROB.CD_ANTIMICROB(+)
)
WHERE CD_ATENDIMENTO = :CAMPO_CD_ATENDIMENTO
)
ORDER BY NM_PRESTADOR,CD_PED_LAB,
CD_EXA_LAB,NM_EXA_LAB,CD_MATERIAL,DS_MATERIAL,
CD_GERMES,DS_GERME

--R_FICHA_INFEC_R_5_Q_6
SELECT * FROM (
SELECT * FROM (
SELECT AVISO_CIRURGIA.CD_ATENDIMENTO
      ,F_VERIF_PRESTADOR(AVISO_CIRURGIA.CD_AVISO_CIRURGIA, CIRURGIA_AVISO.CD_CIRURGIA ) cd_prestador
      ,AVISO_CIRURGIA.DT_REALIZACAO
      ,AVISO_CIRURGIA.DT_INICIO_CIRURGIA
      ,AVISO_CIRURGIA.DT_FIM_CIRURGIA
      ,AVISO_CIRURGIA.CD_SAL_CIR
      ,SAL_CIR.DS_SAL_CIR
      ,DECODE(CONVENIO.TP_CONVENIO,'H',pro_fat_sus.cd_sus,CIRURGIA.CD_PRO_FAT) CD_PRO_FAT
      ,Decode( convenio.tp_convenio, 'H', pro_fat_sus.ds_pro_Fat, PRO_FAT.DS_PRO_FAT ) ds_pro_fat
      ,CIRURGIA.TP_CIRURGIA
      ,PRESTADOR.NM_PRESTADOR
      ,DECODE(AVISO_CIRURGIA.TP_NATUREZA, 'L', 'Limpa', 'C', 'Contaminada', 'I', 'Infectada', 'P', 'Potencialmente Contaminada') TP_NATUREZA
FROM    DBAMV.AVISO_CIRURGIA
       ,DBAMV.CIRURGIA_AVISO
       ,DBAMV.CIRURGIA
       ,DBAMV.AGE_CIR
       ,DBAMV.PRO_FAT
       ,DBAMV.PRO_FAT  PRO_FAT_SUS
       ,DBAMV.TAB_SIASUS
       ,DBAMV.CEN_CIR
       ,DBAMV.CONVENIO
       ,DBAMV.SAL_CIR
       ,DBAMV.PRESTADOR
       ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And AVISO_CIRURGIA.CD_AVISO_CIRURGIA  = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
AND     AVISO_CIRURGIA.CD_CEN_CIR         = CEN_CIR.CD_CEN_CIR
AND     AVISO_CIRURGIA.CD_AVISO_CIRURGIA  = AGE_CIR.CD_AVISO_CIRURGIA(+)
AND     CIRURGIA_AVISO.CD_CIRURGIA        = CIRURGIA.CD_CIRURGIA
AND     CIRURGIA_AVISO.CD_CONVENIO        = CONVENIO.CD_CONVENIO(+)
AND     CIRURGIA.CD_PRO_FAT               = PRO_FAT.CD_PRO_FAT(+)
AND     CIRURGIA.CD_PRO_FAT_SUS           = PRO_FAT_SUS.CD_PRO_FAT(+)
AND     CIRURGIA.CD_SSM                   = TAB_SIASUS.CD_SSM(+)
AND     AVISO_CIRURGIA.CD_SAL_CIR         = SAL_CIR.CD_SAL_CIR
AND     AVISO_CIRURGIA.TP_SITUACAO        = 'R'
AND     PRESTADOR.CD_PRESTADOR          = F_VERIF_PRESTADOR(AVISO_CIRURGIA.CD_AVISO_CIRURGIA, CIRURGIA_AVISO.CD_CIRURGIA )
AND     not exists (SELECT CD_PRO_FAT FROM DBAMV.PRO_INVASIVO  where PRO_FAT.CD_PRO_FAT = pro_invasivo.cd_pro_fat )
AND     NOT exists (SELECT CD_PRO_FAT_SUS FROM DBAMV.PRO_INVASIVO where PRO_FAT_sus.CD_PRO_FAT = pro_invasivo.cd_pro_Fat )
)
WHERE CD_ATENDIMENTO = :CAMPO_CD_ATENDIMENTO
)
ORDER BY CD_PRESTADOR,NM_PRESTADOR
