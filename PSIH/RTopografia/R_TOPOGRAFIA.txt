--R_TOPOGRAFIA
SELECT TOPOGRAFIA.CD_TOPOGRAFIA CODIGO,
       TOPOGRAFIA.DS_TOPOGRAFIA DESCRICAO,
       CLASSIFICACAO_IRAS.CD_SIGLA_IRAS_APRES || ' - ' || CLASSIFICACAO_IRAS.DS_SITIO_PRINCIPAL TP_CATEGORIA
  FROM DBAMV.TOPOGRAFIA,
       DBAMV.CLASSIFICACAO_IRAS
WHERE TOPOGRAFIA.TP_CATEGORIA = CLASSIFICACAO_IRAS.CD_SIGLA_IRAS(+)
      AND  NVL(TOPOGRAFIA.SN_ATIVO,'S')  = 'S'
 ORDER BY DESCRICAO
