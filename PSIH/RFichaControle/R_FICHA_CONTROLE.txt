--R_FICHA_CONTROLE
SELECT ALL ATENDIME.CD_PACIENTE
          ,PACIENTE.NM_PACIENTE
          ,ATENDIME.CD_ATENDIMENTO
          ,PACIENTE.DT_NASCIMENTO
          ,ATENDIME.CD_LEITO
          ,LEITO.DS_LEITO
          ,PACIENTE.NR_FONE
          ,ATENDIME.CD_CID
          ,ATENDIME.CD_SERVICO
          ,ORI_ATE.CD_ORI_ATE
          ,LEITO.cd_unid_int
          ,ORI_ATE.DS_ORI_ATE
          ,DECODE(PRESTADOR_AVISO.CD_PRESTADOR,NULL,ATENDIME.CD_PRESTADOR,PRESTADOR_AVISO.CD_PRESTADOR) CD_PRESTADOR
          ,PRESTADOR.NM_PRESTADOR
          ,CID.DS_CID
          ,SERVICO.DS_SERVICO
          ,AVISO_CIRURGIA.TP_NATUREZA
          ,CIRURGIA.DS_CIRURGIA
     FROM  DBAMV.ORI_ATE
          ,DBAMV.PRESTADOR
          ,DBAMV.PRESTADOR_AVISO
          ,DBAMV.CID
          ,DBAMV.SERVICO
          ,DBAMV.AVISO_CIRURGIA
          ,DBAMV.CIRURGIA_AVISO
          ,DBAMV.CIRURGIA
          ,DBAMV.ATENDIME
          ,DBAMV.PACIENTE
          ,DBAMV.LEITO
WHERE ATENDIME.CD_ORI_ATE                  = ORI_ATE.CD_ORI_ATE
  AND PACIENTE.CD_PACIENTE                 = ATENDIME.CD_PACIENTE
  AND PRESTADOR.CD_PRESTADOR               = DECODE(PRESTADOR_AVISO.CD_PRESTADOR,NULL,ATENDIME.CD_PRESTADOR,PRESTADOR_AVISO.CD_PRESTADOR)
  AND LEITO.CD_LEITO                       = ATENDIME.CD_LEITO
  AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
  AND ATENDIME.CD_CID                      = CID.CD_CID(+)
  AND ATENDIME.CD_SERVICO                  = SERVICO.CD_SERVICO
  AND ATENDIME.CD_ATENDIMENTO              = AVISO_CIRURGIA.CD_ATENDIMENTO(+)
  AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA     = CIRURGIA_AVISO.CD_AVISO_CIRURGIA(+)
  AND CIRURGIA_AVISO.CD_CIRURGIA           = CIRURGIA.CD_CIRURGIA(+)
  AND PRESTADOR_AVISO.CD_AVISO_CIRURGIA(+) = AVISO_CIRURGIA.CD_AVISO_CIRURGIA
  AND ATENDIME.CD_ATENDIMENTO  = :CAMPO_PCD_ATENDIMENTO
  AND ROWNUM = 1
