--R_EVO_INFEC_UNID_INT
SELECT * FROM (
SELECT DS_UNID_INT ds_unid_int, CD_UNID_INT,
               competencia,
               To_Date(competencia,:CAMPO_FORMATO) competencia2,
               SUM(valor) total
 FROM dbamv.grafico_evo_infec
 GROUP BY DS_UNID_INT, CD_UNID_INT,
                    competencia
                    ,To_Date(competencia,:CAMPO_FORMATO)
ORDER BY To_Date(competencia,:CAMPO_FORMATO) , CD_UNID_INT
)
ORDER BY CD_UNID_INT,DS_UNID_INT,
COMPETENCIA2,COMPETENCIA
