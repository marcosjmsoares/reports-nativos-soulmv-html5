--R_CONFIRMA_INFEC_CIRUR
SELECT * FROM (
SELECT CD_PRESTADOR, PRESTADOR, DT_REALIZACAO, AVISO_CIRURGIA,
               ATENDIMENTO, PACIENTE, CIRURGIA, LIMPA, CONTAMINADA, INFECTADA, POT_CONTAMINADA 
FROM (
SELECT PRESTADOR.CD_PRESTADOR CD_PRESTADOR,
       PRESTADOR.NM_PRESTADOR PRESTADOR,
       AVISO_CIRURGIA.DT_REALIZACAO DT_REALIZACAO,
       AVISO_CIRURGIA.CD_AVISO_CIRURGIA AVISO_CIRURGIA,
       AVISO_CIRURGIA.CD_ATENDIMENTO ATENDIMENTO,
       PACIENTE.NM_PACIENTE PACIENTE,
       (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  CD_PACIENTE) NM_SOCIAL_PACIENTE,
       CIRURGIA.DS_CIRURGIA CIRURGIA,
       DECODE(CIRURGIA_AVISO.TP_NATUREZA,'L','X','') LIMPA,
       DECODE(CIRURGIA_AVISO.TP_NATUREZA,'C','X','') CONTAMINADA,
       DECODE(CIRURGIA_AVISO.TP_NATUREZA,'I','X','') INFECTADA,
       DECODE(CIRURGIA_AVISO.TP_NATUREZA,'P','X','') POT_CONTAMINADA
  FROM DBAMV.AVISO_CIRURGIA,
       DBAMV.PACIENTE,
       DBAMV.CIRURGIA,
       DBAMV.CEN_CIR,
       DBAMV.ATI_MED,
       DBAMV.EQUIPE_MEDICA,
       DBAMV.CIRURGIA_AVISO,
       DBAMV.PRESTADOR_AVISO,
       DBAMV.PRESTADOR
 WHERE AVISO_CIRURGIA.CD_AVISO_CIRURGIA  = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
   AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA  = PRESTADOR_AVISO.CD_AVISO_CIRURGIA
   AND AVISO_CIRURGIA.CD_PACIENTE        = PACIENTE.CD_PACIENTE
   AND AVISO_CIRURGIA.CD_CEN_CIR         = CEN_CIR.CD_CEN_CIR
   AND AVISO_CIRURGIA.CD_EQUIPE_MEDICA   = EQUIPE_MEDICA.CD_EQUIPE_MEDICA  (+)
   AND CIRURGIA_AVISO.CD_CIRURGIA_AVISO  = PRESTADOR_AVISO.CD_CIRURGIA_AVISO
   AND PRESTADOR_AVISO.CD_ATI_MED        = ATI_MED.CD_ATI_MED
   AND PRESTADOR_AVISO.CD_PRESTADOR      = PRESTADOR.CD_PRESTADOR
   AND CIRURGIA.CD_CIRURGIA              = CIRURGIA_AVISO.CD_CIRURGIA
   AND AVISO_CIRURGIA.TP_SITUACAO        = 'R'
   AND AVISO_CIRURGIA.DT_REALIZACAO BETWEEN :CAMPO_DT_INICIAL AND 
   AND ATI_MED.TP_FUNCAO = 'C'
   GROUP BY PRESTADOR.CD_PRESTADOR,
       PRESTADOR.NM_PRESTADOR,
       AVISO_CIRURGIA.DT_REALIZACAO,
       AVISO_CIRURGIA.CD_AVISO_CIRURGIA ,
       AVISO_CIRURGIA.CD_ATENDIMENTO ,
       PACIENTE.NM_PACIENTE ,
       PACIENTE.CD_PACIENTE ,
       PACIENTE.NM_SOCIAL_PACIENTE ,
       CIRURGIA.DS_CIRURGIA ,
       DECODE(CIRURGIA_AVISO.TP_NATUREZA,'L','X',''),
       DECODE(CIRURGIA_AVISO.TP_NATUREZA,'C','X',''),
       DECODE(CIRURGIA_AVISO.TP_NATUREZA,'I','X',''),
       DECODE(CIRURGIA_AVISO.TP_NATUREZA,'P','X','')
UNION
SELECT PRESTADOR.CD_PRESTADOR CD_PRESTADOR,
       PRESTADOR.NM_PRESTADOR PRESTADOR,
       AVISO_CIRURGIA.DT_REALIZACAO DT_REALIZACAO,
       AVISO_CIRURGIA.CD_AVISO_CIRURGIA AVISO_CIRURGIA,
       AVISO_CIRURGIA.CD_ATENDIMENTO ATENDIMENTO,
       PACIENTE.NM_PACIENTE PACIENTE,
       (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  CD_PACIENTE) NM_SOCIAL_PACIENTE,
       CIRURGIA.DS_CIRURGIA CIRURGIA,
       DECODE(CIRURGIA_AVISO.TP_NATUREZA,'L','X','') LIMPA,
       DECODE(CIRURGIA_AVISO.TP_NATUREZA,'C','X','') CONTAMINADA,
       DECODE(CIRURGIA_AVISO.TP_NATUREZA,'I','X','') INFECTADA,
       DECODE(CIRURGIA_AVISO.TP_NATUREZA,'P','X','') POT_CONTAMINADA
  FROM DBAMV.AVISO_CIRURGIA,
       DBAMV.PACIENTE,
       DBAMV.CIRURGIA,
       DBAMV.CEN_CIR,
       DBAMV.ATI_MED,
       DBAMV.CIRURGIA_AVISO,
       DBAMV.PRESTADOR_AVISO,
       DBAMV.PRESTADOR,
       DBAMV.EQUIPE_MEDICA
 WHERE AVISO_CIRURGIA.CD_AVISO_CIRURGIA  = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
   AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA  = PRESTADOR_AVISO.CD_AVISO_CIRURGIA
   AND AVISO_CIRURGIA.CD_PACIENTE        = PACIENTE.CD_PACIENTE
   AND AVISO_CIRURGIA.CD_CEN_CIR         = CEN_CIR.CD_CEN_CIR
   AND AVISO_CIRURGIA.CD_EQUIPE_MEDICA   = EQUIPE_MEDICA.CD_EQUIPE_MEDICA(+)
   AND CIRURGIA_AVISO.CD_CIRURGIA_AVISO  = PRESTADOR_AVISO.CD_CIRURGIA_AVISO
   AND PRESTADOR_AVISO.CD_ATI_MED        = ATI_MED.CD_ATI_MED
   AND PRESTADOR_AVISO.CD_PRESTADOR      = PRESTADOR.CD_PRESTADOR
   AND CIRURGIA.CD_CIRURGIA              =  CIRURGIA_AVISO.CD_CIRURGIA
   AND AVISO_CIRURGIA.TP_SITUACAO         = 'R'
   AND AVISO_CIRURGIA.DT_REALIZACAO BETWEEN :CAMPO_DT_INICIAL AND 
   AND ATI_MED.TP_FUNCAO = 'C'
   GROUP BY PRESTADOR.CD_PRESTADOR,
       PRESTADOR.NM_PRESTADOR,
       AVISO_CIRURGIA.DT_REALIZACAO,
       AVISO_CIRURGIA.CD_AVISO_CIRURGIA ,
       AVISO_CIRURGIA.CD_ATENDIMENTO ,
       PACIENTE.NM_PACIENTE ,
       PACIENTE.CD_PACIENTE ,
       PACIENTE.NM_SOCIAL_PACIENTE ,
       CIRURGIA.DS_CIRURGIA ,
       DECODE(CIRURGIA_AVISO.TP_NATUREZA,'L','X',''),
       DECODE(CIRURGIA_AVISO.TP_NATUREZA,'C','X',''),
       DECODE(CIRURGIA_AVISO.TP_NATUREZA,'I','X',''),
       DECODE(CIRURGIA_AVISO.TP_NATUREZA,'P','X','')
)
ORDER BY DT_REALIZACAO
)
ORDER BY CD_PRESTADOR,PRESTADOR,DT_REALIZACAO
