SELECT M.CD_ATENDIMENTO
      ,A.CD_PACIENTE
      ,P.NM_PACIENTE
      ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE PAC WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND P.CD_PACIENTE = PAC.CD_PACIENTE) NM_SOCIAL_PACIENTE
      ,PI.DS_PRO_INVASIVO
      ,Nvl(FA.DS_FOR_APL, :CAMPO_CF_NAO_INFORMADO) DS_FOR_APL
      ,Trunc(Trunc(M.DT_TERMINO) + (M.hr_termino - Trunc(M.hr_termino)) -  Trunc(M.DT_INICIO) + (M.hr_INICIO - Trunc(M.HR_INICIO))) INTERVALO
      ,M.CD_USUARIO_APLICACAO
      ,M.CD_USUARIO_RETIRADA
  FROM DBAMV.MOV_INV M
      ,DBAMV.ATENDIME A
      ,DBAMV.PACIENTE P
      ,DBAMV.PRO_INVASIVO PI
      ,DBAMV.ITPRE_MED IM
      ,DBAMV.FOR_APL FA
 WHERE M.CD_ATENDIMENTO = A.CD_ATENDIMENTO
   AND A.CD_PACIENTE = P.CD_PACIENTE
   AND M.CD_PRO_INVASIVO = PI.CD_PRO_INVASIVO
   AND M.CD_ITPRE_MED = IM.CD_ITPRE_MED (+)
   AND IM.CD_FOR_APL = FA.CD_FOR_APL (+)
ORDER BY M.DT_MOV_INV
