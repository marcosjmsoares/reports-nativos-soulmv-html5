--R_INFEC_UNID_INT
SELECT * FROM (
SELECT DECODE(reg_inf.tp_localizacao,'C',REG_INF.CD_CEN_CIR,'U',REG_INF.CD_UNID_INT,NULL) cd_unid_int,
      nvl(DECODE(reg_inf.tp_localizacao,'C',CEN_CIR.DS_CEN_CIR,'U',UNID_INT2.DS_UNID_INT,NULL),'<Não informado>') ds_unid_int,
               COUNT(*) Total
  FROM  dbamv.reg_inf,
              dbamv.atendime,
              dbamv.leito,
              dbamv.unid_int,
              dbamv.unid_int unid_int2,
              dbamv.CEN_CIR
WHERE reg_inf.cd_atendimento      = atendime.cd_atendimento
     and atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
     and  atendime.cd_leito                = leito.cd_leito
      and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
     and  leito.cd_unid_int                  = unid_int.cd_unid_int
     and reg_inf.cd_unid_int               = unid_int2.cd_unid_int (+)
     and reg_inf.cd_cen_cir                = cen_cir.cd_cen_cir (+)
     and reg_inf.tp_sit                          in ('C','T') 
     and trunc(reg_inf.dt_reg_inf)  between :CAMPO_DT_INICIAL GROUP BY
      DECODE(reg_inf.tp_localizacao,'C',REG_INF.CD_CEN_CIR,'U',REG_INF.CD_UNID_INT,NULL),
      DECODE(reg_inf.tp_localizacao,'C',CEN_CIR.DS_CEN_CIR,'U',UNID_INT2.DS_UNID_INT,NULL)
)
ORDER BY CD_UNID_INT,DS_UNID_INT
