--R_COMPARA_INFEC_CEN_CIR
SELECT TO_CHAR(ANO1) ANO, DECODE(MES1, 1,'Jan', 7,'Jul',
                                     2,'Fev', 8,'Ago',
                                     3,'Mar', 9,'Set',
                                     4,'Abr',10,'Out',
                                     5,'Mai',11,'Nov',
                                     6,'Jun',12,'Dez') MES,
       DECODE(SUM(QTD_CIR_REALIZADO),0,0,SUM(QTD_INFEC_CEN_CIR)/SUM(QTD_CIR_REALIZADO))*100 TAXA
FROM (
SELECT TO_NUMBER(TO_CHAR(R.DT_REG_INF,'YYYY')) ANO1,
       TO_NUMBER(TO_CHAR(R.DT_REG_INF,'MM'))   MES1,
       COUNT(DISTINCT R.CD_AVISO_CIRURGIA) QTD_INFEC_CEN_CIR,
       0 QTD_CIR_REALIZADO
FROM DBAMV.REG_INF R,
     DBAMV.AVISO_CIRURGIA A
WHERE R.CD_AVISO_CIRURGIA = A.CD_AVISO_CIRURGIA
  AND R.TP_SIT IN ('T','C')
  AND A.CD_AVISO_CIRURGIA IN ( SELECT DISTINCT CD_AVISO_CIRURGIA FROM DBAMV.CIRURGIA_AVISO
			                          WHERE TP_NATUREZA LIKE :CAMPO_TP_NATUREZA )
GROUP BY TO_NUMBER(TO_CHAR(R.DT_REG_INF,'YYYY')),
         TO_NUMBER(TO_CHAR(R.DT_REG_INF,'MM')),
         A.CD_CEN_CIR
UNION ALL
SELECT TO_NUMBER(TO_CHAR(A.DT_REALIZACAO,'YYYY')) ANO1,
       TO_NUMBER(TO_CHAR(A.DT_REALIZACAO,'MM'))   MES1,
       0 QTD_INFEC_CEN_CIR,
       COUNT(CD_AVISO_CIRURGIA) QTD_CIR_REALIZADO
FROM DBAMV.AVISO_CIRURGIA A
WHERE A.TP_SITUACAO = 'R'
  AND A.CD_AVISO_CIRURGIA IN ( SELECT DISTINCT CD_AVISO_CIRURGIA FROM DBAMV.CIRURGIA_AVISO
			                          WHERE TP_NATUREZA LIKE :CAMPO_TP_NATUREZA )
GROUP BY TO_NUMBER(TO_CHAR(A.DT_REALIZACAO,'YYYY')),
         TO_NUMBER(TO_CHAR(A.DT_REALIZACAO,'MM')),
         A.CD_CEN_CIR  )
WHERE ANO1 IS NOT NULL
  AND MES1 IS NOT NULL
  AND TO_NUMBER(ANO1||LPAD(MES1,2,0)) BETWEEN TO_NUMBER(TO_CHAR(:CAMPO_DT_INICIAL,'YYYYMM')) AND TO_NUMBER(TO_CHAR(:CAMPO_DT_FINAL,'YYYYMM'))
GROUP BY MES1, ANO1
