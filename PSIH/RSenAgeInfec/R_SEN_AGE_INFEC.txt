SELECT Cultura.CD_GERMES, Cultura.DS_GERME, count(CULTURA.CD_ITPED_LAB) TOTAL_GERME
 FROM dbamv.v_psih_cultura_cultura_manual Cultura
     ,dbamv.atendime
     ,dbamv.leito
where sensibilidade like 'Sens%vel'
  and Cultura.cd_atendimento = atendime.cd_atendimento
  and atendime.tp_atendimento = 'I'
  and leito.cd_leito = nvl(( SELECT cd_leito
                           FROM dbamv.mov_int
                          WHERE cd_mov_int =
                    	                     ( SELECT Max(cd_mov_int)
                      				 FROM dbamv.mov_int
                     				WHERE mov_int.cd_atendimento = atendime.cd_atendimento
                            			  AND tp_mov in ('I','O','F')
                                                  AND TRUNC(Cultura.dt_exame) between TRUNC(MOV_INT.DT_MOV_INT) AND TRUNC(NVL(MOV_INT.DT_LIB_MOV,SYSDATE))
                                              )
                        ),atendime.cd_atendimento)
GROUP BY Cultura.cd_germes,Cultura.DS_GERME
