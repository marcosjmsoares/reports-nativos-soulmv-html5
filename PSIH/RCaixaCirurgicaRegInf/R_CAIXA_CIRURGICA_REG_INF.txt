--R_CAIXA_CIRURGICA_REG_INF
SELECT DISTINCT REG_INF.CD_REG_INF,
       REG_INF.DT_REG_INF,
       AVISO_CIRURGIA.CD_AVISO_CIRURGIA,
       PACIENTE.CD_PACIENTE || ' - ' || PACIENTE.NM_PACIENTE DS_PACIENTE,
       (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(P.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE = P.CD_PACIENTE) NM_SOCIAL_PACIENTE,
       CIRURGIA.CD_CIRURGIA || ' - ' || CIRURGIA.DS_CIRURGIA DS_CIRURGIA,
       PRODUTO.CD_PRODUTO || ' - ' || PRODUTO.DS_PRODUTO DS_PRODUTO,
       ESTERILIZACAO.CD_CARGA_LOTE,
       Nvl(ESTERILIZACAO.DT_FIM_ESTERILIZA,ESTERILIZACAO.DT_INIC_ESTERILIZA) DT_FIM_ESTERILIZA
  FROM DBAMV.REG_INF,
       DBAMV.AVISO_CIRURGIA,
       DBAMV.CIRURGIA_AVISO,
       DBAMV.PACIENTE,
       DBAMV.CIRURGIA,
       DBAMV.IT_AV_CAIXA_CIRURGICA,
       DBAMV.PRODUTO,
       DBAMV.ESTERILIZACAO,
       DBAMV.CARGA_LOTE,
       (SELECT CD_ESTERILIZA, CD_KIT
          FROM DBAMV.IT_MVTO_CME
         GROUP BY CD_ESTERILIZA,CD_KIT) IT_MVTO_CME,
       DBAMV.CX_KIT,
       DBAMV.EMPRESA_PRODUTO
 WHERE REG_INF.CD_AVISO_CIRURGIA = AVISO_CIRURGIA.CD_AVISO_CIRURGIA
   AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
   AND AVISO_CIRURGIA.CD_PACIENTE = PACIENTE.CD_PACIENTE
   AND CIRURGIA_AVISO.CD_CIRURGIA = CIRURGIA.CD_CIRURGIA
   AND CIRURGIA_AVISO.SN_PRINCIPAL = 'S'
   AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = IT_AV_CAIXA_CIRURGICA.CD_AVISO_CIRURGIA
   AND CIRURGIA_AVISO.CD_CIRURGIA_AVISO = IT_AV_CAIXA_CIRURGICA.CD_CIRURGIA_AVISO
   AND CIRURGIA_AVISO.CD_CIRURGIA = IT_AV_CAIXA_CIRURGICA.CD_CIRURGIA
   AND IT_AV_CAIXA_CIRURGICA.CD_PRODUTO = PRODUTO.CD_PRODUTO
   AND IT_MVTO_CME.CD_ESTERILIZA = ESTERILIZACAO.CD_ESTERILIZA
   AND IT_MVTO_CME.CD_KIT = CX_KIT.CD_KIT
   AND ESTERILIZACAO.CD_CARGA_LOTE = CARGA_LOTE.CD_CARGA_LOTE
   AND PRODUTO.SN_KIT = 'E'
   AND EMPRESA_PRODUTO.CD_MULTI_EMPRESA = PKG_MV2000.LE_EMPRESA
   AND EMPRESA_PRODUTO.CD_PRODUTO = PRODUTO.CD_PRODUTO
ORDER BY 7
