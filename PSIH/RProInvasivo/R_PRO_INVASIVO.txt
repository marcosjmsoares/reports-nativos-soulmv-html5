--R_PRO_INVASIVO
Select Pro_Invasivo.Cd_Pro_Invasivo,
           PRO_INVASIVO.DS_PRO_INVASIVO,
           Tip_Presc.Ds_Tip_Presc,
           Cirurgia.Ds_Cirurgia
  From Dbamv.Pro_Invasivo,
          Dbamv.Pro_Inv_PSIH,
          Dbamv.Tip_Presc,
          Dbamv.Cirurgia
Where Pro_Invasivo.Cd_Pro_Invasivo = Pro_Inv_PSIH.Cd_Pro_Invasivo
    And Pro_Inv_PSIH.Cd_Tip_Presc    = Tip_Presc.Cd_Tip_Presc(+)
    And Pro_Inv_PSIH.Cd_Cirurgia     = Cirurgia.Cd_Cirurgia(+)
