--R_IND_CONT_AGEN_INFEC
SELECT sum(culturas) CULTURAS FROM (
SELECT Count(DISTINCT cd_itped_lab) CULTURAS
FROM(
Select DISTINCT  cd_itped_lab
 FROM dbamv.v_psih_cultura_cultura_manual Cultura
     ,dbamv.atendime
     ,dbamv.leito
where Cultura.cd_atendimento = atendime.cd_atendimento
  and atendime.tp_atendimento = 'I'
  AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
  AND cd_itped_lab <> 0
  AND  Trunc(CULTURA.DT_EXAME)  BETWEEN :CAMPO_DT_INICIAL AND 
  and leito.cd_leito =  nvl(( SELECT cd_leito
                           FROM dbamv.mov_int
                          WHERE cd_mov_int =
                    	                     ( SELECT Max(cd_mov_int)
                      				 FROM dbamv.mov_int
                     				WHERE mov_int.cd_atendimento = atendime.cd_atendimento
                            			  AND tp_mov in ('I','O','F')
                                                  AND TRUNC(Cultura.dt_exame) between TRUNC(MOV_INT.DT_MOV_INT) AND TRUNC(NVL(MOV_INT.DT_LIB_MOV,SYSDATE))
                                              )
                        ),atendime.cd_leito)
  :CAMPO_CF_ESPECIALIDADE
 )
      UNION ALL
SELECT Count(DISTINCT cd_cultura_manual)  CULTURAS
FROM(
Select DISTINCT  cd_cultura_manual
 FROM dbamv.v_psih_cultura_cultura_manual Cultura
     ,dbamv.atendime
     ,dbamv.leito
where Cultura.cd_atendimento = atendime.cd_atendimento
  and atendime.tp_atendimento = 'I'
  AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
  AND cd_cultura_manual <> 0
  AND  Trunc(CULTURA.DT_EXAME)  BETWEEN :CAMPO_DT_INICIAL AND 
  and leito.cd_leito =  nvl(( SELECT cd_leito
                           FROM dbamv.mov_int
                          WHERE cd_mov_int =
                    	                     ( SELECT Max(cd_mov_int)
                      				 FROM dbamv.mov_int
                     				WHERE mov_int.cd_atendimento = atendime.cd_atendimento
                            			  AND tp_mov in ('I','O','F')
                                                  AND TRUNC(Cultura.dt_exame) between TRUNC(MOV_INT.DT_MOV_INT) AND TRUNC(NVL(MOV_INT.DT_LIB_MOV,SYSDATE))
                                              )
                        ),atendime.cd_leito)
  :CAMPO_CF_ESPECIALIDADE
 )
 )

--R_IND_CONT_AGEN_INFEC_R_G_CD_GERMES1_Q_3
SELECT CD_GERMES, DS_GERME, Sum(QTD) QTD FROM (
SELECT CD_GERMES, DS_GERME, QTD FROM
(SELECT Cultura.CD_GERMES, Cultura.DS_GERME, count(DISTINCT CULTURA.CD_ITPED_LAB) QTD
 FROM dbamv.v_psih_cultura_cultura_manual Cultura
     ,dbamv.atendime
     ,dbamv.leito
where Cultura.cd_atendimento = atendime.cd_atendimento
 AND Trunc(Cultura.dt_exame) BETWEEN  :CAMPO_DT_INICIAL and 
  and atendime.tp_atendimento = 'I'
    AND CD_ITPED_LAB <> 0
  AND cd_germes IS NOT null
  and leito.cd_leito = nvl(( SELECT cd_leito
                           FROM dbamv.mov_int
                          WHERE cd_mov_int =
                    	                     ( SELECT Max(cd_mov_int)
                      				 FROM dbamv.mov_int
                     				WHERE mov_int.cd_atendimento = atendime.cd_atendimento
                            			  AND tp_mov in ('I','O','F')
                                                  AND TRUNC(Cultura.dt_exame) between TRUNC(MOV_INT.DT_MOV_INT) AND TRUNC(NVL(MOV_INT.DT_LIB_MOV,SYSDATE))
                                              )
                        ),atendime.cd_atendimento)
GROUP BY Cultura.cd_germes,Cultura.DS_GERME
ORDER BY QTD DESC
)
    UNION ALL
SELECT CD_GERMES, DS_GERME, QTD FROM
  (
SELECT Cultura.CD_GERMES, Cultura.DS_GERME, count(DISTINCT CULTURA.cd_cultura_manual) QTD
 FROM dbamv.v_psih_cultura_cultura_manual Cultura
     ,dbamv.atendime
     ,dbamv.leito
where Cultura.cd_atendimento = atendime.cd_atendimento
 AND Trunc(Cultura.dt_exame) BETWEEN  :CAMPO_DT_INICIAL and 
   AND cd_cultura_manual <> 0
  and atendime.tp_atendimento = 'I'
  AND cd_germes IS NOT null
  and leito.cd_leito = nvl(( SELECT cd_leito
                           FROM dbamv.mov_int
                          WHERE cd_mov_int =
                    	                     ( SELECT Max(cd_mov_int)
                      				 FROM dbamv.mov_int
                     				WHERE mov_int.cd_atendimento = atendime.cd_atendimento
                            			  AND tp_mov in ('I','O','F')
                                                  AND TRUNC(Cultura.dt_exame) between TRUNC(MOV_INT.DT_MOV_INT) AND TRUNC(NVL(MOV_INT.DT_LIB_MOV,SYSDATE))
                                              )
                        ),atendime.cd_atendimento)
GROUP BY Cultura.cd_germes,Cultura.DS_GERME
ORDER BY QTD DESC
  )
)
group by CD_GERMES, DS_GERME  ORDER BY ds_germe
