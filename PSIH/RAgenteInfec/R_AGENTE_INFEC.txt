--R_AGENTE_INFEC
select   cd_germes
           ,ds_germe
          ,count(cd_germes) qtd
From (
 select  distinct vgerme.cd_atendimento, vgerme.dt_exame, vgerme.cd_ped_lab, vgerme.cd_germes
        ,vgerme.ds_germe
  from dbamv.v_psih_cultura_germe vgerme
         ,dbamv.atendime
         ,dbamv.leito
         ,dbamv.unid_int
         ,DBAMV.REG_INF
 where vgerme.dt_exame BETWEEN :CAMPO_DT_INICIAL and 
     and atendime.tp_atendimento = 'I'
   and atendime.cd_atendimento = vgerme.cd_atendimento
   and atendime.cd_leito       = leito.cd_leito
   and leito.cd_unid_int       = unid_int.cd_unid_int
   and vgerme.cd_reg_inf    = reg_inf.cd_reg_inf(+)
:CAMPO_CF_TP_CULTURA
)
group by cd_germes, ds_germe
order by qtd desc
