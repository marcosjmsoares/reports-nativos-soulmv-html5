			,( Sum(nvl(PAC_INT_00H,0)) + Sum(nvl(ENT_INTERNADOS,0)) + Sum(nvl(ENT_TRANSF,0))) -
        ( Sum(nvl(SAI_ALTAS,0))   + Sum(nvl(SAI_TRANSFPARA,0)) +  Sum(nvl(SAI_OBITOS,0))) PAC_DIA

				,0                     SAI_OBITOS

      		 FROM DBAMV.CID
      		WHERE ( :CAMPO_DT_INICIAL -1 ) + ROWNUM <= :CAMPO_DT_FINAL ) CONTADOR
