SELECT
NM_SERIE NM_SERIE,
QT_INFEC QTD_INFEC,
QT_REALIZADO QTD_REALIZADO,
QT_TAXA TAXA,
Decode(NR_MES,1,'Jan',2,'Fev',3,'Mar',4,'Abr',5,'Mai',6,'Jun',7,'Jul',8,'Ago',9,'Set',10,'Out',11,'Nov',12,'Dez') || '/'|| NR_ANO ANO
,NR_ANO ANO1
FROM DBAMV.TMPMV_PSIH_INDICD_CMPR_ANUAL
ORDER BY To_Number(ANO1),NR_MES
