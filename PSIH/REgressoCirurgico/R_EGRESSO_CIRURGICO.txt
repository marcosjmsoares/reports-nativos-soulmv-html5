--R_EGRESSO_CIRURGICO
SELECT * FROM (
select   distinct
       antimicrob.cd_antimicrob,
       antimicrob.nm_antimicrob,
       avi_cir.cd_aviso_cirurgia              cd_aviso_cirurgia,
       avi_cir.cd_atendimento                cd_atendimento,
       avi_cir.cd_paciente                      cd_paciente,
       cir_avi.cd_cirurgia_aviso              cd_cirurgia_aviso,
       decode(cir_avi.tp_natureza,'L','Limpa','C','Contaminada','I','Infectada','P','Potencialmente Contaminada')tp_natureza,
       avi_cir.nm_paciente                      nm_paciente,
       to_char(trunc(avi_cir.dt_realizacao),'dd/mm/yyyy')           dt_realizacao,
       to_char(avi_cir.dt_realizacao, 'hh24') hr_inicio,
       dbamv.f_verif_prestador(avi_cir.cd_aviso_cirurgia, cirurgia.cd_cirurgia)  cd_cirurgiao,
       prestador.nm_prestador                 nm_prestador,
       convenio.nm_convenio                  nm_convenio,
       cirurgia.cd_cirurgia                         cd_cirurgia,
       cirurgia.ds_cirurgia                          ds_cirurgia,
       atendime.dt_alta                              dt_alta,
       atendime.dt_alta_medica                 dt_alta_medica
  from dbamv.pre_med,
       dbamv.itpre_med,
       dbamv.tip_presc,
       dbamv.solsai_pro,
       dbamv.itsolsai_pro,
       dbamv.ant_pro,
       dbamv.antimicrob,
       dbamv.mvto_estoque,
       dbamv.itmvto_estoque,
       dbamv.aviso_cirurgia        avi_cir,
       dbamv.cirurgia_aviso        cir_avi,
       dbamv.convenio,
       dbamv.prestador,
       dbamv.cirurgia,
       dbamv.atendime,
       dbamv.produto
 where pre_med.cd_pre_med                  = itpre_med.cd_pre_med
   and itpre_med.cd_tip_presc                 = tip_presc.cd_tip_presc
   and pre_med.cd_pre_med                   = solsai_pro.cd_pre_med
   and solsai_pro.cd_solsai_pro               = itsolsai_pro.cd_solsai_pro
   and tip_presc.cd_produto                    = itsolsai_pro.cd_produto
   and mvto_estoque.cd_mvto_estoque  = itmvto_estoque.cd_mvto_estoque
   and tip_presc.cd_produto                    = itmvto_estoque.cd_produto
   and tip_presc.cd_produto                    = produto.cd_produto
   and tip_presc.cd_produto                    = ant_pro.cd_produto
   and ant_pro.cd_antimicrob                  = antimicrob.cd_antimicrob
   and pre_med.cd_atendimento             = avi_cir.cd_atendimento
   and avi_cir.cd_aviso_cirurgia              = cir_avi.cd_aviso_cirurgia
   and avi_cir.cd_atendimento                = atendime.cd_atendimento
   and avi_cir.tp_situacao                       = 'R'
   and (atendime.dt_alta is not null or atendime.dt_alta_medica is not null)
   and cir_avi.cd_cirurgia                        = cirurgia.cd_cirurgia
   and dbamv.f_verif_prestador(avi_cir.cd_aviso_cirurgia, cirurgia.cd_cirurgia) = prestador.cd_prestador
   and cir_avi.cd_convenio                     = convenio.cd_convenio
   :CAMPO_CF_PERIODO
)
ORDER BY CD_CIRURGIAO,NM_PRESTADOR,
CD_PACIENTE,NM_PACIENTE,NM_CONVENIO,CD_ATENDIMENTO,NM_ANTIMICROB,CD_ANTIMICROB,CD_AVISO_CIRURGIA

--R_EGRESSO_CIRURGICO_R_ANTIMICROB_Q_ANTIMICROBIANO
select   distinct 
       antimicrob.cd_antimicrob,
       antimicrob.nm_antimicrob
  from dbamv.pre_med, 
       dbamv.itpre_med, 
       dbamv.tip_presc, 
       dbamv.solsai_pro, 
       dbamv.itsolsai_pro, 
       dbamv.ant_pro, 
       dbamv.antimicrob, 
       dbamv.mvto_estoque, 
       dbamv.itmvto_estoque,     
       dbamv.aviso_cirurgia        avi_cir, 
       dbamv.cirurgia_aviso        cir_avi, 
       dbamv.convenio, 
       dbamv.prestador, 
       dbamv.cirurgia, 
       dbamv.atendime,
       dbamv.produto
 where pre_med.cd_pre_med                  = itpre_med.cd_pre_med 
   and itpre_med.cd_tip_presc                 = tip_presc.cd_tip_presc 
   and pre_med.cd_pre_med                   = solsai_pro.cd_pre_med 
   and solsai_pro.cd_solsai_pro               = itsolsai_pro.cd_solsai_pro 
   and tip_presc.cd_produto                    = itsolsai_pro.cd_produto   
   and mvto_estoque.cd_mvto_estoque  = itmvto_estoque.cd_mvto_estoque 
   and tip_presc.cd_produto                    = itmvto_estoque.cd_produto 
   and tip_presc.cd_produto                    = produto.cd_produto 
   and tip_presc.cd_produto                    = ant_pro.cd_produto 
   and ant_pro.cd_antimicrob                  = antimicrob.cd_antimicrob   
   and pre_med.cd_atendimento             = avi_cir.cd_atendimento   
   and avi_cir.cd_aviso_cirurgia              = cir_avi.cd_aviso_cirurgia 
   and avi_cir.cd_atendimento                = atendime.cd_atendimento 
   and avi_cir.tp_situacao                       = 'R' 
   and (atendime.dt_alta is not null or atendime.dt_alta_medica is not null) 
   and cir_avi.cd_cirurgia                        = cirurgia.cd_cirurgia   
   and dbamv.f_verif_prestador(avi_cir.cd_aviso_cirurgia, cirurgia.cd_cirurgia) = prestador.cd_prestador       
   and cir_avi.cd_convenio                     = convenio.cd_convenio
