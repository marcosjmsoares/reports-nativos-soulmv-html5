--R_EST_POS_CULTURA_MATERIAL
SELECT * FROM (
	select distinct	unid_int.cd_unid_int
		  ,unid_int.ds_unid_int
		  ,ori_ate.cd_ori_ate
		  ,ori_ate.ds_ori_ate
		  ,itcultura.cd_exa_lab
		  ,itcultura.cd_material
		  ,itcultura.nm_exa_lab
		  ,itcultura.ds_material
	  from dbamv.v_psih_cultura_cultura_manual itcultura
		  ,dbamv.reg_inf
		  ,dbamv.atendime
		  ,dbamv.leito
		  ,dbamv.ori_ate
		  ,dbamv.unid_int
	 where reg_inf.cd_reg_inf(+) = itcultura.cd_reg_inf
	   and itcultura.cd_atendimento = atendime.cd_atendimento
	   and atendime.cd_ori_ate = ori_ate.cd_ori_ate
	   and atendime.cd_leito = leito.cd_leito(+)
	   and leito.cd_unid_int = unid_int.cd_unid_int(+)
	   :CAMPO_CF_TP_ATENDIMENTO
)
order by ds_unid_int,ds_ori_ate,ds_material,nm_exa_lab
