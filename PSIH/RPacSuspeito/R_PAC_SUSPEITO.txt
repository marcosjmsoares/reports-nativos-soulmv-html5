--R_PAC_SUSPEITO
SELECT * FROM (
SELECT NM_PACIENTE,
        (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND V_PAC_SUSPEITOS_INFEC.CD_PACIENTE =  CD_PACIENTE) NM_SOCIAL_PACIENTE,
        CD_ATENDIMENTO,
        NM_PRESTADOR,
        DS_LEITO,
        DT_PREVISTA_ALTA,
        DT_ATENDIMENTO,
        CD_PACIENTE,
        CD_UNID_INT,
        NM_SETOR||' / '||DS_UNID_INT DS_UNID_INT,
        CD_SETOR,
        NM_SETOR,
        CD_CID,
        DS_CID
FROM DBAMV.V_PAC_SUSPEITOS_INFEC
WHERE Dt_Alta Is Null
  AND IDENT = 'I'
  :CAMPO_CF_WHERE
)
ORDER BY NM_SETOR,DS_UNID_INT,CD_SETOR,CD_UNID_INT

		                     , PRODUTO PROD
		                 WHERE ANT_PRO.CD_ANTIMICROB = ANTIMICROB.CD_ANTIMICROB
		                   AND ANT_PRO.CD_PRODUTO  = PROD.CD_PRODUTO
		   AND ANTIMICROB.SN_REL_SUSP_INFEC = 'S'
		   AND PRODUTO.CD_PRODUTO  = PROD.CD_PRODUTO)) OR
		   EXISTS ( SELECT 'X'    
		        FROM ANT_PRO
		          , ANTIMICROB
		          , PRODUTO PROD
		        WHERE ANT_PRO.CD_ANTIMICROB = ANTIMICROB.CD_ANTIMICROB
		        AND ANT_PRO.CD_PRODUTO = PROD.CD_PRODUTO
		        AND ANTIMICROB.SN_REL_SUSP_INFEC = 'S'
		        AND PRODUTO.CD_PRODUTO = PROD.CD_PRODUTO_TEM))
		  AND NOT EXISTS (SELECT 'X'
		              FROM SOLSAI_PRO
		              WHERE CD_PRE_MED =  PRE_MED.CD_PRE_MED)
                  AND atendime.cd_atendimento = :CAMPO_CD_ATENDIMENTO)
ORDER BY  DT_MVTO_ESTOQUE DESC
