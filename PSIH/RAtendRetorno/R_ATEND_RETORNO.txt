--R_ATEND_RETORNO
SELECT PRINCIPAL.CD_ATENDIMENTO CD_ATENDIMENTO_PRINCIPAL,
       PRINCIPAL.DT_ATENDIMENTO + (PRINCIPAL.HR_ATENDIMENTO - Trunc(PRINCIPAL.HR_ATENDIMENTO)) DT_ATENDIMENTO_PRINCIPAL,
       PRINCIPAL.CD_CID CD_CID_PRINCIPAL,
       PRINCIPAL.DT_ALTA + (PRINCIPAL.HR_ALTA - Trunc(PRINCIPAL.HR_ALTA)) DT_ALTA_PRINCIPAL,
       PRINCIPAL.CD_PACIENTE || ' - ' || PACIENTE.NM_PACIENTE DS_PACIENTE,
       (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  CD_PACIENTE) NM_SOCIAL_PACIENTE,
       FILHOS.CD_ATENDIMENTO CD_ATENDIMENTO_FILHO,
       FILHOS.DT_ATENDIMENTO DT_ATENDIMENTO_FILHO,
       FILHOS.CD_CID CD_CID_FILHO,
       FILHOS.DT_ALTA DT_ALTA_FILHO
  FROM (SELECT CD_ATENDIMENTO,
               CD_PACIENTE,
               DT_ATENDIMENTO + (HR_ATENDIMENTO - Trunc(HR_ATENDIMENTO)) DT_ATENDIMENTO,
               CD_CID,
               DT_ALTA + (HR_ALTA - Trunc(HR_ALTA)) DT_ALTA
          FROM DBAMV.ATENDIME
         WHERE DT_ATENDIMENTO = :CAMPO_CF_DT_REFERENCIA) FILHOS,
       DBAMV.ATENDIME PRINCIPAL,
       DBAMV.PACIENTE PACIENTE
  WHERE FILHOS.CD_ATENDIMENTO    <> PRINCIPAL.CD_ATENDIMENTO
    AND FILHOS.CD_PACIENTE       =  PRINCIPAL.CD_PACIENTE
    AND PRINCIPAL.CD_PACIENTE    = PACIENTE.CD_PACIENTE
 ORDER BY PRINCIPAL.DT_ALTA DESC, PACIENTE.NM_PACIENTE
