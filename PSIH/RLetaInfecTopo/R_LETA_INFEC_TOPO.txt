--R_LETA_INFEC_TOPO
Select     count(*)  total,
           topografia.cd_topografia,
           topografia.ds_topografia,
           CLASSIFICACAO_IRAS.cd_sigla_iras_apres || ' - ' || CLASSIFICACAO_IRAS.ds_sitio_principal CATEGORIA,
           Sum(Decode(ATENDIME.sn_obito_infec,'S',1,0)) qt_obito
 from dbamv.reg_inf
     ,dbamv.atendime
     ,dbamv.paciente
     ,dbamv.leito
     ,dbamv.unid_int
     ,dbamv.topografia
     ,DBAMV.Classificacao_Iras
where reg_inf.cd_atendimento                                         = atendime.cd_atendimento
  and atendime.cd_multi_empresa                                      = 1
  and atendime.cd_paciente                                           = paciente.cd_paciente(+)
  and atendime.cd_leito                                              = leito.cd_leito
  and leito.cd_unid_int                                              = unid_int.cd_unid_int(+)
  and topografia.cd_topografia                                       = reg_inf.cd_topografia (+)
  AND atendime.dt_alta is not null
  AND ATENDIME.TP_ATENDIMENTO IN  ('I', 'A', 'E', 'U')
  AND TOPOGRAFIA.TP_CATEGORIA                                        = Classificacao_Iras.Cd_Sigla_Iras(+)
group by  topografia.cd_topografia,
                topografia.ds_topografia,
                CLASSIFICACAO_IRAS.cd_sigla_iras_apres || ' - ' || CLASSIFICACAO_IRAS.ds_sitio_principal
