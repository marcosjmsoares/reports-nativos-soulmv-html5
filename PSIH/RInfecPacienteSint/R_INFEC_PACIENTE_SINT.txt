--R_INFEC_PACIENTE_SINT
SELECT * FROM (
select
       DECODE(reg_inf.tp_localizacao,'C',REG_INF.CD_CEN_CIR,'U',REG_INF.CD_UNID_INT,NULL) cd_unid_int
      ,DECODE(reg_inf.tp_localizacao,'C',CEN_CIR.DS_CEN_CIR,'U',UNID_INT.DS_UNID_INT,NULL) ds_unid_int
      ,topografia.cd_topografia
      ,topografia.ds_topografia
      ,count(*)   qtd_atend
 from dbamv.reg_inf
     ,dbamv.atendime
     ,dbamv.paciente
     ,dbamv.leito
     ,dbamv.unid_int
     ,dbamv.CEN_CIR
     ,dbamv.topografia
where reg_inf.cd_atendimento       = atendime.cd_atendimento
  and atendime.cd_paciente          = paciente.cd_paciente
  and reg_inf.cd_leito                     = leito.cd_leito(+)
  and leito.cd_unid_int                   = unid_int.cd_unid_int(+)
  and reg_inf.CD_CEN_CIR           = CEN_CIR.CD_CEN_CIR (+)
  and topografia.cd_topografia(+)       = reg_inf.cd_topografia 
  and ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
group by
         DECODE(reg_inf.tp_localizacao,'C',REG_INF.CD_CEN_CIR,'U',REG_INF.CD_UNID_INT,NULL),
         DECODE(reg_inf.tp_localizacao,'C',CEN_CIR.DS_CEN_CIR,'U',UNID_INT.DS_UNID_INT,NULL),
         topografia.cd_topografia   ,topografia.ds_topografia
order by cd_unid_int, cd_topografia
)
ORDER BY CD_UNID_INT,DS_UNID_INT, cd_topografia, ds_topografia
