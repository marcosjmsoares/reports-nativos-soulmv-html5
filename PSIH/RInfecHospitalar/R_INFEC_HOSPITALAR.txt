SELECT cen_cir.CD_CEN_CIR,
       cen_cir.ds_cen_cir,
       sal_cir.cd_sal_Cir,
       sal_cir.ds_sal_Cir,
       aviso_cirurgia.cd_aviso_cirurgia,
	   atendime.cd_paciente,
       paciente.cd_identificador_pessoa,
       aviso_cirurgia.nm_paciente,
       (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  CD_PACIENTE) NM_SOCIAL_PACIENTE,
	   (SELECT ds_cirurgia FROM dbamv.cirurgia_aviso, dbamv.cirurgia WHERE cirurgia.cd_Cirurgia = cirurgia_Aviso.cd_cirurgia AND sn_principal = 'S' AND cd_aviso_cirurgia = aviso_cirurgia.CD_AVISO_CIRURGIA) cir_principal,
	   prestador.nm_prestador,
	   reg_inf.cd_cid,
       v.ds_germe
	FROM dbamv.aviso_cirurgia
			,dbamv.atendime
			,dbamv.cirurgia_aviso
			,dbamv.cirurgia
			,dbamv.cen_cir
			,dbamv.sal_cir
			,dbamv.reg_inf
      ,dbamv.v_psih_cultura_germe v
      ,dbamv.prestador
      ,dbamv.paciente
	WHERE AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
		AND aviso_cirurgia.cd_atendimento = atendime.cd_atendimento
		AND AVISO_CIRURGIA.CD_CEN_CIR = CEN_CIR.CD_CEN_CIR
		AND CIRURGIA_AVISO.CD_CIRURGIA = CIRURGIA.CD_CIRURGIA
		AND AVISO_CIRURGIA.CD_SAL_CIR = SAL_CIR.CD_SAL_CIR
		AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = REG_INF.CD_AVISO_CIRURGIA(+)
    AND reg_inf.cd_reg_inf = v.cd_reg_inf
    AND v.tp_resultado = 'P'
    AND prestador.cd_prestador = F_VERIF_PRESTADOR(AVISO_CIRURGIA.CD_AVISO_CIRURGIA, CIRURGIA_AVISO.CD_CIRURGIA)
    AND paciente.cd_paciente = atendime.cd_paciente
    AND TRUNC(reg_inf.dt_reg_inf) BETWEEN :CAMPO_DT_INICIAL GROUP BY cen_Cir.cd_cen_Cir, cen_cir.ds_cen_cir, dbamv.sal_cir.cd_sal_cir, sal_cir.ds_sal_Cir, aviso_cirurgia.cd_aviso_cirurgia, atendime.cd_paciente, paciente.cd_identificador_pessoa, aviso_cirurgia.nm_paciente, prestador.nm_prestador, reg_inf.cd_cid, v.cd_germes, v.ds_germe
