--R_CULTURA_POS
SELECT * FROM (
select distinct v.cd_exa_lab
	     ,v.nm_exa_lab
	     ,trunc(amostra.dt_coleta) dt_coleta
	     ,ate.cd_paciente
	     ,pac.nm_paciente
	     ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ate.CD_PACIENTE =  CD_PACIENTE) NM_SOCIAL_PACIENTE
	     ,amostra.cd_amostra
	     ,v.cd_ped_lab
	     ,v.cd_material
	     ,mat.ds_material
	     ,lei.cd_unid_int
	     ,unid.ds_unid_int
	     ,ate.cd_leito
	     ,lei.ds_leito
  from dbamv.v_psih_cultura_cultura_manual v
         ,dbamv.germes g
         ,dbamv.antimicrob am
         ,dbamv.itped_lab it
         ,dbamv.atendime ate
         ,dbamv.paciente pac
         ,dbamv.leito    lei
         ,dbamv.unid_int unid
         ,dbamv.amostra
         ,dbamv.material mat
         ,dbamv.exa_lab
         ,dbamv.amostra_exa_lab
 where v.cd_germes         = g.cd_germes
    and v.cd_itped_lab      = it.cd_itped_lab(+)
    and v.cd_atendimento = ate.cd_atendimento
    and ate.cd_paciente    = pac.cd_paciente
    and v.cd_antimicrob    = am.cd_antimicrob
    and v.cd_material        = mat.cd_material
    and ate.cd_leito           = lei.cd_leito (+)
    and lei.cd_unid_int      = unid.cd_unid_int (+)
    and it.cd_itped_lab         =  amostra_exa_lab.cd_itped_lab(+)
    and amostra.cd_amostra(+) =  amostra_exa_lab.cd_amostra
    and v.tp_resultado  in ('P','A','F') 
    and exa_lab.cd_exa_lab = v.cd_exa_lab
    and exa_lab.tp_resultado = 'C'   
    :CAMPO_CF_PERIODO
)
ORDER BY CD_UNID_INT,DS_UNID_INT,DT_COLETA,CD_EXA_LAB,NM_EXA_LAB,CD_PACIENTE,NM_PACIENTE,CD_AMOSTRA,CD_PED_LAB,CD_MATERIAL,DS_MATERIAL,CD_LEITO,DS_LEITO
