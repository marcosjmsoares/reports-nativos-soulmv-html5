--R_REV_REPASSE
SELECT *
   			FROM
  	  (SELECT prestador.cd_remessa, prestador.dt_lancamento, prestador.cd_reg_fat, prestador.cd_reg_amb, prestador.cd_lanc_fat, prestador.cd_lanc_amb,
	          prestador.cd_eventos, prestador.cd_aih, prestador.cd_glosas, prestador.conta,prestador.qt_lancamento,prestador.cd_especialid,
	          DECODE (:CAMPO_SN_GRUPO ,'S', DECODE (prestador.cd_prestador_repasse, NULL, prestador.cd_prestador, prestador.cd_prestador_repasse) ,prestador.cd_prestador ) cd_prestador ,
	          prestador.cd_ati_med, prestador.vl_total_conta, prestador.vl_sh, prestador.vl_sp,
	          prestador.vl_sd, prestador.vl_ato, prestador.vl_pontos, prestador.qt_pontos, prestador.cd_sus, prestador.vl_base_repassado, prestador.cd_convenio,
	          DECODE(prestador.nm_convenio, NULL, '- Manual',prestador.nm_convenio)   nm_convenio,
	          prestador.cd_pro_fat, prestador.ds_pro_fat, prestador.cd_gru_pro, prestador.cd_ori_ate, prestador.cd_paciente, prestador.nm_paciente,
	          DECODE (:CAMPO_SN_GRUPO, 'S', NULL, prestador.nm_prestador) nm_prestador, prestador.cd_reg_repasse, prestador.cd_gru_fat, prestador.tp_remessa,
	          prestador.dt_entrega_da_fatura  , prestador.dt_competencia, prestador.sn_pertence_pacote, prestador.cd_tip_acom, prestador.filme, prestador.cd_setor,
	          prestador.vl_outros, prestador.vl_auxiliar, prestador.tp_pagamento, prestador.tipo, DECODE (:CAMPO_SN_GRUPO, 'S', prestador.cd_prestador, TO_NUMBER (NULL) ) cd_prestador_individ      , 
	          DECODE (:CAMPO_SN_GRUPO, 'S', prestador.nm_prestador, NULL) nm_prestador_individ, prestador.tp_atendimento, prestador.qt_ch_unitario, prestador.cd_espec_sus,
	          prestador.cd_repasse, prestador.hr_lancamento hr_lancamento, prestador.cd_atendimento
     	FROM dbamv.v_repasse_prestador2_321 prestador , dbamv.prestador gru_rep
    	WHERE  prestador.cd_prestador_repasse = gru_rep.cd_prestador(+)
 		ORDER BY prestador.cd_remessa)
	ORDER BY CD_PRESTADOR, CD_PRESTADOR_INDIVID, NM_PRESTADOR_INDIVID, NM_CONVENIO, CD_CONVENIO
