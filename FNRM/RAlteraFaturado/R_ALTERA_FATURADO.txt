SELECT Log.cd_prestador_anterior cd_prestador,
       v.cd_reg_amb cd_reg,
       dbamv.pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(pac.cd_paciente, 1) nm_paciente,
       v.cd_pro_fat cd_pro,
       prc.ds_pro_fat ds_pro,
       Log.dt_alterado,
       Log.nm_usuario_alterou,
       Decode(Log.tp_alteracao,'E', 'Excluído', 'C', 'Cancelado', 'P', 'Prestador', 'V', 'Valor') tp_alteracao,
       Log.tp_alteracao tp_alteracao_aux,
       Log.vl_repasse_anterior,
       p1.nm_prestador nm_prestador,
       v.cd_prestador cd_prestador_novo,
       p2.nm_prestador nm_novo_prestador
  FROM dbamv.log_altera_repasse log, dbamv.v_fnrm_repasse_gerado v, dbamv.prestador p1, dbamv.prestador p2, dbamv.atendime ate,
       dbamv.paciente pac, dbamv.pro_fat prc
 WHERE Log.cd_it_repasse = v.cd_it_repasse
   AND Log.cd_prestador_anterior = p1.cd_prestador
   AND v.cd_prestador = p2.cd_prestador
   AND v.cd_atendimento = ate.cd_atendimento
   AND pac.cd_paciente = ate.cd_paciente
   AND v.cd_pro_fat = prc.cd_pro_fat
   AND v.cd_reg_fat IS NULL
   AND log.dt_alterado BETWEEN TO_DATE(:CAMPO_CF_DT1,'DD/MM/YYYY HH24') AND TO_DATE(:CAMPO_CF_DT2,'DD/MM/YYYY HH24')
UNION all
