--R_REPASSE_CREDENCIADO
SELECT * FROM (
SELECT reg_amb.cd_remessa,atendime.dt_atendimento dt_lancamento
                         ,itreg_amb.cd_reg_amb cd_conta
                         ,itreg_amb.cd_lancamento cd_lancamento
                         ,itreg_amb.cd_pro_fat cd_pro_fat
                         ,itreg_amb.cd_convenio cd_convenio
                         ,itreg_amb.qt_lancamento qt_lancamento
                         ,prestador.cd_prestador cd_prestador
                         ,itreg_amb.cd_ati_med cd_ati_med
                         ,NVL (itreg_amb.vl_total_conta, 0) vl_total
                         ,convenio.nm_convenio nm_convenio
                         ,pro_fat.ds_pro_fat ds_pro_fat
                         ,pro_fat.cd_gru_pro cd_gru_pro
                         ,atendime.cd_paciente cd_paciente
                         ,atendime.cd_ori_ate cd_ori_ate
                         ,dbamv.pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(paciente.cd_paciente, 1) nm_paciente
                         ,prestador.nm_prestador nm_prestador
                         ,empresa_prestador.cd_reg_repasse cd_reg_repasse
                         ,itreg_amb.cd_gru_fat cd_gru_fat
                         ,remessa_fatura.sn_paga tp_remessa
                         ,remessa_fatura.dt_entrega_da_fatura
                         ,fatura.dt_competencia
                         ,itreg_amb.sn_pertence_pacote sn_pertence_pacote
                         ,0 cd_tip_acom
                         ,NVL (itreg_amb.vl_filme_unitario * qt_lancamento, 0) filme
                         ,itreg_amb.cd_setor cd_setor
                         ,atendime.tp_atendimento
                         ,reg_amb.cd_multi_empresa
                         ,itreg_amb.qt_ch_unitario qt_ch_unitario
                         ,atendime.cd_atendimento cd_atendimento
                         ,itreg_amb.hr_lancamento hr_lancamento
                         ,ati_med.ds_ati_med
                     FROM dbamv.reg_amb reg_amb
                         ,dbamv.itreg_amb itreg_amb
                         ,dbamv.remessa_fatura remessa_fatura
                         ,dbamv.pro_fat pro_fat
                         ,dbamv.atendime atendime
                         ,dbamv.paciente paciente
                         ,dbamv.prestador prestador
                         ,dbamv.convenio convenio
                         ,dbamv.fatura fatura
                         ,dbamv.ati_med
                         ,dbamv.empresa_prestador
                    WHERE itreg_amb.cd_prestador IS NOT NULL
                      AND  ATENDIME.CD_MULTI_EMPRESA = dbamv.Pkg_Mv2000.Le_Empresa
                      AND NVL (itreg_amb.tp_pagamento, 'P') = 'C'
                      AND itreg_amb.vl_base_repassado IS NULL
                      AND itreg_amb.sn_repassado = 'N'
                      AND NVL (itreg_amb.vl_total_conta, 0) > 0
                      AND reg_amb.cd_remessa = remessa_fatura.cd_remessa
                      AND remessa_fatura.sn_fechada = 'S'
                      AND itreg_amb.sn_fechada = 'S'
                      AND remessa_fatura.cd_fatura = fatura.cd_fatura
                      AND reg_amb.cd_reg_amb = itreg_amb.cd_reg_amb
                      AND itreg_amb.cd_reg_amb = reg_amb.cd_reg_amb
                      AND itreg_amb.cd_convenio = convenio.cd_convenio
                      AND itreg_amb.cd_pro_fat = pro_fat.cd_pro_fat
                      AND itreg_amb.cd_atendimento = atendime.cd_atendimento
                      AND atendime.cd_paciente = paciente.cd_paciente
                      AND itreg_amb.cd_prestador = prestador.cd_prestador
                      AND ITREG_AMB.CD_ATI_MED = ATI_MED.CD_ATI_MED(+)
                      AND prestador.cd_prestador = empresa_prestador.cd_prestador
                      AND empresa_prestador.cd_multi_empresa = reg_amb.cd_multi_empresa
union all
SELECT reg_fat.cd_remessa,itreg_fat.dt_lancamento dt_lancamento
                         ,itreg_fat.cd_reg_fat cd_conta
                         ,itreg_fat.cd_lancamento cd_lancamento
                         ,itreg_fat.cd_pro_fat cd_pro_fat
                         ,reg_fat.cd_convenio cd_convenio
                         ,itreg_fat.qt_lancamento qt_lancamento
                         ,prestador.cd_prestador cd_prestador
                         ,NULL cd_ati_med
                         ,NVL (itreg_fat.vl_total_conta, 0) vl_total
                         ,convenio.nm_convenio nm_convenio
                         ,pro_fat.ds_pro_fat ds_pro_fat
                         ,pro_fat.cd_gru_pro cd_gru_pro
                         ,atendime.cd_paciente cd_paciente
                         ,atendime.cd_ori_ate cd_ori_ate
                         ,dbamv.pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(paciente.cd_paciente, 1) nm_paciente
                         ,prestador.nm_prestador nm_prestador
                         ,empresa_prestador.cd_reg_repasse cd_reg_repasse
                         ,itreg_fat.cd_gru_fat cd_gru_fat
                         ,remessa_fatura.sn_paga tp_remessa
                         ,remessa_fatura.dt_entrega_da_fatura
                         ,fatura.dt_competencia
                         ,itreg_fat.sn_pertence_pacote sn_pertence_pacote
                         ,reg_fat.cd_tip_acom cd_tip_acom
                         ,NVL (itreg_fat.vl_filme_unitario * qt_lancamento, 0) filme
                         ,itreg_fat.cd_setor cd_setor
                         ,atendime.tp_atendimento
                         ,reg_fat.cd_multi_empresa
                         ,itreg_fat.qt_ch_unitario qt_ch_unitario
                         ,atendime.cd_atendimento cd_atendimento
                         ,itreg_fat.hr_lancamento hr_lancamento
                         ,ati_med.ds_ati_med
                     FROM dbamv.reg_fat reg_fat
                         ,dbamv.itreg_fat itreg_fat
                         ,dbamv.remessa_fatura remessa_fatura
                         ,dbamv.pro_fat pro_fat
                         ,dbamv.atendime atendime
                         ,dbamv.paciente paciente
                         ,dbamv.prestador prestador
                         ,dbamv.convenio convenio
                         ,dbamv.ati_med
                         ,dbamv.fatura fatura
                         ,dbamv.empresa_prestador empresa_prestador
                    WHERE itreg_fat.cd_prestador IS NOT NULL
                      AND NVL (itreg_fat.tp_pagamento, 'P') = 'C'
                      AND itreg_fat.vl_base_repassado IS NULL
                      AND itreg_fat.sn_repassado = 'N'
                      AND NVL (itreg_fat.vl_total_conta, 0) > 0
                      AND reg_fat.cd_convenio = convenio.cd_convenio
                      AND reg_fat.cd_remessa = remessa_fatura.cd_remessa
                      AND remessa_fatura.sn_fechada = 'S'
                      AND reg_fat.sn_fechada = 'S'
                      AND remessa_fatura.cd_fatura = fatura.cd_fatura
                      AND reg_fat.cd_reg_fat = itreg_fat.cd_reg_fat
                      AND itreg_fat.cd_reg_fat = reg_fat.cd_reg_fat
                      AND itreg_fat.cd_pro_fat = pro_fat.cd_pro_fat
                      AND  ATENDIME.CD_MULTI_EMPRESA = dbamv.Pkg_Mv2000.Le_Empresa
                      AND reg_fat.cd_atendimento = atendime.cd_atendimento
                      AND atendime.cd_paciente = paciente.cd_paciente
                      AND itreg_fat.cd_prestador = prestador.cd_prestador
                      AND prestador.cd_prestador = empresa_prestador.cd_prestador
                      AND ITREG_FAT.CD_ATI_MED = ATI_MED.CD_ATI_MED(+)
                      AND empresa_prestador.cd_multi_empresa = reg_fat.cd_multi_empresa
union all
SELECT reg_fat.cd_remessa,itreg_fat.dt_lancamento dt_lancamento
                         ,itreg_fat.cd_reg_fat cd_conta
                         ,itreg_fat.cd_lancamento cd_lancamento
                         ,itreg_fat.cd_pro_fat cd_pro_fat
                         ,reg_fat.cd_convenio cd_convenio
                         ,itreg_fat.qt_lancamento qt_lancamento
                         ,prestador.cd_prestador cd_prestador
                         ,itlan_med.cd_ati_med cd_ati_med
                         ,(NVL (itlan_med.vl_liquido, (itlan_med.vl_ato * itreg_fat.qt_lancamento))) vl_total
                         ,convenio.nm_convenio nm_convenio
                         ,pro_fat.ds_pro_fat ds_pro_fat
                         ,pro_fat.cd_gru_pro cd_gru_pro
                         ,atendime.cd_paciente cd_paciente
                         ,atendime.cd_ori_ate cd_ori_ate
                         ,dbamv.pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(paciente.cd_paciente, 1) nm_paciente
                         ,prestador.nm_prestador nm_prestador
                         ,nvl( dbamv.pkg_repasse.fnc_fnrm_reg_rep_auxiliar( itlan_med.cd_prestador ),
						       empresa_prestador.cd_reg_repasse ) cd_reg_repasse
                         ,itreg_fat.cd_gru_fat cd_gru_fat
                         ,remessa_fatura.sn_paga tp_remessa
                         ,remessa_fatura.dt_entrega_da_fatura
                         ,fatura.dt_competencia
                         ,itreg_fat.sn_pertence_pacote sn_pertence_pacote
                         ,reg_fat.cd_tip_acom cd_tip_acom
                         ,0 filme
                         ,itreg_fat.cd_setor cd_setor
                         ,atendime.tp_atendimento
                         ,reg_fat.cd_multi_empresa
                         ,itreg_fat.qt_ch_unitario qt_ch_unitario
                         ,atendime.cd_atendimento cd_atendimento
                         ,itreg_fat.hr_lancamento hr_lancamento
                         ,ati_med.ds_ati_med
                     FROM dbamv.reg_fat reg_fat
                         ,dbamv.itreg_fat itreg_fat
                         ,dbamv.itlan_med itlan_med
                         ,dbamv.remessa_fatura remessa_fatura
                         ,dbamv.pro_fat pro_fat
                         ,dbamv.atendime atendime
                         ,dbamv.paciente paciente
                         ,dbamv.prestador prestador
                         ,dbamv.convenio convenio
                         ,dbamv.ati_med
                         ,dbamv.fatura fatura
                         ,dbamv.empresa_prestador empresa_prestador
                    WHERE itreg_fat.cd_prestador IS NULL
                      AND  ATENDIME.CD_MULTI_EMPRESA = dbamv.Pkg_Mv2000.Le_Empresa
                      AND NVL (itlan_med.tp_pagamento, 'P') = 'C'
                      AND itlan_med.vl_base_repassado IS NULL
                      AND itlan_med.sn_repassado = 'N'
                      AND (NVL (itlan_med.vl_liquido, (itlan_med.vl_ato * itreg_fat.qt_lancamento))) > 0
                      AND reg_fat.cd_remessa = remessa_fatura.cd_remessa
                      AND remessa_fatura.sn_fechada = 'S'
                      AND reg_fat.sn_fechada = 'S'
                      AND remessa_fatura.cd_fatura = fatura.cd_fatura
                      AND reg_fat.cd_reg_fat = itreg_fat.cd_reg_fat
                      AND reg_fat.cd_reg_fat = itlan_med.cd_reg_fat
                      AND itlan_med.cd_reg_fat = reg_fat.cd_reg_fat
                      AND itreg_fat.cd_lancamento = itlan_med.cd_lancamento
                      AND reg_fat.cd_convenio = convenio.cd_convenio
                      AND itreg_fat.cd_pro_fat = pro_fat.cd_pro_fat
                      AND reg_fat.cd_atendimento = atendime.cd_atendimento
                      AND atendime.cd_paciente = paciente.cd_paciente
                      AND ITLAN_MED.CD_ATI_MED = ATI_MED.CD_ATI_MED
                      AND prestador.cd_prestador = dbamv.pkg_repasse.fnc_fnrm_repassa_auxiliar( itlan_med.cd_reg_fat, itlan_med.cd_lancamento, itlan_med.cd_prestador,itlan_med.cd_ati_med )
                      AND prestador.cd_prestador = empresa_prestador.cd_prestador
                      AND empresa_prestador.cd_multi_empresa = reg_fat.cd_multi_empresa
                      :CAMPO_CF_SQL_HOSPITALAR
)
ORDER BY CD_CONTA,CD_ATENDIMENTO,CD_PACIENTE,NM_PACIENTE,TP_ATENDIMENTO
