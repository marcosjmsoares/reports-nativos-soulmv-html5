--R_REPASSE_PAC_VIRTUAL_RESUMO
SELECT * FROM (
SELECT CD_GRU_PRO,
	   DS_GRU_PRO,
   	   SUM(VL_TOTAL) VL_TOTAL,
   	   SUM(VL_REPASSE) VL_REPASSE,
       SUM(VL_TOTAL_CONTA) VL_TOTAL_CONTA,
  	   SUM(VL_FATURADO) VL_FATURADO,
       SUM(VL_FILME) VL_FILME,
       SUM(VL_PERC_REPASSE) VL_PERC_REPASSE,
       SUM(VL_HEMOD) VL_HEMOD,
       CD_SETOR,
       NM_SETOR,
       QUEBRA
from
(
SELECT
	   GRU_PRO.CD_GRU_PRO CD_GRU_PRO,
	   GRU_PRO.DS_GRU_PRO DS_GRU_PRO,
   	   0 VL_TOTAL,
       decode(nvl(itreg_fat.cd_setor,1), 19, 0,IT_REPASSE.VL_REPASSE)  VL_REPASSE,
       decode(nvl(itreg_fat.cd_setor,1), 19, IT_REPASSE.VL_REPASSE, 0)  VL_HEMOD,
       DECODE(ITREG_FAT.SN_PERTENCE_PACOTE, 'N', ITREG_FAT.VL_TOTAL_CONTA, 0)  VL_TOTAL_CONTA,
  	   0 VL_FATURADO,
       IT_REPASSE.VL_FILME VL_FILME,
       IT_REPASSE.VL_PERC_REPASSE VL_PERC_REPASSE,
       SETOR.CD_SETOR,
       SETOR.NM_SETOR,
       DECODE(CD_CONVENIO,116,1,118,1,2) QUEBRA,
       decode(SUBSTR(REG_FAT.CD_PRO_FAT_REALIZADO,1,4),'4001',1,
                  '4002',1,
                  '4003',1,
                  '4004',1,
                  '4005',1,
                  '4006',1,2) CIR
  FROM DBAMV.REPASSE
	  ,DBAMV.IT_REPASSE
	  ,DBAMV.ITREG_FAT
	  ,DBAMV.PRO_FAT
	  ,DBAMV.REG_FAT
	  ,DBAMV.GRU_PRO
      ,DBAMV.SETOR
 WHERE REPASSE.DT_COMPETENCIA BETWEEN TO_DATE('01/'|| :CAMPO_DT_COMPET, 'DD/MM/YYYY')
   AND TO_DATE(TO_CHAR(LAST_DAY(TO_DATE(:CAMPO_DT_COMPET, 'MM/YYYY')), 'DD/MM/YYYY'), 'DD/MM/YYYY')
   AND REPASSE.CD_MULTI_EMPRESA	= :CAMPO_CD_MULTI_EMPRESA
  AND
 REPASSE.CD_REPASSE = IT_REPASSE.CD_REPASSE
   AND IT_REPASSE.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
   AND IT_REPASSE.CD_LANCAMENTO_FAT	= ITREG_FAT.CD_LANCAMENTO
   AND ITREG_FAT.CD_PRO_FAT	= PRO_FAT.CD_PRO_FAT
   AND PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
   AND ITREG_FAT.CD_REG_FAT	= REG_FAT.CD_REG_FAT
   AND ITREG_FAT.CD_SETOR = SETOR.CD_SETOR
   AND NOT EXISTS (SELECT 1
                                      FROM DBAMV.ITLAN_MED
                                   WHERE CD_REG_FAT = IT_REPASSE.CD_REG_FAT
                                        AND CD_LANCAMENTO = IT_REPASSE.CD_LANCAMENTO_FAT
                                        AND CD_ATI_MED = IT_REPASSE.CD_ATI_MED)
UNION ALL
SELECT GRU_PRO.CD_GRU_PRO CD_GRU_PRO,
	   GRU_PRO.DS_GRU_PRO DS_GRU_PRO,
       0 VL_TOTAL,
       decode(nvl(itreg_fat.cd_setor,1), 19, 0,IT_REPASSE.VL_REPASSE)  VL_REPASSE,
       decode(nvl(itreg_fat.cd_setor,1), 19, IT_REPASSE.VL_REPASSE, 0)  VL_HEMOD,
	   DECODE(ITREG_FAT.SN_PERTENCE_PACOTE, 'N', NVL(NVL(ITLAN_MED.VL_LIQUIDO, (ITLAN_MED.VL_ATO * NVL(ITREG_FAT.QT_LANCAMENTO,1))),0), 0) VL_TOTAL_CONTA,
	   0 VL_FATURADO,
       IT_REPASSE.VL_FILME VL_FILME,
       IT_REPASSE.VL_PERC_REPASSE VL_PERC_REPASSE,
       SETOR.CD_SETOR,
       SETOR.NM_SETOR,
       DECODE(CD_CONVENIO,116,1,118,1,2) QUEBRA,
       decode(SUBSTR(REG_FAT.CD_PRO_FAT_REALIZADO,1,4),'4001',1,
                  '4002',1,
                  '4003',1,
                  '4004',1,
                  '4005',1,
                  '4006',1,2) CIR
       FROM DBAMV.REPASSE
	  ,DBAMV.IT_REPASSE
	  ,DBAMV.ITREG_FAT
	  ,DBAMV.ITLAN_MED
	  ,DBAMV.PRO_FAT
	  ,DBAMV.REG_FAT
	  ,DBAMV.GRU_PRO
      ,DBAMV.SETOR
 WHERE REPASSE.DT_COMPETENCIA BETWEEN TO_DATE('01/'|| :CAMPO_DT_COMPET, 'DD/MM/YYYY')
   AND TO_DATE(TO_CHAR(LAST_DAY(TO_DATE(:CAMPO_DT_COMPET, 'MM/YYYY')), 'DD/MM/YYYY'), 'DD/MM/YYYY')
   AND REPASSE.CD_MULTI_EMPRESA	= :CAMPO_CD_MULTI_EMPRESA
  AND
    REPASSE.CD_REPASSE = IT_REPASSE.CD_REPASSE
   AND IT_REPASSE.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
   AND IT_REPASSE.CD_LANCAMENTO_FAT	= ITREG_FAT.CD_LANCAMENTO
   AND IT_REPASSE.CD_ATI_MED = ITLAN_MED.CD_ATI_MED
   AND ITREG_FAT.CD_REG_FAT	= ITLAN_MED.CD_REG_FAT
   AND ITREG_FAT.CD_LANCAMENTO = ITLAN_MED.CD_LANCAMENTO
   AND ITREG_FAT.CD_PRESTADOR IS NULL
   AND ITREG_FAT.CD_PRO_FAT	= PRO_FAT.CD_PRO_FAT
   AND PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
   AND ITREG_FAT.CD_REG_FAT	= REG_FAT.CD_REG_FAT
   AND ITREG_FAT.CD_SETOR = SETOR.CD_SETOR
UNION ALL
SELECT GRU_PRO.CD_GRU_PRO CD_GRU_PRO,
	   GRU_PRO.DS_GRU_PRO DS_GRU_PRO,
       0 VL_TOTAL,
       decode(nvl(itreg_amb.cd_setor,1), 19, 0,IT_REPASSE.VL_REPASSE)  VL_REPASSE,
       decode(nvl(itreg_amb.cd_setor,1), 19, IT_REPASSE.VL_REPASSE, 0)  VL_HEMOD,
	   DECODE(ITREG_AMB.SN_PERTENCE_PACOTE, 'N',ITREG_AMB.VL_TOTAL_CONTA, 0) VL_TOTAL_CONTA,
	   0 VL_FATURADO,
       IT_REPASSE.VL_FILME VL_FILME,
       IT_REPASSE.VL_PERC_REPASSE VL_PERC_REPASSE,
       SETOR.CD_SETOR,
       SETOR.NM_SETOR,
       DECODE(REG_AMB.CD_CONVENIO,116,1,118,1,2) QUEBRA,
       TO_NUMBER('2') CIR
  FROM DBAMV.REPASSE
	  ,DBAMV.IT_REPASSE
	  ,DBAMV.ITREG_AMB
	  ,DBAMV.PRO_FAT
	  ,DBAMV.REG_AMB
	  ,DBAMV.GRU_PRO
      ,DBAMV.SETOR
 WHERE REPASSE.DT_COMPETENCIA BETWEEN TO_DATE('01/'|| :CAMPO_DT_COMPET, 'DD/MM/YYYY')
   AND TO_DATE(TO_CHAR(LAST_DAY(TO_DATE(:CAMPO_DT_COMPET, 'MM/YYYY')), 'DD/MM/YYYY'), 'DD/MM/YYYY')
  AND REPASSE.CD_MULTI_EMPRESA	= :CAMPO_CD_MULTI_EMPRESA
  AND
      REPASSE.CD_REPASSE = IT_REPASSE.CD_REPASSE
   AND IT_REPASSE.CD_REG_AMB = ITREG_AMB.CD_REG_AMB
   AND IT_REPASSE.CD_LANCAMENTO_AMB = ITREG_AMB.CD_LANCAMENTO
   AND ITREG_AMB.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
   AND PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
   AND ITREG_AMB.CD_REG_AMB = REG_AMB.CD_REG_AMB
   AND ITREG_AMB.CD_SETOR = SETOR.CD_SETOR
UNION ALL
SELECT TO_NUMBER(NULL) CD_GRU_PRO,
       '' DS_GRU_PRO,
       decode(nvl(repasse_prestador.cd_setor,1), 19, 0,(NVL(REPASSE_PRESTADOR.VL_REPASSE,0) - NVL(REPASSE_PRESTADOR.VL_DESCONTO,0)))  VL_TOTAL,
	   to_number(null) VL_REPASSE,
       decode(nvl(repasse_prestador.cd_setor,1), 19, (NVL(REPASSE_PRESTADOR.VL_REPASSE,0) - NVL(REPASSE_PRESTADOR.VL_DESCONTO,0)), 0)  VL_HEMOD,
       to_number(null) VL_TOTAL_CONTA,
	   nvl(repasse_prestador.vl_faturado,0)	VL_FATURADO,
       0 VL_FILME,
       0 VL_PERC_REPASSE,
       0  CD_SETOR,
       '' NM_SETOR,
      to_number(null) quebra,
     TO_NUMBER('2') CIR
  FROM DBAMV.REPASSE_PRESTADOR
      ,DBAMV.REPASSE
 WHERE REPASSE.DT_COMPETENCIA BETWEEN TO_DATE('01/'|| :CAMPO_DT_COMPET, 'DD/MM/YYYY')
   AND TO_DATE(TO_CHAR(LAST_DAY(TO_DATE(:CAMPO_DT_COMPET, 'MM/YYYY')), 'DD/MM/YYYY'), 'DD/MM/YYYY')
   AND REPASSE.CD_MULTI_EMPRESA	= :CAMPO_CD_MULTI_EMPRESA
  AND
   REPASSE.CD_REPASSE NOT IN ( SELECT DISTINCT IT_REPASSE.CD_REPASSE
                                     FROM DBAMV.IT_REPASSE  )
   AND REPASSE.CD_REPASSE NOT IN (SELECT DISTINCT IT_REPASSE_SIA.CD_REPASSE
                                    FROM DBAMV.IT_REPASSE_SIA )
   AND REPASSE.CD_REPASSE = REPASSE_PRESTADOR.CD_REPASSE
   AND REPASSE_PRESTADOR.CD_PRESTADOR = 2389
) tabela
GROUP BY CD_GRU_PRO,
	   DS_GRU_PRO,
       CD_SETOR,
       NM_SETOR,
QUEBRA
)
ORDER BY QUEBRA,
CD_GRU_PRO,DS_GRU_PRO
