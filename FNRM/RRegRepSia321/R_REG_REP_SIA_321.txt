--R_REG_REP_SIA_321
SELECT * FROM (
select regra.cd_reg_repasse_sia
     , regra.ds_reg_repasse_sia
     , convenio.cd_convenio
     , convenio.nm_convenio
     , ori_ate.cd_ori_ate
     , ori_ate.ds_ori_ate
     , tip_ate.cd_tip_ate
     , tip_ate.ds_tip_ate
     , especialid.cd_especialid
     , especialid.ds_especialid
     , grupo.cd_grupo_procedimento
     , grupo.ds_grupo_procedimento
     , subgr.cd_sub_grupo_procedimento
     , subgr.ds_sub_grupo_procedimento
     , orgr.cd_organiza_grupo_procedimento
     , orgr.ds_organiza_grupo_procedimento
     , pro.cd_procedimento
	 , pro.ds_procedimento
     , iten.vl_percent_pago
     , iten.vl_procedimento
     , iten.vl_percent_desc
     , iten.vl_desconto
     , iten.tp_valor_base
     , iten.tp_atendimento
     , iten.sn_horario_contratado
  from dbamv.reg_repasse_sia regra
     , dbamv.it_reg_repasse_sia iten
     , dbamv.convenio
     , dbamv.ori_ate
     , dbamv.tip_ate
     , dbamv.especialid
     , dbamv.grupo_procedimentos grupo
     , dbamv.sub_grupo_procedimentos subgr
     , dbamv.organiza_grupo_procedimentos orgr
     , dbamv.procedimento_sus pro
 where regra.cd_reg_repasse_sia                = iten.cd_reg_repasse_sia
   and convenio.cd_convenio (+)                = iten.cd_convenio
   and ori_ate.cd_ori_ate (+)                  = iten.cd_ori_ate
   and tip_ate.cd_tip_ate (+)                  = iten.cd_tip_ate
   and especialid.cd_especialid (+)            = iten.cd_especialid
   and grupo.cd_grupo_procedimento (+)         = iten.cd_grupo_procedimento
   and subgr.cd_grupo_procedimento (+)         = iten.cd_grupo_procedimento
   and subgr.cd_sub_grupo_procedimento (+)     = iten.cd_sub_grupo_procedimento
   and orgr.cd_grupo_procedimento (+)          = iten.cd_grupo_procedimento
   and orgr.cd_sub_grupo_procedimento (+)      = iten.cd_sub_grupo_procedimento
   and orgr.cd_organiza_grupo_procedimento (+) = iten.cd_organiza_grupo_procedimento
   and pro.cd_procedimento (+)                 = iten.cd_procedimento
 order by regra.cd_reg_repasse_sia
        , convenio.cd_convenio
        , ori_ate.cd_ori_ate
        , tip_ate.cd_tip_ate
        , especialid.cd_especialid
        , grupo.cd_grupo_procedimento
        , subgr.cd_sub_grupo_procedimento
        , orgr.cd_organiza_grupo_procedimento
        , pro.cd_procedimento
)
ORDER BY CD_REG_REPASSE_SIA,DS_REG_REPASSE_SIA
