--R_REPASSE_REMESSA
SELECT * FROM (
SELECT 	 CD_GRU_REP
	,DS_GRU_REP
	,CD_PRESTADOR
	,NM_PRESTADOR
	,SUM(VL_REPASSE) 	VL_REPASSE
	,SUM(VL_TOTAL_CONTA)	VL_TOTAL_CONTA
	,CD_REMESSA
	,CD_CONVENIO_AUX
	,NM_CONVENIO
	,SUM(VL_BASE_REPASSADO) VL_BASE_REPASSADO
	,TP_CONVENIO
FROM
(
SELECT GRU_REP.CD_PRESTADOR		CD_GRU_REP
      ,GRU_REP.NM_PRESTADOR		DS_GRU_REP
      ,PRESTADOR.CD_PRESTADOR
      ,PRESTADOR.NM_PRESTADOR
      ,IT_REPASSE.VL_REPASSE
      ,DECODE(ITREG_FAT.CD_PRESTADOR, NULL, 0, ITREG_FAT.VL_TOTAL_CONTA)  VL_TOTAL_CONTA
      ,DECODE(CONVENIO.TP_CONVENIO, 'P', TO_NUMBER(NULL), REG_FAT.CD_REMESSA) CD_REMESSA
      ,CONVENIO.CD_CONVENIO		CD_CONVENIO_AUX
      ,CONVENIO.NM_CONVENIO
      , NVL(ITREG_FAT.VL_BASE_REPASSADO,0)  VL_BASE_REPASSADO
      , CONVENIO.TP_CONVENIO		TP_CONVENIO
FROM    DBAMV.REPASSE
             ,DBAMV.IT_REPASSE
             ,DBAMV.ITREG_FAT
             ,DBAMV.PRESTADOR
             ,DBAMV.PRESTADOR  GRU_REP
             ,DBAMV.REG_FAT
             ,DBAMV.CONVENIO
             ,DBAMV.REMESSA_FATURA
             ,DBAMV.FATURA
WHERE  REMESSA_FATURA.CD_FATURA	= FATURA.CD_FATURA
AND    FATURA.CD_MULTI_EMPRESA	= :CAMPO_CD_MULTI_EMPRESA
AND    REMESSA_FATURA.CD_REMESSA	= REG_FAT.CD_REMESSA
AND    REG_FAT.CD_CONVENIO		= CONVENIO.CD_CONVENIO
AND    REG_FAT.CD_REG_FAT                     	= ITREG_FAT.CD_REG_FAT
AND    NVL(ITREG_FAT.TP_PAGAMENTO, 'X')	<> 'C'
AND    CONVENIO.TP_CONVENIO		<> 'H'
AND    ITREG_FAT.CD_REG_FAT		= IT_REPASSE.CD_REG_FAT(+)
AND    ITREG_FAT.CD_LANCAMENTO   	= IT_REPASSE.CD_LANCAMENTO_FAT(+)
AND    IT_REPASSE.CD_REPASSE                     	= REPASSE.CD_REPASSE(+)
AND    ITREG_FAT.CD_PRESTADOR                	= PRESTADOR.CD_PRESTADOR
AND    PRESTADOR.CD_PRESTADOR_REPASSE = GRU_REP.CD_PRESTADOR(+)
UNION ALL
SELECT GRU_REP.CD_PRESTADOR                      CD_GRU_REP
      ,GRU_REP.NM_PRESTADOR                             DS_GRU_REP
      ,PRESTADOR.CD_PRESTADOR
      ,PRESTADOR.NM_PRESTADOR
      ,IT_REPASSE.VL_REPASSE
      ,decode(ITLAN_MED.CD_PRESTADOR, NULL, 0 , NVL(NVL(ITLAN_MED.VL_LIQUIDO, (ITLAN_MED.VL_ATO * NVL(ITREG_FAT.QT_LANCAMENTO, 1))),0))                              VL_TOTAL_CONTA
      ,DECODE(CONVENIO.TP_CONVENIO, 'P', TO_NUMBER(NULL), REG_FAT.CD_REMESSA) CD_REMESSA
      ,CONVENIO.CD_CONVENIO                                CD_CONVENIO_AUX
      ,CONVENIO.NM_CONVENIO
      , NVL(ITLAN_MED.VL_BASE_REPASSADO,0)     VL_BASE_REPASSADO
      , CONVENIO.TP_CONVENIO                                TP_CONVENIO
FROM    DBAMV.REPASSE
       ,DBAMV.IT_REPASSE
       ,DBAMV.ITREG_FAT
       ,DBAMV.ITLAN_MED
       ,DBAMV.PRESTADOR
       ,DBAMV.PRESTADOR  GRU_REP
       ,DBAMV.REG_FAT
       ,DBAMV.CONVENIO
       ,DBAMV.REMESSA_FATURA
       ,DBAMV.FATURA
WHERE  REMESSA_FATURA.CD_FATURA         	= FATURA.CD_FATURA
AND    FATURA.CD_MULTI_EMPRESA	= :CAMPO_CD_MULTI_EMPRESA
AND    REMESSA_FATURA.CD_REMESSA	= REG_FAT.CD_REMESSA
AND    REG_FAT.CD_CONVENIO       	               	= CONVENIO.CD_CONVENIO
AND    NVL(REG_FAT.SN_DIAGNO, 'N')	= 'N'
AND    REG_FAT.CD_REG_FAT                       	= ITREG_FAT.CD_REG_FAT
AND    NVL(ITREG_FAT.TP_PAGAMENTO, 'X')	<> 'C'
AND    CONVENIO.TP_CONVENIO		<> 'H'
AND    ITREG_FAT.CD_REG_FAT                       	= ITLAN_MED.CD_REG_FAT
AND    ITREG_FAT.CD_LANCAMENTO               	= ITLAN_MED.CD_LANCAMENTO
AND    ITREG_FAT.CD_PRESTADOR                  	IS NULL
AND    ITREG_FAT.CD_REG_FAT		= IT_REPASSE.CD_REG_FAT
AND    ITREG_FAT.CD_LANCAMENTO 	= IT_REPASSE.CD_LANCAMENTO_FAT
AND     ITLAN_MED.CD_ATI_MED		= IT_REPASSE.CD_ATI_MED
AND    IT_REPASSE.CD_REPASSE		= REPASSE.CD_REPASSE
AND    ITLAN_MED.CD_PRESTADOR                	= PRESTADOR.CD_PRESTADOR
AND    PRESTADOR.CD_PRESTADOR_REPASSE  = GRU_REP.CD_PRESTADOR(+)
UNION ALL
SELECT GRU_REP.CD_PRESTADOR                 CD_GRU_REP
      ,GRU_REP.NM_PRESTADOR                        DS_GRU_REP
      ,PRESTADOR.CD_PRESTADOR
      ,PRESTADOR.NM_PRESTADOR
      ,IT_REPASSE.VL_REPASSE
      ,DECODE(ITREG_AMB.CD_PRESTADOR, NULL, 0, ITREG_AMB.VL_TOTAL_CONTA) VL_TOTAL_CONTA
      ,DECODE(CONVENIO.TP_CONVENIO, 'P', TO_NUMBER(NULL), REG_AMB.CD_REMESSA) CD_REMESSA
      ,CONVENIO.CD_CONVENIO		CD_CONVENIO_AUX
      ,CONVENIO.NM_CONVENIO
      , NVL(ITREG_AMB.VL_BASE_REPASSADO,0)      VL_BASE_REPASSADO
      , CONVENIO.TP_CONVENIO                            TP_CONVENIO
FROM    DBAMV.REPASSE
       ,DBAMV.IT_REPASSE
       ,DBAMV.ITREG_AMB
       ,DBAMV.PRESTADOR
       ,DBAMV.PRESTADOR    GRU_REP
       ,DBAMV.REG_AMB
       ,DBAMV.CONVENIO
       ,DBAMV.REMESSA_FATURA
       ,DBAMV.FATURA
WHERE   REMESSA_FATURA.CD_FATURA                	= FATURA.CD_FATURA
AND    FATURA.CD_MULTI_EMPRESA		= :CAMPO_CD_MULTI_EMPRESA
AND    REMESSA_FATURA.CD_REMESSA		= REG_AMB.CD_REMESSA
AND    NVL(REG_AMB.SN_DIAGNO, 'N')		= 'N'
AND    REG_AMB.CD_REG_AMB                          	= ITREG_AMB.CD_REG_AMB
AND    ITREG_AMB.CD_CONVENIO 	                        	= CONVENIO.CD_CONVENIO
AND    NVL(ITREG_AMB.TP_PAGAMENTO, 'X')		<> 'C'
AND    CONVENIO.TP_CONVENIO			<> 'H'
AND    ITREG_AMB.CD_REG_AMB			= IT_REPASSE.CD_REG_AMB(+)
AND    ITREG_AMB.CD_LANCAMENTO		= IT_REPASSE.CD_LANCAMENTO_AMB(+)
AND    IT_REPASSE.CD_REPASSE                               	= REPASSE.CD_REPASSE(+)
AND    ITREG_AMB.CD_PRESTADOR                     	= PRESTADOR.CD_PRESTADOR(+)
AND    PRESTADOR.CD_PRESTADOR_REPASSE 	= GRU_REP.CD_PRESTADOR(+)
union all
SELECT    DISTINCT 0			CD_GRU_REP
	,null     				DS_GRU_REP
      	,0    				CD_PRESTADOR
	,null				NM_PRESTADOR
	,0				VL_REPASSE
	,0				VL_TOTAL_CONTA
 	,DECODE(CONVENIO.TP_CONVENIO, 'P', TO_NUMBER(NULL), REG_FAT.CD_REMESSA) CD_REMESSA
	,CONVENIO.CD_CONVENIO		CD_CONVENIO_AUX
	,CONVENIO.NM_CONVENIO
	, 0       				VL_BASE_REPASSADO
	, CONVENIO.TP_CONVENIO		TP_CONVENIO
FROM    DBAMV.ITREG_FAT
             ,DBAMV.REG_FAT
             ,DBAMV.CONVENIO
             ,DBAMV.REMESSA_FATURA
             ,DBAMV.FATURA
WHERE  REMESSA_FATURA.CD_FATURA	= FATURA.CD_FATURA
AND    FATURA.CD_MULTI_EMPRESA	= :CAMPO_CD_MULTI_EMPRESA
AND    REMESSA_FATURA.CD_REMESSA	= REG_FAT.CD_REMESSA
AND    REG_FAT.CD_CONVENIO		= CONVENIO.CD_CONVENIO
AND    REG_FAT.CD_REG_FAT           = ITREG_FAT.CD_REG_FAT
AND    NVL(ITREG_FAT.TP_PAGAMENTO, 'X')	<> 'C'
AND    ITREG_FAT.SN_PERTENCE_PACOTE = 'N'
AND    CONVENIO.TP_CONVENIO		<> 'H'
AND    ITREG_FAT.CD_PRESTADOR IS NULL
)
GROUP BY CD_GRU_REP
	,DS_GRU_REP
	,CD_PRESTADOR
	,NM_PRESTADOR
	,CD_REMESSA
	,CD_CONVENIO_AUX
	,NM_CONVENIO
	,TP_CONVENIO
)
ORDER BY NM_CONVENIO,CD_CONVENIO_AUX,
CD_REMESSA,TP_CONVENIO,
CD_PRESTADOR,NM_PRESTADOR
