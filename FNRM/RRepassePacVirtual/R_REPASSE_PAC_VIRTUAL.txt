--R_REPASSE_PAC_VIRTUAL
SELECT * FROM (
SELECT RVA.CD_ATENDIMENTO
            ,  RVA.NR_CONTA CONTA
            ,  RVA.DT_INICIO DT_ATENDIMENTO
            ,  RVA.DT_FINAL DT_ALTA
            ,  RVA.CD_CONVENIO
            ,  C.NM_CONVENIO
            ,  dbamv.pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(P.CD_PACIENTE, 1) NM_PACIENTE
            ,  RVA.CD_GRU_PRO
            ,  G.DS_GRU_PRO
            ,  SUM(RVA.VL_TOTAL) VL_TOTAL_CONTA
            ,  SUM(RVA.VL_VIRTUAL) VL_VIRTUAL
            ,  SUM(RVA.VL_HEMODINAMICA) VL_HEMOD
            ,  RVA.CD_SETOR
            ,  S.NM_SETOR
            ,  DECODE(RVA.CD_CONVENIO,116,1,118,1,2) QUEBRA
  FROM DBAMV.REPASSE_VIRTUAL_ATENDIME RVA
           , DBAMV.CONVENIO C
           , DBAMV.PACIENTE P
           , DBAMV.SETOR S
           , DBAMV.GRU_PRO G
  WHERE RVA.DT_COMPETENCIA = TO_DATE('01/'||:CAMPO_DT_COMPET,'DD/MM/YYYY')
        AND RVA.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
        AND RVA.CD_CONVENIO = C.CD_CONVENIO
        AND RVA.CD_PACIENTE = P.CD_PACIENTE
        AND RVA.CD_SETOR = S.CD_SETOR
        AND RVA.CD_GRU_PRO = G.CD_GRU_PRO
GROUP BY RVA.CD_ATENDIMENTO
                 ,  RVA.NR_CONTA
                 ,  RVA.DT_INICIO
                 ,  RVA.DT_FINAL
                 ,  RVA.CD_CONVENIO
                 ,  C.NM_CONVENIO
                 ,  dbamv.pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(P.CD_PACIENTE, 1)
                 ,  RVA.CD_GRU_PRO
                 ,  G.DS_GRU_PRO
                 ,  RVA.CD_SETOR
                 ,  S.NM_SETOR
                 ,  DECODE(RVA.CD_CONVENIO,116,1,118,1,2)
)
ORDER BY QUEBRA DESC,
NM_CONVENIO,NM_PACIENTE,DT_ATENDIMENTO,CD_CONVENIO,CONTA,DT_ALTA,CD_ATENDIMENTO,
CD_GRU_PRO,DS_GRU_PRO
