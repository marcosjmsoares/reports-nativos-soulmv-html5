--R_REG_REP_H9J
SELECT  ITREG_REPASSE_H9J.CD_ITREG_REPASSE_H9J
               ,ITREG_REPASSE_H9J.CD_CONVENIO
               ,ITREG_REPASSE_H9J.CD_ORI_ATE
               ,ITREG_REPASSE_H9J.CD_PRO_FAT
               ,ITREG_REPASSE_H9J.CD_GRU_PRO
               ,ITREG_REPASSE_H9J.VL_PERCENT_PAGO
               ,ITREG_REPASSE_H9J.VL_PROCEDIMENTO
               ,ITREG_REPASSE_H9J.SN_PERTENCE_PACOTE
               ,ITREG_REPASSE_H9J.SN_INSIDE_NO_FILME
               ,ITREG_REPASSE_H9J.CD_SETOR
               ,ITREG_REPASSE_H9J.CD_TIP_ACOM
               ,CONVENIO.NM_CONVENIO
               ,ORI_ATE.DS_ORI_ATE
               ,PRO_FAT.DS_PRO_FAT
               ,GRU_PRO.DS_GRU_PRO
               ,SETOR.NM_SETOR
               ,TIP_ACOM.DS_TIP_ACOM
               ,PRESTADOR.CD_PRESTADOR
               ,ESPECIALID.CD_ESPECIALID
               ,PRESTADOR.NM_PRESTADOR
               ,ESPECIALID.DS_ESPECIALID
FROM      DBAMV.ITREG_REPASSE_H9J
               ,DBAMV.CONVENIO
               ,DBAMV.ORI_ATE
               ,DBAMV.PRO_FAT
               ,DBAMV.GRU_PRO
               ,DBAMV.SETOR
               ,DBAMV.TIP_ACOM
               ,DBAMV.PRESTADOR
               ,DBAMV.ESPECIALID
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And CONVENIO.CD_CONVENIO(+)  =  ITREG_REPASSE_H9J.CD_CONVENIO
AND        ORI_ATE.CD_ORI_ATE(+)  =  ITREG_REPASSE_H9J.CD_ORI_ATE
AND        PRO_FAT.CD_PRO_FAT(+)  =  ITREG_REPASSE_H9J.CD_PRO_FAT
AND        GRU_PRO.CD_GRU_PRO(+)  =  ITREG_REPASSE_H9J.CD_GRU_PRO
AND        SETOR.CD_SETOR(+) = ITREG_REPASSE_H9J.CD_SETOR
AND        TIP_ACOM.CD_TIP_ACOM(+) = ITREG_REPASSE_H9J.CD_TIP_ACOM
AND        ESPECIALID.CD_ESPECIALID(+) = ITREG_REPASSE_H9J.CD_ESPECIALID
AND        PRESTADOR.CD_PRESTADOR(+) = ITREG_REPASSE_H9J.CD_PRESTADOR
