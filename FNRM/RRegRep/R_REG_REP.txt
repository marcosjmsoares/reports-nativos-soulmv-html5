--R_REG_REP
SELECT  distinct RRC.CD_REG_REPASSE CODIGO
               ,RRC.DS_REG_REPASSE DESCRICAO
FROM     dbamv.REG_REPASSE RRC, DBAMV.prestador, DBAMV.empresa_prestador, DBAMV.PRESTADOR PRESTREP
WHERE
     RRC.cd_reg_repasse = EMPRESA_PRESTADOR.CD_REG_REPASSE
     and PRESTREP.cd_prestador = empresa_prestador.cd_prestador
     and PRESTADOR.CD_PRESTADOR = PRESTREP.CD_PRESTADOR_REPASSE
     and empresa_prestador.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
ORDER BY

--R_REG_REP_CG__R_IREG_REP_Q_IT_REGRA
SELECT *
   FROM
  (SELECT ITREG_REPASSE.CD_REG_REPASSE ,
    ITREG_REPASSE.CD_CONVENIO          ,
    ITREG_REPASSE.CD_ORI_ATE           ,
    ITREG_REPASSE.CD_PRO_FAT           ,
    ITREG_REPASSE.CD_GRU_PRO           ,
    ITREG_REPASSE.VL_PERCENT_PAGO      ,
    ITREG_REPASSE.VL_PERCENT_DESC      ,
    ITREG_REPASSE.VL_PROCEDIMENTO      ,
    ITREG_REPASSE.VL_DESCONTO          ,
    ITREG_REPASSE.SN_PERTENCE_PACOTE   ,
    ITREG_REPASSE.SN_INSIDE_NO_FILME   ,
    ITREG_REPASSE.CD_ATI_MED           ,
    ITREG_REPASSE.CD_SETOR             ,
    ITREG_REPASSE.CD_TIP_ACOM          ,
    ITREG_REPASSE.SN_CH_FATURADO ,
    ITREG_REPASSE.QT_CH          ,
    ITREG_REPASSE.TP_ATENDIMENTO ,
    ITREG_REPASSE.SN_HORARIO_CONTRATADO,
    CONVENIO.NM_CONVENIO ,
    ORI_ATE.DS_ORI_ATE   ,
    PRO_FAT.DS_PRO_FAT   ,
    GRU_PRO.DS_GRU_PRO   ,
    ATI_MED.DS_ATI_MED   ,
    SETOR.NM_SETOR       ,
    TIP_ACOM.DS_TIP_ACOM
     FROM DBAMV.ITREG_REPASSE ,
    DBAMV.CONVENIO            ,
    DBAMV.ORI_ATE             ,
    DBAMV.PRO_FAT             ,
    DBAMV.GRU_PRO             ,
    DBAMV.ATI_MED             ,
    DBAMV.SETOR               ,
    DBAMV.TIP_ACOM
    WHERE
    CONVENIO.CD_CONVENIO(+)   = ITREG_REPASSE.CD_CONVENIO
  AND ORI_ATE.CD_ORI_ATE(+)   = ITREG_REPASSE.CD_ORI_ATE
  AND PRO_FAT.CD_PRO_FAT(+)   = ITREG_REPASSE.CD_PRO_FAT
  AND GRU_PRO.CD_GRU_PRO(+)   = ITREG_REPASSE.CD_GRU_PRO
  AND ATI_MED.CD_ATI_MED(+)   = ITREG_REPASSE.CD_ATI_MED
  AND SETOR.CD_SETOR(+)       = ITREG_REPASSE.CD_SETOR
  AND TIP_ACOM.CD_TIP_ACOM(+) = ITREG_REPASSE.CD_TIP_ACOM
  AND (ITREG_REPASSE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA OR ITREG_REPASSE.CD_MULTI_EMPRESA IS NULL)
 ORDER BY ITREG_REPASSE.CD_CONVENIO DESC,
    ITREG_REPASSE.CD_ORI_ATE DESC       ,
    ITREG_REPASSE.CD_GRU_PRO DESC       ,
    ITREG_REPASSE.CD_PRO_FAT DESC       ,
    ITREG_REPASSE.CD_ATI_MED DESC       ,
    ITREG_REPASSE.CD_SETOR DESC         ,
    ITREG_REPASSE.CD_TIP_ACOM DESC
  )
  WHERE CD_REG_REPASSE = :CAMPO_CODIGO
