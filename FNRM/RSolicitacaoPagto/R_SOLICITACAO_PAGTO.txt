  FROM dbamv.con_pag con_pag_principal

                 ,tip_detcon_pag.cd_con_pag_pai cd_con_pag
             FROM dbamv.tip_detcon_pag
                 ,dbamv.con_pag con_pag_impostos
            WHERE tip_detcon_pag.cd_con_pag_filho = con_pag_impostos.cd_con_pag
               AND tip_detcon_pag.cd_con_pag_pai = :CAMPO_CD_CON_PAG
         GROUP BY tip_detcon_pag.cd_con_pag_pai) impostos
      , (SELECT repasse.ds_repasse ds_repasse
               ,repasse.cd_repasse cd_repasse
               ,repasse_prestador_con_pag_rep.cd_con_pag cd_con_pag
               ,repasse_prestador_con_pag_rep.cd_prestador cd_prestador

               ,dbamv.repasse_prestador_con_pag repasse_prestador_con_pag_rep

            AND multi_empresas_pgto_rep.cd_multi_empresa_pag = :CAMPO_CD_MULTI_EMPRESA
            AND repasse.cd_multi_empresa = multi_empresas_pgto_rep.cd_multi_empresa_pag
            AND repasse_prestador_con_pag_rep.cd_con_pag = :CAMPO_CD_CON_PAG
            AND repasse_prestador_con_pag_rep.cd_repasse IN (SELECT   MAX (repasse_prestador_con_pag_sub.cd_repasse) cd_repasse
                                                                 FROM dbamv.repasse_prestador_con_pag repasse_prestador_con_pag_sub
                                                                WHERE repasse_prestador_con_pag_sub.cd_con_pag = repasse_prestador_con_pag_rep.cd_con_pag
                                                             GROUP BY repasse_prestador_con_pag_sub.cd_con_pag)
         UNION
         SELECT repasse.ds_repasse ds_repasse

               ,repasse_prestador.cd_con_pag cd_con_pag
               ,repasse_prestador.cd_prestador cd_prestador
               ,repasse_prestador.cd_prestador_repasse cd_prestador_repasse
           FROM dbamv.repasse_prestador
               ,dbamv.repasse
               ,dbamv.multi_empresas_pgto_rep
          WHERE repasse_prestador.cd_repasse = repasse.cd_repasse
            AND multi_empresas_pgto_rep.cd_multi_empresa_pag = :CAMPO_CD_MULTI_EMPRESA
            AND repasse.cd_multi_empresa = multi_empresas_pgto_rep.cd_multi_empresa_pag
            AND repasse_prestador.cd_con_pag = :CAMPO_CD_CON_PAG
            AND NOT EXISTS (SELECT 1
                              FROM dbamv.repasse_prestador_con_pag
                             WHERE repasse_prestador_con_pag.cd_repasse = repasse_prestador.cd_repasse)) repasse
 WHERE con_pag_principal.cd_con_pag = impostos.cd_con_pag(+)
   AND con_pag_principal.cd_con_pag = repasse.cd_con_pag
   AND empresa_prestador.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
   AND empresa_prestador.cd_prestador = grupo.cd_prestador
   AND prestador.cd_prestador = repasse.cd_prestador
   AND grupo.cd_prestador = repasse.cd_prestador_repasse

                WHERE cd_prestador_repasse = repasse.cd_prestador_repasse AND reg_repasse_rateio.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA)
