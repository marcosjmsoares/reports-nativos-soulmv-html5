--R_REG_REP_SIA_CG__R_IREG_REP_Q_IT_REGRA
SELECT * FROM (
SELECT  IT_REG_REPASSE_SIA.CD_REG_REPASSE_SIA
               ,IT_REG_REPASSE_SIA.CD_CONVENIO
               ,IT_REG_REPASSE_SIA.CD_ORI_ATE
               ,IT_REG_REPASSE_SIA.CD_SSM   cd_pro_fat
               ,IT_REG_REPASSE_SIA.VL_PERCENT_PAGO
               ,IT_REG_REPASSE_SIA.VL_PERCENT_DESC
               ,IT_REG_REPASSE_SIA.VL_PROCEDIMENTO
               ,IT_REG_REPASSE_SIA.VL_DESCONTO
               ,IT_REG_REPASSE_SIA.cd_especialid
               ,IT_REG_REPASSE_SIA.cd_tip_ate
               ,CONVENIO.NM_CONVENIO
               ,ORI_ATE.DS_ORI_ATE
               ,TAB_SIASUS.DS_TITULO   DS_PROCEDIMENTO
               ,DS_ESPECIALID
               ,DS_TIP_ATE
               ,IT_REG_REPASSE_SIA.tp_valor_base
               ,IT_REG_REPASSE_SIA.TP_ATENDIMENTO
               ,IT_REG_REPASSE_SIA.SN_HORARIO_CONTRATADO
FROM     DBAMV.CONVENIO
               ,DBAMV.ORI_ATE
               ,DBAMV.tab_siasus
               ,dbamv.IT_REG_REPASSE_SIA    IT_REG_REPASSE_SIA
               ,DBAMV.TIP_ATE
               ,DBAMV.ESPECIALID
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And CONVENIO.CD_CONVENIO(+)  =  IT_REG_REPASSE_SIA.CD_CONVENIO
AND        ORI_ATE.CD_ORI_ATE(+)  =  IT_REG_REPASSE_SIA.CD_ORI_ATE
AND        tab_siasus.CD_ssm(+)  =  IT_REG_REPASSE_SIA.CD_ssm
AND        ESPECIALID.CD_ESPECIALID(+)  =  IT_REG_REPASSE_SIA.CD_ESPECIALID
AND        TIP_ATE.CD_TIP_ATE(+)  =  IT_REG_REPASSE_SIA.CD_TIP_ATE
ORDER BY  IT_REG_REPASSE_SIA.CD_ORI_ATE DESC,
          IT_REG_REPASSE_SIA.CD_CONVENIO  DESC ,
          IT_REG_REPASSE_SIA.CD_ORI_ATE   DESC,
          IT_REG_REPASSE_SIA.cd_especialid  desc,
          IT_REG_REPASSE_SIA.cd_tip_ate     desc,
          IT_REG_REPASSE_SIA.CD_ssm       DESC
)
WHERE CD_REG_REPASSE_SIA = :CAMPO_CODIGO
