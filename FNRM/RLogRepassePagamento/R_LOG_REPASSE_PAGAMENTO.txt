--R_LOG_REPASSE_PAGAMENTO
SELECT * FROM (
SELECT   log_repasse_pagamento.cd_log_repasse_pagamento
        ,log_repasse_pagamento.cd_prestador_repasse
        ,prestador.nm_prestador
        ,log_repasse_pagamento.cd_prestador
        ,grupo.nm_prestador nm_prestador_grupo
        ,log_repasse_pagamento.dt_competencia_repasse
        ,log_repasse_pagamento.ds_log_repasse_pagamento
        ,log_repasse_pagamento.ds_correcao
        ,log_repasse_pagamento.nm_usuario_baixou
        ,log_repasse_pagamento.dt_criacao_log
    FROM dbamv.log_repasse_pagamento
        ,dbamv.prestador
        ,dbamv.prestador grupo
   WHERE log_repasse_pagamento.cd_prestador = prestador.cd_prestador
     AND log_repasse_pagamento.cd_prestador_repasse = grupo.cd_prestador
     AND log_repasse_pagamento.dt_competencia_repasse BETWEEN TRUNC (TO_DATE(:CAMPO_COMPETENCIA, 'MM/YYYY'), 'MM') AND LAST_DAY (TO_DATE(:CAMPO_COMPETENCIA, 'MM/YYYY'))
ORDER BY log_repasse_pagamento.cd_prestador
,log_repasse_pagamento.nm_usuario_baixou desc
        ,log_repasse_pagamento.cd_log_repasse_pagamento
)
ORDER BY DT_COMPETENCIA_REPASSE,
CD_PRESTADOR,NM_PRESTADOR
