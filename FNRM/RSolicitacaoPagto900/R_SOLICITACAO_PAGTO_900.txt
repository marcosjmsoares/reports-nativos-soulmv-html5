--R_SOLICITACAO_PAGTO_900
SELECT      reg_rep_desc.ds_reg_rep_desc,
            nvl(rep_desc.vl_acrescimo,0)  acresi,
            nvl(rep_DESC.VL_DESCONTO,0)   desco,
            nvl(rep_desc.vl_acrescimo,0)  - nvl(rep_DESC.VL_DESCONTO,0)   tot_acre_desc
    FROM   dbamv.reg_rep_desc reg_rep_desc,
                dbamv.rep_desc     rep_desc
    WHERE  rep_desc.cd_con_pag          =  :CAMPO_CD_CON_PAG
     and   rep_desc.cd_reg_rep_desc =  reg_rep_desc.cd_reg_rep_desc


/*----------------------------------------------------------------------------*/
--R_SOLICITACAO_PAGTO_900_R_PRESTADOR_Q_SOLIC_PAGTO
SELECT * FROM (
SELECT * FROM (
SELECT   mestre.CD_PRESTADOR
              ,( NVL( REPASSE_PRESTADOR.VL_REPASSE, 0 ) - NVL( REPASSE_PRESTADOR.VL_DESCONTO, 0 ) ) VL_REPASSE
              ,MESTRE.NM_PRESTADOR
              ,MESTRE.NR_CPF_CGC
              ,MESTRE.DS_ENDERECO
               ,MESTRE.DS_COMPLEMENTO
               ,MESTRE.DS_BAIRRO
               ,MESTRE.NR_CEP
               ,MESTRE.NR_FONE_CONTATO
               ,MESTRE.NR_FONE_COMERCIAL
               , MESTRE.CD_BANCO
               ,MESTRE.NR_AGENCIA
                ,MESTRE.NM_AGENCIA
                ,MESTRE.NR_CONTA
                ,MESTRE.TP_VINCULO
                ,MESTRE.NR_CCM
                ,MESTRE.CD_FORNECEDOR
              ,REPASSE_PRESTADOR.CD_REPASSE  CD_REPASSE
              ,REPASSE.DS_REPASSE
              ,REPASSE_PRESTADOR.CD_CON_PAG
              ,FILHOS.CD_PRESTADOR     CD_PRESTADOR_FILHO
              ,FILHOS.NM_PRESTADOR    NM_PRESTADOR_FILHO
FROM        DBAMV.REPASSE
            ,DBAMV.REPASSE_PRESTADOR
            ,DBAMV.PRESTADOR mestre
            ,DBAMV.PRESTADOR FILHOS
            ,DBAMV.MULTI_EMPRESAS_PGTO_REP
WHERE     TO_CHAR(REPASSE.DT_COMPETENCIA, 'MM/YYYY')  =  :CAMPO_DT_COMPET
  AND     REPASSE_PRESTADOR.CD_CON_PAG = :CAMPO_CD_CON_PAG
  AND     REPASSE_PRESTADOR.CD_REPASSE = REPASSE.CD_REPASSE
  AND     REPASSE_PRESTADOR.CD_PRESTADOR = FILHOS.CD_PRESTADOR
  AND     REPASSE_PRESTADOR.CD_PRESTADOR_REPASSE = MESTRE.CD_PRESTADOR
  AND     MESTRE.CD_PRESTADOR  =  :CAMPO_CD_PRESTADOR
  AND     MULTI_EMPRESAS_PGTO_REP.CD_MULTI_EMPRESA_PAG = :CAMPO_CD_MULTI_EMPRESA
  AND     REPASSE.CD_MULTI_EMPRESA = MULTI_EMPRESAS_PGTO_REP.CD_MULTI_EMPRESA_REP
UNION
SELECT  mestre.CD_PRESTADOR
              ,( NVL( REPASSE_PRESTADOR.VL_REPASSE, 0 ) - NVL( REPASSE_PRESTADOR.VL_DESCONTO, 0 ) ) VL_REPASSE
               ,MESTRE.NM_PRESTADOR
              ,MESTRE.NR_CPF_CGC
              ,MESTRE.DS_ENDERECO
               ,MESTRE.DS_COMPLEMENTO
               ,MESTRE.DS_BAIRRO
               ,MESTRE.NR_CEP
               ,MESTRE.NR_FONE_CONTATO
               ,MESTRE.NR_FONE_COMERCIAL
               , MESTRE.CD_BANCO
               ,MESTRE.NR_AGENCIA
                ,MESTRE.NM_AGENCIA
                ,MESTRE.NR_CONTA
                ,MESTRE.TP_VINCULO
                ,MESTRE.NR_CCM
                ,MESTRE.CD_FORNECEDOR
              ,REPASSE_PRESTADOR.CD_REPASSE  CD_REPASSE
              ,REPASSE.DS_REPASSE
              ,REPASSE_PRESTADOR.CD_CON_PAG
              ,MESTRE.CD_PRESTADOR CD_PRESTADOR_FILHO
              ,MESTRE.NM_PRESTADOR    NM_PRESTADOR_FILHO
FROM        DBAMV.REPASSE
            ,DBAMV.REPASSE_PRESTADOR
            ,DBAMV.PRESTADOR mestre
            ,DBAMV.MULTI_EMPRESAS_PGTO_REP
WHERE     TO_CHAR(REPASSE.DT_COMPETENCIA, 'MM/YYYY')  =  :CAMPO_DT_COMPET
  AND     REPASSE_PRESTADOR.CD_CON_PAG = :CAMPO_CD_CON_PAG
  AND     REPASSE_PRESTADOR.CD_REPASSE = REPASSE.CD_REPASSE
  AND     REPASSE_PRESTADOR.CD_PRESTADOR = MESTRE.CD_PRESTADOR
  AND     MESTRE.CD_PRESTADOR  =  :CAMPO_CD_PRESTADOR
  AND     MULTI_EMPRESAS_PGTO_REP.CD_MULTI_EMPRESA_PAG = :CAMPO_CD_MULTI_EMPRESA
  AND     REPASSE.CD_MULTI_EMPRESA = MULTI_EMPRESAS_PGTO_REP.CD_MULTI_EMPRESA_REP
order by cd_repasse
)
WHERE CD_PRESTADOR = :CAMPO_CD_PRESTADOR
)
ORDER BY NM_PRESTADOR,CD_PRESTADOR
