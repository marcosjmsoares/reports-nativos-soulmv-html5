--R_13_REPASSE
SELECT * FROM (
SELECT	'	' ds_gru_rep
		  ,mes
		  ,nm_prestador
		  ,SUM (valor) valor
	 FROM (SELECT distinct ' ' ds_gru_rep
					 ,TO_CHAR (repasse.dt_competencia, 'MM') mes
					 ,NVL (itcon_pag.vl_duplicata, 0) valor
					 ,prestador.nm_prestador nm_prestador
					 ,itcon_pag.cd_con_pag cd_con_pag
				FROM dbamv.repasse
					 ,dbamv.repasse_prestador
					 ,dbamv.itcon_pag
					 ,dbamv.con_pag
					 ,dbamv.prestador
			  WHERE TO_CHAR (repasse.dt_competencia, 'YYYY') = :CAMPO_DT_COMPET
				 AND repasse.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
				 AND repasse.cd_repasse = repasse_prestador.cd_repasse
				 AND repasse_prestador.cd_con_pag IS NOT NULL
				 AND repasse_prestador.cd_con_pag = con_pag.cd_con_pag
				 AND con_pag.cd_con_pag = itcon_pag.cd_con_pag
				 AND repasse_prestador.cd_prestador = prestador.cd_prestador
 	          :CAMPO_CF_COND_PRESTADOR
			)
GROUP BY ds_gru_rep
		  ,mes
		  ,nm_prestador
)
ORDER BY DS_GRU_REP,
NM_PRESTADOR,
MES
