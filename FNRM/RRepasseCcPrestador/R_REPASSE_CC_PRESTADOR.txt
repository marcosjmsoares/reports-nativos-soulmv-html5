    	,NM_SETOR

               ,GRU_PRESTADOR.NM_PRESTADOR 					NM_PRESTADOR
               ,NVL(IT_REPASSE.VL_REPASSE,0) - NVL(IT_REPASSE.VL_GLOSA,0)		VL_REPASSE
