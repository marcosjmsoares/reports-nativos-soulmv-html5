--R_ATEND_PAC_VIRTUAL
SELECT DISTINCT ATENDIME.CD_ATENDIMENTO CD_ATENDIMENTO,
       REG_FAT.CD_REG_FAT CONTA,
       ATENDIME.DT_ATENDIMENTO DT_ATENDIMENTO,
	   ATENDIME.DT_ALTA DT_ALTA,
       CONVENIO.CD_CONVENIO	CD_CONVENIO,
       CONVENIO.NM_CONVENIO NM_CONVENIO,
       dbamv.pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(PACIENTE.CD_PACIENTE, 1) NM_PACIENTE
  FROM DBAMV.ATENDIME
	  ,DBAMV.REG_FAT
      ,DBAMV.ITREG_FAT
	  ,DBAMV.CONVENIO
	  ,DBAMV.PACIENTE
 WHERE ATENDIME.DT_ATENDIMENTO BETWEEN :CAMPO_DT_INI and 
   AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
   AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
   AND ATENDIME.CD_ATENDIMENTO = REG_FAT.CD_ATENDIMENTO
   AND REG_FAT.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
   AND REG_FAT.CD_CONVENIO = CONVENIO.CD_CONVENIO
   AND ATENDIME.CD_PACIENTE	= PACIENTE.CD_PACIENTE
   AND ITREG_FAT.CD_SETOR IN (19, 50)
UNION ALL
SELECT DISTINCT ATENDIME.CD_ATENDIMENTO CD_ATENDIMENTO,
       REG_AMB.CD_REG_AMB CONTA,
       ATENDIME.DT_ATENDIMENTO DT_ATENDIMENTO,
	   ATENDIME.DT_ALTA DT_ALTA,
       CONVENIO.CD_CONVENIO	CD_CONVENIO,
       CONVENIO.NM_CONVENIO NM_CONVENIO,
       dbamv.pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(PACIENTE.CD_PACIENTE, 1) NM_PACIENTE
  FROM DBAMV.ATENDIME
	  ,DBAMV.ITREG_AMB
	  ,DBAMV.REG_AMB
	  ,DBAMV.CONVENIO
	  ,DBAMV.PACIENTE
 WHERE ATENDIME.DT_ATENDIMENTO BETWEEN :CAMPO_DT_INI and 
   AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
   AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
   AND ATENDIME.CD_ATENDIMENTO = ITREG_AMB.CD_ATENDIMENTO
   AND ITREG_AMB.CD_REG_AMB = REG_AMB.CD_REG_AMB
   AND REG_AMB.CD_CONVENIO = CONVENIO.CD_CONVENIO
   AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
   AND ITREG_AMB.CD_SETOR IN (19,50)
ORDER BY NM_CONVENIO, NM_PACIENTE
