--R_PRO_FAT_SEM_PRESTADOR
SELECT        PRESTADOR.CD_PRESTADOR
             ,PRESTADOR.NM_PRESTADOR
             ,pro_fat_sem_prestador.cd_gru_pro
             ,pro_fat_sem_prestador.cd_pro_fat
             ,pro_fat_sem_prestador.cd_servico
             ,pro_fat_sem_prestador.cd_setor
             ,gru_pro.ds_gru_pro
             ,pro_fat.ds_pro_fat
             ,servico.ds_servico
             ,setor.nm_setor
             ,pro_fat_sem_prestador.sn_valor_lucro
FROM   DBAMV.PRESTADOR,
             DBAMV.GRU_PRO,
             DBAMV.PRO_FAT,
             DBAMV.SETOR,
             DBAMV.SERVICO,
             dbamv.pro_fat_sem_prestador
Where  pro_fat_sem_prestador.cd_prestador = prestador.cd_prestador
     and pro_fat_sem_prestador.cd_gru_pro = gru_pro.cd_gru_pro(+)
     and pro_fat_sem_prestador.cd_pro_fat = pro_fat.cd_pro_fat(+)
     and pro_fat_sem_prestador.cd_servico = servico.cd_servico(+)
     and pro_fat_sem_prestador.cd_setor = setor.cd_setor(+)
order by     pro_fat_sem_prestador.cd_gru_pro
             ,pro_fat_sem_prestador.cd_pro_fat
             ,pro_fat_sem_prestador.cd_setor
