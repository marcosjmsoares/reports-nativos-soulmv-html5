--R_REPASSE_CONTA
SELECT * FROM (
SELECT DISTINCT reg_amb.cd_remessa
                         ,atendime.dt_atendimento
                         ,reg_amb.cd_reg_amb cd_conta
                         ,reg_amb.cd_convenio
                         ,prestador.cd_prestador
                         ,REPASSE_CONTA.vl_faturado
                         ,REPASSE_CONTA.vl_repassado
                         ,empresa_prestador.VL_PERC_REPASSE_CONTA
                         ,convenio.nm_convenio
                         ,atendime.cd_paciente
                         ,atendime.cd_ori_ate
                         ,ori_ate.ds_ori_ate
                         ,dbamv.pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(paciente.cd_paciente, 1) nm_paciente
                         ,prestador.nm_prestador
                         ,fatura.dt_competencia
                         ,empresa_prestador.cd_prestador_repasse
                         ,prestador_repasse.nm_prestador nm_prestador_repasse
                         ,decode(atendime.tp_atendimento,'A','Ambulatório','E','Externo','U','Urgência') tp_atendimento
                         ,reg_amb.cd_multi_empresa
                         ,atendime.cd_atendimento
                         ,con_pla.cd_con_pla
                         ,con_pla.ds_con_pla
                         ,REPASSE.CD_REPASSE
                         ,REPASSE.DT_REPASSE
                         ,REPASSE.DS_REPASSE
    FROM   DBAMV.reg_amb,
           DBAMV.itreg_amb,
           DBAMV.prestador,
           DBAMV.prestador prestador_repasse,
           DBAMV.empresa_prestador,
           DBAMV.remessa_fatura,
           DBAMV.fatura,
           DBAMV.paciente,
           DBAMV.atendime,
           DBAMV.con_pla,
           DBAMV.convenio,
           DBAMV.ori_ate,
           DBAMV.REPASSE_CONTA,
           DBAMV.REPASSE,
           DBAMV.REPASSE_PRESTADOR
    WHERE  atendime.cd_multi_empresa     =  dbamv.Pkg_Mv2000.Le_Empresa
    AND    itreg_amb.cd_reg_amb          =  reg_amb.cd_reg_amb
    AND    remessa_fatura.cd_fatura      =  fatura.cd_fatura
    AND    con_pla.cd_con_pla            =  atendime.cd_con_pla
    AND    convenio.cd_convenio          =  atendime.cd_convenio
    AND    con_pla.cd_convenio           =  atendime.cd_convenio
    AND    paciente.cd_paciente          =  atendime.cd_paciente
    AND    remessa_fatura.cd_remessa     =  reg_amb.cd_remessa
    AND    ori_ate.cd_ori_ate            =  atendime.cd_ori_ate
    AND    atendime.cd_prestador         =  empresa_prestador.cd_prestador
    AND    prestador.cd_prestador        =  empresa_prestador.cd_prestador
    AND    atendime.cd_multi_empresa     =  empresa_prestador.cd_multi_empresa
    AND    itreg_amb.cd_atendimento      =  atendime.cd_atendimento
    AND    prestador_repasse.cd_prestador = empresa_prestador.cd_prestador_repasse
    AND    REPASSE_CONTA.CD_REG_FAT = REG_AMB.cd_REG_AMB
    AND    REPASSE_CONTA.CD_REPASSE = REPASSE.CD_REPASSE
    AND    REPASSE_PRESTADOR.CD_REPASSE = REPASSE.CD_REPASSE
    AND    REPASSE_PRESTADOR.cd_prestador = prestador_repasse.cd_prestador
    AND    REPASSE_PRESTADOR.cd_prestador_REPASSE = prestador_repasse.cd_prestador_REPASSE
    AND    TO_CHAR(REPASSE.DT_COMPETENCIA,'MM/YYYY') = :CAMPO_DT_COMPET
  UNION ALL
   SELECT REG_FAT.cd_remessa
                         ,atendime.dt_atendimento
                         ,REG_FAT.cd_REG_FAT cd_conta
                         ,REG_FAT.cd_convenio
                         ,prestador.cd_prestador
                         ,REPASSE_CONTA.vl_faturado
                         ,REPASSE_CONTA.vl_repassado
                         ,empresa_prestador.VL_PERC_REPASSE_CONTA
                         ,convenio.nm_convenio
                         ,atendime.cd_paciente
                         ,atendime.cd_ori_ate
                         ,ori_ate.ds_ori_ate
                         ,dbamv.pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(paciente.cd_paciente, 1) nm_paciente
                         ,prestador.nm_prestador
                         ,fatura.dt_competencia
                         ,empresa_prestador.cd_prestador_repasse
                         ,prestador_repasse.nm_prestador nm_prestador_repasse
                         ,decode(atendime.tp_atendimento,'I','Internação','H','Home-Care') tp_atendimento
                         ,REG_FAT.cd_multi_empresa
                         ,atendime.cd_atendimento
                         ,con_pla.cd_con_pla
                         ,con_pla.ds_con_pla
                         ,REPASSE.CD_REPASSE
                         ,REPASSE.DT_COMPETENCIA
                         ,REPASSE.DS_REPASSE
    FROM   DBAMV.REG_FAT,
           DBAMV.prestador,
           DBAMV.empresa_prestador,
           DBAMV.remessa_fatura,
           DBAMV.fatura,
           DBAMV.prestador prestador_repasse,
           DBAMV.paciente,
           DBAMV.atendime,
           DBAMV.con_pla,
           DBAMV.convenio,
           DBAMV.ori_ate,
           DBAMV.REPASSE_CONTA,
           DBAMV.REPASSE,
           DBAMV.REPASSE_PRESTADOR
    WHERE  atendime.cd_multi_empresa     =  dbamv.Pkg_Mv2000.Le_Empresa
    AND    remessa_fatura.cd_fatura      =  fatura.cd_fatura
    AND    con_pla.cd_con_pla            =  atendime.cd_con_pla
    AND    convenio.cd_convenio          =  atendime.cd_convenio
    AND    con_pla.cd_convenio           =  atendime.cd_convenio
    AND    paciente.cd_paciente          =  atendime.cd_paciente
    AND    remessa_fatura.cd_remessa     =  REG_FAT.cd_remessa
    AND    ori_ate.cd_ori_ate            =  atendime.cd_ori_ate
    AND    atendime.cd_prestador         =  empresa_prestador.cd_prestador
    AND    prestador.cd_prestador        =  empresa_prestador.cd_prestador
    AND    atendime.cd_multi_empresa     =  empresa_prestador.cd_multi_empresa
    AND    prestador_repasse.cd_prestador = empresa_prestador.cd_prestador_repasse
    AND    REG_FAT.cd_atendimento        =  atendime.cd_atendimento
    AND    REPASSE_CONTA.CD_REG_FAT = REG_FAT.cd_REG_FAT
    AND    REPASSE_CONTA.CD_REPASSE = REPASSE.CD_REPASSE
    AND    REPASSE_PRESTADOR.CD_REPASSE = REPASSE.CD_REPASSE
    AND    REPASSE_PRESTADOR.cd_prestador = prestador_repasse.cd_prestador
    AND    REPASSE_PRESTADOR.cd_prestador_REPASSE = prestador_repasse.cd_prestador_REPASSE
    AND    TO_CHAR(REPASSE.DT_COMPETENCIA,'MM/YYYY') = :CAMPO_DT_COMPET
:CAMPO_CF_COND_REPASSE
)
ORDER BY CD_PRESTADOR_REPASSE,NM_PRESTADOR_REPASSE,
CD_PRESTADOR,NM_PRESTADOR,
CD_REPASSE,DS_REPASSE,DT_REPASSE
