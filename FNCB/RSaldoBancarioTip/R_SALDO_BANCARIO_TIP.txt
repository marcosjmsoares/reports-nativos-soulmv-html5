--R_SALDO_BANCARIO_TIP
SELECT  'Empresa ' || e.cd_multi_empresa || ' - ' || e.ds_multi_empresa Empresa
      ,e.cd_multi_empresa  cd_multi_empresa
      ,e.cd_multi_empresa || ' - ' || e.ds_multi_empresa Empresa_o
       ,   CON_COR.CD_CON_COR                                               CD_CON_COR1
       ,CON_COR.DS_CON_COR                                                      DS_CON_COR
       ,CON_COR.CD_AGENCIA || ' - ' || CON_COR.DS_AGENCIA    DS_AGENCIA
       ,CON_COR.NR_CONTA                                                           NR_CONTA
       ,BANCO.CD_BANCO                                                                 CD_BANCO
       ,BANCO.NM_BANCO                                                                DS_BANCO
      ,CON_COR.VL_SALDO_ATUAL                                                VL_SALDO_ATUAL
       ,NVL(CON_COR.VL_CHEQUES_DEPOSITADOS_NAO_COM,0)   - NVL(CON_COR.VL_CHEQUES_EMITIDOS_NAO_COMPEN,0)	vl_diferenca
       ,CON_COR.CD_TIP_CONTA_CORRENTE                               CD_TIP_CONTA_CORRENTE
       ,TIP_CONTA_CORRENTE.DS_TIP_CONTA_CORRENTE      DS_TIP_CONTA_CORRENTE
       ,TIP_CONTA_CORRENTE.NR_ORDEM                                  NR_ORDEM
       ,CON_COR.VL_LIMITE_CHEQUE_ESPECIAL VL_LIMITE_CHEQUE_ESPECIAL
       ,CON_COR.CD_MOEDA
FROM    dbamv.CON_COR                                                              CON_COR
             ,dbamv.BANCO                                                                   BANCO
             ,dbamv.TIP_CONTA_CORRENTE                                     TIP_CONTA_CORRENTE
             ,dbamv.multi_empresas e
             ,dbamv.USUARIO_MULTI_EMPRESA UME
WHERE CON_COR.CD_BANCO = BANCO.CD_BANCO
AND  CON_COR.CD_TIP_CONTA_CORRENTE = TIP_CONTA_CORRENTE.CD_TIP_CONTA_CORRENTE
and   CON_COR.cd_multi_empresa = e.cd_multi_empresa
AND CON_COR.cd_multi_empresa = UME.cd_multi_empresa AND ume.cd_id_usuario = user
ORDER BY NR_ORDEM,  NM_BANCO
