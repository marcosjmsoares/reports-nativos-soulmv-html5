--R_ARQUIVOS
SELECT * FROM (
select cb.cd_convenio_bancario
          , cb.ds_convenio_bancario
          , cb.cd_multi_empresa
          , lote_rem.cd_itcon_rec
          , lote_rem.vl_total_desconto
          , lote_rem.vl_total_acrescimo
          , lote_rem.vl_tarifa_bancaria
          , lote_rem.vl_total_recebido
          , arq_ret.cd_arquivo_retorno_pgto
          , arq_ret.dt_arquivo_retorno_pgto
          , arq_ret.cd_con_cor
          , arq_ret.qt_registro
          , arq_ret.qt_baixado
          , nvl(arq_ret.qt_registro,0) - nvl(arq_ret.qt_baixado,0) qt_rejeitado
          , tar.cd_con_pag
          , con.nr_documento
          , it.nr_parcela
   from dbamv.convenio_bancario cb
        , dbamv.lote_rem_itcon_rec lote_rem
        , dbamv.arquivo_retorno_pgto arq_ret
        , dbamv.tarifa_bancaria_ret tar
        , dbamv.con_rec con
        , dbamv.itcon_rec it
   where cb.cd_convenio_bancario = arq_ret.cd_convenio_bancario
     and arq_ret.cd_arquivo_retorno_pgto = lote_rem.cd_arquivo_retorno_pgto
     and arq_ret.cd_arquivo_retorno_pgto = tar.cd_arquivo_retorno_pgto(+)
     and con.cd_con_rec = it.cd_con_rec
     and lote_rem.cd_itcon_rec =  it.cd_itcon_rec
   :CAMPO_CF_DATA
)
ORDER BY CD_MULTI_EMPRESA,
CD_CONVENIO_BANCARIO,DS_CONVENIO_BANCARIO,CD_ARQUIVO_RETORNO_PGTO,DT_ARQUIVO_RETORNO_PGTO,CD_CON_COR,QT_REGISTRO,QT_BAIXADO,QT_REJEITADO,CD_CON_PAG
