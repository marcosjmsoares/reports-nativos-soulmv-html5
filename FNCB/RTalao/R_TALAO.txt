--R_TALAO
SELECT * FROM (
SELECT  CON_COR.DS_CON_COR   DESCRICAO
           ,   CON_COR.CD_CON_COR  CODIGO
	       ,   CON_COR.NR_CONTA       NR_CONTA
           ,   CON_COR.CD_AGENCIA    CD_AGENCIA
           ,   CON_COR.DS_AGENCIA    DS_AGENCIA
           ,   BANCO.CD_BANCO            CD_BANCO
           ,   BANCO.NM_BANCO            NM_BANCO
 FROM   dbamv.CON_COR  CON_COR
         ,    dbamv.BANCO       BANCO
WHERE BANCO.CD_BANCO                          = CON_COR.CD_BANCO
AND       CON_COR.CD_MULTI_EMPRESA   = :CAMPO_CD_MULTI_EMPRESA
:CAMPO_COND_NR_CONTA
)
order by DESCRICAO, CODIGO
