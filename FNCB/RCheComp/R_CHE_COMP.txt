--R_CHE_COMP
SELECT * FROM (
SELECT  'Empresa ' || e.cd_multi_empresa || ' - ' || e.ds_multi_empresa Empresa
      ,e.cd_multi_empresa  cd_multi_empresa
      ,e.cd_multi_empresa || ' - ' || e.ds_multi_empresa Empresa_o
       ,  TRUNC(CHEQUE.DT_COMPENSACAO) DT_COMPESACAO
       ,CON_COR.CD_CON_COR CD_CON_COR
       ,CON_COR.DS_CON_COR DS_CON_COR
       ,CON_COR.NR_CONTA NR_CONTA
       ,CON_COR.CD_BANCO CD_BANCO
       , BANCO.NM_BANCO
       ,CON_COR.CD_AGENCIA CD_AGENCIA
       ,CON_COR.DS_AGENCIA DS_AGENCIA
       ,CHEQUE.NR_CHEQUE NR_CHEQUE
       ,CHEQUE.DS_NOME_IMPRESSO DS_NOME_IMPRESSO
       ,CHEQUE.VL_CHEQUE VL_CHEQUE
FROM    dbamv.CHEQUE CHEQUE
              ,dbamv.TALAO TALAO
              ,dbamv.CON_COR CON_COR
              ,dbamv.BANCO BANCO
             ,dbamv.multi_empresas e
WHERE TALAO.CD_TALAO = CHEQUE.CD_TALAO
AND TALAO.CD_CON_COR = CON_COR.CD_CON_COR
AND BANCO.CD_BANCO = CON_COR.CD_BANCO
AND (CHEQUE.SN_CANCELADO <> 'S' OR CHEQUE.SN_CANCELADO IS NULL)
AND TRUNC(CHEQUE.DT_COMPENSACAO) BETWEEN :CAMPO_DATA_INICIAL AND 
and   CON_COR.cd_multi_empresa = e.cd_multi_empresa
ORDER BY CHEQUE.NR_CHEQUE
                 ,  CHEQUE.DT_EMISSAO DESC
)
ORDER BY EMPRESA,CD_MULTI_EMPRESA,
EMPRESA_O,NR_CONTA,CD_CON_COR,DS_CON_COR,CD_BANCO,NM_BANCO,CD_AGENCIA,DS_AGENCIA,
DT_COMPESACAO
