--R_ENTCHEQUE
SELECT * FROM (
SELECT  CHEQUE.DT_EMISSAO                      DT_EMISSAO
               ,CHEQUE.DT_RETIRADA                   DT_RETIRADA
               ,CHEQUE.NR_CHEQUE                      NR_CHEQUE
               ,CHEQUE.DS_NOME_IMPRESSO      DS_NOME_IMPRESSO
               ,CHEQUE.VL_CHEQUE                       VL_CHEQUE
               ,TALAO.CD_CON_COR                       CD_CON_COR
               ,CON_COR.NR_CONTA                      NR_CONTA
FROM    dbamv.CHEQUE                                   CHEQUE
             ,dbamv.TALAO                                      TALAO
             ,dbamv.CON_COR                                CON_COR
WHERE TALAO.CD_TALAO       =   CHEQUE.CD_TALAO
AND       TALAO.CD_CON_COR =  CON_COR.CD_CON_COR
AND       CON_COR.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND       CHEQUE.DT_RETIRADA IS NOT NULL
AND       TRUNC(CHEQUE.DT_RETIRADA) BETWEEN :CAMPO_DATA_INICIAL ORDER BY CHEQUE.DT_RETIRADA DESC
        ,TALAO.CD_CON_COR
        ,CHEQUE.NR_CHEQUE
)
ORDER BY DT_RETIRADA
