select * from (SELECT CCOR.CD_CON_COR,
       CCOR.CD_CON_COR ||' - '|| CCOR.DS_CON_COR as DS_CON_COR,
       CCOR.CD_MULTI_EMPRESA CD_EMPRESA_CCOR,
       EM.DS_MULTI_EMPRESA,
       CHEQ.CD_CHEQUE,
       CHEQ.CD_TALAO,
       CHEQ.NR_CHEQUE,
       TLAO.NR_TALAO,
       CHEQ.DT_EMISSAO,
       CHEQ.DT_PREDATADA,
       CHEQ.VL_CHEQUE,
       CHEQ.NM_USUARIO, 
       CHEQ.DS_NOME_IMPRESSO DS_FORNECEDOR,
       CCOR.CD_MULTI_EMPRESA CD_EMPRESA_CNTA,
       BA.CD_BANCO ||' - '|| BA.NM_BANCO as BANCO,
       (SELECT TP_STATUS FROM (SELECT MAX(CD_CHEQUE_STATUS), CD_CHEQUE, TP_STATUS FROM DBAMV.CHEQUE_STATUS GROUP BY CD_CHEQUE, TP_STATUS ORDER BY 1 DESC) WHERE CD_CHEQUE = CHEQ.CD_CHEQUE AND ROWNUM = 1) as TP_STATUS       
  FROM DBAMV.CON_COR        CCOR,
       DBAMV.BANCO          BA,
       DBAMV.MULTI_EMPRESAS EM,
       DBAMV.TALAO          TLAO,
       DBAMV.CHEQUE         CHEQ
 WHERE CCOR.CD_CON_COR = TLAO.CD_CON_COR
   AND TLAO.CD_TALAO = CHEQ.CD_TALAO
   AND CCOR.CD_MULTI_EMPRESA = EM.CD_MULTI_EMPRESA
   AND CCOR.CD_BANCO = BA.CD_BANCO
   ) where
