--R_VERSO_CHEQUE
select  substr(cq.ds_verso_cheque,1,580) ds_verso_cheque
from    dbamv.cheque cq
       ,dbamv.talao tl
       ,dbamv.con_cor cc
where  cq.cd_talao     =  tl.cd_talao
and    tl.cd_con_cor   =  cc.cd_con_cor
and    nvl( cq.sn_cancelado, 'N' ) = 'N'
and    cc.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
and    cc.cd_con_cor          =  :CAMPO_CD_CON_COR
and    cq.nr_cheque            =  :CAMPO_NR_CHEQUE
and    cq.cd_talao               =  Nvl(:CAMPO_CD_TALAO, cq.cd_talao)
