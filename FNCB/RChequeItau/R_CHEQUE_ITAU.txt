--R_CHEQUE_ITAU
SELECT BANCO.CD_BANCO,
       BANCO.NM_BANCO,
       CHEQUE.NR_CHEQUE,
       CHEQUE.VL_CHEQUE,
       CHEQUE.DS_NOME_IMPRESSO,
       CHEQUE.DT_EMISSAO,
       CHEQUE.DT_PREDATADA,
       BANCO.NR_LIN_DT_PREDATADO,
       BANCO.NR_COL_DT_PREDATADO,
       CHEQUE.CD_CHEQUE,
       CON_COR.SN_UTILIZA_FORM_CONTINUO,
       CON_COR.NR_CONTA,
       CON_COR.CD_CON_COR,
       CHEQUE.DS_VERSO_CHEQUE,
       BANCO.CD_BANCO_IMP_CHEQUE,
       CON_COR.CD_MOEDA
  FROM
       dbamv.CON_COR,
       dbamv.BANCO,
       dbamv.TALAO,
       dbamv.CHEQUE
 WHERE CON_COR.CD_BANCO = BANCO.CD_BANCO
   AND TALAO.CD_CON_COR = CON_COR.CD_CON_COR
   AND CHEQUE.CD_TALAO = TALAO.CD_TALAO
   AND nvl(CHEQUE.SN_IMPRESSO, 'N') <> 'S'
   AND nvl(CHEQUE.SN_CANCELADO, 'N') <> 'S'
   AND CON_COR.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
   AND CON_COR.CD_CON_COR = decode(:CAMPO_CD_CON_COR, '%', CON_COR.CD_CON_COR, :CAMPO_CD_CON_COR)
   AND TALAO.CD_TALAO = NVL(:CAMPO_CD_TALAO, TALAO.CD_TALAO)
   AND CHEQUE.CD_CHEQUE BETWEEN :CAMPO_V_CHEQUE_ANT ORDER BY CHEQUE.NR_CHEQUE
