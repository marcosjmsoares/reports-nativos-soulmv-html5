--R_APLIC
SELECT * FROM (
SELECT  'Empresa ' || e.cd_multi_empresa || ' - ' || e.ds_multi_empresa Empresa
      ,e.cd_multi_empresa  cd_multi_empresa
      ,e.cd_multi_empresa || ' - ' || e.ds_multi_empresa Empresa_o
       , CON_COR.CD_CON_COR			CD_CON_COR
       ,CON_COR.DS_CON_COR			DS_CON_COR
       ,TIP_APLICA.CD_TIP_APLICA			CD_TIP_APLICA
       ,TIP_APLICA.DS_TIP_APLICA			DS_TIP_APLICA
       ,APLICACAO.CD_APLICACAO			CD_APLICACAO
       ,APLICACAO.DS_APLICACAO			DS_APLICACAO
       ,APLICACAO.VL_SALDO_ATUAL			VL_SALDO_ATUAL
       ,Nvl(APLICACAO.VL_PERCENTUAL_RENDIMENTO, 0)  	VL_PERCENTUAL
       ,APLICACAO.DT_ABERTURA			DT_ABERTURA
       ,APLICACAO.DT_FECHAMENTO			DT_FECHAMENTO
       ,APLICACAO.DT_PREVISTA_PARA_RETIRADA
       ,( APLICACAO.VL_SALDO_ATUAL + ( APLICACAO.VL_SALDO_ATUAL * NVL(APLICACAO.VL_PERCENTUAL_RENDIMENTO,0) / 100 )) 	PROJECAO
       ,decode  ( :CAMPO_CONVERTE, 1 , :CAMPO_CD_MOEDA ,  APLICACAO.CD_MOEDA  )	CD_MOEDA
       ,APLICACAO.VL_MOEDA			VL_MOEDA
       ,( APLICACAO.VL_MOEDA + ( APLICACAO.VL_MOEDA * NVL(APLICACAO.VL_PERCENTUAL_RENDIMENTO,0) / 100 )) 			PROJECAO_MOEDA
FROM    DBAMV.TIP_APLICA	TIP_APLICA
       ,DBAMV.CON_COR		CON_COR
       ,DBAMV.APLICACAO		APLICACAO
       ,dbamv.multi_empresas e
WHERE   CON_COR.CD_CON_COR          =  APLICACAO.CD_CON_COR
AND     APLICACAO.CD_TIP_APLICA         =  TIP_APLICA.CD_TIP_APLICA
AND     APLICACAO.DT_ABERTURA       <= TO_DATE( TO_CHAR( :CAMPO_DT_FIM  , 'DD/MM/YYYY')||' 23','DD/MM/YYYY HH24' )
AND     APLICACAO.DT_ABERTURA       >= TO_DATE( TO_CHAR( :CAMPO_DT_INIC , 'DD/MM/YYYY')||' 00','DD/MM/YYYY HH24' )
and   CON_COR.cd_multi_empresa = e.cd_multi_empresa
:CAMPO_CF_ORDEM
)
