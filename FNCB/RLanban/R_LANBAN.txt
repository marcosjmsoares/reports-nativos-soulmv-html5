--R_LANBAN
SELECT LAN_CONCOR.CD_LAN_CONCOR CODIGO
           ,   LAN_CONCOR.DS_LAN_CONCOR DESCRICAO
           ,   LAN_CONCOR.SN_LANC_EXTRATO SN_LANC_EXTRATO
           ,   LAN_CONCOR.SN_LANC_RECEBIMENTO SN_LANC_RECEBIMENTO
           ,   LAN_CONCOR.SN_TRANSFERENCIA SN_TRANSFERENCIA
           ,   LAN_CONCOR.TP_APARECE_NO_EXTRATO TP_APARECE_NO_EXTRATO
           ,   LAN_CONCOR.TP_CONTA_RENDIMENTO TP_CONTA_RENDIMENTO
           ,   LAN_CONCOR.TP_OPERACAO_SALDO_APLICACAO TP_OPERACAO_SALDO_APLICACAO
           ,   LAN_CONCOR.TP_OPERACAO_SALDO_CONTA TP_OPERACAO_SALDO_CONTA
           ,   LAN_CONCOR.SN_PERMITE_LCTO_CONTABIL    SN_PERMITE_LCTO_CONTABIL
           ,   LAN_CONCOR.CD_LAN_CONCOR_TRANSFERENCIA   CD_LAN_CONCOR_TRANSFERENCIA
           ,   EMPRESA_LAN_CONCOR.CD_REDUZIDO   CD_REDUZIDO
           ,   LAN_CONCOR.SN_ATIVO           SN_ATIVO
           ,   LAN_CONCOR.SN_TARIFA         SN_TARIFA
           ,   PLANO_CONTAS.DS_CONTA    DS_CONTA
           ,   SETOR.NM_SETOR     NM_SETOR
  FROM   dbamv.LAN_CONCOR LAN_CONCOR
               ,dbamv.EMPRESA_LAN_CONCOR EMPRESA_LAN_CONCOR
               ,dbamv.PLANO_CONTAS  PLANO_CONTAS
               ,dbamv.SETOR  SETOR
WHERE  EMPRESA_LAN_CONCOR.CD_LAN_CONCOR =  LAN_CONCOR.CD_LAN_CONCOR
      AND  EMPRESA_LAN_CONCOR.CD_REDUZIDO = PLANO_CONTAS.CD_REDUZIDO(+)
      AND  EMPRESA_LAN_CONCOR.CD_SETOR = SETOR.CD_SETOR(+)
      AND  EMPRESA_LAN_CONCOR.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
ORDER BY
