--R_ARQUIVO_RETORNO
select a.cd_arquivo_retorno_pgto,
       a.ds_maquina,
       a.ds_arquivo,
       a.cd_usuario,
       a.dt_arquivo_retorno_pgto,
       a.cd_layout_arquivo,
       a.qt_registro,
       a.qt_baixado,
       a.cd_con_cor,
       o.CD_OCORRENCIA_ARQ_RET_PGTO,
       o.DS_CONTEUDO_CAMPO_CHAVE,
       o.CD_OCORRENCIA_LAY_ARQ,
       l.tp_ocorrencia,
       l.tp_acao,
       l.DS_OCORRENCIA_LAY_ARQ,
       la.ds_layout_arquivo,
       c.ds_con_cor
FROM DBAMV.ARQUIVO_RETORNO_PGTO a,
     DBAMV.OCORRENCIA_LAY_ARQ l,
     DBAMV.OCORRENCIA_ARQ_RET_PGTO o,
     DBAMV.LAYOUT_ARQUIVO la,
     DBAMV.CON_COR c
WHERE a.cd_arquivo_retorno_pgto = o.cd_arquivo_retorno_pgto
     AND a.cd_layout_arquivo = la.cd_layout_arquivo
     AND a.cd_con_cor = c.cd_con_cor
     AND a.CD_MULTI_EMPRESA = dbamv.PKG_MV2000.LE_EMPRESA
     AND o.cd_ocorrencia_lay_arq = l.cd_ocorrencia_lay_arq
     AND a.cd_arquivo_retorno_pgto = :CAMPO_CD_ARQUIVO
ORDER BY l.tp_acao
