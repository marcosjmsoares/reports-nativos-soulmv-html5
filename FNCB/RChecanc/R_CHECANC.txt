--R_CHECANC
SELECT * FROM (
SELECT  'Empresa ' || e.cd_multi_empresa || ' - ' || e.ds_multi_empresa Empresa
      ,e.cd_multi_empresa  cd_multi_empresa
      ,e.cd_multi_empresa || ' - ' || e.ds_multi_empresa Empresa_o
      ,CON_COR.NR_CONTA
     ,CON_COR.CD_CON_COR CD_CONTA
     ,CON_COR.DS_CON_COR DS_CONTA
     ,CON_COR.CD_BANCO CD_BANCO
     ,BANCO.NM_BANCO
     ,CON_COR.CD_AGENCIA CD_AGENCIA
     ,CON_COR.DS_AGENCIA DS_AGENCIA
     ,CHEQUE.NR_CHEQUE
     ,CHEQUE.CD_TALAO
     ,TRUNC(CHEQUE.DT_EMISSAO) DT_EMISSAO
     ,TRUNC(NVL(CHEQUE.DT_CANCELAMENTO,CHEQUE.DT_EMISSAO)) DT_CANCE
     ,MOTIVO_CHE_CANC.DS_MOTIVO_CANC DS_CANCE
     ,CHEQUE.VL_CHEQUE
     ,MOTIVO_CHE_CANC.SN_INUTILIZA_CHEQUE
FROM
     DBAMV.CHEQUE
     ,DBAMV.CON_COR
     ,DBAMV.MOTIVO_CHE_CANC
     ,DBAMV.TALAO
     ,DBAMV.BANCO BANCO
     ,dbamv.multi_empresas e
WHERE 	CHEQUE.SN_CANCELADO 		= 'S'
AND 	CHEQUE.CD_MOTIVO_CANC 	= MOTIVO_CHE_CANC.CD_MOTIVO_CANC(+)
AND 	TO_CHAR(NVL(CHEQUE.DT_CANCELAMENTO,CHEQUE.DT_EMISSAO), 'YYYY/MM/DD') BETWEEN TO_CHAR(:CAMPO_DATA_INICIAL, 'YYYY/MM/DD')  AND TO_CHAR(:CAMPO_DATA_FINAL, 'YYYY/MM/DD')
AND 	CON_COR.CD_CON_COR 		= TALAO.CD_CON_COR
AND 	CHEQUE.CD_TALAO 		= TALAO.CD_TALAO
AND 	BANCO.CD_BANCO 		= CON_COR.CD_BANCO
and   CON_COR.cd_multi_empresa = e.cd_multi_empresa
:CAMPO_CF_EMPRESA
)
ORDER BY CD_MULTI_EMPRESA,EMPRESA,
EMPRESA_O,CD_CONTA,DS_CONTA,NR_CONTA,CD_BANCO,NM_BANCO,CD_AGENCIA,DS_AGENCIA
