--R_BOLETO_ARRECADACAO
SELECT * FROM (
Select cr.cd_multi_empresa,
       cr.cd_con_rec,
       cr.nr_documento,
       cr.nm_cliente,
       ir.nr_parcela,
       ir.cd_itcon_rec,
       cb.cd_convenio_bancario,
       cb.ds_convenio_bancario,
       lrir.dt_emissao_boleto,
       lrir.dt_vencimento_boleto,
       lrir.vl_boleto,
       lrir.nr_nosso_numero_boleto,
       lr.cd_lote_remessa,
       lr.ds_lote_remessa,
       a.cd_usuario,
       a.cd_arquivo_retorno_pgto,
       a.dt_arquivo_retorno_pgto
  From dbamv.con_rec cr,
       dbamv.itcon_rec ir,
       dbamv.convenio_bancario cb,
       dbamv.lote_rec_cobranca lrc,
       dbamv.lote_rec_itcon_rec lrir,
       dbamv.lote_remessa lr,
       dbamv.lote_rem_itcon_rec lri,
       dbamv.arquivo_retorno_pgto a
 Where cr.cd_con_rec = ir.cd_con_rec
   And cb.cd_convenio_bancario  = lrc.cd_convenio_bancario
   And lrc.cd_lote_rec_cobranca = lrir.cd_lote_rec_cobranca
   And lrir.cd_itcon_rec       = ir.cd_itcon_rec
   And cb.cd_convenio_bancario = lr.cd_convenio_bancario
   And lr.cd_lote_remessa      = lri.cd_lote_remessa
   And a.cd_convenio_bancario  = cb.cd_convenio_bancario
   And a.cd_arquivo_retorno_pgto = lri.cd_arquivo_retorno_pgto
   And tp_convenio_bancario    = 'A'
   And lrir.sn_boleto_impresso = 'S'
   :CAMPO_CF_DATA
)
ORDER BY CD_MULTI_EMPRESA,
CD_ARQUIVO_RETORNO_PGTO,CD_LOTE_REMESSA,DS_LOTE_REMESSA
