--R_LANCA_759
SELECT * FROM (
select c.cd_con_cor,
       c.ds_con_cor,
       m.cd_lan_concor,
       l.ds_lan_concor,
       sum(m.vl_movimentacao) vl_movimentacao
  from dbamv.mov_concor m
      ,dbamv.lan_concor l
      ,dbamv.con_cor c
  where m.cd_lan_concor    = l.cd_lan_concor
    and c.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
    and c.cd_con_cor       = m.cd_con_cor
    and  m.dt_movimentacao >= :CAMPO_DATA_INIC
    and  m.dt_movimentacao  <= :CAMPO_DATA_FIM
group by c.cd_con_cor,
         c.ds_con_cor,
         m.cd_lan_concor,
         l.ds_lan_concor
order by c.ds_con_cor,
         l.ds_lan_concor
)
ORDER BY DS_CON_COR,CD_CON_COR
