--R_CON_PAG_BANCARIO_CONTA_R_10_Q_PREVISTA_PAG
SELECT * FROM (
select 'Empresa ' || e.cd_multi_empresa || ' - ' || e.ds_multi_empresa Empresa_prev
      ,e.cd_multi_empresa  cd_multi_empresa
      ,e.cd_multi_empresa || ' - ' || e.ds_multi_empresa Empresa_oprev
       , pagcon_pag.cd_pagcon_pag                                                      pagcon_pag
       ,pagcon_pag.dt_pagamento                                                           dt_pagamento_prev
       ,itcon_pag.dt_prevista_pag                                                               dt_prevista_pag_prev
       ,pagcon_pag.dt_baixa                                                                       dt_baixa_prev
       ,con_pag.dt_emissao                                                                         dt_emissao_prev
       ,itcon_pag.vl_duplicata                                                                      vl_duplicata_prev
       ,pagcon_pag.vl_pago                                                                        vl_pago_prev
       ,con_pag.nr_documento || '/' || itcon_pag.nr_parcela                         nr_documento_prev
      ,con_cor.ds_con_cor || ' - ' || con_cor.cd_banco || '/' || con_cor.cd_agencia || '/' || con_cor.nr_conta  nm_fantasia
       ,decode(pagcon_pag.cd_cheque,'',pagcon_pag.cd_bordero,cheque.nr_cheque)  che_bord_prev
       ,pagcon_pag.tp_pagamento                                                              tp_pagamento_prev
       ,con_cor.cd_banco                                                                           cd_banco_prev
       ,banco.nm_banco                                                                              nm_banco_prev
       ,cheque.dt_compensacao        				dt_compensacao_prev
from dbamv.con_pag
       ,dbamv.itcon_pag
       ,dbamv.pagcon_pag
       ,dbamv.fornecedor
       ,dbamv.con_cor
       ,dbamv.banco
       ,dbamv.cheque
       ,dbamv.bordero
       ,dbamv.tip_doc
       ,dbamv.multi_empresas e
where  con_pag.cd_con_pag                        = itcon_pag.cd_con_pag
and cheque.cd_cheque(+)                             = pagcon_pag.cd_cheque
and  bordero.cd_bordero(+)                            = pagcon_pag.cd_bordero
and pagcon_pag.cd_itcon_pag                     = itcon_pag.cd_itcon_pag
and pagcon_pag.cd_con_cor                       = con_cor.cd_con_cor
and itcon_pag.cd_con_pag_agrup              is null
and con_pag.cd_tip_doc                               =  tip_doc.cd_tip_doc
and con_pag.cd_fornecedor                          = fornecedor.cd_fornecedor(+)
and   CON_COR.cd_multi_empresa = e.cd_multi_empresa
union
select 'Empresa ' || e.cd_multi_empresa || ' - ' || e.ds_multi_empresa Empresa_prev
      ,e.cd_multi_empresa  cd_multi_empresa
      ,e.cd_multi_empresa || ' - ' || e.ds_multi_empresa Empresa_oprev
       ,0                                                      pagcon_pag
       ,mov_concor.dt_movimentacao               dt_pagamento_prev
     , mov_concor.dt_movimentacao                  dt_vencimento_prev
     , mov_concor.dt_movimentacao                  dt_baixa_prev
     , mov_concor.dt_movimentacao                  dt_emissao_prev
     , mov_concor.vl_movimentacao                  vl_duplicata_prev
     , mov_concor.vl_movimentacao                  vl_pago_prev
     , mov_concor.nr_documento_identificacao  nr_documento_prev
     , 'LANÇAMENTOS BANCÁRIOS'                      nm_fantasia
     , 0                                                                 che_bord_prev
     , 'LANCAMENTO'                                         tp_pagamento_prev
     , 0                                                                 cd_banco_prev
     ,mov_concor.ds_movimentacao                  nm_banco_prev
     ,mov_concor.dt_movimentacao                   dt_compensacao_prev
  from dbamv.mov_concor
     , dbamv.lan_concor
       ,dbamv.multi_empresas e
 where lan_concor.cd_lan_concor = mov_concor.cd_lan_concor
   and lan_concor.tp_operacao_saldo_conta = '-'
   and lan_concor.tp_operacao_saldo_aplicacao = '0'
   and mov_concor.cd_bordero is null
   and mov_concor.cd_cheque is null
   and mov_concor.cd_pagcon_pag is null
   and mov_concor.cd_con_cor_transferencia is null
   and mov_concor.cd_mov_concor_transferencia is null
   and   lan_concor.cd_multi_empresa = e.cd_multi_empresa
:CAMPO_CF_ORDER
)
ORDER BY EMPRESA_PREV,CD_MULTI_EMPRESA,
EMPRESA_OPREV,PAGCON_PAG,NM_FANTASIA,
DT_PREVISTA_PAG_PREV,
NR_DOCUMENTO_PREV,VL_DUPLICATA_PREV,DT_EMISSAO_PREV
