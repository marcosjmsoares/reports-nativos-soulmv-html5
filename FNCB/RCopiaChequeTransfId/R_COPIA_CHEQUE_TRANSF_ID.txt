--R_COPIA_CHEQUE_TRANSF_ID
Select   banco.cd_banco                          cd_banco
       ,banco.nm_banco                          nm_banco
       ,con_cor.cd_agencia                      cd_agencia
       ,con_cor.ds_agencia                      ds_agencia
       ,con_cor.nr_conta                        nr_conta
       ,cheque.nr_cheque                        nr_cheque
       ,to_char(cheque.dt_emissao,'DD "de" Month "de"  YYYY')          dt_emis_chq
       ,cheque.dt_emissao        dt_emis_doc
       ,cheque.vl_cheque                        vl_cheque
       ,cidade.nm_cidade                        nm_cidade
       ,cheque.nm_retirante                    nm_retirante
       ,mov_concor.nr_documento_identificacao   nr_doc
       ,mov_concor.ds_movimentacao              ds_movimentacao
       ,mov_concor.cd_con_cor_transferencia     cd_con_cor_trans
From    dbamv.cheque               cheque
       ,dbamv.talao                talao
       ,dbamv.con_cor              con_cor
       ,dbamv.banco                banco
       ,dbamv.cidade               cidade
       ,dbamv.multi_empresas       multi_emp
       ,dbamv.mov_concor           mov_concor
Where cheque.cd_talao              = talao.cd_talao
and   talao.cd_con_cor             = con_cor.cd_con_cor
and   con_cor.cd_banco             = banco.cd_banco
and   con_cor.cd_multi_empresa     = multi_emp.cd_multi_empresa
and   con_cor.cd_multi_empresa     = :CAMPO_CD_MULTI_EMPRESA
and   cidade.cd_cidade             = multi_emp.cd_cidade
and   cheque.cd_cheque             = :CAMPO_CD_CHEQUE
and   cheque.cd_cheque             = mov_concor.cd_cheque
and   con_cor.cd_con_cor           = :CAMPO_CD_CON_COR
