--R_COMPROVANTE_PBSA
SELECT * FROM (
SELECT  solic.cd_solic_sangue
               ,solic.cd_atendimento
               ,paciente.cd_paciente
               ,paciente.nm_paciente
               ,dbamv.FN_IDADE(paciente.dt_nascimento,'a A',SYSDATE) nr_idade
               ,DECODE(paciente.tp_sexo, 'F', 'Feminino', 'Masculino')  tp_sexo
               ,prestador.nm_prestador
               ,setor.nm_setor
               ,convenio.nm_convenio
               ,con_pla.ds_con_pla
               ,solic.ds_observacao
               ,sangue_derivados.cd_sangue_derivados
               ,sangue_derivados.ds_sangue_derivados
               ,tip_fre.ds_tip_fre
               ,itsolic.qt_prescricao
               ,cirurgia.cd_cirurgia
               ,cirurgia.ds_cirurgia
               ,itsolic.qt_solicitada
               ,ITSOLIC.NR_DIFEP
               , solic.tp_solicitacao
               , (SELECT Nvl(uni_presc.ds_unidade_presc, itunidade.ds_itunidade)
                    FROM dbamv.uni_presc, dbamv.itunidade, dbamv.itpre_med
                  WHERE uni_presc.cd_itunidade = itunidade.cd_itunidade
                    AND uni_presc.cd_uni_presc = itpre_med.cd_uni_presc
                    AND itpre_med.cd_itpre_med = itsolic.cd_itpre_med) ds_unidade_presc
FROM    dbamv.solic_sangue solic
             ,dbamv.atendime
             ,dbamv.paciente
             ,dbamv.prestador
             ,dbamv.setor
             ,dbamv.convenio
             ,dbamv.con_pla
            ,dbamv.it_solic_sangue itsolic
            ,dbamv.sangue_derivados
            ,dbamv.tip_fre
            ,dbamv.cirurgia
,Dbamv.Empresa_Con_Pla
Where Empresa_Con_Pla.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And Empresa_Con_Pla.Cd_Convenio = Con_Pla.Cd_Convenio
And Empresa_Con_Pla.Cd_Con_Pla = Con_Pla.Cd_Con_Pla
And solic.cd_atendimento             =  atendime.cd_atendimento
AND         atendime.cd_paciente           =  paciente.cd_paciente
AND          atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
AND         atendime.cd_prestador          =  prestador.cd_prestador
AND         solic.cd_setor                        =  setor.cd_setor(+)
AND         atendime.cd_convenio          =  convenio.cd_convenio
AND         atendime.cd_con_pla            =  con_pla.cd_con_pla(+)
AND         atendime.cd_convenio          =  con_pla.cd_convenio(+)
AND         solic.cd_solic_sangue           =  itsolic.cd_solic_sangue(+)
AND         itsolic.cd_sangue_derivados =  sangue_derivados.cd_sangue_derivados
AND         itsolic.cd_tip_fre                    =  tip_fre.cd_tip_fre(+)
AND         itsolic.cd_cirurgia                  =  cirurgia.cd_cirurgia(+)
AND         solic.cd_solic_sangue           =  :CAMPO_CD_SOLIC_SANGUE
ORDER BY cd_sangue_derivados
)
ORDER BY CD_SOLIC_SANGUE,CD_PACIENTE,NM_PACIENTE,NM_PRESTADOR,NM_SETOR,NM_CONVENIO,DS_CON_PLA,DS_OBSERVACAO,CD_ATENDIMENTO,NR_IDADE,TP_SEXO
