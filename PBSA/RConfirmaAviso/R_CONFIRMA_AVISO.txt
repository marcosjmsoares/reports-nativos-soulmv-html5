--R_CONFIRMA_AVISO
SELECT * FROM (
SELECT AVISO.CD_AVISO_CIRURGIA
              ,AVISO.DT_REALIZACAO
              ,AVISO.VL_TEMPO_DURACAO
              ,AVISO.CD_ATENDIMENTO
              ,PACIENTE.CD_PACIENTE
              ,DECODE(PACIENTE.NM_PACIENTE, NULL, AVISO.NM_PACIENTE, PACIENTE.NM_PACIENTE)   NM_PACIENTE
              ,PACIENTE.DT_NASCIMENTO
              ,DECODE(PACIENTE.TP_SEXO, 'M', 'MASCULINO', 'FEMININO')   DS_SEXO
              ,PACK_INTERNAMENTO.RETORNA_IDADE(PACIENTE.DT_NASCIMENTO, TRUNC(SYSDATE))  DS_IDADE
              ,CID.CD_CID                                             CD_CID
              ,CID.DS_CID                                             DS_CID
              ,SAL_CIR.CD_SAL_CIR                            CD_SAL_CIR
              ,SAL_CIR.DS_SAL_CIR                            DS_SAL_CIR
              ,CEN_CIR.CD_CEN_CIR                           CD_CEN_CIR
              ,CEN_CIR.DS_CEN_CIR                           DS_CEN_CIR
              ,UNID_INT.CD_UNID_INT                        CD_UNID_INT
              ,UNID_INT.DS_UNID_INT                        DS_UNID_INT
              ,LEITO.DS_RESUMO                                DS_RESUMO
              ,CIRURGIA.CD_CIRURGIA                       CD_CIRURGIA
              ,CIRURGIA.DS_CIRURGIA                       DS_CIRURGIA
              ,DERIV.CD_SANGUE_DERIVADOS         CD_PRO_FAT
              ,DERIV.DS_SANGUE_DERIVADOS         DS_PRO_FAT
              ,IT_SOLIC_SANGUE.QT_SOLICITADA    QT_SOLICITADO
              ,IT_SOLIC_SANGUE.QT_REALIZADO     QT_REALIZADO
FROM  DBAMV.AVISO_CIRURGIA   AVISO
           ,DBAMV.SAL_CIR
           ,DBAMV.CEN_CIR
           ,DBAMV.ATENDIME
           ,DBAMV.PACIENTE
           ,DBAMV.CID
           ,DBAMV.LEITO
           ,DBAMV.UNID_INT
           ,DBAMV.SOLIC_SANGUE
           ,DBAMV.IT_SOLIC_SANGUE
           ,DBAMV.SANGUE_DERIVADOS  DERIV
           ,DBAMV.CIRURGIA
WHERE AVISO.CD_AVISO_CIRURGIA =  :CAMPO_PCD_AVISO
AND       AVISO.CD_SAL_CIR = SAL_CIR.CD_SAL_CIR(+)
AND       ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND       AVISO.CD_CEN_CIR = CEN_CIR.CD_CEN_CIR(+)
AND       AVISO.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO(+)
AND       ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND       ATENDIME.CD_CID  =  CID.CD_CID(+)
AND       ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
AND       LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)
AND       AVISO.CD_AVISO_CIRURGIA  =  SOLIC_SANGUE.CD_AVISO_CIRURGIA
AND       SOLIC_SANGUE.CD_SOLIC_SANGUE  =  IT_SOLIC_SANGUE.CD_SOLIC_SANGUE
AND       IT_SOLIC_SANGUE.CD_SANGUE_DERIVADOS = DERIV.CD_SANGUE_DERIVADOS
AND       IT_SOLIC_SANGUE.CD_CIRURGIA = CIRURGIA.CD_CIRURGIA
)
ORDER BY CD_AVISO_CIRURGIA,CD_ATENDIMENTO,DT_REALIZACAO,VL_TEMPO_DURACAO,CD_SAL_CIR,DS_SAL_CIR,CD_UNID_INT,DS_UNID_INT,CD_CEN_CIR,DS_CEN_CIR,CD_CID,DS_CID,DS_RESUMO,CD_PACIENTE,NM_PACIENTE,DT_NASCIMENTO,DS_IDADE,DS_SEXO,
CD_CIRURGIA,DS_CIRURGIA
