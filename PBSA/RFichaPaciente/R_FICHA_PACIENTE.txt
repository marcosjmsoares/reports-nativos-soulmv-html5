--R_FICHA_PACIENTE
SELECT ATENDIME.CD_ATENDIMENTO
             ,ATENDIME.DT_ATENDIMENTO
             ,ATENDIME.HR_ATENDIMENTO
             ,ATENDIME.DT_ALTA
             ,ATENDIME.HR_ALTA
             ,ATENDIME.CD_CONVENIO
             ,CONVENIO.NM_CONVENIO
             ,SERVICO.CD_SERVICO
             ,SERVICO.DS_SERVICO
             ,PRESTADOR.DS_CODIGO_CONSELHO
             ,PRESTADOR.NM_PRESTADOR
             ,LEITO.DS_LEITO
             ,PACIENTE.CD_PACIENTE
             ,PACIENTE.NM_PACIENTE
             ,(PACIENTE.DS_ENDERECO || ' ' || PACIENTE.NR_ENDERECO || PACIENTE.DS_COMPLEMENTO)   ENDERECO
             ,PACIENTE.NM_BAIRRO
             ,CIDADE.NM_CIDADE
             ,CIDADE.CD_UF
             ,PACIENTE.NR_FONE
             ,PACIENTE.NR_CEP
             ,( PACIENTE.NR_IDENTIDADE || ' ' || PACIENTE.DS_OM_IDENTIDADE )   NR_IDENTIDADE
             ,PACIENTE.NR_CPF
             ,ATENDIME.CD_PRO_INT
             ,PRO_FAT.DS_PRO_FAT
             ,CARTEIRA.NM_TITULAR
             ,CARTEIRA.NR_CARTEIRA
             ,CARTEIRA.DT_VALIDADE
             ,CID.CD_CID
             ,CID.DS_CID
             ,CON_PLA.CD_CON_PLA
             ,CON_PLA.DS_CON_PLA
FROM  DBAMV.ATENDIME
           ,DBAMV.PACIENTE
           ,DBAMV.CONVENIO
           ,DBAMV.PRESTADOR
           ,DBAMV.SERVICO
           ,DBAMV.LEITO
           ,DBAMV.CIDADE
           ,DBAMV.PRO_FAT
           ,DBAMV.CARTEIRA
           ,DBAMV.CID
           ,DBAMV.CON_PLA
,Dbamv.Empresa_Con_Pla
Where Empresa_Con_Pla.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And Empresa_Con_Pla.Cd_Convenio = Con_Pla.Cd_Convenio
And Empresa_Con_Pla.Cd_Con_Pla = Con_Pla.Cd_Con_Pla
And ATENDIME.TP_ATENDIMENTO = 'I'
AND        ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND        ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND        ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
AND        ATENDIME.CD_SERVICO = SERVICO.CD_SERVICO(+)
AND        ATENDIME.CD_PRO_INT = PRO_FAT.CD_PRO_FAT(+)
AND        ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
AND        PACIENTE.CD_CIDADE = CIDADE.CD_CIDADE(+)
AND        ATENDIME.CD_PACIENTE = CARTEIRA.CD_PACIENTE(+)
AND        ATENDIME.CD_CONVENIO = CARTEIRA.CD_CONVENIO(+)
AND        ATENDIME.CD_CON_PLA = CARTEIRA.CD_CON_PLA(+)
AND        ATENDIME.NR_CARTEIRA = CARTEIRA.NR_CARTEIRA(+)
AND        ATENDIME.CD_CID = CID.CD_CID(+)
AND        ATENDIME.CD_CONVENIO = CON_PLA.CD_CONVENIO
AND        ATENDIME.CD_CON_PLA  = CON_PLA.CD_CON_PLA
AND        ATENDIME.CD_MULTI_EMPRESA = NVL(DBAMV.PKG_MV2000.LE_EMPRESA, ATENDIME.CD_MULTI_EMPRESA)
