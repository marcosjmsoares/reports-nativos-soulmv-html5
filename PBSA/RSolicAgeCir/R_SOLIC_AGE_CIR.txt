--R_SOLIC_AGE_CIR
SELECT * FROM (
SELECT AGE_CIR.DT_INICIO_AGE_CIR      DT_AGENDA
      ,CEN_CIR.CD_CEN_CIR                   CD_CEN_CIR
      ,CEN_CIR.DS_CEN_CIR                   DS_CEN_CIR
      ,AVISO.CD_AVISO_CIRURGIA         CD_AVISO_CIRURGIA
      ,AVISO.CD_PACIENTE                     CD_PACIENTE
      ,PACIENTE.DT_NASCIMENTO        DT_NASCIMENTO
      ,DECODE(AVISO.CD_PACIENTE, NULL, AVISO.NM_PACIENTE, PACIENTE.NM_PACIENTE)   NM_PACIENTE
      ,AVISO.VL_IDADE                                                                           VL_IDADE
      ,DECODE(AVISO.TP_SEXO, 'M', 'MASCULINO', 'FEMININO')       DS_SEXO
      ,AVISO.NR_TELEFONE_CONTATO                                              NR_TELEFONE_CONTATO
      ,AVISO.DT_PREV_INTER                                                              DT_PREV_INTER
      ,SAL_CIR.CD_SAL_CIR                                                                  CD_SAL_CIR
      ,SAL_CIR.DS_SAL_CIR                                                                  DS_SAL_CIR
      ,SANGUE_DERIVADOS.CD_SANGUE_DERIVADOS                   CD_PRO_FAT
      ,SANGUE_DERIVADOS.DS_SANGUE_DERIVADOS                   DS_PRO_FAT
      ,SANGUE_AVISO.QT_SOLICITADO                                              QT_SOLICITADA
      ,CIRURGIA.CD_CIRURGIA                                                            CD_CIRURGIA
      ,DECODE(CIRURGIA_AVISO.DS_NPADRONIZADO, NULL, CIRURGIA.DS_CIRURGIA, 'N.P.' || CIRURGIA_AVISO.DS_NPADRONIZADO)   DS_CIRURGIA
FROM   DBAMV.AGE_CIR
      ,DBAMV.SAL_CIR
      ,DBAMV.CEN_CIR
      ,DBAMV.AVISO_CIRURGIA  AVISO
      ,DBAMV.PACIENTE
      ,DBAMV.CIRURGIA_AVISO
      ,DBAMV.CIRURGIA
      ,DBAMV.SANGUE_AVISO
      ,DBAMV.SANGUE_DERIVADOS
WHERE  AGE_CIR.CD_AGE_CIR = :CAMPO_PCD_AGE_CIR
  AND  AGE_CIR.CD_AVISO_CIRURGIA  =  AVISO.CD_AVISO_CIRURGIA
  AND  AVISO.CD_SAL_CIR  =  SAL_CIR.CD_SAL_CIR(+)
  AND  AVISO.CD_CEN_CIR  =  CEN_CIR.CD_CEN_CIR(+)
  AND  AVISO.CD_PACIENTE  =  PACIENTE.CD_PACIENTE(+)
  AND  AVISO.CD_MULTI_EMPRESA = NVL(DBAMV.PKG_MV2000.LE_EMPRESA, AVISO.CD_MULTI_EMPRESA)
  AND  AVISO.CD_AVISO_CIRURGIA  =  CIRURGIA_AVISO.CD_AVISO_CIRURGIA
  AND  CIRURGIA_AVISO.CD_CIRURGIA  =  CIRURGIA.CD_CIRURGIA
  AND  CIRURGIA_AVISO.CD_AVISO_CIRURGIA  =  SANGUE_AVISO.CD_AVISO_CIRURGIA(+)
  AND  CIRURGIA_AVISO.CD_CIRURGIA  =  SANGUE_AVISO.CD_CIRURGIA(+)
  AND  SANGUE_AVISO.CD_SANGUE_DERIVADOS  =  SANGUE_DERIVADOS.CD_SANGUE_DERIVADOS(+)
)
ORDER BY CD_AVISO_CIRURGIA,CD_PACIENTE,NM_PACIENTE,DT_NASCIMENTO,VL_IDADE,NR_TELEFONE_CONTATO,DT_AGENDA,DT_PREV_INTER,CD_CEN_CIR,DS_CEN_CIR,CD_SAL_CIR,DS_SAL_CIR,DS_SEXO,
CD_CIRURGIA,DS_CIRURGIA
