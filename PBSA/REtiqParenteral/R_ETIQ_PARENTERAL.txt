--R_ETIQ_PARENTERAL
SELECT *
FROM (	SELECT 	PM.CD_PRE_MED
			,TO_CHAR(DBAMV.FNC_MV_RECUPERA_DATA_HORA(PM.DT_PRE_MED,PM.HR_PRE_MED),'DD/MM/YYYY HH24') as DH_PRE_MED
			,P.NM_PACIENTE
			,L.DS_RESUMO
			,UI.DS_UNID_INT
			,V.CODIGO_BARRAS
			,V.DS_TIP_PRESC
			,TO_CHAR(P.Dt_Nascimento, 'DD/MM/YYYY') Dt_Nascimento
		FROM 	DBAMV.PRE_MED PM
				,DBAMV.ATENDIME A
				,DBAMV.LEITO L
				,DBAMV.UNID_INT UI
				,DBAMV.PACIENTE P
				,(	SELECT 'T'||X.CD_ITPRE_MED CODIGO_BARRAS, TIP_PRESC.DS_TIP_PRESC
					FROM 	DBAMV.ITPRE_MED X, DBAMV.TIP_PRESC
					WHERE 	x.CD_ITPRE_MED = :CAMPO_CD_ITPRE_MED
					AND 	X.CD_TIP_PRESC = TIP_PRESC.CD_TIP_PRESC) V
		WHERE A.CD_PACIENTE     = P.CD_PACIENTE
		AND PM.CD_ATENDIMENTO = A.CD_ATENDIMENTO
		AND L.CD_LEITO        = A.CD_LEITO
		AND UI.CD_UNID_INT    = L.CD_UNID_INT
		AND PM.CD_PRE_MED    IN (	SELECT IPM.CD_PRE_MED
									FROM DBAMV.ITPRE_MED IPM
									WHERE IPM.CD_ITPRE_MED = :CAMPO_CD_ITPRE_MED )
		UNION ALL
		SELECT 	PM.CD_PRE_MED
				,TO_CHAR(DBAMV.FNC_MV_RECUPERA_DATA_HORA(PM.DT_PRE_MED,PM.HR_PRE_MED),'DD/MM/YYYY HH24') as DH_PRE_MED
				,P.NM_PACIENTE
				,'EMERGENCIA' DS_RESUMO
				,O.DS_ORI_ATE
				,V.CODIGO_BARRAS
				,V.DS_TIP_PRESC
				,TO_CHAR(P.Dt_Nascimento, 'DD/MM/YYYY') Dt_Nascimento
		FROM 	DBAMV.PRE_MED PM
				,DBAMV.ATENDIME A
				,DBAMV.ORI_ATE O
				,DBAMV.PACIENTE P
				,(	SELECT 'T'||X.CD_ITPRE_MED CODIGO_BARRAS, TIP_PRESC.DS_TIP_PRESC
					FROM 	DBAMV.ITPRE_MED X, DBAMV.TIP_PRESC
					WHERE x.CD_ITPRE_MED = :CAMPO_CD_ITPRE_MED
					AND X.CD_TIP_PRESC = TIP_PRESC.CD_TIP_PRESC) V
		WHERE 	A.CD_PACIENTE     	= P.CD_PACIENTE
		AND 	PM.CD_ATENDIMENTO	= A.CD_ATENDIMENTO
		AND 	O.CD_ORI_ATE    	= A.CD_ORI_ATE
		AND 	A.TP_ATENDIMENTO IN ('A', 'U', 'E')
		AND 	PM.CD_PRE_MED    IN (	SELECT IPM.CD_PRE_MED
										FROM DBAMV.ITPRE_MED IPM
										WHERE IPM.CD_ITPRE_MED = :CAMPO_CD_ITPRE_MED )
	)
