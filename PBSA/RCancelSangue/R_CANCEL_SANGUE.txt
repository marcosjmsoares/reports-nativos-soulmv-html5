--R_CANCEL_SANGUE
SELECT * FROM (
SELECT PRE_MED.CD_PRE_MED             CD_PRE_MED
              ,PRE_MED.DT_PRE_MED             DT_PRE_MED
              ,PRE_MED.HR_PRE_MED             HR_PRE_MED
              ,PRE_MED.NM_USUARIO              NM_USUARIO
              ,PRE_MED.CD_ATENDIMENTO    CD_ATENDIMENTO
              ,LEITO.DS_RESUMO                      DS_RESUMO
              ,UNID_INT.CD_UNID_INT              CD_UNID_INT
              ,UNID_INT.DS_UNID_INT              DS_UNID_INT
              ,PACIENTE.CD_PACIENTE             CD_PACIENTE
              ,PACIENTE.NM_PACIENTE            NM_PACIENTE
              ,PACIENTE.DT_NASCIMENTO       DT_NASCIMENTO
              ,DECODE(PACIENTE.TP_SEXO, 'M', 'MASCULINO', 'FEMININO')    DS_SEXO
              ,PACK_INTERNAMENTO.RETORNA_IDADE(PACIENTE.DT_NASCIMENTO, TRUNC(SYSDATE))   DS_IDADE
              ,CONVENIO.CD_CONVENIO                            CD_CONVENIO
              ,CONVENIO.NM_CONVENIO                           NM_CONVENIO
              ,PRESTADOR.CD_PRESTADOR                    CD_PRESTADOR
              ,PRESTADOR.NM_PRESTADOR                    NM_PRESTADOR
              ,PRESTADOR.DS_CODIGO_CONSELHO       DS_CODIGO_CONSELHO
              ,CID.CD_CID                                                     CD_CID
              ,CID.DS_CID                                                     DS_CID
              ,CARTEIRA.NR_CARTEIRA                             NR_CARTEIRA
              ,DERIV.CD_SANGUE_DERIVADOS                CD_PRO_FAT
              ,DERIV.DS_SANGUE_DERIVADOS                DS_PRO_FAT
              ,ITPRE_MED.QT_ITPRE_MED                        QT_ITPRE_MED
              ,ITPRE_MED.DS_ITPRE_MED                        DS_ITPRE_MED
              ,ITPRE_MED.CD_ITPRE_MED                        CD_ITPRE_MED
              ,TIP_FRE.DS_TIP_FRE                                    DS_TIP_FRE
              ,SOLIC_SANGUE.CD_SOLIC_SANGUE           PEDIDO
FROM  DBAMV.PRE_MED
            ,DBAMV.PRESTADOR
            ,DBAMV.ATENDIME
            ,DBAMV.LEITO
            ,DBAMV.UNID_INT
            ,DBAMV.PACIENTE
            ,DBAMV.CONVENIO
            ,DBAMV.CID
            ,DBAMV.CARTEIRA
            ,DBAMV.ITPRE_MED
            ,DBAMV.TIP_PRESC
            ,DBAMV.SANGUE_DERIVADOS  DERIV
            ,DBAMV.TIP_FRE
            ,DBAMV.SOLIC_SANGUE
	    ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And PRE_MED.CD_PRE_MED in (Select Distinct Itpre_Med_Canc.cd_Pre_Med Cd_Pre_Med
                                                                  From Dbamv.Itpre_Med Itpre_Med,
                                                                          Dbamv.Itpre_Med Itpre_Med_Canc
                                                               Where  itPre_Med.Cd_Pre_Med = :CAMPO_PCD_PRE_MED
                                                                   and Itpre_Med.Cd_Itpre_Med_Canc = Itpre_Med_Canc.cd_Itpre_Med)
AND       PRE_MED.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
AND       PRE_MED.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
AND       ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
AND       LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)
AND       ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND       ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND       ATENDIME.CD_CID = CID.CD_CID(+)
AND       ATENDIME.CD_CONVENIO = CARTEIRA.CD_CONVENIO(+)
AND      ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND       ATENDIME.CD_CON_PLA = CARTEIRA.CD_CON_PLA(+)
AND       ATENDIME.CD_PACIENTE = CARTEIRA.CD_PACIENTE(+)
AND       PRE_MED.CD_PRE_MED = ITPRE_MED.CD_PRE_MED
AND       ITPRE_MED.CD_TIP_PRESC = TIP_PRESC.CD_TIP_PRESC
AND       TIP_PRESC.CD_SANGUE_DERIVADOS = DERIV.CD_SANGUE_DERIVADOS
AND       ITPRE_MED.CD_TIP_FRE = TIP_FRE.CD_TIP_FRE
AND       PRE_MED.CD_PRE_MED = SOLIC_SANGUE.CD_PRE_MED
)
ORDER BY CD_PRE_MED,DT_PRE_MED,HR_PRE_MED,NM_USUARIO,CD_ATENDIMENTO,CD_UNID_INT,DS_UNID_INT,DS_RESUMO,CD_PACIENTE,NM_PACIENTE,DT_NASCIMENTO,DS_IDADE,DS_SEXO,CD_CONVENIO,NM_CONVENIO,CD_PRESTADOR,NM_PRESTADOR,DS_CODIGO_CONSELHO,CD_CID,DS_CID,NR_CARTEIRA
