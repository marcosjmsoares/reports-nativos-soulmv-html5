--R_CENSO_HCARE
SELECT ATENDIME_HCARE.CD_AREA
      ,AREA.DS_AREA
      ,ATENDIME.CD_ATENDIMENTO
      ,PACIENTE.CD_PACIENTE
      ,PACIENTE.NM_PACIENTE
      ,PACIENTE.DT_NASCIMENTO
      ,ENDERECO.NR_FONE
      ,ATENDIME.CD_CONVENIO
      ,CONVENIO.NM_CONVENIO
      ,ATENDIME.CD_PRESTADOR
      ,PRESTADOR.NM_PRESTADOR
      ,ATENDIME.DT_ATENDIMENTO
      ,ROUND((SYSDATE - ATENDIME.DT_ATENDIMENTO)) DIAS
      ,ATENDIME.CD_TIPO_INTERNACAO
      ,TIPO_INTERNACAO.DS_TIPO_INTERNACAO
  FROM DBAMV.ATENDIME_HCARE
      ,DBAMV.AREA
      ,DBAMV.ATENDIME
      ,DBAMV.PACIENTE
      ,DBAMV.CONVENIO
      ,DBAMV.PRESTADOR
      ,DBAMV.TIPO_INTERNACAO
      ,DBAMV.ENDERECO
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And ATENDIME_HCARE.CD_ATENDIMENTO      = ATENDIME.CD_ATENDIMENTO
   AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
   AND AREA.CD_AREA                       = ATENDIME_HCARE.CD_AREA
   AND PACIENTE.CD_PACIENTE               = ATENDIME.CD_PACIENTE
   AND ATENDIME.DT_ALTA IS NULL
   AND CONVENIO.CD_CONVENIO               = ATENDIME.CD_CONVENIO
   AND TIPO_INTERNACAO.CD_TIPO_INTERNACAO = ATENDIME.CD_TIPO_INTERNACAO
   AND PRESTADOR.CD_PRESTADOR             = ATENDIME.CD_PRESTADOR
   AND ENDERECO.CD_PACIENTE(+)            = ATENDIME_HCARE.CD_PACIENTE
   AND ENDERECO.CD_ENDERECO(+)            = ATENDIME_HCARE.CD_ENDERECO
