--R_ORDEM_SERVICO_HC
SELECT aten.CD_ATENDIMENTO,
       paci.CD_PACIENTE,
       paci.NM_PACIENTE,
       paci.TP_SEXO,
       paci.DT_NASCIMENTO,
       paci.NM_MAE,
       paci.NM_PAI,
       paci.NR_CPF,
       paci.NR_DOCUMENTO,
       ende.DS_ENDERECO,
       ende.NR_ENDERECO,
       ende.DS_COMPLEMENTO,
       ende.NM_BAIRRO,
       ende.NR_FONE,
       ende.DS_PTO_REFERENCIA,
       SubStr(LPad(to_char(loca.NR_CEP,'99999999'),8,'0'),1,5) ||'-'||SubStr(LPad(to_char(loca.NR_CEP,'99999999'),8,'0'),6,8)  NR_CEP ,
       loca.NM_LOCALIDADE,
       loca.CD_UF,
       area.CD_AREA,
       area.DS_AREA,
       pres.CD_PRESTADOR,
       pres.NM_PRESTADOR,
       pres.NR_FONE_CONTATO,
       tipr.NM_TIP_PRESTA,
       cid.CD_CID,
       cid.DS_CID,
       resp.NM_RESPONSAVEL,
       resp.NR_FONE FoneResp,
       tipa.DS_TIP_PAREN,
       hcar.DT_SOLICITACAO,
       hcar.DT_PREV_INTER,
       pre1.CD_PRESTADOR CdEnfer,
       pre1.NM_PRESTADOR NmEnfer
  FROM DBAMV.ATENDIME aten,
       DBAMV.PACIENTE paci,
       DBAMV.ATENDIME_HCARE hcar,
       DBAMV.ENDERECO ende,
       DBAMV.CEP_LOCALIDADES loca,
       DBAMV.AREA area,
       DBAMV.PRESTADOR pres,
       DBAMV.TIP_PRESTA tipr,
       DBAMV.CID cid,
       DBAMV.RESPONSA resp,
       DBAMV.TIP_PAREN tipa,
       DBAMV.PRESTADOR pre1
 WHERE aten.TP_ATENDIMENTO   = 'H'
   AND  ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
   AND aten.CD_PACIENTE      = paci.CD_PACIENTE
   AND hcar.CD_ATENDIMENTO   = aten.CD_ATENDIMENTO
   AND hcar.CD_PACIENTE      = ende.CD_PACIENTE
   AND hcar.CD_ENDERECO      = ende.CD_ENDERECO
   AND ende.CD_LOCALIDADE    = loca.CD_LOCALIDADE(+)
   AND hcar.CD_AREA          = area.CD_AREA
   AND hcar.CD_PRESTADOR_ACO = pres.CD_PRESTADOR
   AND pres.CD_TIP_PRESTA = tipr.CD_TIP_PRESTA
   AND aten.CD_CID           = cid.CD_CID(+)
   AND resp.CD_ATENDIMENTO   = aten.CD_ATENDIMENTO(+)
   AND resp.CD_TIP_PAREN     = tipa.CD_TIP_PAREN(+)
   AND hcar.CD_ENFER_SUPER   = pre1.CD_PRESTADOR
ORDER BY area.DS_AREA,
         paci.NM_PACIENTE
