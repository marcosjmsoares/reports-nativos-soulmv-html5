--R_PROCED_X_POS_RESULT
SELECT * FROM (
SELECT DISTINCT POSSIBILIDADE_RESULTADO.DS_POSSIBILIDADE_RESULTADO
                               ,PROCEDIMENTOS_CUIDADOS.DS_PROCEDIMENTOS_CUIDADOS
                               ,ITENS_EXAMINAR.DS_ITENS_EXAMINAR
FROM  DBAMV.POSSIBILIDADE_RESULTADO
           ,DBAMV.POSSIBILIDADE_PLANO
           ,DBAMV.PROCEDIMENTOS_CUIDADOS
           ,DBAMV.ITENS_EXAMINAR
WHERE POSSIBILIDADE_RESULTADO.CD_POSSIBILIDADE_RESULTADO =  POSSIBILIDADE_PLANO.CD_POSSIBILIDADE_RESULTADO
      AND POSSIBILIDADE_PLANO.CD_PROCEDIMENTOS_CUIDADOS = PROCEDIMENTOS_CUIDADOS.CD_PROCEDIMENTOS_CUIDADOS
     AND  POSSIBILIDADE_RESULTADO.CD_ITENS_EXAMINAR = ITENS_EXAMINAR.CD_ITENS_EXAMINAR
:CAMPO_CF_ITENS_EXAMINAR
)
ORDER BY DS_ITENS_EXAMINAR,DS_POSSIBILIDADE_RESULTADO
