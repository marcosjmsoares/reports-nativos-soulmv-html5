--R_PROD_LOC_ATEND
SELECT * FROM (
SELECT   hcar.CD_ATENDIMENTO                CD_ATENDIMENTO
        ,paci.CD_PACIENTE                   CD_PACIENTE
        ,paci.NM_PACIENTE                   NM_PACIENTE
        ,sols.CD_SOLSAI_PRO                 CD_SOLSAI_PRO
        ,sols.DT_SOLSAI_PRO                 DT_SOLSAI_PRO
        ,sols.CD_USUARIO                    CD_USUARIO
        ,clas.DS_CLASSE                     DS_CLASSE
        ,prod.CD_PRODUTO                    CD_PRODUTO
        ,prod.DS_PRODUTO                    DS_PRODUTO
        ,itlo.QT_LOCADA                     QT_LOCADA
        ,unpr.DS_UNIDADE                    DS_UNIDADE
        ,itlo.DT_LOCACAO                    DT_LOCACAO
        ,forn.CD_FORNECEDOR                 CD_FORNECEDOR
        ,forn.NM_FORNECEDOR                 NM_FORNECEDOR
        ,ordem_dev.DT_DEVOLUCAO             DT_DEVOLUCAO
  FROM DBAMV.ATENDIME aten,
       DBAMV.ATENDIME_HCARE hcar,
       DBAMV.PACIENTE paci,
       DBAMV.SOLSAI_PRO sols,
       DBAMV.PRODUTO prod,
       DBAMV.CLASSE clas,
       DBAMV.ITMVT_LOCACAO itlo,
       DBAMV.UNI_PRO unpr,
       DBAMV.FORNECEDOR forn,
       DBAMV.ORDEM_DEVOLUCAO    ordem_dev,
       DBAMV.ESTOQUE
 WHERE aten.DT_ALTA IS NULL
   AND  ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
   AND hcar.CD_ATENDIMENTO             = aten.CD_ATENDIMENTO
   AND aten.CD_PACIENTE                      = paci.CD_PACIENTE
   AND sols.CD_ATENDIMENTO              = hcar.CD_ATENDIMENTO
   and   hcar.CD_ATENDIMENTO             = ordem_dev.CD_ATENDIMENTO(+)
   AND prod.CD_CLASSE                          = clas.CD_CLASSE
   AND prod.CD_ESPECIE                         = clas.CD_ESPECIE
   AND itlo.CD_SOLSAI_PRO                    = sols.CD_SOLSAI_PRO
   AND itlo.DT_LOCACAO is not null
   AND itlo.CD_PRODUTO                         = prod.CD_PRODUTO
   AND itlo.CD_UNI_PRO                           = unpr.CD_UNI_PRO
   AND itlo.CD_FORNECEDOR                  = forn.CD_FORNECEDOR(+)
   AND sols.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
   AND ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
ORDER BY paci.NM_PACIENTE,
         sols.CD_SOLSAI_PRO,
         clas.DS_CLASSE,
         prod.DS_PRODUTO
)
ORDER BY NM_PACIENTE,CD_ATENDIMENTO,CD_PACIENTE,
CD_SOLSAI_PRO,DT_SOLSAI_PRO,CD_USUARIO,
DS_CLASSE
