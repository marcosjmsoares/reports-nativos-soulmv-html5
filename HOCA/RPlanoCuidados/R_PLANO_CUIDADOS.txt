SELECT * FROM (
SELECT DISTINCT EXAME_FISICO.CD_EXAME_FISICO
      ,EXAME_FISICO.CD_TIPO_PACIENTE
      ,TIPO_PACIENTE.DS_TIPO_PACIENTE
      ,EXAME_FISICO.CD_ATENDIMENTO
      ,EXAME_FISICO.CD_ATENDIMENTO||' - '||PACIENTE.NM_PACIENTE NM_PACIENTE
      ,EXAME_FISICO.DT_EXAME_FISICO
      ,EXAME_FISICO.HR_EXAME_FISICO
      ,EXAME_FISICO.CD_ENFER_SUPER
      ,PRESTADOR.NM_PRESTADOR
  FROM dbamv.EXAME_FISICO
      ,dbamv.TIPO_PACIENTE
      ,dbamv.PRESTADOR
      ,dbamv.PACIENTE
      ,dbamv.ATENDIME
 WHERE ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
   AND TIPO_PACIENTE.CD_TIPO_PACIENTE                     = EXAME_FISICO.CD_TIPO_PACIENTE
   AND PRESTADOR.CD_PRESTADOR                             = EXAME_FISICO.CD_ENFER_SUPER
   AND ATENDIME.CD_ATENDIMENTO                            = EXAME_FISICO.CD_ATENDIMENTO
   AND PACIENTE.CD_PACIENTE                               = ATENDIME.CD_PACIENTE
   AND EXAME_FISICO.CD_EXAME_FISICO                       = :CAMPO_CD_EXAME_FISICO
)
