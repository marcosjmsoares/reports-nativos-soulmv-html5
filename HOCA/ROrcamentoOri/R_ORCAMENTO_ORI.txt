--R_ORCAMENTO_ORI
SELECT * FROM (
SELECT orha.CD_ATENDIMENTO,
       orha.CD_ORCAMENTO,
       paci.CD_PACIENTE,
       paci.NM_PACIENTE,
       conv.CD_CONVENIO,
       conv.NM_CONVENIO,
       copl.CD_CON_PLA,
       copl.DS_CON_PLA,
       orha.DT_ORCAMENTO,
       aten.CD_TIP_ACOM,
       aten.TP_ATENDIMENTO,
       prfa.CD_PRO_FAT,
       prfa.DS_PRO_FAT,
       itor.QTDE
  FROM DBAMV.ATENDIME aten,
       DBAMV.PACIENTE paci,
       DBAMV.ORCAMENTO_HOCA orha,
       DBAMV.CONVENIO conv,
       DBAMV.CON_PLA copl,
       DBAMV.ITORCAMENTO_HOCA itor,
       DBAMV.PRO_FAT prfa
 WHERE orha.CD_ATENDIMENTO = :CAMPO_ATEND
   AND orha.CD_ORCAMENTO = :CAMPO_ORCAM
   AND aten.CD_PACIENTE    = paci.CD_PACIENTE
   AND orha.CD_ATENDIMENTO = aten.CD_ATENDIMENTO
   AND aten.CD_CONVENIO    = conv.CD_CONVENIO
   AND aten.CD_CONVENIO    = copl.CD_CONVENIO
   AND aten.CD_CON_PLA     = copl.CD_CON_PLA
   AND itor.CD_ORCAMENTO   = orha.CD_ORCAMENTO
   AND itor.CD_PRO_FAT     = prfa.CD_PRO_FAT
   AND aten.cd_multi_empresa = nvl(dbamv.pkg_mv2000.le_empresa, aten.cd_multi_empresa)
order by aten.CD_ATENDIMENTO, prfa.DS_PRO_FAT
)
ORDER BY CD_ATENDIMENTO,CD_ORCAMENTO,CD_PACIENTE,NM_PACIENTE,CD_CONVENIO,NM_CONVENIO,CD_CON_PLA,DS_CON_PLA,DT_ORCAMENTO,TP_ATENDIMENTO,CD_TIP_ACOM
