--R_ITENS_EXAMINAR
SELECT * FROM (
SELECT DISTINCT POSSIBILIDADE_RESULTADO.DS_POSSIBILIDADE_RESULTADO
      ,ITENS_EXAMINAR.DS_ITENS_EXAMINAR
FROM  DBAMV.POSSIBILIDADE_RESULTADO
           ,DBAMV.ITENS_EXAMINAR
WHERE POSSIBILIDADE_RESULTADO.CD_ITENS_EXAMINAR =  ITENS_EXAMINAR.CD_ITENS_EXAMINAR
:CAMPO_CF_ITENS_EXAMINAR
)
ORDER BY DS_ITENS_EXAMINAR
