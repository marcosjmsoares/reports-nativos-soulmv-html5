--R_INT_PREVISTOS
SELECT DBAMV.ATENDIME.CD_ATENDIMENTO,
       DBAMV.PACIENTE.CD_PACIENTE,
       DBAMV.PACIENTE.NM_PACIENTE,
       DBAMV.PACIENTE.TP_SEXO,
       DBAMV.ENDERECO.NM_BAIRRO,
       DBAMV.ENDERECO.DS_ENDERECO,
       DBAMV.ENDERECO.NR_ENDERECO,
       DBAMV.ENDERECO.NR_FONE,
       DBAMV.ENDERECO.DS_COMPLEMENTO,
       DBAMV.ENDERECO.DS_PTO_REFERENCIA,
       DBAMV.ATENDIME_HCARE.DT_SOLICITACAO,
       DBAMV.ATENDIME_HCARE.DT_PREV_INTER,
       DBAMV.CEP_LOCALIDADES.NM_LOCALIDADE AS NM_LOCALIDADE
  FROM DBAMV.ATENDIME,
       DBAMV.PACIENTE,
       DBAMV.ATENDIME_HCARE,
       DBAMV.ENDERECO,
       DBAMV.CEP_LOCALIDADES
 WHERE DBAMV.ATENDIME_HCARE.DT_PREV_INTER BETWEEN :CAMPO_DTINIC and 
   AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
   AND DBAMV.ATENDIME.TP_ATENDIMENTO       = 'H'
   AND DBAMV.ATENDIME.CD_PACIENTE          = DBAMV.PACIENTE.CD_PACIENTE
   AND DBAMV.ATENDIME.CD_ATENDIMENTO       = DBAMV.ATENDIME_HCARE.CD_ATENDIMENTO
   AND DBAMV.ATENDIME_HCARE.CD_PACIENTE    = DBAMV.ENDERECO.CD_PACIENTE
   AND DBAMV.ATENDIME_HCARE.CD_ENDERECO    = DBAMV.ENDERECO.CD_ENDERECO
   AND DBAMV.ENDERECO.CD_LOCALIDADE        = DBAMV.CEP_LOCALIDADES.CD_LOCALIDADE(+)
ORDER BY DBAMV.ATENDIME_HCARE.DT_PREV_INTER,
         DBAMV.PACIENTE.NM_PACIENTE
