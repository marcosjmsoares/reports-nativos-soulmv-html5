--R_PACIENTE_ENFERMEIRA
SELECT * FROM (
SELECT pres.CD_PRESTADOR,
       pres.NM_PRESTADOR,
       aten.CD_ATENDIMENTO,
       aten.DT_ATENDIMENTO,
       paci.CD_PACIENTE,
       paci.NM_PACIENTE,
       ende.NR_FONE,
       ende.NM_BAIRRO,
       loca.NM_LOCALIDADE,
       area.CD_AREA,
       area.DS_AREA
  FROM DBAMV.ATENDIME aten,
       DBAMV.PACIENTE paci,
       DBAMV.ATENDIME_HCARE care,
       DBAMV.ENDERECO ende,
       DBAMV.CEP_LOCALIDADES loca,
       DBAMV.PRESTADOR pres,
       DBAMV.AREA area
 WHERE aten.TP_ATENDIMENTO = 'H'
   AND  ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
   AND aten.CD_PACIENTE    = paci.CD_PACIENTE
   AND care.CD_ATENDIMENTO = aten.CD_ATENDIMENTO
   AND care.CD_ENDERECO    = ende.CD_ENDERECO
   AND care.CD_PACIENTE    = ende.CD_PACIENTE
   AND ende.CD_LOCALIDADE  = loca.CD_LOCALIDADE(+)
   AND care.CD_ENFER_SUPER = pres.CD_PRESTADOR
   AND care.CD_AREA        = area.CD_AREA
   AND aten.DT_ATENDIMENTO BETWEEN  :CAMPO_DTINI ORDER BY pres.NM_PRESTADOR,
         paci.NM_PACIENTE
)
ORDER BY NM_PRESTADOR,CD_PRESTADOR,NM_PACIENTE
