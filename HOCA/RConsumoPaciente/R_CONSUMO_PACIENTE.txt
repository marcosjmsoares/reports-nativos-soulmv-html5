--R_CONSUMO_PACIENTE
SELECT * FROM (
SELECT ATENDIME.CD_ATENDIMENTO
             ,ATENDIME.CD_PACIENTE
             ,PACIENTE.NM_PACIENTE
             ,PRODUTO.CD_PRODUTO
             ,PRODUTO.DS_PRODUTO
             ,V_PROD_ATEND.DS_UNIDADE
             ,(NVL(V_PROD_ATEND.QT_SAIDA,0) - (NVL(V_PROD_ATEND.QT_CONSUMIDA,0) + NVL(V_PROD_ATEND.QT_DEVOLVIDA,0)))  QT_ATUAL
FROM DBAMV.V_PROD_ATEND
           ,DBAMV.PRODUTO
           ,DBAMV.ATENDIME
           ,DBAMV.PACIENTE
WHERE           ATENDIME.TP_ATENDIMENTO               = 'H'
           AND      ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
           AND      V_PROD_ATEND.CD_PRODUTO             =  PRODUTO.CD_PRODUTO
           AND      V_PROD_ATEND.CD_ATENDIMENTO     =  ATENDIME.CD_ATENDIMENTO
          AND Exists ( SELECT Empresa_Produto.Cd_Produto FROM Dbamv.Empresa_Produto WHERE Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA AND Empresa_Produto.Cd_Produto = Produto.Cd_Produto )
           AND      ATENDIME.CD_PACIENTE                        =  PACIENTE.CD_PACIENTE
ORDER BY ATENDIME.CD_ATENDIMENTO
                   ,PACIENTE.CD_PACIENTE
                   ,PRODUTO.DS_PRODUTO
)
ORDER BY CD_PACIENTE,CD_ATENDIMENTO,NM_PACIENTE
