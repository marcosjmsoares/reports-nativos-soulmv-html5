--R_ORCAMENTO
select * from(
   SELECT ORCAMENTO_HOCA.CD_ORCAMENTO,
       PACIENTE.CD_PACIENTE,
       PACIENTE.NM_PACIENTE,
       CONVENIO.CD_CONVENIO,
       CONVENIO.NM_CONVENIO,
       CON_PLA.CD_CON_PLA,
       CON_PLA.DS_CON_PLA,
       ORCAMENTO_HOCA.DT_ORCAMENTO,
       ORCAMENTO_HOCA.NR_CARTEIRA,
       ORCAMENTO_HOCA.DT_VALIDADE,
       ORCAMENTO_HOCA.DT_INIC_PERIODO,
       SUBSTR( TO_CHAR( ORCAMENTO_HOCA.DT_INIC_PERIODO, 'DAY' ),1,3 ) DIA_INIC_PERIODO,
       ORCAMENTO_HOCA.DT_TERM_PERIODO,
       SUBSTR( TO_CHAR( ORCAMENTO_HOCA.DT_TERM_PERIODO, 'DAY' ),1,3 ) DIA_TERM_PERIODO,
       ORCAMENTO_HOCA.CD_TIPO_INTERNACAO,
       TIPO_INTERNACAO.DS_TIPO_INTERNACAO,
       ORCAMENTO_HOCA.TP_MODALIDADE_ATENDI,
       ORCAMENTO_HOCA.DS_DIAGNOSTICO,
       ORCAMENTO_HOCA.NM_USUARIO,
       ITORCAMENTO_AVALIACAO.DS_AVALIACAO_ADMISSIONAL,
       ITORCAMENTO_PROPOSTA.DS_PROPOSTA_TERAPEUTICA,
       ITFOR_APRE.CD_SEQUENCIA,
       GRU_PRO.DS_GRU_PRO,
       PRO_FAT.CD_PRO_FAT,
       ITORCAMENTO_HOCA.DS_PRO_FAT,
		dbamv.fnc_retorna_fnc_ffcv_gera_tiss(convenio.cd_convenio, 'H','C', pro_fat.cd_pro_fat) CD_TUSS,
		dbamv.fnc_retorna_fnc_ffcv_gera_tiss(convenio.cd_convenio, 'H','D', pro_fat.cd_pro_fat) DS_TUSS,
		22 TISS,       
		ITORCAMENTO_HOCA.DS_UNIDADE,
       ITORCAMENTO_HOCA.QTDE,
       ITORCAMENTO_HOCA.VL_PRO_FAT VL_PRO_FAT,
       ROUND ((ITORCAMENTO_HOCA.VL_PRO_FAT), 2) VL_TOTAL,
       ITFOR_APRE.SN_IMPRIME_UNIDADE,
       ITFOR_APRE.SN_IMPRIME_CODIGO_COBRANCA,
       ITFOR_APRE.SN_IMPRIME_UNITARIO,
       ITFOR_APRE.SN_IMPRIME_TOTAL,
       Decode(ORCAMENTO_HOCA.SITUACAO_ORCAMENTO,'E','Espera','A','Aprovado','R','Recusado','S','Substituido') SITUACAO_ORCAMENTO
  FROM DBAMV.PACIENTE PACIENTE,
       DBAMV.ORCAMENTO_HOCA ORCAMENTO_HOCA,
       DBAMV.CONVENIO CONVENIO,
       DBAMV.CON_PLA CON_PLA,
       DBAMV.ITORCAMENTO_HOCA ITORCAMENTO_HOCA,
       DBAMV.ITORCAMENTO_PROPOSTA,
       DBAMV.PRO_FAT PRO_FAT,
       DBAMV.GRU_PRO,
       DBAMV.GRU_FAT,
       DBAMV.ITORCAMENTO_AVALIACAO,
       DBAMV.ITFOR_APRE,
       DBAMV.TIPO_INTERNACAO,
       DBAMV.EMPRESA_CON_PLA
WHERE EMPRESA_CON_PLA.CD_MULTI_EMPRESA = dbamv.pkg_mv2000.le_empresa
   AND EMPRESA_CON_PLA.CD_CONVENIO = CON_PLA.CD_CONVENIO
   AND EMPRESA_CON_PLA.CD_CON_PLA = CON_PLA.CD_CON_PLA
   AND PACIENTE.CD_PACIENTE                  = ORCAMENTO_HOCA.CD_PACIENTE
   AND CONVENIO.CD_CONVENIO                  = ORCAMENTO_HOCA.CD_CONVENIO
   AND CON_PLA.CD_CONVENIO                   = CONVENIO.CD_CONVENIO
   AND CON_PLA.CD_CON_PLA                    = ORCAMENTO_HOCA.CD_CON_PLA
   AND ITORCAMENTO_HOCA.CD_ORCAMENTO         = ORCAMENTO_HOCA.CD_ORCAMENTO
   AND ITORCAMENTO_AVALIACAO.CD_ORCAMENTO(+) = ORCAMENTO_HOCA.CD_ORCAMENTO
   AND ITORCAMENTO_PROPOSTA.CD_ORCAMENTO(+) = ORCAMENTO_HOCA.CD_ORCAMENTO
   AND ITORCAMENTO_HOCA.CD_PRO_FAT           = PRO_FAT.CD_PRO_FAT
   AND GRU_PRO.CD_GRU_PRO                    = PRO_FAT.CD_GRU_PRO
   AND GRU_FAT.CD_GRU_FAT                    = GRU_PRO.CD_GRU_FAT
   AND ITFOR_APRE.CD_FOR_APRE                = CONVENIO.CD_FOR_APRE
   AND ITFOR_APRE.CD_GRU_FAT                 = GRU_FAT.CD_GRU_FAT
   AND TIPO_INTERNACAO.CD_TIPO_INTERNACAO(+) = ORCAMENTO_HOCA.CD_TIPO_INTERNACAO
ORDER BY ORCAMENTO_HOCA.CD_ORCAMENTO
        ,ITFOR_APRE.CD_SEQUENCIA
        ,GRU_PRO.DS_GRU_PRO
        ,PRO_FAT.DS_PRO_FAT
)
