--R_CONFIG_PENDENCIA
SELECT * FROM (
SELECT DISTINCT CONFIG_PEND_HCARE.CD_PENDENCIA
      ,CONFIG_PEND_HCARE.DS_PENDENCIA
      ,CONFIG_PEND_HCARE.CD_TIPO_PENDENCIA
      ,TIPO_PENDENCIA.DS_TIPO_PENDENCIA
      ,IT_PENDENCIA.CD_IT_PENDENCIA
      ,DECODE(CONFIG_PEND_HCARE.TP_ACAO_PENDENCIA,'I','Internação'
                                                                                                      ,'M','Manutenção'
                                                                                                      ,'T','Transferência'
                                                                                                      ,'A','Alta Médica'
                                                                                                      ,'H','Alta Hospitalar') TP_ACAO_PENDENCIA
FROM DBAMV.CONFIG_PEND_HCARE
          ,DBAMV.TIPO_PENDENCIA
          ,DBAMV.IT_PENDENCIA
WHERE CONFIG_PEND_HCARE.CD_TIPO_PENDENCIA = TIPO_PENDENCIA.CD_TIPO_PENDENCIA
      AND CONFIG_PEND_HCARE.CD_PENDENCIA = IT_PENDENCIA.CD_PENDENCIA(+)
:CAMPO_CF_PENDENCIA
)
ORDER BY CD_PENDENCIA,DS_PENDENCIA,CD_TIPO_PENDENCIA,DS_TIPO_PENDENCIA,TP_ACAO_PENDENCIA
