--R_MANUTENCAO_PACIENTE
SELECT * FROM (
SELECT area.CD_AREA,
       area.DS_AREA,
       paci.CD_PACIENTE,
       paci.NM_PACIENTE,
       tipe.CD_TIPO_PENDENCIA,
       tipe.DS_TIPO_PENDENCIA,
       DECODE(peat.DT_BAIXA_PENDENCIA, NULL, '/', 'X') Situacao
  FROM DBAMV.PENDENCIA_ATENDIMENTO peat,
       DBAMV.ATENDIME_HCARE hcar,
       DBAMV.ATENDIME aten,
       DBAMV.PACIENTE paci,
       DBAMV.AREA area,
       DBAMV.TIPO_PENDENCIA tipe
 WHERE  hcar.CD_ATENDIMENTO = aten.CD_ATENDIMENTO
   AND  ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
   AND aten.DT_ALTA is null
   AND hcar.CD_PACIENTE       = paci.CD_PACIENTE
   AND hcar.CD_AREA           = area.CD_AREA
   AND hcar.CD_ATENDIMENTO    = peat.CD_ATENDIMENTO
   AND peat.CD_TIPO_PENDENCIA = tipe.CD_TIPO_PENDENCIA
ORDER BY area.DS_AREA,
         paci.NM_PACIENTE,
         tipe.CD_TIPO_PENDENCIA
)
ORDER BY DS_AREA,CD_AREA,
NM_PACIENTE,CD_PACIENTE,
CD_TIPO_PENDENCIA,DS_TIPO_PENDENCIA
