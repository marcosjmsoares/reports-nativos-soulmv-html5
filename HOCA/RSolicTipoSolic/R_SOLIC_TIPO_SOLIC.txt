--R_SOLIC_TIPO_SOLIC
SELECT * FROM (
SELECT SOLSAI_PRO.CD_ATENDIMENTO
      ,PACIENTE.NM_PACIENTE
      ,SOLSAI_PRO.CD_TP_SOLICITACAO        CD_TP_SOLICITACAO
      ,TIPO_SOLICITACAO.DS_TP_SOLICITACAO  DS_TP_SOLICITACAO
      ,COUNT(SOLSAI_PRO.CD_TP_SOLICITACAO) QT_SOLICITACAO
  FROM DBAMV.SOLSAI_PRO
      ,DBAMV.TIPO_SOLICITACAO
      ,DBAMV.PACIENTE
      ,DBAMV.ATENDIME
      ,DBAMV.ESTOQUE
 WHERE TIPO_SOLICITACAO.CD_TP_SOLICITACAO = SOLSAI_PRO.CD_TP_SOLICITACAO
   AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
   AND ATENDIME.CD_ATENDIMENTO            = SOLSAI_PRO.CD_ATENDIMENTO
   AND PACIENTE.CD_PACIENTE               = ATENDIME.CD_PACIENTE
   AND SOLSAI_PRO.DT_SOLSAI_PRO BETWEEN :CAMPO_DATA_INICIAL AND 
   AND SOLSAI_PRO.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
   AND ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY SOLSAI_PRO.CD_ATENDIMENTO
        ,PACIENTE.NM_PACIENTE
        ,SOLSAI_PRO.CD_TP_SOLICITACAO
        ,TIPO_SOLICITACAO.DS_TP_SOLICITACAO
)
ORDER BY CD_ATENDIMENTO,NM_PACIENTE
