--R_PACIENTE_TRANSFERIDO
SELECT * FROM (
SELECT atendi.CD_ATENDIMENTO,
       pacien.CD_PACIENTE,
       pacien.NM_PACIENTE,
       endatu.DS_ENDERECO,
       endatu.NR_ENDERECO,
       endatu.DS_COMPLEMENTO,
       endatu.NM_BAIRRO,
       endatu.NR_FONE,
       endatu.DS_PTO_REFERENCIA,
       locatu.NR_CEP,
       locatu.NM_LOCALIDADE,
       locatu.CD_UF,
       endant.DS_ENDERECO DS_ENDERECO2,
       endant.DS_COMPLEMENTO DS_COMPLEMENTO2,
       endant.NR_ENDERECO NR_ENDERECO2,
       endant.NM_BAIRRO NM_BAIRRO2,
       endant.NR_FONE NR_FONE2,
       endant.DS_PTO_REFERENCIA DS_PTO_REFERENCIA2,
       locant.NR_CEP NR_CEP2,
       locant.NM_LOCALIDADE NM_LOCALIDADE2,
       locant.CD_UF CD_UF2,
       areatu.CD_AREA,
       areatu.DS_AREA,
       areant.CD_AREA CD_AREA2,
       areant.DS_AREA DS_AREA2,
       transf.USUARIO,
       transf.MOTIVO,
       transf.DT_TRANSF
  FROM DBAMV.ENDERECO endatu,
       DBAMV.ENDERECO endant,
       DBAMV.CEP_LOCALIDADES locatu,
       DBAMV.CEP_LOCALIDADES locant,
       DBAMV.TRANSF_HOCA transf,
       DBAMV.AREA areatu,
       DBAMV.AREA areant,
       DBAMV.PACIENTE pacien,
       DBAMV.ATENDIME_HCARE atenhc,
       DBAMV.ATENDIME atendi
 WHERE transf.CD_ATENDIMENTO       = atenhc.CD_ATENDIMENTO
   AND  ATENDI.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
   AND atenhc.SN_INT_EFETIVADA     = 'S'
   AND transf.CD_ATENDIMENTO       = atendi.CD_ATENDIMENTO
   AND atendi.DT_ALTA IS NULL
   AND transf.CD_PACIENTE          = endatu.CD_PACIENTE
   AND transf.CD_ENDERECO          = endatu.CD_ENDERECO
   AND endatu.CD_LOCALIDADE        = locatu.CD_LOCALIDADE
   AND transf.CD_PACIENTE          = endant.CD_PACIENTE
   AND transf.CD_ENDERECO_ANTERIOR = endant.CD_ENDERECO
   AND endant.CD_LOCALIDADE        = locant.CD_LOCALIDADE
   AND transf.CD_AREA              = areatu.CD_AREA
   AND transf.CD_AREA_ANTERIOR     = areant.CD_AREA
   AND transf.CD_PACIENTE          = pacien.CD_PACIENTE
ORDER BY transf.DT_TRANSF,
         pacien.NM_PACIENTE,
         transf.CD_TRANSF_HOCA
)
ORDER BY DT_TRANSF
