--R_DEVOLUCOES_FORNECEDOR
SELECT * FROM (
SELECT DBAMV.FORNECEDOR.CD_FORNECEDOR,
       DBAMV.FORNECEDOR.NM_FORNECEDOR,
       DBAMV.ATENDIME_HCARE.CD_ATENDIMENTO,
       DBAMV.PACIENTE.CD_PACIENTE,
       DBAMV.PACIENTE.NM_PACIENTE,
       DBAMV.ORDEM_DEVOLUCAO.CD_ORDEM_DEVOLUCAO,
       DBAMV.ORDEM_DEVOLUCAO.DT_DEVOLUCAO,
       DBAMV.ORDEM_DEVOLUCAO.USUARIO,
       DBAMV.CLASSE.DS_CLASSE,
       DBAMV.PRODUTO.CD_PRODUTO,
       DBAMV.PRODUTO.DS_PRODUTO,
       DBAMV.UNI_PRO.DS_UNIDADE,
       DBAMV.ITORDEM_DEVOLUCAO.QT_DEVOLVIDA
  FROM DBAMV.ATENDIME,
       DBAMV.ATENDIME_HCARE,
       DBAMV.PRODUTO,
       DBAMV.CLASSE,
       DBAMV.PACIENTE,
       DBAMV.UNI_PRO,
       DBAMV.FORNECEDOR,
       DBAMV.ORDEM_DEVOLUCAO,
       DBAMV.ITORDEM_DEVOLUCAO
 WHERE DBAMV.ATENDIME.DT_ALTA IS NULL
  AND Exists ( SELECT Empresa_Produto.Cd_Produto FROM Dbamv.Empresa_Produto WHERE Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA AND Empresa_Produto.Cd_Produto = Produto.Cd_Produto )
  AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
  AND DBAMV.ATENDIME_HCARE.CD_ATENDIMENTO        = DBAMV.ATENDIME.CD_ATENDIMENTO
   AND DBAMV.ATENDIME.CD_PACIENTE                 = DBAMV.PACIENTE.CD_PACIENTE
   AND DBAMV.UNI_PRO.CD_PRODUTO                   = DBAMV.PRODUTO.CD_PRODUTO
   AND DBAMV.ORDEM_DEVOLUCAO.CD_ATENDIMENTO       = DBAMV.ATENDIME.CD_ATENDIMENTO
   AND DBAMV.ORDEM_DEVOLUCAO.CD_FORNECEDOR        = DBAMV.FORNECEDOR.CD_FORNECEDOR
   AND DBAMV.ITORDEM_DEVOLUCAO.CD_ORDEM_DEVOLUCAO = DBAMV.ORDEM_DEVOLUCAO.CD_ORDEM_DEVOLUCAO
   AND DBAMV.ITORDEM_DEVOLUCAO.CD_PRODUTO         = DBAMV.PRODUTO.CD_PRODUTO
   AND DBAMV.PRODUTO.CD_CLASSE                    = DBAMV.CLASSE.CD_CLASSE
   AND DBAMV.PRODUTO.CD_ESPECIE                   = DBAMV.CLASSE.CD_ESPECIE
ORDER BY DBAMV.FORNECEDOR.CD_FORNECEDOR,
         DBAMV.PACIENTE.NM_PACIENTE,
         DBAMV.ORDEM_DEVOLUCAO.CD_ORDEM_DEVOLUCAO,
         DBAMV.CLASSE.DS_CLASSE,
         DBAMV.PRODUTO.DS_PRODUTO
)
ORDER BY NM_FORNECEDOR,CD_FORNECEDOR,
NM_PACIENTE,CD_PACIENTE,CD_ATENDIMENTO,
CD_ORDEM_DEVOLUCAO,DT_DEVOLUCAO,USUARIO,
DS_CLASSE
