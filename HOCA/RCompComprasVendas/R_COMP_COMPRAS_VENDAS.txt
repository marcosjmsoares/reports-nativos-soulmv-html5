--R_COMP_COMPRAS_VENDAS
SELECT * FROM (
SELECT MVTO_ESTOQUE.CD_ATENDIMENTO           CD_ATENDIMENTO
      ,PACIENTE.NM_PACIENTE                  NM_PACIENTE
      ,ATENDIME.CD_CONVENIO                CD_CONVENIO
      ,ATENDIME.CD_CON_PLA                  CD_CON_PLA
      ,ESPECIE.CD_ESPECIE                    CD_ESPECIE
      ,ESPECIE.DS_ESPECIE                    DS_ESPECIE
      ,ITMVTO_ESTOQUE.CD_PRODUTO             CD_PRODUTO
      ,PRODUTO.DS_PRODUTO                    DS_PRODUTO
      ,PRODUTO.CD_PRO_FAT
      ,COUNT(ITMVTO_ESTOQUE.QT_MOVIMENTACAO) QT_MOVIMENTACAO
      ,VERIF_VL_CUSTO_MEDIO( ITMVTO_ESTOQUE.CD_PRODUTO
                            ,MVTO_ESTOQUE.DT_MVTO_ESTOQUE
                            ,'M'
                            ,PRODUTO.VL_CUSTO_MEDIO
                            ,MVTO_ESTOQUE.HR_MVTO_ESTOQUE ) * UNI_PRO.VL_FATOR VL_CUSTO_UNIT
  FROM DBAMV.MVTO_ESTOQUE
      ,DBAMV.ATENDIME
      ,DBAMV.PACIENTE
      ,DBAMV.ITMVTO_ESTOQUE
      ,DBAMV.PRODUTO
      ,DBAMV.ESPECIE
      ,DBAMV.UNI_PRO
     ,DBAMV.ESTOQUE 
 WHERE ATENDIME.CD_ATENDIMENTO        = :CAMPO_CD_ATENDIMENTO
   AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
   AND MVTO_ESTOQUE.DT_MVTO_ESTOQUE BETWEEN :CAMPO_DT_INICIO AND 
   AND ATENDIME.CD_ATENDIMENTO        = MVTO_ESTOQUE.CD_ATENDIMENTO
   AND PACIENTE.CD_PACIENTE           = ATENDIME.CD_PACIENTE
   AND ITMVTO_ESTOQUE.CD_MVTO_ESTOQUE = MVTO_ESTOQUE.CD_MVTO_ESTOQUE
   AND MVTO_ESTOQUE.CD_ESTOQUE  = ESTOQUE.CD_ESTOQUE
   AND ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
   AND PRODUTO.CD_PRODUTO             = ITMVTO_ESTOQUE.CD_PRODUTO
   AND ESPECIE.CD_ESPECIE             = PRODUTO.CD_ESPECIE
   AND UNI_PRO.CD_UNI_PRO             = ITMVTO_ESTOQUE.CD_UNI_PRO
GROUP BY MVTO_ESTOQUE.CD_ATENDIMENTO
        ,PACIENTE.NM_PACIENTE
        ,ATENDIME.CD_CONVENIO
        ,ATENDIME.CD_CON_PLA
        ,ESPECIE.CD_ESPECIE
        ,ESPECIE.DS_ESPECIE
        ,ITMVTO_ESTOQUE.CD_PRODUTO
        ,PRODUTO.DS_PRODUTO
        ,PRODUTO.CD_PRO_FAT
        ,VERIF_VL_CUSTO_MEDIO( ITMVTO_ESTOQUE.CD_PRODUTO
                              ,MVTO_ESTOQUE.DT_MVTO_ESTOQUE
                              ,'M'
                              ,PRODUTO.VL_CUSTO_MEDIO
                              ,MVTO_ESTOQUE.HR_MVTO_ESTOQUE ) * UNI_PRO.VL_FATOR
ORDER BY MVTO_ESTOQUE.CD_ATENDIMENTO
        ,PACIENTE.NM_PACIENTE
        ,ESPECIE.CD_ESPECIE
        ,ESPECIE.DS_ESPECIE
        ,ITMVTO_ESTOQUE.CD_PRODUTO
        ,PRODUTO.DS_PRODUTO
)
ORDER BY CD_ATENDIMENTO,NM_PACIENTE,CD_ESPECIE,DS_ESPECIE
