--R_PROCED_CUIDADOS
SELECT DISTINCT PROCEDIMENTOS_CUIDADOS.CD_PROCEDIMENTOS_CUIDADOS
                               ,PROCEDIMENTOS_CUIDADOS.DS_PROCEDIMENTOS_CUIDADOS
                               ,PROCEDIMENTOS_CUIDADOS.CD_TIP_FRE
 ,DECODE(PROCEDIMENTOS_CUIDADOS.TP_PROCEDIMENTOS_CUIDADOS, 'P', 'Procedimento'
                                                                                                                                  , 'Monitoração') TP_PROCEDIMENTOS_CUIDADOS
FROM  DBAMV.PROCEDIMENTOS_CUIDADOS
WHERE PROCEDIMENTOS_CUIDADOS.CD_PROCEDIMENTOS_CUIDADOS =  PROCEDIMENTOS_CUIDADOS.CD_PROCEDIMENTOS_CUIDADOS
ORDER BY 1
