--R_EST_INTER_AREA
SELECT area.CD_AREA,
       area.DS_AREA,
       COUNT(*) INTERNADOS
  FROM DBAMV.ATENDIME aten,
       DBAMV.ATENDIME_HCARE hcar,
       DBAMV.AREA area
 WHERE aten.DT_ATENDIMENTO between :CAMPO_DATINI AND 
   AND  aten.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
   AND aten.TP_ATENDIMENTO = 'H'
   AND hcar.CD_ATENDIMENTO = aten.CD_ATENDIMENTO
   AND hcar.CD_AREA        = area.CD_AREA
GROUP BY area.DS_AREA ,area.CD_AREA
