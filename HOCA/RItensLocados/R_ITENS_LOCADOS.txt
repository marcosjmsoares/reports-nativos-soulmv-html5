--R_ITENS_LOCADOS
SELECT * FROM (
SELECT DISTINCT ATENDIME.CD_ATENDIMENTO
      ,PACIENTE.NM_PACIENTE
      ,ITENS_LOCADOS.CD_PRODUTO
      ,PRODUTO.DS_PRODUTO
      ,UNI_PRO.DS_UNIDADE
      ,ITENS_LOCADOS.DT_LOCACAO
      ,ITENS_LOCADOS.QT_LOCADA
FROM DBAMV.ATENDIME
    ,DBAMV.PACIENTE
    ,DBAMV.V_ITENS_LOCADOS ITENS_LOCADOS
    ,DBAMV.PRODUTO
    ,DBAMV.UNI_PRO
WHERE  ATENDIME.CD_PACIENTE                = PACIENTE.CD_PACIENTE
   AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
   AND Exists ( SELECT Empresa_Produto.Cd_Produto FROM Dbamv.Empresa_Produto WHERE Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA AND Empresa_Produto.Cd_Produto = Produto.Cd_Produto )
   AND ATENDIME.CD_ATENDIMENTO           = ITENS_LOCADOS.CD_ATENDIMENTO
   AND ITENS_LOCADOS.CD_PRODUTO        = PRODUTO.CD_PRODUTO
   AND ITENS_LOCADOS.CD_UNI_PRO          = UNI_PRO.CD_UNI_PRO
:CAMPO_CF_PACIENTE
)
ORDER BY CD_ATENDIMENTO,NM_PACIENTE
