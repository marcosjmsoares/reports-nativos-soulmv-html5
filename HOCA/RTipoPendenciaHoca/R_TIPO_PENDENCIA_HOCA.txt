--R_TIPO_PENDENCIA_HOCA
SELECT DISTINCT TIPO_PENDENCIA.CD_TIPO_PENDENCIA
               ,TIPO_PENDENCIA.DS_TIPO_PENDENCIA
               ,SETOR.NM_SETOR
               ,DECODE(TIPO_PENDENCIA.SN_IMPEDE_ATENDIMENTO, 'S', 'Sim', 'Não') SN_IMPEDE_ATENDIMENTO
               ,DECODE(TIPO_PENDENCIA.SN_QUADRO_PENDENCIA, 'S', 'Sim', 'Não') SN_QUADRO_PENDENCIA
FROM  DBAMV.TIPO_PENDENCIA
     ,DBAMV.SETOR
WHERE TIPO_PENDENCIA.CD_SETOR = SETOR.CD_SETOR(+)
ORDER BY 1
