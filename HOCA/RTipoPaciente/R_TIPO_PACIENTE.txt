--R_TIPO_PACIENTE
SELECT * FROM (
SELECT DISTINCT TIPO_PACIENTE.DS_TIPO_PACIENTE
      ,ITENS_EXAMINAR.DS_ITENS_EXAMINAR
FROM DBAMV.TIPO_PACIENTE
    ,DBAMV.TIPO_PACIENTE_ITENS_EXAMINAR
    ,DBAMV.ITENS_EXAMINAR
WHERE TIPO_PACIENTE.CD_TIPO_PACIENTE = TIPO_PACIENTE_ITENS_EXAMINAR.CD_TIPO_PACIENTE
  AND TIPO_PACIENTE_ITENS_EXAMINAR.CD_ITENS_EXAMINAR = ITENS_EXAMINAR.CD_ITENS_EXAMINAR
:CAMPO_CF_TIPO_PACIENTE
)
ORDER BY DS_TIPO_PACIENTE,DS_ITENS_EXAMINAR
