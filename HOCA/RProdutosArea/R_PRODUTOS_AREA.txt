--R_PRODUTOS_AREA
SELECT * FROM (
SELECT area.CD_AREA,
       area.DS_AREA,
       clas.DS_CLASSE,
       prod.CD_PRODUTO,
       prod.DS_PRODUTO,
       SUM(itso.QT_SOLICITADO) QUANT
  FROM DBAMV.ATENDIME aten,
       DBAMV.ATENDIME_HCARE hcar,
       DBAMV.AREA area,
       DBAMV.SOLSAI_PRO sols,
       DBAMV.ITSOLSAI_PRO itso,
       DBAMV.PRODUTO prod,
       DBAMV.CLASSE clas,
       DBAMV.ESTOQUE
 WHERE aten.TP_ATENDIMENTO = 'H'
   AND  ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
   AND aten.DT_ALTA IS NULL
   AND hcar.CD_ATENDIMENTO = aten.CD_ATENDIMENTO
   AND hcar.CD_AREA        = area.CD_AREA
   AND hcar.CD_ATENDIMENTO = sols.CD_ATENDIMENTO
   AND itso.CD_SOLSAI_PRO  = sols.CD_SOLSAI_PRO
   AND itso.CD_PRODUTO     = prod.CD_PRODUTO
   AND prod.CD_CLASSE      = clas.CD_CLASSE
   AND prod.CD_ESPECIE     = clas.CD_ESPECIE
   AND sols.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
   AND ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY area.DS_AREA,
         clas.DS_CLASSE,
         prod.DS_PRODUTO,
         area.CD_AREA,
         prod.CD_PRODUTO
)
ORDER BY DS_AREA,CD_AREA,
DS_CLASSE
