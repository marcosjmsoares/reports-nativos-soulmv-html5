--R_RENTABILIDADE_CONTA
SELECT * FROM (
SELECT REG_FAT.CD_ATENDIMENTO
      ,PACIENTE.NM_PACIENTE
      ,REG_FAT.CD_CONVENIO
      ,CONVENIO.NM_CONVENIO
      ,REG_FAT.CD_CON_PLA
      ,CON_PLA.DS_CON_PLA
      ,REG_FAT.CD_REMESSA
      ,REG_FAT.CD_REG_FAT
      ,REG_FAT.DT_FINAL
      ,GRU_FAT.CD_GRU_FAT
      ,GRU_FAT.DS_GRU_FAT
      ,GRU_FAT.TP_GRU_FAT
      ,ITREG_FAT.CD_PRO_FAT
      ,PRO_FAT.DS_PRO_FAT
      ,PRO_FAT.DS_UNIDADE
      ,SUM(ITREG_FAT.QT_LANCAMENTO)   QT_LANCAMENTO
      ,SUM(DECODE( NVL(ITREG_FAT.SN_PERTENCE_PACOTE,'N'), 'S',0, ITREG_FAT.VL_TOTAL_CONTA ) )  VL_TOTAL_CONTA
  FROM DBAMV.REG_FAT
      ,DBAMV.ORCAMENTO_HOCA ORCAMENTO_HOCA
      ,DBAMV.ATENDIME
      ,DBAMV.PACIENTE
      ,DBAMV.CONVENIO
      ,DBAMV.CON_PLA
      ,DBAMV.ITREG_FAT
      ,DBAMV.PRO_FAT
      ,DBAMV.GRU_PRO
      ,DBAMV.GRU_FAT
 WHERE ATENDIME.CD_ATENDIMENTO = REG_FAT.CD_ATENDIMENTO
   AND PACIENTE.CD_PACIENTE    = ORCAMENTO_HOCA.CD_PACIENTE
   AND PACIENTE.CD_PACIENTE    = ATENDIME.CD_PACIENTE
   AND CONVENIO.CD_CONVENIO    = REG_FAT.CD_CONVENIO
   AND CON_PLA.CD_CONVENIO     = CONVENIO.CD_CONVENIO
   AND CON_PLA.CD_CON_PLA      = REG_FAT.CD_CON_PLA
   AND ITREG_FAT.CD_REG_FAT    = REG_FAT.CD_REG_FAT
   AND PRO_FAT.CD_PRO_FAT      = ITREG_FAT.CD_PRO_FAT
   AND GRU_PRO.CD_GRU_PRO      = PRO_FAT.CD_GRU_PRO
   AND GRU_FAT.CD_GRU_FAT      = GRU_PRO.CD_GRU_FAT
  GROUP BY REG_FAT.CD_ATENDIMENTO
      ,PACIENTE.NM_PACIENTE
      ,REG_FAT.CD_CONVENIO
      ,CONVENIO.NM_CONVENIO
      ,REG_FAT.CD_CON_PLA
      ,CON_PLA.DS_CON_PLA
      ,REG_FAT.CD_REMESSA
      ,REG_FAT.CD_REG_FAT
      ,REG_FAT.DT_FINAL
      ,GRU_FAT.CD_GRU_FAT
      ,GRU_FAT.DS_GRU_FAT
      ,GRU_FAT.TP_GRU_FAT
      ,ITREG_FAT.CD_PRO_FAT
      ,PRO_FAT.DS_PRO_FAT
      ,PRO_FAT.DS_UNIDADE
)
ORDER BY CD_ATENDIMENTO,NM_PACIENTE,CD_CONVENIO,NM_CONVENIO,CD_CON_PLA,DS_CON_PLA,CD_REMESSA,CD_REG_FAT,DT_FINAL,
DS_GRU_FAT,CD_GRU_FAT,TP_GRU_FAT
