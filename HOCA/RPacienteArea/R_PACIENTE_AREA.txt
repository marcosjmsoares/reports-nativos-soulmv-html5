--R_PACIENTE_AREA
SELECT care.CD_AREA,
       area.DS_AREA,
       aten.CD_ATENDIMENTO,
       paci.CD_PACIENTE,
       paci.NM_PACIENTE,
       aten.DT_ATENDIMENTO,
       ende.NR_FONE,
       ende.NM_BAIRRO,
       loca.NM_LOCALIDADE,
       care.CD_PRESTADOR_ACO,
       pres.NM_PRESTADOR NM_PRESTADOR1,
       care.CD_ENFER_SUPER,
       pre1.NM_PRESTADOR,
       care.CD_AUX_ENFER,
       pre2.NM_PRESTADOR NM_PRESTADOR2
  FROM DBAMV.ATENDIME aten,
       DBAMV.PACIENTE paci,
       DBAMV.ATENDIME_HCARE care,
       DBAMV.ENDERECO ende,
       DBAMV.CEP_LOCALIDADES loca,
       DBAMV.AREA area,
       DBAMV.PRESTADOR pres,
       DBAMV.PRESTADOR pre1,
       DBAMV.PRESTADOR pre2
 WHERE aten.TP_ATENDIMENTO   = 'H'
   AND  ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
   AND aten.CD_PACIENTE      = paci.CD_PACIENTE
   AND care.CD_ATENDIMENTO   = aten.CD_ATENDIMENTO
   AND care.CD_PACIENTE      = ende.CD_PACIENTE
   AND care.CD_ENDERECO      = ende.CD_ENDERECO
   AND ende.CD_LOCALIDADE    = loca.CD_LOCALIDADE(+)
   AND care.CD_PRESTADOR_ACO = pres.CD_PRESTADOR
   AND care.CD_AREA          = area.CD_AREA
   AND care.CD_ENFER_SUPER   = pre1.CD_PRESTADOR(+)
   AND care.CD_AUX_ENFER     = pre2.CD_PRESTADOR(+)
  and aten.dt_alta is null
ORDER BY area.DS_AREA,
         paci.NM_PACIENTE
