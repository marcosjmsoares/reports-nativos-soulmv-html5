--R_PENDENCIA_PACIENTE_HC
SELECT * FROM (
SELECT tipo.DS_TIPO_PENDENCIA,
       paci.CD_PACIENTE,
       paci.NM_PACIENTE,
       paci.NR_FONE,
       aten.CD_ATENDIMENTO,
       aten.DT_ATENDIMENTO,
       pend.NM_RESPONSAVEL
FROM   DBAMV.ATENDIME aten,
       DBAMV.PENDENCIA_ATENDIMENTO pend,
       DBAMV.TIPO_PENDENCIA tipo,
       DBAMV.PACIENTE paci
WHERE  pend.DT_BAIXA_PENDENCIA is null
      AND  ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
      AND  pend.CD_ATENDIMENTO      = aten.CD_ATENDIMENTO
      AND  aten.TP_ATENDIMENTO='H'
      AND  aten.DT_ALTA is null
      AND  pend.CD_TIPO_PENDENCIA = tipo.CD_TIPO_PENDENCIA
      AND  aten.CD_PACIENTE               = paci.CD_PACIENTE
ORDER BY tipo.DS_TIPO_PENDENCIA,
         paci.NM_PACIENTE
)
