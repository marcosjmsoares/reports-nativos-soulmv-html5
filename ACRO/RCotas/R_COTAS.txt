--R_COTAS
SELECT cd_unid_int,
               cd_tp_pecas,
               cd_comp_kit,
               qt_quantidade,
               Descricao,
               ds_unid_int
    FROM(
SELECT acro_cotas.cd_unid_int                        cd_unid_int,
               acro_det_cotas.cd_tp_pecas              cd_tp_pecas,
               acro_det_cotas.cd_comp_kit               cd_comp_kit,
               acro_det_cotas.qt_quantidade             qt_quantidade,
               NVL(tp_pecas.ds_tp_pecas, kit.ds_kit) Descricao,
               unid_int.ds_unid_int                              ds_unid_int
   FROM dbamv.acro_cotas,
              dbamv.acro_det_cotas,
              dbamv.unid_int,
              dbamv.tp_pecas,
              dbamv.kit
WHERE acro_cotas.cd_cotas              = acro_det_cotas.cd_cotas
      AND acro_cotas.cd_unid_int          = unid_int.cd_unid_int
      AND acro_det_cotas.cd_tp_pecas = tp_pecas.cd_tp_pecas (+)
      AND acro_det_cotas.cd_comp_kit = kit.cd_comp_kit (+)
:CAMPO_CF_WHERE_TIPO
)
