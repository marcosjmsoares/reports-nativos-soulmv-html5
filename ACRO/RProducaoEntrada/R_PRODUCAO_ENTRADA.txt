--R_PRODUCAO_ENTRADA
SELECT VDIC_PRODUCAO_ENTRADA.DATA_ENTRADA,
       VDIC_PRODUCAO_ENTRADA.ESTOQUE,
       VDIC_PRODUCAO_ENTRADA.DESCRICAO_ESTOQUE,
       sum(PESO_KG * QUANTIDADE) sum_PESO_KG_QUANTIDADE,
       (select user from sys.dual) select_user_from_sys_dua,
       (select hospital.ds_razao from dbamv.hospital where cd_multi_empresa = dbamv.pkg_mv2000.le_empresa) select_hospital_ds_razao,
       (select hospital.ds_rodape_reports from dbamv.hospital where  cd_multi_empresa = dbamv.pkg_mv2000.le_empresa) select_hospital_ds_rodap
FROM
     DBAMV.VDIC_PRODUCAO_ENTRADA VDIC_PRODUCAO_ENTRADA
WHERE
       VDIC_PRODUCAO_ENTRADA.DATA_ENTRADA BETWEEN :CAMPO_DATA_ENTRADA_INI GROUP BY VDIC_PRODUCAO_ENTRADA.DATA_ENTRADA,
         VDIC_PRODUCAO_ENTRADA.ESTOQUE,
         VDIC_PRODUCAO_ENTRADA.DESCRICAO_ESTOQUE
ORDER BY VDIC_PRODUCAO_ENTRADA.DATA_ENTRADA
