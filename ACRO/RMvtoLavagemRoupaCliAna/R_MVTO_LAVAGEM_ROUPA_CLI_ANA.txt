--R_MVTO_LAVAGEM_ROUPA_CLI_ANA
SELECT LAVAGEM_ROUPA_CLIENTE.CD_LAVAGEM_ROUPA_CLIENTE
     , LAVAGEM_ROUPA_CLIENTE.DH_ENTRADA
     , DECODE(LAVAGEM_ROUPA_CLIENTE.TP_SITUACAO
            , 'R',:CAMPO_TRAD_RECEB_LAV
            , 'L',:CAMPO_TRAD_ENV_LAVAND
            , 'F',:CAMPO_TRAD_LAV_FIN
            , 'E',:CAMPO_TRAD_ENTREGUE
            , 'C',:CAMPO_TRAD_CANCELADA
            , LAVAGEM_ROUPA_CLIENTE.TP_SITUACAO) AS TP_SITUACAO
     , DECODE(LAVAGEM_ROUPA_CLIENTE.TP_SITUACAO
            , 'R',DH_ENTRADA
            , 'L',DH_ENVIO_LAVAGEM
            , 'F',DH_CONCLUSAO_LAVAGEM
            , 'E',DH_ENTREGA_CLIENTE
            , 'C',DH_CANCELAMENTO
            , LAVAGEM_ROUPA_CLIENTE.DH_ENTRADA) AS DH_SITUACAO
     , FORNECEDOR.NM_FORNECEDOR                 AS NM_CLIENTE
     , LAVANDERIA.DS_LAVANDERIA
     , LAVAGEM_ROUPA_CLIENTE.QT_PESO_TOTAL_ROUPA
     , IT_LAVAGEM_ROUPA_CLIENTE.CD_PRODUTO
     , PRODUTO.DS_PRODUTO
     , IT_LAVAGEM_ROUPA_CLIENTE.DS_TAMANHO      AS DS_TAMANHO_PRODUTO
     , IT_LAVAGEM_ROUPA_CLIENTE.QT_MOVIMENTO    AS QT_MOVIMENTO_PRODUTO
     , IT_LAVAGEM_ROUPA_CLIENTE.VL_UNITARIO     AS VL_UNITARIO_PRODUTO
     , IT_LAVAGEM_ROUPA_CLIENTE.VL_TOTAL        AS VL_TOTAL_PRODUTO
  FROM DBAMV.LAVAGEM_ROUPA_CLIENTE
     , DBAMV.IT_LAVAGEM_ROUPA_CLIENTE
     , DBAMV.FORNECEDOR
     , DBAMV.LAVANDERIA
     , DBAMV.PRODUTO
     , DBASGU.USUARIOS
 WHERE LAVAGEM_ROUPA_CLIENTE.CD_CLIENTE               = FORNECEDOR.CD_FORNECEDOR
   AND LAVANDERIA.CD_LAVANDERIA(+)                    = LAVAGEM_ROUPA_CLIENTE.CD_LAVANDERIA
   AND LAVAGEM_ROUPA_CLIENTE.CD_USU_CONTA_ROUPA       = USUARIOS.CD_USUARIO
   AND LAVAGEM_ROUPA_CLIENTE.CD_LAVAGEM_ROUPA_CLIENTE = IT_LAVAGEM_ROUPA_CLIENTE.CD_LAVAGEM_ROUPA_CLI
   AND IT_LAVAGEM_ROUPA_CLIENTE.CD_PRODUTO            = PRODUTO.CD_PRODUTO
GROUP BY LAVAGEM_ROUPA_CLIENTE.CD_LAVAGEM_ROUPA_CLIENTE
       , LAVAGEM_ROUPA_CLIENTE.DH_ENTRADA
       , LAVAGEM_ROUPA_CLIENTE.TP_SITUACAO
       , LAVAGEM_ROUPA_CLIENTE.DH_ENTRADA
       , LAVAGEM_ROUPA_CLIENTE.DH_ENVIO_LAVAGEM
       , LAVAGEM_ROUPA_CLIENTE.DH_CONCLUSAO_LAVAGEM
       , LAVAGEM_ROUPA_CLIENTE.DH_ENTREGA_CLIENTE
       , LAVAGEM_ROUPA_CLIENTE.DH_CANCELAMENTO
       , FORNECEDOR.NM_FORNECEDOR
       , LAVANDERIA.DS_LAVANDERIA
       , LAVAGEM_ROUPA_CLIENTE.QT_PESO_TOTAL_ROUPA
       , IT_LAVAGEM_ROUPA_CLIENTE.CD_PRODUTO
       , PRODUTO.DS_PRODUTO
       , IT_LAVAGEM_ROUPA_CLIENTE.DS_TAMANHO
       , IT_LAVAGEM_ROUPA_CLIENTE.QT_MOVIMENTO
       , IT_LAVAGEM_ROUPA_CLIENTE.VL_UNITARIO
       , IT_LAVAGEM_ROUPA_CLIENTE.VL_TOTAL
