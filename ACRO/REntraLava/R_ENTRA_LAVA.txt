--R_ENTRA_LAVA
SELECT * FROM (
SELECT Ent_roupa.cd_fornecedor cd_fornecedor,
       ent_roupa.cd_transacao          cd_transacao,
       DECODE( NVL(Ent_roupa.tp_int_ext, 'I'), 'I', 'Interna', 'Externa') tp_int_ext,
       Nvl(Ent_roupa.Cd_usuario_Inc, Ent_roupa.cd_usuario_alt) cd_usuario,
       SYSDATE                            Data_atual,
       Ent_det_roupa.cd_tp_pecas   cd_tp_pecas,
       Ent_det_roupa.qt_peso qt_peso,
       Ent_det_roupa.qt_quantidade qt_quantidade,
       tp_pecas.ds_tp_pecas        ds_tp_pecas,
       usuarios.nm_usuario            nm_usuario,
       Fornecedor.nm_fornecedor nm_fornecedor
  FROM dbamv.ent_roupa,
       dbamv.ent_det_roupa,
       dbamv.tp_pecas,
       dbasgu.usuarios,
       dbamv.fornecedor
 WHERE Ent_roupa.cd_transacao = Ent_det_roupa.cd_transacao
       AND tp_pecas.cd_tp_pecas   = ent_det_roupa.cd_tp_pecas
       AND usuarios.cd_usuario = NVL(ent_roupa.cd_usuario_alt, ent_roupa.cd_usuario_inc)
       AND fornecedor.cd_fornecedor = ent_roupa.cd_fornecedor
       AND ent_roupa.tp_tipo_es = 'E'
:CAMPO_CF_WHERE_TRANSACAO
)
ORDER BY CD_FORNECEDOR,DATA_ATUAL,CD_USUARIO,NM_USUARIO,TP_INT_EXT,NM_FORNECEDOR,CD_TRANSACAO
