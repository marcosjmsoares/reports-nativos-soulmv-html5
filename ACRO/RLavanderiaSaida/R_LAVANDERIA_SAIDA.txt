--R_LAVANDERIA_SAIDA
SELECT
       VDIC_ENTRADA_ROUPA_ESTOQUE.CODIGO_FORNECEDOR,
       VDIC_ENTRADA_ROUPA_ESTOQUE.NOME_FORNECEDOR,
       VDIC_ENTRADA_ROUPA_ESTOQUE.NUMERO_REGISTRO,
       VDIC_ENTRADA_ROUPA_ESTOQUE.SETOR,
       VDIC_ENTRADA_ROUPA_ESTOQUE.NOME_SETOR,
       VDIC_ENTRADA_ROUPA_ESTOQUE.QUANTIDADE_DE_PESO,
       VDIC_ENTRADA_ROUPA_ESTOQUE.USUARIO,
       VDIC_ENTRADA_ROUPA_ESTOQUE.NOME_USUARIO,
       VDIC_ENTRADA_ROUPA_ESTOQUE.TIPO_DE_LAVANDERIA,
       (select user from sys.dual) select_user_from_sys_dua,
       (select hospital.ds_razao from dbamv.hospital where cd_multi_empresa = dbamv.pkg_mv2000.le_empresa) select_hospital_ds_razao,
       (select hospital.ds_rodape_reports from dbamv.hospital where  cd_multi_empresa = dbamv.pkg_mv2000.le_empresa) select_hospital_ds_rodap
FROM
     DBAMV.VDIC_ENTRADA_ROUPA_ESTOQUE VDIC_ENTRADA_ROUPA_ESTOQUE
WHERE
      VDIC_ENTRADA_ROUPA_ESTOQUE.DATA_SAIDA BETWEEN :CAMPO_DATA_SAIDA_INI ORDER BY
         VDIC_ENTRADA_ROUPA_ESTOQUE.NUMERO_REGISTRO
