--R_PRODUCAO_SAIDA
SELECT VDIC_PRODUCAO_SAIDA.CODIGO_COMPOSICAO_KIT,
       VDIC_PRODUCAO_SAIDA.CODIGO_TIPO_PECAS,
       VDIC_PRODUCAO_SAIDA.DESCRICAO,
       VDIC_PRODUCAO_SAIDA.QUANTIDADE,
       VDIC_PRODUCAO_SAIDA.PESO_KG,
       VDIC_PRODUCAO_SAIDA.DATA_SAIDA,
       VDIC_PRODUCAO_SAIDA.DESCRICAO_ESTOQUE,
       VDIC_PRODUCAO_SAIDA.NOME_SETOR,
       VDIC_PRODUCAO_SAIDA.UNID_INTERNACAO,
       VDIC_PRODUCAO_SAIDA.DESCRICAO_UNID_INT,
       VDIC_PRODUCAO_SAIDA.ESTOQUE,
       VDIC_PRODUCAO_SAIDA.SETOR,
       (select hospital.ds_rodape_reports from dbamv.hospital where  cd_multi_empresa = dbamv.pkg_mv2000.le_empresa) select_hospital_ds_rodap,
       (select user from sys.dual) select_user_from_sys_dua,
       (select hospital.ds_razao from dbamv.hospital where cd_multi_empresa = dbamv.pkg_mv2000.le_empresa) select_hospital_ds_razao,
       (PESO_KG * QUANTIDADE) PESO_KG_QUANTIDADE,
       (PESO_KG) PESO_KG_2,
       (TRUNC(VDIC_PRODUCAO_SAIDA.DATA_SAIDA)) TRUNC_VDIC_PRODUCAO_SAID
FROM DBAMV.VDIC_PRODUCAO_SAIDA VDIC_PRODUCAO_SAIDA
WHERE
    trunc(VDIC_PRODUCAO_SAIDA.DATA_SAIDA) BETWEEN :CAMPO_DATA_SAIDA_INI ORDER BY VDIC_PRODUCAO_SAIDA.ESTOQUE,
         VDIC_PRODUCAO_SAIDA.SETOR,
         VDIC_PRODUCAO_SAIDA.UNID_INTERNACAO
