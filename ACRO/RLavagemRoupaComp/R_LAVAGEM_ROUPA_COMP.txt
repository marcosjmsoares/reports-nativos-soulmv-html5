SELECT  lavagem_roupa_cliente.CD_LAVAGEM_ROUPA_CLIENTE
      , lavagem_roupa_cliente.TP_SITUACAO
      , lavagem_roupa_cliente.CD_CLIENTE
      , lavagem_roupa_cliente.CD_LAVANDERIA
      , lavagem_roupa_cliente.CD_SETOR_CLIENTE
      , lavagem_roupa_cliente.QT_PESO_TOTAL_ROUPA
      , lavagem_roupa_cliente.DS_OBSERVACAO
  FROM dbamv.lavagem_roupa_cliente
    WHERE CD_LAVAGEM_ROUPA_CLIENTE = :CAMPO_CD_LAVAGEM_ROUPA_CLIENTE
