--R_COTAS_SETOR
SELECT * FROM (
SELECT cd_setor,
               cd_tp_pecas,
               cd_comp_kit,
               qt_quantidade,
               nm_setor,
               descricao
FROM(
            SELECT acro_cotas_setor.cd_setor                        cd_setor,
                           acro_det_cotas_setor.cd_tp_pecas          cd_tp_pecas,
                           acro_det_cotas_setor.cd_comp_kit          cd_comp_kit,
                           acro_det_cotas_setor.qt_quantidade        qt_quantidade,
                           setor.nm_setor                                           nm_setor,
                           NVL(kit.ds_kit, tp_pecas.ds_tp_pecas)     descricao
               FROM dbamv.acro_cotas_setor,
                          dbamv.acro_det_cotas_setor,
                          dbamv.setor,
                          dbamv.tp_pecas,
                          dbamv.kit
            WHERE acro_cotas_setor.cd_cotas_setor     = acro_det_cotas_setor.cd_cotas_setor
                  AND acro_cotas_setor.cd_setor                = setor.cd_setor
                  AND acro_det_cotas_setor.cd_tp_pecas  = tp_pecas.cd_tp_pecas (+)
                  AND acro_det_cotas_setor.cd_comp_kit  = kit.cd_comp_kit (+)
            :CAMPO_CF_WHERE_TIPO
         )
:CAMPO_CF_WHERE_ORDEM
)
ORDER BY CD_SETOR,NM_SETOR
