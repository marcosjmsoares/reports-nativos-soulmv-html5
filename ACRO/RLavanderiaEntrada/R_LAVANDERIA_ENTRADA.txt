--R_LAVANDERIA_ENTRADA
SELECT VDIC_SAIDA_LAVANDERIA.NUMERO_REGISTRO,
       VDIC_SAIDA_LAVANDERIA.CODIGO_FORNECEDOR,
       VDIC_SAIDA_LAVANDERIA.FORNECEDOR,
       VDIC_SAIDA_LAVANDERIA.TIPO_DE_LAVANDERIA,
       VDIC_SAIDA_LAVANDERIA.CODIGO_TIPO_PECAS,
       VDIC_SAIDA_LAVANDERIA.PESO_ENVIADO_LAVANDERIA,
       VDIC_SAIDA_LAVANDERIA.DESCRICAO_PECAS,
       VDIC_SAIDA_LAVANDERIA.QUANTIDADE,
       VDIC_SAIDA_LAVANDERIA.USUARIO,
       VDIC_SAIDA_LAVANDERIA.NOME_USUARIO,
       (select user from sys.dual) select_user_from_sys_dua,
       (select hospital.ds_razao from dbamv.hospital where cd_multi_empresa = dbamv.pkg_mv2000.le_empresa) select_hospital_ds_razao,
       (select hospital.ds_rodape_reports from dbamv.hospital where  cd_multi_empresa = dbamv.pkg_mv2000.le_empresa) select_hospital_ds_rodap
FROM
     DBAMV.VDIC_SAIDA_LAVANDERIA VDIC_SAIDA_LAVANDERIA
WHERE
      trunc(VDIC_SAIDA_LAVANDERIA.DATA_ENTRADA) BETWEEN :CAMPO_DATA_ENTRADA_INI ORDER BY VDIC_SAIDA_LAVANDERIA.CODIGO_FORNECEDOR, VDIC_SAIDA_LAVANDERIA.DESCRICAO_PECAS
