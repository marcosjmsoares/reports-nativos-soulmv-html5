--R_TAB_KITS
SELECT * FROM (
SELECT KIT.CD_COMP_KIT                                 Codigo_Kit,
              DS_KIT                                                     Nome_Kit,
             decode(KIT.sn_ativo,'S','SIM','NAO')          Ativo,
              DET_KIT.CD_COMP_KIT                         CodKit_no_Det,
              DET_KIT.CD_TP_PECAS                         Codigo_da_Peca,
              TP_PECAS.DS_TP_PECAS                      Descricao_Peca,
              QT_PC_KIT                                               Quant_no_Kit
FROM DBAMV.KIT,
           DBAMV.DET_KIT,
           DBAMV.TP_PECAS
WHERE KIT.CD_COMP_KIT = DET_KIT.CD_COMP_KIT
      AND DET_KIT.CD_TP_PECAS=TP_PECAS.CD_TP_PECAS
	:CAMPO_CF_ATIVO
)
ORDER BY CODIGO_KIT,NOME_KIT,ATIVO
