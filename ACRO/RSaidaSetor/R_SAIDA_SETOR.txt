--R_SAIDA_SETOR
SELECT VDIC_SAIDA_ROUPA_ESTOQUE.ESTOQUE,
       VDIC_SAIDA_ROUPA_ESTOQUE.DESCRICAO_ESTOQUE,
       VDIC_SAIDA_ROUPA_ESTOQUE.CODIGO_COMPOSICAO_KIT,
       VDIC_SAIDA_ROUPA_ESTOQUE.CODIGO_TIPO_PECAS,
       VDIC_SAIDA_ROUPA_ESTOQUE.DESCRICAO,
       VDIC_SAIDA_ROUPA_ESTOQUE.QUANTIDADE,
       VDIC_SAIDA_ROUPA_ESTOQUE.NOME_USUARIO,
       VDIC_SAIDA_ROUPA_ESTOQUE.SETOR,
       VDIC_SAIDA_ROUPA_ESTOQUE.NOME_SETOR,
       VDIC_SAIDA_ROUPA_ESTOQUE.UNID_INTERNACAO,
       VDIC_SAIDA_ROUPA_ESTOQUE.DESCRICAO_UNID_INT,
       VDIC_SAIDA_ROUPA_ESTOQUE.DATA_SAIDA,
       (select user from sys.dual) select_user_from_sys_dua,
	(select hospital.ds_razao from dbamv.hospital where cd_multi_empresa = dbamv.pkg_mv2000.le_empresa) select_hospital_ds_razao,
       (select hospital.ds_rodape_reports from dbamv.hospital where  cd_multi_empresa = dbamv.pkg_mv2000.le_empresa) select_hospital_ds_rodap
FROM
     DBAMV.VDIC_SAIDA_ROUPA_ESTOQUE VDIC_SAIDA_ROUPA_ESTOQUE
WHERE
     VDIC_SAIDA_ROUPA_ESTOQUE.DATA_SAIDA BETWEEN :CAMPO_DATA_SAIDA_INI ORDER BY VDIC_SAIDA_ROUPA_ESTOQUE.ESTOQUE,
         VDIC_SAIDA_ROUPA_ESTOQUE.SETOR,
         VDIC_SAIDA_ROUPA_ESTOQUE.UNID_INTERNACAO
