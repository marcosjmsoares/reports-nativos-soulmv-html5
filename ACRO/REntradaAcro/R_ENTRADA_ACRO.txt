--R_ENTRADA_ACRO
SELECT * FROM (
SELECT  SYSDATE                     DATA_ATUAL,
        USUARIOS.NM_USUARIO         NM_USUARIO,
        ROUPA_ESTOQUE.CD_MVTO_EST   CD_MVTO_EST ,
        ROUPA_ESTOQUE.NR_NOTA_FISCAL   NR_NOTA_FISCAL ,
        DET_ESTOQUE.CD_TP_PECAS     CD_TP_PECAS,
        DET_ESTOQUE.QT_QUANTIDADE   QT_QUANTIDADE,
        TP_PECAS.DS_TP_PECAS        DS_TP_PECAS,
        'Peças'                         TP_RELATORIO
FROM DBAMV.ROUPA_ESTOQUE,
    DBASGU.USUARIOS,
    DBAMV.DET_ESTOQUE,
    DBAMV.TP_PECAS
WHERE USUARIOS.CD_USUARIO             = NVL(ROUPA_ESTOQUE.CD_USUARIO_ALT, ROUPA_ESTOQUE.CD_USUARIO_INC)
  AND ROUPA_ESTOQUE.TP_ENTRADA_SAIDA  = 'E'
  AND DET_ESTOQUE.CD_MVTO_EST         = ROUPA_ESTOQUE.CD_MVTO_EST
  AND DET_ESTOQUE.CD_TP_PECAS         = TP_PECAS.CD_TP_PECAS
UNION ALL
SELECT SYSDATE                  DATA_ATUAL,
    USUARIOS.NM_USUARIO         NM_USUARIO,
    ROUPA_ESTOQUE.CD_MVTO_EST   CD_MVTO_EST,
    ROUPA_ESTOQUE.NR_NOTA_FISCAL   NR_NOTA_FISCAL ,
    DET_ESTOQUE.CD_COMP_KIT     CD_COMP_KIT,
    DET_ESTOQUE.QT_QUANTIDADE   QT_QUANTIDADE,
    KIT.DS_KIT                  DS_KIT,
    'Kit'                         TP_RELATORIO
FROM DBAMV.ROUPA_ESTOQUE,
    DBASGU.USUARIOS,
    DBAMV.DET_ESTOQUE,
    DBAMV.KIT
WHERE USUARIOS.CD_USUARIO             = NVL(ROUPA_ESTOQUE.CD_USUARIO_ALT, ROUPA_ESTOQUE.CD_USUARIO_INC)
  AND ROUPA_ESTOQUE.TP_ENTRADA_SAIDA  = 'E'
  AND DET_ESTOQUE.CD_MVTO_EST         = ROUPA_ESTOQUE.CD_MVTO_EST
  AND DET_ESTOQUE.CD_COMP_KIT         =  KIT.CD_COMP_KIT
:CAMPO_CF_WHERE_MVTO_EST
)
ORDER BY DATA_ATUAL,NM_USUARIO,
TP_RELATORIO,CD_MVTO_EST
