--R_TIPO_PECAS
SELECT * FROM (
SELECT
tp_pecas.cd_tp_pecas                                       Codigo_Tp_Peca,
tp_pecas.ds_tp_pecas                                        Descricao_Tp_Peca,
tp_pecas.ds_tamanho                                         Tamanho_Peca,
tp_pecas.fator_custos                                         Fator_Custos,
tp_pecas.nr_peso                                                Peso_Peca,
decode(Tp_Pecas.SN_ATIVO,'S','SIM','NAO')    Sn_Ativo
FROM DbaMv.Tp_Pecas
WHERE sn_ativo  like decode (:CAMPO_ATIVO, '%', sn_ativo,  :CAMPO_ATIVO)
:CAMPO_CF_ORDEM
)
