--R_MVTO_LAVAGEM_ROUPA_CLI_SINT
SELECT CD_LAVAGEM_ROUPA_CLIENTE,
       DH_ENTRADA,
       DECODE (TP_SITUACAO,
               'R', :CAMPO_TRAD_RECEB_LAV,
               'L', :CAMPO_TRAD_ENV_LAVAND,
               'F', :CAMPO_TRAD_LAV_FIN,
               'E', :CAMPO_TRAD_ENTREGUE,
               'C', :CAMPO_TRAD_CANCELADA,
               TP_SITUACAO
              ) TP_SITUACAO,
       DECODE (TP_SITUACAO,
               'R', DH_ENTRADA,
               'L', DH_ENVIO_LAVAGEM,
               'F', DH_CONCLUSAO_LAVAGEM,
               'E', DH_ENTREGA_CLIENTE,
               'C', DH_CANCELAMENTO,
               DH_ENTRADA
              ) DH_SITUACAO,
       f.nm_fornecedor,
       l.ds_lavanderia,
       u.nm_usuario,
       QT_PESO_TOTAL_ROUPA,
       VL_TOTAL_LAVAGEM
  FROM dbamv.lavagem_roupa_cliente lrc,
       dbamv.it_lavagem_roupa_cliente ilrc,
       dbamv.fornecedor f,
       dbamv.lavanderia l,
       dbasgu.usuarios u
 WHERE lrc.cd_cliente = f.cd_fornecedor
   AND l.cd_lavanderia(+) = lrc.cd_lavanderia
   AND lrc.CD_USU_CONTA_ROUPA = u.cd_usuario
   AND lrc.cd_lavagem_roupa_cliente = ilrc.cd_lavagem_roupa_cli
GROUP BY CD_LAVAGEM_ROUPA_CLIENTE,
       DH_ENTRADA,
       tp_situacao,
       dh_entrada,dh_envio_lavagem,dh_conclusao_lavagem,dh_entrega_cliente,dh_cancelamento,
       f.nm_fornecedor,
       l.ds_lavanderia,
       u.nm_usuario,
       QT_PESO_TOTAL_ROUPA,
       VL_TOTAL_LAVAGEM
