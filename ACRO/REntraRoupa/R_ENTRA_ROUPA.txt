--R_ENTRA_ROUPA
SELECT CD_TP_ENTRADA Codigo,
               DS_ENTRADA       Descricao,
              decode(SN_ATIVO,'S','SIM','NAO')     Sn_Ativo
FROM DBAMV.TP_ENTRADA
WHERE SN_ATIVO like decode (:CAMPO_ATIVO, '%', SN_ATIVO,  :CAMPO_ATIVO)
