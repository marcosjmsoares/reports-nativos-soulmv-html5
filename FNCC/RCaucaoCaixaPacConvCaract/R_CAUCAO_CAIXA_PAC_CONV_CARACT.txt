--R_CAUCAO_CAIXA_PAC_CONV_CARACT
select initcap(p.nm_paciente)     nm_paciente
      ,co.nm_convenio             nm_convenio
      ,trunc(ca.dt_caucao)               dt_caucao
      ,sum(ca.vl_caucao)          vl_movimentacao
from   dbamv.mov_caixa  m
      ,dbamv.caucao     ca
      ,dbamv.atendime   a
      ,dbamv.paciente   p
      ,dbamv.convenio   co
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Co.Cd_Convenio
And  a.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And m.cd_caucao = ca.cd_caucao
and    ca.cd_atendimento = a.cd_atendimento
and    a.cd_paciente = p.cd_paciente
and    a.cd_convenio = co.cd_convenio
and    m.dt_cancelamento is null
and    m.cd_caucao is not null
and    m.tp_movimentacao = 'E'
and    m.tp_origem_mov = 'CA'
and    trunc(m.dt_movimentacao) = :CAMPO_DATA
group by p.nm_paciente, co.nm_convenio, trunc(ca.dt_caucao)
minus
select saida_nm_paciente
      ,saida_nm_convenio
      ,saida_dt_caucao
      ,sum(saida_vl_movimentacao) saida_vl_movimentacao
from (
select initcap(p.nm_paciente)     saida_nm_paciente
      ,co.nm_convenio             saida_nm_convenio
      ,trunc(ca.dt_caucao)               saida_dt_caucao
      ,sum(ca.vl_caucao)          saida_vl_movimentacao
from   dbamv.caucao     ca
      ,dbamv.mov_caixa  m
      ,dbamv.doc_caixa  d
      ,dbamv.atendime   a
      ,dbamv.paciente   p
      ,dbamv.convenio   co
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Co.Cd_Convenio
And  a.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And m.cd_doc_caixa = d.cd_doc_caixa
and    d.cd_caucao = ca.cd_caucao
and    ca.cd_atendimento = a.cd_atendimento
and    a.cd_paciente = p.cd_paciente
and    a.cd_convenio = co.cd_convenio
and    m.dt_cancelamento is null
and    m.tp_movimentacao = 'S'
and    m.tp_origem_mov in ('CA','DE')
and    trunc(m.dt_movimentacao) = :CAMPO_DATA
group by p.nm_paciente, co.nm_convenio, trunc(ca.dt_caucao)
union
select initcap(p.nm_paciente)     saida_nm_paciente
      ,co.nm_convenio             saida_nm_convenio
      ,trunc(ca.dt_caucao)               saida_dt_caucao
      ,sum(ca.vl_caucao)          saida_vl_movimentacao
from   dbamv.caucao     ca
      ,dbamv.mov_caixa  m
      ,dbamv.atendime   a
      ,dbamv.paciente   p
      ,dbamv.convenio   co
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Co.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and  a.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And m.cd_caucao = ca.cd_caucao
and    ca.cd_atendimento = a.cd_atendimento
and    a.cd_paciente = p.cd_paciente
and    a.cd_convenio = co.cd_convenio
and    not exists (select 'x' from dbamv.doc_caixa where doc_caixa.cd_caucao = ca.cd_caucao)
and    m.dt_cancelamento is null
and    m.tp_movimentacao = 'S'
and    m.tp_origem_mov in ('CA','DE')
and    trunc(m.dt_movimentacao) = :CAMPO_DATA
group by p.nm_paciente, co.nm_convenio, trunc(ca.dt_caucao)
)
group by saida_nm_paciente, saida_nm_convenio, saida_dt_caucao
