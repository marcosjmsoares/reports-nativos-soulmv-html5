        DBAMV.MOTIVO_LCTO ML,
       DBAMV.MOV_CAIXA   MC,

      MC.CD_MOTIVO_LCTO = ML.CD_MOTIVO_LCTO(+)  AND

	   ,Nvl(MC.CD_MOEDA, DBAMV.MOEDA_PADRAO(MC.CD_MULTI_EMPRESA))CD_MOEDA
    FROM
      DBAMV.MOTIVO_LCTO ML,
      DBAMV.MOV_CAIXA   MC,
      DBAMV.LOTE_CAIXA  LC
    WHERE
      MC.CD_MOTIVO_LCTO = ML.CD_MOTIVO_LCTO AND
      MC.CD_LOTE_CAIXA  = LC.CD_LOTE_CAIXA  AND
      MC.DT_CANCELAMENTO IS NULL  AND
      MC.TP_ORIGEM_MOV  IN  ('DD', 'PA')     and  :CAMPO_TIPO = 'D'  and :CAMPO_CHECK_SEMMOVIMENTO = 'N' and
      trunc(mc.dt_movimentacao)             between :CAMPO_DATA_INIC UNION ALL
     SELECT
        LC.CD_CAIXA
       ,ML.CD_MOTIVO_LCTO
       ,ML.DS_MOTIVO_LCTO
       ,0 	   RECEITA

    FROM
      DBAMV.MOTIVO_LCTO ML,
      DBAMV.MOV_CAIXA   MC,
      DBAMV.LOTE_CAIXA  LC
    WHERE
      MC.CD_MOTIVO_LCTO = ML.CD_MOTIVO_LCTO(+)  AND
      MC.CD_LOTE_CAIXA  = LC.CD_LOTE_CAIXA  AND
      MC.DT_CANCELAMENTO IS NULL  AND
      MC.TP_ORIGEM_MOV  IN  ('DD')     and  :CAMPO_TIPO = 'D'  and :CAMPO_CHECK_SEMMOVIMENTO = 'S' and
      trunc(mc.dt_movimentacao)             between :CAMPO_DATA_INIC UNION ALL
     SELECT
        LC.CD_CAIXA
       ,ML.CD_MOTIVO_LCTO
       ,ML.DS_MOTIVO_LCTO
       ,0 	   RECEITA
       ,Nvl(MC.VL_MOEDA,MC.VL_MOVIMENTACAO) DESPESA
	   ,Nvl(MC.CD_MOEDA, DBAMV.MOEDA_PADRAO(MC.CD_MULTI_EMPRESA))CD_MOEDA
    FROM
      DBAMV.MOTIVO_LCTO ML,
      DBAMV.MOV_CAIXA   MC,
      DBAMV.LOTE_CAIXA  LC
    WHERE
      MC.CD_MOTIVO_LCTO = ML.CD_MOTIVO_LCTO AND
      MC.CD_LOTE_CAIXA  = LC.CD_LOTE_CAIXA  AND
      MC.DT_CANCELAMENTO IS NULL  AND
      MC.TP_ORIGEM_MOV  IN  ('DD', 'PA')     and  :CAMPO_TIPO = 'A'  and :CAMPO_CHECK_SEMMOVIMENTO = 'N' and
      trunc(mc.dt_movimentacao)             between :CAMPO_DATA_INIC UNION ALL
     SELECT
        LC.CD_CAIXA
       ,ML.CD_MOTIVO_LCTO
       ,ML.DS_MOTIVO_LCTO
       ,0 	   RECEITA
       ,Nvl(MC.VL_MOEDA,MC.VL_MOVIMENTACAO) DESPESA
	   ,Nvl(MC.CD_MOEDA, DBAMV.MOEDA_PADRAO(MC.CD_MULTI_EMPRESA))CD_MOEDA
    FROM
      DBAMV.MOTIVO_LCTO ML,
      DBAMV.MOV_CAIXA   MC,
      DBAMV.LOTE_CAIXA  LC
    WHERE
      MC.CD_MOTIVO_LCTO = ML.CD_MOTIVO_LCTO(+)  AND
      MC.CD_LOTE_CAIXA  = LC.CD_LOTE_CAIXA  AND
      MC.DT_CANCELAMENTO IS NULL  AND
      MC.TP_ORIGEM_MOV  IN  ('DD', 'PA')     and  :CAMPO_TIPO = 'A'  and :CAMPO_CHECK_SEMMOVIMENTO = 'S' and
      trunc(mc.dt_movimentacao)             between :CAMPO_DATA_INIC UNION ALL
     SELECT
        LC.CD_CAIXA
       ,ML.CD_MOTIVO_LCTO
       ,ML.DS_MOTIVO_LCTO
       ,Nvl(MC.VL_MOEDA,MC.VL_MOVIMENTACAO) RECEITA
       ,0 DESPESA
	   ,Nvl(MC.CD_MOEDA, DBAMV.MOEDA_PADRAO(MC.CD_MULTI_EMPRESA))CD_MOEDA

       MC.CD_MOTIVO_LCTO = ML.CD_MOTIVO_LCTO AND

WHERE CX.CD_CAIXA = MOV.CD_CAIXA
AND   CX.CD_MULTI_EMPRESA = EMP.CD_MULTI_EMPRESA

    CX.CD_CAIXA,
    CX.DS_CAIXA,
