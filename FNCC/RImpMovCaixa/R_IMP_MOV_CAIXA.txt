--R_IMP_MOV_CAIXA
SELECT * FROM (
select  i.ds_arquivo
       ,i.cd_usuario
       ,i.dt_importa_mov_caixa
       ,i.cd_lote_caixa
       ,it.nr_sequencial
       ,it.ds_ocorrencia
       ,it.tp_situacao
       ,it.cd_usuario||' / '||to_char(it.dt_desprezado,'dd/mm/yyyy hh24') usuario_data
  from  dbamv.importa_mov_caixa   i
       ,dbamv.itimporta_mov_caixa it
 where  i.cd_importa_mov_caixa = it.cd_importa_mov_caixa
   and  i.cd_importa_mov_caixa   = :CAMPO_CD_IMPORTA_MOV_CAIXA
   and  i.cd_multi_empresa           = :CAMPO_CD_MULTI_EMPRESA
)
ORDER BY NR_SEQUENCIAL
