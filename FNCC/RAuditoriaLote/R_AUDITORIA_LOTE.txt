--R_AUDITORIA_LOTE
SELECT * FROM (
select Caixa.Cd_Caixa,
       Caixa.ds_Caixa,
       Decode( :CAMPO_AGRUPAR_EMPRESA, 'S', Multi_Empresas.cd_multi_empresa, null )   cd_multi_empresa,
       Decode( :CAMPO_AGRUPAR_EMPRESA, 'S', Multi_Empresas.ds_multi_empresa, null )   ds_multi_empresa,
       Decode( :CAMPO_AGRUPAR_EMPRESA, 'S', Caixa.Ds_Caixa, Caixa.Ds_Caixa||'    Empresa: '||Multi_Empresas.Ds_Multi_Empresa ) Ds_Caixa_Empresa
      ,Lote_Caixa.Cd_Lote_Caixa
      ,Lote_Caixa.Dt_Abertura
      ,Lote_Caixa.Cd_Usuario
      ,Lote_Caixa.Dt_Fechamento
      ,Lote_Caixa.Cd_Usuario_Fechamento
      ,Lote_Caixa.vl_saldo_inicial
      ,Lote_Caixa.vl_saldo_final
      ,Lote_Caixa.vl_saldo_dinheiro
      ,Mov_Caixa.cd_doc_caixa
      ,Mov_Caixa.ds_mov_caixa
      ,Mov_Caixa.dt_movimentacao
      ,Nvl(Mov_Caixa.vl_moeda,Mov_Caixa.vl_movimentacao) vl_movimentacao
      ,Mov_Caixa.tp_origem_mov
      ,Nvl(Mov_Caixa.cd_moeda,dbamv.moeda_padrao(Mov_Caixa.cd_multi_empresa)) cd_moeda
from dbamv.multi_empresas
    ,dbamv.caixa
    ,dbamv.lote_caixa
    ,dbamv.mov_caixa
Where lote_caixa.cd_caixa      = caixa.cd_caixa
and   mov_caixa.cd_lote_caixa = Lote_Caixa.cd_lote_caixa
And   trunc(lote_caixa.dt_abertura) Between trunc(:CAMPO_DATA_INICIAL) and trunc(:CAMPO_DATA_FIM)
And   caixa.cd_multi_empresa = multi_empresas.cd_multi_empresa
:CAMPO_CF_MULTI_EMPRESA
)
Order By  Cd_moeda
	,cd_multi_empresa
	 ,Cd_Caixa
         ,Cd_Lote_Caixa
         ,Dt_Abertura
         ,dt_movimentacao
