--R_HISTORICO_LOTE
SELECT * FROM (
select  Decode( :CAMPO_AGRUPAR_EMPRESA, 'S', Multi_Empresas.Cd_Multi_Empresa, Null )   Cd_Multi_Empresa,
       Multi_Empresas.Ds_Multi_Empresa,
       Caixa.Cd_Caixa
      ,Caixa.ds_Caixa
      ,Decode( :CAMPO_AGRUPAR_EMPRESA, 'S', ds_caixa, ds_caixa||'   Empresa: '||multi_Empresas.Ds_Multi_Empresa )    Ds_Caixa_Empresa
      ,Abre_Caixa.cd_lote_caixa
      ,Abre_Caixa.Dt_Abertura
      ,Abre_Caixa.Cd_Usuario_Abertura
      ,Abre_Caixa.Dt_Fechamento
      ,Abre_Caixa.Cd_Usuario_Fechamento
      ,Abre_Caixa.Ds_Justificativa
      ,Abre_Caixa.Sn_Reabertura
from dbamv.caixa
      ,dbamv.lote_caixa
      ,dbamv.abre_caixa
      ,dbamv.multi_empresas
Where CAIXA.CD_MULTI_EMPRESA = MULTI_EMPRESAS.CD_MULTI_EMPRESA
And   lote_caixa.cd_caixa      = caixa.cd_caixa
and   abre_caixa.cd_lote_caixa = Lote_Caixa.cd_lote_caixa
And   trunc(abre_caixa.dt_abertura) Between :CAMPO_DATA_INICIAL and 
:CAMPO_CF_MULTI_EMPRESA
)
Order by Cd_Caixa, Cd_Lote_Caixa, Dt_Abertura
