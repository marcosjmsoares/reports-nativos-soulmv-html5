--R_RECIBO_CAUCAO
SELECT '1' AS VIA
      ,'1ª VIA CLIENTE' AS DS_VIA
      ,'Este recibo só terá valor de quitação após compensação bancária do cheque mencionado neste documento.' AS TEXTO
      ,p.NM_PACIENTE
      ,c.CD_CAUCAO
      ,c.NM_PROPRIETARIO
      ,c.VL_CAUCAO
      ,c.dt_caucao
      ,case
          when tp_pagamento = 'P' then  DBAMV.PKG_RMI_TRADUCAO.EXTRAIR_PKG_MSG('MSG_04', 'DBAMV.VDIC_RECIBO_TESOURARIA', 'Nota Promissória')
          when tp_pagamento = 'D' THEN  DBAMV.PKG_RMI_TRADUCAO.EXTRAIR_PKG_MSG('MSG_03', 'DBAMV.VDIC_RECIBO_TESOURARIA', 'Dinheiro')
          when tp_pagamento = 'Q' THEN DBAMV.PKG_RMI_TRADUCAO.EXTRAIR_PKG_MSG('MSG_02', 'DBAMV.VDIC_RECIBO_TESOURARIA', 'Cheque')
          when tp_pagamento = 'C' and tp_debito_credito = 'C' THEN DBAMV.PKG_RMI_TRADUCAO.EXTRAIR_PKG_MSG('MSG_01', 'DBAMV.VDIC_RECIBO_TESOURARIA', 'Cartão de Crédito' ||' - Crédito' )
          when tp_pagamento = 'C' and tp_debito_credito = 'D' then DBAMV.PKG_RMI_TRADUCAO.EXTRAIR_PKG_MSG('MSG_01', 'DBAMV.VDIC_RECIBO_TESOURARIA', 'Cartão de Crédito' ||' - Débito')
          when tp_pagamento = 'I' then DBAMV.PKG_RMI_TRADUCAO.EXTRAIR_PKG_MSG('MSG_02', 'DBAMV.VDIC_RECIBO_TESOURARIA', 'Pix')
        end DESCRICAO
      ,dbamv.PACK_FNCP.EXTENSOS(c.VL_CAUCAO) AS EXTENSO
      ,NVL(c.NR_PARCELAS, 1) AS NR_PARCELAS
      ,NVL(c.VL_PARCELA, c.VL_CAUCAO) AS VL_PARCELA
  FROM dbamv.caucao c                                                                      
INNER JOIN dbamv.atendime a ON c.cd_atendimento = a.cd_atendimento
INNER JOIN dbamv.paciente p ON a.cd_paciente = p.cd_paciente
WHERE a.cd_atendimento =  :CAMPO_CD_ATENDIMENTO
  AND c.cd_caucao = :CAMPO_CD_CAUCAO
UNION ALL
SELECT '2' AS VIA
      ,'2ª VIA CLIENTE' AS DS_VIA
      ,'Este recibo só terá valor de quitação após compensação bancária do cheque mencionado neste documento.' AS TEXTO
      ,p.NM_PACIENTE
      ,c.CD_CAUCAO
      ,c.NM_PROPRIETARIO
      ,c.VL_CAUCAO
      ,c.dt_caucao
       ,case
          when tp_pagamento = 'P' then DBAMV.PKG_RMI_TRADUCAO.EXTRAIR_PKG_MSG('MSG_04', 'DBAMV.VDIC_RECIBO_TESOURARIA', 'Nota Promissória')
          when tp_pagamento = 'D' then DBAMV.PKG_RMI_TRADUCAO.EXTRAIR_PKG_MSG('MSG_03', 'DBAMV.VDIC_RECIBO_TESOURARIA', 'Dinheiro')
          when tp_pagamento = 'Q' then DBAMV.PKG_RMI_TRADUCAO.EXTRAIR_PKG_MSG('MSG_02', 'DBAMV.VDIC_RECIBO_TESOURARIA', 'Cheque')
          when tp_pagamento = 'C' and tp_debito_credito = 'C' THEN DBAMV.PKG_RMI_TRADUCAO.EXTRAIR_PKG_MSG('MSG_01', 'DBAMV.VDIC_RECIBO_TESOURARIA', 'Cartão de Crédito' || ' - Crédito' )
          when tp_pagamento = 'C' and tp_debito_credito = 'D' then DBAMV.PKG_RMI_TRADUCAO.EXTRAIR_PKG_MSG('MSG_01', 'DBAMV.VDIC_RECIBO_TESOURARIA', 'Cartão de Crédito' || ' - Débito')
          when tp_pagamento = 'I' then DBAMV.PKG_RMI_TRADUCAO.EXTRAIR_PKG_MSG('MSG_02', 'DBAMV.VDIC_RECIBO_TESOURARIA', 'Pix')
        end DESCRICAO
      ,dbamv.PACK_FNCP.EXTENSOS(c.VL_CAUCAO) AS EXTENSO
      ,NVL(c.NR_PARCELAS, 1) AS NR_PARCELAS
      ,NVL(c.VL_PARCELA, c.VL_CAUCAO) AS VL_PARCELA
     FROM dbamv.caucao c
INNER JOIN dbamv.atendime a ON c.cd_atendimento = a.cd_atendimento
INNER JOIN dbamv.paciente p ON a.cd_paciente = p.cd_paciente
WHERE a.cd_atendimento =  :CAMPO_CD_ATENDIMENTO
  AND c.cd_caucao = :CAMPO_CD_CAUCAO
