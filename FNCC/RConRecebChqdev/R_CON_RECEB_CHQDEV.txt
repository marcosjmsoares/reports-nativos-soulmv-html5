--R_CON_RECEB_CHQDEV
SELECT * FROM (
select
  doc_caixa.tp_doc_caixa
  , nvl(doc_caixa.vl_moeda,doc_caixa.vl_doc_caixa) vl_recebido
  , con_cor.cd_banco
  , con_cor.cd_agencia
  , con_cor.nr_conta
  , mov_caixa.dt_movimentacao
  , doc_caixa.nr_documento
  , responsavel.nm_responsavel
  , con_cor.cd_con_cor
  , con_cor.ds_con_cor
  , max(mov_caixa.dt_movimentacao) dt_movimentacao1
  , mov_caixa.cd_mot_dev
  , doc_caixa.tp_status_doc ds_status
  , Nvl(doc_caixa.cd_moeda,dbamv.moeda_padrao(doc_caixa.cd_multi_empresa)) cd_moeda
from dbamv.mov_caixa       mov_caixa
        , dbamv.doc_caixa     doc_caixa
        , dbamv.con_cor         con_cor
        , dbamv.responsavel   responsavel
where doc_caixa.cd_doc_caixa = mov_caixa.cd_doc_caixa
    and mov_caixa.cd_con_cor = con_cor.cd_con_cor
    and responsavel.cd_resposavel(+) = doc_caixa.cd_resposavel
    and mov_caixa.cd_doc_caixa is not null
    and mov_caixa.dt_cancelamento is null
    and mov_caixa.cd_con_cor is not null
    and mov_caixa.cd_mov_caixa_transf is null
group by
   Nvl(doc_caixa.cd_moeda,dbamv.moeda_padrao(doc_caixa.cd_multi_empresa))
  , con_cor.cd_con_cor
  , con_cor.ds_con_cor
  , doc_caixa.tp_doc_caixa
  , mov_caixa.dt_movimentacao
  , nvl(doc_caixa.vl_moeda,doc_caixa.vl_doc_caixa)
  , con_cor.cd_banco
  , con_cor.cd_agencia
  , con_cor.nr_conta
  , doc_caixa.nr_documento
  , responsavel.nm_responsavel
  , mov_caixa.cd_mot_dev
  , doc_caixa.tp_status_doc
)
ORDER BY CD_MOEDA,CD_BANCO,CD_AGENCIA,NR_CONTA,CD_CON_COR,DS_CON_COR
