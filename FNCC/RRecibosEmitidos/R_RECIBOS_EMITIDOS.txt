select recibo.cd_recibo,
       recibo.cd_usuario,
       recibo.tp_status,
       recibo.tp_recibo,
       recibo.tp_origem,
       recibo.dt_emissao,
       caucao.dt_caucao dt_vencimento,
       to_number( null ) setor,
       caucao.cd_atendimento atendimento,
       decode( caucao.tp_pagamento, 'C', 'Cartão',
                                                           'Q', 'Cheque',
                                                           'D', 'Dinheiro',
                                                           'P', 'Duplicata' ) tp_recebimento,
	   dbamv.fnc_fnfi_calcula_moeda(:CAMPO_DTFIM, PKG_MV2000.LE_EMPRESA,dbamv.moeda_padrao(PKG_MV2000.LE_EMPRESA),:CAMPO_CD_MOEDA,(recibo.vl_recibo)) vl_recibo,
	recibo.ds_observacao
  from dbamv.recibo,
       dbamv.caucao
 where recibo.vl_recibo > 0
     and recibo.cd_caucao = caucao.cd_caucao
     and caucao.cd_caucao is not null
union

       recibo.cd_usuario,
       recibo.tp_status,
       recibo.tp_recibo,
       recibo.tp_origem,
       recibo.dt_emissao,
       itcon_rec.dt_vencimento dt_vencimento,
       itreg_fat.cd_setor setor,
       con_rec.cd_atendimento atendimento,
       decode( reccon_rec.tp_recebimento, '1', 'Cheque',
                                                                  '2', 'Cartão',
                                                                  '3', 'Dinheiro',
                                                                  '4', 'Crédido C/C',
                                                                  '5', 'Borderô',
                                                                  '6', 'Boleto',
                                                                  '7', 'DOC',
                                                                  '8', 'Duplicata',
                                                                  '9', 'Nota Promissória',
                                                                  'E', 'Encontro de Contas',
                                                                  'B', 'Baixa Contábil',
                                                                  '11', 'Vale a Vista' ) tp_recebimento,
       dbamv.fnc_fnfi_calcula_moeda(:CAMPO_DTFIM, PKG_MV2000.LE_EMPRESA,dbamv.moeda_padrao(PKG_MV2000.LE_EMPRESA),:CAMPO_CD_MOEDA,(recibo.vl_recibo)) vl_recibo,
       recibo.ds_observacao
  from dbamv.recibo,
          dbamv.con_rec,
          dbamv.itcon_rec,
          dbamv.reccon_rec,
          dbamv.reg_fat,
          dbamv.itreg_fat
 where recibo.vl_recibo > 0
     and recibo.cd_reccon_rec is not null
     and recibo.cd_reccon_rec = reccon_rec.cd_reccon_rec
     and reccon_rec.cd_itcon_rec = itcon_rec.cd_itcon_rec
     and itcon_rec.cd_con_rec = con_rec.cd_con_rec
     and con_rec.cd_reg_fat = reg_fat.cd_reg_fat
     and reg_fat.cd_reg_fat = itreg_fat.cd_reg_fat
union
