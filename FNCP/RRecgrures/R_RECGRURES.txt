--R_RECGRURES_R_1_Q_LANC_EXTRATO
select  sum( ( mov_concor.vl_movimentacao - nvl(Mov_concor.vl_conciliado, 0 ) )  ) vl_total_extrato
            from dbamv.mov_concor,
                 dbamv.con_cor,
                 dbamv.lan_concor,
                 dbamv.empresa_lan_concor,
                 dbamv.item_res,
                 dbamv.gru_res,
                 dbamv.usuario_multi_empresa 
           where mov_concor.dt_movimentacao >= :CAMPO_DT_INIC
             and mov_concor.dt_movimentacao <= :CAMPO_DT_FIM
             and mov_concor.cd_con_cor = con_cor.cd_con_cor
             and lan_concor.cd_lan_concor = mov_concor.cd_lan_concor
             and lan_concor.cd_lan_concor = empresa_lan_concor.cd_lan_concor
             and empresa_lan_concor.cd_multi_empresa  =  con_cor.cd_multi_empresa
             and item_res.cd_gru_res = gru_res.cd_gru_res(+)
             and empresa_lan_concor.cd_item_res = item_res.cd_item_res(+)
			 and empresa_lan_concor.cd_multi_empresa = usuario_multi_empresa.cd_multi_empresa
             and usuario_multi_empresa.cd_id_usuario = user
             and ( (  mov_concor.cd_aplicacao is not null and nvl( lan_concor.tp_operacao_saldo_conta, '0' )  = '0'
                    and nvl( lan_concor.tp_operacao_saldo_aplicacao, '0' )  <> '-' )
                 or  (mov_concor.cd_aplicacao is null and lan_concor.tp_operacao_saldo_conta = '+') )
             and mov_concor.cd_bordero is null
             and mov_concor.cd_con_cor_transferencia is null
             and Mov_concor.cd_mov_concor_transferencia is null
             and Mov_concor.cd_mov_caixa is null
             and Mov_concor.cd_pagcon_pag is null
             and Mov_concor.cd_cheque is null
             and Mov_concor.vl_movimentacao  - nvl( Mov_concor.vl_conciliado, 0 ) > 0
             and (not exists( select cd_mov_concor
                               from dbamv.rec_mov_con r
                              where r.cd_mov_concor = Mov_concor.cd_mov_concor )
                  or exists( select cd_mov_concor
                               from dbamv.rec_mov_con r
                              where r.cd_mov_concor = Mov_concor.cd_mov_concor
                            and mov_concor.cd_fornecedor is not null
                            and mov_concor.cd_lan_concor  in ( select conf.cd_lanca_credito_convenio
                                                                 from dbamv.config_financ conf
                                                                where conf.cd_multi_empresa = Mov_concor.cd_multi_empresa)))

--R_RECGRURES_R_GRU_RES1_CG__Q_ITCON_REC
SELECT * FROM (
SELECT   GRU_RES.CD_GRU_RES                       CD_GRU_RES
        ,GRU_RES.DS_GRU_RES                               DS_GRU_RES
        ,RAT_CONREC.CD_ITEM_RES                       CODIGO
        ,ITEM_RES.DS_ITEM_RES                             DESCRICAO
        ,reccon_rec.DT_recebimento                           DT_RECEBIMENTO
        ,itCON_REC.DT_PREVISTA_recebimento      DATA_PREV_PAGAMENTO
        ,CON_REC.NR_DOCUMENTO                         NR_DOCUMENTO
        ,CON_REC.DS_CON_REC                        DS_CON_REC
        ,ltrim(CON_REC.DS_CON_REC || ' ' || HIST_P.DS_HISTORICO_PADRAO)   DS_HIST_P
        ,decode(CON_REC.CD_FORNECEDOR,'',CON_REC.NM_CLIENTE,FORNECEDOR.NM_FANTASIA)     NM_FANTASIA
        ,CON_COR.NR_CONTA                                  NR_CONTA
        ,RECCON_REC.VL_ACRESCIMO        VL_ACRESCIMO
        ,RECCON_REC.VL_DESCONTO         VL_DESCONTO
        ,RAT_CONREC.CD_CON_REC           CD_RAT_CONREC
        ,CON_REC.VL_PREVISTO                  VL_PREVISTO
        ,DECODE(RAT_CONREC.VL_RATEIO,NULL,itCON_REC.VL_duplicata,RAT_CONREC.VL_RATEIO)                           VL_RATEIO
        ,reccon_rec.VL_recebido           VL_MOVIMENTACAO
         , dbamv.calcula_valor_rateio ( con_rec.cd_con_rec , rat_conrec.rowid , Reccon_rec.Vl_Recebido , Rat_conrec.vl_Rateio)
            VL_RECEBIDO
        ,CON_REC.CD_CON_REC
FROM  dbamv.itcon_rec
          , dbamv.CON_REC                          CON_REC
        ,dbamv.RAT_CONREC                      RAT_CONREC
        ,dbamv.ITEM_RES                            ITEM_RES
        ,dbamv.reccon_rec                           reccon_rec
        ,dbamv.FORNECEDOR                     FORNECEDOR
        ,dbamv.CON_COR                             CON_COR
        ,dbamv.HISTORICO_PADRAO          HIST_P
        ,dbamv.GRU_RES                             GRU_RES
        ,dbamv.SETOR                                  SETOR
		,dbamv.USUARIO_MULTI_EMPRESA                  UME
WHERE   CON_REC.CD_FORNECEDOR              = FORNECEDOR.CD_FORNECEDOR(+)
AND     itcon_rec.cd_itcon_rec = RECCON_REC.cd_itcon_rec
AND      CON_REC.CD_HISTORICO_PADRAO   = HIST_P.CD_HISTORICO_PADRAO(+)
AND      ITCON_REC.TP_QUITACAO IN ('Q','P','G')
AND      CON_REC.CD_PREVISAO IS NULL
AND     ITCON_REC.CD_CON_REC  = CON_REC.CD_CON_REC
AND     ITCON_REC.CD_ITCON_REC =  RECCON_REC.CD_ITCON_REC
AND      ITEM_RES.CD_ITEM_RES(+)                 = RAT_CONREC.CD_ITEM_RES
AND      ITEM_RES.CD_GRU_RES                     =  GRU_RES.CD_GRU_RES(+)
AND      RAT_CONREC.CD_CON_REC(+)               = CON_REC.CD_CON_REC
AND      RAT_CONREC.CD_SETOR                   = SETOR.CD_SETOR(+)
AND      RECCON_REC.CD_CON_COR             = CON_COR.CD_CON_COR(+)
AND      RECCON_REC.CD_MULTI_EMPRESA  =  CON_COR.CD_MULTI_EMPRESA(+)
AND      RECCON_REC.CD_MULTI_EMPRESA  =  UME.CD_MULTI_EMPRESA
AND      UME.CD_ID_USUARIO           =   USER
and      Reccon_Rec.dt_estorno is null
AND      Decode(:CAMPO_CONSIDERAR_DT_CHEQUE,'S',Nvl(reccon_rec.dt_cheque, RECCON_REC.DT_RECEBIMENTO),reccon_rec.dt_recebimento)     BETWEEN :CAMPO_DT_INIC AND 
And        ( :CAMPO_CONSIDERAR_CHEQUES_COMPENSA = 'N'  Or
                  ( Reccon_Rec.tp_recebimento <> '1'  Or  Reccon_Rec.cd_caixa IS NULL )  )
UNION ALL
SELECT  GRU_RES.CD_GRU_RES                       CD_GRU_RES
        ,GRU_RES.DS_GRU_RES                               DS_GRU_RES
        ,RAT_CONREC.CD_ITEM_RES                       CODIGO
        ,ITEM_RES.DS_ITEM_RES                             DESCRICAO
        ,reccon_rec.DT_recebimento                           DT_RECEBIMENTO
        ,itCON_REC.DT_PREVISTA_recebimento      DATA_PREV_PAGAMENTO
        ,CON_REC.NR_DOCUMENTO                         NR_DOCUMENTO
        ,CON_REC.DS_CON_REC                               DS_CON_REC
        ,ltrim(CON_REC.DS_CON_REC || ' ' || HIST_P.DS_HISTORICO_PADRAO)   DS_HIST_P
        ,decode(CON_REC.CD_FORNECEDOR,'',CON_REC.NM_CLIENTE,FORNECEDOR.NM_FANTASIA)     NM_FANTASIA
        ,CON_COR.NR_CONTA                                  NR_CONTA
        ,RECCON_REC.VL_ACRESCIMO        VL_ACRESCIMO
        ,RECCON_REC.VL_DESCONTO         VL_DESCONTO
        ,RAT_CONREC.CD_CON_REC           CD_RAT_CONREC
        ,CON_REC.VL_PREVISTO                  VL_PREVISTO
        ,DECODE(RAT_CONREC.VL_RATEIO,NULL,itCON_REC.VL_duplicata,RAT_CONREC.VL_RATEIO)                           VL_RATEIO
        ,reccon_rec.VL_recebido           VL_MOVIMENTACAO
         , dbamv.calcula_valor_rateio ( con_rec.cd_con_rec , rat_conrec.rowid , Reccon_rec.Vl_Recebido , Rat_conrec.vl_Rateio)
            VL_RECEBIDO
        ,CON_REC.CD_CON_REC
  FROM dbamv.con_rec con_rec
      ,dbamv.itcon_rec itcon_rec
      ,dbamv.rat_conrec rat_conrec
      ,dbamv.item_res item_res
      ,dbamv.reccon_rec reccon_rec
      ,dbamv.gru_res gru_res
      ,dbamv.doc_caixa doc
      ,dbamv.mov_concor cc
      ,dbamv.mov_caixa
      ,dbamv.CON_COR                             CON_COR
      ,dbamv.HISTORICO_PADRAO          HIST_P
      ,dbamv.FORNECEDOR                     FORNECEDOR
	  ,dbamv.USUARIO_MULTI_EMPRESA                  UME
  WHERE :CAMPO_CONSIDERAR_CHEQUES_COMPENSA= 'S'
    And  CON_REC.CD_FORNECEDOR        = FORNECEDOR.CD_FORNECEDOR(+)
    AND CON_REC.CD_HISTORICO_PADRAO   = HIST_P.CD_HISTORICO_PADRAO(+)
    And con_rec.cd_con_rec = itcon_rec.cd_con_rec
    AND reccon_rec.cd_itcon_rec     = itcon_rec.cd_itcon_rec
    AND reccon_rec.tp_recebimento   = '1'
    AND reccon_rec.cd_caixa IS NOT NULL
    AND RECCON_REC.CD_CON_COR             = CON_COR.CD_CON_COR(+)
    AND      RECCON_REC.CD_MULTI_EMPRESA  =  CON_COR.CD_MULTI_EMPRESA(+)
	AND      RECCON_REC.CD_MULTI_EMPRESA  =  UME.CD_MULTI_EMPRESA
	AND      UME.CD_ID_USUARIO           =   USER
    AND reccon_rec.cd_doc_caixa = doc.cd_doc_caixa
    AND doc.cd_doc_caixa        = mov_caixa.cd_doc_caixa
    AND cc.cd_mov_caixa = mov_caixa.cd_mov_caixa
    AND cc.cd_mov_caixa = ( select max( to_number(mc.cd_mov_caixa) )
                              from dbamv.mov_caixa mc
                              where mc.cd_doc_caixa = doc.cd_doc_caixa
                                and cd_motivo_canc is null )
    AND doc.tp_status_doc = 'R'
    AND doc.cd_con_cor IS NOT NULL
    AND cc.sn_conciliado  = 'S'
    AND Decode(:CAMPO_CONSIDERAR_DT_CHEQUE,'S',Nvl(reccon_rec.dt_cheque, cc.dt_movimentacao),cc.dt_movimentacao)
        BETWEEN TO_DATE (To_Char(:CAMPO_DT_INIC,'dd/mm/yyyy') ||' 00', 'dd/mm/yyyy hh24')
            AND TO_DATE (To_Char(:CAMPO_DT_FIM,'dd/mm/yyyy') ||' 23', 'dd/mm/yyyy hh24')
    AND itcon_rec.tp_quitacao IN ('Q', 'P', 'G')
    AND item_res.cd_item_res (+) = rat_conrec.cd_item_res
    AND gru_res.cd_gru_res   (+) = item_res.cd_gru_res
    AND rat_conrec.cd_con_rec(+) = con_rec.cd_con_rec
:CAMPO_CF_ORDEM
)
ORDER BY DS_GRU_RES,CD_GRU_RES,
CODIGO,DESCRICAO,
NM_FANTASIA,NR_DOCUMENTO,DS_HIST_P,NR_CONTA,DS_CON_REC,VL_RATEIO,DATA_PREV_PAGAMENTO

--R_RECGRURES_R_GRU_RES2_CG__Q_ITCON_REC
select   CD_GRU_RES
,DS_GRU_RES
,CODIGO
,DESCRICAO
,DATA_PREV_PAGAMENTO
,NR_DOCUMENTO
,DS_CON_REC
,DS_HIST_P
,NM_FANTASIA
,NR_CONTA
,Sum(VL_ACRESCIMO) VL_ACRESCIMO
,Sum(VL_DESCONTO) VL_DESCONTO
,vl_previsto
,Decode(cd_rank, 1, vl_rateio, null) VL_RATEIO
,Sum(VL_MOVIMENTACAO) VL_MOVIMENTACAO
,Sum(VL_RECEBIDO) VL_RECEBIDO
,CD_CON_REC
 from (
SELECT   NVL(GRU_RES.CD_GRU_RES, 0)                       CD_GRU_RES
        ,Nvl (gru_res.ds_gru_res, 'GRUPO NÃO INFORMADO')                                DS_GRU_RES
        ,NVL(RAT_CONREC.CD_ITEM_RES, 0)                       CODIGO
        ,NVL(ITEM_RES.DS_ITEM_RES, 'ITEM DE RESULTADO NÃO INFORMADO')                            DESCRICAO
        ,reccon_rec.DT_recebimento                           DT_RECEBIMENTO
        ,itCON_REC.DT_PREVISTA_recebimento      DATA_PREV_PAGAMENTO
        ,CON_REC.NR_DOCUMENTO                         NR_DOCUMENTO
        ,CON_REC.DS_CON_REC                               DS_CON_REC
        ,HIST_P.DS_HISTORICO_PADRAO                DS_HIST_P
        ,decode(CON_REC.CD_FORNECEDOR,'',CON_REC.NM_CLIENTE,FORNECEDOR.NM_FANTASIA)     NM_FANTASIA
        ,CON_COR.NR_CONTA                                  NR_CONTA
        ,RECCON_REC.VL_ACRESCIMO        VL_ACRESCIMO
        ,RECCON_REC.VL_DESCONTO         VL_DESCONTO
        ,RAT_CONREC.CD_CON_REC           CD_RAT_CONREC
        ,CON_REC.VL_PREVISTO                  VL_PREVISTO
        ,DECODE(RAT_CONREC.VL_RATEIO,NULL,itCON_REC.VL_duplicata,RAT_CONREC.VL_RATEIO)                           VL_RATEIO
        ,reccon_rec.VL_recebido           VL_MOVIMENTACAO
	, dbamv.Calcula_valor_rateio (con_rec.cd_con_rec, rat_conrec.ROWID, ( Nvl (reccon_rec.vl_recebido, 0)
		     - Nvl (reccon_rec.vl_acrescimo, 0) + Nvl ( reccon_rec.vl_imposto_retido, 0) ), rat_conrec.vl_rateio)  VL_RECEBIDO
	,CON_REC.CD_CON_REC CD_CON_REC
	,Decode('S','S',(Rank() OVER ( PARTITION BY con_rec.cd_con_rec,ITEM_RES.cd_ITEM_RES ORDER BY (rat_conrec.nr_linha || reccon_rec.cd_reccon_rec))),1) cd_rank
FROM  dbamv.itcon_rec
          , dbamv.CON_REC                          CON_REC
        ,dbamv.RAT_CONREC                      RAT_CONREC
        ,dbamv.ITEM_RES                            ITEM_RES
        ,dbamv.reccon_rec                           reccon_rec
        ,dbamv.FORNECEDOR                     FORNECEDOR
        ,dbamv.CON_COR                             CON_COR
        ,dbamv.HISTORICO_PADRAO          HIST_P
        ,dbamv.GRU_RES                             GRU_RES
        ,dbamv.SETOR                                  SETOR
		,dbamv.USUARIO_MULTI_EMPRESA                  UME
WHERE   CON_REC.CD_FORNECEDOR              = FORNECEDOR.CD_FORNECEDOR(+)
AND      CON_REC.CD_HISTORICO_PADRAO   = HIST_P.CD_HISTORICO_PADRAO(+)
AND      ITCON_REC.TP_QUITACAO IN ('Q','P','G')
AND      CON_REC.CD_PREVISAO IS NULL
AND     ITCON_REC.CD_CON_REC  = CON_REC.CD_CON_REC
AND     ITCON_REC.CD_ITCON_REC =  RECCON_REC.CD_ITCON_REC
AND      ITEM_RES.CD_ITEM_RES(+)                 = RAT_CONREC.CD_ITEM_RES
AND      ITEM_RES.CD_GRU_RES                     =  GRU_RES.CD_GRU_RES(+)
AND      RAT_CONREC.CD_CON_REC(+)               = CON_REC.CD_CON_REC
AND      RAT_CONREC.CD_SETOR                   = SETOR.CD_SETOR(+)
AND      RECCON_REC.CD_CON_COR             = CON_COR.CD_CON_COR(+)
AND      RECCON_REC.CD_MULTI_EMPRESA  =  CON_COR.CD_MULTI_EMPRESA(+)
AND      RECCON_REC.CD_MULTI_EMPRESA  =  UME.CD_MULTI_EMPRESA
AND      UME.CD_ID_USUARIO           =   USER
and      Reccon_Rec.dt_estorno is null
AND      Decode(:CAMPO_CONSIDERAR_DT_CHEQUE,'S',Nvl(reccon_rec.dt_cheque, reccon_rec.dt_recebimento), reccon_rec.dt_recebimento)  BETWEEN :CAMPO_DT_INIC AND  
And        ( :CAMPO_CONSIDERAR_CHEQUES_COMPENSA = 'N'  Or
                  ( Reccon_Rec.tp_recebimento <> '1'  Or  Reccon_Rec.cd_caixa IS NULL )  )
union all
SELECT   NVL(GRU_RES.CD_GRU_RES, 0)                       CD_GRU_RES
        ,Nvl (gru_res.ds_gru_res, 'GRUPO NÃO INFORMADO')                                DS_GRU_RES
        ,NVL(ratrec_desc_acres.CD_ITEM_RES, 0)                       CODIGO
        ,NVL(ITEM_RES.DS_ITEM_RES, 'ITEM DE RESULTADO NÃO INFORMADO')                            DESCRICAO
        ,reccon_rec.DT_recebimento                           DT_RECEBIMENTO
        ,itCON_REC.DT_PREVISTA_recebimento      DATA_PREV_PAGAMENTO
        ,CON_REC.NR_DOCUMENTO                         NR_DOCUMENTO
        ,CON_REC.DS_CON_REC                               DS_CON_REC
        ,HIST_P.DS_HISTORICO_PADRAO                DS_HIST_P
        ,decode(CON_REC.CD_FORNECEDOR,'',CON_REC.NM_CLIENTE,FORNECEDOR.NM_FANTASIA)     NM_FANTASIA
        ,CON_COR.NR_CONTA                                  NR_CONTA
        ,0       VL_ACRESCIMO
        ,0         VL_DESCONTO
        ,con_rec.CD_CON_REC           CD_RAT_CONREC
        ,0                  VL_PREVISTO
        ,0                           VL_RATEIO
        ,0           VL_MOVIMENTACAO
		, ratrec_desc_acres.vl_ratrec_desc_acres  VL_RECEBIDO
		,CON_REC.CD_CON_REC CD_CON_REC
		,Decode('S','S',(Rank() OVER ( PARTITION BY itcon_rec.cd_itcon_rec,ITEM_RES.cd_ITEM_RES ORDER BY reccon_rec.cd_reccon_rec)),1) cd_rank
FROM  dbamv.itcon_rec
        , dbamv.CON_REC                          CON_REC
        ,dbamv.ITEM_RES                            ITEM_RES
        ,dbamv.reccon_rec                           reccon_rec
        ,dbamv.FORNECEDOR                     FORNECEDOR
        ,dbamv.CON_COR                             CON_COR
        ,dbamv.HISTORICO_PADRAO          HIST_P
        ,dbamv.GRU_RES                             GRU_RES
        ,dbamv.SETOR                                  SETOR
		,dbamv.rec_desc_acres   rec_desc_acres
