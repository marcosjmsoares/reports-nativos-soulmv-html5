--R_CUS_CENCUS_PS_PREV
SELECT   PS.CD_REDUZIDO_PS     CD_CONTABIL
                ,V.DATA                DATA
                ,nvl(PS.DS_CONTA_PS,'NÃO INFORMADA')          DS_CONTA
               ,SUM( V.VL_ENTRADA )    VL_ENTRADA
               ,SUM( V.VL_SAIDA )           VL_SAIDA
FROM     DBAMV.V_CUSTO_CONTA_PREV  V
              ,DBAMV.PLANO_SINTETICO  PS
WHERE    V.CD_MULTI_EMPRESA  = :CAMPO_CD_MULTI_EMPRESA
AND          V.CD_REDUZIDO_PS    = PS.CD_REDUZIDO_PS(+)
AND V.DATA  <=  To_date ( To_char ( :CAMPO_DT_FINAL   ,'DD/MM/YYYY') || ' 23'  , 'DD/MM/YYYY HH24')
AND V.DATA  >=  To_date ( To_char ( :CAMPO_DT_INICIAL ,'DD/MM/YYYY') || ' 00'  , 'DD/MM/YYYY HH24')
GROUP BY PS.CD_REDUZIDO_PS
                   ,V.DATA
                  , PS.DS_CONTA_PS
ORDER BY CD_CONTABIL
