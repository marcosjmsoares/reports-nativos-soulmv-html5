--R_CONTA_RECEB_CONV_CONCOR
SELECT * FROM (
SELECT
CON_COR.CD_CON_COR    CON_COR
, CON_COR.NR_CONTA    NR_CONTA
, CON_COR.DS_CON_COR  DS_CON_COR
, REC.DT_RECEBIMENTO DT_RECEBIMENTO
, CONV.NM_CONVENIO    CONVENIO
, NF.CD_NOTA_FISCAL   NF
, IT.DT_VENCIMENTO    VENCIMENTO
, NF.DT_EMISSAO      EMISSAO_NF
, NF.VL_TOTAL_NOTA VALOR_NF_ORIGINAL
, DECODE(AGRUP.CD_CON_REC_AGRUP,NULL,REC.VL_RECEBIDO,
   (AGRUP.VL_DUPLICATA / TOT_AGRUP.VL_TOT_DUPLICATA) * REC.VL_RECEBIDO ) VL_RECEBIDO
FROM
   (SELECT I.CD_CON_REC_AGRUP
           , C.CD_NOTA_FISCAL
           , SUM(I.VL_DUPLICATA) VL_DUPLICATA
	FROM  DBAMV.CON_REC C
          , DBAMV.ITCON_REC I
	WHERE I.CD_CON_REC_AGRUP IS NOT NULL
	AND   C.CD_CON_REC = I.CD_CON_REC
	GROUP BY I.CD_CON_REC_AGRUP
             , C.CD_NOTA_FISCAL) AGRUP
   , (SELECT I.CD_CON_REC_AGRUP
             , SUM(I.VL_DUPLICATA) VL_TOT_DUPLICATA
      FROM  DBAMV.ITCON_REC I
      WHERE I.CD_CON_REc_AGRUP IS NOT NULL
	  GROUP BY I.CD_CON_REC_AGRUP) TOT_AGRUP
    , DBAMV.ITCON_REC   IT
    , DBAMV.CON_REC     CON
    , DBAMV.NOTA_FISCAL NF
    , DBAMV.CONVENIO    CONV
    , DBAMV.RECCON_REC  REC
    , DBAMV.FORNECEDOR  FORNEC
    , DBAMV.CON_COR     CON_COR
WHERE
CON.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND CON.TP_CON_REC = 'C'
AND IT.CD_CON_REC = CON.CD_CON_REC
AND IT.CD_CON_REC_AGRUP IS NULL
AND AGRUP.CD_CON_REC_AGRUP     (+)= CON.CD_CON_REC
AND TOT_AGRUP.CD_CON_REC_AGRUP (+)= CON.CD_CON_REC
AND CONV.CD_CONVENIO = NF.CD_CONVENIO
AND NF.CD_NOTA_FISCAL = CON.CD_NOTA_FISCAL
and con_cor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA 
AND IT.CD_ITCON_REC = REC.CD_ITCON_REC
AND REC.CD_CON_COR = CON_COR.CD_CON_COR(+)
AND CON.CD_FORNECEDOR = FORNEC.CD_FORNECEDOR(+)
AND CON.CD_PREVISAO IS NULL
ORDER BY CONV.NM_CONVENIO, IT.DT_VENCIMENTO, NF.CD_NOTA_FISCAL, NF.DT_EMISSAO
)
ORDER BY NR_CONTA,DS_CON_COR,CON_COR,
DT_RECEBIMENTO
