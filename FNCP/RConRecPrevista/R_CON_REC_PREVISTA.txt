--R_CON_REC_PREVISTA
SELECT * FROM (
SELECT
       TRUNC( ITCON_REC.DT_PREVISTA_RECEBIMENTO )          DT_VENCIMENTO
      ,CON_REC.NR_DOCUMENTO    NR_DOCUMENTO
      ,DECODE( CON_REC.CD_FORNECEDOR, '', CON_REC.NM_CLIENTE, FORNEC.NM_FANTASIA )   NM_FORNECEDOR
      ,CON_REC.CD_CON_REC                                      CD_CON_REC
      ,CON_REC.DT_EMISSAO                                      DT_EMISSAO
      ,ITCON_REC.VL_DUPLICATA-nvl(itcon_rec.vl_soma_recebido,0)                            VL_PREVISTO
      ,DECODE (:CAMPO_CONVERTE,1, :CAMPO_CD_MOEDA , ITCON_REC.CD_MOEDA)             CD_MOEDA_PREV
      ,ITCON_REC.VL_MOEDA      VL_MOEDA_PREV
      ,CON_REC.DT_LANCAMENTO     DT_LANCAMENTO
FROM  DBAMV.ITCON_REC
       , DBAMV.CON_REC
      ,DBAMV.FORNECEDOR FORNEC
      ,DBAMV.TIP_DOC
WHERE  CON_REC.CD_PREVISAO  IS not NULL
AND    CON_REC.CD_FORNECEDOR    = FORNEC.CD_FORNECEDOR(+)
AND    ITCON_REC.TP_QUITACAO<>'Q'
AND    CON_REC.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND    CON_REC.CD_TIP_DOC       =  TIP_DOC.CD_TIP_DOC
AND    ITCON_REC.CD_CON_REC   =  CON_REC.CD_CON_REC
AND    TRUNC( CON_REC.DT_EMISSAO)  BETWEEN TRUNC( :CAMPO_DATA_INICIAL )   AND  TRUNC( :CAMPO_DATA_FINAL )
AND    TRUNC( ITCON_REC.DT_PREVISTA_RECEBIMENTO) BETWEEN TRUNC( :CAMPO_DATA_PREV_INICIAL)  AND TRUNC( :CAMPO_DATA_PREV_FINAL)
ORDER BY TRUNC( DT_VENCIMENTO ) , NM_FORNECEDOR
)
