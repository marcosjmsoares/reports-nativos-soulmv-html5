--R_LANCAMENTOS_IVA
SELECT
Decode (TP_REGISTRO,'CPAG', 'CREDITO A PAGAR','CREC','CREDITO A RECEBER')TP_REGISTRO,
Decode(TP_LANCAMENTO,'ENTR', 'ENTRADA','BAIX','BAIXA', 'ESTO-ENTR', 'ESTORNO DE ENTRADA','ESTO-BAIX', 'ESTORNO DE BAIXA')TP_LANCAMENTO,
CD_EMPRESA,
CD_CONTA,
CD_BAIXA,
NR_DOCUMENTO,
CD_PESSOA,
NM_PESSOA,
DT_LANCAMENTO,
VL_BASE,
VL_IVA,
CD_REDUZIDO,
CD_CONTABIL,
DS_MOVIMENTO
FROM
DBAMV.V_IVA_FINANCEIRO
WHERE
CD_EMPRESA = :CAMPO_CD_MULTI_EMPRESA1
ORDER BY
TP_REGISTRO,
TP_LANCAMENTO,
NM_PESSOA,
DT_LANCAMENTO,
CD_CONTA,
CD_BAIXA
