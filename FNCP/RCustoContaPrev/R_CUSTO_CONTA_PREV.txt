--R_CUSTO_CONTA_PREV
SELECT * FROM (
SELECT   V.CD_CONTA
        ,V.DS_CONTA
        ,V.DATA
        ,V.HISTORICO
        ,V.NR_DOC
        ,V.VL_ENTRADA
        ,V.VL_SAIDA
        ,V.SITUACAO
FROM     DBAMV.V_CUSTO_CONTA_PREV   V
WHERE    V.CD_MULTI_EMPRESA  =    :CAMPO_CD_MULTI_EMPRESA
AND ( TRUNC( V.DATA ) BETWEEN :CAMPO_DT_INICIAL and :CAMPO_DT_FINAL )
)
ORDER BY CD_CONTA,DS_CONTA
