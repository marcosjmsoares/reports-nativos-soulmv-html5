select   con_pag.ds_fornecedor,
           con_pag.ds_con_pag,
           con_pag.cd_con_pag,
           pagcon_pag.vl_pago vl_soma_pago,
           CON_PAG.VL_BRUTO_CONTA   vl_bruto_adiant
from dbamv.con_pag,
        dbamv.itcon_pag,
        dbamv.pagcon_pag
where itcon_pag.cd_itcon_pag = pagcon_pag.cd_itcon_pag
          and itcon_pag.cd_con_pag = con_pag.cd_con_pag
          and con_pag.tp_adiantamento in ('P','F')
	  and con_pag.cd_multi_empresa  = dbamv.pkg_mv2000.le_empresa
