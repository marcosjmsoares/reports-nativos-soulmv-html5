--R_CONTA_REC_DIV
Select sysdate from dual


/*----------------------------------------------------------------------------*/
--R_CONTA_REC_DIV_R_TP_ATENDIMENTO4_Q_CONTA_REC_PAC3
SELECT * FROM (
SELECT DISTINCT
     CON_REC.NM_CLIENTE NM_CLIENTE4
     ,CON_REC.DS_CON_REC DS_CON_REC4
     ,CON_REC.CD_CON_REC CD_CON_REC4
     ,MOV_CONCOR.DT_MOVIMENTACAO DT_MOVIMENTACAO4
     ,RECCON_REC.TP_RECEBIMENTO TP_RECEBIMENTO4
     ,DECODE(RECCON_REC.TP_RECEBIMENTO, '1' , 'CHEQUE',
                                     '3' , 'DINHEIRO',
                                     '4' , 'CREDITO C/C',
                                     '2', 'CARTAO')   RECEBIMENTO4
     ,CON_REC.VL_PREVISTO VL_PREVISTO4
     ,NVL(CON_REC.VL_RECEBIDO,0) VL_RECEBIDO4
     ,CON_REC.NR_DOCUMENTO	NR_DOCUMENTO4
     ,CON_REC.CD_ATENDIMENTO CD_ATENDIMENTO4
     ,ATENDIME.TP_ATENDIMENTO TP_ATENDIMENTO4
FROM    DBAMV.CON_REC
        ,DBAMV.MOV_CONCOR
        ,DBAMV.ATENDIME
        ,DBAMV.REC_MOV_CON
        ,DBAMV.RECCON_REC
        ,DBAMV.ITCON_REC
WHERE MOV_CONCOR.CD_MOV_CONCOR = REC_MOV_CON.CD_MOV_CONCOR
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND RECCON_REC.CD_RECCON_REC   = REC_MOV_CON.CD_RECCON_REC
AND RECCON_REC.CD_ITCON_REC    = ITCON_REC.CD_ITCON_REC
AND ITCON_REC.CD_CON_REC       = CON_REC.CD_CON_REC
AND CON_REC.CD_ATENDIMENTO     = ATENDIME.CD_ATENDIMENTO
AND CON_REC.TP_CON_REC = 'P'
AND CON_REC.CD_PREVISAO IS NULL
AND CON_REC.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND :CAMPO_ORDEM  = 'Recebimento'
ORDER BY DT_MOVIMENTACAO4  ASC
)
ORDER BY TP_ATENDIMENTO4,
DT_MOVIMENTACAO4


/*----------------------------------------------------------------------------*/
--R_CONTA_REC_DIV_R_TP_PACIENTE_Q_CONTA_REC_PAC
SELECT * FROM (
SELECT   DISTINCT
                 CON_REC.NM_CLIENTE      		NM_CLIENTE1
                ,CON_REC.DS_CON_REC     	DS_CON_REC1
                ,CON_REC.CD_CON_REC     	CD_CON_REC1
                ,ITCON_REC.DT_PREVISTA_RECEBIMENTO	DT_PREVISTA_PAGAMENTO1
                ,RECCON_REC.TP_RECEBIMENTO	TP_RECEBIMENTO1
                ,DECODE(RECCON_REC.TP_RECEBIMENTO, '1' , 'CHEQUE', '3' , 'DINHEIRO', '4' , 'CREDITO C/C', '2', 'CARTAO')   RECEBIMENTO1
                ,CON_REC.VL_PREVISTO		VL_PREVISTO1
                ,NVL(CON_REC.VL_RECEBIDO,0)		VL_RECEBIDO1
                ,CON_REC.NR_DOCUMENTO	NR_DOCUMENTO1
                ,CON_REC.CD_ATENDIMENTO	CD_ATENDIMENTO1
                ,ATENDIME.TP_ATENDIMENTO           TP_ATENDIMENTO1
FROM      DBAMV.CON_REC
                ,DBAMV.ATENDIME
                ,DBAMV.ITCON_REC
                ,DBAMV.RECCON_REC
WHERE   CON_REC.CD_CON_REC = ITCON_REC.CD_CON_REC
AND         ITCON_REC.CD_ITCON_REC = RECCON_REC.CD_ITCON_REC
AND         CON_REC.TP_CON_REC = 'P'
AND         CON_REC.CD_PREVISAO IS NULL
AND	CON_REC.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND         ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND         CON_REC.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
AND	:CAMPO_ORDEM = 'Paciente'
ORDER BY TP_ATENDIMENTO
                  ,NM_CLIENTE1
                  ,CD_ATENDIMENTO1
                  ,DT_PREVISTA_PAGAMENTO1
)
ORDER BY TP_ATENDIMENTO1,
NM_CLIENTE1,DS_CON_REC1,CD_CON_REC1
