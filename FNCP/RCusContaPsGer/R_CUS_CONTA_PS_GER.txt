--R_CUS_CONTA_PS_GER
SELECT A.CD_CONTA
              ,A.DATA             DATA
             ,A.DS_CONTA
             ,SUM(A.VL_ENTRADA) VL_ENTRADA
             ,SUM(A.VL_SAIDA)  VL_SAIDA
FROM (
SELECT   PS.CD_REDUZIDO_PS                                    CD_CONTA
                ,V.DATA                          DATA
               ,nvl(PS.DS_CONTA_PS,'NÃO INFORMADA')   DS_CONTA
              , V.VL_ENTRADA                                               VL_ENTRADA
              , V.VL_SAIDA                                                      VL_SAIDA
FROM     DBAMV.V_CUSTO_CONTA    V
              ,DBAMV.PLANO_SINTETICO  PS
WHERE    V.CD_MULTI_EMPRESA  =       :CAMPO_CD_MULTI_EMPRESA
      AND      TRUNC( V.DATA )     BETWEEN :CAMPO_DT_INICIAL and 
      AND      V.CD_REDUZIDO_PS    =       PS.CD_REDUZIDO_PS(+)
UNION ALL
SELECT   PS.CD_REDUZIDO_PS                                    CD_CONTA
                ,V.DATA              DATA
               ,nvl(PS.DS_CONTA_PS,'NÃO INFORMADA')   DS_CONTA
              , V.VL_ENTRADA                                               VL_ENTRADA
              , V.VL_SAIDA                                                      VL_SAIDA
FROM     DBAMV.V_CUSTO_CONTA_PREV    V
              ,DBAMV.PLANO_SINTETICO  PS
WHERE    V.CD_MULTI_EMPRESA  =       :CAMPO_CD_MULTI_EMPRESA
      AND      TRUNC( V.DATA )     BETWEEN :CAMPO_DT_INICIAL and 
      AND      V.CD_REDUZIDO_PS    =       PS.CD_REDUZIDO_PS(+)
:CAMPO_COND_CD_CONTABIL_PS
) A
GROUP BY  A.CD_CONTA
                     ,A.DATA
                    ,A.DS_CONTA
ORDER BY A.CD_CONTA
