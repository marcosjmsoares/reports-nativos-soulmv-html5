--R_AJUSTE_RECEB_PART
SELECT con_rec.cd_con_rec
    ,reccon_rec.cd_reccon_rec
    ,con_rec.cd_atendimento
    ,paciente.nm_paciente
    ,reccon_rec.nm_responsavel
    ,reccon_rec.nr_id_responsavel
    ,paciente.dt_nascimento
    ,reccon_rec.dt_recebimento
    ,con_rec.dt_emissao
    ,reccon_rec.dt_estorno
    ,reccon_rec.cd_reduzido
    ,decode(reccon_rec.tp_recebimento,'1','Cheque',
    '2','Cartão','3','Dinheiro','4','Crédito C/C',
    '5','Borderô','6','Boleto','7','DOC',
    '8','Duplicata','9','Nota Promissória','B','Baixa Contábil') TP_RECEBIMENTO
    ,reccon_rec.cd_multi_empresa,
     paciente.cd_paciente,
     paciente.nm_paciente,
     Decode(paciente.tp_sexo,'M','Masc.','F','Fem.',paciente.tp_sexo) tp_sexo,
     paciente.email,
     paciente.nr_fone,
     paciente.ds_endereco,
     paciente.nr_endereco,
     paciente.dt_nascimento,
     paciente.nm_bairro,
     paciente.nr_cep,
     paciente.nr_cpf,
     cidade.nm_cidade,
     cidade.cd_uf
FROM dbamv.con_rec
    ,dbamv.itcon_rec
    ,dbamv.reccon_rec
    ,dbamv.atendime
    ,dbamv.paciente
    ,dbamv.processo
    ,dbamv.cidade
WHERE con_rec.cd_con_rec = itcon_rec.cd_con_rec
 AND itcon_rec.cd_itcon_rec = reccon_rec.cd_itcon_rec
 AND con_rec.cd_atendimento = atendime.cd_atendimento
 AND atendime.cd_paciente = paciente.cd_paciente
 AND Nvl(reccon_rec.vl_recebido,0) <> 0
 AND con_rec.cd_processo = processo.cd_processo
 AND con_rec.tp_con_rec = 'P'
 AND processo.cd_estrutural = '3.1.1.2.2'
 AND cidade.cd_cidade(+)    = paciente.cd_cidade
 AND reccon_rec.dt_recebimento BETWEEN :CAMPO_DT_INI
