--R_CONTA_REC_PAC
SELECT   *
    FROM (SELECT DISTINCT con_rec.nm_cliente nm_cliente1,
                          con_rec.ds_con_rec ds_con_rec1,
                          con_rec.cd_con_rec cd_con_rec1,
                          NVL (con_rec.vl_desconto, 0) vl_desconto1,
                          NVL (con_rec.vl_acrescimo, 0) vl_acrescimo1,
                          itcon_rec.cd_itcon_rec,
                          itcon_rec.dt_prevista_recebimento dt_prevista_pagamento1,
                          reccon_rec.tp_recebimento tp_recebimento1,
                          DECODE (reccon_rec.tp_recebimento,
                                  '1', 'CHEQUE',
                                  '3', 'DINHEIRO',
                                  '4', 'CREDITO C/C',
                                  '2', 'CARTAO',
                                  '5', 'BORDERO',
                                  '6', 'BOLETO',
                                  '7', 'DOC',
                                  '8', 'DUPLICATA',
                                  '9', 'PROMISSÓRIA',
                                  'E', 'ENCONTRO CONTA',
                                  'B', 'BAIXA CONTABIL'
                                 ) recebimento1,
                          Decode(Rank() OVER ( PARTITION BY con_rec.cd_con_rec ORDER BY reccon_rec.cd_reccon_rec),1,itcon_rec.vl_duplicata,0) vl_previsto1,
                          DECODE (itcon_rec.dt_cancelamento,
                                  NULL, reccon_rec.vl_recebido,
                                  0
                                 ) vl_recebido1,
                          con_rec.nr_documento nr_documento1,
                          con_rec.cd_atendimento cd_atendimento1,
                          atendime.tp_atendimento tp_atendimento1,
                          DECODE (con_rec.dt_cancelamento,
                                  NULL, itcon_rec.tp_quitacao,
                                  'L'
                                 ) status1,
                          itcon_rec.cd_moeda cd_moeda_prev1,
                          itcon_rec.vl_moeda vl_moeda_prev1,
                          reccon_rec.cd_moeda cd_moeda_rec1,
                          reccon_rec.vl_moeda vl_moeda_rec1,
                          con_rec.dt_lancamento dt_lancamento1,
                          reccon_rec.dt_recebimento dt_recebimento1,
                          reccon_rec.cd_reccon_rec cd_reccon_rec1,
                          itcon_rec.nr_codigo_barra codbarra1,
                          (SELECT nr_cpf
                             FROM dbamv.paciente
                            WHERE cd_paciente =
                                          (SELECT cd_paciente
                                             FROM dbamv.atendime
                                            WHERE cd_atendimento = con_rec.cd_atendimento))
                                                                                      cpf,
                          (SELECT ds_endereco || ' - ' || nm_bairro
                             FROM dbamv.paciente
                            WHERE cd_paciente =
                                          (SELECT cd_paciente
                                             FROM dbamv.atendime
                                            WHERE cd_atendimento = con_rec.cd_atendimento))
                                                                                 endereco
                     FROM dbamv.itcon_rec, dbamv.con_rec,
                          dbamv.atendime,
			 (SELECT DISTINCT
						r.cd_atendimento
					   ,r.cd_convenio
								FROM dbamv.reg_fat r
									,dbamv.atendime a
							WHERE r.cd_atendimento = a.cd_atendimento
								AND tp_atendimento = 'I'
                                 UNION ALL
				SELECT DISTINCT r.cd_atendimento,r.cd_convenio
								FROM dbamv.itreg_amb r
									 ,dbamv.atendime a
                            WHERE r.cd_atendimento = a.cd_atendimento
                                AND tp_atendimento <> 'I') amb,
			  (SELECT cd_itcon_rec,
                                  cd_moeda,
                                  vl_moeda,
                                  dt_recebimento,
                                  vl_recebido,
                                  cd_reccon_rec,
                                  tp_recebimento,
                                  cd_caixa,
                                  CD_DOC_CAIXA
                             FROM dbamv.reccon_rec
                            WHERE dt_estorno IS NULL
                      	      AND Trunc(dt_recebimento) <= :CAMPO_DATA_BASE
                              AND ((dt_estorno IS NULL) OR ((dt_estorno IS NOT NULL )
                              AND (Trunc(dt_estorno)    > :CAMPO_DATA_BASE)))
                             ) reccon_rec
                    WHERE con_rec.tp_con_rec = 'P'
                      AND con_rec.cd_previsao IS NULL
                      AND con_rec.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
                      AND con_rec.cd_atendimento = atendime.cd_atendimento(+)
                      AND itcon_rec.cd_con_rec = con_rec.cd_con_rec
                      AND reccon_rec.cd_itcon_rec(+) = itcon_rec.cd_itcon_rec
					  AND con_rec.cd_atendimento(+) = amb.cd_atendimento
                      AND :CAMPO_ORDEM = 'PACIENTE'
                  ORDER BY tp_atendimento,
                           nm_cliente1,
                           cd_atendimento1,
                           dt_prevista_pagamento1)
ORDER BY tp_atendimento1,
         nm_cliente1,
         ds_con_rec1,
         cd_con_rec1,
         dt_lancamento1,
         cd_moeda_prev1,
         vl_moeda_prev1,
         cd_atendimento1,
         dt_prevista_pagamento1,
         nr_documento1,
         cd_itcon_rec,
         status1,
         codbarra1,
         dt_recebimento1,
         cd_reccon_rec1

                             FROM dbamv.reccon_rec
                            WHERE dt_estorno IS NULL
                      	      AND Trunc(dt_recebimento) <= :CAMPO_DATA_BASE
                              AND ((dt_estorno IS NULL) OR ((dt_estorno IS NOT NULL )
                              AND (Trunc(dt_estorno)    > :CAMPO_DATA_BASE)))
                             ) reccon_rec
                    WHERE con_rec.tp_con_rec = 'P'
                      AND con_rec.cd_previsao IS NULL
                      AND con_rec.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
                      AND con_rec.cd_atendimento = atendime.cd_atendimento(+)
                      AND itcon_rec.cd_con_rec = con_rec.cd_con_rec
					  AND con_rec.cd_atendimento(+) = amb.cd_atendimento 	
                      AND itcon_rec.cd_itcon_rec = reccon_rec.cd_itcon_rec(+)
                      AND :CAMPO_ORDEM = 'VENCIMENTO'
                      :CAMPO_CF_CONVENIO
         )
ORDER BY tp_atendimento2,
         dt_prevista_pagamento2,
         codbarra2,
         dt_lancamento2,
         cd_moeda_prev2,
         vl_moeda_prev2,
         nm_cliente2,
         ds_con_rec2,
         cd_con_rec2,
         nr_documento2,
         status2,
