--R_FNFI_CNTA_FINANCEIRA
SELECT fina_conta.CD_CONTA_FINANCEIRA CODIGO, fina_conta.DS_CONTA DESCRICAO, fina_conta.TP_CONTA TIPO,fina_conta.NR_CONTA
    , fina_portador.DS_PORTADOR PESSOA
       , fina_conta.TP_FINANCEIRO,
         fina_conta.DH_ABERTURA, fina_conta.DH_FECHAMENTO,
        fina_conta.CD_REDUZIDO, fina_conta.CD_EMPRESA, fina_conta.CD_MOEDA, fina_conta.NM_GERENTE,fina_conta.CD_ORIGEM
FROM dbamv.fina_conta, dbamv.fina_portador, dbamv.fina_administradora
WHERE fina_portador.cd_portador = fina_conta.cd_portador
AND dbamv.fina_administradora.cd_administradora = fina_portador.cd_administradora
