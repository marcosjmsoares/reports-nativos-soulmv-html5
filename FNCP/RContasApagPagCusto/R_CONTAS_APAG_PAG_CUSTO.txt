SELECT nvl(sum( ( nvl(mov_concor.vl_movimentacao, 0) - nvl(Mov_concor.vl_conciliado, 0 ) ) ), 0) vl_total_extrato
  FROM dbamv.mov_concor
      ,dbamv.con_cor
      ,dbamv.lan_concor
      ,dbamv.empresa_lan_concor
      ,dbamv.item_res
      ,dbamv.gru_res
  WHERE mov_concor.dt_movimentacao >= :CAMPO_DT_INIC
    AND mov_concor.dt_movimentacao <= :CAMPO_DT_FIM
    AND con_cor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
    AND mov_concor.cd_con_cor = con_cor.cd_con_cor
    AND lan_concor.cd_lan_concor = mov_concor.cd_lan_concor
    AND lan_concor.cd_lan_concor = empresa_lan_concor.cd_lan_concor
    AND empresa_lan_concor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
    AND item_res.cd_gru_res = gru_res.cd_gru_res (+)
    AND empresa_lan_concor.cd_item_res = item_res.cd_item_res(+)
    AND ( (     mov_concor.cd_aplicacao IS NOT NULL
            AND lan_concor.tp_operacao_saldo_conta = '+' )
             OR mov_concor.cd_aplicacao IS NULL )
    AND mov_concor.cd_bordero IS NULL
    AND mov_concor.cd_con_cor_transferencia IS NULL
    AND Mov_concor.cd_mov_concor_transferencia IS NULL
    AND Mov_concor.cd_mov_caixa IS NULL
    AND Mov_concor.cd_pagcon_pag IS NULL
    AND Mov_concor.cd_cheque IS NULL
    And Lan_concor.tp_operacao_saldo_conta = '-'
    AND Mov_concor.vl_movimentacao - nvl(Mov_concor.vl_conciliado,0) > 0
    AND (    not exists( SELECT cd_mov_concor
                           FROM    dbamv.rec_mov_con r
                           WHERE   r.cd_mov_concor = Mov_concor.cd_mov_concor)
          OR  exists ( SELECT  'x'
                        FROM  dbamv.config_financ
                        WHERE cd_lanca_credito_convenio = lan_concor.cd_lan_concor
                          AND cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA) )
