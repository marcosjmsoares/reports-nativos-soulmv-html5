--R_RECURSO_GLOSA_ANALITICO
SELECT * FROM (
SELECT CONVENIO.NM_CONVENIO   NM_CONVENIO
      ,TRUNC( ITCON_REC.DT_PREVISTA_RECEBIMENTO )  DT_PREVISTA_PAGAMENTO
      ,DECODE(AGRUP.CD_CON_REC_AGRUP,NULL,ITCON_REC.VL_DUPLICATA,
                 (AGRUP.VL_DUPLICATA /TOT_AGRUP.VL_TOT_DUPLICATA) * ITCON_REC.VL_DUPLICATA )    VL_PREVISTO
      ,SUM(DECODE(AGRUP.CD_CON_REC_AGRUP,NULL,RECCON_REC.VL_RECEBIDO,
                 (AGRUP.VL_DUPLICATA /TOT_AGRUP.VL_TOT_DUPLICATA) * RECCON_REC.VL_RECEBIDO ) )VL_RECEBIDO
      , DECODE(AGRUP.CD_CON_REC_AGRUP,NULL,NVL(ITCON_REC.VL_GLOSA,0) ,
                 (AGRUP.VL_DUPLICATA /TOT_AGRUP.VL_TOT_DUPLICATA) * (NVL(ITCON_REC.VL_GLOSA,0) )) VL_GLOSA
      , DECODE(AGRUP.CD_CON_REC_AGRUP,NULL,NVL(ITCON_REC.VL_RECURSO,0) ,
                 (AGRUP.VL_DUPLICATA /TOT_AGRUP.VL_TOT_DUPLICATA) * (NVL(ITCON_REC.VL_RECURSO,0) )) VL_RECURSO
      ,CON_REC.NR_DOCUMENTO
      ,CON_REC.CD_REMESSA  CD_REMESSA
      ,CON_REC.CD_CON_REC
      ,ITCON_REC.CD_ITCON_REC
      ,DECODE(AGRUP.CD_CON_REC_AGRUP,NULL,'N','S') Agrupada
FROM   (SELECT PAR.CD_CON_REC_AGRUP , REC.CD_REMESSA , SUM(PAR.VL_DUPLICATA) VL_DUPLICATA
	FROM  DBAMV.CON_REC REC , DBAMV.ITCON_REC PAR
	WHERE PAR.CD_CON_REc_AGRUP IS NOT NULL
	AND   REC.CD_CON_REC        = PAR.CD_CON_REC
	GROUP BY PAR.CD_CON_REC_AGRUP , REC.CD_REMESSA) AGRUP ,
	(SELECT PAR.CD_CON_REC_AGRUP , SUM(PAR.VL_DUPLICATA) VL_TOT_DUPLICATA
	FROM  DBAMV.ITCON_REC PAR
	WHERE PAR.CD_CON_REc_AGRUP IS NOT NULL
	GROUP BY PAR.CD_CON_REC_AGRUP)     TOT_AGRUP ,
        DBAMV.ITCON_REC
      ,DBAMV.CON_REC
     ,DBAMV.REMESSA_FATURA
     ,DBAMV.FATURA
     ,DBAMV.CONVENIO
     ,DBAMV.RECCON_REC
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And NVL(AGRUP.CD_REMESSA,CON_REC.CD_REMESSA) = REMESSA_FATURA.CD_REMESSA
  AND  REMESSA_FATURA.CD_FATURA    =  FATURA.CD_FATURA
  AND  FATURA.CD_CONVENIO          =  CONVENIO.CD_CONVENIO
  AND  ITCON_REC.CD_ITCON_REC          =  RECCON_REC.CD_ITCON_REC
  AND  CON_REC.TP_CON_REC          = 'C'
  AND  CON_REC.CD_PREVISAO IS NULL
  AND  NVL(RECCON_REC.SN_RECURSO,'N') = 'S'
  AND  ITCON_REC.CD_CON_REC =   CON_REC.CD_CON_REC
  AND  TRUNC(ITCON_REC.DT_PREVISTA_RECEBIMENTO)  BETWEEN :CAMPO_DATA_INI AND 
 AND     ITCON_REC.CD_CON_REC_AGRUP IS NULL
AND     AGRUP.CD_CON_REC_AGRUP     (+)= CON_REC.CD_CON_REC
AND     TOT_AGRUP.CD_CON_REC_AGRUP (+)= cON_REC.CD_CON_REC
GROUP BY  CONVENIO.NM_CONVENIO
      ,TRUNC( ITCON_REC.DT_PREVISTA_RECEBIMENTO )
      ,DECODE(AGRUP.CD_CON_REC_AGRUP,NULL,ITCON_REC.VL_DUPLICATA,
                 (AGRUP.VL_DUPLICATA /TOT_AGRUP.VL_TOT_DUPLICATA) * ITCON_REC.VL_DUPLICATA )
      , DECODE(AGRUP.CD_CON_REC_AGRUP,NULL,NVL(ITCON_REC.VL_GLOSA,0) ,
                 (AGRUP.VL_DUPLICATA /TOT_AGRUP.VL_TOT_DUPLICATA) * (NVL(ITCON_REC.VL_GLOSA,0) ))
      , DECODE(AGRUP.CD_CON_REC_AGRUP,NULL,NVL(ITCON_REC.VL_RECURSO,0) ,
                 (AGRUP.VL_DUPLICATA /TOT_AGRUP.VL_TOT_DUPLICATA) * (NVL(ITCON_REC.VL_RECURSO,0) ))
      ,CON_REC.NR_DOCUMENTO
      ,CON_REC.CD_REMESSA
      ,CON_REC.CD_CON_REC
      ,ITCON_REC.CD_ITCON_REC
      ,DECODE(AGRUP.CD_CON_REC_AGRUP,NULL,'N','S')
ORDER BY NM_CONVENIO, CD_REMESSA, TRUNC( ITCON_REC.DT_PREVISTA_RECEBIMENTO )
)
ORDER BY NM_CONVENIO
