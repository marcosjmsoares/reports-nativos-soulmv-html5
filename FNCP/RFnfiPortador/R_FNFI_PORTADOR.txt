--R_FNFI_PORTADOR
SELECT fina_portador.CD_PORTADOR CODIGO,fina_portador.DS_PORTADOR DESCRICAO,
      fina_portador.TP_PORTADOR tipo,
      fina_portador.NR_AGENCIA NR_AGENCIA,
      Nvl(fornecedor.nm_fornecedor,fornecedor.cd_fornecedor || ' - ' || fornecedor.nm_fornecedor)  PESSOA,
      Nvl(fina_administradora.DS_ADMINISTRADORA,fina_administradora.CD_ADMINISTRADORA || ' - ' || fina_administradora.DS_ADMINISTRADORA) ADMINISTRADORA
FROM dbamv.fina_portador fina_portador , dbamv.fina_administradora fina_administradora , dbamv.fornecedor fornecedor
WHERE   fina_administradora.CD_ADMINISTRADORA = fina_portador.CD_ADMINISTRADORA
AND fornecedor.cd_fornecedor(+) = fina_portador.cd_pessoa
