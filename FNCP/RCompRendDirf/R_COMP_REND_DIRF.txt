--R_COMP_REND_DIRF
select  nr_cgc_cpf,
        tp_fornec,
        sum(vl_detalhamento) total_imp_retido,
        sum(vl_rendimento)   total_rendimento,
        sum(vl_deducao)      total_inss
from dbamv.dirf
where dirf.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
and nr_cgc_cpf <> 0
and nr_cgc_cpf is not null
group by nr_cgc_cpf, tp_fornec
order by nr_cgc_cpf
