--R_CON_APAG_REP
SELECT * FROM (
select  rp.cd_prestador_repasse
      , pr.nm_prestador
      , cp.cd_con_pag
      , sum(rp.vl_faturado)    vl_faturado
      , cp.vl_bruto_conta     vl_bruto_conta
      , cp.nr_documento
      , cp.dt_emissao
      , it.dt_vencimento
  from dbamv.con_pag cp
     , dbamv.itcon_pag it
     , dbamv.repasse_prestador rp
     , dbamv.prestador pr
 where rp.cd_con_pag             = cp.cd_con_pag
   and it.cd_con_pag           = cp.cd_con_pag
   and rp.cd_prestador_repasse = pr.cd_prestador
   and cp.cd_con_pag_dev is null
   and cp.vl_bruto_conta > 0
   and it.tp_quitacao  in ('V', 'C', 'L', 'P')
   and it.cd_con_pag_agrup is null
   and nvl(it.vl_duplicata,0) > 0
   and cp.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
group by  rp.cd_prestador_repasse
             , pr.nm_prestador
             , cp.cd_con_pag
             , cp.vl_bruto_conta
             , cp.nr_documento
             , cp.dt_emissao
             , it.dt_vencimento
order by pr.nm_prestador
)
ORDER BY CD_PRESTADOR_REPASSE,NM_PRESTADOR,
DT_EMISSAO
