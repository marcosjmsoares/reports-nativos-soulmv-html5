--R_CON_RECEB_959
SELECT * FROM (
SELECT  TRUNC( MVTO.DT_MOVIMENTACAO )                                                           DT_MVTO_DT
        ,FORNECEDOR.CD_FORNECEDOR                                                        CD_CLIENTE_DT
        ,DECODE( CON_REC.CD_FORNECEDOR, '', CON_REC.NM_CLIENTE, FORNECEDOR.NM_FANTASIA ) NM_CLIENTE_DT
        ,LTRIM( RTRIM( HIST_PADRAO.DS_HISTORICO_PADRAO ) || ' ' || CON_REC.DS_CON_REC )  DS_CON_REC_DT
        ,DECODE( CON_REC.TP_CON_REC, 'C', 'CONVENIO', 'P', 'PACIENTE', 'F', 'FUNCIONARIO',
                                     'A', 'FIN. CARTAO CREDITO', 'DIVERSOS' )            TP_CON_REC_DT
        ,CTA_RESULT.DS_CONTA                                                             DS_CONTA_RESULTADO_DT
        ,PC.DS_CONTA                                                                     DS_CONTA_CONTABIL_DT
        ,CON_REC.NR_DOCUMENTO                                                            NR_DOCUMENTO_DT
        ,CON_REC.VL_PREVISTO                                                             VL_PREVISTO_DT
        ,MVTO.VL_MOVIMENTACAO                                                         VL_RECEBIDO_DT
        ,CON_REC.VL_ACRESCIMO                                                            VL_ACRESCIMO_DT
        ,CON_REC.VL_DESCONTO                                                             VL_DESCONTO_DT
FROM
         DBAMV.CON_REC           CON_REC
        ,DBAMV.ITCON_REC        ITCON_REC
        ,DBAMV.RECCON_REC    RECCON_REC
        ,DBAMV.MOV_CONCOR    MVTO
        ,DBAMV.REC_MOV_CON  REC_MOV_CON
        ,DBAMV.FORNECEDOR
        ,DBAMV.HISTORICO_PADRAO  HIST_PADRAO
        ,DBAMV.PLANO_CONTAS      PC
        ,DBAMV.PLANO_CONTAS      CTA_RESULT
WHERE                                               = 2
AND      ITCON_REC.TP_QUITACAO                       IN ( 'Q', 'P','G' )
AND      CON_REC.CD_PREVISAO IS NULL
AND      CON_REC.CD_MULTI_EMPRESA           = :CAMPO_CD_MULTI_EMPRESA
AND      CON_REC.CD_CON_REC                        = ITCON_REC.CD_CON_REC
AND      ITCON_REC.CD_ITCON_REC                 = RECCON_REC.CD_ITCON_REC
AND      RECCON_REC.CD_RECCON_REC         = REC_MOV_CON.CD_RECCON_REC
AND      MVTO.CD_MOV_CONCOR                      = REC_MOV_CON.CD_MOV_CONCOR
AND      CON_REC.CD_FORNECEDOR                 = FORNECEDOR.CD_FORNECEDOR(+)
AND      CON_REC.CD_HISTORICO_PADRAO     = HIST_PADRAO.CD_HISTORICO_PADRAO(+)
AND      CON_REC.CD_REDUZIDO                       = PC.CD_REDUZIDO(+)
AND      PC.CD_REDUZIDO_RESULTADO            = CTA_RESULT.CD_REDUZIDO(+)
:CAMPO_CF_DATA
)
ORDER BY DT_MVTO_DT,
TP_CON_REC_DT
