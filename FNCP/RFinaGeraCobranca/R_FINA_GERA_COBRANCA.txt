--R_FINA_GERA_COBRANCA
SELECT c.nr_documento, (c.cd_fornecedor)||' - '||(f.nm_fornecedor) AS ds_fornecedor ,
         i.DT_VENCIMENTO, i.DT_NOVO_VENCIMENTO, i.VL_BRUTO,i.VL_DESCONTO, i.VL_ACRESCIMO,
         i.CD_CONTA_FINANCEIRA, i.CD_NOVA_CONTA, i.TP_REGISTRO, i.CD_DOCUMENTO
  FROM dbamv.fina_lote_cobr_it i, dbamv.con_rec c , dbamv.fornecedor f, dbamv.itcon_rec r
  WHERE r.cd_con_rec = c.cd_con_rec
  AND  i.cd_itcon_rec = r.cd_itcon_rec
  AND f.cd_fornecedor = c.cd_fornecedor
  AND i.CD_LOTE_COBR = :CAMPO_CF_FILTROS
ORDER BY i.DT_VENCIMENTO
