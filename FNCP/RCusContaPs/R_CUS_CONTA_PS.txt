--R_CUS_CONTA_PS
SELECT * FROM (
SELECT   PS.CD_REDUZIDO_PS  CD_CONTA
        ,V.DATA             DATA
        ,nvl(PS.DS_CONTA_PS,'NÃO INFORMADA')   DS_CONTA
        ,SUM( V.VL_ENTRADA ) VL_ENTRADA
        ,SUM( V.VL_SAIDA   ) VL_SAIDA
FROM     DBAMV.V_CUSTO_CONTA    V
        ,DBAMV.PLANO_SINTETICO  PS
WHERE    V.CD_MULTI_EMPRESA  = :CAMPO_CD_MULTI_EMPRESA
AND      TRUNC( V.DATA ) BETWEEN :CAMPO_DT_INICIAL and 
AND      V.CD_REDUZIDO_PS = PS.CD_REDUZIDO_PS(+)
GROUP BY  PS.CD_REDUZIDO_PS
         ,V.DATA
         ,PS.DS_CONTA_PS
ORDER BY PS.CD_REDUZIDO_PS
)
ORDER BY CD_CONTA,DATA,DS_CONTA,VL_ENTRADA,VL_SAIDA
