--R_RECURSO_GLOSA_ANALI
SELECT * FROM (
SELECT CONVENIO.NM_CONVENIO   NM_CONVENIO
	,TRUNC( CON_REC.DT_EMISSAO)              DT_EMISSAO
	,TRUNC( ITCON_REC.DT_PREVISTA_RECEBIMENTO )  DT_PREVISTA_PAGAMENTO
	,DECODE(AGRUP.CD_CON_REC_AGRUP,NULL,ITCON_REC.VL_DUPLICATA,
	   (AGRUP.VL_DUPLICATA /TOT_AGRUP.VL_TOT_DUPLICATA) * ITCON_REC.VL_DUPLICATA )      VL_PREVISTO
	,Nvl(Decode(AGRUP.CD_CON_REC_AGRUP,NULL,ITCON_REC.VL_GLOSA,
               ( AGRUP.VL_DUPLICATA /TOT_AGRUP.VL_TOT_DUPLICATA) * ITCON_REC.VL_GLOSA ),0)       VL_GLOSA
	,DECODE(AGRUP.CD_CON_REC_AGRUP,NULL,ITCON_REC.VL_RECURSO,
	   (AGRUP.VL_DUPLICATA /TOT_AGRUP.VL_TOT_DUPLICATA) * ITCON_REC.VL_RECURSO )      VL_RECURSO
	,CON_REC.NR_DOCUMENTO
	,NVL(CON_REC.CD_REMESSA,0)   CD_REMESSA
	,CON_REC.CD_CON_REC
	,ITCON_REC.CD_ITCON_REC
	,DECODE(AGRUP.CD_CON_REC_AGRUP,NULL,'N','S') Agrupada
      ,Nvl((SELECT Sum(dbamv.reccon_rec.vl_recebido)
             FROM dbamv.reccon_rec
            WHERE NVL(sn_recurso, 'N') = 'N'
              AND cd_itcon_rec = itcon_rec.cd_itcon_rec),0) vl_recebido
      ,Nvl((SELECT Sum(dbamv.reccon_rec.vl_recebido)
      FROM dbamv.reccon_rec
      WHERE NVL(sn_recurso, 'N') = 'S'
      AND cd_itcon_rec = itcon_rec.cd_itcon_rec),0) vl_recebido_recurso
FROM (SELECT PAR.CD_CON_REC_AGRUP , REC.CD_REMESSA , SUM(PAR.VL_DUPLICATA) VL_DUPLICATA
	FROM  DBAMV.CON_REC REC , DBAMV.ITCON_REC PAR
	WHERE PAR.CD_CON_REc_AGRUP IS NOT NULL
	AND   REC.CD_CON_REC        = PAR.CD_CON_REC
	GROUP BY PAR.CD_CON_REC_AGRUP , REC.CD_REMESSA) AGRUP ,
	(SELECT PAR.CD_CON_REC_AGRUP , SUM(PAR.VL_DUPLICATA) VL_TOT_DUPLICATA
	FROM  DBAMV.ITCON_REC PAR
	WHERE PAR.CD_CON_REc_AGRUP IS NOT NULL
	GROUP BY PAR.CD_CON_REC_AGRUP)     TOT_AGRUP ,
	 DBAMV.ITCON_REC
	, DBAMV.CON_REC
	,DBAMV.REMESSA_FATURA
	,DBAMV.FATURA
	,DBAMV.CONVENIO
	,DBAMV.RECCON_REC
	,Dbamv.Empresa_Convenio
	Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
	And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
	And NVL(AGRUP.CD_REMESSA,CON_REC.CD_REMESSA) = REMESSA_FATURA.CD_REMESSA
	AND  REMESSA_FATURA.CD_FATURA    =   FATURA.CD_FATURA
	AND  FATURA.CD_CONVENIO          = CONVENIO.CD_CONVENIO
	AND  ITCON_REC.CD_ITCON_REC          =  RECCON_REC.CD_ITCON_REC
	AND  CON_REC.TP_CON_REC          = 'C'
	AND  CON_REC.CD_PREVISAO IS NULL
	AND  ITCON_REC.CD_CON_REC =   CON_REC.CD_CON_REC
	AND     AGRUP.CD_CON_REC_AGRUP     (+)= CON_REC.CD_CON_REC
	AND     TOT_AGRUP.CD_CON_REC_AGRUP (+)= cON_REC.CD_CON_REC
GROUP BY CONVENIO.NM_CONVENIO
        ,TRUNC( CON_REC.DT_EMISSAO )
        ,TRUNC( ITCON_REC.DT_PREVISTA_RECEBIMENTO)
        ,DECODE(AGRUP.CD_CON_REC_AGRUP,NULL,ITCON_REC.VL_DUPLICATA,
                 (AGRUP.VL_DUPLICATA /TOT_AGRUP.VL_TOT_DUPLICATA) * ITCON_REC.VL_DUPLICATA )
        ,Nvl(Decode(AGRUP.CD_CON_REC_AGRUP,NULL,ITCON_REC.VL_GLOSA,
               ( AGRUP.VL_DUPLICATA /TOT_AGRUP.VL_TOT_DUPLICATA) * ITCON_REC.VL_GLOSA ),0)
        ,DECODE(AGRUP.CD_CON_REC_AGRUP,NULL,ITCON_REC.VL_RECURSO,
                 (AGRUP.VL_DUPLICATA /TOT_AGRUP.VL_TOT_DUPLICATA) * ITCON_REC.VL_RECURSO )
        ,CON_REC.NR_DOCUMENTO
        ,CON_REC.CD_REMESSA
        ,CON_REC.CD_CON_REC
        ,ITCON_REC.CD_ITCON_REC
      ,DECODE(AGRUP.CD_CON_REC_AGRUP,NULL,'N','S')
UNION
SELECT CONVENIO.NM_CONVENIO NM_CONVENIO
      ,TRUNC( CON_REC.DT_EMISSAO) DT_EMISSAO
      ,TRUNC( ITCON_REC.DT_PREVISTA_RECEBIMENTO ) DT_PREVISTA_PAGAMENTO
      ,DECODE(AGRUP.CD_CON_REC_AGRUP,NULL,ITCON_REC.VL_DUPLICATA,
       (AGRUP.VL_DUPLICATA /TOT_AGRUP.VL_TOT_DUPLICATA) * ITCON_REC.VL_DUPLICATA ) VL_PREVISTO
      ,Nvl(Decode(AGRUP.CD_CON_REC_AGRUP,NULL,ITCON_REC.VL_GLOSA,
               ( AGRUP.VL_DUPLICATA /TOT_AGRUP.VL_TOT_DUPLICATA) * ITCON_REC.VL_GLOSA ),0)      VL_GLOSA
      ,DECODE(AGRUP.CD_CON_REC_AGRUP,NULL,ITCON_REC.VL_RECURSO,
       (AGRUP.VL_DUPLICATA /TOT_AGRUP.VL_TOT_DUPLICATA) * ITCON_REC.VL_RECURSO ) VL_RECURSO
      ,CON_REC.NR_DOCUMENTO
      ,0 CD_REMESSA
      ,CON_REC.CD_CON_REC
      ,ITCON_REC.CD_ITCON_REC
      ,DECODE(AGRUP.CD_CON_REC_AGRUP,NULL,'N','S') Agrupada
      ,Nvl((SELECT Sum(dbamv.reccon_rec.vl_recebido)
             FROM dbamv.reccon_rec
            WHERE NVL(sn_recurso, 'N') = 'N'
              AND cd_itcon_rec = itcon_rec.cd_itcon_rec),0) vl_recebido
      ,Nvl((SELECT Sum(dbamv.reccon_rec.vl_recebido)
      FROM dbamv.reccon_rec
      WHERE NVL(sn_recurso, 'N') = 'S'
      AND cd_itcon_rec = itcon_rec.cd_itcon_rec),0) vl_recebido_recurso
FROM   (SELECT PAR.CD_CON_REC_AGRUP, ITF.CD_REMESSA, SUM(PAR.VL_DUPLICATA) VL_DUPLICATA
          FROM DBAMV.CON_REC REC, DBAMV.ITCON_REC PAR,
               DBAMV.NOTA_FISCAL NF, DBAMV.ITFAT_NOTA_FISCAL ITF
         WHERE PAR.CD_CON_REC_AGRUP IS NOT NULL
	       AND REC.CD_CON_REC     = PAR.CD_CON_REC
	       AND REC.CD_NOTA_FISCAL = NF.CD_NOTA_FISCAL
	       AND NF.CD_NOTA_FISCAL  = ITF.CD_NOTA_FISCAL
	     GROUP BY PAR.CD_CON_REC_AGRUP, ITF.CD_REMESSA) AGRUP,
       (SELECT PAR.CD_CON_REC_AGRUP, SUM(PAR.VL_DUPLICATA) VL_TOT_DUPLICATA
	      FROM DBAMV.ITCON_REC PAR
	     WHERE PAR.CD_CON_REC_AGRUP IS NOT NULL
	     GROUP BY PAR.CD_CON_REC_AGRUP) TOT_AGRUP
       ,DBAMV.ITCON_REC
       ,DBAMV.CON_REC
       ,DBAMV.REMESSA_FATURA
       ,DBAMV.FATURA
       ,DBAMV.CONVENIO
       ,DBAMV.RECCON_REC
       ,Dbamv.Empresa_Convenio
       ,DBAMV.NOTA_FISCAL NF
       ,DBAMV.ITFAT_NOTA_FISCAL ITF
Where Empresa_Convenio.Cd_Convenio             = Convenio.Cd_Convenio
  And Empresa_Convenio.Cd_Multi_Empresa        = :CAMPO_CD_MULTI_EMPRESA
  And NVL(AGRUP.CD_REMESSA,ITF.CD_REMESSA) = REMESSA_FATURA.CD_REMESSA
  AND REMESSA_FATURA.CD_FATURA  = FATURA.CD_FATURA
  AND FATURA.CD_CONVENIO        = CONVENIO.CD_CONVENIO
  AND ITCON_REC.CD_ITCON_REC    =  RECCON_REC.CD_ITCON_REC
  AND CON_REC.TP_CON_REC        = 'C'
  AND CON_REC.CD_PREVISAO IS NULL
  AND ITCON_REC.CD_CON_REC      =   CON_REC.CD_CON_REC
  AND AGRUP.CD_CON_REC_AGRUP     (+) = CON_REC.CD_CON_REC
  AND TOT_AGRUP.CD_CON_REC_AGRUP (+) = cON_REC.CD_CON_REC
  AND CON_REC.CD_NOTA_FISCAL = NF.CD_NOTA_FISCAL
  AND NF.CD_NOTA_FISCAL = ITF.CD_NOTA_FISCAL
  AND NF.CD_CONVENIO = CONVENIO.CD_CONVENIO
GROUP BY CONVENIO.NM_CONVENIO
        ,TRUNC( CON_REC.DT_EMISSAO )
        ,TRUNC( ITCON_REC.DT_PREVISTA_RECEBIMENTO)
        ,DECODE(AGRUP.CD_CON_REC_AGRUP,NULL,ITCON_REC.VL_DUPLICATA,
                 (AGRUP.VL_DUPLICATA /TOT_AGRUP.VL_TOT_DUPLICATA) * ITCON_REC.VL_DUPLICATA )
        ,Nvl(Decode(AGRUP.CD_CON_REC_AGRUP,NULL,ITCON_REC.VL_GLOSA,
               ( AGRUP.VL_DUPLICATA /TOT_AGRUP.VL_TOT_DUPLICATA) * ITCON_REC.VL_GLOSA ),0)
        ,DECODE(AGRUP.CD_CON_REC_AGRUP,NULL,ITCON_REC.VL_RECURSO,
                 (AGRUP.VL_DUPLICATA /TOT_AGRUP.VL_TOT_DUPLICATA) * ITCON_REC.VL_RECURSO )
        ,CON_REC.NR_DOCUMENTO
        ,ITF.CD_REMESSA
        ,CON_REC.CD_CON_REC
        ,ITCON_REC.CD_ITCON_REC
      ,DECODE(AGRUP.CD_CON_REC_AGRUP,NULL,'N','S')
ORDER BY NM_CONVENIO, CD_REMESSA,  DT_PREVISTA_PAGAMENTO
)
ORDER BY NM_CONVENIO, CD_REMESSA,  DT_PREVISTA_PAGAMENTO
