--R_GRURES
SELECT * FROM (
SELECT  GRU_RES.CD_GRU_RES Codigo
                ,GRU_RES.DS_GRU_RES Descricao
                ,DECODE( GRU_RES.TP_OPERACIONAL, 'S', 'Operacionais', 'Não Operacionais') TP_OPERACIONAL
FROM    dbamv.GRU_RES GRU_RES
)
ORDER BY TP_OPERACIONAL DESC, :CAMPO_ORDEM
