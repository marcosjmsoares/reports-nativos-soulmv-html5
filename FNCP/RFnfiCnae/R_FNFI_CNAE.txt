--R_FNFI_CNAE
SELECT pai.cd_cnae        as CD_CNAE_PAI,
						pai.nr_cnae        as NR_CNAE_PAI,
						pai.ds_cnae	       as DS_CNAE_PAI,
						pai.sn_icms        as SN_ICMS_PAI,
						pai.sn_iss  	   as SN_ISS_PAI,
						pai.tp_registro    as TP_REGISTRO_PAI,	
						filho.cd_cnae      as CD_CNAE_FILHO,
						filho.nr_cnae      as NR_CNAE_FILHO,
						filho.ds_cnae      as DS_CNAE_FILHO,
						filho.sn_icms      as SN_ICMS_FILHO,
						filho.sn_iss  	   as SN_ISS_FILHO,
						filho.tp_registro  as TP_REGISTRO_FILHO,
						filho.sn_bloqueado as SN_BLOQUEADO
						FROM dbamv.cnae filho,
						dbamv.cnae pai
						WHERE filho.cd_pai = pai.cd_cnae (+)
