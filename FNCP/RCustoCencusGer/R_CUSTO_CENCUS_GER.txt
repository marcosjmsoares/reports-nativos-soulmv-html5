--R_CUSTO_CENCUS_GER
SELECT * FROM (
SELECT   V.CD_CONTA
        ,V.DS_CONTA
        ,V.DATA
        ,V.HISTORICO
        ,V.NR_DOC
        ,sum(V.VL_ENTRADA) vl_entrada
        ,sum(V.VL_SAIDA)        vl_saida
        ,V.SITUACAO
FROM     DBAMV.V_CUSTO_CONTA   V
WHERE    V.CD_MULTI_EMPRESA   =  :CAMPO_CD_MULTI_EMPRESA
AND ( TRUNC( V.DATA ) BETWEEN :CAMPO_DT_INICIAL and :CAMPO_DT_FINAL )
group by
  V.CD_CONTA
        ,V.DS_CONTA
        ,V.DATA
        ,V.HISTORICO
        ,V.NR_DOC
        ,V.SITUACAO
UNION ALL
SELECT   V.CD_CONTA
        ,V.DS_CONTA
        ,V.DATA
        ,V.HISTORICO
        ,V.NR_DOC
        ,sum(V.VL_ENTRADA) vl_entrada
        ,sum(V.VL_SAIDA)       vl_saida
        ,V.SITUACAO
FROM     DBAMV.V_CUSTO_CONTA_PREV   V
WHERE    V.CD_MULTI_EMPRESA   =  :CAMPO_CD_MULTI_EMPRESA
AND ( TRUNC( V.DATA ) BETWEEN :CAMPO_DT_INICIAL and :CAMPO_DT_FINAL )
group by
  V.CD_CONTA
        ,V.DS_CONTA
        ,V.DATA
        ,V.HISTORICO
        ,V.NR_DOC
        ,V.SITUACAO
)
ORDER BY DS_CONTA,CD_CONTA
