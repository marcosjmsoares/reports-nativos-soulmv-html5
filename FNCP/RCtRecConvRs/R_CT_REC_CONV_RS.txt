--R_CT_REC_CONV_RS_R_ATENDIMENTO1_Q_SEQUENCIAL_PRE_IMPRESSO
SELECT cd_nota_fiscal      cd_nota_fiscal_seq,
       cd_conta            cd_conta_seq,
       cd_reg_fat          cd_reg_fat_seq,
       cd_remessa          cd_remessa_seq,
       cd_atendimento      cd_atendimento_seq,
       nr_seq_preimpre     nr_seq_preimpre_ate,
       tp_seq_preimpre,
       nr_seq_preimpre_itamb,
       nr_seq_preimpre_regf,
       Sum(vl_itfat_nf)    vl_previsto_seq,
       Sum(vl_recebido + vl_recebido_recurso)    vl_recebido_seq_proc,
       Sum(vl_imposto)     vl_imposto_retido_seq_proc,
       Sum(vl_desconto)    vl_desconto_seq_proc,
       Sum(vl_acrescimo)   vl_acrescimo_seq_proc,
       Sum(vl_glosa)       vl_glosa_seq,
       Sum(vl_recurso)     vl_recurso_seq,
       (CASE
          WHEN (Trunc(Max(dt_recebimento))  >=  Nvl(Max(dt_recebido_recurso), Trunc(Max(dt_recebimento))))
            THEN Max(dt_recebimento)
          ELSE Max(dt_recebido_recurso)
        END) dt_recebimento_seq_proc,
       Sum(vl_aceite)      vl_aceite_seq_proc,
       Sum(vl_itfat_nf - (vl_liquido + vl_aceite + vl_recebido_recurso)) vl_receber_seq_proc
  FROM ( SELECT ifat.cd_nota_fiscal,
                DECODE(ifat.cd_reg_fat, NULL, 'A', 'I') tp_conta,
                NVL(ifat.cd_reg_fat, ifat.cd_reg_amb) cd_conta,
                ifat.cd_reg_fat,
                ifat.cd_atendimento,
                ifat.cd_remessa,
                ifat.cd_pro_fat,
                ifat.cd_prestador,
                prof.ds_pro_fat,
                grup.cd_gru_pro,
                grup.ds_gru_pro,
                gruf.cd_gru_fat,
                gruf.ds_gru_fat,
                paci.nm_paciente,
                conv.tp_seq_preimpre,
                ated.nr_seq_preimpre,
                (SELECT nr_seq_preimpre
                   FROM dbamv.reg_fat
                  WHERE cd_reg_fat = ifat.cd_reg_fat
                    AND ROWNUM = 1) nr_seq_preimpre_regf,
                (SELECT nr_seq_preimpre
                   FROM dbamv.itreg_amb
                  WHERE cd_reg_amb = ifat.cd_reg_fat
                    AND cd_lancamento = ifat.cd_lancamento_amb
                    AND cd_convenio = conv.cd_convenio
                    AND ROWNUM = 1) nr_seq_preimpre_itamb,
                Nvl(ifat.vl_itfat_nf,0)  vl_itfat_nf,
                irec.dt_recebimento,
                Nvl(irec.vl_recebido,0)  vl_recebido,
                Nvl(irec.vl_acrescimo,0) vl_acrescimo,
                Nvl(irec.vl_desconto,0)  vl_desconto,
                Nvl(irec.vl_imposto,0)   vl_imposto,
                Nvl(irec.vl_glosa,0)     vl_glosa,
                Nvl(irec.vl_liquido,0)   vl_liquido,
                Nvl((SELECT Sum(Nvl(vl_reapresentado,0))
                        FROM dbamv.glosas
                      WHERE NVL(cd_reg_fat, cd_reg_amb) = NVL(ifat.cd_reg_fat, ifat.cd_reg_amb)
                        AND cd_atendimento = ifat.cd_atendimento
                        AND cd_itfat_nf = ifat.cd_itfat_nf
                        AND cd_glosas_pai IS NULL
                        AND cd_remessa_glosa IS NOT NULL),0) vl_recurso,
                (SELECT Max(dt_recebido)
                        FROM dbamv.glosas
                      WHERE NVL(cd_reg_fat, cd_reg_amb) = NVL(ifat.cd_reg_fat, ifat.cd_reg_amb)
                        AND cd_atendimento = ifat.cd_atendimento
                        AND cd_itfat_nf = ifat.cd_itfat_nf
                        AND cd_remessa_glosa IS NOT NULL) dt_recebido_recurso,
                Nvl((SELECT Sum(Nvl(vl_recebido,0))
                        FROM dbamv.glosas
                      WHERE NVL(cd_reg_fat, cd_reg_amb) = NVL(ifat.cd_reg_fat, ifat.cd_reg_amb)
                        AND cd_atendimento = ifat.cd_atendimento
                        AND cd_itfat_nf = ifat.cd_itfat_nf
                        AND cd_remessa_glosa IS NOT NULL),0) vl_recebido_recurso,
                Nvl((SELECT Sum(Nvl(vl_glosa,0))
                        FROM dbamv.itmov_glosas
                      WHERE cd_itfat_nf = ifat.cd_itfat_nf),0) vl_aceite,
                Nvl((SELECT Sum(Nvl(vl_retira_recurso,0))
                        FROM dbamv.itmov_glosas
                      WHERE cd_itfat_nf = ifat.cd_itfat_nf),0) vl_retira_recurso
            FROM dbamv.itfat_nota_fiscal ifat,
                 dbamv.nota_fiscal       nota,
                 dbamv.convenio          conv,
                 dbamv.pro_fat           prof,
                 dbamv.gru_pro           grup,
                 dbamv.gru_fat           gruf,
                 dbamv.atendime          ated,
                 dbamv.paciente          paci,
                 (SELECT t.cd_itfat_nf,
                         Max(r.dt_recebimento)         dt_recebimento,
                         Sum(Nvl(t.vl_recebido,0))       vl_recebido,
                         Sum(Nvl(t.vl_acrescimo,0))      vl_acrescimo,
                         Sum(Nvl(t.vl_desconto,0))       vl_desconto,
                         Sum(Nvl(t.vl_imposto_retido,0)) vl_imposto,
                         Sum(Nvl(t.vl_glosa,0))          vl_glosa,
                         Sum(  Nvl(t.vl_recebido,0)
                             - Nvl(t.vl_acrescimo,0)
                             + Nvl(t.vl_desconto,0))     vl_liquido
                     FROM dbamv.it_recebimento t,
                          dbamv.reccon_rec     r,
                          dbamv.itcon_rec      i,
                          dbamv.con_rec        c
                    WHERE t.cd_reccon_rec  = r.cd_reccon_rec
                      AND i.cd_itcon_rec   = r.cd_itcon_rec
                      AND i.cd_con_rec     = c.cd_con_rec
                      AND c.cd_nota_fiscal = :CAMPO_CD_NOTA_FISCAL_CONTA
                 GROUP BY cd_itfat_nf)   irec
           WHERE ifat.cd_atendimento = ated.cd_atendimento
             AND ifat.cd_nota_fiscal = nota.cd_nota_fiscal
             AND nota.cd_convenio    = conv.cd_convenio
             AND ated.cd_paciente    = paci.cd_paciente
             AND ifat.cd_pro_fat     = prof.cd_pro_fat
             AND prof.cd_gru_pro     = grup.cd_gru_pro
             AND grup.cd_gru_fat     = gruf.cd_gru_fat
             AND ifat.cd_itfat_nf    = irec.cd_itfat_nf(+)
             AND :CAMPO_TP_RELATORIO = 'A'
             AND :CAMPO_TP_EXIBIR = 'S'
             AND ifat.cd_nota_fiscal = :CAMPO_CD_NOTA_FISCAL_CONTA
             AND (ifat.cd_reg_fat = :CAMPO_CD_CONTA OR ifat.cd_reg_amb = :CAMPO_CD_CONTA)
             AND ifat.cd_remessa = :CAMPO_CD_REMESSA_CONTA
             AND ifat.cd_atendimento = :CAMPO_CD_ATENDIMENTO_CONTA
			 :CAMPO_CF_EMPRESA_ATENDIMENTO
)
GROUP BY  cd_nota_fiscal,
          cd_conta,
          cd_reg_fat,
          cd_remessa,
          cd_atendimento,
          nr_seq_preimpre,
          tp_seq_preimpre,
          nr_seq_preimpre_itamb,
          nr_seq_preimpre_regf
ORDER BY  cd_nota_fiscal,
          cd_conta,
          cd_remessa,
          cd_atendimento

--R_CT_REC_CONV_RS_R_ATENDIMENTO6_Q_DETALHAMENTO_CONTA
SELECT cd_nota_fiscal      cd_nota_fiscal_conta,
       tp_conta,
       cd_conta,
       cd_remessa          cd_remessa_conta,
       cd_atendimento      cd_atendimento_conta,
       vl_itfat_nf         vl_itfat_nf_conta,
       nm_paciente         nm_paciente_conta,
       vl_itfat_nf         vl_previsto_conta,
       (vl_recebido + vl_recebido_recurso)         vl_recebido_conta,
       vl_glosa            vl_glosa_conta,
       vl_desconto         vl_desconto_conta,
       vl_acrescimo        vl_acrescimo_conta,
       vl_recurso          vl_recurso_conta,
       vl_imposto          vl_imposto_retido_conta,
       (CASE
          WHEN (Trunc(dt_recebimento)  >=  Nvl(dt_recebido_recurso, Trunc(dt_recebimento)))
            THEN dt_recebimento
          ELSE dt_recebido_recurso
        END) dt_recebimento_conta,
       vl_aceite           vl_aceite_conta,
       (vl_itfat_nf - (vl_liquido + vl_aceite + vl_recebido_recurso)) vl_receber_conta
  FROM ( SELECT ifat.cd_nota_fiscal,
                DECODE(ifat.cd_reg_fat, NULL, 'A', 'I') tp_conta,
                NVL(ifat.cd_reg_fat, ifat.cd_reg_amb) cd_conta,
                ifat.cd_atendimento,
                ifat.cd_remessa,
                paci.nm_paciente,
                Nvl(Sum(ifat.vl_itfat_nf),0)  vl_itfat_nf,
                Max(irec.dt_recebimento) dt_recebimento,
                Nvl(Sum(irec.vl_recebido),0)  vl_recebido,
                Nvl(Sum(irec.vl_acrescimo),0) vl_acrescimo,
                Nvl(Sum(irec.vl_desconto),0)  vl_desconto,
                Nvl(Sum(irec.vl_imposto),0)   vl_imposto,
                Nvl(Sum(irec.vl_glosa),0)     vl_glosa,
                Nvl(Sum(irec.vl_liquido),0)   vl_liquido,
                Nvl((SELECT Sum(Nvl(vl_reapresentado,0))
                        FROM dbamv.glosas b
                      WHERE (b.cd_reg_fat = ifat.cd_reg_fat OR b.cd_reg_amb = ifat.cd_reg_amb)
                        AND cd_atendimento = ifat.cd_atendimento
                        AND cd_glosas_pai IS NULL
                        AND cd_remessa_glosa IS NOT NULL),0) vl_recurso,
                (SELECT Max(dt_recebido)
                        FROM dbamv.glosas b
                      WHERE (b.cd_reg_fat = ifat.cd_reg_fat OR b.cd_reg_amb = ifat.cd_reg_amb)
                        AND cd_atendimento = ifat.cd_atendimento
                        AND cd_remessa_glosa IS NOT NULL) dt_recebido_recurso,
                Nvl((SELECT Sum(Nvl(vl_recebido,0))
                        FROM dbamv.glosas b
                      WHERE (b.cd_reg_fat = ifat.cd_reg_fat OR b.cd_reg_amb = ifat.cd_reg_amb)
                        AND cd_atendimento = ifat.cd_atendimento
                        AND cd_remessa_glosa IS NOT NULL),0) vl_recebido_recurso,
                Nvl((SELECT Sum(Nvl(a.vl_glosa,0))
                        FROM dbamv.itmov_glosas a,
                            dbamv.itfat_nota_fiscal b
                      WHERE a.cd_itfat_nf = b.cd_itfat_nf
                        AND (b.cd_reg_fat = ifat.cd_reg_fat OR b.cd_reg_amb = ifat.cd_reg_amb)
                        AND b.cd_remessa = ifat.cd_remessa),0) vl_aceite,
                Nvl((SELECT Sum(Nvl(a.vl_retira_recurso,0))
                        FROM dbamv.itmov_glosas a,
                            dbamv.itfat_nota_fiscal b
                      WHERE a.cd_itfat_nf = b.cd_itfat_nf
                        AND (b.cd_reg_fat = ifat.cd_reg_fat OR b.cd_reg_amb = ifat.cd_reg_amb)
                        AND b.cd_remessa = ifat.cd_remessa),0) vl_retira_recurso
            FROM dbamv.itfat_nota_fiscal ifat,
                 dbamv.pro_fat           prof,
                 dbamv.gru_pro           grup,
                 dbamv.atendime          ated,
                 dbamv.paciente          paci,
                 (SELECT t.cd_itfat_nf,
                         Max(r.dt_recebimento)         dt_recebimento,
                         Sum(Nvl(t.vl_recebido,0))       vl_recebido,
                         Sum(Nvl(t.vl_acrescimo,0))      vl_acrescimo,
                         Sum(Nvl(t.vl_desconto,0))       vl_desconto,
                         Sum(Nvl(t.vl_imposto_retido,0)) vl_imposto,
                         Sum(Nvl(t.vl_glosa,0))          vl_glosa,
                         Sum(  Nvl(t.vl_recebido,0)
                             - Nvl(t.vl_acrescimo,0)
                             + Nvl(t.vl_desconto,0))     vl_liquido
                     FROM dbamv.it_recebimento t,
                          dbamv.reccon_rec     r,
                          dbamv.itcon_rec      i,
                          dbamv.con_rec        c
                    WHERE t.cd_reccon_rec  = r.cd_reccon_rec
                      AND i.cd_itcon_rec   = r.cd_itcon_rec
                      AND i.cd_con_rec     = c.cd_con_rec
                      AND c.cd_nota_fiscal = :CAMPO_CD_NOTA_FISCAL
                 GROUP BY cd_itfat_nf)   irec
           WHERE ifat.cd_atendimento = ated.cd_atendimento
             AND ated.cd_paciente    = paci.cd_paciente
             AND ifat.cd_pro_fat     = prof.cd_pro_fat
             AND prof.cd_gru_pro     = grup.cd_gru_pro
             AND ifat.cd_itfat_nf    = irec.cd_itfat_nf(+)
             AND ifat.cd_nota_fiscal = :CAMPO_CD_NOTA_FISCAL
             AND :CAMPO_TP_RELATORIO  = 'A'
        GROUP BY ifat.cd_nota_fiscal,
                 DECODE(cd_reg_fat, NULL, 'A', 'I'),
                 ifat.cd_reg_fat,
                 ifat.cd_reg_amb,
                 ifat.cd_atendimento,
                 ifat.cd_remessa,
                 paci.nm_paciente
        :CAMPO_CF_ORDER_CONTAS
)

                    r.vl_recebido + NVL (r.vl_imposto_retido, 0) vl_movimentacao_conv,
                    NVL (r.vl_imposto_retido, 0) vl_imposto_retido,
                    r.dt_recebimento dt_pagamento_conv,
                    TO_CHAR (NVL (n.nr_nota_fiscal_nfe, n.nr_id_nota_fiscal)) nr_documento_conv,
                    NVL (i.vl_glosa, 0) vl_glosa_conv,
                    NVL (i.vl_glosa_aceita, 0) vl_glosa_aceita_conv,
                    NVL (i.vl_recurso, 0) vl_recurso_conv,
                    i.tp_quitacao status_conv,
                    r.cd_reccon_rec cd_reccon_rec,
                    c.cd_nota_fiscal,
                    c.cd_con_rec cd_con_rec_conv,
                    (i.vl_duplicata - i.vl_soma_recebido) vl_receber_conrec,
                    NVL (r.vl_acrescimo, 0) vl_acrescimo_conv,
                    NVL (r.vl_desconto, 0) vl_desconto_conv,
                    i.cd_itcon_rec,
                    Nvl((SELECT SUM(vl_glosa)
                           FROM dbamv.it_recebimento
                          WHERE cd_reccon_rec = r.cd_reccon_rec),0) vl_glosa_rec,
                    Nvl((SELECT SUM(glosa.vl_reapresentado) vl_glosa
                           FROM dbamv.it_recebimento    itrec,
                                dbamv.itfat_nota_fiscal itfat,
                                dbamv.glosas            glosa
                          WHERE itrec.cd_reccon_rec = r.cd_reccon_rec
                            AND itrec.cd_itfat_nf   = itfat.cd_itfat_nf
                            AND itfat.cd_itfat_nf   = glosa.cd_itfat_nf
                            AND glosa.cd_remessa_glosa IS NOT NULL
                            AND glosa.cd_glosas_pai IS NULL),0)        vl_recurso_rec,
                    Decode(:CAMPO_DETALHE,'S',(Rank() OVER ( PARTITION BY c.cd_con_rec ORDER BY r.cd_reccon_rec)),1) cd_rank,
                    c.cd_multi_empresa cd_multi_empresa_conv,
                    e.ds_multi_empresa ds_multi_empresa_conv
               FROM dbamv.convenio co,
                    dbamv.multi_empresas e,
					dbamv.USUARIO_MULTI_EMPRESA ume,
                    dbamv.nota_fiscal n,
                    dbamv.con_rec c,
                    dbamv.itcon_rec i,
                    dbamv.reccon_rec r
              WHERE co.cd_convenio = n.cd_convenio
                AND n.cd_nota_fiscal = c.cd_nota_fiscal
                AND e.cd_multi_empresa = c.cd_multi_empresa
				and e.cd_multi_empresa = UME.cd_multi_empresa 
				AND ume.cd_id_usuario = USER
                AND c.cd_con_rec = i.cd_con_rec
                AND i.cd_con_rec_agrup IS NULL
                AND i.cd_itcon_rec = r.cd_itcon_rec(+)
                AND c.cd_nota_fiscal IS NOT NULL
                AND r.dt_estorno IS NULL
                AND UPPER(:CAMPO_ORDEM) in ('CONVENIO','RECEBIMENTO', 'PACIENTERECEBIMENTO', 'NOTAFISCAL')
                AND :CAMPO_TP_DATA   = 'N'
		:CAMPO_CF_SN_NOTA_CANC
                )
:CAMPO_CF_ORDER_REMESSA
         )
GROUP BY nm_convenio_conv,
         cd_remessa,
         dt_emissao_conv,
         dt_prevista_pagamento_conv,
         vl_previsto_conv,
         nr_documento_conv,
         vl_glosa_conv,
         vl_glosa_aceita_conv,
         vl_recurso_conv,
         status_conv,
         cd_rank,
         cd_nota_fiscal,
         cd_con_rec_conv,
         :CAMPO_CF_DETALHAMENTO_RECEBIMENTO
         vl_receber_conrec,
         cd_itcon_rec,
         cd_multi_empresa_conv,
         ds_multi_empresa_conv

         dt_emissao,
         cd_remessa,
         dt_prevista_pagamento,
         Decode(cd_rank, 1, vl_previsto, 0) vl_previsto,
         vl_previsto vl_previsto_calculo,
         SUM (vl_movimentacao) vl_movimentacao,
         MAX (dt_pagamento) AS dt_pagamento,
         nr_documento,
         Decode(:CAMPO_DETALHE, 'S', sum(vl_glosa_rec), vl_glosa) vl_glosa,
         Decode(:CAMPO_DETALHE, 'S', sum(vl_recurso_rec), vl_recurso_venc) vl_recurso_venc,

         status,
         cd_con_rec_venc,
         cd_nota_fiscal_venc,
         vl_receber_conrec_venc,
         SUM (vl_acrescimo_venc) vl_acrescimo_venc,
         SUM (vl_desconto_venc) vl_desconto_venc,
         SUM (vl_imposto_retido_venc) vl_imposto_retido_venc,
         cd_itcon_rec,
         cd_multi_empresa_venc,
         ds_multi_empresa_venc
    FROM ((SELECT co.nm_convenio nm_convenio,
                  TRUNC (c.dt_emissao) dt_emissao,
                  c.cd_remessa,
                  i.dt_vencimento dt_prevista_pagamento,
                  c.vl_previsto,
                  r.vl_recebido + NVL (r.vl_imposto_retido, 0) vl_movimentacao,

                  c.nr_documento nr_documento,
                  NVL (i.vl_glosa, 0) vl_glosa,

				  dbamv.USUARIO_MULTI_EMPRESA ume,
                  dbamv.remessa_fatura,
                  dbamv.fatura,
                  dbamv.convenio co,
                  dbamv.reccon_rec r
            WHERE remessa_fatura.cd_fatura = fatura.cd_fatura
              AND fatura.cd_convenio = co.cd_convenio
              AND e.cd_multi_empresa = c.cd_multi_empresa
			  and e.cd_multi_empresa = UME.cd_multi_empresa 
			  AND ume.cd_id_usuario = USER
              AND c.cd_remessa = remessa_fatura.cd_remessa
              AND c.cd_nota_fiscal IS NULL
              AND i.cd_con_rec = c.cd_con_rec
              AND i.cd_con_rec_agrup IS NULL
              AND r.cd_itcon_rec(+) = i.cd_itcon_rec
              AND r.dt_estorno IS NULL
              AND :CAMPO_ORDEM = 'Vencimento'
              :CAMPO_CF_SN_RECURSO
          )
          UNION
          (SELECT   co.nm_convenio nm_convenio,
                    c.dt_emissao dt_emissao,
                    0 cd_remessa,
                    i.dt_vencimento dt_prevista_pagamento,
                    c.vl_previsto,
                    r.vl_recebido + NVL (r.vl_imposto_retido, 0) vl_movimentacao,
                    r.dt_recebimento dt_pagamento,
                    TO_CHAR (n.nr_id_nota_fiscal) nr_documento,
                    NVL (i.vl_glosa, 0) vl_glosa,
                    NVL (i.vl_glosa_aceita, 0) vl_glosa_aceita,
                    NVL (i.vl_recurso, 0) vl_recurso_venc,
                    i.tp_quitacao status,
                    r.cd_reccon_rec cd_reccon_rec_venc,
                    c.cd_con_rec cd_con_rec_venc,
                    c.cd_nota_fiscal cd_nota_fiscal_venc,
                    (NVL (i.vl_duplicata, 0) - NVL (i.vl_soma_recebido, 0)) vl_receber_conrec_venc,
                    NVL (r.vl_acrescimo, 0) vl_acrescimo_venc,
                    NVL (r.vl_desconto, 0) vl_desconto_venc,
                    NVL (r.vl_imposto_retido, 0) vl_imposto_retido_venc,
                    i.cd_itcon_rec,
                    Nvl((SELECT SUM(vl_glosa)
                           FROM dbamv.it_recebimento
                          WHERE cd_reccon_rec = r.cd_reccon_rec),0) vl_glosa_rec,
                    Nvl((SELECT SUM(glosa.vl_reapresentado) vl_glosa
                           FROM dbamv.it_recebimento    itrec,
                                dbamv.itfat_nota_fiscal itfat,
                                dbamv.glosas            glosa
                          WHERE itrec.cd_reccon_rec = r.cd_reccon_rec
                            AND itrec.cd_itfat_nf   = itfat.cd_itfat_nf
                            AND itfat.cd_itfat_nf   = glosa.cd_itfat_nf
                            AND glosa.cd_remessa_glosa IS NOT NULL

                    Decode(:CAMPO_DETALHE,'S',(Rank() OVER ( PARTITION BY c.cd_con_rec ORDER BY r.cd_reccon_rec)),1) cd_rank,
