--R_CON_AUTORIZA_759
SELECT * FROM (
SELECT  CON_PAG.CD_CON_PAG
       ,CON_PAG.DS_CON_PAG DS_CON_PAG
       ,CON_PAG.NR_DOCUMENTO
       ,CON_PAG.DT_EMISSAO
       ,CON_PAG.VL_BRUTO_CONTA
       ,ITCON_PAG.CD_ITCON_PAG
       ,ITCON_PAG.NR_PARCELA
       ,ITCON_PAG.DT_VENCIMENTO
       ,ITCON_PAG.VL_DUPLICATA
       ,ITCON_PAG.VL_DUPLICATA                  VL_DESTAQUE
       ,NVL( FORNECEDOR.NM_FORNECEDOR, CON_PAG.DS_FORNECEDOR ) NM_FANTASIA
       ,TIP_DOC.DS_TIP_DOC
FROM    DBAMV.CON_PAG
       ,DBAMV.TIP_DOC
       ,DBAMV.ITCON_PAG
       ,DBAMV.FORNECEDOR
WHERE   TIP_DOC.CD_TIP_DOC         	 =  CON_PAG.CD_TIP_DOC
AND     ITCON_PAG.CD_CON_PAG        	 =  CON_PAG.CD_CON_PAG
AND     CON_PAG.DT_EMISSAO       <= TO_DATE( TO_CHAR( :CAMPO_DT_FINAL  , 'DD/MM/YYYY')||' 23','DD/MM/YYYY HH24' )
AND     CON_PAG.DT_EMISSAO       >= TO_DATE( TO_CHAR( :CAMPO_DT_INICIAL, 'DD/MM/YYYY')||' 00','DD/MM/YYYY HH24' )
AND     FORNECEDOR.CD_FORNECEDOR(+) 	 =  CON_PAG.CD_FORNECEDOR
AND     CON_PAG.CD_MULTI_EMPRESA    	 =  :CAMPO_CD_MULTI_EMPRESA
ORDER BY CON_PAG.CD_CON_PAG
)
ORDER BY CD_CON_PAG,DS_CON_PAG,DT_EMISSAO,NR_DOCUMENTO,DS_TIP_DOC,VL_BRUTO_CONTA,NM_FANTASIA,VL_DUPLICATA
