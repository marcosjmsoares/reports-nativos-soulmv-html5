--R_ARECGRURES
SELECT * FROM (
SELECT  GRU_RES.CD_GRU_RES                       CD_GRU_RES
        ,GRU_RES.DS_GRU_RES                               DS_GRU_RES
        ,RAT_CONREC.CD_ITEM_RES                       CODIGO
        ,ITEM_RES.DS_ITEM_RES                             DESCRICAO
        ,CON_REC.NR_DOCUMENTO                         NR_DOCUMENTO
        ,CON_REC.DS_CON_REC                               DS_CON_REC
        ,itCON_REC.DT_PREVISTA_RECEBIMENTO      DT_PREVISTA_PAGAMENTO
        ,CON_REC.DT_EMISSAO                                DT_EMISSAO
        ,HIST_P.DS_HISTORICO_PADRAO                DS_HIST_P
        ,decode(CON_REC.CD_FORNECEDOR,'',CON_REC.NM_CLIENTE,FORNECEDOR.NM_FANTASIA)   nM_FANTASIA ,
    dbamv.Calcula_valor_Rateio(Rat_ConREc.Cd_Con_rec ,Rat_ConRec.RowID , ItCon_rec.Vl_Duplicata , Rat_conRec.Vl_Rateio )   Vl_Rateio
    ,CON_REC.CD_CON_REC
From
         dbamv.itcon_rec
          ,dbamv.CON_REC                   CON_REC
        ,dbamv.RAT_CONREC                RAT_CONREC
        ,dbamv.ITEM_RES                  ITEM_RES
        ,dbamv.FORNECEDOR                FORNECEDOR
        ,dbamv.HISTORICO_PADRAO          HIST_P
        ,dbamv.GRU_RES                   GRU_RES
        ,dbamv.SETOR                     SETOR
WHERE    CON_REC.CD_FORNECEDOR              = FORNECEDOR.CD_FORNECEDOR(+)
AND      CON_REC.CD_HISTORICO_PADRAO     = HIST_P.CD_HISTORICO_PADRAO(+)
AND      CON_REC.CD_MULTI_EMPRESA           =  :CAMPO_CD_MULTI_EMPRESA
AND      CON_REC.CD_CON_REC                        = RAT_CONREC.CD_CON_REC
AND      ITCON_REC.CD_CON_REC			=   CON_REC.CD_CON_REC
AND      ITEM_RES.CD_ITEM_RES                      = RAT_CONREC.CD_ITEM_RES
AND      ITEM_RES.CD_GRU_RES                       =  GRU_RES.CD_GRU_RES
AND      ITCON_REC.DT_PREVISTA_RECEBIMENTO   BETWEEN :CAMPO_DT_INIC ORDER BY  DT_PREVISTA_PAGAMENTO, NM_FANTASIA
)
ORDER BY DS_GRU_RES,CD_GRU_RES,
CODIGO,DESCRICAO,
DT_PREVISTA_PAGAMENTO
