--R_CUSTO_CONTA_GER
SELECT * FROM (
SELECT   V.CD_CONTA
        ,V.DS_CONTA
        ,V.DATA
        ,V.HISTORICO
        ,V.NR_DOC
        ,nvl(V.VL_ENTRADA,0) VL_ENTRADA
        ,nvl(V.VL_SAIDA,0) VL_SAIDA
        ,V.SITUACAO
        ,V.CD_RATEIO
FROM     DBAMV.V_CUSTO_CONTA_PREV   V
WHERE    V.CD_MULTI_EMPRESA  =    :CAMPO_CD_MULTI_EMPRESA
AND ( TRUNC( V.DATA ) BETWEEN :CAMPO_DT_INICIAL and :CAMPO_DT_FINAL )
UNION ALL
SELECT   V.CD_CONTA
        ,V.DS_CONTA
        ,V.DATA
        ,V.HISTORICO
        ,V.NR_DOC
        ,nvl(V.VL_ENTRADA, 0) VL_ENTRADA
        ,nvl(V.VL_SAIDA, 0) VL_SAIDA
        ,V.SITUACAO
        ,V.CD_RATEIO
FROM     DBAMV.V_CUSTO_CONTA   V
WHERE    V.CD_MULTI_EMPRESA  =    :CAMPO_CD_MULTI_EMPRESA
AND ( TRUNC( V.DATA ) BETWEEN :CAMPO_DT_INICIAL and :CAMPO_DT_FINAL )
)
ORDER BY CD_CONTA,DS_CONTA, DATA, HISTORICO, NR_DOC, VL_ENTRADA, VL_SAIDA, SITUACAO
