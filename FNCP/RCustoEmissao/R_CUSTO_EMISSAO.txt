--R_CUSTO_EMISSAO
SELECT * FROM (
SELECT trunc(CP.DT_EMISSAO) DT_EMISSAO
      ,ITEM.CD_ITEM_RES
      ,ITEM.DS_ITEM_RES
      ,SETOR.CD_SETOR
      ,SETOR.NM_SETOR
      ,RAT.VL_RATEIO
FROM DBAMV.CON_PAG          CP
    ,DBAMV.RATCON_PAG       RAT
    ,DBAMV.SETOR            SETOR
    ,DBAMV.ITEM_RES         ITEM
WHERE CP.CD_MULTI_EMPRESA            =   :CAMPO_CD_MULTI_EMPRESA 
AND   CP.CD_CON_PAG            	      =   RAT.CD_CON_PAG
AND   RAT.CD_ITEM_RES           	      =   ITEM.CD_ITEM_RES
AND   RAT.CD_SETOR                	      =   SETOR.CD_SETOR
AND   SETOR.SN_ACEITA_LANCAMENTO    =   'S'
AND   CP.DT_EMISSAO  		 <=To_date(To_char(:CAMPO_DT_FIM,'DD/MM/YYYY')||' 23'  ,'DD/MM/YYYY HH24')
AND   CP.DT_EMISSAO		 >=To_date(To_char(:CAMPO_DT_INIC,'DD/MM/YYYY')||' 00' ,'DD/MM/YYYY HH24')
:CAMPO_CF_ITEM_RES
)
ORDER BY DT_EMISSAO,
NM_SETOR,CD_SETOR,
DS_ITEM_RES,CD_ITEM_RES
