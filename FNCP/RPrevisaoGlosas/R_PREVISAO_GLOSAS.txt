--R_PREVISAO_GLOSAS
SELECT * FROM (
SELECT RG.CD_REMESSA                                               CD_REMESSA
            , RG.CD_CON_REC					   CD_CON_REC
	    , TRUNC(RG.DT_ENTREGA)                                 DT_EMISSAO
            , TRUNC(RG.DT_PREVISTA_PARA_PGTO)         DT_PREVISTA_PAGAMENTO
            , RG.CD_CONVENIO                                              CD_CONVENIO
            , RG.VL_TOTAL_REMESSA                                   VL_TOTAL_REMESSA
            , CONVENIO.NM_CONVENIO                                NM_CONVENIO
FROM  DBAMV.REMESSA_GLOSA  RG
          , DBAMV.CONVENIO               CONVENIO
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And RG.CD_CONVENIO    =  CONVENIO.CD_CONVENIO
      AND  NVL(RG.SN_PAGA,'N') <> 'S'
      AND  TRUNC(RG.DT_PREVISTA_PARA_PGTO)  BETWEEN :CAMPO_DATA_INI GROUP BY CONVENIO.NM_CONVENIO
                 ,  RG.CD_REMESSA
                 , TRUNC(RG.DT_ENTREGA)
                 , TRUNC(RG.DT_PREVISTA_PARA_PGTO)
                 , RG.CD_CONVENIO
                 , RG.VL_TOTAL_REMESSA
		 , RG.CD_CON_REC
ORDER BY NM_CONVENIO, CD_REMESSA,  TRUNC( DT_PREVISTA_PARA_PGTO ),CD_CON_REC
)
ORDER BY NM_CONVENIO,CD_CONVENIO,
CD_REMESSA,CD_CON_REC
