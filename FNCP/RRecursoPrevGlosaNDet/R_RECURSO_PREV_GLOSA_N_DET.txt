--R_RECURSO_PREV_GLOSA_N_DET
SELECT *
   FROM
  (SELECT convenio.nm_convenio nm_convenio                            ,
    TRUNC(con_rec.dt_emissao) dt_emissao                              ,
    TRUNC(itcon_rec.dt_prevista_recebimento ) dt_prevista_pagamento   ,
    itcon_rec.vl_duplicata vl_previsto                                ,
    itcon_rec.vl_glosa vl_glosa_fnfi                                  ,
    NVL(itcon_rec.vl_recurso,0) vl_recurso_fnfi                       ,
    NVL(remessa_glosa.vl_total_remessa,0) vl_recurso                  ,
    NVL(ABS(remessa_glosa.vl_recurso_recebido),0) vl_recurso_recebido ,
    NULL nr_documento                                                 ,
    con_rec.cd_con_rec cd_con_rec                                     ,
    itcon_rec.cd_itcon_rec cd_itcon_rec                               ,
    con_rec.cd_remessa cd_remessa                                     ,
    remessa_glosa.cd_remessa_glosa cd_remessa_glosa                   ,
    TRUNC(remessa_glosa.dt_prevista_para_pgto) dt_prevista_para_pgto  ,
    remessa_glosa.vl_total_remessa vl_total_remessa_glosa             ,
    GLOSA.TOT_GLOSA_PRE_ACEITA                                        ,
    DECODE(NVL(con_rec.SN_GLOSA_ACEITA,'N'), 'S', '*', NULL) SN_GLOSA_ACEITA
     FROM dbamv.itcon_rec  ,
    dbamv.con_rec          ,
    dbamv.convenio         ,
    dbamv.remessa_glosa    ,
    dbamv.remessa_fatura   ,
    dbamv.fatura           ,
    Dbamv.Empresa_Convenio ,
    (SELECT CD_REMESSA     ,
      SUM(TOT_GLOSA_PRE_ACEITA) TOT_GLOSA_PRE_ACEITA
       FROM
      (SELECT REG_FAT.CD_REMESSA ,
        SUM(NVL(GLOSAS.VL_GLOSA_PRE_ACEITA,0)) TOT_GLOSA_PRE_ACEITA
         FROM DBAMV.ITFAT_NOTA_FISCAL ,
        DBAMV.REG_FAT                 ,
        DBAMV.GLOSAS
        WHERE ITFAT_NOTA_FISCAL.CD_REMESSA = REG_FAT.CD_REMESSA
      AND REG_FAT.CD_REG_FAT               = GLOSAS.CD_REG_FAT
     GROUP BY REG_FAT.CD_REMESSA
    UNION ALL
       SELECT REG_AMB.CD_REMESSA ,
        SUM(NVL(GLOSAS.VL_GLOSA_PRE_ACEITA,0)) TOT_GLOSA_PRE_ACEITA
         FROM DBAMV.ITFAT_NOTA_FISCAL ,
        DBAMV.REG_AMB                 ,
        DBAMV.GLOSAS
        WHERE ITFAT_NOTA_FISCAL.CD_REMESSA = REG_AMB.CD_REMESSA
      AND REG_AMB.CD_REG_AMB               = GLOSAS.CD_REG_AMB
     GROUP BY REG_AMB.CD_REMESSA
      )
   GROUP BY CD_REMESSA
    ) GLOSA
    WHERE Empresa_Convenio.Cd_Convenio  = Convenio.Cd_Convenio
  AND Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
  AND fatura.cd_convenio                = convenio.cd_convenio
  AND fatura.cd_fatura                  = remessa_fatura.cd_fatura
  AND convenio.cd_convenio              = remessa_glosa.cd_convenio
  AND con_rec.cd_remessa                = remessa_fatura.cd_remessa
  AND con_rec.tp_con_rec                = 'C'
  AND con_rec.cd_previsao              IS NULL
  AND itcon_rec.cd_con_rec              = con_rec.cd_con_rec
  AND remessa_fatura.cd_remessa         = remessa_glosa.cd_remessa (+)
  AND CON_REC.CD_REMESSA                = GLOSA.CD_REMESSA
 GROUP BY convenio.nm_convenio                ,
    con_rec.cd_remessa                        ,
    TRUNC(con_rec.dt_emissao)                 ,
    TRUNC(itcon_rec.dt_prevista_recebimento ) ,
    itcon_rec.vl_duplicata                    ,
    itcon_rec.vl_glosa                        ,
    itcon_rec.vl_recurso                      ,
    remessa_glosa.vl_total_remessa            ,
    remessa_glosa.vl_recurso_recebido         ,
    con_rec.cd_con_rec                        ,
    itcon_rec.cd_itcon_rec                    ,
    remessa_glosa.cd_remessa_glosa            ,
    remessa_glosa.dt_prevista_para_pgto       ,
    remessa_glosa.vl_total_remessa            ,
    itcon_rec.vl_soma_recebido                ,
    GLOSA.TOT_GLOSA_PRE_ACEITA                ,
    DECODE(NVL(con_rec.SN_GLOSA_ACEITA,'N'), 'S', '*', NULL)
 ORDER BY nm_convenio                          ,
    cd_remessa                                 ,
    TRUNC(remessa_glosa.dt_prevista_para_pgto) ,
    remessa_glosa.cd_remessa_glosa
  )
ORDER BY NM_CONVENIO   ,
  DT_PREVISTA_PAGAMENTO,
  TOT_GLOSA_PRE_ACEITA ,
  SN_GLOSA_ACEITA      ,
  CD_ITCON_REC         ,
  CD_REMESSA           ,
  DT_EMISSAO           ,
  NR_DOCUMENTO         ,
  VL_PREVISTO          ,
  CD_CON_REC           ,
  VL_GLOSA_FNFI        ,
  VL_RECURSO_FNFI
