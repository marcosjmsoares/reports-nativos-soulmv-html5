select   c.ds_fornecedor                     ds_fornecedor,
           c.ds_con_pag                        ds_con_pag,
           c.cd_con_pag                        cd_con_pag,
           c.nr_documento                     nr_documento_conta,
           c.vl_bruto_conta                    vl_bruto_conta,
           td.ds_tip_doc                         ds_tip_doc,
           i.dt_vencimento                      dt_vencimento,
           p.cd_pagcon_pag                  cd_pagcon_pag,
           p.vl_pago                               vl_pago_bruto,
           nvl(p.vl_pago,0) + nvl(p.vl_desconto,0) - nvl(p.vl_acrescimo,0)         vl_pago_liquido,
           p.dt_pagamento                     dt_pagamento,
           f.nm_fornecedor                     nm_fornecedor,
           f.nm_fantasia                          nm_fantasia,
           f.nr_cgc_cpf                           nr_cgc_cpf,
           f.nr_inscricao_inss                  nr_inscricao_inss,
           f.nr_pis_pasep                        nr_pis_pasep,
           f.ds_endereco                        ds_endereco,
           f.ds_bairro                              ds_bairro,
           f.nr_cep                                  nr_cep,
           f.cd_cidade                            cd_cidade
from    dbamv.con_pag               c,
           dbamv.itcon_pag             i,
           dbamv.pagcon_pag        p,
           dbamv.fornecedor           f,
           dbamv.tip_doc                td,
	   dbamv.lote_rem_itcon_pag lt,
	   dbamv.usuario_multi_empresa ume
where  i.cd_con_pag   = c.cd_con_pag
and      i.cd_itcon_pag = p.cd_itcon_pag
and      c.cd_fornecedor = f.cd_fornecedor (+)
and      i.cd_itcon_pag  = lt.cd_itcon_pag(+)
and      c.cd_tip_doc = td.cd_tip_doc
and c.cd_multi_empresa = ume.cd_multi_empresa 
and ume.cd_id_usuario = user 
and      i.cd_con_pag_agrup is null
and      p.dt_estorno is null
