--R_TIP_DET
SELECT  TIP_DETALHE.CD_DETALHAMENTO CODIGO
       ,TIP_DETALHE.DS_DETALHAMENTO DESCRICAO
       ,TIP_DETALHE.NR_DIAS_VENCTO NR_DIAS_VENCTO
       ,TIP_DETALHE.TP_VENCIMENTO TP_VENCIMENTO
       ,TIP_DETALHE.VL_PERCENTUAL VL_PERCENTUAL
       ,TIP_DETALHE.CD_FORNECEDOR CD_FORNECEDOR
       ,FORNECEDOR.NM_FORNECEDOR NM_FORNECEDOR
FROM    dbamv.TIP_DETALHE TIP_DETALHE
       ,dbamv.FORNECEDOR FORNECEDOR
WHERE TIP_DETALHE.CD_FORNECEDOR = FORNECEDOR.CD_FORNECEDOR(+)
ORDER BY
