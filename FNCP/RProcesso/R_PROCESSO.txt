--R_PROCESSO
select	P.cd_estrutural,
        	P.cd_processo,
        	P.tp_processo,
        	P.ds_processo,
         	P.ds_complemento,
	ME.sn_utiliza_processo_me
      from	dbamv.processo		P,
                dbamv.multi_empresas	ME
     where  P.cd_multi_empresa = ME.cd_multi_empresa
       and	substr(P.cd_estrutural, 1, 2) <> '99'
       and	P.sn_processo_multi_empresa in ('N',nvl(ME.sn_utiliza_processo_me,'N'))
order by	P.cd_processo
