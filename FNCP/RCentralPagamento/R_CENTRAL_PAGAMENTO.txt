--R_CENTRAL_PAGAMENTO
SELECT cd_con_pag
      ,cd_multi_empresa
      ,nr_documento
      ,nr_parcela
      ,ds_fornecedor
      ,dt_vencimento
      ,qt_dias_atraso
      ,Decode(tp_quitacao,'Q','Quitado','V','Previsto','C','Aberto','P','Parcial','N','Cancelado','T','Liq. Total','Liq. Parcial' ) tp_quitacao
      ,Decode(sn_autorizado,'S','Sim','Não') sn_autorizado
      ,cd_lote_pgto
      ,cd_moeda
      ,vl_duplicata
      ,vl_soma_pago
      ,vl_pendente
      ,dt_prevista_pag
FROM DBAMV.V_CONSULTA_PARCELA_PAGAR
WHERE 1=1
 ORDER BY cd_con_pag, dt_vencimento DESC
