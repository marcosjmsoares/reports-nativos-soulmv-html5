--R_FORNEC_R_NM_CONTATO_CG__Q_FORNECEDOR
SELECT * FROM (
SELECT  FORNECEDOR.CD_FORNECEDOR CODIGO
       ,FORNECEDOR.NM_FORNECEDOR NM_FORNECEDOR
       ,FORNECEDOR.NM_FANTASIA FANTASIA
       ,FORNECEDOR.DS_ENDERECO DS_ENDERECO
       ,FORNECEDOR.DS_BAIRRO DS_BAIRRO
       ,CIDADE.NM_CIDADE CIDADE
       ,CONTATO_FORNEC.NM_CONTATO NM_CONTATO
       ,FORNECEDOR.NR_CEP NR_CEP
       ,FORNECEDOR.TP_FORNECEDOR
       ,FORNECEDOR.CD_CNAE CD_CNAE
,CASE  Nvl((SELECT CONFIGURACAO.VALOR FROM DBAMV.CONFIGURACAO  WHERE CONFIGURACAO.CHAVE = 'CD_PAIS'
                AND CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.le_empresa AND CD_SISTEMA = 'GLOBAL'), 'BR') WHEN 'BR' then
                decode( tp_fornecedor, 'J', substr(lpad(nvl(nr_cgc_cpf, '0'),14, '0'),1,2)||'.'||
                            substr(lpad(nvl(nr_cgc_cpf, '0'),14, '0'),3,3)||'.'||
                            substr(lpad(nvl(nr_cgc_cpf, '0'),14, '0'),6,3)||'/'||
                            substr(lpad(nvl(nr_cgc_cpf, '0'),14, '0'),9,4)||'-'||
                            substr(lpad(nvl(nr_cgc_cpf, '0'),14, '0'),13,2),
                       'F', substr(lpad(nvl(nr_cgc_cpf, '0'),14, '0'),4,3)||'.'||
                            substr(lpad(nvl(nr_cgc_cpf, '0'),14, '0'),7,3)||'.'||
                            substr(lpad(nvl(nr_cgc_cpf, '0'),14, '0'),10,3)||'-'||
                            substr(lpad(nvl(nr_cgc_cpf, '0'),14, '0'),13,2))   ELSE CD_IDENTIFICADOR_PESSOA END nr_cgc_cpf
      , FORNECEDOR.SN_ATIVO
FROM DBAMV.FORNECEDOR FORNECEDOR
       ,DBAMV.CIDADE CIDADE
       ,DBAMV.CONTATO_FORNEC CONTATO_FORNEC
WHERE CIDADE.CD_CIDADE = FORNECEDOR.CD_CIDADE
      AND FORNECEDOR.CD_FORNECEDOR = CONTATO_FORNEC.CD_FORNECEDOR(+)
      AND FORNECEDOR.SN_ATIVO LIKE :CAMPO_SN_ATIVO
     AND (FORNECEDOR.TP_CLIENTE_FORN = :CAMPO_TP_FORNEC OR :CAMPO_TP_FORNEC = '0')
)
