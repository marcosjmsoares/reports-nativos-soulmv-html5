--R_GPS
SELECT   MULTI_EMP.DS_MULTI_EMPRESA || ' / ' || NR_TELEFONE_FNFI    NM_EMPRESA
        ,AGRUP.DT_EMISSAO                       DT_EMISSAO
        ,AGRUP.CD_CON_PAG
        ,nvl(to_char(MULTI_EMP.CD_CGC,'99999999999999'),MULTI_EMP.CD_IEST) NR_CGC
        ,IT_AGRUP.DT_VENCIMENTO                 DT_VENC
        ,IT_AGRUP.VL_DUPLICATA                  VL_DUPLIC
        ,IT_AGRUP.VL_DUPLICATA                  VL_TOTAL
        ,TIP_DETALHE.CD_ATIVIDADE
        ,MULTI_EMP.DS_ENDERECO||', '||CIDADE.NM_CIDADE||'\'||CIDADE.CD_UF  DS_LOGRADOURO
        ,TIP_DETCON_PAG.CD_CON_PAG_PAI
FROM   DBAMV.CON_PAG              AGRUP
        ,DBAMV.ITCON_PAG          IT_AGRUP
        ,DBAMV.ITCON_PAG          ITCON_PAG
        ,DBAMV.CON_PAG            CON_PAG
        ,DBAMV.MULTI_EMPRESAS     MULTI_EMP
        ,DBAMV.TIP_DETALHE        TIP_DETALHE
        ,DBAMV.TIP_DETCON_PAG     TIP_DETCON_PAG
        ,DBAMV.CIDADE
WHERE   AGRUP.CD_CON_PAG            = IT_AGRUP.CD_CON_PAG
AND     IT_AGRUP.CD_CON_PAG         = ITCON_PAG.CD_CON_PAG_AGRUP(+)
AND     ITCON_PAG.CD_CON_PAG        = CON_PAG.CD_CON_PAG(+)
AND     AGRUP.CD_MULTI_EMPRESA      = MULTI_EMP.CD_MULTI_EMPRESA
AND     MULTI_EMP.CD_CIDADE         = CIDADE.CD_CIDADE(+)
AND     NVL(CON_PAG.CD_CON_PAG,AGRUP.CD_CON_PAG) = TIP_DETCON_PAG.CD_CON_PAG_FILHO
AND     TIP_DETCON_PAG.CD_DETALHAMENTO = TIP_DETALHE.CD_DETALHAMENTO
