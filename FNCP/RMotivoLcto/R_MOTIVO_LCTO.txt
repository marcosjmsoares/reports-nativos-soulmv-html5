--R_MOTIVO_LCTO
SELECT
  CODIGO,
  DESCRICAO,
  TIPO
FROM
(
SELECT
  MOTIVO_LCTO.CD_MOTIVO_LCTO CODIGO
 ,MOTIVO_LCTO.DS_MOTIVO_LCTO DESCRICAO
 ,'Receita' TIPO
FROM
 DBAMV.MOTIVO_LCTO
WHERE
  MOTIVO_LCTO.TP_UTILIZACAO = 'R'
UNION ALL
SELECT
  MOTIVO_LCTO.CD_MOTIVO_LCTO CODIGO
 ,MOTIVO_LCTO.DS_MOTIVO_LCTO DESCRICAO
 ,'Despesa' TIPO
FROM
 DBAMV.MOTIVO_LCTO
WHERE
  MOTIVO_LCTO.TP_UTILIZACAO = 'D'
UNION ALL
SELECT
  MOTIVO_LCTO.CD_MOTIVO_LCTO CODIGO
 ,MOTIVO_LCTO.DS_MOTIVO_LCTO DESCRICAO
 ,'Ambos' TIPO
FROM
 DBAMV.MOTIVO_LCTO
WHERE
  MOTIVO_LCTO.TP_UTILIZACAO = 'A' )
ORDER BY
