--R_VISAO_CONTA
SELECT * FROM (
Select visao.cd_visao                cd_visao
             ,visao.ds_visao                ds_visao
             ,hierarq.numlinha              numlinha
             ,hierarq.cd_plano_sintetico    cd_plano_sintetico
             ,hierarq.cd_conta              cd_conta
             ,hierarq.nome_conta            ds_conta
             ,hierarq.tp_conta            tp_conta
             ,hierarq.grau                  grau
       From   dbamv.visao,
              (select rownum numLinha
                     ,cd_visao     cd_visao
                     ,cd_conta cd_conta
                     ,ds_conta nome_conta
                     ,cd_plano_sintetico cd_plano_sintetico
                     ,tp_conta  tp_conta
                     ,level grau
               from   dbamv.plano_sintetico_visao
               start with cd_pai is null
               connect by prior cd_plano_sintetico = cd_pai) hierarq
       where  visao.cd_visao         = hierarq.cd_visao
       and    visao.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
       order by cd_visao, cd_conta
)
ORDER BY CD_VISAO,DS_VISAO
