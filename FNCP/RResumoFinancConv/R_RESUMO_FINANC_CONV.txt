--R_RESUMO_FINANC_CONV
SELECT DT_EMISSAO
, CD_CONVENIO
, NVL(NM_CONVENIO, 'z - RECEBIMENTO DIRETO') NM_CONVENIO
, SUM(VL_PREVISTO) VL_PREVISTO
, SUM(VL_RECUPERADO_RECURSO) VL_RECUPERADO_RECURSO
, SUM(VL_RECEBIDO_SEM_RECURSO) + SUM(VL_IMPOSTO) VL_RECEBIDO_SEM_RECURSO
, SUM(VL_TOTAL_RECEBIDO) VL_TOTAL_RECEBIDO
, SUM(VL_GLOSA_FNFI) VL_GLOSA_FNFI
, SUM(VL_RECURSO) VL_RECURSO
, SUM(VL_IMPOSTO) VL_IMPOSTO
, SUM(vl_glosa_aceita)VL_GLOSA_ACEITA
FROM (
SELECT
      CON_REC_CONV.CD_CON_REC
      , TO_CHAR( TRUNC(CON_REC_CONV.DT_EMISSAO,'YYYY'),'YYYY')	DT_EMISSAO
      ,CONVENIO.CD_CONVENIO   CD_CONVENIO
      ,CONVENIO.NM_CONVENIO 	NM_CONVENIO
      , ITCON_REC.VL_DUPLICATA VL_PREVISTO
      ,SUM(DECODE(NVL(RECCON_REC.SN_RECURSO,'N'),'S',NVL(RECCON_REC.VL_RECEBIDO,0),0))        VL_RECUPERADO_RECURSO
      ,SUM(DECODE(NVL(RECCON_REC.SN_RECURSO,'N'),'N',NVL(RECCON_REC.VL_RECEBIDO,0),0))        VL_RECEBIDO_SEM_RECURSO
      ,SUM(NVL(RECCON_REC.VL_RECEBIDO,0)) VL_TOTAL_RECEBIDO
      ,NVL(ITCON_REC.VL_GLOSA,0) VL_GLOSA_FNFI
      ,NVL(ITCON_REC.VL_RECURSO,0) VL_RECURSO
      ,SUM(NVL(RECCON_REC.VL_IMPOSTO_RETIDO,0)) VL_IMPOSTO
      ,NVL(ITCON_REC.vl_glosa_aceita,0) VL_GLOSA_ACEITA
  FROM  DBAMV.ITCON_REC
      , (
        SELECT CON_REC.CD_CON_REC
        , CON_REC.DT_EMISSAO
        , CON_REC.CD_PROCESSO
        , NVL(NOTA_FISCAL.CD_CONVENIO, FATURA.CD_CONVENIO) CD_CONVENIO
        , CON_REC.CD_FORNECEDOR
        FROM DBAMV.CON_REC
        , DBAMV.NOTA_FISCAL
        , DBAMV.REMESSA_FATURA
        , DBAMV.FATURA
        , DBAMV.PROCESSO
        WHERE CON_REC.TP_CON_REC = 'C'
        AND CON_REC.CD_PREVISAO IS NULL
        AND CON_REC.CD_PROCESSO = PROCESSO.CD_PROCESSO
        AND CON_REC.CD_REMESSA = REMESSA_FATURA.CD_REMESSA(+)
        AND REMESSA_FATURA.CD_FATURA = FATURA.CD_FATURA(+)
        AND CON_REC.CD_NOTA_FISCAL = NOTA_FISCAL.CD_NOTA_FISCAL(+)
        :CAMPO_CF_CONFIRMA_VIRTUAL
      ) CON_REC_CONV
      , DBAMV.CONVENIO
      , DBAMV.RECCON_REC
      , DBAMV.EMPRESA_CONVENIO
WHERE 1=1
AND CON_REC_CONV.CD_CONVENIO = CONVENIO.CD_CONVENIO(+)
AND CONVENIO.CD_CONVENIO = EMPRESA_CONVENIO.CD_CONVENIO(+)
AND EMPRESA_CONVENIO.CD_MULTI_EMPRESA(+) = :CAMPO_CD_MULTI_EMPRESA
AND CON_REC_CONV.CD_CONVENIO IS NOT NULL
AND ITCON_REC.CD_CON_REC = CON_REC_CONV.CD_CON_REC
AND ITCON_REC.CD_ITCON_REC = RECCON_REC.CD_ITCON_REC (+)
GROUP BY CON_REC_CONV.CD_CON_REC, TRUNC(CON_REC_CONV.DT_EMISSAO,'YYYY'), CONVENIO.CD_CONVENIO, NM_CONVENIO
, ITCON_REC.VL_DUPLICATA
, ITCON_REC.VL_GLOSA
, ITCON_REC.VL_RECURSO
, ITCON_REC.vl_glosa_aceita
)
GROUP BY DT_EMISSAO
, CD_CONVENIO
, NM_CONVENIO
ORDER BY NM_CONVENIO,CD_CONVENIO
