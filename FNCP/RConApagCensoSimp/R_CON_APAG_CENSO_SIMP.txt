--R_CON_APAG_CENSO_SIMP
SELECT * FROM (
SELECT  UNIQUE  TRUNC(ITCON_PAG.DT_VENCIMENTO)      DT_VENCIMENTO
       ,DECODE(CON_PAG.CD_FORNECEDOR, NULL, CON_PAG.DS_FORNECEDOR, FORNECEDOR.NM_FANTASIA )  NM_FANTASIA
       ,TRUNC (CON_PAG.DT_EMISSAO )                         DT_EMISSAO
       ,CON_PAG.DS_CON_PAG                                        DS_CON_PAG
       ,CON_PAG.CD_CON_PAG                                        CD_CON_PAG
       ,CON_PAG.NR_DOCUMENTO || '/' ||  to_char(ITCON_PAG.NR_PARCELA,'000')     NR_DOCUMENTO
       ,DBAMV.QTD_ITENS_CON_PAG( 1,CON_PAG.CD_CON_PAG )            QTD
       ,ITCON_PAG.TP_QUITACAO                                   TP_QUITACAO
       ,ITCON_PAG.VL_DUPLICATA                                  VL_DUPLICATA
from  dbamv.con_pag con_pag
      ,dbamv.itcon_pag itcon_pag
      ,dbamv.fornecedor fornecedor
      ,dbamv.multi_empresas multi_empresa
       ,dbamv.TIP_DOC
where (ITCON_PAG.DT_BAIXA IS NULL OR TO_CHAR(ITCON_PAG.DT_BAIXA, 'YYYY/MM/DD') > :CAMPO_DT_SITUACAO)
          and con_pag.cd_multi_empresa                           = :CAMPO_CD_MULTI_EMPRESA
          and con_pag.cd_multi_empresa                     = multi_empresa.cd_multi_empresa
          and   CON_PAG.CD_TIP_DOC                     =  TIP_DOC.CD_TIP_DOC
          and  con_pag.cd_con_pag                           = itcon_pag.cd_con_pag
          and itcon_pag.cd_itcon_pag_agrup is null
order by  dt_vencimento, nm_fantasia
)
ORDER BY DT_VENCIMENTO
