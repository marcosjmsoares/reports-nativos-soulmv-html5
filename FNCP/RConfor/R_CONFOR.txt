            AND ((pagcon_pag.cd_caixa is not null :CAMPO_CF_MULTI_EMPRESA_CAIXA) or  pagcon_pag.cd_caixa is null)
            :CAMPO_CF_BAIXA_DEVOLUCAO) pagamento  
  where con_pag.cd_historico_padrao = hist_p.cd_historico_padrao(+)
    AND con_pag.cd_con_pag_dev IS NULL
    AND con_pag.tp_adiantamento IS NULL
    AND con_pag.vl_bruto_conta <> 0
    and con_pag.cd_con_pag = pagamento.cd_con_pag(+)
  UNION
  SELECT 'C'                                                              tp
      ,Decode (con_pag.cd_fornecedor, '', Decode(:CAMPO_CF_PSN_N_CAD_FOR, 'S', 'FORNECEDOR NÃO CADASTRADO',CON_PAG.DS_FORNECEDOR),fornec.nm_fornecedor) nm_fornec
      , fornec.cd_fornecedor                                             cd_fornec
      , Trunc(:CAMPO_DT_INIC)                              dt_lancamento
      , NULL dt_vencimento
      , NULL dt_pagamento
      , 'SALDO MÊS ANTERIOR'                                             ds_hist
      , NULL                                                             nr_doc
      , 0                                                                vl_credito
      , 0                                                                vl_debito
      , con_pag.cd_multi_empresa                                         cd_multi_empresa
    FROM dbamv.con_pag
      , dbamv.historico_padrao hist_p
      , dbamv.fornecedor fornec
      , (SELECT pagcon_pag.cd_itcon_pag cd_itcon_pag
              , it.cd_con_pag
              , pagcon_pag.vl_pago - nvl(pagcon_pag.vl_acrescimo,0) + nvl(pagcon_pag.vl_desconto,0) + Nvl(imposto_pg.vl_detalhamento,0) vl_pago
              , pagcon_pag.vl_pago vl_pago_liquido
              , TRUNC(pagcon_pag.dt_pagamento) dt_pagamento
              , TRUNC(pagcon_pag.dt_baixa) dt_baixa
              , decode( pagcon_pag.tp_pagamento, '1', 'BORDERO'
                                                , '2', 'BORDERO'
                                                , '3', 'CHQ'
                                                , '4', 'DINHEIRO'
                                                , '5', 'DEBITO C/C'
                                                , '6', 'PRT. CONTAS'
                                                , '7', 'QUIT. DEVOL'
                                                , '8', 'TED'
                                                , 'B', 'BX CONTABIL'
                                                , 'Q', 'BX DESC'
                                                , 'E', 'ENC. CONTAS'
                                                ,'10', 'PIX')  ds_doc_pagto
              , decode(pagcon_pag.tp_pagamento,'1',bordero.cd_bordero
                                              ,'2',bordero.cd_bordero
                                              ,'3',cheque.nr_cheque
                                              ,'4',to_number(null)
                                              ,'10',to_number(null)) cd_doc_pagto
              , pagcon_pag.cd_pagcon_pag  cd_pagcon_pag
              , pagcon_pag.ds_pagcon_pag
              , pagcon_pag.cd_multi_empresa
            FROM dbamv.pagcon_pag
              , dbamv.itcon_pag it
              , dbamv.con_cor
              , dbamv.bordero
              , dbamv.cheque
              , dbamv.mov_caixa
              , (SELECT cd_pagcon_pag, Sum(VL_DETALHAMENTO) vl_detalhamento
                    FROM dbamv.tip_detcon_pag
                GROUP BY cd_pagcon_pag) imposto_pg
          WHERE pagcon_pag.cd_con_cor = con_cor.cd_con_cor(+)
            AND pagcon_pag.cd_bordero = bordero.cd_bordero(+)
            AND pagcon_pag.cd_cheque = cheque.cd_cheque(+)
            and it.cd_itcon_pag = pagcon_pag.cd_itcon_pag(+)
            AND pagcon_pag.cd_pagcon_pag =  imposto_pg.cd_pagcon_pag(+)
            And Not Exists(Select 'X'
                              from dbamv.itcon_pag itcp
                            where itcp.cd_con_pag_agrup = it.cd_con_pag
                          )
            AND pagcon_pag.cd_pagcon_pag =  mov_caixa.cd_pagcon_pag(+)
            AND ((pagcon_pag.cd_con_cor is not null :CAMPO_CF_MULTI_EMPRESA_CONCOR) or  pagcon_pag.cd_con_cor is null)  
            AND ((pagcon_pag.cd_caixa is not null :CAMPO_CF_MULTI_EMPRESA_CAIXA) or  pagcon_pag.cd_caixa is null)
            :CAMPO_CF_BAIXA_DEVOLUCAO) pagamento 
  where con_pag.cd_historico_padrao = hist_p.cd_historico_padrao(+)
    AND con_pag.cd_con_pag_dev IS NULL
    AND con_pag.tp_adiantamento IS NULL
    AND con_pag.vl_bruto_conta <> 0
    and con_pag.cd_con_pag = pagamento.cd_con_pag
    :CAMPO_CF_COND_REDUZIDO
    )
  )
