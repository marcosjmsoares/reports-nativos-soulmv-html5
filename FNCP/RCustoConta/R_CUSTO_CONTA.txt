--R_CUSTO_CONTA
SELECT * FROM (
SELECT   V.CD_CONTA Cd_Conta
        ,V.DS_CONTA Ds_Conta
        ,V.DATA Data
        ,V.HISTORICO Historico
        ,V.NR_DOC Nr_Doc
        ,V.VL_ENTRADA Vl_Entrada
        ,V.VL_SAIDA Vl_Saida
        ,V.SITUACAO Situacao
        ,V.Cd_Con_Finac Cd_Con_Finac
FROM     DBAMV.V_CUSTO_CONTA   V
WHERE    V.CD_MULTI_EMPRESA  =    :CAMPO_CD_MULTI_EMPRESA
AND ( TRUNC( V.DATA ) BETWEEN :CAMPO_DT_INICIAL and :CAMPO_DT_FINAL )
)
ORDER BY CD_CONTA,DS_CONTA,
CD_CON_FINAC,DATA,HISTORICO,NR_DOC,SITUACAO
