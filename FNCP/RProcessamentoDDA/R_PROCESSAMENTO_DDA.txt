--R_PROCESSAMENTO_DDA
SELECT CD_MULTI_EMPRESA,
       DS_MULTI_EMPRESA,
       EMPRESA,
       NR_SEQUENCIAL_ARQUIVO,
       DATA_GERACAO_ARQUIVO,
       DATA_IMPORTACAO_ARQUIVO,
       CD_TITULO_BANCARIO,
       NR_CGC_CPF,
       NM_FORNECEDOR,
       NR_TITULO_BANCARIO,
       DT_VENCIMENTO,
       DT_EMISSAO,
       DT_DESCONTO,
       VL_TITULO_BANCARIO,
       VL_DESCONTO,
       NR_SEQUENCIA,
       NR_CODIGO_BARRAS,
       TP_CARTEIRA_BANCARIA,
       TP_MOVIMENTO,
       CD_AGENCIA,
       SN_CONCILIADO,
       CD_ITCON_PAG,
       NOME_ARQUIVO
  FROM (SELECT TB.CD_MULTI_EMPRESA,
               DS_MULTI_EMPRESA,
               TB.CD_MULTI_EMPRESA||' - '||DS_MULTI_EMPRESA EMPRESA,
               NR_SEQUENCIAL_ARQUIVO,
               DATA_GERACAO_ARQUIVO,
               DATA_IMPORTACAO_ARQUIVO,
               CD_TITULO_BANCARIO,
               NR_CGC_CPF,
               NM_FORNECEDOR,
               NR_TITULO_BANCARIO,
               DT_VENCIMENTO,
               DT_EMISSAO,
               DT_DESCONTO,
               VL_TITULO_BANCARIO,
               VL_DESCONTO,
               NR_SEQUENCIA,
               NR_CODIGO_BARRAS,
               TP_CARTEIRA_BANCARIA,
               TP_MOVIMENTO,
               CD_AGENCIA,
               NVL(SN_CONCILIADO,'N') SN_CONCILIADO,
               CD_ITCON_PAG,
               NOME_ARQUIVO
          FROM DBAMV.TITULO_BANCARIO TB, DBAMV.MULTI_EMPRESAS ME
         WHERE TB.CD_MULTI_EMPRESA = ME.CD_MULTI_EMPRESA (+)
           AND NR_SEQUENCIAL_ARQUIVO IS NOT NULL)
   ORDER BY CD_MULTI_EMPRESA,
                     NR_SEQUENCIAL_ARQUIVO,
                     DATA_GERACAO_ARQUIVO,
                     DATA_IMPORTACAO_ARQUIVO,
                     DT_VENCIMENTO,
                     NM_FORNECEDOR
