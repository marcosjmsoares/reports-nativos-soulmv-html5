--R_CON_RECEB_SIMP_R_14_Q_1
SELECT * FROM (
SELECT * FROM (
select f.cd_fornecedor
,decode(c.cd_fornecedor,'', c.nm_cliente, f.nm_fantasia)	nm_fornec
,sum (VL_TRIBUTO)  vl_detalhamento
,tp.cd_detalhamento
from dbamv.tip_det_reccon_rec tp,  dbamv.reccon_rec rr , dbamv.fornecedor f , dbamv.con_rec c, itcon_rec i
 where  1=1
 AND i.CD_ITCON_REC = rr.CD_ITCON_REC
 and c.cd_fornecedor = f.cd_fornecedor(+)
 and i.CD_CON_REC = c.CD_CON_REC
 AND tp.CD_RECCON_REC = rr.CD_RECCON_REC
 and tp.CD_RECCON_REC IN(
SELECT  RECCON_REC.cd_reccon_rec
FROM   DBAMV.ITCON_REC
            ,DBAMV.CON_REC           CON_REC
            ,DBAMV.FORNECEDOR
            ,DBAMV.HISTORICO_PADRAO  HIST_PADRAO
            ,DBAMV.RECCON_REC
            ,DBAMV.PLANO_CONTAS      PC
            ,DBAMV.PLANO_CONTAS      CTA_RESULT
WHERE                                                     = 1
and        :CAMPO_SN_IMPOSTO = 'S'
AND      ITCON_REC.CD_CON_REC                             = CON_REC.CD_CON_REC
AND      CON_REC.CD_PREVISAO IS NULL
AND      CON_REC.CD_MULTI_EMPRESA                   = :CAMPO_CD_MULTI_EMPRESA
AND      ITCON_REC.CD_ITCON_REC                          = RECCON_REC.CD_ITCON_REC
AND      CON_REC.CD_FORNECEDOR                         = FORNECEDOR.CD_FORNECEDOR(+)
AND      CON_REC.CD_HISTORICO_PADRAO             = HIST_PADRAO.CD_HISTORICO_PADRAO(+)
AND      CON_REC.CD_REDUZIDO                               = PC.CD_REDUZIDO(+)
AND      PC.CD_REDUZIDO_RESULTADO                   = CTA_RESULT.CD_REDUZIDO(+)
AND      ITCON_REC .TP_QUITACAO                           <> 'L'
and        con_rec.cd_con_rec = c.cd_con_rec
and        con_rec.cd_multi_empresa                  = :CAMPO_CD_MULTI_EMPRESA
and        RECCON_REC.dt_estorno is null
:CAMPO_CF_PF_TP_CONTA
)
GROUP BY f.cd_fornecedor
,decode(c.cd_fornecedor,'', c.nm_cliente, f.nm_fantasia)
,tp.cd_detalhamento
)
WHERE CD_FORNECEDOR = :CAMPO_CD_FORNECEDOR_CLI
)
ORDER BY CD_FORNECEDOR,NM_FORNEC
