--R_CON_APAG_AGRUP
SELECT * FROM (
SELECT    AGRUP.CD_CON_PAG                    	CD_CON_PAG_G
         ,AGRUP.DS_CON_PAG                    	DS_CON_PAG_G
         ,DECODE(AGRUP.CD_FORNECEDOR, '',AGRUP.DS_FORNECEDOR, FORNEC_G.NM_FORNECEDOR ) FORNEC_G
         ,HIST_PADRAO_G.DS_HISTORICO_PADRAO   DS_HIST_PADRAO_G
         ,AGRUP.DT_LANCAMENTO                 	DT_LANC_G
         ,AGRUP.DT_EMISSAO                    	DT_EMISSAO_G
         ,AGRUP.NR_DOCUMENTO                  	NR_DOC_G
         ,ITAGRUP.DT_VENCIMENTO               	DT_VENCTO_G
         ,ITAGRUP.TP_QUITACAO                 	TP_QUITACAO_G
         ,ITAGRUP.VL_DUPLICATA                	VL_DUPLIC_G
         ,(ITAGRUP.VL_DUPLICATA - NVL(ITAGRUP.VL_SOMA_BAIXADA,0))           VL_DEVIDO_G
         ,CON_PAG_D.CD_CON_PAG                    	CD_CON_PAG_D
         ,CON_PAG_D.DS_CON_PAG                    	DS_CON_PAG_D
         ,DECODE(CON_PAG_D.CD_FORNECEDOR, '',CON_PAG_D.DS_FORNECEDOR, FORNEC_D.NM_FORNECEDOR ) FORNEC_D
         ,HIST_PADRAO_D.DS_HISTORICO_PADRAO       DS_HIST_PADRAO_D
         ,CON_PAG_D.DT_LANCAMENTO                 DT_LANC_D
         ,CON_PAG_D.DT_EMISSAO                    	DT_EMISSAO_D
         ,CON_PAG_D.NR_DOCUMENTO                 NR_DOC_D
         ,ITCON_PAG_D.DT_VENCIMENTO              DT_VENCTO_D
         ,ITCON_PAG_D.TP_QUITACAO                 	TP_QUITACAO_D
         ,ITCON_PAG_D.VL_DUPLICATA                	VL_DUPLIC_D
FROM      DBAMV.CON_PAG                       	AGRUP
         ,DBAMV.ITCON_PAG                     	ITAGRUP
         ,DBAMV.HISTORICO_PADRAO              	HIST_PADRAO_G
         ,DBAMV.FORNECEDOR                    	FORNEC_G
         ,DBAMV.ITCON_PAG                     	ITCON_PAG_D
         ,DBAMV.CON_PAG                       		CON_PAG_D
         ,DBAMV.HISTORICO_PADRAO              	HIST_PADRAO_D
         ,DBAMV.FORNECEDOR                    	FORNEC_D
         ,DBAMV.TIP_DOC			TIP_DOC
WHERE     AGRUP.CD_MULTI_EMPRESA           = :CAMPO_CD_MULTI_EMPRESA
AND       ITAGRUP.TP_QUITACAO             	<> 'Q'
AND  AGRUP.CD_CON_PAG                 	= ITAGRUP.CD_CON_PAG
AND  AGRUP.CD_HISTORICO_PADRAO        	= HIST_PADRAO_G.CD_HISTORICO_PADRAO(+)
AND  AGRUP.CD_FORNECEDOR              	= FORNEC_G.CD_FORNECEDOR(+)
AND ITCON_PAG_D.CD_CON_PAG_AGRUP 	IS NOT NULL
AND ITAGRUP.CD_CON_PAG             	= VERIF_CD_AGRUP_MAX(ITCON_PAG_D.CD_CON_PAG_AGRUP)
AND NOT EXISTS (SELECT CD_CON_PAG_AGRUP
                                FROM DBAMV.ITCON_PAG ITCON_PAG_PAI
                                WHERE CD_CON_PAG_AGRUP  = ITCON_PAG_D.CD_CON_PAG)
AND  ITCON_PAG_D.CD_CON_PAG          	= CON_PAG_D.CD_CON_PAG
AND  CON_PAG_D.CD_HISTORICO_PADRAO    = HIST_PADRAO_D.CD_HISTORICO_PADRAO(+)
AND  CON_PAG_D.CD_TIP_DOC          	= TIP_DOC.CD_TIP_DOC
AND  CON_PAG_D.CD_FORNECEDOR          	= FORNEC_D.CD_FORNECEDOR(+)
ORDER BY  FORNEC_G, DT_VENCTO_G
)
ORDER BY FORNEC_G,
DT_VENCTO_G,DT_EMISSAO_G,DT_LANC_G,DS_CON_PAG_G,DS_HIST_PADRAO_G,CD_CON_PAG_G,NR_DOC_G,TP_QUITACAO_G,VL_DUPLIC_G,VL_DEVIDO_G, FORNEC_G, DT_VENCTO_G
