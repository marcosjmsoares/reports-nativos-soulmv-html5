--R_RECIBO_PAGAMENTO
select   con_pag.ds_fornecedor       ds_fornecedor,
           con_pag.ds_con_pag          ds_con_pag,
           con_pag.cd_con_pag          cd_con_pag,
           pagcon_pag.vl_pago vl_soma_pago,
           pagcon_pag.dt_pagamento   dt_pagamento,
           fornecedor.nm_fornecedor    nm_fornecedor,
           fornecedor.nr_cgc_cpf       nr_cgc_cpf,
           con_pag.nr_documento     nr_documento
from dbamv.con_pag,
        dbamv.itcon_pag,
        dbamv.pagcon_pag,
        dbamv.fornecedor
where  :CAMPO_CHEQUE = 'N'
    and pagcon_pag.cd_pagcon_pag = :CAMPO_CD_PAGCON_PAG
    and itcon_pag.cd_con_pag   = con_pag.cd_con_pag
    and itcon_pag.cd_itcon_pag = pagcon_pag.cd_itcon_pag
    and itcon_pag.cd_con_pag_agrup is null
    and con_pag.cd_fornecedor = fornecedor.cd_fornecedor (+)
