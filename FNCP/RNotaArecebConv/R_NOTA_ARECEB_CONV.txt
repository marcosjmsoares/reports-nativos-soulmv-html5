--R_NOTA_ARECEB_CONV
SELECT * FROM (
SELECT
CONV.NM_CONVENIO    CONVENIO
, IT.DT_VENCIMENTO  VENCIMENTO
, NF.CD_NOTA_FISCAL NF
, NF.NR_ID_NOTA_FISCAL
, NF.DT_EMISSAO    EMISSAO_NF
, NF.VL_TOTAL_NOTA  VALOR_NF_ORIGINAL
, DECODE (IT.TP_QUITACAO, 'V', 'Previsto', 'C', 'Comprometido', 'P', 'Parc. Quitado', 'Q', 'Quitado', 'L', 'Cancelado',
'R','Protestado', 'G', 'Quitado com Glosa', IT.TP_QUITACAO) TIPO_BAIXA
, NVL(NVL(IT.VL_DUPLICATA,0) - NVL(IT.VL_SOMA_RECEBIDO,0),0) SALDO_PENDENTE
FROM DBAMV.ITCON_REC   IT
     , DBAMV.CON_REC     CON
     , DBAMV.NOTA_FISCAL NF
     , DBAMV.CONVENIO    CONV
WHERE CON.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND CON.TP_CON_REC = 'C'
AND IT.CD_CON_REC = CON.CD_CON_REC
AND IT.CD_CON_REC_AGRUP IS NULL
AND CONV.CD_CONVENIO = NF.CD_CONVENIO
AND NF.CD_NOTA_FISCAL = CON.CD_NOTA_FISCAL
AND CON.CD_PREVISAO IS NULL
AND CON.dt_cancelamento IS NULL
ORDER BY CONV.NM_CONVENIO, IT.DT_VENCIMENTO, NF.CD_NOTA_FISCAL, NF.DT_EMISSAO
)
ORDER BY CONVENIO,EMISSAO_NF,VENCIMENTO,NR_ID_NOTA_FISCAL
