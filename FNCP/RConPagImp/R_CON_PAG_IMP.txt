--R_CON_PAG_IMP
SELECT  'CREC'                                   tp_conta
       ,lote.cd_lote                             cd_lote 
       ,lote.ds_lote                             ds_lote
       ,lote.cd_multi_empresa                    cd_multi_empresa       
       ,emp.ds_multi_empresa                     ds_multi_empresa
       ,lote.cd_detalhamento                     cd_detalhamento
       ,det.ds_detalhamento                      ds_detalhamento
       ,cnta.cd_fornecedor ||' - ' || cnta.nm_cliente nm_fornecedor 
       ,lote.dt_competencia_ini
       ,lote.dt_competencia_fim          
       ,lote.dt_abertura
       ,lote.cd_usuario_abert
       ,lote.dt_fechamento
       ,lote.cd_usuario_fecha
       ,lote.cd_con_pag                              cd_con_pag 
       ,(SELECT Max(x.dt_recebimento) FROM dbamv.itcon_rec y, dbamv.reccon_rec x WHERE y.cd_con_rec = cnta.cd_con_rec AND y.cd_itcon_rec = x.cd_itcon_rec AND x.dt_estorno IS NULL) dt_lancamento
       ,cnta.cd_con_rec                              cd_con_pag_it
       ,cnta.cd_fornecedor || '' || cnta.nm_cliente  nm_fornecedor
       ,cnta.nr_documento                            nr_documento
       ,cnta.nr_serie                                nr_serie
       ,cnta.cd_moeda                                cd_moeda     
       ,item.vl_base                                 vl_base      
       ,item.vl_imposto                              vl_imposto
       ,item.vl_editado                              vl_editado
       ,item.vl_percentual                           vl_percentual      
       ,item.sn_editado                              sn_editado
       ,lote.tp_geracao                              tp_geracao
FROM   dbamv.finan_lote_imposto lote,
       dbamv.finan_lote_imposto_it item,
       dbamv.con_rec cnta,      
       dbamv.tip_detcon_rec deta,
       dbamv.multi_empresas emp,
       dbamv.tip_detalhe det
WHERE  lote.cd_lote = item.cd_lote  
AND lote.cd_multi_empresa = emp.cd_multi_empresa
  AND item.cd_con_pag = cnta.cd_con_rec   
  AND  cnta.cd_con_rec = deta.cd_con_rec
  AND  lote.cd_detalhamento = det.cd_detalhamento
  AND lote.cd_lote = :CAMPO_CD_LOTE
UNION ALL
SELECT  'CPAG'                               tp_conta
       ,lote.cd_lote                             cd_lote 
       ,lote.ds_lote                             ds_lote
       ,lote.cd_multi_empresa                    cd_multi_empresa       
       ,emp.ds_multi_empresa                     ds_multi_empresa
       ,lote.cd_detalhamento                     cd_detalhamento
       ,det.ds_detalhamento                      ds_detalhamento  
       ,cnta.cd_fornecedor ||' - '|| cnta.ds_fornecedor nm_fornecedor
       ,lote.dt_competencia_ini
       ,lote.dt_competencia_fim
       ,lote.dt_abertura
       ,lote.cd_usuario_abert
       ,lote.dt_fechamento
       ,lote.cd_usuario_fecha
       ,lote.cd_con_pag                              cd_con_pag       
       ,(SELECT Max(x.dt_pagamento) FROM dbamv.itcon_pag y, dbamv.pagcon_pag x WHERE y.cd_con_pag = cnta.cd_con_pag AND y.cd_itcon_pag = x.cd_itcon_pag AND x.dt_estorno IS NULL) dt_lancamento
       ,cnta.cd_con_pag                              cd_con_pag_it
       ,cnta.cd_fornecedor || '' || cnta.ds_fornecedor  nm_fornecedor
       ,cnta.nr_documento                            nr_documento
       ,cnta.nr_serie                                nr_serie
       ,cnta.cd_moeda                                cd_moeda       
       ,item.vl_base                                 vl_base      
       ,item.vl_imposto                              vl_imposto
       ,item.vl_editado                              vl_editado
       ,item.vl_percentual                           vl_percentual      
       ,item.sn_editado                              sn_editado
       ,lote.tp_geracao                              tp_geracao
FROM   dbamv.finan_lote_imposto lote,
       dbamv.finan_lote_imposto_it item,         
       dbamv.con_pag cnta,     
       dbamv.multi_empresas emp,
       dbamv.tip_detalhe det
WHERE lote.cd_lote = item.cd_lote  
  AND lote.cd_multi_empresa = emp.cd_multi_empresa
  AND item.cd_con_pag = cnta.cd_con_pag    
  AND  lote.cd_detalhamento = det.cd_detalhamento  
  AND  lote.cd_lote = :CAMPO_CD_LOTE
