--R_RECIBO_DISTRI_CONTA
SELECT a.cd_reccon_rec   cd_movimento,
       c.cd_reduzido     cd_conta,
       d.cd_contabil     cd_contabil,
       d.ds_conta        ds_conta,
       To_Number(NULL)   cd_ccusto,
       To_Number(NULL)   vl_debito,
       a.vl_recebido     vl_credito
FROM   dbamv.reccon_rec a,
       dbamv.itcon_rec b,
       dbamv.con_rec c,
       dbamv.plano_contas d
WHERE  a.cd_itcon_rec = b.cd_itcon_rec
  AND  b.cd_con_rec = c.cd_con_rec
  AND  c.cd_reduzido = d.cd_reduzido
  AND  a.cd_reccon_rec = :CAMPO_CF_CONTA
UNION ALL
SELECT a.cd_reccon_rec   cd_movimento,
       c.cd_reduzido     cd_conta,
       d.cd_contabil     cd_contabil,
       d.ds_conta        ds_conta,
       e.cd_item_res     cd_ccusto,
       e.vl_rat_reccon_rec vl_debito,
       To_Number(NULL)   vl_credito
FROM   dbamv.reccon_rec a,
       dbamv.itcon_rec b,
       dbamv.con_rec c,
       dbamv.plano_contas d,
       dbamv.rat_reccon_rec e
       , (SELECT i.cd_con_rec
            FROM dbamv.reccon_rec r
               , dbamv.itcon_rec i
           WHERE r.cd_itcon_rec = i.cd_itcon_rec
             AND r.cd_reccon_rec = :CAMPO_CF_CONTA) f
WHERE  a.cd_itcon_rec = b.cd_itcon_rec
  AND  b.cd_con_rec = c.cd_con_rec
  AND  a.cd_reccon_rec = e.cd_reccon_rec
  AND  e.cd_reduzido = d.cd_reduzido
  AND  (f.cd_con_rec = b.cd_itcon_rec OR f.cd_con_rec = b.cd_con_rec_agrup)
