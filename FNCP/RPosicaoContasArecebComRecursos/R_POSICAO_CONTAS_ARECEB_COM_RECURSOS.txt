--R_POSICAO_CONTAS_ARECEB_COM_RECURSOS
SELECT *
   FROM
  (SELECT CD_MULTI_EMPRESA      CD_MULTI_EMPRESA,
    DS_MULTI_EMPRESA      DS_MULTI_EMPRESA,
    CD_FONTE_PAGADORA CD_FONTE_PAGADORA,
    NM_FANTASIA NOME_FANTASIA                ,
    CD_CON_REC,
    NR_DOCUMENTO NR_DOCUMENTO                ,
    SUM(VL_A_VENCER) VL_A_VENCER             ,
    SUM(VL_VENCIDO) VL_VENCIDO               ,
    SUM(TOTAL_A_RECEBER) TOTAL_A_RECEBER     ,
    SUM(VENCIDO_01_30) VENCIDO_01_30         ,
    SUM(VENCIDO_31_60) VENCIDO_31_60         ,
    SUM(VENCIDO_61_90) VENCIDO_61_90         ,
    SUM(VENCIDO_91_180) VENCIDO_91_180       ,
    SUM(VENCIDO_181_360) VENCIDO_181_360     ,
    SUM(VENCIDO_361) VENCIDO_361             ,
    SUM(TOTAL_VENCIDO) TOTAL_VENCIDO         ,
    SUM(A_VENCER_01_30) A_VENCER_01_30       ,
    SUM(A_VENCER_31_60) A_VENCER_31_60       ,
    SUM(A_VENCER_61_90) A_VENCER_61_90       ,
    SUM(A_VENCER_91_180) A_VENCER_91_180     ,
    SUM(A_VENCER_181_360) A_VENCER_181_360   ,
    SUM(A_VENCER_361) A_VENCER_361           ,
    SUM(TOTAL_A_VENCER ) TOTAL_A_VENCER
     FROM
    (SELECT
    CD_MULTI_EMPRESA      CD_MULTI_EMPRESA,
    DS_MULTI_EMPRESA      DS_MULTI_EMPRESA,
    CD_CON_REC                   ,
      CD_FONTE_PAGADORA CD_FONTE_PAGADORA,
      NM_FANTASIA NM_FANTASIA            ,
      NR_DOCUMENTO NR_DOCUMENTO          ,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM(VL_A_VENCER)
        ELSE 0
      END VL_A_VENCER,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM(VL_VENCIDO)
        ELSE 0
      END VL_VENCIDO,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN (SUM(VL_A_VENCER) + SUM(VL_VENCIDO))
        ELSE 0
      END TOTAL_A_RECEBER,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P01_30 * DECODE(STATUS,'VENCIDO',1,0)) )
        ELSE 0
      END VENCIDO_01_30,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P31_60 * DECODE(STATUS,'VENCIDO',1,0)) )
        ELSE 0
      END VENCIDO_31_60,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P61_90 * DECODE(STATUS,'VENCIDO',1,0)) )
        ELSE 0
      END VENCIDO_61_90,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P91_180 * DECODE(STATUS,'VENCIDO',1,0)) )
        ELSE 0
      END VENCIDO_91_180,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P181_360 * DECODE(STATUS,'VENCIDO',1,0)) )
        ELSE 0
      END VENCIDO_181_360,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P361 * DECODE(STATUS,'VENCIDO',1,0)) )
        ELSE 0
      END VENCIDO_361,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P01_30 * DECODE(STATUS,'VENCIDO',1,0))) + SUM( (P31_60 * DECODE(STATUS,'VENCIDO',1,0)) ) + SUM( (P61_90 * DECODE(STATUS,'VENCIDO',1,0))) + SUM( (P91_180 * DECODE(STATUS,'VENCIDO',1,0)) ) + SUM( (P181_360 * DECODE(STATUS,'VENCIDO',1,0)))+ SUM( (P361 * DECODE(STATUS,'VENCIDO',1,0)) )
        ELSE 0
      END TOTAL_VENCIDO,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P01_30 * DECODE(STATUS,'VENCIDO',0,1)) )
        ELSE 0
      END A_VENCER_01_30,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P31_60 * DECODE(STATUS,'VENCIDO',0,1)) )
        ELSE 0
      END A_VENCER_31_60,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P61_90 * DECODE(STATUS,'VENCIDO',0,1)) )
        ELSE 0
      END A_VENCER_61_90,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P91_180 * DECODE(STATUS,'VENCIDO',0,1)) )
        ELSE 0
      END A_VENCER_91_180,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P181_360 * DECODE(STATUS,'VENCIDO',0,1)) )
        ELSE 0
      END A_VENCER_181_360,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P361 * DECODE(STATUS,'VENCIDO',0,1)) )
        ELSE 0
      END A_VENCER_361,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P01_30 * DECODE(STATUS,'VENCIDO',0,1)) ) + SUM( (P31_60 * DECODE(STATUS,'VENCIDO',0,1)) ) + SUM( (P61_90 * DECODE(STATUS,'VENCIDO',0,1)) ) + SUM( (P91_180 * DECODE(STATUS,'VENCIDO',0,1)) ) + SUM( (P181_360 * DECODE(STATUS,'VENCIDO',0,1)))+ SUM( (P361 * DECODE(STATUS,'VENCIDO',0,1)) )
        ELSE 0
      END TOTAL_A_VENCER
       FROM
      (SELECT
        M.CD_MULTI_EMPRESA      CD_MULTI_EMPRESA,
        M.DS_MULTI_EMPRESA      DS_MULTI_EMPRESA,
        C.CD_FORNECEDOR CD_FONTE_PAGADORA                                                       ,
        F.NM_FANTASIA NM_FANTASIA                                                                     ,
        I.DT_VENCIMENTO DT_VENCIMENTO                                                                 ,
        NVL(I.VL_DUPLICATA,0) - NVL(R.VL_SOMA_RECEBIDO,0) - NVL(Aceite.vl_movimentacao,0) VL_PREVISTO ,
        C.CD_CON_REC CD_CON_REC                                                                       ,
        I.CD_ITCON_REC CD_ITCON_REC                                                                   ,
        AGRUPAMENTO.CD_CON_REC CD_CON_REC_AGRUP                                                       ,
        DECODE ('','S',NULL, C.NR_DOCUMENTO) NR_DOCUMENTO                             ,
        CASE
          WHEN TRUNC((TRUNC(:CAMPO_DT_BASE)-TRUNC(I.DT_VENCIMENTO))) > 0
          THEN 'VENCIDO'
          ELSE 'A_VENCER'
        END AS "STATUS" ,
        CASE
          WHEN TRUNC((TRUNC(:CAMPO_DT_BASE)-TRUNC(I.DT_VENCIMENTO))) > 0
          THEN 0
          ELSE NVL(I.VL_DUPLICATA,0) - NVL(R.VL_SOMA_RECEBIDO,0) - NVL(Aceite.vl_movimentacao,0)
        END AS "VL_A_VENCER" ,
        CASE
          WHEN TRUNC((TRUNC(:CAMPO_DT_BASE)-TRUNC(I.DT_VENCIMENTO))) > 0
          THEN NVL(I.VL_DUPLICATA,0)      - NVL(R.VL_SOMA_RECEBIDO,0) - NVL(Aceite.vl_movimentacao,0)
          ELSE 0
        END AS "VL_VENCIDO" ,
        CASE
          WHEN ABS(TRUNC(:CAMPO_DT_BASE) - TRUNC(I.DT_VENCIMENTO)) BETWEEN 0 AND 30
          THEN NVL(I.VL_DUPLICATA,0)    - NVL(R.VL_SOMA_RECEBIDO,0) - NVL(Aceite.vl_movimentacao,0)
          ELSE 0
        END AS "P01_30" ,
        CASE
          WHEN ABS(TRUNC(:CAMPO_DT_BASE) - TRUNC(I.DT_VENCIMENTO)) BETWEEN 31 AND 60
          THEN NVL(I.VL_DUPLICATA,0)    - NVL(R.VL_SOMA_RECEBIDO,0) - NVL(Aceite.vl_movimentacao,0)
          ELSE 0
        END AS "P31_60" ,
        CASE
          WHEN ABS(TRUNC(:CAMPO_DT_BASE) - TRUNC(I.DT_VENCIMENTO)) BETWEEN 61 AND 90
          THEN NVL(I.VL_DUPLICATA,0)    - NVL(R.VL_SOMA_RECEBIDO,0) - NVL(Aceite.vl_movimentacao,0)
          ELSE 0
        END AS "P61_90" ,
        CASE
          WHEN ABS(TRUNC(:CAMPO_DT_BASE) - TRUNC(I.DT_VENCIMENTO)) BETWEEN 91 AND 180
          THEN NVL(I.VL_DUPLICATA,0)    - NVL(R.VL_SOMA_RECEBIDO,0) - NVL(Aceite.vl_movimentacao,0)
          ELSE 0
        END AS "P91_180" ,
        CASE
          WHEN ABS(TRUNC(:CAMPO_DT_BASE) - TRUNC(I.DT_VENCIMENTO)) BETWEEN 181 AND 360
          THEN NVL(I.VL_DUPLICATA,0)    - NVL(R.VL_SOMA_RECEBIDO,0) - NVL(Aceite.vl_movimentacao,0)
          ELSE 0
        END AS "P181_360" ,
        CASE
          WHEN ABS(TRUNC(:CAMPO_DT_BASE) - TRUNC(I.DT_VENCIMENTO)) > 360
          THEN NVL(I.VL_DUPLICATA,0)    - NVL(R.VL_SOMA_RECEBIDO,0) - NVL(Aceite.vl_movimentacao,0)
          ELSE 0
        END AS "P361"
         FROM
        (SELECT RECEBIMENTO.CD_ITCON_REC CD_ITCON_REC,
          SUM (NVL(RECEBIMENTO.VL_RECEBIDO,0) ) VL_SOMA_RECEBIDO
           FROM
          (SELECT RECCON_REC.CD_ITCON_REC,
            NVL(RECCON_REC.VL_RECEBIDO,0) + NVL(RECCON_REC.VL_DESCONTO,0) - NVL(RECCON_REC.VL_ACRESCIMO,0) + NVL(RECCON_REC.VL_IMPOSTO_RETIDO,0) VL_RECEBIDO
             FROM DBAMV.RECCON_REC
            WHERE 1=1
            and decode(:CAMPO_SN_DATA_CRED_CONV,'S',TRUNC (nvl(DBAMV.PACK_FNCP.fnc_Retorna_Dt_Cred_Convenio(RECCON_REC.cd_reccon_rec), RECCON_REC.DT_RECEBIMENTO)),
            TRUNC(RECCON_REC.DT_RECEBIMENTO)) <= :CAMPO_DT_BASE
          AND ( ( RECCON_REC.DT_ESTORNO            IS NULL )
          OR ( ( RECCON_REC.DT_ESTORNO             IS NOT NULL )
          AND (TRUNC(RECCON_REC.DT_ESTORNO)         > :CAMPO_DT_BASE) ))
          ) RECEBIMENTO
       GROUP BY RECEBIMENTO.CD_ITCON_REC
        ) R                  ,
        DBAMV.CON_REC C      ,
        DBAMV.ITCON_REC I    ,
        DBAMV.FORNECEDOR F   ,
        DBAMV.MULTI_EMPRESAS M,
        (SELECT CR.CD_CON_REC,
          CR.DT_LANCAMENTO
           FROM DBAMV.CON_REC CR
          WHERE CR.CD_PROCESSO IN
          (SELECT CD_PROCESSO
             FROM DBAMV.PROCESSO
            WHERE CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
          AND CD_ESTRUTURAL       IN ( '1.4.1.2.1','1.4.1.1.8','1.4.1.1.9','1.4.1.1.10' ,'1.4.1.1.11','1.4.1.1.12','1.4.1.1.13','1.4.1.1.14' )
          )
        AND TRUNC(CR.DT_EMISSAO) <= TRUNC(:CAMPO_DT_BASE)
        ) AGRUPAMENTO          ,
        (SELECT mg.cd_itcon_rec,
          SUM(NVL(mg.vl_movimentacao,0)) vl_movimentacao
           FROM dbamv.mov_glosas mg,
          dbamv.processo p
          WHERE mg.cd_processo         = p.cd_processo
        AND p.cd_estrutural            = '3.1.2.1.2'
        AND TRUNC(mg.dt_movimentacao) <= :CAMPO_DT_BASE
       GROUP BY mg.cd_itcon_rec
        ) Aceite
        WHERE C.CD_CON_REC     = I.CD_CON_REC
      AND C.CD_FORNECEDOR      = F.CD_FORNECEDOR
      AND I.CD_CON_REC_AGRUP   = AGRUPAMENTO.CD_CON_REC(+)
      AND C.CD_MULTI_EMPRESA   = M.CD_MULTI_EMPRESA
      AND C.VL_PREVISTO        > 0
      AND C.CD_PREVISAO       IS NULL
      AND TRUNC(C.DT_EMISSAO) <= :CAMPO_DT_BASE
      AND R.CD_ITCON_REC (+)   = I.CD_ITCON_REC
      AND i.cd_itcon_rec       = Aceite.cd_itcon_rec (+)
      AND C.CD_PROCESSO NOT   IN
        (SELECT CD_PROCESSO
           FROM DBAMV.PROCESSO
          WHERE CD_ESTRUTURAL       IN ( '1.4.1.2.1','1.4.1.1.8','1.4.1.1.9', '1.4.1.1.10','1.4.1.1.11','1.4.1.1.12', '1.4.1.1.13','1.4.1.1.14' )
        )
     GROUP BY M.CD_MULTI_EMPRESA,
        M.DS_MULTI_EMPRESA,
         C.CD_FORNECEDOR                                                             ,
        F.NM_FANTASIA                                                                     ,
        I.DT_VENCIMENTO                                                                   ,
        NVL(I.VL_DUPLICATA,0) - NVL(R.VL_SOMA_RECEBIDO,0) - NVL(Aceite.vl_movimentacao,0) ,
        C.CD_CON_REC                                                                      ,
        I.CD_ITCON_REC                                                                    ,
        AGRUPAMENTO.CD_CON_REC                                                            ,
        DECODE ('','S',NULL, C.NR_DOCUMENTO)
      )
   GROUP BY
    CD_MULTI_EMPRESA,
    DS_MULTI_EMPRESA,
    CD_FONTE_PAGADORA,
      NM_FANTASIA            ,
      NR_DOCUMENTO           ,
      CD_CON_REC             ,
      CD_CON_REC_AGRUP
    HAVING SUM(vl_vencido) >= 0
      UNION
     SELECT
      CD_MULTI_EMPRESA,
      DS_MULTI_EMPRESA,
      CD_CON_REC                                                                                                                                                                                                                                                                                                   ,
      CD_FONTE_PAGADORA CD_FONTE_PAGADORA                                                                                                                                                                                                                                                                                ,
      NM_FANTASIA NM_FANTASIA                                                                                                                                                                                                                                                                                            ,
      NR_DOCUMENTO NR_DOCUMENTO                                                                                                                                                                                                                                                                                          ,
      SUM(VL_A_VENCER) VL_A_VENCER                                                                                                                                                                                                                                                                                       ,
      SUM(VL_VENCIDO) VL_VENCIDO                                                                                                                                                                                                                                                                                         ,
      (SUM(VL_A_VENCER) + SUM(VL_VENCIDO)) TOTAL_A_RECEBER                                                                                                                                                                                                                                                               ,
      SUM( (P01_30      * DECODE(STATUS,'VENCIDO',1,0)) ) VENCIDO_01_30                                                                                                                                                                                                                                                  ,
      SUM( (P31_60      * DECODE(STATUS,'VENCIDO',1,0)) ) VENCIDO_31_60                                                                                                                                                                                                                                                  ,
      SUM( (P61_90      * DECODE(STATUS,'VENCIDO',1,0)) ) VENCIDO_61_90                                                                                                                                                                                                                                                  ,
      SUM( (P91_180     * DECODE(STATUS,'VENCIDO',1,0)) ) VENCIDO_91_180                                                                                                                                                                                                                                                 ,
      SUM( (P181_360    * DECODE(STATUS,'VENCIDO',1,0)) ) VENCIDO_181_360                                                                                                                                                                                                                                                ,
      SUM( (P361        * DECODE(STATUS,'VENCIDO',1,0)) ) VENCIDO_361                                                                                                                                                                                                                                                    ,
      SUM( (P01_30      * DECODE(STATUS,'VENCIDO',1,0))) + SUM( (P31_60 * DECODE(STATUS,'VENCIDO',1,0)) ) + SUM( (P61_90 * DECODE(STATUS,'VENCIDO',1,0))) + SUM( (P91_180 * DECODE(STATUS,'VENCIDO',1,0)) ) + SUM( (P181_360 * DECODE(STATUS,'VENCIDO',1,0)))+ SUM( (P361 * DECODE(STATUS,'VENCIDO',1,0)) ) TOTAL_VENCIDO,
      SUM( (P01_30      * DECODE(STATUS,'VENCIDO',0,1)) ) A_VENCER_01_30                                                                                                                                                                                                                                                 ,
      SUM( (P31_60      * DECODE(STATUS,'VENCIDO',0,1)) ) A_VENCER_31_60                                                                                                                                                                                                                                                 ,
      SUM( (P61_90      * DECODE(STATUS,'VENCIDO',0,1)) ) A_VENCER_61_90                                                                                                                                                                                                                                                 ,
      SUM( (P91_180     * DECODE(STATUS,'VENCIDO',0,1)) ) A_VENCER_91_180                                                                                                                                                                                                                                                ,
      SUM( (P181_360    * DECODE(STATUS,'VENCIDO',0,1)) ) A_VENCER_181_360                                                                                                                                                                                                                                               ,
      SUM( (P361        * DECODE(STATUS,'VENCIDO',0,1)) ) A_VENCER_361                                                                                                                                                                                                                                                   ,
      SUM( (P01_30      * DECODE(STATUS,'VENCIDO',0,1)) ) + SUM( (P31_60 * DECODE(STATUS,'VENCIDO',0,1)) ) + SUM( (P61_90 * DECODE(STATUS,'VENCIDO',0,1)) ) + SUM( (P91_180 * DECODE(STATUS,'VENCIDO',0,1)) ) + SUM( (P181_360 * DECODE(STATUS,'VENCIDO',0,1)))+ SUM( (P361 * DECODE(STATUS,'VENCIDO',0,1)) ) TOTAL_A_VENCER
       FROM
      (SELECT
        M.CD_MULTI_EMPRESA      CD_MULTI_EMPRESA,
        M.DS_MULTI_EMPRESA      DS_MULTI_EMPRESA,
        C.CD_FORNECEDOR CD_FONTE_PAGADORA                           ,
        F.NM_FANTASIA NM_FANTASIA                                         ,
        I.DT_VENCIMENTO DT_VENCIMENTO                                     ,
        NVL(I.VL_DUPLICATA,0) - NVL(R.VL_SOMA_RECEBIDO,0) VL_PREVISTO     ,
        C.CD_CON_REC CD_CON_REC                                           ,
        I.CD_ITCON_REC CD_ITCON_REC                                       ,
        DECODE ('','S',NULL, C.NR_DOCUMENTO) NR_DOCUMENTO ,
        CASE
          WHEN TRUNC((TRUNC(:CAMPO_DT_BASE)-TRUNC(I.DT_VENCIMENTO))) > 0
          THEN 'VENCIDO'
          ELSE 'A_VENCER'
        END AS "STATUS" ,
        CASE
          WHEN TRUNC((TRUNC(:CAMPO_DT_BASE)-TRUNC(I.DT_VENCIMENTO))) > 0
          THEN 0
          ELSE NVL(I.VL_DUPLICATA,0) - NVL(R.VL_SOMA_RECEBIDO,0)
        END AS "VL_A_VENCER" ,
        CASE
          WHEN TRUNC((TRUNC(:CAMPO_DT_BASE)-TRUNC(I.DT_VENCIMENTO))) > 0
          THEN NVL(I.VL_DUPLICATA,0)      - NVL(R.VL_SOMA_RECEBIDO,0)
          ELSE 0
        END AS "VL_VENCIDO" ,
        CASE
          WHEN ABS(TRUNC(:CAMPO_DT_BASE) - TRUNC(I.DT_VENCIMENTO)) BETWEEN 0 AND 30
          THEN NVL(I.VL_DUPLICATA,0)    - NVL(R.VL_SOMA_RECEBIDO,0)
          ELSE 0
        END AS "P01_30" ,
        CASE
          WHEN ABS(TRUNC(:CAMPO_DT_BASE) - TRUNC(I.DT_VENCIMENTO)) BETWEEN 31 AND 60
          THEN NVL(I.VL_DUPLICATA,0)    - NVL(R.VL_SOMA_RECEBIDO,0)
          ELSE 0
        END AS "P31_60" ,
        CASE
          WHEN ABS(TRUNC(:CAMPO_DT_BASE) - TRUNC(I.DT_VENCIMENTO)) BETWEEN 61 AND 90
          THEN NVL(I.VL_DUPLICATA,0)    - NVL(R.VL_SOMA_RECEBIDO,0)
          ELSE 0
        END AS "P61_90" ,
        CASE
          WHEN ABS(TRUNC(:CAMPO_DT_BASE) - TRUNC(I.DT_VENCIMENTO)) BETWEEN 91 AND 180
          THEN NVL(I.VL_DUPLICATA,0)    - NVL(R.VL_SOMA_RECEBIDO,0)
          ELSE 0
        END AS "P91_180" ,
        CASE
          WHEN ABS(TRUNC(:CAMPO_DT_BASE) - TRUNC(I.DT_VENCIMENTO)) BETWEEN 181 AND 360
          THEN NVL(I.VL_DUPLICATA,0)    - NVL(R.VL_SOMA_RECEBIDO,0)
          ELSE 0
        END AS "P181_360" ,
        CASE
          WHEN ABS(TRUNC(:CAMPO_DT_BASE) - TRUNC(I.DT_VENCIMENTO)) > 360
          THEN NVL(I.VL_DUPLICATA,0)    - NVL(R.VL_SOMA_RECEBIDO,0)
          ELSE 0
        END AS "P361"
         FROM
        (SELECT RECEBIMENTO.CD_ITCON_REC CD_ITCON_REC,
          SUM (NVL(RECEBIMENTO.VL_RECEBIDO,0)) VL_SOMA_RECEBIDO
           FROM
          (SELECT RECCON_REC.CD_ITCON_REC,
            NVL(RECCON_REC.VL_RECEBIDO,0) + NVL(RECCON_REC.VL_DESCONTO,0) - NVL(RECCON_REC.VL_ACRESCIMO,0) + NVL(RECCON_REC.VL_IMPOSTO_RETIDO,0) VL_RECEBIDO
             FROM DBAMV.RECCON_REC
            WHERE 1=1
            and decode(:CAMPO_SN_DATA_CRED_CONV,'S',TRUNC (nvl(DBAMV.PACK_FNCP.fnc_Retorna_Dt_Cred_Convenio(RECCON_REC.cd_reccon_rec), RECCON_REC.DT_RECEBIMENTO)),
            TRUNC(RECCON_REC.DT_RECEBIMENTO)) <= :CAMPO_DT_BASE
          AND ( ( RECCON_REC.DT_ESTORNO            IS NULL )
          OR ( ( RECCON_REC.DT_ESTORNO             IS NOT NULL )
          AND (TRUNC(RECCON_REC.DT_ESTORNO)         > :CAMPO_DT_BASE) ))
          ) RECEBIMENTO
       GROUP BY RECEBIMENTO.CD_ITCON_REC
        ) R                              ,
        (SELECT CR.CD_CON_REC            ,
          CR.DT_EMISSAO                  ,
          CR.DT_PREVISTA_PAGAMENTO       ,
          CR.TP_QUITACAO                 ,
          CR.TP_CON_REC                  ,
          CR.DS_CON_REC                  ,
          CR.VL_RECEBIDO                 ,
          CR.CD_FIN_CAR                  ,
          CR.CD_REMESSA                  ,
          CR.VL_IMPOSTO                  ,
          CR.VL_DESCONTO                 ,
          CR.VL_ACRESCIMO                ,
          CR.CD_FORNECEDOR               ,
          CR.CD_REDUZIDO                 ,
          CR.CD_HISTORICO_PADRAO         ,
          CR.CD_MULTI_EMPRESA            ,
          CR.CD_TIP_DOC                  ,
          CR.NR_DOCUMENTO                ,
          CR.NR_SERIE                    ,
          CR.TP_RECEBIMENTO              ,
          CR.NM_CLIENTE                  ,
          CR.CD_EXP_CONTABILIDADE        ,
          CR.CD_REG_FAT                  ,
          CR.CD_REG_AMB                  ,
          CR.CD_ATENDIMENTO              ,
          CR.CD_BANCO                    ,
          CR.CD_CAIXA                    ,
          CR.CD_MOT_SITUACAO             ,
          CR.VL_GLOSA                    ,
          CR.SN_CONTABILIZA              ,
          CR.CD_CON_REC_ESTORNO          ,
          CR.VL_ESTORNO                  ,
          CR.CD_PREVISAO                 ,
          CR.DT_CON_REC_ALT_DADOS_ORA_APP,
          CR.CD_USUARIO                  ,
          CR.DT_LANCAMENTO               ,
          CR.TP_VENCIMENTO               ,
          CR.CD_DESCONTO                 ,
          CR.CD_ACRESCIMO                ,
          CR.SN_AGRUPAMENTO              ,
          CR.DS_OBSERVACAO               ,
          CR.VL_TROCO                    ,
          CR.SN_CONVENIO_TERC            ,
          CR.VL_MOEDA                    ,
          CR.VL_MOEDA_DESCONTO           ,
          CR.VL_MOEDA_ACRESCIMO          ,
          CR.CD_MOEDA                    ,
          CR.CD_NOTA_FISCAL              ,
          CR.VL_DEVOLVIDO                ,
          CR.CD_CON_REC_INTEGRA          ,
          CR.DT_INTEGRA                  ,
          CR.CD_SEQ_INTEGRA              ,
          CR.CD_AGENCIA                  ,
          CR.NR_CONTA                    ,
          CR.CD_PROCESSO                 ,
          CR.DT_CANCELAMENTO             ,
          CR.CD_MOTIVO_CANC              ,
          CR.CD_PROCESSO_SEC             ,
          CR.DS_CANCELAMENTO             ,
          CR.CD_EXP_CONTABILIDADE_CANC   ,
          CR.CD_LOG_EXCL_ATENDIME        ,
          CR.SN_GLOSA_ACEITA             ,
          CR.CD_MULTI_EMPRESA_ORIGEM     ,
          CR.CD_CON_REC_RENEGOCIADO      ,
          CR.CD_CONTRATO_ADIANT          ,
          NVL(
          (SELECT SUM(C.VL_PREVISTO) + SUM(NVL(C.VL_ACRESCIMO,0)) - SUM(NVL(C.VL_DESCONTO,0))
             FROM DBAMV.CON_REC C,
            DBAMV.ITCON_REC IT
            WHERE C.CD_CON_REC     = IT.CD_CON_REC
          AND IT.CD_CON_REC_AGRUP  = CR.CD_CON_REC
          AND TRUNC(C.DT_EMISSAO) <= TRUNC(:CAMPO_DT_BASE)
          ),0) VL_PREVISTO
           FROM DBAMV.CON_REC CR
          WHERE CR.CD_PROCESSO IN
          (SELECT CD_PROCESSO
             FROM DBAMV.PROCESSO
            WHERE CD_ESTRUTURAL       IN ( '1.4.1.2.1','1.4.1.1.8','1.4.1.1.9','1.4.1.1.10' ,'1.4.1.1.11','1.4.1.1.12','1.4.1.1.13','1.4.1.1.14' )
          )
        ) C              ,
        DBAMV.ITCON_REC I,
        DBAMV.FORNECEDOR F,
        DBAMV.MULTI_EMPRESAS M
        WHERE C.CD_CON_REC     = I.CD_CON_REC
      AND C.CD_FORNECEDOR      = F.CD_FORNECEDOR
      AND C.CD_MULTI_EMPRESA   = M.CD_MULTI_EMPRESA
      AND C.CD_PREVISAO       IS NULL
      AND TRUNC(C.DT_EMISSAO) <= :CAMPO_DT_BASE
      AND R.CD_ITCON_REC (+)   = I.CD_ITCON_REC
     GROUP BY
     M.CD_MULTI_EMPRESA      ,
        M.DS_MULTI_EMPRESA      ,
     C.CD_FORNECEDOR                             ,
        F.NM_FANTASIA                                     ,
        I.DT_VENCIMENTO                                   ,
        NVL(I.VL_DUPLICATA,0) - NVL(R.VL_SOMA_RECEBIDO,0) ,
        C.CD_CON_REC                                      ,
        I.CD_ITCON_REC                                    ,
        DECODE ('','S',NULL, C.NR_DOCUMENTO)
      )
   GROUP BY
      CD_MULTI_EMPRESA,
      DS_MULTI_EMPRESA,
      CD_FONTE_PAGADORA,
      NM_FANTASIA            ,
      NR_DOCUMENTO           ,
      CD_CON_REC
    )
 GROUP BY
    CD_MULTI_EMPRESA,
    DS_MULTI_EMPRESA,
    CD_FONTE_PAGADORA,
    NM_FANTASIA            ,
    CD_CON_REC,
    NR_DOCUMENTO
  HAVING SUM(VL_A_VENCER) > 0
  OR SUM(VL_VENCIDO)      >0 :CAMPO_CF_ORDER_BY
  )
ORDER BY
  CD_MULTI_EMPRESA,
  DS_MULTI_EMPRESA,
  NOME_FANTASIA,
  CD_FONTE_PAGADORA   ,
  NR_DOCUMENTO

--R_POSICAO_CONTAS_ARECEB_COM_RECURSOS_NOTAS_FISCAIS
SELECT CD_MULTI_EMPRESA,
       DS_MULTI_EMPRESA,
       CD_FONTE_PAGADORA,
       NOME_FANTASIA,
       CD_CON_REC,
       NR_DOCUMENTO,
       CD_NOTA_FISCAL,
       CD_REMESSA_GLOSA,
       VL_TOTAL_NOTA,
       CD_CON_REC_AGRUP,
       Sum(VL_A_VENCER) VL_A_VENCER,
       Sum(VL_VENCIDO)  VL_VENCIDO,
       Sum(TOTAL_A_RECEBER) TOTAL_A_RECEBER,
       Sum(VENCIDO_01_30)   VENCIDO_01_30,
       Sum(VENCIDO_31_60)   VENCIDO_31_60,
       Sum(VENCIDO_61_90)   VENCIDO_61_90,
       Sum(VENCIDO_91_180)  VENCIDO_91_180,
       Sum(VENCIDO_181_360) VENCIDO_181_360,
       Sum(VENCIDO_361)    VENCIDO_361,
       Sum(TOTAL_VENCIDO)  TOTAL_VENCIDO,
       Sum(A_VENCER_01_30) A_VENCER_01_30,
       Sum(A_VENCER_31_60) A_VENCER_31_60,
       Sum(A_VENCER_61_90) A_VENCER_61_90,
       Sum(A_VENCER_91_180) A_VENCER_91_180,
       Sum(A_VENCER_181_360) A_VENCER_181_360,
       Sum(A_VENCER_361) A_VENCER_361,
       Sum(TOTAL_A_VENCER) TOTAL_A_VENCER
  FROM (SELECT CD_MULTI_EMPRESA,
       DS_MULTI_EMPRESA,
       CD_FONTE_PAGADORA,
       NOME_FANTASIA,
       CD_CON_REC,
       NR_DOCUMENTO,
       CD_NOTA_FISCAL,
       CD_REMESSA_GLOSA,
       VL_TOTAL_NOTA,
       CD_CON_REC_AGRUP,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM(VL_A_VENCER)
        ELSE 0
      END VL_A_VENCER,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM(VL_VENCIDO)
        ELSE 0
      END VL_VENCIDO,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN (SUM(VL_A_VENCER) + SUM(VL_VENCIDO))
        ELSE 0
      END TOTAL_A_RECEBER,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P01_30 * DECODE(STATUS,'VENCIDO',1,0)) )
        ELSE 0
      END VENCIDO_01_30,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P31_60 * DECODE(STATUS,'VENCIDO',1,0)) )
        ELSE 0
      END VENCIDO_31_60,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P61_90 * DECODE(STATUS,'VENCIDO',1,0)) )
        ELSE 0
      END VENCIDO_61_90,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P91_180 * DECODE(STATUS,'VENCIDO',1,0)) )
        ELSE 0
      END VENCIDO_91_180,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P181_360 * DECODE(STATUS,'VENCIDO',1,0)) )
        ELSE 0
      END VENCIDO_181_360,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P361 * DECODE(STATUS,'VENCIDO',1,0)) )
        ELSE 0
      END VENCIDO_361,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P01_30 * DECODE(STATUS,'VENCIDO',1,0))) + SUM( (P31_60 * DECODE(STATUS,'VENCIDO',1,0)) ) + SUM( (P61_90 * DECODE(STATUS,'VENCIDO',1,0))) + SUM( (P91_180 * DECODE(STATUS,'VENCIDO',1,0)) ) + SUM( (P181_360 * DECODE(STATUS,'VENCIDO',1,0)))+ SUM( (P361 * DECODE(STATUS,'VENCIDO',1,0)) )
        ELSE 0
      END TOTAL_VENCIDO,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P01_30 * DECODE(STATUS,'VENCIDO',0,1)) )
        ELSE 0
      END A_VENCER_01_30,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P31_60 * DECODE(STATUS,'VENCIDO',0,1)) )
        ELSE 0
      END A_VENCER_31_60,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P61_90 * DECODE(STATUS,'VENCIDO',0,1)) )
        ELSE 0
      END A_VENCER_61_90,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P91_180 * DECODE(STATUS,'VENCIDO',0,1)) )
        ELSE 0
      END A_VENCER_91_180,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P181_360 * DECODE(STATUS,'VENCIDO',0,1)) )
        ELSE 0
      END A_VENCER_181_360,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P361 * DECODE(STATUS,'VENCIDO',0,1)) )
        ELSE 0
      END A_VENCER_361,
      CASE
        WHEN CD_CON_REC_AGRUP IS NULL
        THEN SUM( (P01_30 * DECODE(STATUS,'VENCIDO',0,1)) ) + SUM( (P31_60 * DECODE(STATUS,'VENCIDO',0,1)) ) + SUM( (P61_90 * DECODE(STATUS,'VENCIDO',0,1)) ) + SUM( (P91_180 * DECODE(STATUS,'VENCIDO',0,1)) ) + SUM( (P181_360 * DECODE(STATUS,'VENCIDO',0,1)))+ SUM( (P361 * DECODE(STATUS,'VENCIDO',0,1)) )
        ELSE 0
      END TOTAL_A_VENCER
       FROM
          (SELECT  DISTINCT
                M.CD_MULTI_EMPRESA                                        CD_MULTI_EMPRESA,
                M.DS_MULTI_EMPRESA                                        DS_MULTI_EMPRESA,
                C.cd_fornecedor                                              CD_FONTE_PAGADORA,
                C.NR_DOCUMENTO                                          NR_DOCUMENTO,
                F.NM_FANTASIA                                               NOME_FANTASIA,
                C.CD_CON_REC                                               CD_CON_REC,
                G.CD_REMESSA_GLOSA,
                N.NR_ID_NOTA_FISCAL		    CD_NOTA_FISCAL,
                N.VL_TOTAL_NOTA,
                I.CD_CON_REC_AGRUP,
                CASE
                   WHEN TRUNC((TRUNC(:CAMPO_DT_BASE)-TRUNC(G.DT_PREVISTA_PARA_PGTO))) > 0
                    THEN 'VENCIDO'
                     ELSE 'A_VENCER'
                END AS "STATUS" ,
                       CASE
          WHEN TRUNC((TRUNC(:CAMPO_DT_BASE)-TRUNC(G.DT_PREVISTA_PARA_PGTO))) > 0
          THEN 0
          ELSE NVL(G.VL_TOTAL_REMESSA,0) - NVL(G.VL_RECURSO_RECEBIDO,0)
        END AS "VL_A_VENCER" ,
        CASE
          WHEN TRUNC((TRUNC(:CAMPO_DT_BASE)-TRUNC(G.DT_PREVISTA_PARA_PGTO))) > 0
          THEN NVL(G.VL_TOTAL_REMESSA,0) - NVL(G.VL_RECURSO_RECEBIDO,0)
          ELSE 0
        END AS "VL_VENCIDO" ,
        CASE
          WHEN ABS(TRUNC(:CAMPO_DT_BASE) - TRUNC(G.DT_PREVISTA_PARA_PGTO)) BETWEEN 0 AND 30
          THEN NVL(G.VL_TOTAL_REMESSA,0) - NVL(G.VL_RECURSO_RECEBIDO,0)
          ELSE 0
        END AS "P01_30" ,
        CASE
          WHEN ABS(TRUNC(:CAMPO_DT_BASE) - TRUNC(G.DT_PREVISTA_PARA_PGTO)) BETWEEN 31 AND 60
          THEN NVL(G.VL_TOTAL_REMESSA,0) - NVL(G.VL_RECURSO_RECEBIDO,0) 
          ELSE 0
        END AS "P31_60" ,
        CASE
          WHEN ABS(TRUNC(:CAMPO_DT_BASE) - TRUNC(G.DT_PREVISTA_PARA_PGTO)) BETWEEN 61 AND 90
          THEN NVL(G.VL_TOTAL_REMESSA,0) - NVL(G.VL_RECURSO_RECEBIDO,0)
          ELSE 0
        END AS "P61_90" ,
        CASE
          WHEN ABS(TRUNC(:CAMPO_DT_BASE) - TRUNC(G.DT_PREVISTA_PARA_PGTO)) BETWEEN 91 AND 180
          THEN NVL(G.VL_TOTAL_REMESSA,0) - NVL(G.VL_RECURSO_RECEBIDO,0)
          ELSE 0
        END AS "P91_180" ,
        CASE
          WHEN ABS(TRUNC(:CAMPO_DT_BASE) - TRUNC(G.DT_PREVISTA_PARA_PGTO)) BETWEEN 181 AND 360
          THEN NVL(G.VL_TOTAL_REMESSA,0) - NVL(G.VL_RECURSO_RECEBIDO,0)
          ELSE 0
        END AS "P181_360" ,
        CASE
          WHEN ABS(TRUNC(:CAMPO_DT_BASE) - TRUNC(G.DT_PREVISTA_PARA_PGTO)) > 360
          THEN NVL(G.VL_TOTAL_REMESSA,0) - NVL(G.VL_RECURSO_RECEBIDO,0)
          ELSE 0
        END AS "P361"      
           FROM DBAMV.CON_REC C
               ,DBAMV.ITCON_REC I
               ,DBAMV.MULTI_EMPRESAS M
               ,DBAMV.NOTA_FISCAL N
               ,DBAMV.ITFAT_NOTA_FISCAL T
               ,DBAMV.FORNECEDOR F
               ,DBAMV.REMESSA_GLOSA G
               ,DBAMV.MOV_GLOSAS MG
               ,DBAMV.PROCESSO P
               ,(SELECT CR.CD_CON_REC,
          CR.DT_LANCAMENTO
           FROM DBAMV.CON_REC CR
          WHERE CR.CD_PROCESSO IN
          (SELECT CD_PROCESSO
             FROM DBAMV.PROCESSO
            WHERE CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
          AND CD_ESTRUTURAL       IN ( '1.4.1.2.1','1.4.1.1.8','1.4.1.1.9','1.4.1.1.10' ,'1.4.1.1.11','1.4.1.1.12','1.4.1.1.13','1.4.1.1.14' )
          )
        AND TRUNC(CR.DT_EMISSAO) <= TRUNC(:CAMPO_DT_BASE)
        ) AGRUPAMENTO
           WHERE I.CD_CON_REC = C.CD_CON_REC
            AND M.CD_MULTI_EMPRESA = C.CD_MULTI_EMPRESA
            AND N.CD_MULTI_EMPRESA = C.CD_MULTI_EMPRESA
            AND T.CD_NOTA_FISCAL = N.CD_NOTA_FISCAL
            AND N.CD_NOTA_FISCAL = C.CD_NOTA_FISCAL
            AND F.CD_FORNECEDOR = C.CD_FORNECEDOR
            AND G.CD_REMESSA = T.CD_REMESSA
            AND G.CD_MULTI_EMPRESA = C.CD_MULTI_EMPRESA
            AND G.CD_REMESSA_GLOSA IS NOT NULL
            AND MG.CD_REMESSA_GLOSA = G.CD_REMESSA_GLOSA
            AND C.CD_CON_REC = G.CD_CON_REC
            AND P.CD_PROCESSO = MG.CD_PROCESSO
            AND P.CD_MULTI_EMPRESA = C.CD_MULTI_EMPRESA
            AND C.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
            AND C.TP_CON_REC                      = 'C'
            AND C.CD_PREVISAO                     is null
            AND C.DT_CANCELAMENTO                 is NULL
            AND I.CD_CON_REC_AGRUP   = AGRUPAMENTO.CD_CON_REC(+)         
				:CAMPO_CF_MULTI_EMPRESA
        )GROUP BY CD_MULTI_EMPRESA,
         DS_MULTI_EMPRESA,
         CD_FONTE_PAGADORA,
         NOME_FANTASIA,
         CD_REMESSA_GLOSA,
         CD_NOTA_FISCAL,
         NR_DOCUMENTO,
         CD_CON_REC,
         VL_TOTAL_NOTA,
         CD_CON_REC_AGRUP
)
WHERE    TOTAL_A_RECEBER > 0
GROUP BY CD_MULTI_EMPRESA,
         DS_MULTI_EMPRESA,
         CD_FONTE_PAGADORA,
         NOME_FANTASIA,
         CD_REMESSA_GLOSA,
         CD_NOTA_FISCAL,
         NR_DOCUMENTO,
         CD_CON_REC,
         VL_TOTAL_NOTA,
         CD_CON_REC_AGRUP
ORDER BY CD_MULTI_EMPRESA,
         DS_MULTI_EMPRESA,
         NOME_FANTASIA,
         CD_FONTE_PAGADORA,
         CD_NOTA_FISCAL,
         CD_REMESSA_GLOSA
