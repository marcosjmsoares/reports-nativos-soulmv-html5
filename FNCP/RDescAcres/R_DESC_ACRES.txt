--R_DESC_ACRES_R_G_TIPO_PG_RE2_Q_CUSTO
SELECT * FROM (
Select 'PA'                                     Tipo_Pg_Re
      ,RATPAG_DESC_ACRES.CD_ITEM_RES            CCusto
      ,PAG_DESC_ACRES.CD_DESC_ACRES             Cd_DC
      ,PAG_DESC_ACRES.TP_DESC_ACRES             Tipo_DC
      ,DESCONTOS_ACRESCIMOS.DS_DESC_ACRES       Desc_Acres
      ,RATPAG_DESC_ACRES.VL_RATPAG_DESC_ACRES   Valor
From   DBAMV.PAGCON_PAG
      ,DBAMV.PAG_DESC_ACRES
      ,DBAMV.RATPAG_DESC_ACRES
      ,DBAMV.DESCONTOS_ACRESCIMOS
Where  PAG_DESC_ACRES.CD_PAGCON_PAG = PAGCON_PAG.CD_PAGCON_PAG
  And  RATPAG_DESC_ACRES.CD_PAG_DESC_ACRES = PAG_DESC_ACRES.CD_PAG_DESC_ACRES
  And  DESCONTOS_ACRESCIMOS.CD_DESC_ACRES = PAG_DESC_ACRES.CD_DESC_ACRES
  And  TRUNC( PAGCON_PAG.DT_PAGAMENTO ) >= :CAMPO_DT_INICIAL
  And  TRUNC( PAGCON_PAG.DT_PAGAMENTO ) <= :CAMPO_DT_FINAL
  And  :CAMPO_TP_MOVIMENTACAO in ('PA','AM')
Union All
Select 'RE'                                     Tipo_Pg_Re
      ,RATREC_DESC_ACRES.CD_ITEM_RES            CCusto
      ,REC_DESC_ACRES.CD_DESC_ACRES             Cd_DC
      ,REC_DESC_ACRES.TP_DESC_ACRES             Tipo_DC
      ,DESCONTOS_ACRESCIMOS.DS_DESC_ACRES       Desc_Acres
      ,RATREC_DESC_ACRES.VL_RATREC_DESC_ACRES   Valor
From   DBAMV.RECCON_REC
      ,DBAMV.REC_DESC_ACRES
      ,DBAMV.RATREC_DESC_ACRES
      ,DBAMV.DESCONTOS_ACRESCIMOS
Where  REC_DESC_ACRES.CD_RECCON_REC = RECCON_REC.CD_RECCON_REC
  And  RATREC_DESC_ACRES.CD_REC_DESC_ACRES = REC_DESC_ACRES.CD_REC_DESC_ACRES
  And  DESCONTOS_ACRESCIMOS.CD_DESC_ACRES = REC_DESC_ACRES.CD_DESC_ACRES
  And  TRUNC( RECCON_REC.DT_RECEBIMENTO ) >= :CAMPO_DT_INICIAL
  And  TRUNC( RECCON_REC.DT_RECEBIMENTO ) <= :CAMPO_DT_FINAL
  And  :CAMPO_TP_MOVIMENTACAO in ('RE','AM')
Order by 1,2,4,3
)
ORDER BY TIPO_PG_RE,
CCusto
