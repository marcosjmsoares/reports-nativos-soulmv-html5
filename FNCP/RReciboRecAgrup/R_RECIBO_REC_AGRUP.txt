--R_RECIBO_REC_AGRUP
SELECT NR_RECIBO,
       VL_RECEBIDO,
       NM_RESPONSAVEL,
       DBAMV.PACK_FNCP.EXTENSOS(VL_RECEBIDO) EXTENSO,
       VL_RECEBIDO - VL_ACRESCIMO + VL_DESCONTO VL_TITULO,
       VL_DESCONTO,
       VL_ACRESCIMO
FROM (SELECT Max(RCR.CD_AGRUP_IMPRESSAO_RECIBO) NR_RECIBO,
             NVL(SUM(NVL(RCR.VL_RECEBIDO,0)),0) VL_RECEBIDO,
             Nvl(MIN(R.NM_RESPONSAVEL),MIN(C.NM_CLIENTE)) NM_RESPONSAVEL,
             NVL(SUM(NVL(RCR.VL_DESCONTO,0)),0) VL_DESCONTO,
             NVL(SUM(NVL(RCR.VL_ACRESCIMO,0)),0) VL_ACRESCIMO
         FROM DBAMV.RECCON_REC RCR,
             DBAMV.ITCON_REC I,
             DBAMV.CON_REC C,
             DBAMV.RESPONSAVEL R
       WHERE RCR.CD_ITCON_REC = I.CD_ITCON_REC
         AND I.CD_CON_REC = C.CD_CON_REC
         AND RCR.CD_RESPOSAVEL = R.CD_RESPOSAVEL (+)
         AND RCR.CD_AGRUP_IMPRESSAO_RECIBO IN (SELECT CD_AGRUP_IMPRESSAO_RECIBO
                                             FROM DBAMV.RECCON_REC
                                            WHERE CD_RECCON_REC = :CAMPO_CD_RECCON_REC))
