--R_VISAO
SELECT * FROM (
select nvl(psv_06.cd_plano_sintetico,
           nvl(psv_05.cd_plano_sintetico,
               nvl(psv_04.cd_plano_sintetico,
                   nvl(psv_03.cd_plano_sintetico,
                       nvl(psv_02.cd_plano_sintetico,
                           psv_01.cd_plano_sintetico)))))   cd_plano_sintetico
      ,psv_01.cd_visao    cd_visao
      ,visao.ds_visao        ds_visao
      ,psv_01.cd_conta    cd_conta_01
      ,psv_01.ds_conta    ds_conta_01
      ,psv_02.cd_conta    cd_conta_02
      ,psv_02.ds_conta    ds_conta_02
      ,psv_03.cd_conta    cd_conta_03
      ,psv_03.ds_conta    ds_conta_03
      ,psv_04.cd_conta    cd_conta_04
      ,psv_04.ds_conta    ds_conta_04
      ,psv_05.cd_conta    cd_conta_05
      ,psv_05.ds_conta    ds_conta_05
      ,psv_06.cd_conta    cd_conta_06
      ,psv_06.ds_conta    ds_conta_06
from   dbamv.plano_sintetico_visao    psv_01
      ,dbamv.plano_sintetico_visao    psv_02
      ,dbamv.plano_sintetico_visao    psv_03
      ,dbamv.plano_sintetico_visao    psv_04
      ,dbamv.plano_sintetico_visao    psv_05
      ,dbamv.plano_sintetico_visao    psv_06
      ,dbamv.visao                              visao
where  psv_06.cd_pai(+)   = psv_05.cd_plano_sintetico
and    psv_05.cd_pai(+)   = psv_04.cd_plano_sintetico
and    psv_04.cd_pai(+)   = psv_03.cd_plano_sintetico
and    psv_03.cd_pai(+)   = psv_02.cd_plano_sintetico
and    psv_02.cd_pai(+)   = psv_01.cd_plano_sintetico
and    psv_01.cd_pai is null
and    visao.cd_visao = psv_01.cd_visao
and    visao.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
ORDER BY 1,2,4,6,8,10,12
)
ORDER BY CD_VISAO,DS_VISAO,
CD_CONTA_01,DS_CONTA_01,
CD_CONTA_02,DS_CONTA_02,
CD_CONTA_03,DS_CONTA_03,
CD_CONTA_04,DS_CONTA_04,
CD_CONTA_05,DS_CONTA_05,
CD_CONTA_06,DS_CONTA_06
