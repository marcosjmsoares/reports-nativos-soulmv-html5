--R_CON_ARECEB_POSICAO_GLO
SELECT * 
FROM 
    (SELECT tp_conta,
       nm_cliente,
       cd_fornecedor,
       nr_documento,
       nr_ncf,
       cd_con_rec,
       cd_reduzido,
       dt_emissao,
       dt_vencimento,
       vl_previsto,
       vl_glosa,
       vl_glosa_aceite,
       vl_recurso,
       vl_recurso_recebido,
       vl_recebido_total,
       vl_a_receber,
       vl_saldo_glosa,
       vl_recebido_glosa,
       vl_a_receber_dif,
       cd_multi_empresa,
       ds_multi_empresa,
       ds_conta_contabil
FROM 
(SELECT        tp_conta,
               nm_cliente,
               cd_fornecedor,
               nr_documento,
               nr_ncf,
               cd_con_rec,
               cd_reduzido,
               dt_emissao,
               dt_vencimento,
               vl_previsto,
               vl_glosa,
               vl_glosa_aceite,
               vl_recurso,
               vl_recurso_recebido,
               (Nvl(vl_recebido_total,0) - Nvl(vl_recurso_recebido,0))                                                vl_recebido_total,
               (Nvl (vl_previsto, 0) - Nvl (vl_recebido_total, 0) - (Nvl (vl_glosa,0) - Nvl(vl_recurso_recebido,0))) vl_a_receber,
               (Nvl(vl_glosa,0) - Nvl(vl_glosa_aceite,0) - Nvl(vl_recurso_recebido,0))                                vl_saldo_glosa,
			   (Nvl(vl_recebido_total,0) - Nvl(vl_recurso_recebido,0) + Nvl(vl_glosa,0))                              vl_recebido_glosa,
			   (Nvl (vl_previsto, 0) - (Nvl(vl_recebido_total,0) - Nvl(vl_recurso_recebido,0) + Nvl(vl_glosa,0))) vl_a_receber_dif,
			cd_multi_empresa,
	       ds_multi_empresa,
	       cd_reduzido || ' - ' || cd_contabil || ' | ' || ds_conta ds_conta_contabil
      FROM    (SELECT con_rec.tp_con_rec            tp_conta,
                      con_rec.nm_cliente            nm_cliente,
                      con_rec.cd_fornecedor,
                      con_rec.nr_documento          nr_documento,
                      con_rec.nr_ncf          nr_ncf,
                      con_rec.cd_multi_empresa      cd_multi_empresa,
					  multi_empresas.ds_multi_empresa,
                      plano_contas.cd_contabil,
                      plano_contas.ds_conta,
                      con_rec.cd_con_rec            cd_con_rec,
                      con_rec.cd_reduzido           cd_reduzido,
                      itcon_rec.vl_duplicata        vl_previsto,
					  itcon_rec.tp_quitacao         tp_quitacao,
                      itcon_rec.dt_vencimento       dt_vencimento,
                      itcon_rec.dt_cancelamento     dt_cancelamento,
                      Nvl(reccon_rec.vl_recebido,0) vl_recebido_total,
                      Nvl(reccon_rec.vl_glosa,0)    vl_glosa,
                      Nvl(reccon_rec.vl_recurso,0)  vl_recurso_recebido ,
                      Nvl(itcon_rec.vl_recurso,0)   vl_recurso,
                      Nvl(itcon_rec.vl_aceite,0)    vl_glosa_aceite,
					  Nvl(itcon_rec.vl_recurso_it,0)   vl_recurso_it,
                      con_rec.dt_emissao            dt_emissao
               FROM (SELECT   recebimento.cd_itcon_rec,
                              Sum(Nvl(recebimento.vl_recebido,0))                              vl_recebido,
                              Sum(Nvl(recebimento.vl_glosa ,0))                                vl_glosa,
                              Sum(Decode(Nvl(recebimento.cd_remessa_glosa,0),0,0,vl_recebido)) vl_recurso
                     FROM    (SELECT reccon_rec.cd_itcon_rec,
                                     Nvl(reccon_rec.vl_recebido,0)  +
                                     Nvl(reccon_rec.vl_desconto,0)  -
                                     Nvl(reccon_rec.vl_acrescimo,0) +
                                     Nvl(reccon_rec.vl_imposto_retido,0)   vl_recebido,
                                     reccon_rec.vl_recebido                vl_liq_recebido,
                                     reccon_rec.vl_acrescimo               vl_acrescimo,
                                     reccon_rec.vl_desconto                vl_desconto,
                                     Sum(Nvl(it_recebimento.vl_glosa,0))   vl_glosa,
                                     reccon_rec.cd_remessa_glosa           cd_remessa_glosa
                              FROM   dbamv.reccon_rec,
                                     dbamv.it_recebimento
                              WHERE  reccon_rec.cd_reccon_rec  = it_recebimento.cd_reccon_rec (+)
                                AND (reccon_rec.dt_estorno IS NULL or reccon_rec.dt_estorno >  :CAMPO_DATA_BASE )
                              GROUP BY  reccon_rec.cd_itcon_rec,
                                        reccon_rec.dt_recebimento,
                                        reccon_rec.vl_recebido,
                                        reccon_rec.vl_desconto,
                                        reccon_rec.vl_acrescimo,
                                        reccon_rec.vl_imposto_retido,
                                        reccon_rec.cd_reccon_rec,
                                        reccon_rec.cd_remessa_glosa) recebimento
                     GROUP BY recebimento.cd_itcon_rec) reccon_rec,
              (SELECT it.cd_itcon_rec,
                      it.cd_con_rec,
                      it.dt_vencimento,
					  it.tp_quitacao,
                      it.cd_con_rec_agrup,
                      it.vl_duplicata,
                      it.dt_cancelamento,
	              it.vl_recurso vl_recurso_it,
                      a.vl_recurso,
                      a.vl_glosa,
                      a.vl_aceite,
                      it.dt_prevista_recebimento
               FROM  (SELECT cd_itcon_rec,
                             Sum(vl_recurso) vl_recurso,
                             Sum(vl_glosa) vl_glosa,
                             Sum(vl_aceite) vl_aceite
                      FROM (SELECT mg.cd_itcon_rec,
                                   Sum(mg.vl_movimentacao) vl_recurso,
                                   0 vl_glosa,
                                   0 vl_aceite
                            FROM   dbamv.mov_glosas mg,
                                   dbamv.processo pr
                            WHERE  mg.cd_remessa_glosa in (SELECT rg.cd_remessa_glosa
                                                           FROM   dbamv.remessa_glosa rg
                                                           WHERE  rg.cd_remessa_glosa_pai IS NULL)
                              AND  mg.cd_processo = pr.cd_processo
                              AND  pr.cd_estrutural = '3.1.2.1.1'
                              AND  Trunc (mg.dt_movimentacao) <= :CAMPO_DATA_BASE
                            GROUP BY mg.cd_itcon_rec
                            UNION ALL
                            SELECT mg.cd_itcon_rec,
                                   0 vl_recurso,
                                   0 vl_glosa,
                                   Sum(mg.vl_movimentacao) vl_aceite
                            FROM   dbamv.mov_glosas mg,
                                   dbamv.processo pr
                            WHERE  mg.cd_remessa_glosa IS NULL
                              AND  mg.cd_processo = pr.cd_processo
                              AND  pr.cd_estrutural = '3.1.2.1.2'
                              AND  Trunc (mg.dt_movimentacao) <= :CAMPO_DATA_BASE
                            GROUP BY mg.cd_itcon_rec
                            UNION ALL
                            SELECT mg.cd_itcon_rec,
                                   0 vl_recurso ,
                                   (Sum(mg.vl_movimentacao)) vl_glosa,
                                   0 vl_aceite
                            FROM   dbamv.mov_glosas mg,
                                   dbamv.processo pr
                            WHERE  mg.cd_remessa_glosa IS NULL
                              AND  mg.cd_processo = pr.cd_processo
                              AND  pr.cd_estrutural NOT IN( '3.1.2.1.1','3.1.2.1.2','3.1.2.1.3')
                              AND  Trunc (mg.dt_movimentacao) <= :CAMPO_DATA_BASE
                            GROUP BY mg.cd_itcon_rec)
                      GROUP BY cd_itcon_rec) a,
                      dbamv.itcon_rec it
               WHERE it.cd_itcon_rec = a.cd_itcon_rec(+)) itcon_rec,
               dbamv.con_rec, dbamv.multi_empresas, dbamv.plano_contas
      WHERE con_rec.cd_previsao IS NULL
        AND con_rec.cd_multi_empresa = multi_empresas.cd_multi_empresa
        AND con_rec.cd_reduzido(+) = plano_contas.cd_reduzido
        AND itcon_rec.cd_con_rec = con_rec.cd_con_rec
        AND reccon_rec.cd_itcon_rec(+) = itcon_rec.cd_itcon_rec
        AND itcon_rec.cd_con_rec_agrup IS NULL
            :CAMPO_CF_CD_REDUZIDO))
ORDER BY dt_emissao
)
ORDER BY CD_MULTI_EMPRESA,
CD_REDUZIDO,
TP_CONTA,
CD_FORNECEDOR,NM_CLIENTE
