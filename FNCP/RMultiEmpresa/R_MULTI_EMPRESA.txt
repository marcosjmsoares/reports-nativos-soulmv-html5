--R_MULTI_EMPRESA
SELECT  multi.cd_multi_empresa codigo
       ,ds_multi_empresa
       ,cd_ativo
       ,cd_passivo
       ,cd_receita
       ,cd_despesa
       ,cd_cgc
       ,cd_iest
       ,cd_imun
       ,cd_ijunta
       ,ds_endereco
       ,multi.cd_uf
       ,cd_crc_contador
       ,ds_nome_contador
       ,sn_ativo
       ,sn_utiliza_custos
       ,usu.cd_multi_empresa empresa_usuario
       ,usu.cd_id_usuario
       ,decode(multi.cd_cidade,' ',nm_cidade,nm_cidade) cidade
  FROM  dbamv.multi_empresas multi
       ,dbamv.usuario_multi_empresa usu
       ,dbamv.cidade
  WHERE usu.cd_multi_empresa = multi.cd_multi_empresa
    AND cidade.cd_cidade = multi.cd_cidade
ORDER BY
