       reg_fat.cd_remessa       remessa,
       reg_fat.cd_reg_fat       conta,

       pro_fat.cd_pro_fat,
       pro_fat.ds_pro_fat,
       gru_pro.cd_gru_pro,
       gru_pro.ds_gru_pro,
       glosas.vl_glosa,
       glosas.vl_reapresentado,
       glosas.vl_recebido,
       motivo_glosa.ds_motivo_glosa,
			 to_char(glosas.dt_glosa)
from   dbamv.glosas,
       dbamv.motivo_glosa,
       dbamv.reg_fat,
       dbamv.gru_pro,
       dbamv.prestador,
       dbamv.convenio,
       dbamv.empresa_convenio,
       dbamv.remessa_fatura  remessa,
       dbamv.fatura,
       dbamv.atendime,
       dbamv.pro_fat,
       dbamv.paciente
where  Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and glosas.cd_reg_fat        =   reg_fat.cd_reg_fat
and    glosas.cd_prestador    =  prestador.cd_prestador(+)
and    glosas.cd_pro_fat      =  pro_fat.cd_pro_fat
and    glosas.cd_gru_pro     =  gru_pro.cd_gru_pro(+)
and    motivo_glosa.cd_motivo_glosa = glosas.cd_motivo_glosa
and    reg_fat.cd_atendimento    =  atendime.cd_atendimento
and    reg_fat.cd_convenio       =  convenio.cd_convenio
and    reg_fat.cd_remessa        =  remessa.cd_remessa
and    remessa.cd_fatura         =  fatura.cd_fatura
and    atendime.cd_paciente      =  paciente.cd_paciente
and    reg_fat.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
and    convenio.tp_convenio <> 'H'
and    glosas.cd_gru_pro is null
union all
select prestador.cd_prestador,
       decode(prestador.nm_prestador, null, 'Sem Prestador', prestador.nm_prestador) nm_prestador,
       convenio.nm_convenio,
       reg_amb.cd_remessa       remessa,
       reg_amb.cd_reg_amb       conta,
       paciente.nm_paciente,
       pro_fat.cd_pro_fat,
       pro_fat.ds_pro_fat,
       gru_pro.cd_gru_pro,
       gru_pro.ds_gru_pro,
       glosas.vl_glosa,
       glosas.vl_reapresentado,
       glosas.vl_recebido,
       motivo_glosa.ds_motivo_glosa,
			 to_char(glosas.dt_glosa)
from   dbamv.glosas,
       dbamv.motivo_glosa,
       dbamv.reg_amb,
       dbamv.prestador,
       dbamv.convenio,
       dbamv.empresa_convenio,
       dbamv.remessa_fatura  remessa,
       dbamv.fatura,
       dbamv.atendime,
       dbamv.pro_fat,
       dbamv.gru_pro,
       dbamv.paciente
where  Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and glosas.cd_reg_amb  =  reg_amb.cd_reg_amb
and   glosas.cd_atendimento  =  atendime.cd_atendimento
and   reg_amb.cd_convenio     =  convenio.cd_convenio
and   glosas.cd_pro_fat      =  pro_fat.cd_pro_fat
and    glosas.cd_gru_pro     =  gru_pro.cd_gru_pro(+)
and    motivo_glosa.cd_motivo_glosa = glosas.cd_motivo_glosa
and    reg_amb.cd_remessa        =  remessa.cd_remessa
and    remessa.cd_fatura         =  fatura.cd_fatura
and    glosas.cd_prestador      =  prestador.cd_prestador(+)
and    atendime.cd_paciente      =  paciente.cd_paciente
and    reg_amb.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
and    convenio.tp_convenio <> 'H'
and    glosas.cd_gru_pro is null
Union all
select prestador.cd_prestador,
       decode(prestador.nm_prestador, null, 'Sem Prestador', prestador.nm_prestador) nm_prestador,
       convenio.nm_convenio,
       reg_fat.cd_remessa       remessa,
       reg_fat.cd_reg_fat       conta,
       paciente.nm_paciente,
       pro_fat.cd_pro_fat,
       pro_fat.ds_pro_fat,
       gru_pro.cd_gru_pro,
       gru_pro.ds_gru_pro,
       glosas.vl_glosa,
       glosas.vl_reapresentado,
       glosas.vl_recebido,
       motivo_glosa.ds_motivo_glosa,
			 to_char(glosas.dt_glosa)
from   dbamv.glosas,
       dbamv.motivo_glosa,
       dbamv.reg_fat,
       dbamv.gru_pro,
       dbamv.prestador,
       dbamv.convenio,
       dbamv.empresa_convenio,
       dbamv.remessa_fatura  remessa,
       dbamv.fatura,
       dbamv.atendime,
       dbamv.pro_fat,
       dbamv.paciente
where  Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and glosas.cd_reg_fat        =   reg_fat.cd_reg_fat
and    glosas.cd_prestador    =  prestador.cd_prestador(+)
and    glosas.cd_pro_fat      =  pro_fat.cd_pro_fat(+)
and    glosas.cd_gru_pro     =  gru_pro.cd_gru_pro
and    motivo_glosa.cd_motivo_glosa = glosas.cd_motivo_glosa
and    reg_fat.cd_atendimento    =  atendime.cd_atendimento
and    reg_fat.cd_convenio       =  convenio.cd_convenio
and    reg_fat.cd_remessa        =  remessa.cd_remessa
and    remessa.cd_fatura         =  fatura.cd_fatura
and    atendime.cd_paciente      =  paciente.cd_paciente
and    reg_fat.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
and    convenio.tp_convenio <> 'H'
and    glosas.cd_gru_pro is not null
union all
select prestador.cd_prestador,
       decode(prestador.nm_prestador, null, 'Sem Prestador', prestador.nm_prestador) nm_prestador,
       convenio.nm_convenio,
       reg_amb.cd_remessa       remessa,
       reg_amb.cd_reg_amb       conta,
       paciente.nm_paciente,
       pro_fat.cd_pro_fat,
       pro_fat.ds_pro_fat,
       gru_pro.cd_gru_pro,
       gru_pro.ds_gru_pro,
       glosas.vl_glosa,
       glosas.vl_reapresentado,
       glosas.vl_recebido,
       motivo_glosa.ds_motivo_glosa,
			 to_char(glosas.dt_glosa)
from   dbamv.glosas,
       dbamv.motivo_glosa,
       dbamv.reg_amb,
       dbamv.prestador,
       dbamv.convenio,
       dbamv.empresa_convenio,
       dbamv.remessa_fatura  remessa,
       dbamv.fatura,
       dbamv.atendime,
       dbamv.pro_fat,
       dbamv.gru_pro,
       dbamv.paciente
where  Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and glosas.cd_reg_amb  =  reg_amb.cd_reg_amb
and   glosas.cd_atendimento  =  atendime.cd_atendimento
and   reg_amb.cd_convenio     =  convenio.cd_convenio
and   glosas.cd_pro_fat      =  pro_fat.cd_pro_fat(+)
and    glosas.cd_gru_pro     =  gru_pro.cd_gru_pro
and    motivo_glosa.cd_motivo_glosa = glosas.cd_motivo_glosa
and    reg_amb.cd_remessa        =  remessa.cd_remessa
and    remessa.cd_fatura         =  fatura.cd_fatura
and    glosas.cd_prestador      =  prestador.cd_prestador(+)
and    atendime.cd_paciente      =  paciente.cd_paciente
and    reg_amb.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
and    convenio.tp_convenio <> 'H'
and    glosas.cd_gru_pro is not null
union all
select 000 cd_prestador,
       'Sem Prestador - Glosa não identificada' nm_prestador,
       convenio.nm_convenio,
       remessa.cd_remessa       remessa,
       to_number(null)       conta,
       null       nm_paciente,
       null       cd_pro_fat,
       null       ds_pro_fat,
       to_number(null)       cd_gru_pro,
       null       ds_gru_pro,
       remessa_glosa.vl_glosa_nao_identificada vl_glosa,
       0   vl_reapresentado, 
       0          vl_recebido,
       'Glosa não identificada' ds_motivo_glosa,
			 null   dt_glosa
from   dbamv.remessa_glosa,
       dbamv.remessa_fatura  remessa,
       dbamv.fatura,
       dbamv.convenio,
       dbamv.empresa_convenio
where  Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and remessa_glosa.cd_remessa = remessa.cd_remessa
and    remessa_glosa.tp_glosa = 'N'
and    remessa.cd_fatura         =  fatura.cd_fatura
and    fatura.cd_convenio = convenio.cd_convenio
and    fatura.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
and    convenio.tp_convenio <> 'H'
:CAMPO_CF_COND_REMESSA
