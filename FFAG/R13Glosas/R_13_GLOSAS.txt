--R_13_GLOSAS
SELECT * FROM (
select nm_convenio,
       mes,
       sum(vl_glosa)         valor
from (
select convenio.nm_convenio,
       glosas.vl_glosa,
       decode ('' , 'Entrega', to_char(remessa.dt_entrega_da_fatura,'mm'), to_char(fatura.dt_competencia, 'mm')) mes
from   dbamv.glosas,
       dbamv.reg_fat,
       dbamv.convenio,
       dbamv.empresa_convenio,
       dbamv.remessa_fatura  remessa,
       dbamv.fatura,
       dbamv.remessa_glosa
where  Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and reg_fat.cd_remessa    =  remessa.cd_remessa
and    remessa.cd_fatura        =  fatura.cd_fatura
and    reg_fat.cd_reg_fat        =  glosas.cd_reg_fat
and    glosas.cd_remessa_glosa   =  remessa_glosa.cd_remessa_glosa
and    remessa_glosa.cd_convenio =  convenio.cd_convenio
and    convenio.tp_convenio <> 'H'
union all
select convenio.nm_convenio,
       glosas.vl_glosa,
       decode ('' , 'Entrega', to_char(remessa.dt_entrega_da_fatura,'mm'), to_char(fatura.dt_competencia, 'mm')) mes
from   dbamv.glosas,
       dbamv.reg_amb,
       dbamv.convenio,
       dbamv.empresa_convenio,
       dbamv.remessa_fatura  remessa,
       dbamv.fatura,
       dbamv.remessa_glosa
where  Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and reg_amb.cd_remessa        =  remessa.cd_remessa
and    remessa.cd_fatura         =  fatura.cd_fatura
and    reg_amb.cd_reg_amb      =  glosas.cd_reg_amb
and    glosas.cd_remessa_glosa   =  remessa_glosa.cd_remessa_glosa
and    remessa_glosa.cd_convenio =  convenio.cd_convenio
and    reg_amb.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
and    convenio.tp_convenio <> 'H'
Union All
Select convenio.nm_convenio,
           Sum( remessa_glosa.vl_total_remessa ) vl_glosa,
           decode ('' , 'Entrega', to_char(remessa.dt_entrega_da_fatura,'mm'), to_char(fatura.dt_competencia, 'mm')) mes
  From dbamv.convenio,
  dbamv.empresa_convenio,
          dbamv.remessa_fatura remessa,
          dbamv.remessa_glosa,
          dbamv.fatura
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and remessa.cd_remessa = remessa_glosa.cd_remessa
   and remessa.cd_fatura = fatura.cd_fatura
   and fatura.cd_convenio = convenio.cd_convenio
   and convenio.tp_convenio <> 'H'
   and fatura.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
   and remessa_glosa.cd_remessa_glosa not in ( Select distinct cd_remessa_glosa From dbamv.glosas Where cd_remessa_glosa is not null AND GLOSAS.Cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA)
Group by convenio.nm_convenio,
           decode ('' , 'Entrega', to_char(remessa.dt_entrega_da_fatura,'mm'), to_char(fatura.dt_competencia, 'mm'))
)
group by nm_convenio,
         mes
)
ORDER BY NM_CONVENIO,
MES

select mes                        mes_linha,
          nm_convenio         desc_linha,
          valor                       valor_linha,
          cd_convenio
from
(
select mes,
          nm_convenio,
          sum(vl_glosa) valor,
          cd_convenio
from (
select convenio.nm_convenio,
       glosas.vl_glosa,
       to_char(fatura.dt_competencia, 'mm') mes,
       convenio.cd_convenio
from   dbamv.glosas,
       dbamv.reg_fat,
       dbamv.convenio,
       dbamv.empresa_convenio,
       dbamv.remessa_fatura  remessa,
       dbamv.fatura,
       dbamv.remessa_glosa

And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and reg_fat.cd_remessa        =  remessa.cd_remessa
and    remessa.cd_fatura         =  fatura.cd_fatura
and    reg_fat.cd_reg_fat      =  glosas.cd_reg_fat
and    glosas.cd_remessa_glosa   =  remessa_glosa.cd_remessa_glosa

            Sum( remessa_glosa.vl_total_remessa ) vl_glosa,
           to_char(fatura.dt_competencia, 'mm') mes,
           convenio.cd_convenio
  From dbamv.convenio,
   dbamv.empresa_convenio,
          dbamv.remessa_fatura remessa,
          dbamv.remessa_glosa,
          dbamv.fatura
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and remessa.cd_remessa = remessa_glosa.cd_remessa
   and remessa.cd_fatura = fatura.cd_fatura
   and fatura.cd_convenio = convenio.cd_convenio
   and convenio.tp_convenio <> 'H'
   and fatura.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
   and remessa_glosa.cd_remessa_glosa not in ( Select distinct cd_remessa_glosa From dbamv.glosas Where cd_remessa_glosa is not null )

                fatura.dt_competencia,
                convenio.cd_convenio
union all
select convenio.nm_convenio,
       glosas.vl_glosa,
       to_char(fatura.dt_competencia, 'mm') mes,
       convenio.cd_convenio
from   dbamv.glosas,
       dbamv.reg_amb,
       dbamv.convenio,
       dbamv.empresa_convenio,
       dbamv.remessa_fatura  remessa,
       dbamv.fatura,
       dbamv.remessa_glosa
where  Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and reg_amb.cd_remessa        =  remessa.cd_remessa
and    remessa.cd_fatura         =  fatura.cd_fatura
and    reg_amb.cd_reg_amb      =  glosas.cd_reg_amb
and    glosas.cd_remessa_glosa   =  remessa_glosa.cd_remessa_glosa
and    remessa_glosa.cd_convenio =  convenio.cd_convenio
and    convenio.tp_convenio <> 'H'
and    reg_amb.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
:CAMPO_CF_COND_DATA
)
group by mes, nm_convenio, cd_convenio
)
where cd_convenio in
(select distinct cd_convenio
from
(
select cd_convenio, valor
from
( select cd_convenio,
            sum(vl_glosa) valor
  from
(
select glosas.vl_glosa,
          convenio.cd_convenio
from   dbamv.glosas,
       dbamv.reg_fat,
       dbamv.convenio,
       dbamv.empresa_convenio,
       dbamv.remessa_fatura  remessa,
       dbamv.fatura,
       dbamv.remessa_glosa
where  Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and reg_fat.cd_remessa        =  remessa.cd_remessa
and    remessa.cd_fatura         =  fatura.cd_fatura
and    reg_fat.cd_reg_fat      =  glosas.cd_reg_fat
and    glosas.cd_remessa_glosa   =  remessa_glosa.cd_remessa_glosa
and    remessa_glosa.cd_convenio =  convenio.cd_convenio
and    convenio.tp_convenio <> 'H'
Union All
Select Sum( remessa_glosa.vl_total_remessa ) vl_glosa,
           convenio.cd_convenio
  From dbamv.convenio,
  dbamv.empresa_convenio,
          dbamv.remessa_fatura remessa,
          dbamv.remessa_glosa,
          dbamv.fatura
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and remessa.cd_remessa = remessa_glosa.cd_remessa
   and remessa.cd_fatura = fatura.cd_fatura
   and fatura.cd_convenio = convenio.cd_convenio
   and convenio.tp_convenio <> 'H'
   and fatura.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
   and remessa_glosa.cd_remessa_glosa not in ( Select distinct cd_remessa_glosa From dbamv.glosas Where cd_remessa_glosa is not null AND GLOSAS.Cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA)
Group by convenio.cd_convenio
union all
select glosas.vl_glosa,
          convenio.cd_convenio
from   dbamv.glosas,
       dbamv.reg_amb,
       dbamv.convenio,
       dbamv.empresa_convenio,
       dbamv.remessa_fatura  remessa,
       dbamv.fatura,
       dbamv.remessa_glosa
where  Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and reg_amb.cd_remessa        =  remessa.cd_remessa
and    remessa.cd_fatura         =  fatura.cd_fatura
and    reg_amb.cd_reg_amb      =  glosas.cd_reg_amb
and    glosas.cd_remessa_glosa   =  remessa_glosa.cd_remessa_glosa
and    remessa_glosa.cd_convenio =  convenio.cd_convenio
and    convenio.tp_convenio <> 'H'
and    reg_amb.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
:CAMPO_CF_COND_DATA
)
group by cd_convenio
)
where valor >= :CAMPO_CS_MENOR
)
)
ORDER BY MES
