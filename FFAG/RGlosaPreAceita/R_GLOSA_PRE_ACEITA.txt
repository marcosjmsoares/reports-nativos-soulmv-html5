--R_GLOSA_PRE_ACEITA
SELECT * FROM (
SELECT DISTINCT CONVENIO.CD_CONVENIO
     , CONVENIO.NM_CONVENIO
     , REG_FAT.CD_REMESSA
     , FATURA.DT_COMPETENCIA
     , REG_FAT.CD_ATENDIMENTO
     , PACIENTE.CD_PACIENTE
     , PACIENTE.NM_PACIENTE
     , GRU_PRO.CD_GRU_PRO
     , GRU_PRO.DS_GRU_PRO
     , PRO_FAT.CD_PRO_FAT
     , PRO_FAT.DS_PRO_FAT
     , nvl(GLOSAS.VL_GLOSA_PRE_ACEITA,0) VL_GLOSA_PRE_ACEITA
     , 'ds_mot_pre_aceite' ds_mot_pre_aceite
  FROM DBAMV.REG_FAT
     , DBAMV.REMESSA_FATURA
     , DBAMV.CONVENIO
     , DBAMV.FATURA
     , DBAMV.ATENDIME
     , DBAMV.PACIENTE
     , DBAMV.GLOSAS
     , DBAMV.PRO_FAT
     , DBAMV.GRU_PRO
 WHERE GLOSAS.CD_MULTI_EMPRESA   = :CAMPO_CD_MULTI_EMPRESA
   AND REMESSA_FATURA.CD_REMESSA = REG_FAT.CD_REMESSA
   AND CONVENIO.CD_CONVENIO      = FATURA.CD_CONVENIO
   AND FATURA.CD_FATURA          = REMESSA_FATURA.CD_FATURA
   AND ATENDIME.CD_ATENDIMENTO   = REG_FAT.CD_ATENDIMENTO
   AND PACIENTE.CD_PACIENTE      = ATENDIME.CD_PACIENTE
   AND REG_FAT.CD_REG_FAT        = GLOSAS.CD_REG_FAT
   AND PRO_FAT.CD_PRO_FAT        = GLOSAS.CD_PRO_FAT
   AND GRU_PRO.CD_GRU_PRO        = PRO_FAT.CD_GRU_PRO
   and glosas.dt_glosa between :CAMPO_DATA_INI
                           and 
   AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
UNION ALL
SELECT DISTINCT CONVENIO.CD_CONVENIO
     , CONVENIO.NM_CONVENIO
     , REG_AMB.CD_REMESSA
     , FATURA.DT_COMPETENCIA
     , ITREG_AMB.CD_ATENDIMENTO
     , PACIENTE.CD_PACIENTE
     , PACIENTE.NM_PACIENTE
     , GRU_PRO.CD_GRU_PRO
     , GRU_PRO.DS_GRU_PRO
     , PRO_FAT.CD_PRO_FAT
     , PRO_FAT.DS_PRO_FAT
     , nvl(GLOSAS.VL_GLOSA_PRE_ACEITA,0) VL_GLOSA_PRE_ACEITA
     , 'ds_mot_pre_aceite' DS_MOT_PRE_ACEITE
  FROM DBAMV.REG_AMB
     , DBAMV.ITREG_AMB
     , DBAMV.GLOSAS
     , DBAMV.REMESSA_FATURA
     , DBAMV.FATURA
     , DBAMV.CONVENIO
     , DBAMV.ATENDIME
     , DBAMV.PACIENTE
     , DBAMV.PRO_FAT
     , DBAMV.GRU_PRO
 WHERE REG_AMB.CD_REG_AMB        = GLOSAS.CD_REG_AMB
   AND ITREG_AMB.CD_REG_AMB      = GLOSAS.CD_REG_AMB
   AND REMESSA_FATURA.CD_REMESSA = REG_AMB.CD_REMESSA
   AND FATURA.CD_FATURA          = REMESSA_FATURA.CD_FATURA
   AND CONVENIO.CD_CONVENIO      = FATURA.CD_CONVENIO
   AND ATENDIME.CD_ATENDIMENTO   = ITREG_AMB.CD_ATENDIMENTO
   AND PRO_FAT.CD_PRO_FAT        = GLOSAS.CD_PRO_FAT
   AND GRU_PRO.CD_GRU_PRO        = PRO_FAT.CD_GRU_PRO
   AND ATENDIME.CD_PACIENTE      = PACIENTE.CD_PACIENTE
   AND GLOSAS.CD_MULTI_EMPRESA   = :CAMPO_CD_MULTI_EMPRESA
   and glosas.dt_glosa between :CAMPO_DATA_INI
                           and 
   AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
 ORDER BY  2,3,5,8,10
)
ORDER BY NM_CONVENIO,CD_CONVENIO,
CD_REMESSA,DT_COMPETENCIA,
CD_ATENDIMENTO,CD_PACIENTE,NM_PACIENTE,
CD_GRU_PRO,DS_GRU_PRO,CD_PRO_FAT
