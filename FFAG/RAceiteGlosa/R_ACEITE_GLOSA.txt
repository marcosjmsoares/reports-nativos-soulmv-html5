--R_ACEITE_GLOSA
select nf.cd_nota_fiscal
     , nf.nr_id_nota_fiscal
     , nf.vl_total_nota
     , nf.dt_emissao
     , nf.cd_convenio
     , sum(itm.vl_glosa) vl_glosa
  from dbamv.nota_fiscal nf
      ,dbamv.convenio co
      ,dbamv.mov_glosas mg
      ,dbamv.itmov_glosas itm
      ,dbamv.itfat_nota_fiscal itf
  where co.cd_convenio = nf.cd_convenio
    and mg.cd_mov_glosas = itm.cd_mov_glosas
    and itm.cd_itfat_nf = itf.cd_itfat_nf
    and itf.cd_nota_fiscal = nf.cd_nota_fiscal
   and   mg.dt_movimentacao >= to_date(:CAMPO_DATA_INICIAL,'DD/MM/YYYY')
   and   mg.dt_movimentacao <= to_date(:CAMPO_DATA_FINAL,'DD/mm/yyyy')
  group by  nf.cd_nota_fiscal
     , nf.nr_id_nota_fiscal
     , nf.vl_total_nota
     , nf.dt_emissao
     , nf.cd_convenio
