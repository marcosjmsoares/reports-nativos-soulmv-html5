--R_MOTIVO_AUDITORIA
SELECT CD_MOTIVO_AUDITORIA Codigo
      ,DS_MOTIVO_AUDITORIA Descricao
      ,DECODE(TP_MOTIVO_AUDITORIA,'A','Auditoria','O','Operacional') TP_MOTIVO_AUDITORIA
  FROM DBAMV.MOTIVO_AUDITORIA
where TP_MOTIVO_AUDITORIA like :CAMPO_TIPO
 order by
