--R_GLOSA_MOTIVO_DT_LANC
SELECT * FROM (
select distinct motivo.cd_motivo_glosa,
       motivo.ds_motivo_glosa,
       convenio.nm_convenio,
       reg_fat.cd_remessa       remessa,
       reg_fat.cd_reg_fat       conta,
       paciente.nm_paciente,
       pro_fat.cd_pro_fat,
       pro_fat.ds_pro_fat,
       glosas.vl_glosa,
       glosas.vl_reapresentado,
       glosas.vl_recebido,
       glosas.cd_glosas
from   dbamv.glosas,
       dbamv.reg_fat,
       dbamv.itreg_fat,
       dbamv.motivo_glosa    motivo,
       dbamv.convenio,
       dbamv.empresa_Convenio,
       dbamv.atendime,
       dbamv.pro_fat,
       dbamv.paciente
where  Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and reg_fat.cd_reg_fat        =  itreg_fat.cd_reg_fat
and    reg_fat.cd_atendimento    =  atendime.cd_atendimento
and    reg_fat.cd_convenio       =  convenio.cd_convenio
and    itreg_fat.cd_pro_fat      =  pro_fat.cd_pro_fat
and    itreg_fat.cd_reg_fat      =  glosas.cd_reg_fat
and    itreg_fat.cd_pro_fat      =  glosas.cd_pro_fat
and    glosas.cd_motivo_glosa    =  motivo.cd_motivo_glosa
and    atendime.cd_paciente      =  paciente.cd_paciente
and    convenio.tp_convenio <> 'H'
and    atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
and atendime.cd_multi_empresa =  dbamv.Pkg_Mv2000.Le_Empresa
union all
select distinct motivo.cd_motivo_glosa,
       motivo.ds_motivo_glosa,
       convenio.nm_convenio,
       reg_amb.cd_remessa       remessa,
       reg_amb.cd_reg_amb       conta,
       paciente.nm_paciente,
       pro_fat.cd_pro_fat,
       pro_fat.ds_pro_fat,
       glosas.vl_glosa,
       glosas.vl_reapresentado,
       glosas.vl_recebido,
       glosas.cd_glosas
from   dbamv.glosas,
       dbamv.reg_amb,
       dbamv.motivo_glosa    motivo,
       dbamv.convenio,
       dbamv.empresa_convenio,
       dbamv.atendime,
       dbamv.pro_fat,
       dbamv.paciente
where  Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and glosas.cd_reg_amb  =  reg_amb.cd_reg_amb
and    glosas.cd_atendimento  =  atendime.cd_atendimento
and    reg_amb.cd_convenio     =  convenio.cd_convenio
and    glosas.cd_pro_fat      =  pro_fat.cd_pro_fat
and    glosas.cd_motivo_glosa    =  motivo.cd_motivo_glosa
and    atendime.cd_paciente      =  paciente.cd_paciente
and    reg_amb.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
and    convenio.tp_convenio <> 'H'
and    atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
and atendime.cd_multi_empresa =  dbamv.Pkg_Mv2000.Le_Empresa 
union all
select 000 cd_motivo_glosa,
       'Glosa Não Identificada' ds_motivo_glosa,
       convenio.nm_convenio,
       remessa.cd_remessa       remessa,
       to_number(null)       conta,
       null       nm_paciente,
       null       cd_pro_fat,
       null       ds_pro_fat,
       remessa_glosa.vl_glosa_nao_identificada vl_glosa,
       0        vl_reapresentado  ,
       0         vl_recebido,
       0         cd_glosas
from   dbamv.remessa_glosa,
       dbamv.remessa_fatura  remessa,
       dbamv.fatura,
       dbamv.convenio,
       dbamv.empresa_convenio
where  Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and remessa_glosa.cd_remessa = remessa.cd_remessa
and    remessa_glosa.tp_glosa = 'N'
and    remessa.cd_fatura         =  fatura.cd_fatura
and    fatura.cd_convenio = convenio.cd_convenio
and    fatura.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
and    convenio.tp_convenio <> 'H'
:CAMPO_CF_COND_CONVENIO
)
ORDER BY CD_MOTIVO_GLOSA,DS_MOTIVO_GLOSA
