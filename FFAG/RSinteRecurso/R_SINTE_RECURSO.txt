--R_SINTE_RECURSO
SELECT * FROM (
Select nf.cd_convenio,
           co.nm_convenio,
           gl.cd_setor,
           se.nm_setor,
           gl.cd_motivo_recurso_glosa,
           rg.ds_motivo_recurso_glosa,
           sum(nf.vl_itfat_nf) vl_itfat_nf,
           sum(gl.vl_glosa) vl_glosa,
           sum(gl.vl_reapresentado) vl_reapresentado
   from dbamv.convenio co,
           dbamv.glosas gl,
           dbamv.setor se,
           (select nf.nr_id_nota_fiscal
	     ,itnf.cd_remessa
	     ,nf.cd_convenio
	     ,itnf.cd_atendimento
	     ,itnf.cd_prestador
	     ,itnf.cd_reg_amb
	     ,itnf.cd_reg_fat
	     ,itnf.vl_itfat_nf
          from   dbamv.nota_fiscal nf
	  , dbamv.itfat_nota_fiscal itnf
          where nf.cd_nota_fiscal = itnf.cd_nota_fiscal) nf,
           dbamv.motivo_recurso_glosa rg
where co.cd_convenio = nf.cd_convenio
  and gl.cd_setor = se.cd_setor
  and gl.cd_atendimento = nf.cd_atendimento
	and nvl(gl.cd_prestador,0) = nvl(nf.cd_prestador,0)
  and (nf.cd_reg_amb = gl.cd_reg_amb
       or nf.cd_reg_fat = gl.cd_reg_fat)
  and gl.cd_motivo_recurso_glosa = rg.cd_motivo_recurso_glosa
  and gl.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
and trunc(gl.dt_glosa) between to_date(:CAMPO_DT_INICIO, 'dd/mm/yyyy') and to_date(:CAMPO_DT_FINAL, 'dd/mm/yyyy')
group by nf.cd_convenio,
              co.nm_convenio,
              gl.cd_setor,
              se.nm_setor,
              gl.cd_motivo_recurso_glosa,
              rg.ds_motivo_recurso_glosa
order by 1,3
)
ORDER BY CD_CONVENIO,NM_CONVENIO,
CD_SETOR,NM_SETOR
