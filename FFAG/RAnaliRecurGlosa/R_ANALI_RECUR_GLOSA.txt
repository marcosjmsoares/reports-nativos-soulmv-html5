--R_ANALI_RECUR_GLOSA
SELECT * FROM (
select nf.cd_convenio,
       at.cd_paciente,
       co.nm_convenio,
       gl.cd_atendimento,
       at.dt_atendimento,
       gl.cd_setor,
       se.nm_setor,
       nvl(gl.cd_reg_amb,gl.cd_reg_fat) conta,
       it.cd_remessa,
       gl.cd_pro_fat,
       sum(it.vl_itfat_nf) VL_TOTAL_ITEM,
       sum(gl.vl_reapresentado) VL_TOTAL_RECURSO,
       ju.ds_justifica_recurso_glosa,
       gl.cd_prestador
 from  dbamv.glosas gl,
       dbamv.nota_fiscal nf,
       dbamv.atendime at,
       dbamv.itfat_nota_fiscal it,
       dbamv.setor se,
       dbamv.convenio co,
       dbamv.justifica_Recurso_glosa ju
 where gl.cd_setor                                         =  se.cd_setor
     and nf.cd_nota_fiscal                               = it.cd_nota_fiscal
     and co.cd_convenio                                = nf.cd_convenio
     and nvl(gl.cd_prestador,0)                        = nvl(it.cd_prestador,0)
     and gl.cd_reg_fat                                      = it.cd_reg_fat
     and gl.cd_pro_fat                                      = it.cd_pro_fat
     and gl.cd_atendimento                             = it.cd_atendimento
     and gl.cd_atendimento                             = at.cd_atendimento
     and ju.cd_justifica_recurso_glosa             = gl.cd_justifica_recurso_glosa
     and gl.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
     and trunc(gl.dt_glosa) between to_date(:CAMPO_DATA_INI,'dd/mm/yyyy') and to_date(:CAMPO_DATA_FIM,'dd/mm/yyyy')
 group by  nf.cd_convenio,
                at.cd_paciente,
                co.nm_convenio,
                 gl.cd_atendimento,
                at.dt_atendimento,
                gl.cd_setor,
                se.nm_setor,
                 nvl(gl.cd_reg_amb,gl.cd_reg_fat),
                 it.cd_remessa,
                 gl.cd_pro_fat,
                 ju.ds_justifica_recurso_glosa,
                 gl.cd_prestador
union all
select nf.cd_convenio,
          at.cd_paciente,
          co.nm_convenio,
           gl.cd_atendimento,
          at.dt_atendimento,
          gl.cd_setor,
          se.nm_setor,
          nvl(gl.cd_reg_amb,gl.cd_reg_fat) conta,
           it.cd_remessa,
          gl.cd_pro_fat,
          sum(it.vl_itfat_nf) VL_TOTAL_ITEM,
          sum(gl.vl_reapresentado) VL_TOTAL_RECURSO,
          ju.ds_justifica_recurso_glosa,
         gl.cd_prestador
  from dbamv.glosas gl,
         dbamv.itfat_nota_fiscal it,
         dbamv.nota_fiscal nf,
         dbamv.setor se,
         dbamv.atendime at,
         dbamv.convenio co,
         dbamv.justifica_Recurso_glosa ju
 where gl.cd_setor                                          = se.cd_setor
     and nf.cd_nota_fiscal                                = it.cd_nota_fiscal
     and co.cd_convenio                                 = nf.cd_convenio
     and nvl(gl.cd_prestador,0)                         = nvl(it.cd_prestador,0)
     and gl.cd_reg_amb                                      = it.cd_reg_amb
     and gl.cd_pro_fat                                       = it.cd_pro_fat
     and gl.cd_atendimento                               = it.cd_atendimento
     and  gl.cd_atendimento                              = at.cd_atendimento
     and  ju.cd_justifica_recurso_glosa             = gl.cd_justifica_recurso_glosa
     and gl.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
     and trunc(gl.dt_glosa) between to_date(:CAMPO_DATA_INI,'dd/mm/yyyy') and to_date(:CAMPO_DATA_FIM,'dd/mm/yyyy')
 group by  nf.cd_convenio,
                at.cd_paciente,
                co.nm_convenio,
                 gl.cd_atendimento,
                 at.dt_atendimento,
                 gl.cd_setor,
                se.nm_setor,
                nvl(gl.cd_reg_amb,gl.cd_reg_fat),
                 it.cd_remessa,
                gl.cd_pro_fat,
                ju.ds_justifica_recurso_glosa,
                gl.cd_prestador
 order by 10,4,12
)
ORDER BY CD_CONVENIO,NM_CONVENIO,
CD_SETOR,NM_SETOR
