--R_AUDITORIA_PROC
SELECT * FROM (
SELECT P.CD_PACIENTE ||' - '|| P.NM_PACIENTE   CD_PACIENTE,
               R.CD_REG_FAT,
               R.NR_GUIA,
               NVL( TO_CHAR( A.DT_ATENDIMENTO, 'DD/MM/YYYY'), '' ) DT_INTERN,
               NVL( TO_CHAR( A.DT_ALTA, 'DD/MM/YYYY'), '' ) DT_ALTA,
               A.CD_LEITO ||' - '|| L.DS_LEITO    CD_LEITO,
               R.CD_CONVENIO ||' - '|| C.NM_RAZAO_SOCIAL     CD_CONVENIO,
               R.DT_INICIO,
               R.DT_FINAL,
               R.VL_TOTAL_CONTA,
               AC.CD_MOTIVO_AUDITORIA,
               M.DS_MOTIVO_AUDITORIA,
               AC.CD_PRO_FAT,
               PR.DS_PRO_FAT,
               AC.CD_LANCAMENTO_FAT CD_LANCAMENTO,
               AC.CD_AUDITORIA_CONTA CD_AUDITORIA_CONTA_I,
               AC.CD_AUDITORIA_CONTA CD_AUDITORIA_CONTA_F,
               A.CD_ATENDIMENTO,
               P.NM_PACIENTE,
               C.NM_CONVENIO,
               G.DS_GRU_PRO,
               S.NM_SETOR,
               ac.vl_total_conta_ant vl_fatura,
               ac.vl_total_conta - ( ac.vl_total_conta - ( ac.vl_total_conta * (ac.qt_lancamento/ac.qt_lancamento_ant) * ( ac.vl_percentual_multipla/ac.vl_percentual_multipla_ant) ) ) vl_liberado,
              ac.vl_total_conta - ( ac.vl_total_conta * (ac.qt_lancamento/ac.qt_lancamento_ant) * ( ac.vl_percentual_multipla/ac.vl_percentual_multipla_ant) ) vl_glosa,
            decode(NVL( AC.QT_LANCAMENTO_ANT,0) -NVL( AC.QT_LANCAMENTO, 0 ),0,
               NVL( AC.VL_PERCENTUAL_MULTIPLA_ANT/100,0) -NVL( AC.VL_PERCENTUAL_MULTIPLA/100, 0 ),
               NVL( AC.QT_LANCAMENTO_ANT,0) -NVL( AC.QT_LANCAMENTO, 0 ))  QT_LANCAMENTO_ITEM,
              ac.qt_lancamento_ant  qtd_faturado,
              ac.qt_lancamento,
               ac.vl_total_conta - ( ac.vl_total_conta * nvl(ac.qt_lancamento/ac.qt_lancamento_ant,1) * nvl( ac.vl_percentual_multipla/ac.vl_percentual_multipla_ant,1) )   VL_TOTAL_ITEM
FROM    DBAMV.PACIENTE P,
              DBAMV.REG_FAT R,
              DBAMV.ATENDIME A,
              DBAMV.LEITO L,
              DBAMV.GRU_PRO G,
              DBAMV.CONVENIO C,
              DBAMV.AUDITORIA_CONTA AC,
              DBAMV.SETOR S,
              DBAMV.MOTIVO_AUDITORIA M,
              DBAMV.PRO_FAT PR,
	      dbamv.empresa_convenio
WHERE Empresa_Convenio.Cd_Convenio = C.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
      AND A.CD_ATENDIMENTO = R.CD_ATENDIMENTO
      AND R.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
      AND L.CD_LEITO                 = A.CD_LEITO
      AND C.CD_CONVENIO        = R.CD_CONVENIO
      AND R.CD_ATENDIMENTO = :CAMPO_ATENDIMENTO
      AND R.CD_REG_FAT          = :CAMPO_CONTA
      AND AC.CD_REG_FAT                 = R.CD_REG_FAT
      AND AC.CD_ATENDIMENTO = A.CD_ATENDIMENTO
      AND M.CD_MOTIVO_AUDITORIA = AC.CD_MOTIVO_AUDITORIA
      AND P.CD_PACIENTE  = A.CD_PACIENTE
      AND PR.CD_PRO_FAT                  = AC.CD_PRO_FAT
      AND G.CD_GRU_PRO  =  PR.CD_GRU_PRO
      AND AC.DT_CANCELOU IS NULL
      AND AC.CD_SETOR = S.CD_SETOR
      and A.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
      and a.cd_multi_empresa =  dbamv.Pkg_Mv2000.Le_Empresa
UNION ALL
SELECT  P.CD_PACIENTE ||' - '|| P.NM_PACIENTE    CD_PACIENTE,
               R.CD_REG_AMB        CD_REG_FAT,
               ''    NR_GUIA,
               NVL( TO_CHAR( A.DT_ATENDIMENTO, 'DD/MM/YYYY'), '' ) DT_INTERN,
               NVL( TO_CHAR( A.DT_ALTA, 'DD/MM/YYYY'), '' ) DT_ALTA,
               A.CD_LEITO ||' - '|| L.DS_LEITO    CD_LEITO,
               R.CD_CONVENIO ||' - '|| C.NM_RAZAO_SOCIAL     CD_CONVENIO,
               R.DT_LANCAMENTO                      DT_INICIO,
               R.DT_LANCAMENTO_FINAL          DT_FINAL,
               R.VL_TOTAL_CONTA,
               AC.CD_MOTIVO_AUDITORIA,
               M.DS_MOTIVO_AUDITORIA,
               AC.CD_PRO_FAT,
               PR.DS_PRO_FAT,
               AC.CD_LANCAMENTO_AMB CD_LANCAMENTO,
               AC.CD_AUDITORIA_CONTA CD_AUDITORIA_CONTA_I,
               AC.CD_AUDITORIA_CONTA CD_AUDITORIA_CONTA_F,
               A.CD_ATENDIMENTO,
               P.NM_PACIENTE,
               C.NM_CONVENIO,
               G.DS_GRU_PRO,
               S.NM_SETOR,
               ac.vl_total_conta_ant    vl_fatura,
               ac.vl_total_conta - ( ac.vl_total_conta - ( ac.vl_total_conta * (ac.qt_lancamento/ac.qt_lancamento_ant) * ( ac.vl_percentual_multipla/ac.vl_percentual_multipla_ant) ) ) vl_liberado,
               ac.vl_total_conta - ( ac.vl_total_conta * (ac.qt_lancamento/ac.qt_lancamento_ant) * ( ac.vl_percentual_multipla/ac.vl_percentual_multipla_ant) ) vl_glosa,
               decode(NVL( AC.QT_LANCAMENTO_ANT,0) -NVL( AC.QT_LANCAMENTO, 0 ),0,
               NVL( AC.VL_PERCENTUAL_MULTIPLA_ANT/100,0) -NVL( AC.VL_PERCENTUAL_MULTIPLA/100, 0 ),
               NVL( AC.QT_LANCAMENTO_ANT,0) -NVL( AC.QT_LANCAMENTO, 0 ))  QT_LANCAMENTO_ITEM,
               ac.qt_lancamento_ant  qtd_faturado,
               ac.qt_lancamento,
                ac.vl_total_conta - ( ac.vl_total_conta * nvl(ac.qt_lancamento/ac.qt_lancamento_ant,1) * nvl( ac.vl_percentual_multipla/ac.vl_percentual_multipla_ant,1) )   VL_TOTAL_ITEM
FROM    DBAMV.PACIENTE P,
              DBAMV.REG_AMB R,
              DBAMV.ATENDIME A,
              DBAMV.LEITO L,
              DBAMV.CONVENIO C,
              DBAMV.GRU_PRO G,
              DBAMV.AUDITORIA_CONTA AC,
              DBAMV.MOTIVO_AUDITORIA M,
              DBAMV.SETOR S,
              DBAMV.PRO_FAT PR,
	      dbamv.empresa_Convenio
WHERE Empresa_Convenio.Cd_Convenio = C.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
      AND L.CD_LEITO(+)                 = A.CD_LEITO
      AND R.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
      AND C.CD_CONVENIO        = R.CD_CONVENIO
      AND R.CD_REG_AMB          = :CAMPO_CONTA
      AND A.CD_ATENDIMENTO = :CAMPO_ATENDIMENTO
      AND AC.CD_REG_AMB                   = R.CD_REG_AMB
      AND AC.CD_ATENDIMENTO = A.CD_ATENDIMENTO
      AND M.CD_MOTIVO_AUDITORIA = AC.CD_MOTIVO_AUDITORIA
      AND P.CD_PACIENTE  = A.CD_PACIENTE
      AND PR.CD_PRO_FAT                  = AC.CD_PRO_FAT
      AND G.CD_GRU_PRO  =  PR.CD_GRU_PRO
      AND AC.DT_CANCELOU IS NULL
      AND AC.CD_SETOR = S.CD_SETOR
      and A.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
ORDER BY  21, 22, 13
)
ORDER BY CD_PACIENTE,CD_REG_FAT,NR_GUIA,DT_INTERN,DT_ALTA,CD_LEITO,CD_CONVENIO,DT_INICIO,DT_FINAL,NM_CONVENIO,NM_PACIENTE,CD_ATENDIMENTO,VL_TOTAL_CONTA,
DS_GRU_PRO,
NM_SETOR
