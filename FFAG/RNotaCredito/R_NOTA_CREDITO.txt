SELECT Sum(nvl(mg.vl_movimentacao,0)) vl_aceite
      ,(select Sum(nvl(vl_tributo_total,0))
          FROM dbamv.tip_det_reccon_rec td,
               dbamv.reccon_rec re,
               dbamv.itcon_rec ic
         WHERE td.cd_reccon_rec = re.cd_reccon_rec
               AND re.cd_itcon_rec = ic.cd_itcon_rec
               AND ic.cd_con_rec = cr.cd_con_rec) vl_imposto
      ,rc.cd_moeda
      ,user usuario
    FROM dbamv.con_rec cr
        ,dbamv.itcon_rec it
        ,dbamv.mov_glosas mg
        ,dbamv.processo pc
        ,dbamv.reccon_rec rc
  WHERE cr.cd_con_rec = it.cd_con_rec
    AND it.cd_itcon_rec = mg.cd_itcon_rec
    AND mg.cd_processo = pc.cd_processo
    AND pc.cd_estrutural = '3.1.2.1.2'
    AND it.cd_itcon_rec = rc.cd_itcon_rec
    AND cr.cd_con_rec =  :CAMPO_CD_CON_REC
GROUP BY rc.cd_moeda, user, cr.cd_con_rec
