SELECT * FROM (
Select me.cd_multi_empresa || ' - ' || me.ds_multi_empresa Empresa
, S.Cd_Setor
, S.Nm_Setor
,L.Cd_Localidade
,L.Ds_Localidade
, B.Cd_Bem
, B.Ds_Plaqueta
, B.Ds_Bem
, B.Nr_Serie
, B.Ds_Marca
, B.Ds_Modelo
, B.Ds_Acessorios
From Dbamv.Setor S
, Dbamv.Bens B
, dbamv.multi_empresas me
,dbamv.Localidade L
Where S.Cd_Setor = B.Cd_Setor
	and B.Cd_Localidade = L.Cd_Localidade
	and S.Cd_Setor = L.Cd_Setor
    and b.cd_multi_empresa = me.cd_multi_empresa
:CAMPO_CF_ORDEM
)
