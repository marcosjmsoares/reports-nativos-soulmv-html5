Select bens.cd_bem Codigo
      ,bens.ds_plaqueta
      ,bens.ds_bem Descricao
      ,bens.nr_serie
      ,bens.ds_marca
      ,bens.ds_modelo
      ,bens.ds_acessorios
      ,bens.cd_setor
      ,bens.cd_localidade
 from dbamv.bens
     ,dbamv.setor
     ,dbamv.localidade
where bens.cd_setor      = setor.cd_setor
  and setor.cd_setor     = localidade.cd_setor
  and bens.cd_localidade = localidade.cd_localidade
  and bens.hr_baixa is null
  and bens.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
order by  bens.ds_plaqueta
