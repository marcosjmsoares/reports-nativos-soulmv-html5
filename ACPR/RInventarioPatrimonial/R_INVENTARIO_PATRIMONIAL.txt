--R_INVENTARIO_PATRIMONIAL
SELECT DISTINCT
	   CD_INVENTARIO_PATRIMONIO
	  ,DS_INVENTARIO
	  ,DT_INICIO
	  ,DT_FIM
	  ,DT_FECHAMENTO
	  ,CD_USUARIO_RESPONSAVEL
 FROM (
		SELECT CD_INVENTARIO_PATRIMONIO
			  ,DS_INVENTARIO
			  ,DT_INICIO
			  ,DT_FIM
			  ,DT_FECHAMENTO
			  ,CD_USUARIO_RESPONSAVEL
		FROM DBAMV.INVENTARIO_PATRIMONIO
		WHERE Nvl(trunc(DT_FECHAMENTO), trunc(DT_FIM)) BETWEEN :CAMPO_DT_INICIO UNION ALL
	  SELECT 0 CD_INVENTARIO_PATRIMONIO
			,:CAMPO_NAO_INVENTARIADOS DS_INVENTARIO
			,To_Date(NULL) DT_INICIO
			,To_Date(NULL) DT_FIM
			,To_Date(NULL) DT_FECHAMENTO
			,NULL CD_USUARIO_RESPONSAVEL
	  FROM SYS.DUAL
		)
WHERE :CAMPO_SN_INVENTARIO = 'S'
UNION ALL
SELECT NULL, NULL, NULL, NULL, NULL, NULL
FROM SYS.DUAL
WHERE :CAMPO_SN_INVENTARIO = 'N'
ORDER BY DT_INICIO, DS_INVENTARIO

--R_INVENTARIO_PATRIMONIAL_BENS
SELECT DISTINCT
	   CD_BEM
	  ,DS_PLAQUETA
	  ,DS_BEM
	  ,VL_RESIDUAL
	  ,DS_STATUS_ANTERIOR
	  ,DS_STATUS_ATUAL
FROM (
	SELECT B.CD_BEM
		   ,BC.DS_PLAQUETA
		   ,B.DS_BEM
		   ,(B.VL_COMPRA + NVL(DBAMV.PKG_PATRIMONIO.FN_RET_DADOS_REAVALIACAOBEM(B.CD_BEM,NULL, 'V', NULL, NULL),0) - NVL(B.VL_DEPRECIACAO_HISTORICA,0) - (SELECT NVL(ROUND(SUM(VL_LANCAMENTO),2),0) FROM DBAMV.LANCAMENTO_PATRIMONIO WHERE CD_BEM = B.CD_BEM AND TP_LANCAMENTO = 'D')) VL_RESIDUAL
		   ,BC.DS_STATUS_ANTERIOR
		   ,BC.DS_STATUS_ATUAL
		   ,Nvl(IP.CD_INVENTARIO_PATRIMONIO, 0) CD_INVENTARIO_PATRIMONIO
		   ,IP.DT_FECHAMENTO
		   ,IP.DT_FIM
	  FROM DBAMV.CONTAGEM_PATRIMONIO C
		   ,DBAMV.BENS_CONTAGEM BC
		   ,DBAMV.BENS B
		   ,DBAMV.SETOR S
		   ,DBAMV.LOCALIDADE L
		   ,DBAMV.INVENTARIO_PATRIMONIO IP
	  WHERE C.CD_CONTAGEM_PATRIMONIO = BC.CD_CONTAGEM
		AND BC.DS_PLAQUETA = B.DS_PLAQUETA(+)
		AND C.CD_SETOR = S.CD_SETOR
		AND C.CD_LOCALIDADE = L.CD_LOCALIDADE
		AND C.CD_INVENTARIO = IP.CD_INVENTARIO_PATRIMONIO(+)
		AND (Nvl(C.CD_INVENTARIO, 0) = :CAMPO_CD_INVENTARIO_PATRIMONIO OR  IS NULL)
		AND (C.CD_CONTAGEM_PATRIMONIO = :CAMPO_CD_CONTAGEM_PATRIMONIO OR  IS NULL)
		AND Nvl(Trunc(C.DT_FIM_CONTAGEM), Trunc(C.DT_INI_CONTAGEM)) BETWEEN :CAMPO_DT_INICIO AND 
		:CAMPO_CF_CONTAGEM
) WHERE (trunc(DT_FECHAMENTO) BETWEEN :CAMPO_DT_INICIO AND 
				OR (trunc(DT_FECHAMENTO) IS NULL AND DT_FIM BETWEEN :CAMPO_DT_INICIO AND :CAMPO_DT_TERMINO)
				OR (DT_FECHAMENTO IS NULL AND DT_FIM IS NULL))
ORDER BY DS_BEM, DS_PLAQUETA


/*----------------------------------------------------------------------------*/
--R_INVENTARIO_PATRIMONIAL_CONTAGEM
SELECT DISTINCT
	   NM_SETOR
	  ,DS_LOCALIDADE
	  ,Trunc(DT_INI_CONTAGEM) DT_INI_CONTAGEM
	  ,Trunc(DT_FIM_CONTAGEM) DT_FIM_CONTAGEM
	  ,CD_USUARIO
	  ,CD_INVENTARIO
	  ,QTD_BENS_CONTAGEM
	  ,QTD_BENS_PENDENTES
	  ,CD_CONTAGEM_PATRIMONIO
FROM (
	SELECT S.NM_SETOR
	  ,L.DS_LOCALIDADE
	  ,C.DT_INI_CONTAGEM
	  ,C.DT_FIM_CONTAGEM
	  ,C.CD_USUARIO
	  ,Decode(CD_INVENTARIO, NULL, :CAMPO_NAO_INVENTARIADOS, CD_INVENTARIO) CD_INVENTARIO
	  ,(SELECT COUNT('X') FROM DBAMV.BENS_CONTAGEM WHERE CD_CONTAGEM = C.CD_CONTAGEM_PATRIMONIO) QTD_BENS_CONTAGEM
	  ,(SELECT COUNT('X') FROM DBAMV.BENS_CONTAGEM WHERE CD_CONTAGEM = C.CD_CONTAGEM_PATRIMONIO AND (DS_STATUS_ATUAL != 'OK' OR DS_STATUS_ATUAL IS NULL)) QTD_BENS_PENDENTES
	  ,C.CD_CONTAGEM_PATRIMONIO
	  ,IP.DT_FECHAMENTO
	  ,IP.DT_FIM
	FROM DBAMV.CONTAGEM_PATRIMONIO C
	   ,DBAMV.SETOR S
	   ,DBAMV.LOCALIDADE L
	   ,DBAMV.INVENTARIO_PATRIMONIO IP
	WHERE C.CD_SETOR = S.CD_SETOR
	AND C.CD_LOCALIDADE = L.CD_LOCALIDADE
	AND C.CD_INVENTARIO = IP.CD_INVENTARIO_PATRIMONIO(+)
	AND (Nvl(CD_INVENTARIO, 0) = :CAMPO_CD_INVENTARIO_PATRIMONIO OR  IS NULL)
	AND Nvl(Trunc(C.DT_FIM_CONTAGEM), Trunc(C.DT_INI_CONTAGEM)) BETWEEN :CAMPO_DT_INICIO AND :CAMPO_DT_TERMINO
) WHERE (trunc(DT_FECHAMENTO) BETWEEN :CAMPO_DT_INICIO AND 
				OR (trunc(DT_FECHAMENTO) IS NULL AND DT_FIM BETWEEN :CAMPO_DT_INICIO AND :CAMPO_DT_TERMINO)
				OR (DT_FECHAMENTO IS NULL AND DT_FIM IS NULL))
ORDER BY DT_INI_CONTAGEM, NM_SETOR, DS_LOCALIDADE
