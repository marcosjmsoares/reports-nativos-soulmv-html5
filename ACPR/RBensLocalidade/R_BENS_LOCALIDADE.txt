--R_BENS_LOCALIDADE
SELECT * FROM (
Select	Me.cd_multi_empresa|| ' - ' ||Me.ds_multi_empresa Empresa,
                Me.cd_multi_empresa cd_multi_empresa1,
                Ib.Cd_Capitulo Cd_Capitulo,
	Ib.Ds_Capitulo Ds_Capitulo,
	Se.Cd_Setor Cd_CentroCusto,
	Se.Nm_Setor Nm_Setor,
	Be.Cd_Bem Cd_Bem,
	Be.Ds_Plaqueta Ds_Plaqueta,
	Be.Ds_Bem Ds_Bem,
	Be.Cd_Sub_Cap Cd_Sub_Cap,
	Si.Ds_Sub_Cap Ds_Sub_Cap,
	Be.Dt_Compra Dt_Compra,
                Be.Nr_NotaFiscal Nr_NotaFiscal
From	Dbamv.Iob	Ib,
	Dbamv.Sub_Iob	Si,
	Dbamv.Setor	Se,
	Dbamv.Bens	Be,
                Dbamv.Multi_Empresas Me
Where	Be.Cd_Setor	     =	Se.Cd_Setor
And	Be.Cd_Capitulo 	     =	Ib.Cd_Capitulo
And	Be.Cd_Capitulo 	     =	Si.Cd_Capitulo
And	Be.Cd_Sub_Cap	     =	Si.Cd_Sub_Cap
And         Be.Cd_Multi_Empresa   =        Me.Cd_Multi_Empresa
AND dbamv.pkg_patrimonio.fn_retbemativochr(be.cd_bem) = 'A'
Order By Ib.cd_Capitulo
)ORDER BY EMPRESA,
CD_CAPITULO,CD_SUB_CAP,
CD_CENTROCUSTO
