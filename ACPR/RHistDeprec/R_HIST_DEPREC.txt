--R_HIST_DEPREC
SELECT * FROM (
select     e.cd_multi_empresa || ' - ' || e.ds_multi_empresa Empresa
             ,b.cd_bem codigo
             ,b.dt_competencia
             ,b.dt_lancamento dt_deprecia
             ,b.ds_plaqueta
             ,b.ds_bem descricao
             ,b.cd_capitulo
             ,iob.ds_capitulo
             ,b.cd_sub_cap
             ,siob.ds_sub_cap
             ,b.Tx_Depreciacao * 100 tx_depre
             ,b.vl_deprecia_antes vl_anterior
             ,b.vl_deprecia_atual vl_deprecia
from           dbamv.v_lancamento_patrimonio b
              ,dbamv.iob iob
              ,dbamv.sub_iob siob
              ,dbamv.multi_empresas e
where        b.tp_lancamento = 'D'
and          b.cd_capitulo      =  iob.cd_capitulo
and          siob.cd_capitulo  =  iob.cd_capitulo
and          b.cd_sub_cap     =  siob.cd_sub_cap
and          b.dt_lancamento      between    :CAMPO_DT_INICIAL  and  last_day(:CAMPO_DT_FINAL)
and          b.cd_multi_empresa_bem = e.cd_multi_empresa
order by e.cd_multi_empresa,b.ds_bem, b.dt_lancamento,  b.dt_competencia
)
ORDER BY EMPRESA,CODIGO,DT_COMPETENCIA,DS_PLAQUETA,DESCRICAO
