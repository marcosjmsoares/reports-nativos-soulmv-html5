--R_HIST_SEGUROS
SELECT * FROM (
select  distinct bs.cd_bem Codigo
          ,b.ds_bem Descricao
          ,b.ds_plaqueta
          ,'Empresa ' || me.cd_multi_empresa || ' - ' || me.ds_multi_empresa Empresa
 from dbamv.bens_segurados bs,
         dbamv.bens b,
         dbamv.multi_empresas me
where b.cd_bem = bs.cd_bem
and     b.cd_multi_empresa = me.cd_multi_empresa
and     b.hr_baixa is null
:CAMPO_CF_WHERE_EMPRESA
)
