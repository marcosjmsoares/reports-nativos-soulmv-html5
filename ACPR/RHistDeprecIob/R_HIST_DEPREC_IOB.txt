--R_HIST_DEPREC_IOB
SELECT * FROM (
select     me.cd_multi_empresa || ' - ' || me.ds_multi_empresa Empresa
             ,b.dt_lancamento  dt_deprecia
             ,b.cd_capitulo
             ,iob.ds_capitulo
             ,b.cd_sub_cap
             ,siob.ds_sub_cap
              ,b.tx_depreciacao * 100  tx_depre
             ,sum(b.vl_deprecia_antes) vl_anterior
             ,sum(b.vl_deprecia_atual) vl_deprecia
from       dbamv.iob iob
             ,dbamv.sub_iob siob
             ,dbamv.v_lancamento_patrimonio b
             ,dbamv.multi_empresas me
where     b.tp_lancamento  =  'D'
and          b.cd_capitulo       =  iob.cd_capitulo
and          siob.cd_capitulo   = iob.cd_capitulo
and          b.cd_sub_cap      =  siob.cd_sub_cap
and        me.cd_multi_empresa = b.cd_multi_empresa_bem 
and        b.dt_lancamento    between :CAMPO_DT_INICIAL   and  last_day(:CAMPO_DT_FINAL)       
group by me.cd_multi_empresa || ' - ' || me.ds_multi_empresa  
             ,b.cd_capitulo
             ,iob.ds_capitulo
             ,b.cd_sub_cap
             ,siob.ds_sub_cap
              ,b.tx_depreciacao
             ,b.dt_lancamento
order by dt_deprecia
)
ORDER BY EMPRESA,
CD_CAPITULO,DS_CAPITULO,
CD_SUB_CAP,DS_SUB_CAP, dt_deprecia
