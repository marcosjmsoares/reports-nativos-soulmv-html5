--R_BENS_CLASSIF_IOB
SELECT 'Empresa ' || me.cd_multi_empresa || ' - ' || me.ds_multi_empresa Empresa
              ,me.cd_multi_empresa cd_multi_empresa1
              ,IOB.CD_CAPITULO CD_CAPITULO
              ,IOB.DS_CAPITULO DS_CAPITULO
              ,BENS.CD_BEM CD_BEM
              ,BENS.DS_BEM DS_BEM
              ,BENS.DS_PLAQUETA DS_PLAQUETA
              ,BENS.DT_COMPRA DT_AQUISICAO
              ,BENS.VL_COMPRA VL_ORIGINAL,
 NVL(BENS.VL_DEPRECIACAO_HISTORICA, 0) + NVL(BENS.VL_DEPREC_ACUM_ECO, 0)    VL_DEPREC_HISTORICA,
       Dbamv.Pkg_Patrimonio.Fn_RetTotalDeprecBem( Bens.Cd_Bem, Trunc(Sysdate,'MM'), Null, Null, 'C' )    VL_DEPREC_CUSTO,
       Dbamv.Pkg_Patrimonio.Fn_Ret_Dados_ReavaliacaoBem( Bens.Cd_Bem, Null, 'V', Null, Null )         Vl_Total_Reavaliacao,
       Dbamv.Pkg_Patrimonio.Fn_RetTotalDeprecBem( Bens.Cd_Bem, Trunc(Sysdate,'MM'), Null, Null, 'R' ) Vl_Total_Deprec_Reav
  FROM DBAMV.IOB
            ,DBAMV.BENS
            ,DBAMV.ITDEPRECIA IT
            ,dbamv.multi_empresas me
  WHERE IOB.CD_CAPITULO                  = BENS.CD_CAPITULO
        AND BENS.CD_BEM                        = IT.CD_BEM (+)
        AND BENS.CD_MULTI_EMPRESA = ME.CD_MULTI_EMPRESA
        AND dbamv.pkg_patrimonio.fn_retbemativochr(bens.cd_bem) = 'A'
 GROUP BY  IOB.CD_CAPITULO
              ,IOB.DS_CAPITULO
              ,BENS.CD_BEM
              ,BENS.DS_BEM
              ,BENS.DS_PLAQUETA
              ,BENS.DT_COMPRA
              ,BENS.VL_COMPRA
              ,BENS.VL_DEPRECIACAO_HISTORICA
              ,BENS.VL_DEPREC_ACUM_ECO
              ,me.cd_multi_empresa
              ,me.ds_multi_empresa
