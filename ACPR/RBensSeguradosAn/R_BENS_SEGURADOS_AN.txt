--R_BENS_SEGURADOS_AN
SELECT * FROM (
SELECT 'Empresa ' || me.cd_multi_empresa || ' - ' || me.ds_multi_empresa Empresa
      ,' ' ||me.cd_multi_empresa || ' - ' || me.ds_multi_empresa Emp
      ,BS.NR_APOLICE
      ,BS.DT_INICIO
      ,BS.DT_FIM
      ,BS.SN_ATIVO
      ,BS.VL_BENS_SEGURADO
      ,BS.CD_BEM Codigo
      ,BS.CD_SEGURADORA
      ,BS.VL_PAGO_SEGURO
      ,BS.TP_SEGURO
      ,B.DS_MARCA
      ,B.DS_BEM Descricao
      ,B.DS_MODELO
      ,S.DS_SEGURADORA
      ,BS.CD_BENS_SEGURADOS
FROM dbamv.BENS_SEGURADOS BS
         , dbamv.BENS B
         , dbamv.SEGUROS S
         , dbamv.multi_empresas me
WHERE BS.SN_ATIVO = 'S'
     AND BS.CD_BEM = B.CD_BEM
     AND BS.CD_SEGURADORA = S.CD_SEGURADORA(+)
     AND B.HR_BAIXA IS NULL
     AND BS.DT_FIM BETWEEN :CAMPO_DT_INICIAL AND 
     and   me.cd_multi_empresa = b.cd_multi_empresa
UNION
SELECT 'Empresa ' || me.cd_multi_empresa || ' - ' || me.ds_multi_empresa Empresa
      ,' ' ||me.cd_multi_empresa || ' - ' || me.ds_multi_empresa Emp
      ,BS.NR_APOLICE
      ,BS.DT_INICIO
      ,BS.DT_FIM
      ,BS.SN_ATIVO
      ,IOB.VL_SEGURADO
      ,IOB.CD_BENS_SEGURADOS_IOB Codigo
      ,BS.CD_SEGURADORA
      ,BS.VL_PAGO_SEGURO
      ,BS.TP_SEGURO
      ,B.DS_MARCA
      ,B.DS_BEM Descricao
      ,B.DS_MODELO
      ,S.DS_SEGURADORA
      ,BS.CD_BENS_SEGURADOS
FROM dbamv.BENS_SEGURADOS BS
   , dbamv.BENS_SEGURADOS_IOB IOB
   , dbamv.BENS B
   , dbamv.SEGUROS S
   , dbamv.multi_empresas me
WHERE BS.CD_BENS_SEGURADOS = IOB.CD_BENS_SEGURADOS
     AND BS.SN_ATIVO = 'S'
     AND IOB.CD_BENS_SEGURADOS_IOB = B.CD_BEM
     AND IOB.SN_SEGURADO = 'S'
     AND BS.CD_SEGURADORA = S.CD_SEGURADORA(+)
     AND B.HR_BAIXA IS NULL
     AND BS.DT_FIM BETWEEN :CAMPO_DT_INICIAL AND 
     and   me.cd_multi_empresa = b.cd_multi_empresa
:CAMPO_CF_EMPRESA
)
