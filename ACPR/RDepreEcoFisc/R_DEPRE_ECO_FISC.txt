--R_DEPRE_ECO_FISC
SELECT b.cd_bem,
	   b.ds_plaqueta,
	   b.ds_bem,
	   b.sn_bem_laudado,
	   SUM(NVL(i.VL_DEPRECIA_ANTES,0) - NVL(i.VL_DEPRECIA_ATUAL,0)) + (SUM(NVL(i.Vl_Amortizacao_Antes,0)) - SUM(NVL(i.vl_amortizacao_atual,0))) +
         (select nvl(sum(lp.vl_lancamento),0) vl_lancamento
                 from dbamv.lancamento_patrimonio lp where lp.cd_bem = b.cd_bem and lp.TP_LANCAMENTO in ('Y')
                  and lp.Dt_Lancamento between  :CAMPO_DT_INICIO AND  :CAMPO_DT_FIM) VL_CONTABIL,
	   SUM(NVL(i.VL_DEPRECIA_ANTES_ECO, i.VL_DEPRECIA_ANTES) -
                        NVL(i.VL_DEPRECIA_ATUAL_ECO, i.VL_DEPRECIA_ATUAL)) VL_FISCAL,
	  SUM( (NVL(i.VL_DEPRECIA_ANTES,0) - NVL(i.VL_DEPRECIA_ATUAL,0) )  -
                       (NVL(i.VL_DEPRECIA_ANTES_ECO, i.VL_DEPRECIA_ANTES) -
 	   NVL(i.VL_DEPRECIA_ATUAL_ECO, i.VL_DEPRECIA_ATUAL) )) VL_DIFERENCA,
     o.ds_capitulo,
	   s.ds_sub_cap
	   FROM DBAMV.BENS b
	   JOIN DBAMV.ITDEPRECIA i ON i.cd_bem = b.cd_bem
	   JOIN DBAMV.DEPRECIA d ON d.cd_deprecia = i.cd_deprecia
	   JOIN DBAMV.IOB o ON o.cd_capitulo = b.cd_capitulo
	   JOIN DBAMV.SUB_IOB s ON s.cd_sub_cap = b.cd_sub_cap AND s.CD_CAPITULO = o.CD_CAPITULO
	   WHERE d.DT_DEPRECIA between  :CAMPO_DT_INICIO GROUP BY  b.cd_bem,
         b.ds_plaqueta,
         b.ds_bem,
         b.sn_bem_laudado,
         o.ds_capitulo,
         s.ds_sub_cap
