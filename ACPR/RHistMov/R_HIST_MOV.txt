--R_HIST_MOV
SELECT * FROM (
select  bens.cd_bem bem_cd_bem
     , NVL(bens.VL_DEPREC_ACUM_ECO,0)               VL_DEPREC_ACUM_ECO 
     , bens.VL_COMPRA
     , bens.ds_plaqueta bem_ds_plaqueta
     , bens.ds_bem bem_ds_bem
     , setor.cd_setor                                          set_atual_cd_setor
     , setor.nm_setor                                         set_atual_nm_setor
     , localidade.cd_localidade                         loc_atual_cd_localidade
     , localidade.ds_localidade                         loc_atual_ds_localidade
     , v_hist_mov_bem.dt_mov_bem                Data
     , v_hist_mov_bem.tp_alocacao                 Aloc_tp_alocacao
     , v_hist_mov_bem.cd_mov_bem               Mov_cd_mov_bem
     , v_hist_mov_bem.ds_mov_bem               Mov_ds_mov_bem
     , v_hist_mov_bem.cd_empresa_origem ||' - '|| v_hist_mov_bem.ds_empresa_origem EmpOri_emp_origem
     , v_hist_mov_bem.cd_setor_origem ||' - '|| v_hist_mov_bem.ds_setor_origem SetOri_setor_origem
     , v_hist_mov_bem.cd_localidade_origem ||' - '|| v_hist_mov_bem.ds_localidade_origem   LocOri_localidade_origem
     , v_hist_mov_bem.cd_empresa_destino ||' - '|| v_hist_mov_bem.ds_empresa_destino     EmpDest_empresa_destino
     , v_hist_mov_bem.cd_setor ||' - '|| v_hist_mov_bem.nm_setor  SetDest_setor
     , v_hist_mov_bem.cd_localidade ||' - '|| v_hist_mov_bem.ds_localidade        LocDest_ds_localidade
  from dbamv.v_hist_mov_bem
     , dbamv.bens
     , dbamv.localidade
     , dbamv.setor
 where bens.cd_bem = v_hist_mov_bem.cd_bem
   and bens.cd_setor = setor.cd_setor
   and bens.cd_localidade = localidade.cd_localidade
   and bens.hr_baixa is null
   and trunc(v_hist_mov_bem.dt_mov_bem) between :CAMPO_DT_INICIAL and 
:CAMPO_CF_ORDEM
)
ORDER BY BEM_CD_BEM,BEM_DS_BEM,BEM_DS_PLAQUETA,
SET_ATUAL_CD_SETOR,SET_ATUAL_NM_SETOR,LOC_ATUAL_CD_LOCALIDADE,LOC_ATUAL_DS_LOCALIDADE
