--R_SETOR_LOCALIDADE
SELECT * FROM (
Select	Lc.Cd_Setor,
	Se.Nm_Setor,
	Lc.Cd_Localidade,
	Lc.Ds_Localidade,
	Lc.Nm_Responsavel
From	Dbamv.Setor		Se,
	Dbamv.Localidade	Lc
Where	Lc.Cd_Setor	=	Se.Cd_Setor
AND se.cd_multi_empresa = dbamv.pkg_mv2000.le_empresa
Order By 2
)
ORDER BY NM_SETOR,CD_SETOR
