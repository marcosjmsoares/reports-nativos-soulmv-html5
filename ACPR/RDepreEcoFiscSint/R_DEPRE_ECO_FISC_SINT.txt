--R_DEPRE_ECO_FISC_SINT
SELECT
     o.cd_capitulo,
     o.ds_capitulo,
	   SUM(NVL(i.VL_DEPRECIA_ANTES,0) - NVL(i.VL_DEPRECIA_ATUAL,0)) VL_CONTABIL,
	   SUM(NVL(i.VL_DEPRECIA_ANTES_ECO, i.VL_DEPRECIA_ANTES) -
                        NVL(i.VL_DEPRECIA_ATUAL_ECO, i.VL_DEPRECIA_ATUAL)) VL_FISCAL,
	   SUM( (NVL(i.VL_DEPRECIA_ANTES,0) - NVL(i.VL_DEPRECIA_ATUAL,0) )  -
                       (NVL(i.VL_DEPRECIA_ANTES_ECO, i.VL_DEPRECIA_ANTES) -
 	   NVL(i.VL_DEPRECIA_ATUAL_ECO, i.VL_DEPRECIA_ATUAL) )) VL_DIFERENCA
	   FROM DBAMV.BENS b
	   JOIN DBAMV.ITDEPRECIA i ON i.cd_bem = b.cd_bem
	   JOIN DBAMV.DEPRECIA d ON d.cd_deprecia = i.cd_deprecia
	   JOIN DBAMV.IOB o ON o.cd_capitulo = b.cd_capitulo
	   JOIN DBAMV.SUB_IOB s ON s.cd_sub_cap = b.cd_sub_cap AND s.CD_CAPITULO = o.CD_CAPITULO
	   WHERE
       d.DT_DEPRECIA between  :CAMPO_DT_INICIO GROUP BY  o.cd_capitulo, o.ds_capitulo
    ORDER BY  o.cd_capitulo
