--R_BENS_BAIXADOS
SELECT * FROM (
Select	'Empresa: ' || me.cd_multi_empresa || ' - ' || me.ds_multi_empresa Empresa,
               me.cd_multi_empresa,
               Mt.Cd_Motivo,
	Mt.Ds_Motivo,
	Be.Dt_Baixa,
	Be.Cd_Bem,
	Be.Ds_Plaqueta,
	Be.Ds_Bem,
	Be.Dt_Compra,
	Be.Vl_Compra,
	Nvl (VL_DEPRECIACAO_HISTORICA + Dbamv.Pkg_Patrimonio.Fn_RetTotalDeprecBem( Be.Cd_Bem, Be.Dt_Baixa ) ,'0'  )  	Total_Depreciacao,
	Be.DEPREC_HIST,
	Be.VL_DEPREC_ACUM_FISCAL,
	Be.VL_DEPREC_ACUM_ECO
From	Dbamv.Mot_Baixa	Mt,
               dbamv.multi_empresas Me,
	dbamv.v_lancamento_patrimonio		Be
Where	Be.cd_motivo = Mt.cd_motivo
  and   Be.tp_lancamento  =  'B'
  And   Trunc(Be.dt_lancamento) Between :CAMPO_PDTINICIO And 
  And   Me.cd_multi_empresa = Be.cd_multi_empresa_bem
Order By Be.Dt_Baixa,
	 Be.Ds_Bem
)
ORDER BY CD_MULTI_EMPRESA,EMPRESA,
CD_MOTIVO,DS_MOTIVO,
DT_BAIXA
