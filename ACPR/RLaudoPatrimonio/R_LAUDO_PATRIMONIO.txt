--R_LAUDO_PATRIMONIO
SELECT
       l.nr_laudo,
       l.cd_laudo,
       l.nr_cgc_cpf,
       l.nm_fornecedor razao_social,
       l.nm_fantasia,
       To_Char(l.dt_inicio,'DD/MM/YYYY') dt_inicio,
       To_Char(l.dt_fim,'DD/MM/YYYY') dt_fim,
       l.ds_observacao,
       i.cd_bem,
       SUBSTR(b.ds_bem, 1,30) ds_bem,
       SUBSTR(b. ds_plaqueta,1, 15) ds_plaqueta,
       i.tx_aplicada,
       i.vl_recuperavel,
       i.nr_vida_util_aplic,
       i.tx_anterior,
       i.vl_residual,
       nr_vida_util_ant
      FROM dbamv.laudo l
      INNER JOIN dbamv.it_laudo i ON l.cd_laudo = i.cd_laudo
      INNER JOIN dbamv.bens b ON i.cd_bem = b.cd_bem
      WHERE
        l.dt_inicio between  :CAMPO_DT_INICIO AND  
       AND ( (i.tx_aplicada > 0 AND :CAMPO_PTP_MOV_LAUDO = 'T')
        OR (i.vl_recuperavel > 0 AND :CAMPO_PTP_MOV_LAUDO = 'V')
        OR (i.tx_aplicada > 0 AND i.vl_recuperavel > 0 AND :CAMPO_PTP_MOV_LAUDO = 'A')
        OR :CAMPO_PTP_MOV_LAUDO = 'G')
GROUP BY l.nr_laudo,
       l.cd_laudo,
       l.nr_cgc_cpf,
       l.nm_fornecedor,
       l.nm_fantasia,
       To_Char(l.dt_inicio,'DD/MM/YYYY'),
       To_Char(l.dt_fim,'DD/MM/YYYY'),
       l.ds_observacao,
       i.cd_bem,
       SUBSTR(b.ds_bem, 1,30),
       SUBSTR(b. ds_plaqueta,1, 15),
       i.tx_aplicada,
       i.vl_recuperavel,
       i.nr_vida_util_aplic,
       i.tx_anterior,
       i.vl_residual,
       nr_vida_util_ant
ORDER BY l.nr_laudo
