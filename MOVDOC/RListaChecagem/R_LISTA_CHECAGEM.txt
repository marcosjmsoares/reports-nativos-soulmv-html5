--R_LISTA_CHECAGEM
SELECT * FROM (
select distinct
       lc.cd_documento_prot
      ,lc.ds_documento_prot
      ,lc.cd_protocolo_doc
      ,lc.cd_setor
      ,lc.dt_envio
      ,lc.nm_usuario_envio
      ,lc.nm_setor
      ,lc.cd_it_protocolo_doc
      ,lc.dt_realizacao
      ,lc.hr_realizacao
      ,lc.cd_mvto
      ,lc.dt_recebimento
      ,lc.nm_usuario_recebimento
      ,lc.dt_devolucao
      ,lc.nm_usuario_devolucao
      ,lc.cd_paciente
      ,lc.tp_atendimento
      ,lc.nm_paciente
      ,lc.cd_gru_doc
      ,lc.ds_gru_doc
      ,lc.nr_seq_emissao
      ,lc.cd_atendimento
from (
select documento_prot.cd_documento_prot cd_documento_prot
      ,documento_prot.ds_documento_prot ds_documento_prot
      ,protocolo_doc.cd_protocolo_doc cd_protocolo_doc
      ,protocolo_doc.cd_setor cd_setor
      ,protocolo_doc.dt_envio dt_envio
      ,protocolo_doc.nm_usuario_envio nm_usuario_envio
      ,setor.nm_setor nm_setor
      ,it_protocolo_doc.cd_it_protocolo_doc cd_it_protocolo_doc
      ,it_protocolo_doc.dt_realizacao dt_realizacao
      ,it_protocolo_doc.hr_realizacao hr_realizacao
      ,it_protocolo_doc.cd_mvto cd_mvto
      ,it_protocolo_doc.dt_recebimento dt_recebimento
      ,it_protocolo_doc.nm_usuario_recebimento nm_usuario_recebimento
      ,it_protocolo_doc.dt_devolucao dt_devolucao
      ,it_protocolo_doc.nm_usuario_devolucao nm_usuario_devolucao
      ,atendime.cd_paciente cd_paciente
      ,atendime.tp_atendimento tp_atendimento
      ,paciente.nm_paciente nm_paciente
      ,gru_doc.cd_gru_doc cd_gru_doc
      ,gru_doc.ds_gru_doc ds_gru_doc
      ,gru_doc.nr_seq_emissao
      ,atendime.cd_atendimento cd_atendimento
from   dbamv.documento_prot documento_prot
      ,dbamv.it_protocolo_doc it_protocolo_doc
      ,dbamv.protocolo_doc protocolo_doc
      ,dbamv.atendime atendime
      ,dbamv.paciente paciente
      ,dbamv.setor setor
      ,dbamv.gru_doc gru_doc
where  documento_prot.cd_documento_prot = it_protocolo_doc.cd_documento_prot
  and  it_protocolo_doc.cd_atendimento = atendime.cd_atendimento
  and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
  and  paciente.cd_paciente = atendime.cd_paciente
  and  it_protocolo_doc.cd_protocolo_doc = protocolo_doc.cd_protocolo_doc
  and  protocolo_doc.cd_setor = setor.cd_setor
  and  it_protocolo_doc.cd_atendimento = :CAMPO_CD_ATENDIMENTO
  and  (it_protocolo_doc.cd_reg_fat is null or
        it_protocolo_doc.cd_reg_fat = :CAMPO_CD_CONTA)
  and  (it_protocolo_doc.cd_reg_amb is null or
        it_protocolo_doc.cd_reg_amb = :CAMPO_CD_CONTA)
  and  gru_doc.cd_gru_doc = documento_prot.cd_gru_doc
union all
select documento_prot.cd_documento_prot cd_documento_prot
      ,documento_prot.ds_documento_prot ds_documento_prot
      ,to_number(null)
      ,to_number(null)
      ,to_date(null)
      ,null
      ,null
      ,to_number(null)
      ,to_date(null)
      ,to_date(null)
      ,to_number(null)
      ,to_date(null)
      ,null
      ,to_date(null)
      ,null
      ,atendime.cd_paciente cd_paciente
      ,atendime.tp_atendimento tp_atendimento
      ,paciente.nm_paciente nm_paciente
      ,gru_doc.cd_gru_doc cd_gru_doc
      ,gru_doc.ds_gru_doc ds_gru_doc
      ,gru_doc.nr_seq_emissao
      ,atendime.cd_atendimento cd_atendimento
from   dbamv.documento_prot documento_prot
      ,dbamv.atendime atendime
      ,dbamv.paciente paciente
      ,dbamv.gru_doc gru_doc
where  documento_prot.cd_documento_prot not in
       (select it_protocolo_doc.cd_documento_prot
        from   dbamv.it_protocolo_doc it_protocolo_doc
        where  it_protocolo_doc.cd_atendimento = :CAMPO_CD_ATENDIMENTO
          and  (it_protocolo_doc.cd_reg_fat is null or
                it_protocolo_doc.cd_reg_fat = :CAMPO_CD_CONTA)
          and  (it_protocolo_doc.cd_reg_amb is null or
                it_protocolo_doc.cd_reg_amb = :CAMPO_CD_CONTA))
  and  atendime.cd_atendimento = :CAMPO_CD_ATENDIMENTO
  and  paciente.cd_paciente = atendime.cd_paciente
  and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
  and  gru_doc.cd_gru_doc = documento_prot.cd_gru_doc
union all
select distinct documento_prot.cd_documento_prot cd_documento_prot
      ,documento_prot.ds_documento_prot ds_documento_prot
      ,to_number(null)
      ,to_number(null)
      ,to_date(null)
      ,null
      ,null
      ,to_number(null)
      ,itreg_fat.dt_lancamento dt_realizacao
      ,itreg_fat.hr_lancamento hr_realizacao
      ,itreg_fat.cd_mvto cd_mvto
      ,to_date(null)
      ,null
      ,to_date(null)
      ,null
      ,atendime.cd_paciente cd_paciente
      ,atendime.tp_atendimento tp_atendimento
      ,paciente.nm_paciente nm_paciente
      ,gru_doc.cd_gru_doc cd_gru_doc
      ,gru_doc.ds_gru_doc ds_gru_doc
      ,gru_doc.nr_seq_emissao
      ,atendime.cd_atendimento cd_atendimento
from   dbamv.atendime atendime
      ,dbamv.reg_fat reg_fat
      ,dbamv.itreg_fat itreg_fat
      ,dbamv.pro_fat pro_fat
      ,dbamv.paciente paciente
      ,dbamv.convenio convenio
      ,dbamv.documento_proc documento_proc
      ,dbamv.documento_prot documento_prot
      ,dbamv.gru_doc gru_doc
      ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And reg_fat.cd_reg_fat = :CAMPO_CD_CONTA
  and  reg_fat.cd_atendimento = :CAMPO_CD_ATENDIMENTO
  and  reg_fat.cd_atendimento = atendime.cd_atendimento
  and  atendime.cd_paciente = paciente.cd_paciente
  and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
  and  reg_fat.cd_convenio = convenio.cd_convenio
  and  pro_fat.cd_pro_fat = itreg_fat.cd_pro_fat
  and  reg_fat.cd_reg_fat = itreg_fat.cd_reg_fat
  and  documento_proc.cd_pro_fat = pro_fat.cd_pro_fat
  and  documento_proc.cd_documento_prot = documento_prot.cd_documento_prot
  and  gru_doc.cd_gru_doc = documento_prot.cd_gru_doc
  and  not exists (select 'x'
                   from dbamv.it_protocolo_doc it_protocolo_doc
                   where  it_protocolo_doc.cd_atendimento = atendime.cd_atendimento
                     and  it_protocolo_doc.cd_reg_fat = reg_fat.cd_reg_fat
                     and  it_protocolo_doc.cd_lancamento_fat = itreg_fat.cd_lancamento
                     and  it_protocolo_doc.cd_documento_prot = documento_proc.cd_documento_prot)
union all
select distinct documento_prot.cd_documento_prot cd_documento_prot
      ,documento_prot.ds_documento_prot ds_documento_prot
      ,to_number(null)
      ,to_number(null)
      ,to_date(null)
      ,null
      ,null
      ,to_number(null)
      ,atendime.dt_atendimento dt_realizacao
      ,itreg_amb.hr_lancamento hr_realizacao
      ,itreg_amb.cd_mvto cd_mvto
      ,to_date(null)
      ,null
      ,to_date(null)
      ,null
      ,atendime.cd_paciente cd_paciente
      ,atendime.tp_atendimento tp_atendimento
      ,paciente.nm_paciente nm_paciente
      ,gru_doc.cd_gru_doc cd_gru_doc
      ,gru_doc.ds_gru_doc ds_gru_doc
      ,gru_doc.nr_seq_emissao
      ,atendime.cd_atendimento cd_atendimento
from   dbamv.atendime atendime
      ,dbamv.reg_amb reg_amb
      ,dbamv.itreg_amb itreg_amb
      ,dbamv.pro_fat pro_fat
      ,dbamv.paciente paciente
      ,dbamv.convenio convenio
      ,dbamv.documento_proc documento_proc
      ,dbamv.documento_prot documento_prot
      ,dbamv.gru_doc gru_doc
      ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And reg_amb.cd_reg_amb = :CAMPO_CD_CONTA
  and  itreg_amb.cd_atendimento = :CAMPO_CD_ATENDIMENTO
  and  itreg_amb.cd_atendimento = atendime.cd_atendimento
  and  atendime.cd_paciente = paciente.cd_paciente
  and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
  and  reg_amb.cd_convenio = convenio.cd_convenio
  and  pro_fat.cd_pro_fat = itreg_amb.cd_pro_fat
  and  reg_amb.cd_reg_amb = itreg_amb.cd_reg_amb
  and  documento_proc.cd_pro_fat = pro_fat.cd_pro_fat
  and  documento_proc.cd_documento_prot = documento_prot.cd_documento_prot
  and  gru_doc.cd_gru_doc = documento_prot.cd_gru_doc
  and  not exists (select 'x'
                   from dbamv.it_protocolo_doc it_protocolo_doc
                   where  it_protocolo_doc.cd_atendimento = atendime.cd_atendimento
                     and  it_protocolo_doc.cd_reg_amb = reg_amb.cd_reg_amb
                     and  it_protocolo_doc.cd_lancamento_amb = itreg_amb.cd_lancamento
                     and  it_protocolo_doc.cd_documento_prot = documento_proc.cd_documento_prot)) lc
)   ORDER BY CD_GRU_DOC, DS_GRU_DOC, cd_documento_prot, dt_realizacao, hr_realizacao, cd_mvto, cd_protocolo_doc
