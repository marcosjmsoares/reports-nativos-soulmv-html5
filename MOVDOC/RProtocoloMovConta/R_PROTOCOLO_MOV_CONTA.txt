--R_PROTOCOLO_MOV_CONTA
SELECT * FROM (
select itmvto_conta.cd_mvto_conta    cd_mvto_conta
     , mvto_conta.cd_setor_origem  cd_setor_origem
     , mvto_conta.cd_setor_destino cd_setor_destino
     , mvto_conta.cd_usuario         cd_usuario
     , mvto_conta.dt_entrada         dt_entrada
     , itmvto_conta.cd_atendimento   cd_atendimento
     , itmvto_conta.cd_reg_fat       cd_reg_fat
     , paciente.nm_paciente          nm_paciente
     , itmvto_conta.cd_pendencia     cd_pendencia
     , pendencia_conta.ds_pendencia  ds_pendencia
     , itmvto_conta.ds_observacao  ds_observacao
  from dbamv.mvto_conta
     , dbamv.itmvto_conta
     , dbamv.atendime
     , dbamv.paciente
     , dbamv.pendencia_conta
 where mvto_conta.cd_mvto_conta    = itmvto_conta.cd_mvto_conta
   and itmvto_conta.cd_atendimento = atendime.cd_atendimento
   and atendime.cd_paciente        = paciente.cd_paciente
   and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
   and itmvto_conta.cd_pendencia   = pendencia_conta.cd_pendencia(+)
:CAMPO_CF_MVTO_CONTA
)
ORDER BY CD_MVTO_CONTA,CD_SETOR_ORIGEM,CD_SETOR_DESTINO,CD_USUARIO,DT_ENTRADA
