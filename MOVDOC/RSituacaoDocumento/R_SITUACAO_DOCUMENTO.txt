      ,it_protocolo_doc.hr_realizacao hr_realizacao
      ,it_protocolo_doc.dt_recebimento dt_recebimento
      ,it_protocolo_doc.dt_devolucao dt_devolucao

      ,dbamv.itreg_amb itreg_amb
      ,dbamv.ori_ate
      ,dbamv.atendime atendime
      ,dbamv.paciente paciente
      ,dbamv.setor setor
      ,dbamv.convenio convenio

  and  reg_amb.cd_convenio   = convenio.cd_convenio
  and  itreg_amb.cd_atendimento = atendime.cd_atendimento
  and  itreg_amb.cd_reg_amb = reg_amb.cd_reg_amb
  and  ori_ate.cd_setor = setor.cd_setor
  and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
  and  atendime.cd_ori_ate = ori_ate.cd_ori_ate
  and  atendime.tp_atendimento in ('U','A','E')
  and  atendime.dt_atendimento < sysdate
  and  :CAMPO_SN_SEM_PROTOCOLO = 'S'
  and  1 = (select count(*) from dbamv.documento_prot where sn_informa_conta = 'S')
  and decode('','T','T', decode(atendime.tp_atendimento, 'A', 'A', 'E', 'A', 'U', 'A', 'H', 'H', 'I', 'H')) = :CAMPO_TP_CONTA
  and  not exists (Select 'x'
                          From dbamv.it_protocolo_doc
                        Where it_protocolo_doc.cd_reg_fat = reg_amb.cd_reg_amb
                          and it_protocolo_doc.cd_lancamento_amb is null
:CAMPO_CF_CD_REMESSA)
AND (:CAMPO_CD_CONTA IS NULL or :CAMPO_CD_CONTA = '%' ) AND (:CAMPO_CD_ATENDE IS NULL or  :CAMPO_CD_ATENDE = '%' )
:CAMPO_CF_PERIODO_2)
order by 6
)
ORDER BY CD_DOCUMENTO_PROT, CD_SETOR, CD_CONVENIO, SN_INFORMA_CONTA,CD_PROTOCOLO_DOC, DT_ENVIO1

      ,dbamv.convenio convenio

  and  itreg_amb.cd_reg_amb = reg_amb.cd_reg_amb
  and  ori_ate.cd_setor = setor.cd_setor
  and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
  and  atendime.cd_ori_ate = ori_ate.cd_ori_ate
  and  atendime.tp_atendimento in ('U','A','E')
  and  atendime.dt_atendimento < sysdate
  and  :CAMPO_SN_SEM_PROTOCOLO = 'S'

  and decode('','T','T', decode(atendime.tp_atendimento, 'A', 'A', 'E', 'A', 'U', 'A', 'H', 'H', 'I', 'H')) = :CAMPO_TP_CONTA
  and  not exists (Select 'x'
                          From dbamv.it_protocolo_doc
                        Where it_protocolo_doc.cd_reg_fat = reg_amb.cd_reg_amb
                          and it_protocolo_doc.cd_lancamento_amb is null
:CAMPO_CF_CD_REMESSA)
AND (:CAMPO_CD_CONTA IS NULL or :CAMPO_CD_CONTA = '%' ) AND (:CAMPO_CD_ATENDE IS NULL or  :CAMPO_CD_ATENDE = '%' )
:CAMPO_CF_PERIODO_2)
order by 6
)
