--R_MVTO_SETOR
SELECT * FROM (
select mvto_conta.cd_mvto_conta    cd_mvto_conta
     , mvto_conta.dt_entrada       dt_entrada
     , contas.cd_reg_fat           cd_reg_fat
     , reg_fat.cd_atendimento      cd_atendimento
     , reg_fat.dt_inicio           dt_inicio
     , reg_fat.dt_final            dt_final
     , paciente.nm_paciente        nm_paciente
     , mvto_conta.cd_setor_destino cd_setor_destino
     , setor.nm_setor              nm_setor
     , convenio.nm_convenio        nm_convenio
     , con_pla.ds_con_pla          ds_con_pla
  from dbamv.mvto_conta
     , (select itmvto_conta.cd_reg_fat         cd_reg_fat
             , max(itmvto_conta.cd_mvto_conta) cd_max_mvto_conta
          from dbamv.itmvto_conta
      group by itmvto_conta.cd_reg_fat) contas
     , dbamv.atendime
     , dbamv.paciente
     , dbamv.reg_fat
     , dbamv.setor
     , dbamv.convenio
     , dbamv.con_pla
     ,Dbamv.Empresa_Con_Pla
Where Empresa_Con_Pla.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And Empresa_Con_Pla.Cd_Convenio = Con_Pla.Cd_Convenio
And  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
And Empresa_Con_Pla.Cd_Con_Pla = Con_Pla.Cd_Con_Pla
And mvto_conta.cd_mvto_conta    = contas.cd_max_mvto_conta
   and mvto_conta.cd_setor_destino = setor.cd_setor
   and contas.cd_reg_fat           = reg_fat.cd_reg_fat
   and reg_fat.cd_convenio         = convenio.cd_convenio
   and reg_fat.cd_con_pla          = con_pla.cd_con_pla
   and reg_fat.cd_convenio         = con_pla.cd_convenio
   and reg_fat.cd_atendimento      = atendime.cd_atendimento
   and atendime.cd_paciente        = paciente.cd_paciente
 order by setor.nm_setor, contas.cd_reg_fat
)
ORDER BY CD_SETOR_DESTINO,NM_SETOR
