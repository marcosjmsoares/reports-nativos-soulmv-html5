--R_DOCUMENTO_REQ
SELECT * FROM (
select documento_prot.cd_documento_prot cd_documento_prot
      ,documento_prot.ds_documento_prot ds_documento_prot
      ,convenio.nm_convenio nm_convenio
      ,atendime.cd_paciente cd_paciente
      ,atendime.tp_atendimento tp_atendimento
      ,atendime.cd_atendimento cd_atendimento
      ,paciente.nm_paciente nm_paciente
      ,to_number(guia.nr_guia) cd_referencia
      ,con_pla.ds_con_pla ds_con_pla
      ,pro_fat.ds_pro_fat ds_pro_fat
      ,pro_fat.cd_pro_fat cd_pro_fat
from   dbamv.documento_prot documento_prot
      ,dbamv.con_pla con_pla
      ,dbamv.convenio convenio
      ,dbamv.atendime atendime
      ,dbamv.paciente paciente
      ,dbamv.guia guia
      ,dbamv.documento_req documento_req
      ,dbamv.pro_fat pro_fat
      ,dbamv.it_guia it_guia
      ,Dbamv.Empresa_Con_Pla
Where Empresa_Con_Pla.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And Empresa_Con_Pla.Cd_Convenio = Con_Pla.Cd_Convenio
And Empresa_Con_Pla.Cd_Con_Pla = Con_Pla.Cd_Con_Pla
And documento_prot.cd_documento_prot = documento_req.cd_documento_prot
  and  (guia.cd_convenio = documento_req.cd_convenio or
       documento_req.cd_convenio is null)
  and  atendime.cd_atendimento = guia.cd_atendimento
  and  convenio.cd_convenio = guia.cd_convenio
  and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
  and  guia.cd_guia = it_guia.cd_guia
  and  it_guia.cd_pro_fat = documento_req.cd_pro_fat
  and  pro_fat.cd_pro_fat = it_guia.cd_pro_fat
  and  con_pla.cd_con_pla = atendime.cd_con_pla
  and  con_pla.cd_convenio = guia.cd_convenio
  and  documento_req.sn_guia = 'S'
  and  paciente.cd_paciente = atendime.cd_paciente
  and  guia.cd_guia = :CAMPO_CD_REFERENCIA
  and  guia.cd_atendimento = :CAMPO_CD_ATENDIMENTO
  and  :CAMPO_TP_RELATORIO = 'G'
union all
select documento_prot.cd_documento_prot cd_documento_prot
      ,documento_prot.ds_documento_prot ds_documento_prot
      ,convenio.nm_convenio nm_convenio
      ,atendime.cd_paciente cd_paciente
      ,atendime.tp_atendimento tp_atendimento
      ,atendime.cd_atendimento cd_atendimento
      ,paciente.nm_paciente nm_paciente
      ,reg_fat.cd_reg_fat cd_referencia
      ,con_pla.ds_con_pla
      ,pro_fat.ds_pro_fat
      ,pro_fat.cd_pro_fat
from   dbamv.documento_prot documento_prot
      ,dbamv.con_pla con_pla
      ,dbamv.convenio convenio
      ,dbamv.atendime atendime
      ,dbamv.paciente paciente
      ,dbamv.reg_fat reg_fat
      ,dbamv.documento_req documento_req
      ,dbamv.pro_fat pro_fat
      ,dbamv.itreg_fat itreg_fat
      ,Dbamv.Empresa_Con_Pla
Where Empresa_Con_Pla.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And Empresa_Con_Pla.Cd_Convenio = Con_Pla.Cd_Convenio
And Empresa_Con_Pla.Cd_Con_Pla = Con_Pla.Cd_Con_Pla
And documento_prot.cd_documento_prot = documento_req.cd_documento_prot
  and  ((reg_fat.cd_convenio = documento_req.cd_convenio or
       documento_req.cd_convenio is null) and
      (reg_fat.cd_con_pla = documento_req.cd_con_pla or
       documento_req.cd_con_pla is null))
  and  atendime.cd_atendimento = reg_fat.cd_atendimento
  and  convenio.cd_convenio = reg_fat.cd_convenio
  and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
  and  reg_fat.cd_reg_fat = itreg_fat.cd_reg_fat
  and  itreg_fat.cd_pro_fat = documento_req.cd_pro_fat
  and  pro_fat.cd_pro_fat = itreg_fat.cd_pro_fat
  and  con_pla.cd_con_pla = reg_fat.cd_con_pla
  and  con_pla.cd_convenio = reg_fat.cd_convenio
  and  documento_req.sn_fatura = 'S'
  and  paciente.cd_paciente = atendime.cd_paciente
  and  reg_fat.cd_reg_fat = :CAMPO_CD_REFERENCIA
  and  reg_fat.cd_atendimento = :CAMPO_CD_ATENDIMENTO
  and  itreg_fat.sn_pertence_pacote = 'N'
  and  nvl(itreg_fat.tp_pagamento,'X') <> 'C'
  and  :CAMPO_TP_RELATORIO = 'F'
union
  select documento_prot.cd_documento_prot cd_documento_prot
      ,documento_prot.ds_documento_prot ds_documento_prot
      ,convenio.nm_convenio nm_convenio
      ,atendime.cd_paciente cd_paciente
      ,atendime.tp_atendimento tp_atendimento
      ,atendime.cd_atendimento cd_atendimento
      ,paciente.nm_paciente nm_paciente
      ,reg_amb.cd_reg_amb cd_referencia
      ,con_pla.ds_con_pla
      ,pro_fat.ds_pro_fat
      ,pro_fat.cd_pro_fat
from   dbamv.documento_prot documento_prot
      ,dbamv.con_pla con_pla
      ,dbamv.convenio convenio
      ,dbamv.atendime atendime
      ,dbamv.paciente paciente
      ,dbamv.reg_amb reg_amb
      ,dbamv.documento_req documento_req
      ,dbamv.pro_fat pro_fat
      ,dbamv.itreg_amb itreg_amb
      ,Dbamv.Empresa_Con_Pla
Where Empresa_Con_Pla.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And Empresa_Con_Pla.Cd_Convenio = Con_Pla.Cd_Convenio
And Empresa_Con_Pla.Cd_Con_Pla = Con_Pla.Cd_Con_Pla
And documento_prot.cd_documento_prot = documento_req.cd_documento_prot
  and  ((reg_amb.cd_convenio = documento_req.cd_convenio or
       documento_req.cd_convenio is null) and
       (itreg_amb.cd_con_pla = documento_req.cd_con_pla or
       documento_req.cd_con_pla is null))
  and  atendime.cd_atendimento = itreg_amb.cd_atendimento
  and  convenio.cd_convenio = reg_amb.cd_convenio
  and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
  and  reg_amb.cd_reg_amb = itreg_amb.cd_reg_amb
  and  itreg_amb.cd_pro_fat = documento_req.cd_pro_fat
  and  pro_fat.cd_pro_fat = itreg_amb.cd_pro_fat
  and  con_pla.cd_con_pla = itreg_amb.cd_con_pla
  and  con_pla.cd_convenio = reg_amb.cd_convenio
  and  documento_req.sn_fatura = 'S'
  and  paciente.cd_paciente = atendime.cd_paciente
  and  reg_amb.cd_reg_amb = :CAMPO_CD_REFERENCIA
  and  itreg_amb.cd_atendimento = :CAMPO_CD_ATENDIMENTO
  and  itreg_amb.sn_pertence_pacote = 'N'
  and  nvl(itreg_amb.tp_pagamento,'X') <> 'C'
  and  :CAMPO_TP_RELATORIO = 'F'
  order by  6,11,1
)
ORDER BY NM_PACIENTE,CD_PACIENTE,TP_ATENDIMENTO,NM_CONVENIO,CD_REFERENCIA,CD_ATENDIMENTO,DS_CON_PLA,
CD_PRO_FAT,DS_PRO_FAT
