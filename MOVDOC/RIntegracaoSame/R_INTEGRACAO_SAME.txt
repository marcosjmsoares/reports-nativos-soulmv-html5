--R_INTEGRACAO_SAME
SELECT * FROM (
SELECT PROTOCOLO_DOC.CD_PROTOCOLO_DOC,
       IT_PROTOCOLO_DOC.CD_DOCUMENTO_PROT,
       DOCUMENTO_PROT.DS_DOCUMENTO_PROT,
       ATENDIME.CD_ATENDIMENTO,
       Decode(ATENDIME.TP_ATENDIMENTO,'I','Internação','A','Ambulatorial','U','Ambulatoria','Ambulatoria') TP_ATENDIMENTO,
       ATENDIME.CD_PACIENTE,
       PACIENTE.NM_PACIENTE,
       IT_PROTOCOLO_DOC.CD_MVTO,
       IT_PROTOCOLO_DOC.DT_REALIZACAO,
       PROTOCOLO_DOC.DT_ENVIO,
       PROTOCOLO_DOC.NM_USUARIO_ENVIO,
       PROTOCOLO_DOC.CD_SETOR_DESTINO,
       PROTOCOLO_DOC.DS_OBSERVACAO
  FROM DBAMV.PACIENTE,
       DBAMV.ATENDIME,
       DBAMV.IT_SAME,
       DBAMV.DOCUMENTO_PROT,
       DBAMV.IT_PROTOCOLO_DOC,
       DBAMV.PROTOCOLO_DOC
 WHERE IT_SAME.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
   AND ATENDIME.CD_PACIENTE   = PACIENTE.CD_PACIENTE
   AND PROTOCOLO_DOC.CD_PROTOCOLO_DOC= IT_PROTOCOLO_DOC.CD_PROTOCOLO_DOC
   AND IT_PROTOCOLO_DOC.CD_ATENDIMENTO = IT_SAME.CD_ATENDIMENTO
   AND IT_PROTOCOLO_DOC.CD_DOCUMENTO_PROT = DOCUMENTO_PROT.CD_DOCUMENTO_PROT
   AND IT_SAME.SN_AUTOMATICO='S'
   AND PROTOCOLO_DOC.DT_ENVIO BETWEEN :CAMPO_DT_INICIO AND :CAMPO_DT_FIM
)
ORDER BY CD_PROTOCOLO_DOC,CD_PACIENTE,NM_PACIENTE,TP_ATENDIMENTO,CD_ATENDIMENTO,DT_ENVIO,CD_SETOR_DESTINO,DS_OBSERVACAO,NM_USUARIO_ENVIO
