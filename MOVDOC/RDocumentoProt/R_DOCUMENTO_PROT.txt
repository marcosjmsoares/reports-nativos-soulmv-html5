--R_DOCUMENTO_PROT
select documento_prot.cd_documento_prot cd_documento_prot
      ,documento_prot.ds_documento_prot ds_documento_prot
      ,documento_prot.cd_gru_doc cd_gru_doc
      ,decode(documento_prot.sn_protocolo,'S','Sim','Não') sn_protocolo
      ,decode(documento_prot.sn_codigo,'S','Sim','Não') sn_codigo
      ,gru_doc.ds_gru_doc ds_gru_doc
     ,DECODE( DBAMV.DOCUMENTO_PROT.SN_REMESSA, 'S', 'Sim', 'Não' )  SN_REMESSA
from   dbamv.documento_prot documento_prot
      ,dbamv.gru_doc gru_doc
where  documento_prot.cd_gru_doc = gru_doc.cd_gru_doc
