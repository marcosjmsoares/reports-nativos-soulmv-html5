      ,To_Char(nvl(it_protocolo_doc.hr_realizacao, protocolo_doc.dt_envio),'hh24')   hr_realizacao

      ,paciente.nm_paciente                            nm_paciente

      ,to_char(atendime.dt_atendimento,'dd/mm/yyyy')||' '||to_char(atendime.hr_atendimento,'hh24') dh_atendimento
from   dbamv.documento_prot documento_prot
      ,dbamv.it_protocolo_doc it_protocolo_doc
      ,dbamv.protocolo_doc protocolo_doc
      ,dbamv.atendime atendime
      ,dbamv.paciente paciente
      ,dbamv.setor setor
      ,dbamv.setor setor_dest
      ,dbamv.convenio
      ,dbamv.motivo_envio_mvto
where  documento_prot.cd_documento_prot = it_protocolo_doc.cd_documento_prot
  and  it_protocolo_doc.cd_atendimento = atendime.cd_atendimento
  and  paciente.cd_paciente = atendime.cd_paciente
  and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
  and  it_protocolo_doc.cd_protocolo_doc = protocolo_doc.cd_protocolo_doc
  and  protocolo_doc.cd_setor = setor.cd_setor
  and  protocolo_doc.cd_setor_destino = setor_dest.cd_setor
  and  protocolo_doc.cd_protocolo_doc   =  :CAMPO_CD_PROTOCOLO
  and  it_protocolo_doc.cd_atendimento is not null
  and  it_protocolo_doc.CD_REMESSA IS NULL
  AND  convenio.cd_convenio(+) = atendime.cd_convenio
  and  it_protocolo_doc.cd_motenvio = motivo_envio_mvto.cd_motenvio(+)
UNION ALL
select documento_prot.cd_documento_prot   cd_documento_prot
      ,SubStr(documento_prot.ds_documento_prot,1,20)||' - Conta: '||nvl(it_protocolo_doc.cd_reg_fat,it_protocolo_doc.cd_reg_amb)       ds_documento_prot
      ,protocolo_doc.cd_protocolo_doc            cd_protocolo_doc
      ,protocolo_doc.cd_setor                           cd_setor
      ,protocolo_doc.dt_envio                           dt_envio

      ,protocolo_doc.CD_ETIQUETA                      cd_etiqueta

      ,protocolo_doc.NR_LACRE2                        nr_lacre2
      ,setor.nm_setor                                         nm_setor
      ,it_protocolo_doc.cd_it_protocolo_doc    cd_it_protocolo_doc
      ,it_protocolo_doc.cd_atendimento           cd_atendimento
      ,it_protocolo_doc.cd_remessa                 cd_remessa
      ,To_Char(nvl(it_protocolo_doc.dt_realizacao, protocolo_doc.dt_envio),'dd/mm/yyyy') dt_realizacao
      ,To_Char(nvl(it_protocolo_doc.hr_realizacao, protocolo_doc.dt_envio),'hh24')   hr_realizacao

      ,'  '                   nm_paciente
      ,'  '                   nm_convenio

from   dbamv.documento_prot documento_prot
      ,dbamv.it_protocolo_doc it_protocolo_doc
      ,dbamv.protocolo_doc protocolo_doc
      ,dbamv.setor setor
      ,dbamv.setor setor_dest
      ,dbamv.motivo_envio_mvto
where  documento_prot.cd_documento_prot = it_protocolo_doc.cd_documento_prot
  and  it_protocolo_doc.cd_protocolo_doc = protocolo_doc.cd_protocolo_doc
  and  protocolo_doc.cd_setor = setor.cd_setor
  and  protocolo_doc.cd_setor_destino = setor_dest.cd_setor
  and  protocolo_doc.cd_protocolo_doc   =  :CAMPO_CD_PROTOCOLO
  and  it_protocolo_doc.cd_atendimento is null
  and  it_protocolo_doc.CD_REMESSA IS NULL
  and  it_protocolo_doc.cd_motenvio = motivo_envio_mvto.cd_motenvio(+)

      ,documento_prot.ds_documento_prot       ds_documento_prot
      ,protocolo_doc.cd_protocolo_doc             cd_protocolo_doc
      ,protocolo_doc.cd_setor                            cd_setor
      ,protocolo_doc.dt_envio                            dt_envio
      ,protocolo_doc.nm_usuario_envio             nm_usuario_envio
      ,protocolo_doc.ds_observacao
      ,protocolo_doc.CD_ETIQUETA                      cd_etiqueta
      ,protocolo_doc.NR_LACRE1                        nr_lacre1
      ,protocolo_doc.NR_LACRE2                        nr_lacre2
      ,setor.nm_setor                                          nm_setor
      ,it_protocolo_doc.cd_it_protocolo_doc     cd_it_protocolo_doc
      ,it_protocolo_doc.cd_atendimento            cd_atendimento
      ,it_protocolo_doc.cd_remessa                  cd_remessa
      ,To_Char(nvl(it_protocolo_doc.dt_realizacao, protocolo_doc.dt_envio),'dd/mm/yyyy') dt_realizacao
      ,To_Char(nvl(it_protocolo_doc.hr_realizacao, protocolo_doc.dt_envio),'hh24')   hr_realizacao

      ,convenio.cd_convenio                            cd_convenio
      ,''                                                               tp_atendimento
      ,''                                                               nm_paciente
      ,convenio.nm_convenio                           nm_convenio
      ,it_protocolo_doc.nr_seq_emissao          nr_seq_emissao 
      ,setor_dest.nm_setor nm_setor_destino
      ,it_protocolo_doc.cd_motenvio||'-'||motivo_envio_mvto.ds_motenvio mot_envio
      ,'' dh_atendimento
from   dbamv.documento_prot  documento_prot
      ,dbamv.it_protocolo_doc it_protocolo_doc
      ,dbamv.protocolo_doc protocolo_doc
      ,dbamv.setor setor
      ,DBAMV.convenio
      ,DBAMV.remessa_fatura
      ,DBAMV.fatura
      ,DBAMV.empresa_convenio
      ,dbamv.setor setor_dest
      ,dbamv.motivo_envio_mvto
where       protocolo_doc.cd_protocolo_doc    = :CAMPO_CD_PROTOCOLO
  and it_protocolo_doc.cd_atendimento is null
  and it_protocolo_doc.cd_remessa is not null
  and it_protocolo_doc.cd_protocolo_doc   = protocolo_doc.cd_protocolo_doc
  and documento_prot.cd_documento_prot = it_protocolo_doc.cd_documento_prot
  and protocolo_doc.cd_setor                      = setor.cd_setor
  and  protocolo_doc.cd_setor_destino = setor_dest.cd_setor
  and remessa_fatura.cd_remessa                = it_protocolo_doc.cd_remessa
  and  fatura.cd_fatura		     = remessa_fatura.cd_fatura
  and convenio.cd_convenio	   	     = fatura.cd_convenio
  and empresa_convenio.cd_convenio	     = convenio.cd_convenio
  and empresa_convenio.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
  and fatura.cd_multi_empresa 	     = empresa_convenio.cd_multi_empresa
  and  it_protocolo_doc.cd_motenvio = motivo_envio_mvto.cd_motenvio(+)
order by cd_it_protocolo_doc
        ,cd_protocolo_doc
        ,cd_atendimento
        ,cd_documento_prot
        ,nm_convenio
        ,cd_remessa
)
ORDER BY CD_PROTOCOLO_DOC,CD_IT_PROTOCOLO_DOC,CD_SETOR,DT_ENVIO,NM_USUARIO_ENVIO,NM_SETOR,DS_OBSERVACAO,
NR_SEQ_EMISSAO,CD_ATENDIMENTO,NM_PACIENTE,CD_PACIENTE,TP_ATENDIMENTO,CD_REMESSA,CD_CONVENIO,NM_CONVENIO
