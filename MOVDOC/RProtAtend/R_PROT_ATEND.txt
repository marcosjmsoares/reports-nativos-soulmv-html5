select sysdate from dual

--R_PROT_ATEND_RG_PROT_REMESSA1_Q_PROTOCOLO_REMESSA
SELECT * FROM (
select documento_prot.cd_documento_prot cd_documento_prot
      ,documento_prot.ds_documento_prot ds_documento_prot
      ,protocolo_doc.cd_protocolo_doc cd_protocolo_doc
      ,protocolo_doc.cd_setor cd_setor
      ,protocolo_doc.dt_envio dt_envio
      ,protocolo_doc.nm_usuario_envio nm_usuario_envio
      ,setor.nm_setor nm_setor
      ,it_protocolo_doc.cd_it_protocolo_doc cd_it_protocolo_doc
      ,it_protocolo_doc.dt_realizacao dt_realizacao
      ,it_protocolo_doc.hr_realizacao hr_realizacao
      ,it_protocolo_doc.cd_mvto cd_mvto
      ,it_protocolo_doc.dt_recebimento dt_recebimento
      ,it_protocolo_doc.nm_usuario_recebimento nm_usuario_recebimento
      ,it_protocolo_doc.dt_devolucao dt_devolucao
      ,it_protocolo_doc.nm_usuario_devolucao nm_usuario_devolucao
      ,it_protocolo_doc.cd_remessa cd_remessa
  from dbamv.documento_prot documento_prot
      ,dbamv.it_protocolo_doc it_protocolo_doc
      ,dbamv.protocolo_doc protocolo_doc
      ,dbamv.setor setor
 where documento_prot.cd_documento_prot = it_protocolo_doc.cd_documento_prot
   and it_protocolo_doc.cd_protocolo_doc = protocolo_doc.cd_protocolo_doc
   and protocolo_doc.cd_setor = setor.cd_setor
   and protocolo_doc.dt_envio >= sysdate - 180
   and it_protocolo_doc.cd_atendimento is null
   and it_protocolo_doc.cd_remessa is not null
order by cd_remessa
        ,cd_protocolo_doc
        ,cd_documento_prot
)
ORDER BY CD_REMESSA,
CD_PROTOCOLO_DOC,CD_SETOR,NM_SETOR,DT_ENVIO,NM_USUARIO_ENVIO

--R_PROT_ATEND_R_G_DOCUMENTO_PROT3_Q_PROTOCOLO
SELECT * FROM (
select documento_prot.cd_documento_prot cd_documento_prot
      ,documento_prot.ds_documento_prot ds_documento_prot
      ,protocolo_doc.cd_protocolo_doc cd_protocolo_doc
      ,protocolo_doc.cd_setor cd_setor
      ,protocolo_doc.dt_envio dt_envio
      ,protocolo_doc.nm_usuario_envio nm_usuario_envio
      ,setor.nm_setor nm_setor
      ,it_protocolo_doc.cd_it_protocolo_doc cd_it_protocolo_doc
      ,it_protocolo_doc.cd_atendimento cd_atendimento
      ,To_Char(it_protocolo_doc.dt_realizacao,'dd/mm/yyyy') dt_realizacao
      ,To_Char(it_protocolo_doc.hr_realizacao,'hh24') hr_realizacao
      ,it_protocolo_doc.cd_mvto cd_mvto
      ,it_protocolo_doc.dt_recebimento dt_recebimento
      ,it_protocolo_doc.nm_usuario_recebimento nm_usuario_recebimento
      ,it_protocolo_doc.dt_devolucao dt_devolucao
      ,it_protocolo_doc.nm_usuario_devolucao nm_usuario_devolucao
      ,atendime.cd_paciente cd_paciente
      ,paciente.nm_paciente nm_paciente
      ,atendime.tp_atendimento tp_atendimento
  from dbamv.documento_prot documento_prot
      ,dbamv.it_protocolo_doc it_protocolo_doc
      ,dbamv.protocolo_doc protocolo_doc
      ,dbamv.setor setor
      ,dbamv.atendime atendime
      ,dbamv.paciente paciente
 where documento_prot.cd_documento_prot = it_protocolo_doc.cd_documento_prot
   and it_protocolo_doc.cd_protocolo_doc = protocolo_doc.cd_protocolo_doc
   and protocolo_doc.cd_setor = setor.cd_setor
   and protocolo_doc.dt_envio >= sysdate - 180
   and it_protocolo_doc.cd_atendimento IS NOT null
   and it_protocolo_doc.cd_remessa is NULL
   AND it_protocolo_doc.cd_atendimento = atendime.cd_atendimento
   and  paciente.cd_paciente = atendime.cd_paciente
   and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   and setor.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
order by cd_remessa
        ,cd_protocolo_doc
        ,cd_documento_prot
)
ORDER BY cd_atendimento,
CD_PROTOCOLO_DOC,CD_SETOR,NM_SETOR,DT_ENVIO,NM_USUARIO_ENVIO
