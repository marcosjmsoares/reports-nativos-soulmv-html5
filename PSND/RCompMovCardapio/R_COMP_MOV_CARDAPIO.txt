--R_COMP_MOV_CARDAPIO
SELECT * FROM
  (SELECT mc.dt_mov_cardapio,
          tp_ref.ds_tipo_refeicao,
          tp_dieta.ds_tipo_dieta,
          mc.cd_mov_cardapio,
          mc.cd_usuario_entrega,
          mc.dt_entrega,
          ti.ds_tipo_ocorrencia||' - '||od.ds_ocorrencia ocorrencia,
          DECODE (mc.tp_cardapio,
                  'P', 'PACIENTE',
                  'A', 'ACOMPANHANTE',
                  ' ') tp_cardapio,
          mc.cd_atendimento,
          imc.cd_opcao,
          opc.ds_opcao,
          imc.qt_cardapio,
          leito.ds_resumo,
          unid_int.ds_unid_int,
          pac.dt_nascimento DT_NASCIMENTO,
          fn_idade (dt_nascimento) || ' ' || fn_idade (dt_nascimento, 'A') idade,
          mc.cd_usuario,
          mc.cd_setor,
          substr(setor.nm_setor, 0, 22) nm_setor,
          SUBSTR (nm_convenio, 1, 60) convenio,
          ds_con_pla ds_plano,
          mc.ds_observacao,
          mc.obs_nutricao,
          mc.cd_tipo_dieta,
          mc.cd_tipo_refeicao,
          pt.ds_prato,
          pt.cd_prato
   FROM dbamv.mov_cardapio mc,
        dbamv.tipo_refeicao tp_ref,
        dbamv.tipo_dieta tp_dieta,
        dbamv.ocorrencia_dieta od,
        dbamv.tipo_ocorrencia ti,
        dbamv.itmov_cardapio imc,
        dbamv.itmov_lactario iml,
        dbamv.opcao_cardapio opc,
        dbamv.atendime aten,
        dbamv.paciente pac,
        dbamv.convenio con,
        dbamv.con_pla pla,
        dbamv.leito,
        dbamv.unid_int,
        dbamv.setor,
        dbamv.empresa_con_pla,
        dbamv.prato pt
   WHERE empresa_con_pla.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
     AND empresa_con_pla.cd_convenio = pla.cd_convenio
     AND empresa_con_pla.cd_con_pla = pla.cd_con_pla
     AND mc.tp_cardapio IN ('P', 'A')
     AND aten.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
     AND mc.cd_mov_cardapio = imc.cd_mov_cardapio(+)
     AND mc.cd_mov_cardapio = iml.cd_mov_cardapio(+)
     AND mc.cd_tipo_refeicao = tp_ref.cd_tipo_refeicao
     AND aten.cd_convenio = con.cd_convenio(+)
     AND aten.cd_paciente = pac.cd_paciente(+)
     AND aten.cd_convenio = pla.cd_convenio(+)
     AND mc.cd_mov_cardapio = od.cd_mov_cardapio(+)
     AND ti.cd_tipo_ocorrencia(+) = od.cd_tipo_ocorrencia
     AND aten.cd_con_pla = pla.cd_con_pla(+)
     AND mc.cd_tipo_dieta = tp_dieta.cd_tipo_dieta(+)
     AND imc.cd_opcao = opc.cd_opcao(+)
     AND mc.cd_atendimento = aten.cd_atendimento(+)
     AND aten.cd_leito = leito.cd_leito(+)
     AND leito.cd_unid_int = unid_int.cd_unid_int(+)
     AND mc.cd_setor = setor.cd_setor(+)   
   UNION 
   SELECT mc.dt_mov_cardapio,
          tp_ref.ds_tipo_refeicao,
          tp_dieta.ds_tipo_dieta,
          mc.cd_mov_cardapio,
          mc.cd_usuario_entrega,
          mc.dt_entrega,
          ti.ds_tipo_ocorrencia||' - '||od.ds_ocorrencia ocorrencia,
          DECODE (mc.tp_cardapio,
                  'F', 'FUNCIONARIO',
                  'B', 'BANDEJAO',
                  'S', 'SETOR',
                  'M', 'MEDICO',
                  ' ') tp_cardapio,
          mc.cd_atendimento,
          imc.cd_opcao,
          opc.ds_opcao,
          imc.qt_cardapio,
          leito.ds_resumo,
          unid_int.ds_unid_int,
          pac.dt_nascimento DT_NASCIMENTO,
          fn_idade (dt_nascimento) || ' ' || fn_idade (dt_nascimento, 'A') idade,
          mc.cd_usuario,
          mc.cd_setor,
          substr(setor.nm_setor, 0, 22) nm_setor,
          SUBSTR (nm_convenio, 1, 60) convenio,
          ds_con_pla ds_plano,
          mc.ds_observacao,
          mc.obs_nutricao,
          mc.cd_tipo_dieta,
          mc.cd_tipo_refeicao,
          pt.ds_prato,
          pt.cd_prato
   FROM dbamv.mov_cardapio mc,
        dbamv.tipo_refeicao tp_ref,
        dbamv.tipo_dieta tp_dieta,
        dbamv.ocorrencia_dieta od,
        dbamv.tipo_ocorrencia ti,
        dbamv.itmov_cardapio imc,
        dbamv.itmov_lactario iml,
        dbamv.opcao_cardapio opc,
        dbamv.atendime aten,
        dbamv.paciente pac,
        dbamv.convenio con,
        dbamv.con_pla pla,
        dbamv.leito,
        dbamv.unid_int,
        dbamv.setor,
        dbamv.prato pt
   WHERE mc.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
     AND mc.tp_cardapio NOT IN ('P', 'A')
     AND mc.cd_mov_cardapio = imc.cd_mov_cardapio(+)
     AND mc.cd_mov_cardapio = iml.cd_mov_cardapio(+)
     AND mc.cd_mov_cardapio = od.cd_mov_cardapio(+)
     AND ti.cd_tipo_ocorrencia(+) = od.cd_tipo_ocorrencia
     AND mc.cd_tipo_refeicao = tp_ref.cd_tipo_refeicao
     AND aten.cd_convenio = con.cd_convenio(+)
     AND aten.cd_paciente = pac.cd_paciente(+)
     AND aten.cd_convenio = pla.cd_convenio(+)
     AND aten.cd_con_pla = pla.cd_con_pla(+)
     AND mc.cd_tipo_dieta = tp_dieta.cd_tipo_dieta(+)
     AND imc.cd_opcao = opc.cd_opcao(+)
     AND mc.cd_atendimento = aten.cd_atendimento(+) 
     AND aten.cd_leito = leito.cd_leito(+)
     AND leito.cd_unid_int = unid_int.cd_unid_int(+)
     AND mc.cd_setor = setor.cd_setor(+) 
   ORDER BY dt_mov_cardapio,
            cd_tipo_dieta,
            cd_tipo_refeicao)
ORDER BY DT_MOV_CARDAPIO,
         DS_TIPO_REFEICAO,
         DS_TIPO_DIETA,
         CD_ATENDIMENTO,
         TP_CARDAPIO,
         DS_UNID_INT,
         DS_RESUMO,
         CD_SETOR,
         NM_SETOR,
         CD_USUARIO,
         OBS_NUTRICAO,
         DT_NASCIMENTO,
         IDADE,
         DS_PLANO,
         CONVENIO,
         DS_OBSERVACAO
