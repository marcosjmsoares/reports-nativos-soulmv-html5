SELECT tipo_refeicao.ds_tipo_refeicao,
       atendime.cd_atendimento,
       paciente.cd_paciente,
       mov_cardapio.cd_mov_cardapio,
       dbamv.fnc_get_pac_nome(atendime.cd_atendimento) nm_paciente,
       unid_int.ds_unid_int,
       leito.ds_resumo || ' / ' || leito.ds_enfermaria ds_resumo,
       convenio.nm_convenio,
       RTRIM(DBAMV.PACK_INTERNAMENTO.RETORNA_IDADE(PACIENTE.DT_NASCIMENTO,SysDate),' ') idade,
       tipo_dieta.cd_tipo_dieta,
       'Geral' ds_tipo_dieta,
       mov_cardapio.obs_nutricao ds_observacao,
       mov_cardapio.dt_mov_cardapio
FROM   dbamv.atendime,
       dbamv.paciente,
       dbamv.mov_cardapio,
       dbamv.leito,
       dbamv.unid_int,
       dbamv.convenio,
       dbamv.tipo_dieta,
       dbamv.tipo_refeicao,
       dbamv.copa
WHERE  atendime.tp_atendimento = 'I'
AND    atendime.dt_alta IS NULL
AND  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
AND    atendime.cd_paciente = paciente.cd_paciente
AND    atendime.cd_atendimento = mov_cardapio.cd_atendimento
AND    mov_cardapio.cd_tipo_refeicao = tipo_refeicao.cd_tipo_refeicao
AND    atendime.cd_leito = leito.cd_leito
AND    leito.cd_unid_int = unid_int.cd_unid_int
AND    atendime.cd_convenio = convenio.cd_convenio
AND    mov_cardapio.cd_tipo_dieta = tipo_dieta.cd_tipo_dieta(+)
AND   MOV_CARDAPIO.TP_MOVIMENTACAO = 'R'
AND   tipo_refeicao.tp_status_refeicao = 'R'
AND    leito.cd_copa = copa.cd_copa(+)
AND   mov_cardapio.tp_cardapio = 'A'
AND   atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
