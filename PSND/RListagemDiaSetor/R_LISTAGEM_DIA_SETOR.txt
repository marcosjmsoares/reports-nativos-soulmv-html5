--R_LISTAGEM_DIA_SETOR
SELECT * FROM (
SELECT	MV.CD_SETOR 		CD_SETOR
         	,SETOR.NM_SETOR 	NM_SETOR
         	,C_C.CD_CLAS_CARDAPIO 	CD_CLAS_CARDAPIO
         	,C_C.DS_CLAS_CARDAPIO 	DS_CLAS_CARDAPIO
         	,IMV.CD_OPCAO 		CD_OPCAO
         	,OPC.DS_OPCAO 		DS_OPCAO
         	,MV.CD_MOV_CARDAPIO 	CD_MOV_CARDAPIO
         	,IMV.CD_ITMOV_CARDAPIO CD_ITMOV_CARDAPIO
         	,IMV.QT_CARDAPIO 	QT_CARDAPIO
         	,TP_D.CD_TIPO_DIETA 	CD_TIPO_DIETA
         	,TP_D.DS_TIPO_DIETA 	DS_TIPO_DIETA
FROM      DBAMV.MOV_CARDAPIO    		MV
         	,DBAMV.ITMOV_CARDAPIO  	IMV
         	,DBAMV.TIPO_DIETA      		TP_D
         	,DBAMV.OPCAO_CARDAPIO  	OPC
         	,DBAMV.CLAS_CARDAPIO   		C_C
         	,DBAMV.SETOR			SETOR
WHERE TRUNC(MV.DT_MOV_CARDAPIO)   BETWEEN   :CAMPO_DT_INIC  AND  :CAMPO_DT_FIM+.99999
 AND MV.CD_SETOR                    = SETOR.CD_SETOR
      AND MV.CD_MOV_CARDAPIO    = IMV.CD_MOV_CARDAPIO
      AND IMV.CD_OPCAO                   = OPC.CD_OPCAO
      AND MV.CD_TIPO_DIETA           = TP_D.CD_TIPO_DIETA(+)
      AND OPC.CD_CLAS_CARDAPIO = C_C.CD_CLAS_CARDAPIO
      AND MV.CD_MULTI_EMPRESA = NVL(DBAMV.PKG_MV2000.LE_EMPRESA, MV.CD_MULTI_EMPRESA)
ORDER BY
	SETOR.NM_SETOR
	,C_C.DS_CLAS_CARDAPIO
	,IMV.CD_OPCAO
	,OPC.DS_OPCAO
)
ORDER BY
	NM_SETOR
	,DS_CLAS_CARDAPIO
	,CD_OPCAO
	,DS_OPCAO
