--R_OPCAO_CARDAPIO
SELECT   OPC.CD_OPCAO
        ,OPC.DS_OPCAO
        ,UNI.DS_UNIDADE
        ,CD_CLAS_CARDAPIO
        ,VL_PROTEINAS
        ,VL_CALORIAS
        ,VL_SAIS
FROM     DBAMV.OPCAO_CARDAPIO OPC
        ,DBAMV.UNIDADE        UNI
WHERE    OPC.CD_UNIDADE   = UNI.CD_UNIDADE
AND OPC.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
