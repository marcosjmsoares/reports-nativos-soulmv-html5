--R_MOV_CARDAPIO_PROD
SELECT * FROM (
select mov_cardapio.cd_mov_cardapio         cd_mov_cardapio
     , mov_cardapio.dt_mov_cardapio         dt_mov_cardapio
     , mov_cardapio.tp_cardapio             tp_cardapio
     , decode(mov_cardapio.tp_cardapio, 'P', 'Paciente',
                                        'A', 'Acompanhante',
                                        'F', 'Funcionário',
                                        'B', 'Bandejão',
                                        'S', 'Setor',
                                        'M', 'Médico') nm_tp_cardapio
     , mov_cardapio.cd_atendimento          cd_atendimento
     , itmov_cardapio.cd_opcao              cd_opcao
     , opcao_cardapio.ds_opcao              ds_opcao
     , itprato_opcao_cardapio.cd_prato      cd_prato
     , prato.ds_prato                       ds_prato
     , ITMOV_FICHA_CARDAPIO.cd_produto          cd_produto
     , produto.ds_produto                   ds_produto
     , ITMOV_FICHA_CARDAPIO.cd_uni_pro          cd_uni_pro
     , itmov_cardapio.qt_cardapio           qt_cardapio
     , ITMOV_FICHA_CARDAPIO.QT_FICHA_DE_PREPARO QT_FICHA_DE_PREPARO
     , uni_pro.ds_unidade                   ds_unidade
     , decode(mov_cardapio.tp_cardapio, 'F','Funcionário: '|| funcionario.nm_func,'M','Médico','Atendimento: '|| mov_cardapio.cd_atendimento) ds_consumo
  from dbamv.mov_cardapio
     , dbamv.itmov_cardapio
     , dbamv.opcao_cardapio
     , dbamv.itprato_opcao_cardapio
     , dbamv.prato
     , dbamv.ITMOV_FICHA_CARDAPIO
     , dbamv.produto
     , dbamv.uni_pro
     , dbamv.unidade
     , (Select distinct(cd_func) cd_func, nm_func From dbamv.funcionario) funcionario
 where mov_cardapio.cd_mov_cardapio = itmov_cardapio.cd_mov_cardapio(+)
 and funcionario.cd_func (+) = mov_cardapio.cd_func
   and itprato_opcao_cardapio.cd_mov_cardapio = mov_cardapio.cd_mov_cardapio
   and  mov_cardapio.cd_mov_cardapio = ITMOV_FICHA_CARDAPIO.cd_mov_cardapio
   and itmov_cardapio.cd_opcao = opcao_cardapio.cd_opcao(+)
   and opcao_cardapio.cd_opcao = itprato_opcao_cardapio.cd_opcao(+)
   and itprato_opcao_cardapio.cd_prato = prato.cd_prato(+)
   and prato.cd_prato = ITMOV_FICHA_CARDAPIO.cd_prato(+)
   and ITMOV_FICHA_CARDAPIO.cd_produto = produto.cd_produto(+)
   and ITMOV_FICHA_CARDAPIO.cd_uni_pro = uni_pro.cd_uni_pro(+)
   and uni_pro.cd_unidade = unidade.cd_unidade(+)
   and mov_cardapio.cd_multi_empresa = nvl(dbamv.pkg_mv2000.le_empresa,mov_cardapio.cd_multi_empresa)
 union
    select  mov_cardapio.cd_mov_cardapio         cd_mov_cardapio
         , mov_cardapio.dt_mov_cardapio         dt_mov_cardapio
         , mov_cardapio.tp_cardapio             tp_cardapio
         , decode(mov_cardapio.tp_cardapio, 'P', 'Paciente',
                                            'A', 'Acompanhante',
                                            'F', 'Funcionário',
                                            'B', 'Bandejão',
                                            'S', 'Setor',
                                            'M', 'Médico') nm_tp_cardapio
         , mov_cardapio.cd_atendimento          cd_atendimento
         , itmov_cardapio.cd_opcao              cd_opcao
         , opcao_cardapio.ds_opcao              ds_opcao
         , itprato_opcao_cardapio.cd_prato      cd_prato
         , prato.ds_prato                       ds_prato
         , ITMOV_FICHA_CARDAPIO.cd_produto          cd_produto
         , produto.ds_produto                   ds_produto
         , ITMOV_FICHA_CARDAPIO.cd_uni_pro          cd_uni_pro
         , itmov_cardapio.qt_cardapio           qt_cardapio
         , ITMOV_FICHA_CARDAPIO.QT_FICHA_DE_PREPARO QT_FICHA_DE_PREPARO
         , uni_pro.ds_unidade                   ds_unidade
	 , decode(mov_cardapio.tp_cardapio, 'F','Funcionário: '|| funcionario.nm_func,'M','Médico','Atendimento: '|| mov_cardapio.cd_atendimento) ds_consumo
  from dbamv.mov_cardapio
     , dbamv.itmov_cardapio
     , dbamv.opcao_cardapio
     , dbamv.itprato_opcao_cardapio
     , dbamv.prato
     , dbamv.ITMOV_FICHA_CARDAPIO
     , dbamv.produto
     , dbamv.uni_pro
     , dbamv.unidade
     , dbamv.atendime
     , (Select distinct(cd_func) cd_func, nm_func From dbamv.funcionario) funcionario
 where mov_cardapio.cd_mov_cardapio = itmov_cardapio.cd_mov_cardapio(+)
and  atendime.Cd_Multi_Empresa = nvl(dbamv.pkg_mv2000.le_empresa,atendime.cd_multi_empresa)
and funcionario.cd_func (+) = mov_cardapio.cd_func
   and itprato_opcao_cardapio.cd_mov_cardapio = mov_cardapio.cd_mov_cardapio
   and  mov_cardapio.cd_mov_cardapio = ITMOV_FICHA_CARDAPIO.cd_mov_cardapio
   and itmov_cardapio.cd_opcao = opcao_cardapio.cd_opcao(+)
   and opcao_cardapio.cd_opcao = itprato_opcao_cardapio.cd_opcao(+)
   and itprato_opcao_cardapio.cd_prato = prato.cd_prato(+)
   and prato.cd_prato = ITMOV_FICHA_CARDAPIO.cd_prato(+)
   and ITMOV_FICHA_CARDAPIO.cd_produto = produto.cd_produto(+)
   and ITMOV_FICHA_CARDAPIO.cd_uni_pro = uni_pro.cd_uni_pro(+)
   and uni_pro.cd_unidade = unidade.cd_unidade(+)
   and atendime.tp_atendimento <> 'I'
   and atendime.cd_atendimento = mov_cardapio.cd_atendimento
   AND   MOV_CARDAPIO.CD_MOV_CARDAPIO = (SELECT MAX(MOV_CARD.CD_MOV_CARDAPIO)
                                      FROM   DBAMV.MOV_CARDAPIO   MOV_CARD
                                      WHERE  MOV_CARD.CD_TIPO_REFEICAO = MOV_CARDAPIO.CD_TIPO_REFEICAO
                                      AND    MOV_CARD.CD_ATENDIMENTO = MOV_CARDAPIO.CD_ATENDIMENTO
                                      AND    MOV_CARD.CD_TIPO_REFEICAO = MOV_CARDAPIO.CD_TIPO_REFEICAO
                                      AND    TRUNC(MOV_CARD.DT_MOV_CARDAPIO) BETWEEN :CAMPO_DT_INI AND :CAMPO_DT_FIM)
   and mov_cardapio.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
)
ORDER BY CD_MOV_CARDAPIO,DT_MOV_CARDAPIO,CD_ATENDIMENTO,TP_CARDAPIO,NM_TP_CARDAPIO,
CD_OPCAO,DS_OPCAO,CD_PRATO,DS_PRATO,QT_CARDAPIO
