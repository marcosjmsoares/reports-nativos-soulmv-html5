--R_COMP_OPCAO_CARDAPIO
SELECT * FROM (
SELECT   OPC.CD_OPCAO
        ,OPC.DS_OPCAO
        ,POPC.CD_PRATO
        ,P.DS_PRATO
FROM     DBAMV.OPCAO_CARDAPIO OPC
        ,DBAMV.PRATO_OPCAO_CARDAPIO POPC
        ,DBAMV.PRATO  P
WHERE   OPC.CD_OPCAO   = POPC.CD_OPCAO
AND       POPC.CD_PRATO  = P.CD_PRATO
AND OPC.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
ORDER BY 1, 3
)
ORDER BY CD_OPCAO,DS_OPCAO
