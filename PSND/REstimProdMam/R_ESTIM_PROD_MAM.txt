--R_ESTIM_PROD_MAM
SELECT * FROM (
SELECT tipo_refeicao.cd_tipo_refeicao,
       tipo_refeicao.ds_tipo_refeicao,
       prato.cd_prato,
       prato.ds_prato,
       Sysdate data,
       ficha_de_preparo.cd_produto,
       produto.ds_produto || ' (' || uni_pro.cd_unidade || ')' produto,
       SUM((ficha_de_preparo.qt_ficha_de_preparo * uni_pro.vl_fator * ficha_nutricao_lactario.nr_volume) / 1000),
       SUM(ficha_nutricao_lactario.nr_volume) nr_volume
FROM   dbamv.tipo_refeicao,
       dbamv.ficha_nutricao_lactario,
       dbamv.prato,
       dbamv.ficha_de_preparo,
       dbamv.produto,
       dbamv.uni_pro,
       dbamv.atendime
WHERE  atendime.cd_atendimento = ficha_nutricao_lactario.cd_atendimento
AND    atendime.tp_atendimento = 'I'
AND    atendime.dt_alta IS NULL
AND   atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
and     tipo_refeicao.cd_tipo_refeicao = ficha_nutricao_lactario.cd_tipo_refeicao
AND    ficha_nutricao_lactario.cd_prato = prato.cd_prato
AND    prato.cd_prato = ficha_de_preparo.cd_prato
AND    ficha_de_preparo.cd_produto = produto.cd_produto
AND    ficha_de_preparo.cd_uni_pro = uni_pro.cd_uni_pro
and     atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
And Exists ( Select Empresa_Produto.Cd_Produto
                     From Dbamv.Empresa_Produto
                   Where Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
                        and Empresa_Produto.Cd_Produto = Produto.Cd_Produto )
GROUP BY tipo_refeicao.cd_tipo_refeicao,
       tipo_refeicao.ds_tipo_refeicao,
       prato.cd_prato,
       prato.ds_prato,
       Sysdate,
       ficha_de_preparo.cd_produto,
       produto.ds_produto,
       uni_pro.cd_unidade,
       ficha_nutricao_lactario.nr_volume
)
ORDER BY CD_TIPO_REFEICAO,DS_TIPO_REFEICAO,
DS_PRATO,NR_VOLUME,
PRODUTO
