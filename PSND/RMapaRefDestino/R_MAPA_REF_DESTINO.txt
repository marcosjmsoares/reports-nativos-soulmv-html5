--R_MAPA_REF_DESTINO
SELECT * FROM (
                    SELECT  CD_SETOR,
                            NM_SETOR,
                            DIA,
                            TP_REF,
                            SUM(QTDADE) QTDADE
                    FROM (
                         SELECT  MC.CD_SETOR                                                        CD_SETOR
                                ,SETOR.NM_SETOR                                                     NM_SETOR
                                ,TO_NUMBER(TO_CHAR( MC.DT_MOV_CARDAPIO, 'DD' ))                     DIA
                                ,TP_REF.DS_TIPO_REFEICAO                                            TP_REF
                                ,SUM(NVL(IT.QT_CARDAPIO,0 ))                                        QTDADE
                         FROM    DBAMV.MOV_CARDAPIO        MC
                                ,DBAMV.ITMOV_CARDAPIO      IT
                                ,DBAMV.TIPO_REFEICAO       TP_REF
                                ,DBAMV.SETOR
                         WHERE   MC.CD_MOV_CARDAPIO  = IT.CD_MOV_CARDAPIO
                         AND     MC.CD_TIPO_REFEICAO = TP_REF.CD_TIPO_REFEICAO
                         AND     TO_CHAR( MC.DT_MOV_CARDAPIO, 'MM/YYYY' ) = TO_CHAR(:CAMPO_DT_COMPETE,'MM/YYYY')
                         AND     MC.CD_SETOR = SETOR.CD_SETOR
                        GROUP BY MC.CD_SETOR, SETOR.NM_SETOR, TO_NUMBER(TO_CHAR( MC.DT_MOV_CARDAPIO, 'DD' )), TP_REF.DS_TIPO_REFEICAO
                    )
                GROUP BY CD_SETOR, NM_SETOR, DIA, TP_REF
                ORDER BY CD_SETOR
            )
        ORDER BY CD_SETOR,NM_SETOR, TP_REF, DIA
