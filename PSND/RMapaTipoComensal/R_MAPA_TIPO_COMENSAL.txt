--R_MAPA_TIPO_COMENSAL
SELECT * FROM (
     SELECT  DECODE(MC.TP_CARDAPIO, 'P' ,'PACIENTE' ,
                                                                      'A' ,'ACOMPANHANTE' ,
                                                                      'F' ,'FUNCIONARIO' ,
                                                                      'S' ,'SETOR' ,
                                                                      'M' ,'MEDICO' ,
                                                                      'B' ,'BANDEIJAO' ,'OUTROS' )    TP_COMENSAL
            ,NVL(TP_REF.DS_TIPO_REFEICAO, 'NAI INFORMADO')                  TP_REF
            ,SUM(NVL(IT.QT_CARDAPIO,1))                                                         QT_T
     FROM    DBAMV.MOV_CARDAPIO        MC
                   ,DBAMV.ITMOV_CARDAPIO    IT
                   ,DBAMV.TIPO_REFEICAO       TP_REF
     WHERE   MC.CD_MOV_CARDAPIO  = IT.CD_MOV_CARDAPIO(+)
     AND     MC.CD_TIPO_REFEICAO = TP_REF.CD_TIPO_REFEICAO(+)
     AND     TO_CHAR( MC.DT_MOV_CARDAPIO, 'MM/YYYY' ) = TO_CHAR(:CAMPO_DT_COMPETE,'MM/YYYY')
     AND     MC.CD_MULTI_EMPRESA = NVL(DBAMV.PKG_MV2000.LE_EMPRESA,MC.CD_MULTI_EMPRESA)
GROUP BY  DECODE(MC.TP_CARDAPIO, 'P' ,'PACIENTE' ,
                                                                      'A' ,'ACOMPANHANTE' ,
                                                                      'F' ,'FUNCIONARIO' ,
                                                                      'S' ,'SETOR' ,
                                                                      'M' ,'MEDICO' ,
                                                                      'B' ,'BANDEIJAO' ,'OUTROS' )
,NVL(TP_REF.DS_TIPO_REFEICAO, 'NAI INFORMADO')
)
ORDER BY TP_COMENSAL,
TP_REF
