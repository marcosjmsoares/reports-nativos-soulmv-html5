--R_MOV_ATENDIME
SELECT * FROM (
SELECT   MC.DT_MOV_CARDAPIO
        ,TP_REF.DS_TIPO_REFEICAO
        ,TP_DIETA.DS_TIPO_DIETA
        ,MC.TP_CARDAPIO
        ,MC.CD_ATENDIMENTO
        ,IMC.CD_OPCAO
        ,OPC.DS_OPCAO
        ,IMC.QT_CARDAPIO
FROM     DBAMV.MOV_CARDAPIO    MC
        ,DBAMV.TIPO_REFEICAO  TP_REF
        ,DBAMV.TIPO_DIETA  TP_DIETA
        ,DBAMV.ITMOV_CARDAPIO  IMC
        ,DBAMV.OPCAO_CARDAPIO  OPC
        ,DBAMV.ATENDIME   ATEN
WHERE    MC.CD_MOV_CARDAPIO   = IMC.CD_MOV_CARDAPIO(+)
        AND      MC.CD_TIPO_REFEICAO = TP_REF.CD_TIPO_REFEICAO(+)
        AND      MC.CD_TIPO_DIETA  =  TP_DIETA.CD_TIPO_DIETA(+)
        AND      IMC.CD_OPCAO         = OPC.CD_OPCAO(+)
        AND      ATEN.CD_MULTI_EMPRESA = NVL(DBAMV.PKG_MV2000.LE_EMPRESA, ATEN.CD_MULTI_EMPRESA)
        AND      trunc(MC.DT_MOV_CARDAPIO)   BETWEEN TO_DATE(:CAMPO_DT_INIC, 'DD/MM/YYYY')  AND TO_DATE(:CAMPO_DT_FINAL, 'DD/MM/YYYY')
        AND      MC.CD_ATENDIMENTO = ATEN.CD_ATENDIMENTO
ORDER BY MC.DT_MOV_CARDAPIO, MC.CD_TIPO_DIETA, MC.CD_TIPO_REFEICAO
)
