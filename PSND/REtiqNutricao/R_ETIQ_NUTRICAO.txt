--R_ETIQ_NUTRICAO
SELECT  'Unidade:'                                    ATENDIMENTO,
        TIPO_REFEICAO.DS_TIPO_REFEICAO                PRATO,
        LEITO.CD_UNID_INT                             CD_UNID_INT,
        LEITO.DS_ENFERMARIA                           DS_ENFERMARIA,
        LEITO.DS_RESUMO                               DS_LEITO,
        UNID_INT.DS_UNID_INT                          DS_UNID_INT,
        LEITO.CD_COPA                                 CD_COPA,
        COPA.DS_COPA                                  DS_COPA,
        PACIENTE.CD_PACIENTE                          CD_PACIENTE,
        dbamv.fnc_get_pac_nome(ATENDIME.CD_ATENDIMENTO)           NM_PACIENTE,
        CONVENIO.NM_CONVENIO                          NM_CONVENIO,
        ATENDIME.DT_ATENDIMENTO                       DT_ATENDIMENTO,
        ATENDIME.CD_ATENDIMENTO                       CD_ATENDIMENTO,
        RTRIM(DBAMV.PACK_INTERNAMENTO.RETORNA_IDADE(PACIENTE.DT_NASCIMENTO,SYSDATE),' ')  NR_IDADE,
        MOV_CARDAPIO.OBS_NUTRICAO ,
        TIPO_REFEICAO.CD_TIPO_REFEICAO                CD_TIPO_REFEICAO
  FROM  DBAMV.V_INTERNADOS                         V_INTERNADOS,
        DBAMV.LEITO                                LEITO,
        DBAMV.UNID_INT                             UNID_INT,
        DBAMV.COPA                                 COPA,
        DBAMV.ATENDIME                             ATENDIME,
        DBAMV.PACIENTE                             PACIENTE,
        DBAMV.CONVENIO                             CONVENIO,
        DBAMV.MOV_CARDAPIO                         MOV_CARDAPIO,
        DBAMV.EMPRESA_CONVENIO                     EMPRESA_CONVENIO,
        DBAMV.TIPO_REFEICAO                        TIPO_REFEICAO
  Where  LEITO.TP_OCUPACAO NOT IN ('M','L','R','A','V')
	AND LEITO.CD_LEITO (+) = V_INTERNADOS.CD_LEITO
	AND COPA.CD_COPA (+) = LEITO.CD_COPA
	AND UNID_INT.CD_UNID_INT (+) = LEITO.CD_UNID_INT
	AND UNID_INT.SN_ATIVO = 'S'
	AND ATENDIME.CD_ATENDIMENTO = V_INTERNADOS.CD_ATENDIMENTO
	AND ATENDIME.CD_MULTI_EMPRESA = NVL(DBAMV.PKG_MV2000.LE_EMPRESA, ATENDIME.CD_MULTI_EMPRESA)
	AND PACIENTE.CD_PACIENTE = ATENDIME.CD_PACIENTE
	AND CONVENIO.CD_CONVENIO = ATENDIME.CD_CONVENIO
	AND EMPRESA_CONVENIO.CD_CONVENIO = CONVENIO.CD_CONVENIO
	AND EMPRESA_CONVENIO.CD_MULTI_EMPRESA = NVL(DBAMV.PKG_MV2000.LE_EMPRESA, ATENDIME.CD_MULTI_EMPRESA)
	AND ATENDIME.CD_ATENDIMENTO = MOV_CARDAPIO.CD_ATENDIMENTO(+)
	AND TIPO_REFEICAO.CD_TIPO_REFEICAO (+) = MOV_CARDAPIO.CD_TIPO_REFEICAO
	AND (MOV_CARDAPIO.CD_MOV_CARDAPIO = (SELECT MAX(MOV_CARD.CD_MOV_CARDAPIO)
			FROM DBAMV.MOV_CARDAPIO MOV_CARD
			WHERE MOV_CARDAPIO.CD_ATENDIMENTO = MOV_CARD.CD_ATENDIMENTO
			AND	NVL(MOV_CARD.SN_PRESCRICAO_SUSPENSA,'N') = 'N'
			AND trunc(MOV_CARD.DT_MOV_CARDAPIO) BETWEEN  :CAMPO_DATA AND :CAMPO_DATA_FINAL )
	      :CAMPO_CF_TP_ATENDIMENTO1 )
		UNION ALL
SELECT          'Amb/Urg:'                                    	ATENDIMENTO,
                TIPO_REFEICAO.DS_TIPO_REFEICAO                  PRATO,
                0                                               CD_UNID_INT,
                ''                                              DS_ENFERMARIA,
                ''                                              DS_LEITO,
                ORI_ATE.DS_ORI_ATE                              DS_UNID_INT,
                ORI_ATE.CD_COPA                                 CD_COPA,
                COPA.DS_COPA                                    DS_COPA,
                PACIENTE.CD_PACIENTE                            CD_PACIENTE,
                PACIENTE.NM_PACIENTE                            NM_PACIENTE,
                CONVENIO.NM_CONVENIO                            NM_CONVENIO,
                ATENDIME.DT_ATENDIMENTO                         DT_ATENDIMENTO,
                ATENDIME.CD_ATENDIMENTO                         CD_ATENDIMENTO,
                RTRIM(DBAMV.PACK_INTERNAMENTO.RETORNA_IDADE(PACIENTE.DT_NASCIMENTO,SYSDATE),' ')  NR_IDADE,
                MOV_CARDAPIO.OBS_NUTRICAO,
				TIPO_REFEICAO.CD_TIPO_REFEICAO                  CD_TIPO_REFEICAO
      FROM      DBAMV.COPA                              COPA,
                DBAMV.ATENDIME                          ATENDIME,
                DBAMV.PACIENTE                          PACIENTE,
                DBAMV.CONVENIO                          CONVENIO,
                DBAMV.MOV_CARDAPIO                      MOV_CARDAPIO,
                DBAMV.ORI_ATE                           ORI_ATE,
                DBAMV.TIPO_REFEICAO                     TIPO_REFEICAO
               ,DBAMV.EMPRESA_CONVENIO
Where 	EMPRESA_CONVENIO.CD_CONVENIO (+) = CONVENIO.CD_CONVENIO
		AND NVL(EMPRESA_CONVENIO.CD_MULTI_EMPRESA,1)  = NVL(DBAMV.PKG_MV2000.LE_EMPRESA, ATENDIME.CD_MULTI_EMPRESA)
		AND CONVENIO.CD_CONVENIO = ATENDIME.CD_CONVENIO
		AND PACIENTE.CD_PACIENTE = ATENDIME.CD_PACIENTE
		AND COPA.CD_COPA (+) = ORI_ATE.CD_COPA
		AND ORI_ATE.CD_ORI_ATE (+) = ATENDIME.CD_ORI_ATE
		AND ATENDIME.TP_ATENDIMENTO IN ('U','A')
		AND ATENDIME.CD_MULTI_EMPRESA = NVL(DBAMV.PKG_MV2000.LE_EMPRESA, ATENDIME.CD_MULTI_EMPRESA)
		AND TRUNC(ATENDIME.DT_ATENDIMENTO) BETWEEN  :CAMPO_DATA AND 
		AND MOV_CARDAPIO.CD_ATENDIMENTO (+) = ATENDIME.CD_ATENDIMENTO
		AND ( MOV_CARDAPIO.CD_MOV_CARDAPIO = (SELECT MAX(MOV_CARD.CD_MOV_CARDAPIO)
			FROM DBAMV.MOV_CARDAPIO MOV_CARD
			WHERE MOV_CARDAPIO.CD_ATENDIMENTO = MOV_CARD.CD_ATENDIMENTO
			AND NVL(MOV_CARD.SN_PRESCRICAO_SUSPENSA,'N') = 'N'
			AND TRUNC(MOV_CARD.DT_MOV_CARDAPIO) BETWEEN  :CAMPO_DATA AND :CAMPO_DATA_FINAL )
		:CAMPO_CF_TP_ATENDIMENTO2 )
		AND TIPO_REFEICAO.CD_TIPO_REFEICAO (+) = MOV_CARDAPIO.CD_TIPO_REFEICAO
		ORDER BY   DS_UNID_INT, DS_LEITO, PRATO
