--R_COPEIRA_NUTRI
SELECT 	A.NM_SETOR,
		A.CD_SETOR,
		A.CD_PACIENTE,
		A.NM_PACIENTE,
		A.DS_COPA,
		A.DS_LEITO,
		A.NR_IDADE,
		A.NM_CONVENIO,
		A.DT_NASCIMENTO,
		A.PESO,
		A.ALTURA,
		A.DS_CID,
		A.CD_ATENDIMENTO
FROM (	SELECT  COPA.DS_COPA,
				UNID_INT.DS_UNID_INT,
				LEITO.DS_RESUMO DS_LEITO,
				PACIENTE.CD_PACIENTE,
				DBAMV.FNC_GET_PAC_NOME(ATENDIME.CD_ATENDIMENTO) NM_PACIENTE,
				RTRIM(PACK_INTERNAMENTO.RETORNA_IDADE(PACIENTE.DT_NASCIMENTO,SYSDATE),' ')  NR_IDADE,
				CONVENIO.NM_CONVENIO,
				PACIENTE.DT_NASCIMENTO,
				DBAMV.FNC_PAGU_SUPER_CORPOREA_PESO(ATENDIME.CD_ATENDIMENTO) PESO,
				DBAMV.FNC_PAGU_SUPER_CORPOREA_ALTURA(ATENDIME.CD_ATENDIMENTO) ALTURA,
				CID.DS_CID,
				SETOR.NM_SETOR,
				SETOR.CD_SETOR,
				ATENDIME.CD_ATENDIMENTO
		FROM 	DBAMV.ATENDIME,
				DBAMV.PACIENTE,
				DBAMV.CONVENIO,
				DBAMV.CID,
				DBAMV.COPA,
				DBAMV.UNID_INT,
				DBAMV.LEITO,
				DBAMV.SETOR,
				DBAMV.V_INTERNADOS
		WHERE 	V_INTERNADOS.CD_LEITO           = LEITO.CD_LEITO
		AND 	V_INTERNADOS.CD_ATENDIMENTO     = ATENDIME.CD_ATENDIMENTO
		AND   	ATENDIME.CD_PACIENTE        = PACIENTE.CD_PACIENTE
		AND 	ATENDIME.CD_CONVENIO        = CONVENIO.CD_CONVENIO
		AND 	ATENDIME.CD_CID            	= CID.CD_CID(+)
		AND   	COPA.CD_COPA                = LEITO.CD_COPA
		AND 	LEITO.CD_UNID_INT           = UNID_INT.CD_UNID_INT
		AND 	SETOR.CD_SETOR              = UNID_INT.CD_SETOR
		AND 	ATENDIME.CD_MULTI_EMPRESA   = DBAMV.PKG_MV2000.LE_EMPRESA
		UNION ALL
		SELECT 	COPA.DS_COPA,
				'AMBULATÓRIO',
				'AMB' DS_LEITO,
				PACIENTE.CD_PACIENTE,
				dbamv.fnc_get_pac_nome(ATENDIME.CD_ATENDIMENTO),
				RTRIM(PACK_INTERNAMENTO.RETORNA_IDADE(PACIENTE.DT_NASCIMENTO,SYSDATE),' ') NR_IDADE,
				CONVENIO.NM_CONVENIO,
				PACIENTE.DT_NASCIMENTO,
				DBAMV.FNC_PAGU_SUPER_CORPOREA_PESO(ATENDIME.CD_ATENDIMENTO) PESO,
				DBAMV.FNC_PAGU_SUPER_CORPOREA_ALTURA(ATENDIME.CD_ATENDIMENTO) ALTURA,
				CID.DS_CID,
				SETOR.NM_SETOR,
				SETOR.CD_SETOR,
				ATENDIME.CD_ATENDIMENTO
		FROM DBAMV.ATENDIME,
			DBAMV.PACIENTE,
			DBAMV.CONVENIO,
			DBAMV.CID,
			DBAMV.COPA,
			DBAMV.ORI_ATE,
			DBAMV.SETOR
		WHERE 	ATENDIME.CD_PACIENTE        = PACIENTE.CD_PACIENTE
		AND 	ATENDIME.CD_CONVENIO        = CONVENIO.CD_CONVENIO
		AND 	SETOR.CD_SETOR              = ORI_ATE.CD_SETOR
		AND 	ATENDIME.CD_MULTI_EMPRESA   = DBAMV.PKG_MV2000.LE_EMPRESA
		AND 	ATENDIME.CD_CID             = CID.CD_CID(+)
		AND 	ATENDIME.CD_ORI_ATE         = ORI_ATE.CD_ORI_ATE
		AND 	ORI_ATE.CD_COPA             = COPA.CD_COPA
		AND 	ATENDIME.TP_ATENDIMENTO 	IN ('A','E')
		AND 	TRUNC(ATENDIME.DT_ALTA) IS NULL
		:CAMPO_CF_COPA
		) A
		ORDER BY A.NM_SETOR, A.DS_LEITO
