--R_DIETA_OBS_LACTARIO
SELECT * FROM (
SELECT leito.cd_unid_int               cd_unid_int,
       leito.ds_enfermaria             ds_enfermaria,
       leito.ds_resumo                 ds_leito,
       unid_int.ds_unid_int            ds_unid_int,
       paciente.cd_paciente            cd_paciente,
       dbamv.fnc_get_pac_nome(atendime.cd_atendimento, 'AP')            nm_paciente,
       atendime.cd_atendimento         cd_atendimento,
       RTRIM(dbamv.pack_internamento.retorna_idade(paciente.dt_nascimento,SysDate),' ')  nr_idade,
       paciente.tp_sexo                tp_sexo,
       nvl(atendime_obs_lactario.ds_obs_lactario,'Sem Mamadeira') ds_obs_lactario
FROM   dbamv.atendime,
       dbamv.paciente,
       dbamv.leito,
       dbamv.unid_int,
       dbamv.atendime_obs_lactario
WHERE  atendime.tp_atendimento = 'I'
AND    atendime.dt_alta IS NULL
AND    atendime.cd_paciente = paciente.cd_paciente
AND    atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
AND    atendime.cd_leito = leito.cd_leito
AND    leito.cd_unid_int = unid_int.cd_unid_int
AND    unid_int.sn_ativo = 'S'
AND    atendime.cd_atendimento = atendime_obs_lactario.cd_atendimento(+)
AND    atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
ORDER BY  DS_UNID_INT, DS_LEITO, ds_enfermaria
)
ORDER BY DS_UNID_INT,CD_UNID_INT, ds_leito, ds_enfermaria
