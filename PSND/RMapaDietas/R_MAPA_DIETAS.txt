--R_MAPA_DIETAS
SELECT * FROM (
Select LEITO.CD_UNID_INT                    CD_UNID_INT
      ,UNID_INT.DS_UNID_INT                 DS_UNID_INT
      ,PACIENTE.CD_PACIENTE                 CD_PACIENTE
      ,dbamv.fnc_get_pac_nome(ATENDIME.CD_ATENDIMENTO)    NM_PACIENTE
      ,ATENDIME.DT_ATENDIMENTO              DT_ATENDIMENTO
      ,ATENDIME.CD_ATENDIMENTO              CD_ATENDIMENTO
      ,RTRIM(PACK_INTERNAMENTO.RETORNA_IDADE(PACIENTE.DT_NASCIMENTO,SysDate),' ')  NR_IDADE
      ,CID.DS_CID                           DS_CID
      ,LEITO.DS_RESUMO
      ,ATENDIME.SN_ACOMPANHANTE
      ,CONVENIO.NM_CONVENIO
      ,fnc_mv_recupera_data_hora(ATENDIME.DT_ATENDIMENTO,ATENDIME.HR_ATENDIMENTO) DTHR_INTERNACAO
FROM  DBAMV.ATENDIME
     ,DBAMV.PACIENTE
     ,DBAMV.CID
     ,DBAMV.LEITO
     ,DBAMV.UNID_INT
     ,DBAMV.CONVENIO
WHERE  ATENDIME.TP_ATENDIMENTO = 'I'
AND   (ATENDIME.DT_ALTA IS NULL )
AND   ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND   ATENDIME.CD_MULTI_EMPRESA = NVL(DBAMV.PKG_MV2000.LE_EMPRESA, ATENDIME.CD_MULTI_EMPRESA)
AND   ATENDIME.CD_CID = CID.CD_CID(+)
AND   ATENDIME.CD_LEITO = LEITO.CD_LEITO
AND   LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT
AND   UNID_INT.SN_ATIVO = 'S'
AND   CONVENIO.CD_CONVENIO = ATENDIME.CD_CONVENIO
Union All
Select LEITO.CD_UNID_INT                    CD_UNID_INT
      ,UNID_INT.DS_UNID_INT                 DS_UNID_INT
      ,0                 CD_PACIENTE
      ,Null               NM_PACIENTE
      ,SYSDATE              DT_ATENDIMENTO
      ,0             CD_ATENDIMENTO
      ,Null                NR_IDADE
      ,Null                        DS_CID
      ,LEITO.DS_RESUMO
      ,Null
      ,Null
      ,to_date(Null, 'dd/mm/yyyy')
FROM DBAMV.LEITO
     ,DBAMV.UNID_INT
WHERE LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT
AND   UNID_INT.SN_ATIVO = 'S'
AND TP_OCUPACAO =  'V'
AND :CAMPO_LEITO = 'S'
:CAMPO_CF_UNID_INT
)
ORDER BY DS_UNID_INT, 9
