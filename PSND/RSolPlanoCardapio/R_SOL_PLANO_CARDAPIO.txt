--R_SOL_PLANO_CARDAPIO
SELECT * FROM (
SELECT
    SOLICITACAO_CARDAPIO_PRODUTO.CD_PRODUTO CD_PRODUTO
    , SOLICITACAO_CARDAPIO_PRODUTO.CD_UNI_PRO CD_UNI_PRO
    , SOLICITACAO_CARDAPIO_PRODUTO.QT_SOLICITADA
    , PRODUTO.DS_PRODUTO DS_PRODUTO
    , UNI_PRO.DS_UNIDADE                DS_UNIDADE
    , SOLICITACAO_CARDAPIO.CD_SOLICITACAO_CARDAPIO CD_SOLICITACAO_CARDAPIO
    , SOLICITACAO_CARDAPIO.DT_SOLICITACAO_CARDAPIO DT_SOLICITACAO_CARDAPIO
    , ESTOQUE.DS_ESTOQUE DS_ESTOQUE
    , SETOR.NM_SETOR
    , SOLICITACAO_CARDAPIO.CD_SOLSAI_PRO CD_SOLSAI_PRO
    , SOLICITACAO_CARDAPIO_PLAN_CRDP.CD_PLANO_CARDAPIO
    , PLANO_CARDAPIO.DS_PLANO_CARDAPIO
    , SOLICITACAO_CARDAPIO_PRODUTO.CD_ITSOLSAI_PRO
		FROM DBAMV.SOLICITACAO_CARDAPIO
			 , DBAMV.SOLICITACAO_CARDAPIO_PLAN_CRDP
			 , DBAMV.SOLICITACAO_CARDAPIO_OPCAO
			 , DBAMV.SOLICITACAO_CARDAPIO_PRATO
			 , DBAMV.SOLICITACAO_CARDAPIO_PRODUTO
			 , DBAMV.PLANO_CARDAPIO
			 , DBAMV.PRODUTO
			 , DBAMV.UNI_PRO
			 , DBAMV.SETOR
			 , DBAMV.ESTOQUE
		WHERE SOLICITACAO_CARDAPIO.CD_SOLICITACAO_CARDAPIO  =	SOLICITACAO_CARDAPIO_PLAN_CRDP.CD_SOLICITACAO_CARDAPIO
			AND SOLICITACAO_CARDAPIO_PLAN_CRDP.CD_SOLICITACAO_CRDP_PL_CRDP  = SOLICITACAO_CARDAPIO_OPCAO.CD_SOLICITACAO_CRDP_PL_CRDP
			AND SOLICITACAO_CARDAPIO_OPCAO.CD_SOLICITACAO_CARDAPIO_OPCAO = SOLICITACAO_CARDAPIO_PRATO.CD_SOLICITACAO_CARDAPIO_OPCAO
      AND SOLICITACAO_CARDAPIO_PRATO.CD_SOLICITACAO_CARDAPIO_PRATO = SOLICITACAO_CARDAPIO_PRODUTO.CD_SOLICITACAO_CARDAPIO_PRATO
			AND PLANO_CARDAPIO.CD_PLANO_CARDAPIO  = SOLICITACAO_CARDAPIO_PLAN_CRDP.CD_PLANO_CARDAPIO
			AND SOLICITACAO_CARDAPIO.CD_SETOR   = SETOR.CD_SETOR
			AND SOLICITACAO_CARDAPIO.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
			AND SOLICITACAO_CARDAPIO_PRODUTO.CD_PRODUTO	  =  PRODUTO.CD_PRODUTO
			AND SOLICITACAO_CARDAPIO_PRODUTO.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO
                        AND ESTOQUE.CD_MULTI_EMPRESA = NVL(DBAMV.PKG_MV2000.LE_EMPRESA, ESTOQUE.CD_MULTI_EMPRESA)
			AND SOLICITACAO_CARDAPIO.DT_SOLICITACAO_CARDAPIO BETWEEN :CAMPO_DT_INICIAL AND 
        :CAMPO_CF_TP_SOLICITACAO
) ORDER BY CD_SOLICITACAO_CARDAPIO, CD_PLANO_CARDAPIO, DS_PLANO_CARDAPIO, DS_PRODUTO,DS_UNIDADE, CD_ITSOLSAI_PRO
