--R_EST_REF_DESTINO
SELECT * FROM (
SELECT NM_SETOR NM_SETOR,
       DS_TP_REF DS_TP_REF,
 sum(QT_A) QT_A,
 sum(QT_F) QT_F,
 sum(QT_S) QT_S,
 sum(QT_P) QT_P,
 sum(QT_B) QT_B,
 sum(QT_M) QT_M,
 sum(QT_T) QT_T
from(
SELECT   SETOR.NM_SETOR                                  NM_SETOR
        ,TP_REF.DS_TIPO_REFEICAO                         DS_TP_REF
        ,DECODE( MC.TP_CARDAPIO, 'A', IT.QT_CARDAPIO,0 ) QT_A
        ,DECODE( MC.TP_CARDAPIO, 'F', IT.QT_CARDAPIO,0 ) QT_F
        ,DECODE( MC.TP_CARDAPIO, 'S', IT.QT_CARDAPIO,0 ) QT_S
        ,DECODE( MC.TP_CARDAPIO, 'P', IT.QT_CARDAPIO,0 ) QT_P
        ,DECODE( MC.TP_CARDAPIO, 'B', IT.QT_CARDAPIO,0 ) QT_B
        ,DECODE( MC.TP_CARDAPIO, 'M', IT.QT_CARDAPIO,0 ) QT_M
        ,IT.QT_CARDAPIO  QT_T
FROM     DBAMV.MOV_CARDAPIO    MC
        ,DBAMV.ITMOV_CARDAPIO  IT
        ,DBAMV.TIPO_REFEICAO   TP_REF
        ,DBAMV.SETOR
WHERE    MC.CD_MOV_CARDAPIO              = IT.CD_MOV_CARDAPIO
AND MC.SN_PRESCRICAO_SUSPENSA = 'N'
AND      MC.CD_TIPO_REFEICAO                  = TP_REF.CD_TIPO_REFEICAO
AND     TRUNC( MC.DT_MOV_CARDAPIO)  between TRUNC(:CAMPO_DTINIC) and TRUNC(:CAMPO_DTFIM)
AND      SETOR.CD_SETOR                           = MC.CD_SETOR
AND MC.CD_MULTI_EMPRESA = NVL(DBAMV.PKG_MV2000.LE_EMPRESA, MC.CD_MULTI_EMPRESA)
)
GROUP BY NM_SETOR,
       DS_TP_REF,
 QT_A,
 QT_F,
 QT_S,
 QT_P,
 QT_B,
 QT_M,
 QT_T
)
ORDER BY NM_SETOR,
DS_TP_REF
