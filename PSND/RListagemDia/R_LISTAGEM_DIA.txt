--R_LISTAGEM_DIA
SELECT * FROM (
SELECT C_C.DS_CLAS_CARDAPIO
         ,IMV.CD_OPCAO
         ,OPC.DS_OPCAO
         ,IMV.QT_CARDAPIO
         ,TP_D.DS_TIPO_DIETA
FROM      DBAMV.MOV_CARDAPIO    MV
         ,DBAMV.ITMOV_CARDAPIO  IMV
         ,DBAMV.TIPO_DIETA      TP_D
         ,DBAMV.OPCAO_CARDAPIO  OPC
         ,DBAMV.CLAS_CARDAPIO   C_C
WHERE TRUNC(MV.DT_MOV_CARDAPIO)   BETWEEN  TRUNC( :CAMPO_DT_INIC  )  AND TRUNC( :CAMPO_DT_FIM )
AND       MV.CD_MOV_CARDAPIO    = IMV.CD_MOV_CARDAPIO
AND       IMV.CD_OPCAO                   = OPC.CD_OPCAO
AND       MV.CD_TIPO_DIETA           = TP_D.CD_TIPO_DIETA(+)
AND       OPC.CD_CLAS_CARDAPIO = C_C.CD_CLAS_CARDAPIO
AND       MV.CD_MULTI_EMPRESA  = NVL(DBAMV.PKG_MV2000.LE_EMPRESA, MV.CD_MULTI_EMPRESA)
ORDER BY  C_C.DS_CLAS_CARDAPIO
)
ORDER BY DS_CLAS_CARDAPIO
