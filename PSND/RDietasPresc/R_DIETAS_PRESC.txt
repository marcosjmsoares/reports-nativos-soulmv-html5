--R_DIETAS_PRESC
SELECT
	Pkg_Mv_Variaveis.fnc_get_usuario() AS CD_USUARIO
        ,MC.CD_USUARIO as OLD_CD_USUARIO
	,MC.CD_ATENDIMENTO
        , 'teste'   DS_RESUMO
        ,UNID_INT.DS_UNID_INT
        ,UNID_INT.CD_UNID_INT
        ,SETOR.NM_SETOR
        ,dbamv.fnc_get_pac_nome(MC.CD_ATENDIMENTO) NM_PACIENTE
        ,CONVENIO.CD_CONVENIO
        ,CONVENIO.NM_CONVENIO
        ,PRE_MED.CD_PRE_MED
        ,MC.CD_COPA
        ,TIPO_DIETA.DS_TIPO_DIETA
        ,TIPO_REFEICAO.CD_TIPO_REFEICAO
        ,TIPO_REFEICAO.DS_TIPO_REFEICAO
        ,PRESTADOR.NM_PRESTADOR
        ,PRE_MED.DT_PRE_MED
        ,NVL( MC.DS_OBSERVACAO, ITPRE_MED.DS_ITPRE_MED  ) DS_ITPRE_MED
        ,MC.OBS_NUTRICAO
        ,COPA.DS_COPA
        ,ITPRE_MED.QT_ITPRE_MED
        ,ITPRE_MED.CD_UNI_PRO
        ,Nvl(UNI_PRO.DS_UNIDADE,UNI_PRESC.CD_UNIDADE)  DS_UNIDADE
        ,ITPRE_MED.CD_FOR_APL
        ,ITPRE_MED.CD_TIP_FRE
        ,TIP_FRE.DS_TIP_FRE
FROM     DBAMV.MOV_CARDAPIO    MC
        ,DBAMV.ATENDIME   ATEN
        ,DBAMV.UNID_INT
        ,DBAMV.SETOR
        ,DBAMV.PACIENTE
        ,DBAMV.ITPRE_MED
        ,DBAMV.TIPO_DIETA
        ,DBAMV.TIPO_REFEICAO
        ,DBAMV.PRE_MED
        ,DBAMV.PRESTADOR
        ,DBAMV.COPA
        ,DBAMV.CONVENIO
        ,DBAMV.UNI_PRO
        ,DBAMV.UNI_PRESC
        ,DBAMV.TIP_FRE
WHERE MC.CD_ATENDIMENTO = ATEN.CD_ATENDIMENTO(+)
      AND ATEN.CD_PACIENTE = PACIENTE.CD_PACIENTE
      AND  ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
      AND MC.CD_SETOR = SETOR.CD_SETOR(+)
      AND ATEN.CD_CONVENIO = CONVENIO.CD_CONVENIO(+)
      AND MC.CD_TIPO_REFEICAO = TIPO_REFEICAO.CD_TIPO_REFEICAO(+)
      AND MC.CD_ITPRE_MED = ITPRE_MED.CD_ITPRE_MED(+)
      AND MC.CD_TIPO_DIETA = TIPO_DIETA.CD_TIPO_DIETA(+)
      AND ITPRE_MED.CD_PRE_MED = PRE_MED.CD_PRE_MED
      AND PRE_MED.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
      AND ATEN.CD_ATENDIMENTO = PRE_MED.CD_ATENDIMENTO(+)
      AND PRE_MED.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)
      AND MC.CD_COPA = COPA.CD_COPA(+)
     AND ATEN.CD_ATENDIMENTO = :CAMPO_CD_ATENDIME
     AND ( PRE_MED.CD_PRE_MED = :CAMPO_CD_PRE_MED
                or
                 is null)
     AND PRE_MED.DT_REFERENCIA = :CAMPO_DT_REFERENCIA
    AND UNI_PRO.CD_UNI_PRO(+) = ITPRE_MED.CD_UNI_PRO
    AND UNI_PRESC.CD_UNI_PRESC(+) = ITPRE_MED.CD_UNI_PRESC
   AND TIP_FRE.CD_TIP_FRE(+) = ITPRE_MED.CD_TIP_FRE
ORDER BY MC.DT_MOV_CARDAPIO, MC.CD_TIPO_DIETA, MC.CD_TIPO_REFEICAO
