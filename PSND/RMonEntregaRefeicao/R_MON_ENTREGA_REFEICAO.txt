--R_MON_ENTREGA_REFEICAO
SELECT MC.CD_ATENDIMENTO, 
       MC.DT_ENTREGA,
       PC.NM_PACIENTE,
       LT.DS_RESUMO,
       CVN.NM_CONVENIO,
       PC.DT_NASCIMENTO,
       TR.DS_TIPO_REFEICAO,
       PC.DT_NASCIMENTO,
       MC.TP_CARDAPIO,
       MC.DT_MOV_CARDAPIO        
FROM DBAMV.MOV_CARDAPIO MC, 
     DBAMV.PACIENTE PC, 
     DBAMV.LEITO LT,
     DBAMV.ATENDIME ATEN,
     DBAMV.CONVENIO CVN,
     DBAMV.TIPO_REFEICAO TR
WHERE MC.CD_ATENDIMENTO = ATEN.CD_ATENDIMENTO
  AND LT.CD_LEITO(+) = ATEN.CD_LEITO
  AND ATEN.CD_CONVENIO = CVN.CD_CONVENIO
  AND ATEN.CD_PACIENTE = PC.CD_PACIENTE
  AND MC.CD_TIPO_REFEICAO = TR.CD_TIPO_REFEICAO
  AND MC.TP_CARDAPIO = 'A'
ORDER BY  CD_ATENDIMENTO,DT_MOV_CARDAPIO, DS_TIPO_REFEICAO
