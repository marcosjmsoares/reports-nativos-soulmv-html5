--R_COMPROVANTE_ACOMPANHANTE
SELECT
       leito.cd_unid_int               cd_unid_int,
       leito.ds_enfermaria             ds_enfermaria,
       leito.ds_resumo                 ds_leito,
       unid_int.ds_unid_int            ds_unid_int,
       paciente.cd_paciente            cd_paciente,
       dbamv.fnc_get_pac_nome(atendime.cd_atendimento)       nm_paciente,
       atendime.cd_atendimento         cd_atendimento,
       mov_cardapio.cd_tipo_refeicao cd_tipo_refeicao,
       tipo_refeicao.ds_tipo_refeicao  ds_tipo_refeicao,
       mov_cardapio.obs_nutricao    obs_nutricao,
       mov_cardapio.ds_observacao  ds_observacao,
       tipo_dieta.cd_tipo_dieta     cd_tipo_dieta,
       Decode(mov_cardapio.tp_cardapio,'A','Geral',tipo_dieta.ds_tipo_dieta)      ds_tipo_dieta,
       mov_cardapio.cd_mov_cardapio cd_mov_cardapio,
       mov_cardapio.dt_mov_cardapio dt_mov_cardapio,
       PACK_INTERNAMENTO.RETORNA_IDADE( PACIENTE.DT_NASCIMENTO, SYSDATE)  NR_IDADE,
       convenio.cd_convenio                cd_convenio,
       convenio.nm_convenio                nm_convenio
FROM   dbamv.atendime,
       dbamv.paciente,
       dbamv.leito,
       dbamv.unid_int,
       dbamv.tipo_refeicao,
       dbamv.mov_cardapio,
       dbamv.tipo_dieta,
       dbamv.convenio
WHERE  atendime.tp_atendimento = 'I'
AND    atendime.dt_alta IS NULL
AND    atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
AND    atendime.cd_paciente = paciente.cd_paciente
AND    atendime.cd_leito = leito.cd_leito
AND    leito.cd_unid_int = unid_int.cd_unid_int
AND    unid_int.sn_ativo = 'S'
AND    atendime.cd_atendimento = mov_cardapio.cd_atendimento
AND    mov_cardapio.cd_tipo_refeicao = tipo_refeicao.cd_tipo_refeicao(+)
AND    atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
AND    mov_cardapio.cd_tipo_dieta = tipo_dieta.cd_tipo_dieta(+)
AND   atendime.cd_convenio = convenio.cd_convenio
union all
SELECT
       0  cd_unid_int,
       '' ds_enfermaria,
       '' ds_leito,
       '' ds_unid_int,
       mov_cardapio.cd_func cd_paciente,
       nm_func nm_paciente,
       0  cd_atendimento,
       mov_cardapio.cd_tipo_refeicao cd_tipo_refeicao,
       tipo_refeicao.ds_tipo_refeicao  ds_tipo_refeicao,
       mov_cardapio.obs_nutricao    obs_nutricao,
       mov_cardapio.ds_observacao  ds_observacao,
       tipo_dieta.cd_tipo_dieta     cd_tipo_dieta,
       Decode(mov_cardapio.tp_cardapio,'A','Geral',tipo_dieta.ds_tipo_dieta)      ds_tipo_dieta,
       mov_cardapio.cd_mov_cardapio cd_mov_cardapio,
       mov_cardapio.dt_mov_cardapio dt_mov_cardapio,
       ''  NR_IDADE,
       0 cd_convenio,
       '' nm_convenio
FROM   dbamv.tipo_refeicao,
       dbamv.mov_cardapio,
       dbamv.tipo_dieta,
       dbamv.funcionario
where  mov_cardapio.cd_tipo_refeicao = tipo_refeicao.cd_tipo_refeicao(+)
AND    mov_cardapio.cd_tipo_dieta = tipo_dieta.cd_tipo_dieta(+)
and    mov_cardapio.tp_cardapio not in ('P','A')
and    mov_cardapio.cd_func = funcionario.cd_func (+)
Order by cd_unid_int, ds_leito
