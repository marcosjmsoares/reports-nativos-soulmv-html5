--R_EST_MAMADEIRA_UNID_INT
SELECT * FROM (
SELECT prato.cd_prato,
       prato.ds_prato,
       unid_int.ds_unid_int,
       COUNT(*) valor
FROM   dbamv.ficha_nutricao_lactario,
       dbamv.prato,
       dbamv.atendime,
       dbamv.unid_int,
       dbamv.leito
WHERE  atendime.cd_atendimento = ficha_nutricao_lactario.cd_atendimento
AND    atendime.tp_atendimento = 'I'
AND    atendime.Cd_Multi_Empresa = nvl(dbamv.pkg_mv2000.le_empresa, atendime.Cd_Multi_Empresa)
AND    atendime.dt_alta IS NULL
AND    ficha_nutricao_lactario.cd_prato = prato.cd_prato
AND    atendime.cd_leito = leito.cd_leito
AND    leito.cd_unid_int = unid_int.cd_unid_int
GROUP BY prato.cd_prato,
       prato.ds_prato,
       unid_int.ds_unid_int
)
ORDER BY DS_PRATO,
DS_UNID_INT
