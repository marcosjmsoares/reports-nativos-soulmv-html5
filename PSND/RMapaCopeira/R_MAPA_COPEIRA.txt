--R_MAPA_COPEIRA
SELECT * FROM (
SELECT  LEITO.CD_UNID_INT                                CD_UNID_INT,
                LEITO.CD_LEITO CD_LEITO,
                LEITO.DS_ENFERMARIA                         DS_ENFERMARIA,
                LEITO.DS_RESUMO                                 DS_LEITO,
                UNID_INT.DS_UNID_INT                         DS_UNID_INT,
                LEITO.CD_COPA                                       CD_COPA,
                LEITO.CD_COPA||' - '||COPA.DS_COPA                  DS_COPA,
                PACIENTE.CD_PACIENTE                       CD_PACIENTE,
                dbamv.fnc_get_pac_nome(ATENDIME.CD_ATENDIMENTO, 'AP')     NM_PACIENTE,
                CONVENIO.NM_CONVENIO                     NM_CONVENIO,
                PRESTADOR.NM_PRESTADOR              NM_PRESTADOR,
                ATENDIME.DT_ATENDIMENTO              DT_ATENDIMENTO,
                ATENDIME.CD_ATENDIMENTO              CD_ATENDIMENTO,
                PACIENTE.DT_NASCIMENTO               DT_NASCIMENTO,
                DBAMV.PACK_INTERNAMENTO.RETORNA_IDADE(PACIENTE.DT_NASCIMENTO, TRUNC(SYSDATE))    NR_IDADE
FROM      DBAMV.V_INTERNADOS                          V_INTERNADOS,
                DBAMV.LEITO                                           LEITO,
                DBAMV.UNID_INT                                    UNID_INT,
                DBAMV.COPA                                            COPA,
                DBAMV.ATENDIME                                   ATENDIME,
                DBAMV.PACIENTE                                    PACIENTE,
                DBAMV.CONVENIO                                   CONVENIO,
                DBAMV.PRESTADOR                               PRESTADOR
WHERE   ATENDIME.CD_MULTI_EMPRESA = NVL(DBAMV.PKG_MV2000.LE_EMPRESA, ATENDIME.CD_MULTI_EMPRESA)
      AND   V_INTERNADOS.CD_LEITO (+) =  LEITO.CD_LEITO
      AND   LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT
      AND   UNID_INT.SN_ATIVO = 'S'
      AND   V_INTERNADOS.CD_ATENDIMENTO =  ATENDIME.CD_ATENDIMENTO (+)
      AND   ATENDIME.CD_PACIENTE  =  PACIENTE.CD_PACIENTE (+)
      AND   ATENDIME.CD_CONVENIO   =   CONVENIO.CD_CONVENIO (+)
      AND   PRESTADOR.CD_PRESTADOR (+) =  ATENDIME.CD_PRESTADOR
      AND   LEITO.CD_COPA = COPA.CD_COPA(+)
      AND ATENDIME.DT_ATENDIMENTO(+) <= :CAMPO_DATA
ORDER BY  UNID_INT.DS_UNID_INT, LEITO.CD_LEITO
)
ORDER BY CD_COPA,DS_COPA,DS_UNID_INT, CD_LEITO,CD_UNID_INT
