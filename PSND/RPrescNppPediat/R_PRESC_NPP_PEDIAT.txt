--R_PRESC_NPP_PEDIAT
SELECT PRE_MED.CD_PRE_MED,
		 		To_Date((To_Char(PRE_MED.DT_PRE_MED, 'dd/mm/yy') || ' ' ||to_char(PRE_MED.HR_PRE_MED ,'HH24')), 'dd/mm/yy HH24') DT_PRE_MED,
		 		PRE_MED.CD_ATENDIMENTO,
		 		PACIENTE.CD_PACIENTE,
		 		PACIENTE.NM_PACIENTE,
		 		PACIENTE.NR_DOCUMENTO,
 				UNID_INT.DS_UNID_INT,
 				LEITO.DS_RESUMO,
        NPP_PEDIATRICO.SN_PERIFERICA,
        NPP_PEDIATRICO.SN_CENTRAL,
        NPP_PEDIATRICO.VL_PPA_PESO,
        NPP_PEDIATRICO.VL_PPA_BH,
        NPP_PEDIATRICO.VL_PPA_PROTEINA,
        NPP_PEDIATRICO.VL_PPA_LIPIDEO,
        NPP_PEDIATRICO.VL_PPA_GLICOSE,
        NPP_PEDIATRICO.VL_PPA_NA,
        NPP_PEDIATRICO.VL_PPA_K,
        NPP_PEDIATRICO.VL_PPA_CA,
        NPP_PEDIATRICO.VL_PPA_PO4,
        NPP_PEDIATRICO.VL_PPA_MG,
        NPP_PEDIATRICO.VL_PPA_ACETATO_SODIO,
        NPP_PEDIATRICO.VL_PPA_OLIGOELEMENTOS,
        NPP_PEDIATRICO.VL_PPA_POLIVITAMINICOA,
        NPP_PEDIATRICO.VL_PPA_POLIVITAMINICOB,
        NPP_PEDIATRICO.VL_PPA_ZINCO,
        NPP_PEDIATRICO.VL_PPA_SELENIO,
        NPP_PEDIATRICO.VL_PPA_POLIVITAMINICOAB,
        NPP_PEDIATRICO.VL_PPA_GLUTAMINA,
        NPP_PEDIATRICO.VL_PPA_HEPARINA,
        NPP_PEDIATRICO.VL_PPA_FC
FROM DBAMV.PRE_MED, DBAMV.ATENDIME, DBAMV.PACIENTE, DBAMV.UNID_INT, DBAMV.LEITO, DBAMV.NPP_PEDIATRICO
WHERE ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
  AND NPP_PEDIATRICO.CD_PRE_MED = PRE_MED.CD_PRE_MED
	AND PRE_MED.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
  AND UNID_INT.CD_UNID_INT =  PRE_MED.CD_UNID_INT
  AND ATENDIME.CD_LEITO = LEITO.CD_LEITO
  AND ATENDIME.CD_MULTI_EMPRESA = NVL(DBAMV.PKG_MV2000.LE_EMPRESA, ATENDIME.CD_MULTI_EMPRESA)
  AND PRE_MED.CD_PRE_MED in (:CAMPO_CD_PRE_MED)
