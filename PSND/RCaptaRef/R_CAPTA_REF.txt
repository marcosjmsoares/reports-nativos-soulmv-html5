--R_CAPTA_REF
SELECT    TR.DS_TIPO_REFEICAO                                          DS_REF
         , TO_CHAR(MVTO.DT_MOV_CARDAPIO,'YYYY/MM')     DT_COMPETE
         , TO_CHAR(ADD_MONTHS( MVTO.DT_MOV_CARDAPIO, -1 ),'YYYY/MM')   DT_ANT
         ,COUNT(*)                                    QTD_REF
         ,VL_PERFIL_PERCAPTA( MVTO.CD_SETOR, TR.CD_TIPO_REFEICAO, TO_CHAR( MVTO.DT_MOV_CARDAPIO, 'YYYY/MM' ) )                VL_UNIT
         ,( COUNT(*) * VL_PERFIL_PERCAPTA( MVTO.CD_SETOR, TR.CD_TIPO_REFEICAO, TO_CHAR( MVTO.DT_MOV_CARDAPIO, 'YYYY/MM' ) ) ) VL_TOT
         ,MVTO.CD_SETOR         CD_SETOR
         ,TR.CD_TIPO_REFEICAO                         CD_REF
FROM      DBAMV.TIPO_REFEICAO  TR
         ,DBAMV.MOV_CARDAPIO   MVTO
         ,DBAMV.SETOR                  SETOR
WHERE     TR.CD_TIPO_REFEICAO                = MVTO.CD_TIPO_REFEICAO
AND        MVTO.CD_SETOR                             = SETOR.CD_SETOR
AND       SETOR.CD_MULTI_EMPRESA          = :CAMPO_CD_MULTI_EMPRESA
AND       TO_CHAR( MVTO.DT_MOV_CARDAPIO, 'YYYY/MM' ) BETWEEN TO_CHAR(:CAMPO_DT_COMPETE_I,'YYYY/MM')  AND TO_CHAR(:CAMPO_DT_COMPETE_F, 'YYYY/MM')
GROUP BY  TR.CD_TIPO_REFEICAO
         ,TR.DS_TIPO_REFEICAO
         ,MVTO.CD_SETOR
         , TO_CHAR(MVTO.DT_MOV_CARDAPIO,'YYYY/MM')
         , TO_CHAR(ADD_MONTHS( MVTO.DT_MOV_CARDAPIO, -1 ),'YYYY/MM')
         ,VL_PERFIL_PERCAPTA( MVTO.CD_SETOR, TR.CD_TIPO_REFEICAO, TO_CHAR( MVTO.DT_MOV_CARDAPIO, 'YYYY/MM' ) )
ORDER BY    TO_CHAR(MVTO.DT_MOV_CARDAPIO,'YYYY/MM') desc
                     ,  TR.DS_TIPO_REFEICAO
