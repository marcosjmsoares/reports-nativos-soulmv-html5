SELECT * FROM (
select unid_int.cd_unid_int,
       unid_int.ds_unid_int
       ,leito.cd_leito cd_leito
      ,leito.ds_leito
      ,tipo_dieta.cd_tipo_dieta
      ,tipo_dieta.ds_tipo_dieta
      ,cd_opcao
     ,ds_opcao
     ,' '   dado
from dbamv.unid_int,dbamv.leito,dbamv.tipo_dieta,dbamv.opcao_cardapio
where unid_int.cd_unid_int = leito.cd_unid_int
order by unid_int.cd_unid_int
)
ORDER BY DS_UNID_INT,
CD_LEITO,
DS_TIPO_DIETA,
DS_OPCAO
