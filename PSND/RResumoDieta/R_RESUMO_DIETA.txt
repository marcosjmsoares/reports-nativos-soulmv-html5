--R_RESUMO_DIETA
SELECT * FROM (
select   mc.cd_tipo_dieta       cd_tipo_dieta
           ,td.ds_tipo_dieta        ds_tipo_dieta
           ,st.cd_setor                cd_setor
           ,st.nm_setor               nm_setor
           ,count(*)                     qtd
           ,DBAMV.PSND_QTD_DIETA(mc.cd_tipo_dieta, :CAMPO_DT_INIC,  :CAMPO_DT_FINAL ) qtd_dieta
from    dbamv.mov_cardapio mc
       ,dbamv.tipo_dieta   td
       ,dbamv.setor        st
where  mc.cd_tipo_dieta  = td.cd_tipo_dieta(+)
and     mc.cd_setor       = st.cd_setor
and   mc.cd_multi_empresa = nvl(dbamv.pkg_mv2000.le_empresa, mc.cd_multi_empresa)
and     trunc( mc.dt_mov_cardapio ) between   :CAMPO_DT_INIC group by  mc.cd_tipo_dieta
               ,td.ds_tipo_dieta
               ,st.cd_setor
               ,st.nm_setor
order by QTD_DIETA, td.ds_tipo_dieta, st.nm_setor
)
ORDER BY QTD_DIETA DESC,DS_TIPO_DIETA,CD_TIPO_DIETA,
NM_SETOR,CD_SETOR
