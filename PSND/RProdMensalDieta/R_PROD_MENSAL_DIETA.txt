--R_PROD_MENSAL_DIETA
SELECT * FROM (
SELECT Decode(mov_cardapio.tp_cardapio,
              'S','Setor',
              'P','Paciente',
              'A','Acompanhante',
              'F','Funcionário') Tipo,
       'Unidade de Internacao' Titulo,
       unid_int.ds_unid_int Descricao,
       tipo_refeicao.cd_tipo_refeicao,
       tipo_refeicao.ds_tipo_refeicao,
       tipo_dieta.cd_tipo_dieta,
       tipo_dieta.ds_tipo_dieta,
       count(nvl(mov_cardapio.cd_tipo_dieta,1))  qtde,
       NVL(excecao_preco_refeicao.vl_refeicao,tipo_refeicao.vl_refeicao) valor
FROM   dbamv.mov_cardapio,
       dbamv.unid_int,
       dbamv.atendime,
       dbamv.leito,
       dbamv.tipo_refeicao,
       dbamv.tipo_dieta,
       dbamv.itmov_cardapio,
       dbamv.excecao_preco_refeicao
WHERE  mov_cardapio.cd_atendimento = atendime.cd_atendimento(+)
AND    atendime.cd_leito = leito.cd_leito(+)
AND    leito.cd_unid_int = unid_int.cd_unid_int(+)
AND   atendime.Cd_Multi_Empresa = nvl(dbamv.pkg_mv2000.le_empresa, atendime.Cd_Multi_Empresa)
AND    mov_cardapio.cd_tipo_refeicao = tipo_refeicao.cd_tipo_refeicao(+)
AND    mov_cardapio.cd_tipo_dieta = tipo_dieta.cd_tipo_dieta(+)
AND    mov_cardapio.cd_mov_cardapio = itmov_cardapio.cd_mov_cardapio(+)
AND    mov_cardapio.tp_movimentacao = 'R'
AND    NVL(itmov_cardapio.cd_itmov_cardapio,0) = (SELECT NVL(MIN(itmov.cd_itmov_cardapio),0)
                                                                                    FROM dbamv.itmov_cardapio itmov
                                                                                    WHERE itmov.cd_mov_cardapio = mov_cardapio.cd_mov_cardapio)
AND    mov_cardapio.cd_tipo_refeicao = excecao_preco_refeicao.cd_tipo_refeicao(+)
AND    mov_cardapio.cd_tipo_dieta = excecao_preco_refeicao.cd_tipo_dieta(+)
AND    TO_CHAR(MOV_CARDAPIO.DT_MOV_CARDAPIO,'MM/YYYY') = TO_CHAR(:CAMPO_DATA,'MM/YYYY')
GROUP BY Decode(mov_cardapio.tp_cardapio,
              'S','Setor',
              'P','Paciente',
              'A','Acompanhante',
              'F','Funcionário'),
       'Unidade de Internacao',
       unid_int.ds_unid_int,
       tipo_refeicao.cd_tipo_refeicao,
       tipo_refeicao.ds_tipo_refeicao,
       tipo_dieta.cd_tipo_dieta,
       tipo_dieta.ds_tipo_dieta,
       NVL(excecao_preco_refeicao.vl_refeicao,tipo_refeicao.vl_refeicao)
UNION
SELECT Decode(mov_cardapio.tp_cardapio,
              'S','Setor',
              'P','Paciente',
              'A','Acompanhante',
              'F','Funcionário') Tipo,
       'Setor' Titulo,
       setor.nm_setor Descricao,
       tipo_refeicao.cd_tipo_refeicao,
       tipo_refeicao.ds_tipo_refeicao,
       tipo_dieta.cd_tipo_dieta,
       tipo_dieta.ds_tipo_dieta,
       count(nvl(mov_cardapio.cd_tipo_dieta,1))  qtde,
       NVL(excecao_preco_refeicao.vl_refeicao,tipo_refeicao.vl_refeicao) valor
FROM   dbamv.mov_cardapio,
       dbamv.tipo_refeicao,
       dbamv.tipo_dieta,
       dbamv.itmov_cardapio,
       dbamv.excecao_preco_refeicao,
       dbamv.setor
WHERE  mov_cardapio.cd_setor = setor.cd_setor
AND    mov_cardapio.cd_tipo_refeicao = tipo_refeicao.cd_tipo_refeicao(+)
AND    mov_cardapio.cd_tipo_dieta = tipo_dieta.cd_tipo_dieta(+)
AND    mov_cardapio.cd_mov_cardapio = itmov_cardapio.cd_mov_cardapio(+)
AND    mov_cardapio.tp_movimentacao = 'R'
AND    NVL(itmov_cardapio.cd_itmov_cardapio,0) = (SELECT NVL(MIN(itmov.cd_itmov_cardapio),0)
                                           FROM dbamv.itmov_cardapio itmov
                                           WHERE itmov.cd_mov_cardapio = mov_cardapio.cd_mov_cardapio)
AND    mov_cardapio.cd_tipo_refeicao = excecao_preco_refeicao.cd_tipo_refeicao(+)
AND    mov_cardapio.cd_tipo_dieta = excecao_preco_refeicao.cd_tipo_dieta(+)
AND    TO_CHAR(MOV_CARDAPIO.DT_MOV_CARDAPIO,'MM/YYYY') = TO_CHAR(:CAMPO_DATA,'MM/YYYY')
GROUP BY Decode(mov_cardapio.tp_cardapio,
              'S','Setor',
              'P','Paciente',
              'A','Acompanhante',
              'F','Funcionário'),
       'Setor',
       setor.nm_setor,
       tipo_refeicao.cd_tipo_refeicao,
       tipo_refeicao.ds_tipo_refeicao,
       tipo_dieta.cd_tipo_dieta,
       tipo_dieta.ds_tipo_dieta,
       NVL(excecao_preco_refeicao.vl_refeicao,tipo_refeicao.vl_refeicao)
UNION
SELECT 'Mamadeira' Tipo,
       'Prato' Titulo,
       Prato.ds_prato Descricao,
       tipo_refeicao.cd_tipo_refeicao,
       tipo_refeicao.ds_tipo_refeicao,
       tipo_dieta.cd_tipo_dieta,
       tipo_dieta.ds_tipo_dieta,
       SUM(1) qtde,
       NVL(excecao_preco_refeicao.vl_refeicao,tipo_refeicao.vl_refeicao) valor
FROM   dbamv.mov_cardapio,
       dbamv.Prato,
       dbamv.tipo_refeicao,
       dbamv.tipo_dieta,
       dbamv.itmov_lactario,
       dbamv.excecao_preco_refeicao,
       dbamv.atendime,
       dbamv.leito,
       dbamv.unid_int
WHERE  mov_cardapio.cd_tipo_refeicao = tipo_refeicao.cd_tipo_refeicao(+)
AND    mov_cardapio.cd_tipo_dieta = tipo_dieta.cd_tipo_dieta(+)
AND    mov_cardapio.cd_tipo_refeicao = excecao_preco_refeicao.cd_tipo_refeicao(+)
AND    mov_cardapio.cd_tipo_dieta = excecao_preco_refeicao.cd_tipo_dieta(+)
AND    mov_cardapio.cd_atendimento = atendime.cd_atendimento
AND   atendime.Cd_Multi_Empresa = nvl(dbamv.pkg_mv2000.le_empresa, atendime.Cd_Multi_Empresa)
AND    atendime.cd_leito = leito.cd_leito
AND    leito.cd_unid_int = unid_int.cd_unid_int
AND    mov_cardapio.tp_movimentacao = 'M'
AND    mov_cardapio.cd_mov_cardapio = itmov_lactario.cd_mov_cardapio(+)
AND    itmov_lactario.cd_prato = prato.cd_prato
AND    TO_CHAR(MOV_CARDAPIO.DT_MOV_CARDAPIO,'MM/YYYY') = TO_CHAR(:CAMPO_DATA,'MM/YYYY')
GROUP BY Decode(mov_cardapio.tp_cardapio,
              'S','Setor',
              'P','Paciente',
              'A','Acompanhante',
              'F','Funcionário'),
       'Prato',
       Prato.Ds_Prato,
       tipo_refeicao.cd_tipo_refeicao,
       tipo_refeicao.ds_tipo_refeicao,
       tipo_dieta.cd_tipo_dieta,
       tipo_dieta.ds_tipo_dieta,
       NVL(excecao_preco_refeicao.vl_refeicao,tipo_refeicao.vl_refeicao)
)
ORDER BY TIPO,TITULO,
DESCRICAO
