--R_PLANO_CARDAPIO
select * from (
select clas_cardapio.ds_clas_cardapio	   ds_clas_cardapio
        ,clas_cardapio.cd_clas_cardapio    cd_clas_cardapio
	,plano_cardapio.cd_setor           cd_setor
	,plano_cardapio.qt_porcao          qt_porcao
        ,setor.nm_setor                    nm_setor
	,plano_cardapio.cd_plano_cardapio  cd_plano_cardapio
        ,plano_cardapio.ds_plano_cardapio  ds_plano_cardapio
        ,plano_cardapio_opcao.cd_opcao     cd_opcao
	,opcao_cardapio.ds_opcao	   ds_opcao
	,plano_cardapio.dt_previsao_cardapio dt_previsao_cardapio
,PLANO_CARDAPIO_OPCAO.CD_PLANO_CARDAPIO_OPCAO CD_PLANO_CARDAPIO_OPCAO
      	from dbamv.plano_cardapio
	    ,dbamv.plano_cardapio_opcao
	    ,dbamv.opcao_cardapio
	    ,dbamv.clas_cardapio
	    ,dbamv.setor
	 where plano_cardapio.cd_clas_cardapio          =  clas_cardapio.cd_clas_cardapio
         and plano_cardapio.cd_setor                  = setor.cd_setor
         and plano_cardapio_opcao.cd_plano_cardapio   = plano_cardapio.cd_plano_cardapio
         and plano_cardapio_opcao.cd_opcao                = opcao_cardapio.cd_opcao
         :CAMPO_CF_PLANO
)
order by cd_plano_cardapio, ds_plano_cardapio
