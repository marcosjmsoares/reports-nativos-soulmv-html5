SELECT
    mov_cardapio.cd_copa,
    to_char(mov_cardapio.dt_mov_cardapio, 'dd/mm/yyyy') dt_mov_cardapio,
    mov_cardapio.cd_tipo_refeicao,
    mov_cardapio.tp_cardapio,
    mov_cardapio.cd_multi_empresa,
    to_char(atendime.dt_alta_medica, 'dd/mm/yyyy hh24') dt_alta_medica,
    atendime.dh_alta_medica_lancada,
    decode(atendime.tp_atendimento, 'I', 'Unid.Internação', 'Ambulatório / Urgência / Externo') tp_atendimento,
    atendime.cd_atendimento,
    atendime.cd_ori_ate,
    atendime.cd_leito,
    copa.ds_copa,
    convenio.nm_convenio,
    leito.ds_enfermaria,
    leito.cd_unid_int,
    leito.ds_resumo ds_leito,
    Decode(leito.tp_ocupacao,'I', '(Infectado)', NULL) sn_infectado,
    leito.tp_ocupacao,
    unid_int.ds_unid_int,
    paciente.nm_paciente,
    paciente.dt_nascimento,
    rtrim(pack_internamento.retorna_idade(paciente.dt_nascimento,sysdate),' ')  nr_idade
  FROM dbamv.atendime
    , dbamv.mov_cardapio
    , dbamv.copa copa
    , dbamv.paciente
    , dbamv.convenio
    , dbamv.leito
    , dbamv.unid_int
    , dbamv.tipo_dieta
 WHERE tipo_dieta.cd_tipo_dieta = mov_cardapio.cd_tipo_dieta
   AND convenio.cd_convenio = atendime.cd_convenio
   AND unid_int.cd_unid_int (+) = leito.cd_unid_int
   AND leito.cd_leito (+)    = atendime.cd_leito
   AND paciente.cd_paciente = atendime.cd_paciente
   AND copa.cd_copa(+) = mov_cardapio.cd_copa
   AND atendime.dt_alta IS NULL
   AND atendime.cd_atendimento = mov_cardapio.cd_atendimento
   AND mov_cardapio.SN_PRESCRICAO_SUSPENSA = 'N'
   AND mov_cardapio.tp_movimentacao = 'R'
   AND mov_cardapio.tp_cardapio = 'P'
   AND mov_cardapio.cd_multi_empresa = pkg_mv2000.le_empresa
GROUP BY
   mov_cardapio.cd_copa,
   to_char(mov_cardapio.dt_mov_cardapio, 'dd/mm/yyyy'),
   mov_cardapio.cd_tipo_refeicao,
   mov_cardapio.tp_cardapio,
   mov_cardapio.cd_multi_empresa,
   to_char(atendime.dt_alta_medica, 'dd/mm/yyyy hh24'),
   atendime.dh_alta_medica_lancada,
   decode(atendime.tp_atendimento, 'I', 'Unid.Internação', 'Ambulatório / Urgência / Externo'),
   atendime.cd_atendimento,
   atendime.cd_ori_ate,
   atendime.cd_leito,
   copa.ds_copa,
   convenio.nm_convenio,
   leito.ds_enfermaria,
   leito.cd_unid_int,
   leito.ds_resumo,
   Decode(leito.tp_ocupacao,'I', '(Infectado)', NULL),
   leito.tp_ocupacao,
   unid_int.ds_unid_int,
   paciente.nm_paciente,
   paciente.dt_nascimento,
   rtrim(pack_internamento.retorna_idade(paciente.dt_nascimento,sysdate),' ')
