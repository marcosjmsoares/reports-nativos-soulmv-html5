--R_PRESC_NPP_ADULTO
SELECT PRE_MED.CD_PRE_MED,
		 		To_Date((To_Char(PRE_MED.DT_PRE_MED, 'dd/mm/yy') || ' ' ||to_char(PRE_MED.HR_PRE_MED ,'HH24')), 'dd/mm/yy HH24') DT_PRE_MED,
		 		PRE_MED.CD_ATENDIMENTO,
		 		PACIENTE.CD_PACIENTE,
		 		PACIENTE.NM_PACIENTE,
		 		PACIENTE.NR_DOCUMENTO,
 				UNID_INT.DS_UNID_INT,
 				LEITO.DS_RESUMO,
        NPP_ADULTO.SN_PERIFERICA,
        NPP_ADULTO.SN_CENTRAL,
        NPP_ADULTO.NR_DIA_PRESCRICAO,
        NPP_ADULTO.VL_PPA_PESO,
        NPP_ADULTO.VL_PPA_CALORIA,
        NPP_ADULTO.VL_PPA_PROTEINA,
        NPP_ADULTO.VL_PPA_LIPIDEO,
        NPP_ADULTO.VL_PPA_ZINCO,
        NPP_ADULTO.VL_PPA_POLIVITAMINICOA,
        NPP_ADULTO.VL_PPA_POLIVITAMINICOB,
        NPP_ADULTO.VL_PPA_POLIVITAMINICOAB,
        NPP_ADULTO.VL_COMPLEXO_B,
        NPP_ADULTO.VL_ACETATO_SODIO VL_PPA_ACETATO_SODIO,
        NPP_ADULTO.VL_CLORETO_SODIO,
        NPP_ADULTO.VL_CLORETO_POTASSIO,
        NPP_ADULTO.VL_PPA_FOSFATO_POTASSIO,
        NPP_ADULTO.VL_FOSFATO_ORGANICO,
        NPP_ADULTO.VL_SULF_MAGNESIO,
        NPP_ADULTO.VL_GLUCONATO_CALCIO,
        NPP_ADULTO.VL_AGUA_DESTILADA,
      	NPP_ADULTO.VL_PPA_OLIGOELEMENTOS,
      	NPP_ADULTO.VL_PORC_GLICOSE_OSMO,
      	NPP_ADULTO.VL_PORC_LIPIDEO_OSMO,
        NPP_ADULTO.DS_OUTROS_OLIGOELEMENTOS,
	NPP_ADULTO.VL_OUTROS_OLIGOELEMENTOS,
	NPP_ADULTO.DS_OUTROS_VITAMINAS,
	NPP_ADULTO.VL_OUTROS_VITAMINAS,
	NPP_ADULTO.DS_OUTROS_VITAMINAS1,
	NPP_ADULTO.VL_OUTROS_VITAMINAS1,
        NPP_ADULTO.CD_PRESTADOR_GRAVA_CALCULO,
        NPP_ADULTO.VL_INSULINA
FROM DBAMV.PRE_MED, DBAMV.ATENDIME, DBAMV.PACIENTE, DBAMV.UNID_INT, DBAMV.LEITO, DBAMV.NPP_ADULTO
WHERE ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
  AND NPP_ADULTO.CD_PRE_MED = PRE_MED.CD_PRE_MED
	AND PRE_MED.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
  AND UNID_INT.CD_UNID_INT =  PRE_MED.CD_UNID_INT
  AND ATENDIME.CD_LEITO = LEITO.CD_LEITO
  AND ATENDIME.CD_MULTI_EMPRESA = NVL(DBAMV.PKG_MV2000.LE_EMPRESA, ATENDIME.CD_MULTI_EMPRESA)
  AND PRE_MED.CD_PRE_MED in (:CAMPO_CD_PRE_MED)
