--R_EST_ENTRADA_PRODUTO_SINT
SELECT * FROM (
select fornecedor.cd_fornecedor,
       fornecedor.nm_fornecedor  ,
       tip_doc.ds_tip_doc,
       ent_pro.nr_documento ,
       ent_pro.nr_serie ,
       ent_pro.dt_emissao dt_emissao,
       ent_pro.dt_entrada,
      sum(  nvl( itent_pro.vl_custo_real, 0) * nvl( itent_pro.qt_entrada,0)) vl_total
from dbamv.fornecedor,
       dbamv.tip_doc,
       dbamv.itent_pro,
       dbamv.ent_pro,
       dbamv.estoque
WHERE  ent_pro.cd_ent_pro    = itent_pro.cd_ent_pro
  and  Fornecedor.Tp_Cliente_Forn In  ( 'F', 'A', 'R', 'T' )
  and  ent_pro.cd_fornecedor = fornecedor.cd_fornecedor
  and  ent_pro.cd_tip_doc = tip_doc.cd_tip_doc
  and ent_pro.cd_fornecedor is not null
  and ent_pro.cd_estoque = estoque.cd_estoque
  and Estoque.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
Group by fornecedor.cd_fornecedor, fornecedor.nm_fornecedor, tip_doc.ds_tip_doc, ent_pro.nr_documento, ent_pro.nr_serie, ent_pro.dt_emissao, ent_pro.dt_entrada
:CAMPO_CF_TP_ORDEM
)
ORDER BY CD_FORNECEDOR,NM_FORNECEDOR, ds_tip_doc, nr_documento , nr_serie , dt_emissao, dt_entrada, vl_total
