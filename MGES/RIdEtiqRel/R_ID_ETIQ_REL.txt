--R_ID_ETIQ_REL
SELECT * FROM (
select distinct lp.cd_estoque
               ,e.ds_estoque
               ,id.cd_produto
               ,nvl(p.ds_produto_resumido,p.ds_produto) ds_produto
               ,id.cd_identificador
               ,up.ds_unidade
               ,id.cd_lote
               ,id.dt_validade
from            dbamv.lot_pro lp
               ,dbamv.estoque e
               ,dbamv.identificador_etiqueta id
               ,dbamv.produto p
               ,dbamv.uni_pro up
where           id.cd_produto                = lp.cd_produto
and             nvl(id.cd_lote,'ok')         = nvl(lp.cd_lote,'ok')
and             nvl(id.dt_validade, sysdate) = nvl(lp.dt_validade, sysdate)
and             lp.cd_estoque                = e.cd_estoque
and             id.cd_produto                = p.cd_produto
and             id.cd_uni_pro                = up.cd_uni_pro
and             lp.qt_estoque_atual <> 0
and             p.sn_imprime_etiqueta in ('S', 'F')
and             e.Cd_Multi_Empresa     = :CAMPO_CD_MULTI_EMPRESA
AND             id.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
order by lp.cd_estoque,
nvl(p.ds_produto_resumido,p.ds_produto), id.cd_identificador
)
ORDER BY CD_ESTOQUE,DS_ESTOQUE,DS_PRODUTO,CD_IDENTIFICADOR
