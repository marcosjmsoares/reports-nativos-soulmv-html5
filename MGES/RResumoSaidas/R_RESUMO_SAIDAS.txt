--R_RESUMO_SAIDAS
SELECT * FROM (
Select  itmvto_estoque.cd_produto
           ,produto.ds_produto
           ,sum((itmvto_estoque.qt_movimentacao *  uni_pro.vl_fator)/ verif_vl_fator_prod(produto.cd_produto)) qt_movimentacao
           ,setor.cd_setor
           ,setor.nm_setor
           ,SUM(( itmvto_estoque.qt_movimentacao *  uni_pro.vl_fator) *
                     Verif_Vl_Custo_Medio( ItMvto_Estoque.Cd_Produto, Mvto_Estoque.Dt_Mvto_Estoque,
                                                          :CAMPO_TP_CUSTO, produto.vl_custo_medio, Mvto_Estoque.Hr_Mvto_Estoque )) vl_total
           ,verif_ds_unid_prod(produto.cd_produto) ds_unidade
   From dbamv.mvto_estoque
           ,dbamv.itmvto_estoque
           ,dbamv.setor
           ,dbamv.produto
           ,dbamv.uni_pro
           ,dbamv.estoque
Where mvto_estoque.tp_mvto_estoque  = 'S'
   AND estoque.cd_multi_empresa           = :CAMPO_CD_MULTI_EMPRESA
   AND mvto_estoque.dt_mvto_estoque  BETWEEN :CAMPO_DTINIC
                                                                           AND 
   AND mvto_estoque.cd_mvto_estoque = itmvto_estoque.cd_mvto_estoque
   AND mvto_estoque.cd_setor                = setor.cd_setor
   AND produto.cd_produto                      = itmvto_estoque.cd_produto
   AND uni_pro.cd_uni_pro                       = itmvto_estoque.cd_uni_pro
   AND mvto_estoque.cd_estoque           = estoque.cd_estoque
Group by produto.ds_produto
              ,itmvto_estoque.cd_produto
              ,setor.cd_setor
              ,setor.nm_setor
              ,verif_ds_unid_prod(produto.cd_produto)
Order by setor.nm_setor
              ,produto.ds_produto
)
ORDER BY NM_SETOR,CD_SETOR, DS_PRODUTO
