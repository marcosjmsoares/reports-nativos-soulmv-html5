--R_LISTA_DCB
SELECT * FROM (
SELECT E.CD_LISTA_CODIGO_MEDICAMENTO CD_CODIGO,
               E.DS_LISTA_CODIGO_MEDICAMENTO DS_DESCRICAO,
               PRODUTO.CD_PRODUTO,
               PRODUTO.DS_PRODUTO,
               PRODUTO.CD_DCB
FROM DBAMV.PRODUTO,
           DBAMV.LISTA_CODIGO_MEDICAMENTO E
WHERE PRODUTO.CD_LISTA_CODIGO_MEDICAMENTO = E.CD_LISTA_CODIGO_MEDICAMENTO
And Exists ( Select Empresa_Produto.Cd_Produto
                     From Dbamv.Empresa_Produto
                   Where Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
                        and Empresa_Produto.Cd_Produto = Produto.Cd_Produto )
:CAMPO_CF_P_DETALHES
)
ORDER BY :CAMPO_CF_ORDENA, DS_PRODUTO
