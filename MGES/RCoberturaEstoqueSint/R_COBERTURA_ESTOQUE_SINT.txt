--R_COBERTURA_ESTOQUE_SINT
SELECT
  indCobSalSIN.cd_estoque,
  estSIN.ds_estoque,
  Count(itIndCodSalSIN.cd_produto) qt_estoque,
  (SELECT  Count( DISTINCT itIndCodSalINT.cd_produto) FROM
				dbamv.item_indice_cobertura_saldo itIndCodSalINT,
				dbamv.indice_cobertura_saldo indCobSalINT,
				dbamv.estoque estINT,
				dbamv.produto prodINT
			WHERE itIndCodSalINT.cd_indice = indCobSalINT.cd_indice
			AND itIndCodSalINT.cd_produto = prodINT.cd_produto
			AND indCobSalINT.cd_estoque = estINT.cd_estoque
			AND itIndCodSalINT.sn_tem_saldo = 'S'
			AND indCobSalINT.cd_estoque = indCobSalSIN.cd_estoque) qt_estoque_c_saldo
FROM
  dbamv.item_indice_cobertura_saldo itIndCodSalSIN,
  dbamv.indice_cobertura_saldo indCobSalSIN,
  dbamv.estoque estSIN
WHERE itIndCodSalSIN.cd_indice = indCobSalSIN.cd_indice
AND indCobSalSIN.cd_estoque = estSIN.cd_estoque
GROUP BY
  indCobSalSIN.cd_estoque,
  estSIN.ds_estoque
ORDER BY
  indCobSalSIN.cd_estoque
