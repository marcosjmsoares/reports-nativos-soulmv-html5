--R_REQUISICAO_PRODUTO
SELECT 	req.cd_requisicao cd_requisicao,
	req.cd_estoque_solic cd_estoque_solic,
         est_sol.ds_estoque ds_estoque_solic,
        req.cd_estoque_dest cd_estoque_dest,
         est_des.ds_estoque ds_estoque_dest,
         req.cd_setor_destino  cd_setor_destino,
         setor.nm_setor  nm_setor,
          To_Char(req.dt_requisicao, 'DD/MM/YYYY HH24') dt_requisicao,
          req.ds_observacao ds_observacao,
    req.cd_usuario_ativo cd_usuario_ativo,
       Decode(req.tp_status, 'A','ABERTA',
                             'L', 'LIBERADA',
                             'C', 'CANCELADA',
                             'T', 'ABERTA PARCIALMENTE ATENDIDA',
                             'P', 'LIBERADA PARCIAL',
                             'E', 'ATENDIDA',
                             'R', 'ATENDIDA PARCIAL',
                             'N', 'ENVIADA PARA REGIONAL',
                             'V', 'ENVIADA PARCIALMENTE',
                             'S', 'RECEBIDA NA REGIONAL',
                             'B', 'RECEBIDA PARCIALMENTE') tp_status
FROM dbamv.requisicao_produto req,
            dbamv.estoque est_sol,
            dbamv.estoque est_des,
            dbamv.setor setor
WHERE cd_requisicao 
AND est_sol.cd_estoque = req.cd_estoque_solic
AND est_des.cd_estoque(+) = req.cd_estoque_dest
AND setor.cd_setor(+) = req.cd_setor_destino
