--R_SAI_DEV_ESP_CONSO
SELECT * FROM (
Select especie.cd_especie
          ,especie.ds_especie
          ,c_conset.cd_produto
          ,produto.ds_produto ds_produto
          ,verif_ds_unid_prod(c_conset.cd_produto)  ds_unidade
          ,sum(nvl(qt_saida_setor,0)
          + nvl( qt_saida_paciente,0)
          + nvl( qt_venda,0)) qt_saida
          ,sum( nvl(qt_devolvida_setor,0)
          + nvl( qt_devolvida_paciente,0)
          + nvl( qt_devolucao_venda,0)) qt_devolvida
           ,estoque.cd_estoque
           ,estoque.ds_estoque
           ,sum(nvl( vl_saida_setor,0)
           + nvl( vl_saida_paciente,0)
           + nvl( vl_venda,0)) vl_saida
           ,sum(nvl( vl_devolvida_setor,0)
           + nvl( vl_devolvida_paciente,0)
           + nvl( vl_devolucao_venda,0)) vl_devolvida
           ,verif_vl_fator_prod(c_conset.cd_produto) vl_fator
          From
           dbamv.c_conset
         , dbamv.setor
         , dbamv.produto
         , dbamv.especie
         , dbamv.estoque
Where
          c_conset.cd_ano|| to_char(c_conset.cd_mes,'fm09') between to_char(:CAMPO_DTINIC,'YYYYMM') AND to_char(:CAMPO_DTFIM ,'YYYYMM')
and       c_conset.cd_setor                = setor.cd_setor
and       produto.cd_produto                      = c_conset.cd_produto
and       produto.cd_especie                      = especie.cd_especie
and       c_conset.cd_Estoque = estoque.cd_estoque
and       estoque.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
group by
           especie.cd_especie
          ,especie.ds_especie
          ,c_conset.cd_produto
          ,produto.ds_produto
          ,verif_ds_unid_prod(c_conset.cd_produto)
          ,estoque.cd_estoque
           ,estoque.ds_estoque
order by
  produto.ds_produto
)
ORDER BY CD_ESTOQUE,DS_ESTOQUE,
CD_ESPECIE,DS_ESPECIE, DS_PRODUTO
