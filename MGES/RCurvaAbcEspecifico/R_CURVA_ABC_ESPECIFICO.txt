SELECT   ESPECIE.DS_ESPECIE DS_ESPECIE,
     SUM(C_CONEST.vl_ESTOQUE_INICIAL)		QT_ESTOQUE_INICIAL, 
     SUM(C_CONEST.vl_ESTOQUE_FINAL)		QT_ESTOQUE_FINAL,
     SUM(C_CONEST.vl_ENTRADA)			QT_ENTRADA,
     SUM(C_CONEST.vl_SAIDA_SETOR)		QT_SAIDA_SETOR,
     SUM(C_CONEST.vl_SAIDA_PACIENTE)		QT_SAIDA_PACIENTE,
     SUM(C_CONEST.vl_BAIXA)			QT_BAIXA,
     SUM(C_CONEST.vl_EMPRESTIMO)			QT_EMPRESTIMO_SAIDA,
     SUM(C_CONEST.vl_EMPRESTIMO_entrada)   	QT_EMPRESTIMO_ENTRADA,
     SUM(C_CONEST.vl_DEVOLUCAO_SETOR)		QT_DEVOLUCAO_SETOR, 
     SUM(C_CONEST.vl_DEVOLUCAO_PACIENTE)	QT_DEVOLUCAO_PACIENTE,
     SUM(C_CONEST.vl_DOACAO)			QT_DOACAO,
     SUM(C_CONEST.vl_CONTAGEM_ENTRADA)		QT_CONTAGEM_ENTRADA,
     SUM(C_CONEST.vl_CONTAGEM_SAIDA)		QT_CONTAGEM_SAIDA,
     SUM(C_CONEST.vl_DEVOLVIDA_FORNEC)		QT_DEVOLVIDA_FORNEC
FROM Dbamv.C_CONEST,
           Dbamv.PRODUTO,
           DBAMV.ESPECIE,
           Dbamv.ESTOQUE
WHERE C_CONEST.CD_PRODUTO            = PRODUTO.CD_PRODUTO
      AND ESPECIE.CD_ESPECIE                   = PRODUTO.CD_ESPECIE       
      AND  PRODUTO.SN_MESTRE               = 'N'
      AND ESTOQUE.CD_ESTOQUE              = C_CONEST.CD_ESTOQUE 
      AND C_CONEST.CD_MES                      = :CAMPO_DATA_INI
      AND C_CONEST.CD_ANO                      = :CAMPO_DATA_FIM
      AND ESTOQUE.CD_MULTI_EMPRESA  =  :CAMPO_CD_MULTI_EMPRESA
GROUP BY ESPECIE.DS_ESPECIE
ORDER BY ESPECIE.DS_ESPECIE
