--R_LISTA_PRODUTOS
SELECT * FROM (
	SELECT  PRODUTO.CD_PRODUTO CODIGO
	       ,PRODUTO.DS_PRODUTO DESCRICAO
	       ,PRODUTO.SN_MESTRE SN_MESTRE
	       ,PRODUTO.CD_SUB_CLA CD_SUB_CLA
	       ,PRODUTO.CD_CLASSE CD_CLASSE
	       ,PRODUTO.CD_ESPECIE CD_ESPECIE
	FROM    dbamv.PRODUTO PRODUTO,
			dbamv.Empresa_Produto EMPRESA_PRODUTO
	WHERE  DECODE(DBAMV.PKT_CONFIGEST.RETORNA_CAMPO('SN_LOTE_VAL_PRD_EMP'), 'S',EMPRESA_PRODUTO.SN_MOVIMENTACAO,PRODUTO.SN_MOVIMENTACAO) = 'S'
	And EMPRESA_PRODUTO.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
	AND EMPRESA_PRODUTO.Cd_Produto = PRODUTO.Cd_Produto 
	ORDER BY  CD_ESPECIE, CD_CLASSE, CD_SUB_CLA
	)
	ORDER BY CD_ESPECIE,
	CD_CLASSE,
	CD_SUB_CLA
