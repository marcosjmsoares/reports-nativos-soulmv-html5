   And Exists (Select 1
                 From DBAMV.ITENT_PRO_IMPOSTO
                Where ITENT_PRO_IMPOSTO.CD_ITENT_PRO = ITENT_PRO.CD_ITENT_PRO)
Inner Join DBAMV.ESTOQUE On
       ESTOQUE.CD_ESTOQUE = ENT_PRO.CD_ESTOQUE
Inner Join DBAMV.MULTI_EMPRESAS On
       MULTI_EMPRESAS.CD_MULTI_EMPRESA = ESTOQUE.CD_MULTI_EMPRESA
Inner Join DBAMV.TIP_DOC On

Inner Join DBAMV.PRODUTO On

Inner Join DBAMV.UNI_PRO On
       UNI_PRO.CD_UNI_PRO = ITENT_PRO.CD_UNI_PRO
Left Join DBAMV.CFOP On
       CFOP.CD_CFOP = ITENT_PRO.CD_CFOP
Left Join DBAMV.ORIGEM_MERCADORIA On
       ORIGEM_MERCADORIA.CD_ORIGEM_MERCADORIA = ITENT_PRO.CD_ORIGEM_MERCADORIA
Left Join DBAMV.ENT_PRO_MOEDA On
       ENT_PRO_MOEDA.CD_ENT_PRO = ENT_PRO.CD_ENT_PRO
Left Join DBAMV.FORNECEDOR On
       FORNECEDOR.CD_FORNECEDOR = ENT_PRO.CD_FORNECEDOR

       ENT_PRO_ACERTO.CD_ENT_PRO = ENT_PRO.CD_ENT_PRO

       MOTIVO_ACERTO_ESTOQUE.CD_MOTIVO_ACERTO_ESTOQUE = ENT_PRO_ACERTO.CD_MOTIVO_ACERTO_ESTOQUE

       ORD_COM.CD_ORD_COM = ENT_PRO.CD_ORD_COM

                              And Trunc(:CAMPO_DTFIM) + 0.99999
