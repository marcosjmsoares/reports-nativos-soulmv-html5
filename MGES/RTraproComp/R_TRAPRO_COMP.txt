--R_TRAPRO_COMP
SELECT
       MVTO_ESTOQUE.CD_USUARIO,
       MVTO_ESTOQUE.CD_MVTO_ESTOQUE,
       MVTO_ESTOQUE.CD_SOLSAI_PRO,
       MVTO_ESTOQUE.CD_ESTOQUE       CD_ESTOQUE_ORIGEM,
       ESTOQUE_ORIGEM.DS_ESTOQUE  DS_ESTOQUE_ORIGEM ,
       MVTO_ESTOQUE.CD_ESTOQUE_DESTINO,
       ESTOQUE_DESTINO.DS_ESTOQUE   DS_ESTOQUE_DESTINO,
       MVTO_ESTOQUE.DT_MVTO_ESTOQUE,
       MVTO_ESTOQUE.HR_MVTO_ESTOQUE,
       MVTO_ESTOQUE.DT_IMPRESSAO,
       MVTO_ESTOQUE.NR_DOCUMENTO,
       MVTO_ESTOQUE.DS_OBS_EMPRESTIMO  DS_OBSERVACAO,
       MOTIVO_DIVERG_ATEND.DS_MOTIVO_DIVERG_ATEND,       
       ITMVTO_ESTOQUE.CD_PRODUTO,
       PRODUTO.DS_PRODUTO,
       ITMVTO_ESTOQUE.CD_PRODUTO_KIT              CD_PRODUTO_KIT,
       PRODUTO_KIT.DS_PRODUTO                             DS_PRODUTO_KIT,
       UNI_PRO.DS_UNIDADE,
       ITMVTO_ESTOQUE.CD_LOTE,
       ITMVTO_ESTOQUE.DT_VALIDADE,
       SUM(ITMVTO_ESTOQUE.QT_MOVIMENTACAO) QT_MOVIMENTACAO1,
       VERIF_VL_CUSTO_MEDIO(ITMVTO_ESTOQUE.CD_PRODUTO,MVTO_ESTOQUE.DT_MVTO_ESTOQUE,'H',NULL,MVTO_ESTOQUE.HR_MVTO_ESTOQUE) * UNI_PRO.VL_FATOR VL_CUSTO_MEDIO
FROM
       DBAMV.MVTO_ESTOQUE    MVTO_ESTOQUE,
       DBAMV.ITMVTO_ESTOQUE  ITMVTO_ESTOQUE,
       DBAMV.ESTOQUE         ESTOQUE_ORIGEM,
       DBAMV.UNI_PRO         UNI_PRO,
       DBAMV.PRODUTO         PRODUTO,
       DBAMV.PRODUTO         PRODUTO_KIT,
       DBAMV.ESTOQUE         ESTOQUE_DESTINO,
       DBAMV.ITSOLSAI_PRO    ITSOLSAI_PRO, 
       DBAMV.MOTIVO_DIVERG_ATEND MOTIVO_DIVERG_ATEND 
WHERE
       MVTO_ESTOQUE.TP_MVTO_ESTOQUE    =  'T'
AND    MVTO_ESTOQUE.CD_MVTO_ESTOQUE    =  ITMVTO_ESTOQUE.CD_MVTO_ESTOQUE
AND    MVTO_ESTOQUE.CD_ESTOQUE         =  ESTOQUE_ORIGEM.CD_ESTOQUE
AND    MVTO_ESTOQUE.CD_ESTOQUE_DESTINO =  ESTOQUE_DESTINO.CD_ESTOQUE
AND    ITMVTO_ESTOQUE.CD_ITSOLSAI_PRO  =  ITSOLSAI_PRO.CD_ITSOLSAI_PRO(+) 
AND    ITSOLSAI_PRO.CD_MOTIVO_DIVERG_ATEND = MOTIVO_DIVERG_ATEND.CD_MOTIVO_DIVERG_ATEND(+)   
AND    ITMVTO_ESTOQUE.CD_UNI_PRO       =  UNI_PRO.CD_UNI_PRO(+)
AND    ITMVTO_ESTOQUE.CD_PRODUTO       =  PRODUTO.CD_PRODUTO(+)
AND    MVTO_ESTOQUE.CD_MVTO_ESTOQUE    =  :CAMPO_CD_MVTO_ESTOQUE
AND    ITMVTO_ESTOQUE.CD_PRODUTO_KIT   =  PRODUTO_KIT.CD_PRODUTO(+)
GROUP BY
       MVTO_ESTOQUE.CD_USUARIO,
       MVTO_ESTOQUE.CD_MVTO_ESTOQUE,
       MVTO_ESTOQUE.CD_SOLSAI_PRO,
       MVTO_ESTOQUE.CD_ESTOQUE,
       ESTOQUE_ORIGEM.DS_ESTOQUE,
       MVTO_ESTOQUE.CD_ESTOQUE_DESTINO,
       ESTOQUE_DESTINO.DS_ESTOQUE,
       MVTO_ESTOQUE.DT_MVTO_ESTOQUE,
       MVTO_ESTOQUE.HR_MVTO_ESTOQUE,
       MVTO_ESTOQUE.DT_IMPRESSAO,
       MVTO_ESTOQUE.NR_DOCUMENTO,
       MVTO_ESTOQUE.DS_OBS_EMPRESTIMO,
       MOTIVO_DIVERG_ATEND.DS_MOTIVO_DIVERG_ATEND,   
       ITMVTO_ESTOQUE.CD_PRODUTO,
       PRODUTO.DS_PRODUTO,
       ITMVTO_ESTOQUE.CD_PRODUTO_KIT,
       PRODUTO_KIT.DS_PRODUTO,
       UNI_PRO.DS_UNIDADE,
       ITMVTO_ESTOQUE.CD_LOTE,
       ITMVTO_ESTOQUE.DT_VALIDADE,
       VERIF_VL_CUSTO_MEDIO( ITMVTO_ESTOQUE.CD_PRODUTO
                           , MVTO_ESTOQUE.DT_MVTO_ESTOQUE, 'H', NULL
                           , MVTO_ESTOQUE.HR_MVTO_ESTOQUE) * UNI_PRO.VL_FATOR
ORDER BY CD_MVTO_ESTOQUE,CD_PRODUTO,CD_ESTOQUE_ORIGEM,DS_ESTOQUE_ORIGEM,DS_ESTOQUE_DESTINO,CD_ESTOQUE_DESTINO,CD_USUARIO,DT_MVTO_ESTOQUE,HR_MVTO_ESTOQUE,CD_SOLSAI_PRO,DT_IMPRESSAO,NR_DOCUMENTO,
DS_PRODUTO

--R_TRAPRO_COMP_R_PRODUTO_KIT_Q_MOV_ESTOQUE
SELECT
       MVTO_ESTOQUE.CD_USUARIO,
       MVTO_ESTOQUE.CD_MVTO_ESTOQUE,
       MVTO_ESTOQUE.CD_SOLSAI_PRO,
       MVTO_ESTOQUE.CD_ESTOQUE       CD_ESTOQUE_ORIGEM,
       ESTOQUE_ORIGEM.DS_ESTOQUE  DS_ESTOQUE_ORIGEM ,
       MVTO_ESTOQUE.CD_ESTOQUE_DESTINO,
       ESTOQUE_DESTINO.DS_ESTOQUE   DS_ESTOQUE_DESTINO,
       MVTO_ESTOQUE.DT_MVTO_ESTOQUE,
       MVTO_ESTOQUE.HR_MVTO_ESTOQUE,
       MVTO_ESTOQUE.DT_IMPRESSAO,
       MVTO_ESTOQUE.NR_DOCUMENTO,
       MOTIVO_DIVERG_ATEND.DS_MOTIVO_DIVERG_ATEND,       
       ITMVTO_ESTOQUE.CD_PRODUTO,
       PRODUTO.DS_PRODUTO,
       ITMVTO_ESTOQUE.CD_PRODUTO_KIT              CD_PRODUTO_KIT,
       PRODUTO_KIT.DS_PRODUTO                             DS_PRODUTO_KIT,
       UNI_PRO.DS_UNIDADE,
       ITMVTO_ESTOQUE.CD_LOTE,
       ITMVTO_ESTOQUE.DT_VALIDADE,
       ITMVTO_ESTOQUE.DS_ENDERECO,
       SUM(ITMVTO_ESTOQUE.QT_MOVIMENTACAO) QT_MOVIMENTACAO1,
       VERIF_VL_CUSTO_MEDIO(ITMVTO_ESTOQUE.CD_PRODUTO,MVTO_ESTOQUE.DT_MVTO_ESTOQUE,'H',NULL,MVTO_ESTOQUE.HR_MVTO_ESTOQUE) * UNI_PRO.VL_FATOR VL_CUSTO_MEDIO,
       (NVL(ITSOLSAI_PRO.QT_SOLICITADO,0) * dbamv.mges.f_checa_uni_pro(ITSOLSAI_PRO.CD_UNI_PRO,ITSOLSAI_PRO.CD_PRODUTO))/UNI_PRO.VL_FATOR QT_SOLICITADO,
       (NVL(ITSOLSAI_PRO.QT_ATENDIDA  ,0) * dbamv.mges.f_checa_uni_pro(ITSOLSAI_PRO.CD_UNI_PRO,ITSOLSAI_PRO.CD_PRODUTO))/UNI_PRO.VL_FATOR QT_ATENDIDA,
       DECODE(ITSOLSAI_PRO.SN_CONF_DETERM_USU,'S','Sim','Não')      SN_CONF_DETERM_USU
FROM
       DBAMV.MVTO_ESTOQUE    MVTO_ESTOQUE,
       DBAMV.ITMVTO_ESTOQUE  ITMVTO_ESTOQUE,
       DBAMV.ESTOQUE         ESTOQUE_ORIGEM,
       DBAMV.UNI_PRO         UNI_PRO,
       DBAMV.PRODUTO         PRODUTO,
       DBAMV.PRODUTO         PRODUTO_KIT,
       DBAMV.ESTOQUE         ESTOQUE_DESTINO,
       DBAMV.ITSOLSAI_PRO    ITSOLSAI_PRO, 
       DBAMV.MOTIVO_DIVERG_ATEND MOTIVO_DIVERG_ATEND 
WHERE
       MVTO_ESTOQUE.TP_MVTO_ESTOQUE    =  'T'
AND    MVTO_ESTOQUE.CD_MVTO_ESTOQUE    =  ITMVTO_ESTOQUE.CD_MVTO_ESTOQUE
AND    MVTO_ESTOQUE.CD_ESTOQUE         =  ESTOQUE_ORIGEM.CD_ESTOQUE
AND    MVTO_ESTOQUE.CD_ESTOQUE_DESTINO =  ESTOQUE_DESTINO.CD_ESTOQUE
AND    ITMVTO_ESTOQUE.CD_ITSOLSAI_PRO  =  ITSOLSAI_PRO.CD_ITSOLSAI_PRO(+) 
AND    ITSOLSAI_PRO.CD_MOTIVO_DIVERG_ATEND = MOTIVO_DIVERG_ATEND.CD_MOTIVO_DIVERG_ATEND(+)   
AND    ITMVTO_ESTOQUE.CD_UNI_PRO       =  UNI_PRO.CD_UNI_PRO(+)
AND    ITMVTO_ESTOQUE.CD_PRODUTO       =  PRODUTO.CD_PRODUTO(+)
AND    MVTO_ESTOQUE.CD_MVTO_ESTOQUE    =  :CAMPO_CD_MVTO_ESTOQUE
AND    ITMVTO_ESTOQUE.CD_PRODUTO_KIT   =  PRODUTO_KIT.CD_PRODUTO(+)
GROUP BY
       MVTO_ESTOQUE.CD_USUARIO,
       MVTO_ESTOQUE.CD_MVTO_ESTOQUE,
       MVTO_ESTOQUE.CD_SOLSAI_PRO,
       MVTO_ESTOQUE.CD_ESTOQUE,
       ESTOQUE_ORIGEM.DS_ESTOQUE,
       MVTO_ESTOQUE.CD_ESTOQUE_DESTINO,
       ESTOQUE_DESTINO.DS_ESTOQUE,
       MVTO_ESTOQUE.DT_MVTO_ESTOQUE,
       MVTO_ESTOQUE.HR_MVTO_ESTOQUE,
       MVTO_ESTOQUE.DT_IMPRESSAO,
       MVTO_ESTOQUE.NR_DOCUMENTO,
       MOTIVO_DIVERG_ATEND.DS_MOTIVO_DIVERG_ATEND,   
       ITMVTO_ESTOQUE.CD_PRODUTO,
       PRODUTO.DS_PRODUTO,
       ITMVTO_ESTOQUE.CD_PRODUTO_KIT,
       PRODUTO_KIT.DS_PRODUTO,
       UNI_PRO.DS_UNIDADE,
       ITMVTO_ESTOQUE.CD_LOTE,
       ITMVTO_ESTOQUE.DT_VALIDADE,
       ITMVTO_ESTOQUE.DS_ENDERECO,
       VERIF_VL_CUSTO_MEDIO( ITMVTO_ESTOQUE.CD_PRODUTO
                           , MVTO_ESTOQUE.DT_MVTO_ESTOQUE, 'H', NULL
                           , MVTO_ESTOQUE.HR_MVTO_ESTOQUE) * UNI_PRO.VL_FATOR,
       (NVL(ITSOLSAI_PRO.QT_SOLICITADO,0) * dbamv.mges.f_checa_uni_pro(ITSOLSAI_PRO.CD_UNI_PRO,ITSOLSAI_PRO.CD_PRODUTO))/UNI_PRO.VL_FATOR ,
       (NVL(ITSOLSAI_PRO.QT_ATENDIDA  ,0) * dbamv.mges.f_checa_uni_pro(ITSOLSAI_PRO.CD_UNI_PRO,ITSOLSAI_PRO.CD_PRODUTO))/UNI_PRO.VL_FATOR ,
       DECODE(ITSOLSAI_PRO.SN_CONF_DETERM_USU,'S','Sim','Não')
ORDER BY ITMVTO_ESTOQUE.CD_PRODUTO, DS_ENDERECO, CD_MVTO_ESTOQUE,CD_ESTOQUE_ORIGEM,DS_ESTOQUE_ORIGEM,DS_ESTOQUE_DESTINO,CD_ESTOQUE_DESTINO,CD_USUARIO,DT_MVTO_ESTOQUE,HR_MVTO_ESTOQUE,CD_SOLSAI_PRO,DT_IMPRESSAO,NR_DOCUMENTO,
DS_PRODUTO
