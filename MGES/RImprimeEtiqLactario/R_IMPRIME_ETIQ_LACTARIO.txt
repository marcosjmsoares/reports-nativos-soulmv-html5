--R_IMPRIME_ETIQ_LACTARIO
SELECT 
SP.CD_SOLSAI_PRO,
A.CD_ATENDIMENTO,
To_CHAR(ITLOT.DT_VALIDADE,'DD/MM/YYYY HH24') DT_VALIDADE,
P.NM_PACIENTE,
P.DT_NASCIMENTO,
L.DS_RESUMO DS_LEITO,
ITPM.CD_ITPRE_MED,
EMP.CD_MULTI_EMPRESA,
EMP.CD_PRODUTO,
TO_CHAR(ENT.DT_ENTRADA, 'DD/MM/YYYY') || ' ' || TO_CHAR(ENT.HR_ENTRADA, 'HH24') DT_FABRICACAO,
FA.DS_FOR_APL DS_VIA,
D.DS_DISPOSITIVO,
TO_CHAR(HRITSP.DH_MEDICACAO) DH_MEDICACAO,
TO_CHAR(HRITSP.DH_MEDICACAO, 'DD/MM/YYYY') DT_MEDICACAO,
TO_CHAR(HRITSP.DH_MEDICACAO, 'HH24') HR_MEDICACAO,
ITPM.QT_INFUSAO,
UP.CD_UNIDADE || '/' || UP.TP_TEMPO UND_INFUSAO,
'Manter refrigerado. Retirar 30 minutos antes da administração.' || CHR(10) || ' Válido por 1h após atingir temperatura ambiente' ESTABILIDADE,
Trim('2' || Substr(Replace(To_Char(To_Number(DBAMV.FNC_MGES_VALIDA_COD_BARRA(I.CD_IDENTIFICADOR)),'99999999999'),' ','0'),2, 11)||'1') DSP_CODIGO_DE_BARRAS,
PRESTADOR.NM_PRESTADOR || ' - CRN: ' || PRESTADOR.DS_CODIGO_CONSELHO NUTRICIONISTA,
TP.DS_TIP_PRESC,
UP.DS_UNIDADE,
ITPM.QT_ITPRE_MED,
ELE.DS_DENSIDADE_CALORICA,
ELE.DS_DISTRIBUICAO_CALORICA,
ELE.DS_CARBOIDRATO,
ELE.DS_PROTEINA,
ELE.DS_LIPIDIO,
ELE.DS_OSMOLARIDADE
FROM
DBAMV.SOLSAI_PRO SP,
DBAMV.ATENDIME A,
DBAMV.PRE_MED PM,
DBAMV.ITPRE_MED ITPM,
DBAMV.HRITPRE_MED HRITPM,
DBAMV.HRITSOLSAI_PRO HRITSP,
DBAMV.ITSOLSAI_PRO ITSP,
DBAMV.ITSOLSAI_PRO_ATENDIDO ITSPAT,
DBAMV.SETOR ST,
DBAMV.UNID_INT UI,
DBAMV.LEITO L,
DBAMV.PACIENTE P,
DBAMV.PW_DISPOSITIVO D,
DBAMV.FOR_APL FA,
DBAMV.UNI_PRO UP,
DBAMV.FORMULA F,
DBAMV.EMPRESA_PRODUTO EMP,
DBAMV.ITMVTO_ESTOQUE ITMVTO,
DBAMV.ITLOT_ENT ITLOT,
DBAMV.ITENT_PRO ITENT,
DBAMV.ENT_PRO ENT,
DBAMV.NUTRICIONISTA_EMPRESA NE,
DBAMV.PRESTADOR,
DBAMV.TIP_PRESC TP,
DBAMV.ETIQUETA_LACTARIO_EMPRESA ELE,
DBAMV.IDENTIFICADOR_ETIQUETA I,
DBAMV.PRODUTO PD
WHERE A.CD_ATENDIMENTO = PM.CD_ATENDIMENTO
AND PM.CD_PRE_MED = ITPM.CD_PRE_MED
AND ITPM.CD_ITPRE_MED = HRITPM.CD_ITPRE_MED
AND HRITSP.CD_ITSOLSAI_PRO = ITSP.CD_ITSOLSAI_PRO
AND ITPM.CD_ITPRE_MED = HRITSP.CD_ITPRE_MED                                                    
AND ITSP.CD_SOLSAI_PRO = SP.CD_SOLSAI_PRO
AND SP.CD_ATENDIMENTO = A.CD_ATENDIMENTO
AND HRITPM.DH_MEDICACAO = HRITSP.DH_MEDICACAO
AND A.CD_MULTI_EMPRESA = ST.CD_MULTI_EMPRESA
AND ST.CD_SETOR = UI.CD_SETOR
AND SP.CD_UNID_INT = UI.CD_UNID_INT
AND A.CD_LEITO = L.CD_LEITO
AND A.CD_PACIENTE = P.CD_PACIENTE
AND ITPM.CD_DISPOSITIVO = D.CD_DISPOSITIVO
AND ITPM.CD_FOR_APL = FA.CD_FOR_APL
AND ITPM.CD_UNI_PRO = UP.CD_UNI_PRO
AND ITSP.CD_ITSOLSAI_PRO = ITSPAT.CD_ITSOLSAI_PRO
AND F.CD_PRODUTO(+) = ITSPAT.CD_PRODUTO_ATENDIDO
AND A.CD_MULTI_EMPRESA = EMP.CD_MULTI_EMPRESA
AND ITSPAT.CD_PRODUTO_ATENDIDO = EMP.CD_PRODUTO
AND ITSPAT.CD_ITMVTO_ESTOQUE = ITMVTO.CD_ITMVTO_ESTOQUE
AND ITMVTO.CD_ITMVTO_ESTOQUE = ITSPAT.CD_ITMVTO_ESTOQUE 
AND ITMVTO.CD_LOTE = ITLOT.CD_LOTE                                               
AND ITLOT.CD_ITENT_PRO = ITENT.CD_ITENT_PRO                                                 
AND ITENT.CD_ENT_PRO = ENT.CD_ENT_PRO
AND EMP.SN_IMPRIME_ETIQUETA_LACTARIO = 'S'
AND EMP.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA                                                                  
AND NE.CD_MULTI_EMPRESA = EMP.CD_MULTI_EMPRESA
AND NE.CD_NUTRICIONISTA_RESP = PRESTADOR.CD_PRESTADOR                                                            
AND ITPM.CD_TIP_PRESC = TP.CD_TIP_PRESC
AND ITPM.CD_UNI_PRO = UP.CD_UNI_PRO                                   
AND ELE.CD_PRODUTO = EMP.CD_PRODUTO
AND ELE.CD_MULTI_EMPRESA = EMP.CD_MULTI_EMPRESA
AND PD.CD_PRODUTO = ITSP.CD_PRODUTO
AND I.CD_ITSOLSAI_PRO = ITSP.CD_ITSOLSAI_PRO
AND I.CD_ITMVTO_ESTOQUE = ITMVTO.CD_ITMVTO_ESTOQUE
AND HRITSP.cd_itsolsai_pro = ITSPAT.cd_itsolsai_pro
and To_Char(I.NR_RECEBIMENTO) = To_Char(HRITSP.DH_MEDICACAO, 'DD/MM/YYYY HH24')
AND I.SN_ATIVO = 'S'
AND To_Date(ITLOT.DT_VALIDADE) = To_Date(I.DT_VALIDADE)
ORDER BY I.CD_IDENTIFICADOR, HR_MEDICACAO, SP.CD_SOLSAI_PRO
