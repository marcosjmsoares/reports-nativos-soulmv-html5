--R_MODIFICA_PRODUTO
SELECT m.cd_modifica_produto,
        m.cd_produto,
        p.ds_produto,
        m.sn_lote_antigo,
        m.sn_lote_novo,
        m.sn_controle_validade_antigo,
        m.sn_controle_validade_novo,
        m.dt_abertura_modificacao dt_modificacao,
        m.cd_usuario_abertura cd_usuario,
        p.sn_mestre,
        i.cd_contagem,
        c.cd_estoque,
        c.dt_contagem ,
        c.sn_aberto,
        c.dt_geracao,
        u.nm_usuario
 FROM
       dbamv.modifica_produto m,
       dbamv.itmodifica_produto i,
       dbamv.produto p,
       dbamv.contagem c,
       dbamv.estoque e,
       dbasgu.usuarios u
 WHERE m.cd_modifica_produto = i.cd_modifica_produto
       AND p.cd_produto = m.cd_produto
       AND i.cd_contagem = c.cd_contagem
       AND c.cd_estoque = e.cd_estoque
       AND m.cd_usuario_abertura = u.cd_usuario
       AND m.cd_modifica_produto = :CAMPO_CD_MODIFICA
