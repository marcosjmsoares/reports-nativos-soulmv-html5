--R_LIST_PROD_GET
SELECT * FROM (
SELECT estoque.cd_estoque,
       estoque.ds_estoque,
       especie.cd_especie,
       especie.ds_especie,
       classe.cd_classe,
       classe.ds_classe,
       sub_clas.cd_sub_cla,
       sub_clas.ds_sub_cla,
       produto.cd_produto,
       produto.ds_produto,
       NVL(est_pro.qt_estoque_maximo,0) AS qt_estoque_maximo,
       NVL(est_pro.qt_estoque_minimo,0) AS qt_estoque_minimo,
       NVL(produto.vl_custo_medio,0) AS vl_custo_medio,
       NVL(est_pro.qt_estoque_atual,0) AS qt_estoque_atual
FROM   dbamv.estoque,
       dbamv.est_pro,
       dbamv.produto,
       dbamv.sub_clas,
       dbamv.classe,
       dbamv.especie
WHERE  estoque.cd_estoque  = est_pro.cd_estoque
AND    est_pro.cd_produto  = produto.cd_produto
AND    produto.cd_sub_cla  = sub_clas.cd_sub_cla
AND    produto.cd_classe     = sub_clas.cd_classe
AND    produto.cd_especie  = sub_clas.cd_especie
AND    sub_clas.cd_classe   = classe.cd_classe
AND    sub_clas.cd_especie = classe.cd_especie
AND    classe.cd_especie     = especie.cd_especie
AND    estoque.cd_estoque  LIKE :CAMPO_ESTOQUE
AND    especie.cd_especie   LIKE :CAMPO_ESPECIE
AND    classe.cd_classe        LIKE :CAMPO_CLASSE
AND    sub_clas.cd_sub_cla  LIKE :CAMPO_SUB_CLAS
AND    Estoque.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
)
ORDER BY CD_ESTOQUE,DS_ESTOQUE,
CD_ESPECIE,DS_ESPECIE,
CD_CLASSE,DS_CLASSE,
CD_SUB_CLA,DS_SUB_CLA
