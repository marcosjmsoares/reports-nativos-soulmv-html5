--R_EMPRESTIMO_ENTRADA_DIF
Select Fornecedor.nm_fornecedor                       Fornecedor,
            Produto.cd_produto                                  Produto,
            Produto.ds_produto                                  Ds_Produto,
            Verif_Ds_unid_prod(produto.cd_produto) Ds_unidade,
            Itent_pro.qt_entrada                                 Qt_entrada,
            Itent_pro.vl_total                                       Vl_total,
            ItEnt_Pro.Cd_Itent_Pro                             Cd_ItEnt_Pro
   From Dbamv.Ent_pro                            Ent_pro,
            Dbamv.Itent_pro                          Itent_pro,
            Dbamv.Tip_doc                           Tip_doc,
            Dbamv.Fornecedor                      Fornecedor,
            Dbamv.Produto                            Produto,
            Dbamv.Uni_pro                            Uni_pro
Where Fornecedor.cd_fornecedor           = Ent_pro.cd_fornecedor
    And Ent_Pro.cd_tip_doc                      = tip_doc.cd_tip_doc
    And Ent_pro.cd_ent_pro                      = Itent_pro.cd_ent_pro
    And Itent_pro.cd_produto                     = Produto.cd_produto
    And Itent_pro.cd_uni_pro                     = Uni_pro.cd_uni_pro
    And Tip_Doc.tp_entrada                      = 'E'
    And Exists (Select Cd_Itent_Pro From Dbamv.Itmvto_Ent Where itmvto_ent.Cd_Itent_Pro = Itent_pro.cd_itent_Pro)
    And Ent_pro.dt_entrada     Between :CAMPO_DT_INIC
                                                     And 
   And Ent_Pro.Cd_Estoque in ( Select Estoque.Cd_Estoque
			            From Dbamv.Estoque
                             Where Estoque.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA )
 order by  Fornecedor.nm_fornecedor
