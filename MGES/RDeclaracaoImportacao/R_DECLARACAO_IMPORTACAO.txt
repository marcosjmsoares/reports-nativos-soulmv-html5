--R_DECLARACAO_IMPORTACAO
SELECT
DI.CD_DECLARACAO_IMPORTACAO AS CD_DECLARACAO_IMPORTACAO,
DI.DS_DOCUMENTO AS DS_DOCUMENTO,
DI.DS_SERIE AS DS_SERIE,
DI.DT_EMISSAO AS DT_EMISSAO,
DI.DT_ENTRADA AS DT_ENTRADA,
DI.DT_ENTRADA AS DT_ENTRADA_HORA,
M.CD_MOEDA AS CD_MOEDA,
M.DS_MOEDA AS DS_MOEDA,
E.CD_ESTOQUE AS CD_ESTOQUE,
E.DS_ESTOQUE AS DS_ESTOQUE,
F.CD_FORNECEDOR AS CD_FORNECEDOR,
F.NM_FANTASIA AS NM_FANTASIA,
MEP.CD_MODELO_ENTRADA AS CD_MODELO_ENTRADA,
MEP.DS_MODELO_ENTRADA AS DS_MODELO_ENTRADA,
DI.DS_OBSERVACAO AS DS_OBSERVACAO,
DI.CD_UF_DESEMBARACO AS CD_UF_DESEMBARACO,
DI.DS_LOCAL_DESEMBARACO AS DS_LOCAL_DESEMBARACO,
DI.DT_DESEMBARACO AS DT_DESEMBARACO,
CI.CD_IMPORTACAO_DI AS CD_IMPORTACAO_DI,
CI.DS_IMPORTACAO_DI AS DS_IMPORTACAO_DI,
DI.NM_EXPORTADOR AS NM_EXPORTADOR,
MTD.CD_MODALIDADE_TRANSPORTE_DI AS CD_MODALIDADE_TRANSPORTE_DI,
MTD.DS_MODALIDADE_TRANSPORTE_DI AS DS_MODALIDADE_TRANSPORTE_DI,
CSDI.CD_CSOSN,
CSDI.DS_CSOSN,
F1.NM_FANTASIA NM_FANTASIA_FORN_FRETE,
TR.NM_FANTASIA NM_FANTASIA_TRANSP_FRETE
,VR.DS_MODELO || '-' || VR.DS_MARCA || '-' || DS_PLACA DS_MODELO
,Decode(FDI.TP_FRETE_DI, 'C','CIF','F','FOB','I','INTERNACIONAL','OUTROS') TP_FRETE_DI_FRETE
,MTD1.DS_MODALIDADE_TRANSPORTE_DI DS_MODALIDADE_TRANSP_DI_FRETE
,FDI.NR_VOLUMES
,FDI.CD_NUMERACAO
,FDI.CD_ESPECIE
,FDI.CD_IDENTIFICACAO
,FDI.CD_IDENTIFICACAO_LACRE
,FDI.NR_PESO_BRUTO
,FDI.DS_MEDIDA
,FDI.NR_PESO_LIQUIDO
FROM
DBAMV.DECLARACAO_IMPORTACAO DI,
DBAMV.MOEDA M,
DBAMV.ESTOQUE E,
DBAMV.FORNECEDOR F,
DBAMV.MODELO_ENTRADA_PRODUTO MEP,
DBAMV.CODIGO_IMPORTACAO_DI CI,
DBAMV.MODALIDADE_TRANSPORTE_DI MTD,
DBAMV.CSOSN CSDI,
DBAMV.FRETE_DECLARACAO_IMPORTACAO FDI,
DBAMV.FORNECEDOR F1,
DBAMV.TRANSPORTADORA_ROMANEIO TR,
DBAMV.VEICULO_ROMANEIO VR,
DBAMV.MODALIDADE_TRANSPORTE_DI MTD1
WHERE DI.CD_MOEDA = M.CD_MOEDA
AND DI.CD_ESTOQUE = E.CD_ESTOQUE
AND DI.CD_FORNECEDOR = F.CD_FORNECEDOR
AND DI.CD_MODELO_ENTRADA = MEP.CD_MODELO_ENTRADA(+)
AND DI.CD_IMPORTACAO_DI = CI.CD_IMPORTACAO_DI
AND DI.CD_MODALIDADE_TRANSPORTE_DI = MTD.CD_MODALIDADE_TRANSPORTE_DI
AND DI.CD_CSOSN = CSDI.CD_CSOSN
AND DI.CD_DECLARACAO_IMPORTACAO = FDI.CD_DECLARACAO_IMPORTACAO(+)
AND FDI.CD_FORNECEDOR = F1.CD_FORNECEDOR(+)
AND FDI.CD_TRANSPORTADORA = TR.CD_TRANSPORTADORA(+)
AND FDI.CD_VEICULO = VR.CD_VEICULO(+)
AND FDI.CD_MODALIDADE_TRANSPORTE_DI = MTD1.CD_MODALIDADE_TRANSPORTE_DI(+)
