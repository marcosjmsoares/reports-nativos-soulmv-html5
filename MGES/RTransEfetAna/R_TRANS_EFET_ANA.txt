Select B.Cd_Produto
         , C.Ds_Produto
         , A.dt_mvto_estoque
         , A.cd_mvto_estoque
         , Verif_Ds_Unid_Prod( B.Cd_Produto ) Ds_Unidade
         , D.Ds_Estoque Estoque_Ori
         , E.Ds_Estoque Estoque_Des
         ,Sum( B.Qt_Movimentacao * F.Vl_Fator / Verif_Vl_Fator_Prod( B.Cd_Produto ) ) Qt_Movimentacao
         ,Sum( B.Qt_Movimentacao * F.Vl_Fator  * Verif_Vl_Custo_Medio( B.Cd_Produto, A.Dt_Mvto_Estoque, :CAMPO_TP_CUSTO, C.Vl_Custo_Medio, A.Hr_Mvto_Estoque ) ) vl_total
          ,Sum( B.Qt_Movimentacao * F.Vl_Fator  * Verif_Vl_Custo_Medio( B.Cd_Produto, A.Dt_Mvto_Estoque, :CAMPO_TP_CUSTO, C.Vl_Custo_Medio, A.Hr_Mvto_Estoque ) ) / Sum( B.Qt_Movimentacao * F.Vl_Fator / Verif_Vl_Fator_Prod( B.Cd_Produto ) ) Vl_Unitario
 From Dbamv.Uni_Pro F
        , Dbamv.Estoque E
        , Dbamv.Estoque D
        , Dbamv.Produto C
        , Dbamv.Itmvto_Estoque B
        , Dbamv.Mvto_Estoque A
 Where A.Dt_Mvto_Estoque  Between :CAMPO_DTINIC and 
   and A.Tp_Mvto_Estoque	= 'T'
   and B.Cd_Mvto_Estoque	= A.Cd_Mvto_Estoque
   and C.Cd_Produto	= B.Cd_Produto
   and D.Cd_Estoque	= A.Cd_Estoque
   and E.Cd_Estoque	= A.Cd_Estoque_Destino
   and E.Cd_Multi_Empresa	= :CAMPO_CD_MULTI_EMPRESA
   and F.Cd_Uni_Pro	= B.Cd_Uni_Pro
 Group By B.Cd_Produto
               , C.Ds_Produto
               , A.dt_mvto_estoque
               , A.cd_mvto_estoque
               , D.Ds_Estoque
               , E.Ds_Estoque
               , C.Vl_Custo_Medio
               , F.Vl_Fator
