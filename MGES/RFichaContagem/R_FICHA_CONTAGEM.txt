--R_FICHA_CONTAGEM
SELECT PRODUTO,
       CD_PRODUTO,
       DS_PRODUTO,
       ESTOQUE,
       ESPECIE,
       UNIDADE,
       CONTAGEM,
       CD_LOCALIZACAO
FROM
  (SELECT PRODUTO.CD_PRODUTO || ' - ' || PRODUTO.DS_PRODUTO AS PRODUTO,
          PRODUTO.CD_PRODUTO,
          PRODUTO.DS_PRODUTO,
          ESTOQUE.CD_ESTOQUE || ' - ' || ESTOQUE.DS_ESTOQUE AS ESTOQUE,
          ESPECIE.CD_ESPECIE || ' - ' || ESPECIE.DS_ESPECIE AS ESPECIE,
          DBAMV.Verif_Ds_Unid_Prod(PRODUTO.CD_PRODUTO) AS UNIDADE,
          0 AS CONTAGEM,
          EST_PRO.CD_LOCALIZACAO
   FROM DBAMV.EST_PRO,
        DBAMV.ESPECIE,
        DBAMV.ESTOQUE,
     (SELECT PRODUTO.CD_PRODUTO,
             PRODUTO.DS_PRODUTO,
             PRODUTO.CD_ESPECIE,
             EMPRESA_PRODUTO.CD_MULTI_EMPRESA,
             PRODUTO.CD_CLASSE,
             PRODUTO.CD_SUB_CLA,
             PRODUTO.CD_PRO_FAT,
             PRODUTO.SN_CONSIGNADO,
             PRODUTO.SN_ADMITIDO,
             Decode(DBAMV.PKT_CONFIGEST.RetornarControleLoteValEmpPrd(), 'S', EMPRESA_PRODUTO.SN_MOVIMENTACAO, PRODUTO.SN_MOVIMENTACAO) AS SN_MOVIMENTACAO,
             Decode(DBAMV.PKT_CONFIGEST.RetornarControleLoteValEmpPrd(), 'S', EMPRESA_PRODUTO.SN_BLOQUEIO_DE_COMPRA, PRODUTO.SN_BLOQUEIO_DE_COMPRA) AS SN_BLOQUEIO_DE_COMPRA,
             Decode(DBAMV.PKT_CONFIGEST.RetornarPadronizadoEmp(), 'S', EMPRESA_PRODUTO.SN_PADRONIZADO, PRODUTO.SN_PADRONIZADO) AS SN_PADRONIZADO,
             Nvl(Decode(DBAMV.PKT_CONFIGEST.RetornarControleLoteValEmpPrd(), 'S', EMPRESA_PRODUTO.SN_CONTROLE_VALIDADE, PRODUTO.SN_CONTROLE_VALIDADE), 'N') AS SN_CONTROLE_VALIDADE,
             Nvl(Decode(DBAMV.PKT_CONFIGEST.RetornarControleLoteValEmpPrd(), 'S', EMPRESA_PRODUTO.SN_LOTE, PRODUTO.SN_LOTE), 'N') AS SN_LOTE
      FROM DBAMV.EMPRESA_PRODUTO,
           DBAMV.PRODUTO
      WHERE PRODUTO.CD_PRODUTO = EMPRESA_PRODUTO.CD_PRODUTO
        AND EMPRESA_PRODUTO.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
        AND Nvl(PRODUTO.SN_MESTRE, 'N') = 'N'
        AND Decode(DBAMV.PKT_CONFIGEST.RetornarControleLoteValEmpPrd(), 'S', EMPRESA_PRODUTO.SN_MOVIMENTACAO, PRODUTO.SN_MOVIMENTACAO) = 'S'   :CAMPO_CF_SUB_CLASSE) PRODUTO
   WHERE PRODUTO.CD_ESPECIE = ESPECIE.CD_ESPECIE
     AND PRODUTO.CD_PRODUTO = EST_PRO.CD_PRODUTO
     AND ESTOQUE.CD_ESTOQUE = EST_PRO.CD_ESTOQUE
     AND ESTOQUE.CD_MULTI_EMPRESA = PRODUTO.CD_MULTI_EMPRESA
     AND DBAMV.MGES.f_configuracao_estoque(ESTOQUE.CD_ESTOQUE, ESPECIE.CD_ESPECIE, PRODUTO.CD_CLASSE, PRODUTO.CD_SUB_CLA, PRODUTO.CD_PRODUTO) > 0
     AND ((PRODUTO.SN_CONTROLE_VALIDADE = 'N'
           AND PRODUTO.SN_LOTE = 'N')
          OR Nvl(
              (SELECT Max(1)
                FROM DBAMV.LOT_PRO LP
                WHERE LP.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
                  AND LP.CD_PRODUTO = PRODUTO.CD_PRODUTO :CAMPO_CF_DESCONSIDERA_PRODUTO_TODOS_LOTES_ZERADOS), 0) = 1)   
   UNION ALL SELECT PRODUTO.CD_PRODUTO || ' - ' || PRODUTO.DS_PRODUTO AS PRODUTO,
                    PRODUTO.CD_PRODUTO,
                    PRODUTO.DS_PRODUTO,
                    ESTOQUE.CD_ESTOQUE || ' - ' || ESTOQUE.DS_ESTOQUE AS ESTOQUE,
                    ESPECIE.CD_ESPECIE || ' - ' || ESPECIE.DS_ESPECIE AS ESPECIE,
                    DBAMV.Verif_Ds_Unid_Prod(PRODUTO.CD_PRODUTO) AS UNIDADE,
                    CONTAGEM.CD_CONTAGEM AS CONTAGEM,
                    EST_PRO.CD_LOCALIZACAO
   FROM DBAMV.ITCONTAGEM_DA_ABERTURA,
        DBAMV.CONTAGEM,
        DBAMV.EST_PRO,
        DBAMV.ESPECIE,
        DBAMV.ESTOQUE,
     (SELECT PRODUTO.CD_PRODUTO,
             PRODUTO.DS_PRODUTO,
             PRODUTO.CD_ESPECIE,
             EMPRESA_PRODUTO.CD_MULTI_EMPRESA,
             PRODUTO.CD_CLASSE,
             PRODUTO.CD_SUB_CLA,
             PRODUTO.CD_PRO_FAT,
             PRODUTO.SN_CONSIGNADO,
             PRODUTO.SN_ADMITIDO,
             Decode(DBAMV.PKT_CONFIGEST.RetornarControleLoteValEmpPrd(), 'S', EMPRESA_PRODUTO.SN_MOVIMENTACAO, PRODUTO.SN_MOVIMENTACAO) AS SN_MOVIMENTACAO,
             Decode(DBAMV.PKT_CONFIGEST.RetornarControleLoteValEmpPrd(), 'S', EMPRESA_PRODUTO.SN_BLOQUEIO_DE_COMPRA, PRODUTO.SN_BLOQUEIO_DE_COMPRA) AS SN_BLOQUEIO_DE_COMPRA,
             Decode(DBAMV.PKT_CONFIGEST.RetornarPadronizadoEmp(), 'S', EMPRESA_PRODUTO.SN_PADRONIZADO, PRODUTO.SN_PADRONIZADO) AS SN_PADRONIZADO,
             Nvl(Decode(DBAMV.PKT_CONFIGEST.RetornarControleLoteValEmpPrd(), 'S', EMPRESA_PRODUTO.SN_CONTROLE_VALIDADE, PRODUTO.SN_CONTROLE_VALIDADE), 'N') AS SN_CONTROLE_VALIDADE,
             Nvl(Decode(DBAMV.PKT_CONFIGEST.RetornarControleLoteValEmpPrd(), 'S', EMPRESA_PRODUTO.SN_LOTE, PRODUTO.SN_LOTE), 'N') AS SN_LOTE
      FROM DBAMV.EMPRESA_PRODUTO,
           DBAMV.PRODUTO
      WHERE PRODUTO.CD_PRODUTO = EMPRESA_PRODUTO.CD_PRODUTO
        AND EMPRESA_PRODUTO.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
        AND Nvl(PRODUTO.SN_MESTRE, 'N') = 'N'
        AND Decode(DBAMV.PKT_CONFIGEST.RetornarControleLoteValEmpPrd(), 'S', EMPRESA_PRODUTO.SN_MOVIMENTACAO, PRODUTO.SN_MOVIMENTACAO) = 'S'   :CAMPO_CF_SUB_CLASSE) PRODUTO
   WHERE PRODUTO.CD_ESPECIE = ESPECIE.CD_ESPECIE
     AND PRODUTO.CD_PRODUTO = EST_PRO.CD_PRODUTO
     AND ESTOQUE.CD_ESTOQUE = EST_PRO.CD_ESTOQUE
     AND ESTOQUE.CD_MULTI_EMPRESA = PRODUTO.CD_MULTI_EMPRESA
     AND CONTAGEM.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
     AND CONTAGEM.CD_CONTAGEM = ITCONTAGEM_DA_ABERTURA.CD_CONTAGEM
     AND ITCONTAGEM_DA_ABERTURA.CD_PRODUTO = PRODUTO.CD_PRODUTO
     AND DBAMV.MGES.f_configuracao_estoque(ESTOQUE.CD_ESTOQUE, ESPECIE.CD_ESPECIE, PRODUTO.CD_CLASSE, PRODUTO.CD_SUB_CLA, PRODUTO.CD_PRODUTO) > 0
     AND ((PRODUTO.SN_CONTROLE_VALIDADE = 'N'
           AND PRODUTO.SN_LOTE = 'N')
          OR  Nvl((SELECT Max(1)
                    FROM DBAMV.LOT_PRO LP
                    WHERE LP.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
                    AND LP.CD_PRODUTO = PRODUTO.CD_PRODUTO :CAMPO_CF_DESCONSIDERA_PRODUTO_TODOS_LOTES_ZERADOS), 0) = 1)
                       :CAMPO_CF_CONT2)
