--R_ITENS_SEM_VERIF_ROTATIVA
SELECT * FROM (
Select estoque.cd_estoque,
           estoque.ds_estoque,
           est_pro.cd_produto,
           produto.ds_produto
  from dbamv.estoque,
           dbamv.est_pro,
          dbamv.produto
where estoque.cd_estoque = est_pro.cd_estoque
    and est_pro.cd_produto = produto.cd_produto
    and Not Exists( Select 1
                                from  dbamv.verificacao_rotativa_saldo,
                                         dbamv.item_verificacao_rotativa_sald
                               where verificacao_rotativa_saldo.cd_verificacao_rotativa_saldo = item_verificacao_rotativa_sald.cd_verificacao_rotativa_saldo
                                  and verificacao_rotativa_saldo.dh_verificacao Between :CAMPO_DT_INI and 
                                  and  verificacao_rotativa_saldo.cd_estoque = est_pro.cd_estoque
                                  and item_verificacao_rotativa_sald.cd_produto = est_pro.cd_produto)
    and estoque.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
    Order by produto.ds_produto
)
ORDER BY CD_ESTOQUE,DS_ESTOQUE,ds_produto
