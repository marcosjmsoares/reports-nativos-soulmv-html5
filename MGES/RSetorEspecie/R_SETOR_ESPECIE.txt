--R_SETOR_ESPECIE
SELECT * FROM (
Select    setor.cd_setor cd_Setor_especie 
              ,setor.nm_setor nm_setor_especie
              ,especie.cd_especie cd_especie
              ,especie.ds_especie ds_especie
              ,sum( ( nvl( c_conset.vl_saida_paciente, 0 )          + nvl( c_conset.vl_saida_setor  ,0 ) ) - 
                       ( nvl( c_conset.vl_devolvida_paciente, 0 )   + nvl( c_conset.vl_devolvida_setor,0 ) ) ) vl_total
   From   dbamv.c_conset
            , dbamv.setor
            , dbamv.produto
            , dbamv.especie
            ,dbamv.estoque
Where   c_conset.cd_mes = TO_NUMBER( TO_CHAR( :CAMPO_DTINIC, 'MM' ) ) 
   and    c_conset.cd_ano  = TO_NUMBER( TO_CHAR( :CAMPO_DTINIC, 'YYYY' ) ) 
   and     produto.cd_especie        = especie.cd_especie
   and     c_conset.cd_setor          = setor.cd_setor 
   and     produto.cd_produto        = c_conset.cd_produto  
   and     estoque.cd_estoque      = c_conset.cd_estoque
   and     estoque.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA  
group by setor.cd_setor
              ,setor.nm_setor
              ,especie.cd_especie
              ,especie.ds_especie
)
ORDER BY NM_SETOR_ESPECIE,
DS_ESPECIE
