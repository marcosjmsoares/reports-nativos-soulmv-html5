--R_CONT_PROD_ESTOQ
Select
   ESPECIE.CD_ESPECIE,
   ESPECIE.DS_ESPECIE,
   PRODUTO.CD_PRODUTO,
   PRODUTO.DS_PRODUTO,
   PRODUTO.CD_PRODUTO_FILHO,
   PRODUTO.DS_PRODUTO_FILHO 
From
   DBAMV.ESPECIE,
   (
      Select
         PAI.CD_PRODUTO,
         PAI.DS_PRODUTO,
         FILHO.CD_PRODUTO As CD_PRODUTO_FILHO,
         FILHO.DS_PRODUTO As DS_PRODUTO_FILHO,
         PAI.CD_ESPECIE,
         PAI.CD_CLASSE,
         PAI.CD_SUB_CLA,
         Nvl(FILHO.SN_MOVIMENTACAO, PAI.SN_MOVIMENTACAO) As SN_MOVIMENTACAO,
         Nvl(FILHO.SN_PADRONIZADO, PAI.SN_PADRONIZADO) As SN_PADRONIZADO,
         Nvl(FILHO.CD_PRO_FAT, PAI.CD_PRO_FAT) As CD_PRO_FAT,
         Nvl(FILHO.SN_CONSIGNADO, PAI.SN_CONSIGNADO) As SN_CONSIGNADO,
         Nvl(FILHO.SN_BLOQUEIO_DE_COMPRA, PAI.SN_BLOQUEIO_DE_COMPRA) As SN_BLOQUEIO_DE_COMPRA,
         Nvl(FILHO.SN_ADMITIDO, PAI.SN_ADMITIDO) As SN_ADMITIDO 
      From
         DBAMV.PRODUTO PAI,
         DBAMV.PRODUTO FILHO 
      Where
         PAI.CD_PRODUTO_TEM Is Null 
         And PAI.CD_PRODUTO = FILHO.CD_PRODUTO_TEM(+) 
         :CAMPO_CF_SUBCLA
   )
   PRODUTO,
   DBAMV.EMPRESA_PRODUTO EMPRESA_PRODUTO 
Where
   DBAMV.MGES.f_configuracao_estoque( :CAMPO_CD_ESTOQUE, PRODUTO.CD_ESPECIE, PRODUTO.CD_CLASSE, PRODUTO.CD_SUB_CLA, PRODUTO.CD_PRODUTO) > 0 
   And ESPECIE.CD_ESPECIE = PRODUTO.CD_ESPECIE 
   AND EMPRESA_PRODUTO.CD_PRODUTO = PRODUTO.CD_PRODUTO
   AND EMPRESA_PRODUTO.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
