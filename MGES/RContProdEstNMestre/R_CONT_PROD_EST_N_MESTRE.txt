--R_CONT_PROD_EST_N_MESTRE
SELECT DISTINCT
               ESPECIE.CD_ESPECIE cd_especie,
               ESPECIE.DS_ESPECIE ds_especie,
               PRODUTO.CD_PRODUTO cd_produto,
               PRODUTO.DS_PRODUTO ds_produto
FROM     DBAMV.ESPECIE,
               DBAMV.PRODUTO
WHERE  dbamv.mges.f_configuracao_estoque(:CAMPO_CD_ESTOQUE
                                                             ,produto.cd_especie
                                                             ,produto.cd_classe
                                                             ,produto.cd_sub_cla
                                                             ,produto.cd_produto) > 0
AND        ESPECIE.CD_ESPECIE = PRODUTO.CD_ESPECIE
AND        NVL(PRODUTO.SN_MESTRE,'N') = 'N'
