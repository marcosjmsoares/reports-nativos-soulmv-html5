--R_COTSET_LISTAGEM
Select ESTOQUE.CD_SETOR As CD_SETOR_ORIGEM, SETOR_ESTOQUE.NM_SETOR As DS_SETOR_ORIGEM, COTA_SETOR.CD_SETOR As CD_SETOR_DESTINO,
SETOR.NM_SETOR As DS_SETOR_DESTINO, PRODUTO.CD_PRODUTO, PRODUTO.DS_PRODUTO, LOCALIZACAO.DS_LOCALIZACAO,
DBAMV.Verif_Ds_Unid_Prod(PRODUTO.CD_PRODUTO) As DS_UNIDADE, Round(COTA_SETOR.QT_COTA * UNI_PRO.VL_FATOR /
DBAMV.Verif_Vl_Fator_Prod(PRODUTO.CD_PRODUTO), 4) As QT_COTA, Round((Select Sum(IME.QT_MOVIMENTACAO * UP.VL_FATOR /
DBAMV.Verif_Vl_Fator_Prod(IME.CD_PRODUTO)) From DBAMV.UNI_PRO UP, DBAMV.ITMVTO_ESTOQUE IME, DBAMV.MVTO_ESTOQUE ME Where
IME.CD_MVTO_ESTOQUE = ME.CD_MVTO_ESTOQUE  And IME.CD_UNI_PRO = UP.CD_UNI_PRO And ME.CD_ESTOQUE = COTA_SETOR.CD_ESTOQUE And
ME.CD_ESTOQUE_DESTINO = (Select Min(E.CD_ESTOQUE) From DBAMV.ESTOQUE E Where E.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA And
E.CD_SETOR = COTA_SETOR.CD_SETOR) And ME.TP_MVTO_ESTOQUE = 'T' And IME.CD_PRODUTO = PRODUTO.CD_PRODUTO And
Nvl(ME.TP_STATUS_CONFIRMACAO, 'N') <> 'C'), 4) As QT_PENDENTE, Nvl((Select Round(EP.QT_ESTOQUE_ATUAL /
DBAMV.Verif_Vl_Fator_Prod(EP.CD_PRODUTO), 4) From DBAMV.EST_PRO EP Where EP.CD_ESTOQUE = (Select Min(E.CD_ESTOQUE) From DBAMV.ESTOQUE E
Where E.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA And E.CD_SETOR = COTA_SETOR.CD_SETOR) And EP.CD_PRODUTO = COTA_SETOR.CD_PRODUTO),
0) As QT_ESTOQUE_ATUAL, 0 As QT_KIT, DBAMV.FNC_MGES_CONSUMO_ANTERIOR(PRODUTO.CD_PRODUTO, :CAMPO_QT_DIAS, (Select Min(E.CD_ESTOQUE) From
DBAMV.ESTOQUE E Where E.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA And E.CD_SETOR = COTA_SETOR.CD_SETOR)) As QT_CONSUMO_MEDIO
From DBAMV.LOCALIZACAO, DBAMV.EST_PRO, DBAMV.SETOR SETOR_ESTOQUE, DBAMV.ESTOQUE, (Select D.CD_PRODUTO, D.DS_PRODUTO,
Decode(DBAMV.PKT_CONFIGEST.RetornarPadronizadoEmp(), 'S', C.SN_PADRONIZADO, D.SN_PADRONIZADO) As SN_PADRONIZADO, D.CD_PRO_FAT,
Decode(DBAMV.PKT_CONFIGEST.RetornarControleLoteValEmpPrd(), 'S', C.SN_MOVIMENTACAO, D.SN_MOVIMENTACAO) As SN_MOVIMENTACAO,
D.SN_CONSIGNADO, Decode(DBAMV.PKT_CONFIGEST.RetornarControleLoteValEmpPrd(), 'S', C.SN_BLOQUEIO_DE_COMPRA,
D.SN_BLOQUEIO_DE_COMPRA) As SN_BLOQUEIO_DE_COMPRA, D.SN_ADMITIDO From DBAMV.EMPRESA_PRODUTO C, DBAMV.PRODUTO D Where
C.CD_PRODUTO = D.CD_PRODUTO And C.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA :CAMPO_CF_KIT) PRODUTO, DBAMV.SETOR, DBAMV.UNI_PRO,
DBAMV.COTA_SETOR
Where COTA_SETOR.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE And ESTOQUE.CD_SETOR = SETOR_ESTOQUE.CD_SETOR And COTA_SETOR.CD_SETOR = SETOR.CD_SETOR
And EST_PRO.CD_ESTOQUE = COTA_SETOR.CD_ESTOQUE And EST_PRO.CD_PRODUTO = COTA_SETOR.CD_PRODUTO
And COTA_SETOR.CD_PRODUTO = PRODUTO.CD_PRODUTO And EST_PRO.CD_ESTOQUE = LOCALIZACAO.CD_ESTOQUE(+) And
EST_PRO.CD_LOCALIZACAO = LOCALIZACAO.CD_LOCALIZACAO(+) And COTA_SETOR.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO And (:CAMPO_PS_CD_LOCALIZACAO = '%'
Or EST_PRO.CD_LOCALIZACAO = :CAMPO_PS_CD_LOCALIZACAO) And ((:CAMPO_S_SN_VISUALIZA_PRODUTO_ZERADO = 'N' And Nvl((Select EP.QT_ESTOQUE_ATUAL
From DBAMV.EST_PRO EP Where EP.CD_ESTOQUE = (Select Min(E.CD_ESTOQUE) From DBAMV.ESTOQUE E Where
E.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA And E.CD_SETOR = COTA_SETOR.CD_SETOR) And EP.CD_PRODUTO = COTA_SETOR.CD_PRODUTO), 0) > 0) Or
:CAMPO_S_SN_VISUALIZA_PRODUTO_ZERADO = 'S')
Order By 1, 3, 5
