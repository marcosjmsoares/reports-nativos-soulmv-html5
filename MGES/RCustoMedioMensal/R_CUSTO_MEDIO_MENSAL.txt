--R_CUSTO_MEDIO_MENSAL
Select Custo_medio.cd_produto                              cd_produto,
           Produto.ds_produto                                      ds_produto,
           Custo_medio.dh_custo_medio                      dt_custo,
           Custo_medio.qt_estoque_antes / verif_vl_fator_prod(produto.cd_produto)                 qt_estoque_antes,
           Custo_medio.vl_custo_medio_antes * verif_vl_fator_prod(produto.cd_produto)            vl_custo_medio_antes,
           Custo_medio.vl_custo_medio * verif_vl_fator_prod(produto.cd_produto)                       vl_custo_medio,
           dbamv.VERIF_DS_UNID_PROD(produto.cd_produto) ds_unidade
  From Dbamv.custo_medio_mensal                        Custo_medio,
           Dbamv.produto                                             Produto,
           Dbamv.Especie                                            Especie,
           Dbamv.Classe                                               Classe,
           Dbamv.Sub_clas                                          Sub_clas
Where Custo_medio.cd_produto                          = Produto.cd_produto
    And TRUNC( custo_medio.dh_custo_medio,'MM' ) between trunc(To_date(:CAMPO_DTINIC,'dd/MM/yyyy'),'MM') and last_day(trunc(To_date(:CAMPO_DTFIM,'dd/MM/yyyy'),'MM')) + 0.99999
    And Produto.cd_sub_cla                                  = Sub_clas.cd_sub_cla
    And Produto.cd_classe                                    = Sub_clas.cd_classe
    And Produto.cd_especie                                  = Sub_clas.cd_especie
    And Sub_clas.cd_classe                                  = Classe.cd_classe
    And Sub_clas.cd_especie                                = Classe.cd_especie
    And Classe.cd_especie                                    = Especie.cd_especie
    And CUSTO_MEDIO.cd_multi_empresa          = dbamv.pkg_mv2000.le_empresa
order by cd_produto, dt_custo desc
