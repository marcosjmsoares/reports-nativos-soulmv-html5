--R_VALIDA_CONTAGEM
SELECT * FROM (
SELECT CD_SEQUENCIAL
     , CD_PRODUTO
     , CD_LOTE
     , DT_VALIDADE
     , 'Existem produtos sem quantidade informada.' OBS
     , 0 QTD
  FROM DBAMV.ITEM_CONTAGEM_ABERTA
 WHERE CD_CONTAGEM = :CAMPO_CD_CONTAGEM
   AND NR_SEQUENCIAL = :CAMPO_NR_SEQUENCIAL
   AND ((QT_CONTAGEM1 IS NULL AND :CAMPO_NR_CONTAGEM  = 1)
     OR (QT_CONTAGEM2 IS NULL AND :CAMPO_NR_CONTAGEM  = 2)
     OR (QT_CONTAGEM3 IS NULL AND :CAMPO_NR_CONTAGEM  = 3))
UNION ALL
SELECT CD_SEQUENCIAL
     , CD_PRODUTO
     , CD_LOTE
     , DT_VALIDADE
     , 'Existem produtos com Lote / Serie / Validade Duplicados.' OBS
     , Count(*)
  FROM DBAMV.ITEM_CONTAGEM_ABERTA
 WHERE CD_CONTAGEM = :CAMPO_CD_CONTAGEM
   AND NR_SEQUENCIAL = :CAMPO_NR_SEQUENCIAL
 GROUP BY CD_SEQUENCIAL
        , CD_PRODUTO
        , CD_LOTE
        , DT_VALIDADE
        , 'Existem produtos com Lote / Serie / Validade Duplicados.'
HAVING  Count(*) > 1
UNION ALL
SELECT CD_SEQUENCIAL
     , CD_PRODUTO
     , CD_LOTE
     , null DT_VALIDADE
     , 'Existem produtos com mesmo Lote / Serie, com validades divergentes.' OBS
     , Count(*)
  FROM DBAMV.ITEM_CONTAGEM_ABERTA
 WHERE CD_CONTAGEM = :CAMPO_CD_CONTAGEM
   AND NR_SEQUENCIAL = :CAMPO_NR_SEQUENCIAL
 GROUP BY CD_SEQUENCIAL
        , CD_PRODUTO
        , CD_LOTE
	, null
        , 'Existem produtos com mesmo Lote / Serie, com validades divergentes.'
HAVING  Count(*) > 1
UNION ALL
 SELECT ITEM_CONTAGEM_ABERTA.CD_SEQUENCIAL
      , PRODUTO.CD_PRODUTO
      , ITEM_CONTAGEM_ABERTA.CD_LOTE
      , ITEM_CONTAGEM_ABERTA.DT_VALIDADE
      , 'Existem produtos configurados para controlar lote, sem lote / validade informado.' OBS
      , 0 QTD
   FROM DBAMV.PRODUTO
      , DBAMV.ITEM_CONTAGEM_ABERTA
      , DBAMV.EMPRESA_PRODUTO
      , DBAMV.CONFIGEST
  WHERE PRODUTO.CD_PRODUTO = ITEM_CONTAGEM_ABERTA.CD_PRODUTO
    AND ITEM_CONTAGEM_ABERTA.CD_CONTAGEM = :CAMPO_CD_CONTAGEM
    AND EMPRESA_PRODUTO.CD_PRODUTO = PRODUTO.CD_PRODUTO
    AND NVL(DECODE(NVL(CONFIGEST.SN_LOTE_VAL_PRD_EMP,'N'),'S', EMPRESA_PRODUTO.SN_LOTE, PRODUTO.SN_LOTE),'N') = 'S'
    AND ITEM_CONTAGEM_ABERTA.CD_LOTE IS NULL
    AND CONFIGEST.CD_MULTI_EMPRESA = EMPRESA_PRODUTO.CD_MULTI_EMPRESA
    AND EMPRESA_PRODUTO.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
    AND ITEM_CONTAGEM_ABERTA.NR_SEQUENCIAL = :CAMPO_NR_SEQUENCIAL
UNION ALL
 SELECT ITEM_CONTAGEM_ABERTA.CD_SEQUENCIAL
      , PRODUTO.CD_PRODUTO
      , ITEM_CONTAGEM_ABERTA.CD_LOTE
      , ITEM_CONTAGEM_ABERTA.DT_VALIDADE
      , 'Existem produtos configurados para controlar validade, sem validade informado.' OBS
      , 0 QTD
   FROM DBAMV.PRODUTO
      , DBAMV.ITEM_CONTAGEM_ABERTA
      , DBAMV.EMPRESA_PRODUTO
      , DBAMV.CONFIGEST
  WHERE PRODUTO.CD_PRODUTO = ITEM_CONTAGEM_ABERTA.CD_PRODUTO
    AND ITEM_CONTAGEM_ABERTA.CD_CONTAGEM = :CAMPO_CD_CONTAGEM
    AND EMPRESA_PRODUTO.CD_PRODUTO = PRODUTO.CD_PRODUTO
    AND NVL(DECODE(NVL(configest.SN_LOTE_VAL_PRD_EMP,'N'),'S', EMPRESA_PRODUTO.SN_CONTROLE_VALIDADE, PRODUTO.SN_CONTROLE_VALIDADE),'N') = 'S'
    AND ITEM_CONTAGEM_ABERTA.DT_VALIDADE IS NULL
    AND CONFIGEST.CD_MULTI_EMPRESA = EMPRESA_PRODUTO.CD_MULTI_EMPRESA
    AND EMPRESA_PRODUTO.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
    AND ITEM_CONTAGEM_ABERTA.NR_SEQUENCIAL = :CAMPO_NR_SEQUENCIAL
UNION ALL
   SELECT ITEM_CONTAGEM_ABERTA.CD_SEQUENCIAL
      , PRODUTO.CD_PRODUTO
      , ITEM_CONTAGEM_ABERTA.CD_LOTE
      , ITEM_CONTAGEM_ABERTA.DT_VALIDADE
      , 'Existem produtos configurados para controlar serie, sem serie informado! Verifique sua digitação.' OBS
      , 0 QTD
   FROM DBAMV.PRODUTO
      , DBAMV.ITEM_CONTAGEM_ABERTA
      , DBAMV.EMPRESA_PRODUTO
  WHERE PRODUTO.CD_PRODUTO = ITEM_CONTAGEM_ABERTA.CD_PRODUTO
    AND ITEM_CONTAGEM_ABERTA.CD_CONTAGEM = :CAMPO_CD_CONTAGEM
    AND EMPRESA_PRODUTO.CD_PRODUTO = PRODUTO.CD_PRODUTO
    AND NVL(produto.sn_controla_serie,'N') = 'S'
    AND ITEM_CONTAGEM_ABERTA.CD_SERIE IS NULL
    AND EMPRESA_PRODUTO.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
    AND ITEM_CONTAGEM_ABERTA.NR_SEQUENCIAL = :CAMPO_NR_SEQUENCIAL)
ORDER BY CD_SEQUENCIAL
