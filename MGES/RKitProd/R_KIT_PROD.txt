--R_KIT_PROD
SELECT distinct
       itmvto_kit_produzido.cd_produto
     , produto.ds_produto
     , mvto_kit_produzido.cd_kit
     , produto2.ds_produto ds_kit
     , mvto_kit_produzido.dsp_cd_barras
     , mvto_kit_produzido.cd_mvto_estoque
     , nvl(mvto_kit_produzido.cd_estoque_destino, mvto_kit_produzido.cd_estoque) cd_estoque
     , mvto_kit_produzido.cd_formula
  FROM dbamv.mvto_kit_produzido
     , dbamv.itmvto_kit_produzido
     , dbamv.produto
     , dbamv.produto produto2
     , dbamv.estoque
     , dbamv.identificador_etiqueta
 WHERE mvto_kit_produzido.cd_mvto_estoque      = itmvto_kit_produzido.cd_mvto_estoque
   AND identificador_etiqueta.cd_identificador = dbamv.fnc_mges_valida_cod_barra(mvto_kit_produzido.dsp_cd_barras)
   and identificador_etiqueta.sn_ativo         = 'S'
   AND NVL(mvto_kit_produzido.sn_kit_armazenado,'N') = 'S'
   AND itmvto_kit_produzido.cd_produto         = produto.cd_produto
   AND mvto_kit_produzido.cd_kit               = produto2.cd_produto
   AND nvl(mvto_kit_produzido.cd_estoque_destino, mvto_kit_produzido.cd_estoque) = estoque.cd_estoque
   AND estoque.cd_multi_empresa                = dbamv.pkg_mv2000.le_empresa
ORDER BY cd_produto
       , cd_estoque
       , :CAMPO_CF_ORDEM
       , cd_mvto_estoque
