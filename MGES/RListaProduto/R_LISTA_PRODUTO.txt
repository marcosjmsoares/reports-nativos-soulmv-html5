--R_LISTA_PRODUTO
SELECT  PRODUTO.CD_PRODUTO CODIGO
       ,PRODUTO.DS_PRODUTO DESCRICAO
       ,PRODUTO.SN_MESTRE SN_MESTRE
       ,SUB_CLAS.CD_SUB_CLA CD_SUB_CLA
       ,SUB_CLAS.DS_SUB_CLA DS_SUB_CLA
       ,CLASSE.CD_CLASSE CD_CLASSE
       ,CLASSE.DS_CLASSE DS_CLASSE
       ,ESPECIE.CD_ESPECIE CD_ESPECIE
       ,ESPECIE.DS_ESPECIE DS_ESPECIE
FROM    dbamv.PRODUTO PRODUTO
       ,dbamv.SUB_CLAS SUB_CLAS
       ,dbamv.CLASSE CLASSE
       ,dbamv.ESPECIE ESPECIE
WHERE
           SUB_CLAS.CD_CLASSE = PRODUTO.CD_CLASSE
AND   SUB_CLAS.CD_ESPECIE = PRODUTO.CD_ESPECIE
AND   SUB_CLAS.CD_SUB_CLA = PRODUTO.CD_SUB_CLA
AND   CLASSE.CD_ESPECIE = SUB_CLAS.CD_ESPECIE
AND   CLASSE.CD_CLASSE = SUB_CLAS.CD_CLASSE
AND   ESPECIE.CD_ESPECIE = CLASSE.CD_ESPECIE
AND   PRODUTO.SN_MOVIMENTACAO = 'S'
And Exists ( Select Empresa_Produto.Cd_Produto
                     From Dbamv.Empresa_Produto
                   Where Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
                        and Empresa_Produto.Cd_Produto = Produto.Cd_Produto )
