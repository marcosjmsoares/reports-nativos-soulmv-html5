--R_LISTAGEM1
SELECT * FROM (
		SELECT especie.cd_especie,
		       especie.ds_especie,
		       produto.cd_produto codigo,
		       produto.ds_produto descricao,
		       produto.dt_cadastro,
		       produto.tp_ativo,
		       produto.dt_ultima_entrada,
		       produto.sn_controle_validade,
		       produto.sn_pscotropico,
		       produto.vl_custo_medio,
		       produto.vl_margem_lucro,
		       produto.vl_preco_de_venda,
		       produto.vl_tempo_de_reposicao,
		       produto.sn_mestre,
		       DECODE(DBAMV.PKT_CONFIGEST.RETORNA_CAMPO('SN_LOTE_VAL_PRD_EMP'), 'S',EMPRESA_PRODUTO.SN_MOVIMENTACAO,PRODUTO.SN_MOVIMENTACAO) sn_movimentacao,
		       produto.hr_ultima_entrada,
		       produto.sn_medicamento,
		       produto.tp_classificacao_xyz,
		       DECODE(DBAMV.PKT_CONFIGEST.RETORNA_CAMPO('SN_LOTE_VAL_PRD_EMP'), 'S',EMPRESA_PRODUTO.SN_BLOQUEIO_DE_COMPRA,PRODUTO.SN_BLOQUEIO_DE_COMPRA) sn_bloqueio_de_compra,
		       produto.cd_tip_ativ,
		       DECODE(NVL(DBAMV.PKT_CONFIGEST.RETORNA_CAMPO('SN_LOTE_VAL_PRD_EMP'),'N'),'S',empresa_produto.SN_LOTE, produto.SN_LOTE) sn_lote,
		       produto.sn_imprime_etiqueta,
		       produto_filho.cd_produto            codigo_filho,
		       produto_filho.ds_produto            descricao_filho,
		       produto_filho.dt_cadastro           dt_cadastro_filho,
		       produto_filho.tp_ativo              tp_ativo_filho,
		       produto_filho.dt_ultima_entrada     dt_ultima_entrada_filho,
		       produto_filho.sn_controle_validade  sn_controle_validade_filho,
		       produto_filho.sn_pscotropico        sn_pscotropico_filho,
		       produto_filho.vl_custo_medio        vl_custo_medio_filho,
		       produto_filho.vl_margem_lucro       vl_margem_lucro_filho,
		       produto_filho.vl_preco_de_venda     vl_preco_de_venda_filho,
		       produto_filho.vl_tempo_de_reposicao vl_tempo_de_reposicao_filho,
		       produto_filho.sn_mestre             sn_mestre_filho,
		       DECODE(DBAMV.PKT_CONFIGEST.RETORNA_CAMPO('SN_LOTE_VAL_PRD_EMP'), 'S',EMPRESA_PRODUTO_FILHO.SN_MOVIMENTACAO,PRODUTO_FILHO.SN_MOVIMENTACAO) sn_movimentacao_filho,
		       produto_filho.hr_ultima_entrada     sn_ultima_entrada_filho,
		       produto_filho.sn_medicamento        sn_medicamento_filho,
		       produto_filho.tp_classificacao_xyz  tp_classificacao_xyz_filho,
		       DECODE(DBAMV.PKT_CONFIGEST.RETORNA_CAMPO('SN_LOTE_VAL_PRD_EMP'), 'S',EMPRESA_PRODUTO_FILHO.SN_BLOQUEIO_DE_COMPRA,PRODUTO_FILHO.SN_BLOQUEIO_DE_COMPRA) sn_bloqueio_de_compra_filho,
		       produto_filho.cd_tip_ativ           cd_tip_ativ_filho,
		       DECODE(NVL(DBAMV.PKT_CONFIGEST.RETORNA_CAMPO('SN_LOTE_VAL_PRD_EMP'),'N'),'S',empresa_produto_filho.SN_LOTE, produto_filho.SN_LOTE)  sn_lote_filho,
		       produto_filho.sn_imprime_etiqueta   sn_imprime_etiqueta_filho
		FROM   dbamv.especie,
		       dbamv.produto ,
		       dbamv.produto produto_filho,
			   dbamv.empresa_produto empresa_produto,
			   dbamv.empresa_produto empresa_produto_filho
		WHERE   produto.cd_produto  = produto_filho.cd_produto_tem(+)
		AND    produto.cd_produto_tem is null
		AND    (produto.cd_especie  = :CAMPO_ESPECIE)
		AND    produto.cd_especie = especie.cd_especie
		and    empresa_produto.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
		and    Empresa_Produto.Cd_Produto = Produto.Cd_Produto
		and    empresa_produto_filho.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
		and    empresa_produto_filho.Cd_Produto = produto_filho.Cd_Produto
		ORDER BY produto.cd_especie,
		         produto.ds_produto,
		         produto_filho.ds_produto
		)
		ORDER BY DESCRICAO
