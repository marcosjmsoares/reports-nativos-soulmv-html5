--R_EMPRESTIMO_SAIDA_DIF
Select Itmvto_estoque.cd_produto                                    Produto,
            Produto.ds_produto                                                Ds_Produto,
            Verif_Ds_unid_prod(itmvto_estoque.cd_produto)   Ds_unidade,
            ItMvto_Estoque.Qt_Movimentacao                        Qt_Movimentacao,
            ItMvto_Estoque.Vl_Total_Emprestimo                    Vl_Total,
            Fornecedor.nm_fornecedor                                     Fornecedor,
            itmvto_estoque.cd_itmvto_estoque                         itMvto_estoque
   From Dbamv.Produto                                                       Produto,
            Dbamv.Uni_pro                                                       Uni_pro,
            Dbamv.itmvto_estoque                                           Itmvto_estoque,
            Dbamv.mvto_estoque                                             Mvto_estoque,
            Dbamv.fornecedor                                                  Fornecedor,
	    Dbamv.Estoque                  Estoque
Where itmvto_estoque.cd_produto             	= Produto.cd_produto
    And Itmvto_estoque.cd_uni_pro           	= Uni_pro.cd_uni_pro
    And itmvto_estoque.cd_mvto_estoque      	= mvto_estoque.cd_mvto_estoque
    And Mvto_estoque.cd_fornecedor          	= fornecedor.cd_fornecedor(+)
    AND MVTO_ESTOQUE.CD_ESTOQUE  		= ESTOQUE.CD_ESTOQUE
    AND ESTOQUE.CD_MULTI_EMPRESA 		= :CAMPO_CD_MULTI_EMPRESA
    And Mvto_Estoque.Tp_Mvto_Estoque  		= 'E'
    And Mvto_estoque.dt_mvto_estoque  		Between :CAMPO_DT_INIC order by  Fornecedor.nm_fornecedor
