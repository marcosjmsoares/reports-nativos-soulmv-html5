--R_ESTAT_CANC_SOLIC
SELECT
        ESTOQUE.CD_ESTOQUE
       ,ESTOQUE.DS_ESTOQUE
       ,SOLSAI_PRO.CD_SOLSAI_PRO
       ,SOLSAI_PRO.DT_SOLSAI_PRO
       ,SOLSAI_PRO.TP_SOLSAI_PRO
       ,SOLSAI_PRO.DT_CANCELAMENTO
       ,MOTIVO_CANC_SOL.CD_MOTIVO_CANC
       ,MOTIVO_CANC_SOL.DS_MOTIVO_CANC
       ,UNID_INT.DS_UNID_INT
       ,UNID_INT.CD_UNID_INT
       ,SETOR.NM_SETOR
       ,SETOR.CD_SETOR
FROM
        DBAMV.SOLSAI_PRO       SOLSAI_PRO
       ,DBAMV.MOTIVO_CANC_SOL  MOTIVO_CANC_SOL
       ,DBAMV.ESTOQUE          ESTOQUE
       ,DBAMV.UNID_INT         UNID_INT
       ,DBAMV.SETOR            SETOR
WHERE
      SOLSAI_PRO.CD_MOTIVO_CANC = MOTIVO_CANC_SOL.CD_MOTIVO_CANC
      AND ESTOQUE.CD_ESTOQUE = SOLSAI_PRO.CD_ESTOQUE
      AND UNID_INT.CD_UNID_INT(+) = SOLSAI_PRO.CD_UNID_INT
      AND SETOR.CD_SETOR(+) = SOLSAI_PRO.CD_SETOR
      AND ESTOQUE.CD_MULTI_EMPRESA = Dbamv.Pkg_mv2000.le_empresa
 ORDER BY     SOLSAI_PRO.CD_ESTOQUE, SOLSAI_PRO.CD_MOTIVO_CANC
