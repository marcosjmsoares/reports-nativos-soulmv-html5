--R_CONT_PROD_ESPECIFICO
SELECT * FROM (
SELECT CONTAGEM.CD_CONTAGEM,
CONTAGEM.CD_ESTOQUE,
ESTOQUE.DS_ESTOQUE,
CONTAGEM.DT_CONTAGEM,
CONTAGEM.TP_CONTAGEM,
CONTAGEM.SN_ABERTO,
CONTAGEM.DT_GERACAO,
ITCONTAGEM.CD_LINHA,
ITCONTAGEM.CD_PRODUTO CODIGO,
PRODUTO.DS_PRODUTO DESCRICAO,
ITCONTAGEM.CD_LOTE,
ITCONTAGEM.DT_VALIDADE,
ITCONTAGEM.CD_UNI_PRO,
UNI_PRO.DS_UNIDADE,
ITCONTAGEM.QT_ESTOQUE, 
ITCONTAGEM.QT_ESTOQUE_DOADO,
UNI_PRO.VL_FATOR,
ITCONTAGEM.VL_CUSTO_MEDIO,
ESPECIE.CD_ESPECIE,
ESPECIE.DS_ESPECIE
FROM DBAMV.CONTAGEM,
DBAMV.ESTOQUE,
DBAMV.ITCONTAGEM,
DBAMV.PRODUTO, 
DBAMV.UNI_PRO,
DBAMV.ESPECIE
WHERE ( contagem.dt_geracao, contagem.hr_geracao ) in
( select min( a.dt_geracao ), min( a.hr_geracao )
from dbamv.contagem a, dbamv.itcontagem i
where i.cd_produto = itcontagem.cd_produto and
a.sn_Aberto = 'N' and
i.cd_contagem = a.cd_contagem and 
a.cd_estoque = :CAMPO_CD_ESTOQUE and
a.cd_contagem like :CAMPO_CD_CONTAGEM ) 
AND ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA 
AND CONTAGEM.CD_CONTAGEM = ITCONTAGEM.CD_CONTAGEM
AND CONTAGEM.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
AND ITCONTAGEM.CD_PRODUTO = PRODUTO.CD_PRODUTO 
AND ITCONTAGEM.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO
AND PRODUTO.CD_ESPECIE = ESPECIE.CD_ESPECIE	
AND CONTAGEM.CD_ESTOQUE = :CAMPO_CD_ESTOQUE
AND CONTAGEM.CD_CONTAGEM LIKE :CAMPO_CD_CONTAGEM 
order by ESPECIE.CD_ESPECIE,
ESPECIE.DS_ESPECIE,
PRODUTO.DS_PRODUTO 
)
ORDER BY CD_ESTOQUE,DS_ESTOQUE, 
CD_ESPECIE,DS_ESPECIE
