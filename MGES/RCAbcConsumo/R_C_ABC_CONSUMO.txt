--R_C_ABC_CONSUMO
SELECT PRODUTO.DS_PRODUTO DS_PRODUTO,
       PRODUTO.CD_PRODUTO CD_PRODUTO,
       EMPRESA_PRODUTO.TP_CLASSIFICACAO_XYZ TP_CLASSIFICACAO_XYZ,
       VERIF_DS_UNID_PROD(PRODUTO.CD_PRODUTO)   DS_UNIDADE,
       SUM( RESULTADO.QT_CONSUMO) / VERIF_VL_FATOR_PROD(PRODUTO.CD_PRODUTO) QT_CONSUMO,
       SUM(RESULTADO.VL_CUSTO_PERIODO)  VL_CUSTO_PERIODO
 FROM (SELECT TRUNC( SUM( DECODE( MVTO_ESTOQUE.TP_MVTO_ESTOQUE, 'D', -1, 'C', -1, 'N', -1, 1 ) * (ITMVTO_ESTOQUE.QT_MOVIMENTACAO * UNI_PRO.VL_FATOR)),4) QT_CONSUMO,
              TRUNC( SUM( DECODE( MVTO_ESTOQUE.TP_MVTO_ESTOQUE, 'D', -1, 'C', -1, 'N', -1, 1 ) * (ITMVTO_ESTOQUE.QT_MOVIMENTACAO * UNI_PRO.VL_FATOR) *(VERIF_VL_CUSTO_MEDIO( PRODUTO.CD_PRODUTO, MVTO_ESTOQUE.DT_MVTO_ESTOQUE, :CAMPO_TP_CUSTO,PRODUTO.VL_CUSTO_MEDIO, MVTO_ESTOQUE.HR_MVTO_ESTOQUE,EMPRESA_PRODUTO.CD_MULTI_EMPRESA))),4) VL_CUSTO_PERIODO,
              DECODE('', 'S', PRODUTO.CD_PRODUTO_TEM, PRODUTO.CD_PRODUTO) CD_PRODUTO,
              empresa_produto.cd_multi_empresa
         FROM DBAMV.MVTO_ESTOQUE,
              DBAMV.ITMVTO_ESTOQUE,
              DBAMV.ESTOQUE,
              DBAMV.PRODUTO,
              DBAMV.UNI_PRO,
              dbamv.empresa_produto
        WHERE PRODUTO.TP_ATIVO = 'S'
          AND trunc(MVTO_ESTOQUE.DT_MVTO_ESTOQUE) BETWEEN :CAMPO_DT_INICIAL AND 
          AND MVTO_ESTOQUE.CD_MVTO_ESTOQUE = ITMVTO_ESTOQUE.CD_MVTO_ESTOQUE
          AND ITMVTO_ESTOQUE.CD_PRODUTO    = PRODUTO.CD_PRODUTO
          AND ITMVTO_ESTOQUE.CD_UNI_PRO    = UNI_PRO.CD_UNI_PRO
          AND ITMVTO_ESTOQUE.CD_PRODUTO    = UNI_PRO.CD_PRODUTO
          AND MVTO_ESTOQUE.CD_ESTOQUE      = ESTOQUE.CD_ESTOQUE
          and empresa_produto.cd_produto = produto.cd_produto
          and empresa_produto.cd_multi_empresa = estoque.cd_multi_empresa
        GROUP BY DECODE('', 'S', PRODUTO.CD_PRODUTO_TEM, PRODUTO.CD_PRODUTO), empresa_produto.cd_multi_empresa
       UNION ALL
       SELECT TRUNC( SUM( (ITMVTO_ESTOQUE.QT_MOVIMENTACAO * UNI_PRO.VL_FATOR)*(-1)),4) QT_CONSUMO,
              TRUNC( SUM( (ITMVTO_ESTOQUE.QT_MOVIMENTACAO * UNI_PRO.VL_FATOR) * (VERIF_VL_CUSTO_MEDIO( PRODUTO.CD_PRODUTO, MVTO_ESTOQUE.DT_MVTO_ESTOQUE, :CAMPO_TP_CUSTO, PRODUTO.VL_CUSTO_MEDIO, MVTO_ESTOQUE.HR_MVTO_ESTOQUE,EMPRESA_PRODUTO.CD_MULTI_EMPRESA ))*(-1)),4) VL_CUSTO_PERIODO,
              DECODE('', 'S', PRODUTO.CD_PRODUTO_TEM, PRODUTO.CD_PRODUTO) CD_PRODUTO,
              empresa_produto.cd_multi_empresa
         FROM DBAMV.MVTO_ESTOQUE,
              DBAMV.ITMVTO_ESTOQUE,
              DBAMV.ESTOQUE,
              DBAMV.PRODUTO,
              DBAMV.UNI_PRO,
              dbamv.empresa_produto
        WHERE PRODUTO.TP_ATIVO = 'S'
          AND trunc(MVTO_ESTOQUE.DT_MVTO_ESTOQUE) BETWEEN :CAMPO_DT_INICIAL AND 
          AND MVTO_ESTOQUE.CD_MVTO_ESTOQUE = ITMVTO_ESTOQUE.CD_MVTO_ESTOQUE
          AND ITMVTO_ESTOQUE.CD_PRODUTO    = PRODUTO.CD_PRODUTO
          AND ITMVTO_ESTOQUE.CD_UNI_PRO    = UNI_PRO.CD_UNI_PRO
          AND ITMVTO_ESTOQUE.CD_PRODUTO    = UNI_PRO.CD_PRODUTO
          AND MVTO_ESTOQUE.CD_ESTOQUE_DESTINO = ESTOQUE.CD_ESTOQUE
          and empresa_produto.cd_produto = produto.cd_produto
          and empresa_produto.cd_multi_empresa = estoque.cd_multi_empresa
        GROUP BY DECODE('', 'S', PRODUTO.CD_PRODUTO_TEM, PRODUTO.CD_PRODUTO),empresa_produto.cd_multi_empresa ) RESULTADO,
      DBAMV.PRODUTO,
      DBAMV.SUB_CLAS,
      DBAMV.EMPRESA_PRODUTO
WHERE PRODUTO.CD_SUB_CLA = SUB_CLAS.CD_SUB_CLA
  and empresA_produto.cd_produto = produto.cd_produto
  AND PRODUTO.TP_ATIVO = 'S'
  AND PRODUTO.CD_CLASSE = SUB_CLAS.CD_CLASSE
  AND PRODUTO.CD_ESPECIE = SUB_CLAS.CD_ESPECIE
  AND RESULTADO.CD_PRODUTO = PRODUTO.CD_PRODUTO
  AND empresa_produto.cd_multi_empresa = resultado.cd_multi_empresa
  AND PRODUTO.SN_MESTRE = NVL(:CAMPO_PROD_MESTRE,'N')
  AND RESULTADO.QT_CONSUMO > 0
GROUP BY PRODUTO.DS_PRODUTO,
         PRODUTO.CD_PRODUTO,
         EMPRESA_PRODUTO.TP_CLASSIFICACAO_XYZ,
         VERIF_DS_UNID_PROD(PRODUTO.CD_PRODUTO)
ORDER BY VL_CUSTO_PERIODO DESC
