--R_COTSET
SELECT * FROM (
Select Cota_Setor.Cd_Setor
	,Cota_Setor.Cd_Produto
	,Cota_Setor.Cd_Uni_Pro
	,Cota_Setor.Qt_Cota
	,Cota_Setor.Nr_Dias
	,Produto.Ds_Produto
    ,Produto.Ds_especificacao
	,Uni_Pro.Ds_Unidade
	,Setor.Nm_Setor
  From 	Dbamv.Setor
	,Dbamv.Uni_Pro
	,Dbamv.Produto
	,Dbamv.Cota_Setor
 Where :CAMPO_QUEBRA = 'S' and Setor.Cd_Multi_Empresa = Pkg_MV2000.Le_Empresa
   and Cota_Setor.Cd_Produto	= Produto.Cd_Produto
   AND Exists ( SELECT Empresa_Produto.Cd_Produto FROM Dbamv.Empresa_Produto WHERE Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA AND Empresa_Produto.Cd_Produto = Produto.Cd_Produto )
   and Cota_Setor.Cd_Uni_Pro		= Uni_Pro.Cd_Uni_Pro
   and Cota_Setor.Cd_Setor		= Setor.Cd_Setor
 And (:CAMPO_PS_CD_LOCALIZACAO = '%' Or Exists (Select 1 From DBAMV.EST_PRO EP Where EP.CD_ESTOQUE = COTA_SETOR.CD_ESTOQUE And EP.CD_PRODUTO = PRODUTO.CD_PRODUTO And EP.CD_LOCALIZACAO = :CAMPO_PS_CD_LOCALIZACAO))
UNION ALL
SELECT 0 Cd_Setor
     ,0 Cd_Produto
     ,0 Cd_Uni_Pro
     ,0 Qt_Cota
     ,0 Nr_Dias
     ,'#' Ds_Produto
     ,null ds_especificacao
	,'#' Ds_Unidade
	,'#' Nm_Setor
  FROM DUAL
WHERE '' = 'P'
order by Cd_Setor, Ds_Produto
)
