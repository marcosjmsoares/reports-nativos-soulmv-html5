/*----------------------------------------------------------------------------*/
--R_GER_ESTOQUE
SELECT CD_ESTOQUE,
               DS_ESTOQUE
  FROM DBAMV.ESTOQUE
 WHERE CD_ESTOQUE LIKE :CAMPO_ESTOQUE


/*----------------------------------------------------------------------------*/
--R_GER_ESTOQUE_R_ESTOQUE_Q_ESTOQUE
SELECT * FROM (
SELECT  ESTOQUE.CD_ESTOQUE
	,ESTOQUE.DS_ESTOQUE
	,SETOR.CD_SETOR
	,SETOR.NM_SETOR
	,ESPECIE.CD_ESPECIE
	,ESPECIE.DS_ESPECIE
	,SUM( Decode( Mvto_Estoque.Tp_Mvto_Estoque, 'D', 0, 'C', 0, 'N',0, (ITMVTO_ESTOQUE.QT_MOVIMENTACAO * UNI_PRO.VL_FATOR)  *
		dbamv.VERIF_VL_CUSTO_MEDIO(PRODUTO.CD_PRODUTO, MVTO_ESTOQUE.DT_MVTO_ESTOQUE, :CAMPO_TP_CUSTO, Null, Mvto_Estoque.Hr_Mvto_Estoque  ) ) ) VL_SAIDA
	,SUM( Decode( Mvto_Estoque.Tp_Mvto_Estoque, 'S', 0, 'P', 0, 'V',0, (ITMVTO_ESTOQUE.QT_MOVIMENTACAO * UNI_PRO.VL_FATOR)  *
		dbamv.VERIF_VL_CUSTO_MEDIO(PRODUTO.CD_PRODUTO, MVTO_ESTOQUE.DT_MVTO_ESTOQUE, :CAMPO_TP_CUSTO, Null, Mvto_Estoque.Hr_Mvto_Estoque  ) ) ) VL_DEVOLUCAO
FROM      DBAMV.MVTO_ESTOQUE
               ,DBAMV.SETOR
               ,DBAMV.ESTOQUE
               ,DBAMV.ITMVTO_ESTOQUE
               ,DBAMV.UNI_PRO
               ,DBAMV.PRODUTO
               ,DBAMV.ESPECIE
WHERE  MVTO_ESTOQUE.TP_MVTO_ESTOQUE IN ( 'S', 'P', 'D', 'C','V','N' )
AND        MVTO_ESTOQUE.DT_MVTO_ESTOQUE BETWEEN :CAMPO_DATA_INI AND 
AND        MVTO_ESTOQUE.CD_SETOR = SETOR.CD_SETOR
AND        ITMVTO_ESTOQUE.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO
AND        ESTOQUE.CD_ESTOQUE LIKE :CAMPO_ESTOQUE
AND        MVTO_ESTOQUE.CD_MVTO_ESTOQUE = ITMVTO_ESTOQUE.CD_MVTO_ESTOQUE
AND        ITMVTO_ESTOQUE.CD_PRODUTO = PRODUTO.CD_PRODUTO
AND        PRODUTO.CD_ESPECIE = ESPECIE.CD_ESPECIE
AND        MVTO_ESTOQUE.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
AND        ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY ESTOQUE.CD_ESTOQUE
	,ESTOQUE.DS_ESTOQUE
	,SETOR.CD_SETOR
	,SETOR.NM_SETOR
	,ESPECIE.CD_ESPECIE
	,ESPECIE.DS_ESPECIE
)
ORDER BY CD_SETOR,NM_SETOR


/*----------------------------------------------------------------------------*/
--R_GER_ESTOQUE_R_EST_TRANSF_Q_TRANSF
SELECT * FROM (
SELECT * FROM (
Select A.CD_ESTOQUE,
A.Cd_Est_Transf,
A.Ds_Est_Transf,
A.Cd_Especie_Transf,
A.Ds_Especie_Transf,
Sum( Vl_Transf ) Vl_Transf, Sum( Vl_Receb ) Vl_Receb
  From (
SELECT  ESTOQUE.CD_ESTOQUE CD_ESTOQUE
	,ESTOQUE_DES.CD_ESTOQUE CD_EST_TRANSF
	,ESTOQUE_DES.DS_ESTOQUE DS_EST_TRANSF
	,ESPECIE.CD_ESPECIE CD_ESPECIE_TRANSF
	,ESPECIE.DS_ESPECIE DS_ESPECIE_TRANSF
	,SUM( (ITMVTO_ESTOQUE.QT_MOVIMENTACAO * UNI_PRO.VL_FATOR)  *
		dbamv.VERIF_VL_CUSTO_MEDIO(PRODUTO.CD_PRODUTO, MVTO_ESTOQUE.DT_MVTO_ESTOQUE, :CAMPO_TP_CUSTO, Null, Mvto_Estoque.Hr_Mvto_Estoque  ) ) VL_TRANSF
	, 0 VL_RECEB
FROM	DBAMV.ESTOQUE ESTOQUE_DES
	,DBAMV.MVTO_ESTOQUE
               ,DBAMV.SETOR
               ,DBAMV.ESTOQUE
               ,DBAMV.ITMVTO_ESTOQUE
               ,DBAMV.UNI_PRO
               ,DBAMV.PRODUTO
               ,DBAMV.ESPECIE
WHERE  MVTO_ESTOQUE.TP_MVTO_ESTOQUE IN ( 'T' )
AND        ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND        MVTO_ESTOQUE.DT_MVTO_ESTOQUE BETWEEN :CAMPO_DATA_INI AND 
AND        MVTO_ESTOQUE.CD_SETOR = SETOR.CD_SETOR
AND        ITMVTO_ESTOQUE.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO
AND        MVTO_ESTOQUE.CD_MVTO_ESTOQUE = ITMVTO_ESTOQUE.CD_MVTO_ESTOQUE
AND        ITMVTO_ESTOQUE.CD_PRODUTO = PRODUTO.CD_PRODUTO
AND        PRODUTO.CD_ESPECIE = ESPECIE.CD_ESPECIE
AND        MVTO_ESTOQUE.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
AND	ESTOQUE_DES.CD_ESTOQUE (+)= MVTO_ESTOQUE.CD_ESTOQUE_DESTINO
GROUP BY ESTOQUE.CD_ESTOQUE
	,ESTOQUE_DES.CD_ESTOQUE
	,ESTOQUE_DES.DS_ESTOQUE
	,SETOR.CD_SETOR
	,SETOR.NM_SETOR
	,ESPECIE.CD_ESPECIE
	,ESPECIE.DS_ESPECIE
UNION ALL
SELECT  ESTOQUE_DES.CD_ESTOQUE CD_ESTOQUE
	,ESTOQUE.CD_ESTOQUE CD_EST_TRANSF
	,ESTOQUE.DS_ESTOQUE DS_EST_TRANSF
	,ESPECIE.CD_ESPECIE CD_ESPECIE_TRANSF
	,ESPECIE.DS_ESPECIE DS_ESPECIE_TRANSF
	, 0 VL_TRANSF
	,SUM( (ITMVTO_ESTOQUE.QT_MOVIMENTACAO * UNI_PRO.VL_FATOR)  *
		dbamv.VERIF_VL_CUSTO_MEDIO(PRODUTO.CD_PRODUTO, MVTO_ESTOQUE.DT_MVTO_ESTOQUE, :CAMPO_TP_CUSTO, Null, Mvto_Estoque.Hr_Mvto_Estoque  ) ) VL_RECEB
FROM	DBAMV.ESTOQUE ESTOQUE_DES
	,DBAMV.MVTO_ESTOQUE
               ,DBAMV.SETOR
               ,DBAMV.ESTOQUE
               ,DBAMV.ITMVTO_ESTOQUE
               ,DBAMV.UNI_PRO
               ,DBAMV.PRODUTO
               ,DBAMV.ESPECIE
WHERE  MVTO_ESTOQUE.TP_MVTO_ESTOQUE IN ( 'T' )
AND        ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND        MVTO_ESTOQUE.DT_MVTO_ESTOQUE BETWEEN :CAMPO_DATA_INI AND 
AND        MVTO_ESTOQUE.CD_SETOR = SETOR.CD_SETOR
AND        ITMVTO_ESTOQUE.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO
AND        MVTO_ESTOQUE.CD_MVTO_ESTOQUE = ITMVTO_ESTOQUE.CD_MVTO_ESTOQUE
AND        ITMVTO_ESTOQUE.CD_PRODUTO = PRODUTO.CD_PRODUTO
AND        PRODUTO.CD_ESPECIE = ESPECIE.CD_ESPECIE
AND        MVTO_ESTOQUE.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
AND	ESTOQUE_DES.CD_ESTOQUE (+)= MVTO_ESTOQUE.CD_ESTOQUE_DESTINO
GROUP BY
                 ESTOQUE_DES.CD_ESTOQUE
	,ESTOQUE.CD_ESTOQUE
	,ESTOQUE.DS_ESTOQUE
	,SETOR.CD_SETOR
	,SETOR.NM_SETOR
	,ESPECIE.CD_ESPECIE
	,ESPECIE.DS_ESPECIE
) A
Group By
A.Cd_Estoque,
A.Cd_Est_Transf,
A.Ds_Est_Transf,
A.Cd_Especie_Transf,
A.Ds_Especie_Transf
)
WHERE CD_ESTOQUE = :CAMPO_CD_ESTOQUE
)
ORDER BY CD_ESTOQUE,CD_EST_TRANSF,DS_EST_TRANSF
