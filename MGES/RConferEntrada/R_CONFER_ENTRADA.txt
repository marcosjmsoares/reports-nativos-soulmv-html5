--R_CONFER_ENTRADA
SELECT * FROM (
SELECT FORNECEDOR.CD_FORNECEDOR
              ,FORNECEDOR.NM_FORNECEDOR
              ,DOCUMENTO_ENTRADA.NR_DOCUMENTO
              ,TIP_DOC.CD_TIP_DOC
              ,TIP_DOC.DS_TIP_DOC
              ,DOCUMENTO_ENTRADA.NR_SERIE
              ,DOCUMENTO_ENTRADA.DT_EMISSAO
              ,DOCUMENTO_ENTRADA.VL_FRETE
              ,DOCUMENTO_ENTRADA.VL_TOTAL
              ,DOCUMENTO_ENTRADA.VL_IMPOSTO_ICMS
              ,DOCUMENTO_ENTRADA.VL_PERCENTUAL_ICMS
              ,DOCUMENTO_ENTRADA.VL_DESCONTO
              ,DOCUMENTO_ENTRADA.OBSERVACAO
              ,ITDOC_ENT_ORD_COM.CD_ORD_COM
              ,DOCUMENTO_ENTRADA.DT_ENTRADA
              ,UNI_PRO.DS_UNIDADE
              ,LABORATOR.NM_LABORATOR
              ,PRODUTO.CD_PRODUTO
              ,PRODUTO.DS_PRODUTO
   FROM DBAMV.DOCUMENTO_ENTRADA
              ,DBAMV.FORNECEDOR
              ,DBAMV.TIP_DOC
              ,DBAMV.ITDOC_ENT_ORD_COM
              ,DBAMV.ORD_COM
              ,DBAMV.ITORD_PRO
              ,DBAMV.UNI_PRO
              ,DBAMV.LABORATOR
              ,DBAMV.PRODUTO
WHERE DOCUMENTO_ENTRADA.CD_FORNECEDOR = FORNECEDOR.CD_FORNECEDOR
      AND DOCUMENTO_ENTRADA.CD_TIP_DOC           = TIP_DOC.CD_TIP_DOC
      AND DOCUMENTO_ENTRADA.NR_DOCUMENTO  = ITDOC_ENT_ORD_COM.NR_DOCUMENTO (+)
      AND DOCUMENTO_ENTRADA.NR_SERIE               = ITDOC_ENT_ORD_COM.NR_SERIE (+)
      AND DOCUMENTO_ENTRADA.CD_FORNECEDOR = ITDOC_ENT_ORD_COM.CD_FORNECEDOR (+)
      AND ITDOC_ENT_ORD_COM.CD_ORD_COM          = ORD_COM.CD_ORD_COM (+)
      AND ORD_COM.CD_ORD_COM                                = ITORD_PRO.CD_ORD_COM (+)
      AND ITORD_PRO.CD_UNI_PRO                               = UNI_PRO.CD_UNI_PRO (+)
      AND ITORD_PRO.CD_PRODUTO                             = PRODUTO.CD_PRODUTO (+)
      AND ITORD_PRO.DS_MARCA                                  = LABORATOR.CD_LABORATOR(+)
      AND DOCUMENTO_ENTRADA.NR_DOCUMENTO  = :CAMPO_NR_DOCUMENTO
      AND DOCUMENTO_ENTRADA.NR_SERIE              = :CAMPO_NR_SERIE
      AND DOCUMENTO_ENTRADA.CD_TIP_DOC          = :CAMPO_TIP_DOC
      AND FORNECEDOR.CD_FORNECEDOR                  = :CAMPO_FORNECEDOR
)
ORDER BY CD_FORNECEDOR,NM_FORNECEDOR,NR_DOCUMENTO,CD_TIP_DOC,DS_TIP_DOC,NR_SERIE,DT_EMISSAO,VL_FRETE,VL_TOTAL,VL_DESCONTO,VL_PERCENTUAL_ICMS,VL_IMPOSTO_ICMS,DT_ENTRADA,OBSERVACAO,
CD_ORD_COM
