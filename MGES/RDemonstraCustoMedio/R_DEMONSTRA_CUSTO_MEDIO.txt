--R_DEMONSTRA_CUSTO_MEDIO_ITEM_GRUPO_EMPRESA
SELECT * FROM (
          SELECT ESTOQUE.CD_MULTI_EMPRESA, ITENT_PRO.CD_PRODUTO,
						Sum(((ITENT_PRO.QT_ENTRADA * dbamv.mges.f_checa_uni_pro( ITENT_PRO.CD_UNI_PRO, ITENT_PRO.CD_PRODUTO)) / dbamv.VERIF_VL_FATOR_PROD(ITENT_PRO.cd_produto, 'R'))) QT_ENTRADA,
						 Sum(((ITENT_PRO.VL_UNITARIO/ dbamv.mges.f_checa_uni_pro( ITENT_PRO.CD_UNI_PRO, ITENT_PRO.CD_PRODUTO)) * (ITENT_PRO.QT_ENTRADA * dbamv.mges.f_checa_uni_pro( ITENT_PRO.CD_UNI_PRO, ITENT_PRO.CD_PRODUTO)) / dbamv.VERIF_VL_FATOR_PROD(ITENT_PRO.cd_produto, 'R'))) VL_TOTAL,
						  Sum(((ITENT_PRO.VL_CUSTO_REAL / dbamv.mges.f_checa_uni_pro( ITENT_PRO.CD_UNI_PRO, ITENT_PRO.CD_PRODUTO)) * (ITENT_PRO.QT_ENTRADA * dbamv.mges.f_checa_uni_pro( ITENT_PRO.CD_UNI_PRO, ITENT_PRO.CD_PRODUTO)) / dbamv.VERIF_VL_FATOR_PROD(ITENT_PRO.cd_produto, 'R')))  VL_CUSTO_REAL,
             (SELECT Sum(NVL(C_CONEST.qt_estoque_inicial,0))
              FROM DBAMV.C_CONEST,
                  DBAMV.ESTOQUE ES
              WHERE C_CONEST.CD_ESTOQUE = ES.CD_ESTOQUE
                AND C_CONEST.CD_PRODUTO = ITENT_PRO.cd_produto
                AND ES.CD_MULTI_EMPRESA = ESTOQUE.CD_MULTI_EMPRESA
                AND C_CONEST.CD_ANO = To_Char(trunc((trunc(:CAMPO_DT_ENTRADA,'MM')-1),'YYYY'),'YYYY')
                AND C_CONEST.CD_MES = To_Char(trunc((trunc(:CAMPO_DT_ENTRADA,'MM')-1),'MM'),'MM')) qt_estoque_inicial,
              (SELECT Sum(NVL(C_CONEST.vl_estoque_inicial,0))
              FROM DBAMV.C_CONEST,
                  DBAMV.ESTOQUE ES
              WHERE C_CONEST.CD_ESTOQUE = ES.CD_ESTOQUE
                AND C_CONEST.CD_PRODUTO = ITENT_PRO.cd_produto
                AND ES.CD_MULTI_EMPRESA = ESTOQUE.CD_MULTI_EMPRESA
                AND C_CONEST.CD_ANO = To_Char(trunc((trunc(:CAMPO_DT_ENTRADA,'MM')-1),'YYYY'),'YYYY')
                AND C_CONEST.CD_MES = To_Char(trunc((trunc(:CAMPO_DT_ENTRADA,'MM')-1),'MM'),'MM')) vl_estoque_inicial
						FROM DBAMV.ITENT_PRO, DBAMV.ENT_PRO,DBAMV.ESTOQUE
						WHERE ENT_PRO.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
						AND ITENT_PRO.CD_ENT_PRO = ENT_PRO.CD_ENT_PRO
                        AND ESTOQUE.CD_MULTI_EMPRESA = pkg_mv2000.le_empresa
                        AND ITENT_PRO.CD_PRODUTO = :CAMPO_CD_PRODUTO
						AND TRUNC(DT_ENTRADA,'MONTH') =  Trunc(:CAMPO_DT_ENTRADA,'MONTH')
                        AND 'N' = (SELECT nvl(sn_cons_est_corp,'N') FROM dbamv.multi_empresas WHERE multi_empresas.cd_multi_empresa =  pkg_mv2000.le_empresa)
                        GROUP BY ESTOQUE.CD_MULTI_EMPRESA,ITENT_PRO.CD_PRODUTO
UNION ALL
SELECT ESTOQUE.CD_MULTI_EMPRESA, ITENT_PRO.CD_PRODUTO,
						Sum(((ITENT_PRO.QT_ENTRADA * dbamv.mges.f_checa_uni_pro( ITENT_PRO.CD_UNI_PRO, ITENT_PRO.CD_PRODUTO)) / dbamv.VERIF_VL_FATOR_PROD(ITENT_PRO.cd_produto, 'R'))) QT_ENTRADA,
						 Sum(((ITENT_PRO.VL_UNITARIO/ dbamv.mges.f_checa_uni_pro( ITENT_PRO.CD_UNI_PRO, ITENT_PRO.CD_PRODUTO)) * (ITENT_PRO.QT_ENTRADA * dbamv.mges.f_checa_uni_pro( ITENT_PRO.CD_UNI_PRO, ITENT_PRO.CD_PRODUTO)) / dbamv.VERIF_VL_FATOR_PROD(ITENT_PRO.cd_produto, 'R'))) VL_TOTAL,
						  Sum(((ITENT_PRO.VL_CUSTO_REAL / dbamv.mges.f_checa_uni_pro( ITENT_PRO.CD_UNI_PRO, ITENT_PRO.CD_PRODUTO)) * (ITENT_PRO.QT_ENTRADA * dbamv.mges.f_checa_uni_pro( ITENT_PRO.CD_UNI_PRO, ITENT_PRO.CD_PRODUTO)) / dbamv.VERIF_VL_FATOR_PROD(ITENT_PRO.cd_produto, 'R')))  VL_CUSTO_REAL,
             (SELECT Sum(NVL(C_CONEST.qt_estoque_inicial,0))
              FROM DBAMV.C_CONEST,
                  DBAMV.ESTOQUE ES
              WHERE C_CONEST.CD_ESTOQUE = ES.CD_ESTOQUE
                AND C_CONEST.CD_PRODUTO = ITENT_PRO.cd_produto
                AND ES.CD_MULTI_EMPRESA = ESTOQUE.CD_MULTI_EMPRESA
                AND C_CONEST.CD_ANO = To_Char(trunc((trunc(:CAMPO_DT_ENTRADA,'MM')-1),'YYYY'),'YYYY')
                AND C_CONEST.CD_MES = To_Char(trunc((trunc(:CAMPO_DT_ENTRADA,'MM')-1),'MM'),'MM')) qt_estoque_inicial,
              (SELECT Sum(NVL(C_CONEST.vl_estoque_inicial,0))
              FROM DBAMV.C_CONEST,
                  DBAMV.ESTOQUE ES
              WHERE C_CONEST.CD_ESTOQUE = ES.CD_ESTOQUE
                AND C_CONEST.CD_PRODUTO = ITENT_PRO.cd_produto
                AND ES.CD_MULTI_EMPRESA = ESTOQUE.CD_MULTI_EMPRESA
                AND C_CONEST.CD_ANO = To_Char(trunc((trunc(:CAMPO_DT_ENTRADA,'MM')-1),'YYYY'),'YYYY')
                AND C_CONEST.CD_MES = To_Char(trunc((trunc(:CAMPO_DT_ENTRADA,'MM')-1),'MM'),'MM')) vl_estoque_inicial
						FROM DBAMV.ITENT_PRO, DBAMV.ENT_PRO,DBAMV.ESTOQUE
						WHERE ENT_PRO.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
						AND ITENT_PRO.CD_ENT_PRO = ENT_PRO.CD_ENT_PRO
                        AND ITENT_PRO.CD_PRODUTO = :CAMPO_CD_PRODUTO
						AND TRUNC(DT_ENTRADA,'MONTH') =  Trunc(:CAMPO_DT_ENTRADA,'MONTH')
                        AND ( ESTOQUE.CD_MULTI_EMPRESA IN (SELECT multi_empresas.cd_multi_empresa FROM dbamv.multi_empresas WHERE multi_empresas.cd_multi_empresa_consol = pkg_mv2000.le_empresa)
                                                          OR ESTOQUE.CD_MULTI_EMPRESA = pkg_mv2000.le_empresa
                                                         )
                                                    AND 'S' = (SELECT nvl(sn_cons_est_corp,'N') FROM dbamv.multi_empresas WHERE multi_empresas.cd_multi_empresa =  pkg_mv2000.le_empresa)
						GROUP BY ESTOQUE.CD_MULTI_EMPRESA,ITENT_PRO.CD_PRODUTO)
						ORDER BY CD_MULTI_EMPRESA, CD_PRODUTO

--R_DEMONSTRA_CUSTO_MEDIO_ITENT_PRO
SELECT * FROM (
     SELECT ESTOQUE.CD_MULTI_EMPRESA, ENT_PRO.CD_ENT_PRO, ENT_PRO.DT_ENTRADA,
						  ENT_PRO.CD_FORNECEDOR, ITENT_PRO.QT_ENTRADA, UNI_PRO.DS_UNIDADE CD_UNIDADE,
						  ITENT_PRO.VL_TOTAL, (ITENT_PRO.QT_ENTRADA * ITENT_PRO.VL_CUSTO_REAL) VL_CUSTO_REAL,
						  (FORNECEDOR.cd_fornecedor ||' - '|| FORNECEDOR.nm_fornecedor) DSP_DS_FORNECEDOR,
						   FORNECEDOR.NM_FORNECEDOR, ITENT_PRO.CD_PRODUTO,
						   ESTOQUE.CD_ESTOQUE,ESTOQUE.DS_ESTOQUE,
						   ((ITENT_PRO.QT_ENTRADA * dbamv.mges.f_checa_uni_pro( ITENT_PRO.CD_UNI_PRO, ITENT_PRO.CD_PRODUTO)) / dbamv.VERIF_VL_FATOR_PROD(ITENT_PRO.cd_produto, 'R')) QT_ENTRADA_REFERENCIA,
						   ((ITENT_PRO.VL_UNITARIO/ dbamv.mges.f_checa_uni_pro( ITENT_PRO.CD_UNI_PRO, ITENT_PRO.CD_PRODUTO)) * (ITENT_PRO.QT_ENTRADA * dbamv.mges.f_checa_uni_pro( ITENT_PRO.CD_UNI_PRO, ITENT_PRO.CD_PRODUTO)) / dbamv.VERIF_VL_FATOR_PROD(ITENT_PRO.cd_produto, 'R')) VL_TOTAL_REFERENCIA
						  FROM DBAMV.ITENT_PRO, DBAMV.ENT_PRO, DBAMV.UNI_PRO, DBAMV.ESTOQUE
						  ,DBAMV.FORNECEDOR
						  WHERE ITENT_PRO.CD_ENT_PRO = ENT_PRO.CD_ENT_PRO
						    AND ITENT_PRO.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO
						    AND ENT_PRO.CD_FORNECEDOR = FORNECEDOR.CD_FORNECEDOR
						     AND ITENT_PRO.CD_PRODUTO = :CAMPO_CD_PRODUTO
                             AND ESTOQUE.CD_MULTI_EMPRESA = pkg_mv2000.le_empresa
						     AND TRUNC(DT_ENTRADA,'MONTH') =  Trunc(:CAMPO_DT_ENTRADA,'MONTH')
						    AND ENT_PRO.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
                AND 'N' = (SELECT nvl(sn_cons_est_corp,'N') FROM dbamv.multi_empresas WHERE multi_empresas.cd_multi_empresa =  pkg_mv2000.le_empresa)
                UNION ALL
    SELECT ESTOQUE.CD_MULTI_EMPRESA, ENT_PRO.CD_ENT_PRO, ENT_PRO.DT_ENTRADA,
                              ENT_PRO.CD_FORNECEDOR, ITENT_PRO.QT_ENTRADA, UNI_PRO.DS_UNIDADE CD_UNIDADE,
                              ITENT_PRO.VL_TOTAL, (ITENT_PRO.QT_ENTRADA * ITENT_PRO.VL_CUSTO_REAL) VL_CUSTO_REAL,
                              (FORNECEDOR.cd_fornecedor ||' - '|| FORNECEDOR.nm_fornecedor) DSP_DS_FORNECEDOR,
                               FORNECEDOR.NM_FORNECEDOR, ITENT_PRO.CD_PRODUTO,
                               ESTOQUE.CD_ESTOQUE,ESTOQUE.DS_ESTOQUE,
                               ((ITENT_PRO.QT_ENTRADA * dbamv.mges.f_checa_uni_pro( ITENT_PRO.CD_UNI_PRO, ITENT_PRO.CD_PRODUTO)) / dbamv.VERIF_VL_FATOR_PROD(ITENT_PRO.cd_produto, 'R')) QT_ENTRADA_REFERENCIA,
                               ((ITENT_PRO.VL_UNITARIO/ dbamv.mges.f_checa_uni_pro( ITENT_PRO.CD_UNI_PRO, ITENT_PRO.CD_PRODUTO)) * (ITENT_PRO.QT_ENTRADA * dbamv.mges.f_checa_uni_pro( ITENT_PRO.CD_UNI_PRO, ITENT_PRO.CD_PRODUTO)) / dbamv.VERIF_VL_FATOR_PROD(ITENT_PRO.cd_produto, 'R')) VL_TOTAL_REFERENCIA
                              FROM DBAMV.ITENT_PRO, DBAMV.ENT_PRO, DBAMV.UNI_PRO, DBAMV.ESTOQUE
                              ,DBAMV.FORNECEDOR
                              WHERE ITENT_PRO.CD_ENT_PRO = ENT_PRO.CD_ENT_PRO
                                AND ITENT_PRO.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO
                                AND ENT_PRO.CD_FORNECEDOR = FORNECEDOR.CD_FORNECEDOR
                                 AND ITENT_PRO.CD_PRODUTO = :CAMPO_CD_PRODUTO
                                 AND TRUNC(DT_ENTRADA,'MONTH') =  Trunc(:CAMPO_DT_ENTRADA,'MONTH')
                                AND ENT_PRO.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
                                AND ( ESTOQUE.CD_MULTI_EMPRESA IN (SELECT multi_empresas.cd_multi_empresa FROM dbamv.multi_empresas WHERE multi_empresas.cd_multi_empresa_consol = pkg_mv2000.le_empresa)
                                      OR ESTOQUE.CD_MULTI_EMPRESA = pkg_mv2000.le_empresa
                                     )
                                AND 'S' = (SELECT nvl(sn_cons_est_corp,'N') FROM dbamv.multi_empresas WHERE multi_empresas.cd_multi_empresa =  pkg_mv2000.le_empresa)
                                   )
