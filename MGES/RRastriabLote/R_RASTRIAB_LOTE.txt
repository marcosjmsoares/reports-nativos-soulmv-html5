--R_RASTRIAB_LOTE
SELECT * FROM (
SELECT  NVL(LABORATOR.CD_LABORATOR,'Lote Sem Marca') CD_LABORATOR
       ,NVL(LABORATOR.NM_LABORATOR,' ') NM_LABORATOR
       ,FORNECEDOR.CD_FORNECEDOR
       ,FORNECEDOR.NM_FORNECEDOR
       ,ENT_PRO.DT_ENTRADA
       ,PRODUTO.CD_PRODUTO
       ,PRODUTO.DS_PRODUTO
       ,ITLOT_ENT.CD_LOTE
       ,ITLOT_ENT.DT_VALIDADE
       ,ITLOT_ENT.QT_ENT_PRO
       ,ITLOT_ENT.CD_REGISTRO_OPME CD_REGISTRO_ANVISA
       ,ITLOT_ENT.DT_VALIDADE_REG DT_REGISTRO_ANVISA
FROM DBAMV.ENT_PRO
     ,DBAMV.ITENT_PRO
     ,DBAMV.FORNECEDOR
     ,DBAMV.ITLOT_ENT
     ,DBAMV.LABORATOR
     ,DBAMV.PRODUTO
     ,DBAMV.ESTOQUE  
WHERE ENT_PRO.CD_ENT_PRO                 = ITENT_PRO.CD_ENT_PRO
  AND FORNECEDOR.CD_FORNECEDOR = ENT_PRO.CD_FORNECEDOR
  AND ITENT_PRO.CD_ITENT_PRO          = ITLOT_ENT.CD_ITENT_PRO(+)
  AND LABORATOR.CD_LABORATOR(+)       = ITLOT_ENT.DS_MARCA
  AND PRODUTO.CD_PRODUTO               = ITENT_PRO.CD_PRODUTO
  AND ENT_PRO.DT_ENTRADA BETWEEN :CAMPO_DTINIC AND 
  AND ESTOQUE.CD_ESTOQUE = ENT_PRO.CD_ESTOQUE  
  AND ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
  AND ENT_PRO.TP_DOCUMENTO_ENTRADA NOT IN ('Z')
ORDER BY CD_LABORATOR,DT_ENTRADA
)
ORDER BY CD_LABORATOR,DT_ENTRADA
