SELECT it.CD_PRODUTO 					   cd_produto
      ,p.ds_produto                        ds_produto
      ,it.CD_PRODUTO_TEM                   CD_PRODUTO_TEM
      ,p2.ds_produto                       ds_produto_tem
      ,it.QT_UTILIZADA                     QT_UTILIZADA
      ,it.QT_TARA                          QT_TARA
      ,it.QT_PESADO                        QT_PESADO
      ,it.QT_TOTAL                         QT_TOTAL
      ,it.cd_usuario_registro_pesagem      cd_usuario_registro_pesagem
      ,it.dt_registro_pesagem              dt_registro_pesagem
      ,it.cd_usuario_conferencia_pesagem   cd_usuario_conferencia_pesagem
      ,it.dt_conferencia_pesagem           dt_conferencia_pesagem
      ,u.ds_unidade                        ds_unidade
      ,IT.DS_MARCA                         DS_MARCA
      ,o.cd_lote                           cd_lote
      ,it.dt_validade                       dt_validade
      ,it.cd_lote                          cd_lote_materia
	  ,o.cd_ordem_producao                 cd_ordem_producao
  FROM dbamv.itordem_producao it
     , dbamv.ordem_producao   o
     , dbamv.produto          p
     , dbamv.produto          p2
     , dbamv.uni_pro          u
 WHERE it.cd_ordem_producao = :CAMPO_CD_ORDEM_PRODUCAO
   AND it.cd_ordem_producao = o.cd_ordem_producao
   AND it.cd_produto_tem = :CAMPO_CD_PRODUTO_TEM
   AND it.cd_produto = p.cd_produto
   AND p2.cd_produto = it.cd_produto_tem
   AND it.cd_uni_pro = u.cd_uni_pro
UNION ALL
