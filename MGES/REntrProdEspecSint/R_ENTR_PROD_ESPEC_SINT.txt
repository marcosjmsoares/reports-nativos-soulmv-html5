--R_ENTR_PROD_ESPEC_SINT
SELECT * FROM (
SELECT especie.cd_especie,
               especie.ds_especie,
               SUM(itent_pro.qt_entrada * itent_pro.vl_custo_real) vl_total
   FROM dbamv.tip_doc,
              dbamv.fornecedor,
              dbamv.produto,
              dbamv.uni_pro,
              dbamv.ent_pro,
              dbamv.itent_pro,
              dbamv.estoque,
              dbamv.especie
WHERE ent_pro.cd_ent_pro      = itent_pro.cd_ent_pro
      AND ent_pro.cd_fornecedor = fornecedor.cd_fornecedor
      AND ent_pro.cd_estoque     = estoque.cd_estoque
      AND ent_pro.cd_tip_doc      = tip_doc.cd_tip_doc
      AND itent_pro.cd_produto    = produto.cd_produto
      AND especie.cd_especie     = produto.cd_especie
      AND itent_pro.cd_uni_pro    = uni_pro.cd_uni_pro
      AND trunc(ent_pro.dt_entrada) BETWEEN :CAMPO_DTINIC
                                                                  AND 
      AND ent_pro.cd_fornecedor IS NOT NULL
      AND estoque.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
      AND ( ( :CAMPO_SN_EXIBE_ENT_CONSIG = 'N' AND ent_pro.tp_documento_entrada != 'C')
             OR :CAMPO_SN_EXIBE_ENT_CONSIG = 'S')
GROUP BY  especie.cd_especie,
                     especie.ds_especie
)
ORDER BY CD_ESPECIE
