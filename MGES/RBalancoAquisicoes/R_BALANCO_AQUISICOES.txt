--R_BALANCO_AQUISICOES
SELECT * FROM (
SELECT          ITENT_PRO.CD_PRODUTO            CD_PRODUTO,
                PRODUTO.DS_PRODUTO              DS_PRODUTO,
                UNI_PRO.VL_FATOR,
                ITENT_PRO.QT_ENTRADA            QTD_ENTRADA,
                ITENT_PRO.QT_ENTRADA  * vl_fator qtd_real,
                PRODUTO.CD_LISTA_CODIGO_MEDICAMENTO,
                LISTA_CODIGO_MEDICAMENTO.Ds_LISTA_CODIGO_MEDICAMENTO,
                CD_DCB,
                FORNECEDOR.NM_FORNECEDOR,
                FORNECEDOR.NR_CGC_CPF,
                ENT_PRO.NR_DOCUMENTO || '/' || LTRIM( ENT_PRO.NR_SERIE ) NR_DOCUMENTO,
                ENT_PRO.DT_ENTRADA,
                PORTARIA.CD_PORTARIA,
                PORTARIA.DS_PORTARIA
FROM    dbamv.ITENT_PRO          ITENT_PRO,
              dbamv.ENT_PRO            ENT_PRO,
              dbamv.PRODUTO            PRODUTO,
              dbamv.TIP_DOC            TIP_DOC,
              dbamv.UNI_PRO            UNI_PRO,
              dbamv. LISTA_CODIGO_MEDICAMENTO,
              dbamv.fornecedor,
              dbamv.PORTARIA,
             dbamv.itportaria,
	     dbamv.estoque
WHERE  ITENT_PRO.CD_PRODUTO      =   PRODUTO.CD_PRODUTO
    AND ENT_PRO.CD_ENT_PRO         =   ITENT_PRO.CD_ENT_PRO
    AND ENT_PRO.CD_TIP_DOC           =   TIP_DOC.CD_TIP_DOC
    AND UNI_PRO.CD_UNI_PRO           =   ITENT_PRO.CD_UNI_PRO
    AND ITENT_PRO.CD_UNI_PRO     =  UNI_PRO.CD_UNI_PRO
    AND PORTARIA.CD_PORTARIA(+)          = ITPORTARIA.CD_PORTARIA
    AND itPORTARIA.CD_PRODUTO(+)           = PRODUTO.CD_PRODUTO
    AND  TRUNC( ENT_PRO.DT_ENTRADA, 'MM' )     BETWEEN   to_date (:CAMPO_DTINIC , 'DD/MM/YYYY')  AND   to_date (:CAMPO_DTFIM, 'DD/MM/YYYY')
    AND PRODUTO.CD_LISTA_CODIGO_MEDICAMENTO = LISTA_CODIGO_MEDICAMENTO.CD_LISTA_CODIGO_MEDICAMENTO
    AND FORNECEDOR.CD_FORNECEDOR = ENT_PRO.CD_FORNECEDOR
    AND  FORNECEDOR.TP_CLIENTE_FORN	IN ( 'F', 'A' )
    AND ESTOQUE.CD_ESTOQUE       = ENT_PRO.CD_ESTOQUE
  AND ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
:CAMPO_CF_ORDER_BY
)
