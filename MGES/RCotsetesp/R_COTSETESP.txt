--R_COTSETESP
SELECT * FROM (
				Select Cota_Setor.Cd_Setor Cd_Setor
					,Cota_Setor.Cd_Produto Cd_Produto
					,Cota_Setor.Cd_Uni_Pro Cd_Uni_Pro
					,Cota_Setor.Qt_Cota Qt_Cota
					,Cota_Setor.Nr_Dias
					,Produto.Ds_Produto
				    ,Produto.DS_ESPECIFICACAO
				    ,Produto.cd_especie cd_especie
				    ,especie.ds_especie
					,Uni_Pro.Ds_Unidade
					,Setor.Nm_Setor
				  From 	Dbamv.Setor
					,Dbamv.Uni_Pro
					,Dbamv.Produto
					,Dbamv.Cota_Setor
					,Dbamv.especie
					,Dbamv.Empresa_Produto Empresa_Produto
				 Where Cota_Setor.Cd_Produto	= Produto.Cd_Produto
				   AND Empresa_Produto.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
				   AND Empresa_Produto.Cd_Produto = Produto.Cd_Produto
				   and Cota_Setor.Cd_Uni_Pro		= Uni_Pro.Cd_Uni_Pro
				   and Cota_Setor.Cd_Setor		= Setor.Cd_Setor
				   and Produto.cd_especie                      = Especie.cd_especie
				   AND setor.cd_multi_empresa = Empresa_Produto.Cd_Multi_Empresa
  And (:CAMPO_PS_CD_LOCALIZACAO = '%' Or Exists (Select 1 From DBAMV.EST_PRO EP Where EP.CD_ESTOQUE = COTA_SETOR.CD_ESTOQUE And EP.CD_PRODUTO = PRODUTO.CD_PRODUTO And EP.CD_LOCALIZACAO = :CAMPO_PS_CD_LOCALIZACAO))
				order by Cota_Setor.Cd_Setor, Produto.cd_especie, Produto.Ds_Produto
				)
				ORDER BY CD_SETOR,NM_SETOR,
				CD_ESPECIE,DS_ESPECIE , ds_produto
