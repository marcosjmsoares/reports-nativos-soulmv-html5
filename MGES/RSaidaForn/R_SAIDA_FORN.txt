--R_SAIDA_FORN
Select MVTO_ESTOQUE.CD_USUARIO,
       SOLSAI_PRO.CD_USUARIO As CD_USUARIO_SOLIC,
       MVTO_ESTOQUE.CD_MVTO_ESTOQUE As CD_MVTO_ESTOQUE_ME,
       MVTO_ESTOQUE.CD_ESTOQUE As CD_ESTOQUE_ME,
       MVTO_ESTOQUE.CD_SOLSAI_PRO As CD_SOLSAI_PRO_ME,
       ESTOQUE.DS_ESTOQUE As DS_ESTOQUE_ME,
       MVTO_PACIENTE.CD_PRESTADOR,
       PRESTADOR.NM_MNEMONICO As NM_MNEMONICO_ME,
       PRESTADOR.DS_CODIGO_CONSELHO || ' - ' || NM_PRESTADOR As CRM,
       MVTO_ESTOQUE.CD_SETOR As CD_SETOR_ME,
       SETOR.NM_SETOR As NM_SETOR_ME,
       MVTO_ESTOQUE.CD_UNID_INT As CD_UNID_INT_ME,
       UNID_INT.DS_UNID_INT As DS_UNID_INT_ME,
       MVTO_ESTOQUE.DT_MVTO_ESTOQUE As DT_MVTO_ESTOQUE_ME,
       MVTO_ESTOQUE.HR_MVTO_ESTOQUE As HR_MVTO_ESTOQUE_ME,
       MVTO_ESTOQUE.NR_DOCUMENTO As NR_DOCUMENTO_ME,
       MVTO_ESTOQUE.DT_IMPRESSAO,
       MVTO_ESTOQUE.CD_AVISO_CIRURGIA,
       MVTO_ESTOQUE.DS_OBS_EMPRESTIMO As DS_OBS_EMPRESTIMO_ME,
       MVTO_PACIENTE.NM_PACIENTE As NM_PACIENTE_MP,
       MVTO_PACIENTE.NM_MAE As NM_MAE_MP,
       MVTO_PACIENTE.DT_NASCIMENTO As DT_NASCIMENTO_MP,
       MVTO_PACIENTE.CD_CAD_SAME As CD_CAD_SAME_MP,
       MVTO_PACIENTE.NR_MATRICULA_SAME As NR_MATRICULA_SAME_MP,
       ITMVTO_ESTOQUE.CD_ITMVTO_ESTOQUE As CD_ITMVTO_ESTOQUE_IE,
       ITMVTO_ESTOQUE.CD_LOTE As CD_LOTE_IE,
       ITMVTO_ESTOQUE.CD_MVTO_ESTOQUE As CD_MVTO_ESTOQUE_IE,
       ITMVTO_ESTOQUE.CD_PRODUTO As CD_PRODUTO_IE,
       PRODUTO.DS_PRODUTO As DS_PRODUTO_IE,
       ITMVTO_ESTOQUE.CD_PRODUTO_KIT,
       PRODUTO_KIT.DS_PRODUTO As DS_PRODUTO_KIT,
       Decode(DBAMV.PKG_MGES_CONSIGNADO.Fnc_Sn_Prod_Processo_Consig(ITMVTO_ESTOQUE.CD_PRODUTO),
              'S',
              0,
              DBAMV.Verif_Vl_Custo_Medio(ITMVTO_ESTOQUE.CD_PRODUTO, MVTO_ESTOQUE.DT_MVTO_ESTOQUE, 'H', Null, MVTO_ESTOQUE.HR_MVTO_ESTOQUE) * UNI_PRO.VL_FATOR) As VL_UNITARIO,
       ITMVTO_ESTOQUE.CD_UNI_PRO As CD_UNI_PRO_IE,
       UNI_PRO.DS_UNIDADE As DS_UNIDADE_IE,
       ITMVTO_ESTOQUE.DT_VALIDADE As DT_VALIDADE_IE,
       ITMVTO_ESTOQUE.QT_MOVIMENTACAO,
       ITMVTO_ESTOQUE.QT_PERDA As QT_PERDA_IE,
       SOLSAI_PRO.CD_PACIENTE_INTEGRA,
       SOLSAI_PRO.NM_PACIENTE_INTEGRA,
       SOLSAI_PRO.CD_ATENDIMENTO_INTEGRA,
       MVTO_ESTOQUE.CD_FORNECEDOR,
       MVTO_ESTOQUE.NR_SERIE,
       MVTO_ESTOQUE.CD_CFOP,
       MVTO_ESTOQUE.CD_MOT_SAISET_FORN
  From DBAMV.MVTO_ESTOQUE
Inner Join DBAMV.ESTOQUE On
       ESTOQUE.CD_ESTOQUE = MVTO_ESTOQUE.CD_ESTOQUE
   And ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
Left Join DBAMV.ITMVTO_ESTOQUE On
       ITMVTO_ESTOQUE.CD_MVTO_ESTOQUE = MVTO_ESTOQUE.CD_MVTO_ESTOQUE
Left Join DBAMV.MVTO_PACIENTE On
       MVTO_PACIENTE.CD_MVTO_ESTOQUE = MVTO_ESTOQUE.CD_MVTO_ESTOQUE
Left Join DBAMV.PRESTADOR On
       PRESTADOR.CD_PRESTADOR = MVTO_PACIENTE.CD_PRESTADOR
Left Join DBAMV.UNID_INT On
       UNID_INT.CD_UNID_INT = MVTO_ESTOQUE.CD_UNID_INT
Left Join DBAMV.SETOR On
       SETOR.CD_SETOR = MVTO_ESTOQUE.CD_SETOR
Left Join DBAMV.UNI_PRO On
       UNI_PRO.CD_UNI_PRO = ITMVTO_ESTOQUE.CD_UNI_PRO
Left Join DBAMV.PRODUTO On
       PRODUTO.CD_PRODUTO = ITMVTO_ESTOQUE.CD_PRODUTO
Left Join DBAMV.PRODUTO PRODUTO_KIT On
       PRODUTO_KIT.CD_PRODUTO = ITMVTO_ESTOQUE.CD_PRODUTO_KIT
Left Join DBAMV.ITSOLSAI_PRO On
       ITSOLSAI_PRO.CD_ITSOLSAI_PRO = ITMVTO_ESTOQUE.CD_ITSOLSAI_PRO
Left Join DBAMV.SOLSAI_PRO On
       SOLSAI_PRO.CD_SOLSAI_PRO = MVTO_ESTOQUE.CD_SOLSAI_PRO
 Where MVTO_ESTOQUE.TP_MVTO_ESTOQUE In ('S',
                                        'V')
   And MVTO_ESTOQUE.CD_MVTO_ESTOQUE = Nvl(:CAMPO_CD_MVTO_ESTOQUE,
                                          MVTO_ESTOQUE.CD_MVTO_ESTOQUE)
   And MVTO_ESTOQUE.CD_FORNECEDOR Is Not Null
Order By CD_PRESTADOR,
       CD_ESTOQUE_ME,
       DS_ESTOQUE_ME,
       NM_MNEMONICO_ME,
       CD_SETOR_ME,
       NM_SETOR_ME,
       CRM,
       CD_UNID_INT_ME,
       DS_UNID_INT_ME,
       DT_MVTO_ESTOQUE_ME,
       HR_MVTO_ESTOQUE_ME,
       NR_DOCUMENTO_ME,
       CD_MVTO_ESTOQUE_ME,
       CD_USUARIO,
       CD_SOLSAI_PRO_ME,
       NM_PACIENTE_MP,
       NM_MAE_MP,
       DT_NASCIMENTO_MP,
       DT_IMPRESSAO,
       CD_AVISO_CIRURGIA,
       CD_PRODUTO_KIT,
       DS_PRODUTO_KIT
