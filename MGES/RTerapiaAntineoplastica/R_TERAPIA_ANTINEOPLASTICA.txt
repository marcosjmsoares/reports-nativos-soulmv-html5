--R_TERAPIA_ANTINEOPLASTICA
Select Mvto_Estoque.Cd_Mvto_Estoque         Cd_Mvto_Estoque
     , Mvto_Estoque.Cd_Estoque                       Cd_Estoque
     , Estoque.Ds_Estoque                                 Ds_Estoque
     , Mvto_Estoque.Dt_Mvto_Estoque             Dt_Mvto_Estoque
     , Mvto_Estoque.Hr_Mvto_Estoque             Hr_Mvto_Estoque
     , Mvto_Estoque.Cd_Atendimento                Cd_Atendimento
     , Atendime.Cd_Paciente                              Cd_Paciente
     , Paciente.Nm_Paciente                             Nm_Paciente
     , Mvto_Estoque.Cd_Unid_Int                      Cd_Unid_Int
     , Unid_Int.Ds_Unid_Int                                Ds_Unid_Int
     , Mvto_Estoque.Cd_Prestador                    Cd_Prestador
     , Prestador.Nm_Prestador                           Nm_Prestador
     , Mvto_Estoque.Cd_Setor                          Cd_Setor
     , Setor.Nm_Setor                                        Nm_Setor
     , Leito.Cd_Leito                                          Cd_Leito
     , Leito.Ds_Leito                                          Ds_Leito
     , Atendime.Cd_Convenio                           Cd_Convenio
     , Convenio.Nm_Convenio                         Nm_Convenio
     , Mvto_Estoque.Cd_SolSai_Pro                Cd_SolSai_Pro
     , Mvto_Estoque.Cd_Usuario                     Cd_Usuario
    , Solsai_pro.cd_pre_med                           Cd_Pre_Med
  From Dbamv.Mvto_Estoque
     , Dbamv.Estoque
     , Dbamv.Atendime
     , Dbamv.Paciente
     , Dbamv.Unid_Int
     , Dbamv.Prestador
     , Dbamv.Setor
     , Dbamv.Convenio
     , Dbamv.Leito
     , Dbamv.Solsai_pro
 Where Mvto_Estoque.Cd_Mvto_Estoque = :CAMPO_CD_MVTO_ESTOQUE
   And Mvto_Estoque.Cd_Estoque = Estoque.Cd_Estoque
   And Mvto_Estoque.Cd_Atendimento = Atendime.Cd_Atendimento
   And Atendime.Cd_Paciente = Paciente.Cd_Paciente
   And Mvto_Estoque.Cd_Unid_Int = Unid_Int.Cd_Unid_Int(+)
   And Mvto_Estoque.Cd_Prestador = Prestador.Cd_Prestador
   And Mvto_Estoque.Cd_Setor = Setor.Cd_Setor
   And Atendime.Cd_Convenio = Convenio.Cd_Convenio
   And Atendime.Cd_Leito = Leito.Cd_Leito(+)
  And Solsai_pro.cd_solsai_pro = Mvto_estoque.cd_solsai_pro
  and atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA

--R_TERAPIA_ANTINEOPLASTICA_R_3_Q_ITPRE_MED
SELECT * FROM (
Select ItPre_Med.Cd_ItPre_Med Cd_ItPre_Med
     , ItPre_Med.Cd_Tip_Presc Cd_Tip_Presc
     , Tip_Presc.Ds_Tip_Presc Ds_Tip_Presc
     , ItPre_Med.Qt_ItPre_Med Qt_ItPre_Med
     , ItPre_Med.Cd_Unidade Cd_Unidade
     , nvl(Unidade.Ds_Unidade , Uni_Pro.Ds_Unidade) Ds_Unidade_it
     , ItPre_Med.Cd_Uni_Pro Cd_Uni_Pro
     , ItPre_Med.Cd_For_Apl Cd_For_Apl
     , ItPre_Med.Cd_Tip_Fre Cd_Tip_Fre
     , Tip_Fre.Ds_Tip_Fre Ds_Tip_Fre
     , Nvl2(ItPre_Med.Ds_Justificativa,  'Justificativa: '||ItPre_Med.Ds_Justificativa, null ) Ds_Justificativa
     , Nvl2(ItPre_Med.Ds_ItPre_Med,  'Obs.: '||ItPre_Med.Ds_ItPre_Med , null)                  Ds_ItPre_Med
     , Tip_Presc.ds_obs_adicional Observacao
     , ItPre_Med.Cd_Pre_Med Cd_Pre_Med
     , Uni_Pro.vl_fator Vl_Fator
     , Pre_med.cd_atendimento Cd_Atendimento
     , Uni_Pro.Cd_Produto Cd_Produto_It
  From Dbamv.ItPre_Med
     , Dbamv.Tip_Presc
     , Dbamv.Tip_Fre
     , Dbamv.Uni_Pro
     , Dbamv.Unidade
     , Dbamv.Pre_med
 Where ItPre_Med.Cd_Tip_Presc = Tip_Presc.Cd_Tip_Presc
   And ItPre_Med.Cd_Tip_Fre = Tip_Fre.Cd_Tip_Fre
   And ItPre_Med.Cd_Uni_Pro = Uni_Pro.Cd_Uni_Pro(+)
   And ItPre_Med.Cd_Unidade = Unidade.Cd_Unidade(+)
   And ItPre_Med.Cd_Pre_Med = Pre_Med.Cd_Pre_Med
Order By ItPre_Med.Cd_ItPre_Med
)
WHERE CD_PRE_MED = :CAMPO_CD_PRE_MED

--R_TERAPIA_ANTINEOPLASTICA_R_5_Q_1
SELECT * FROM (
Select ItMvto_Estoque.Cd_Produto
     , Produto.Ds_Produto
     , Lot_Pro.Ds_Marca
     , ItMvto_Estoque.Cd_Uni_Pro
     , Uni_Pro.Ds_Unidade
     , ItMvto_Estoque.Cd_Lote
     , ItMvto_Estoque.Dt_Validade
     , sum(ItMvto_Estoque.Qt_Movimentacao) qt_mov
     , Mvto_Estoque.Cd_Mvto_Estoque
  From Dbamv.ItMvto_Estoque
     , Dbamv.Mvto_Estoque
     , Dbamv.Produto
     , Dbamv.Lot_Pro
     , Dbamv.Uni_Pro
     , Dbamv.ItSolSai_Pro
 Where  ItMvto_Estoque.Cd_Mvto_Estoque = Mvto_Estoque.Cd_Mvto_Estoque
   And ItMvto_Estoque.Cd_Produto = Produto.Cd_Produto
   And ItMvto_Estoque.Cd_Lote = Lot_Pro.Cd_Lote(+)
   And ItMvto_Estoque.Dt_Validade = Lot_Pro.Dt_Validade(+)
   And ItMvto_Estoque.Cd_Produto = Lot_Pro.Cd_Produto(+)
   And Mvto_Estoque.Cd_Estoque = Lot_Pro.Cd_Estoque
   And ItMvto_Estoque.Cd_Uni_Pro = Uni_Pro.Cd_Uni_Pro
   And ItMvto_Estoque.Cd_ItSolSai_Pro = ItSolSai_Pro.Cd_ItSolSai_Pro
group by  ItMvto_Estoque.Cd_Produto
     , Produto.Ds_Produto
     , Lot_Pro.Ds_Marca
     , ItMvto_Estoque.Cd_Uni_Pro
     , Uni_Pro.Ds_Unidade
     , ItMvto_Estoque.Cd_Lote
     , ItMvto_Estoque.Dt_Validade
     , Mvto_Estoque.Cd_Mvto_Estoque
)
WHERE CD_MVTO_ESTOQUE = :CAMPO_CD_MVTO_ESTOQUE
