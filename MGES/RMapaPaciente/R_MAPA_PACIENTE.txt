--R_MAPA_PACIENTE_PRODUTO
SELECT ITPRE_MED.CD_PRE_MED,
ITPRE_MED.CD_ITPRE_MED,
PACIENTE.CD_PACIENTE,
PACIENTE.NM_PACIENTE,
PRE_MED.CD_ATENDIMENTO,
PREPARACAO_LOTE.DS_STATUS,
SOLSAI_PRO.CD_SOLSAI_PRO,
To_CHAR(DT_PRE_MED,'DD/MM/YYYY') DT_PRE_MED, 
To_CHAR(HR_PRE_MED,'HH24') HR_PRE_MED,
To_CHAR(PREPARACAO_LOTE.DT_CONCLUSAO, 'DD/MM/YYYY HH24' ) DT_CONCLUSAO,
PREPARACAO_LOTE.CD_USUARIO_LIB
FROM DBAMV.PRE_MED,
DBAMV.ITPRE_MED,
DBAMV.ATENDIME,
DBAMV.PACIENTE,
DBAMV.SOLSAI_PRO,
DBAMV.TIP_PRESC,
DBAMV.PRODUTO,
(SELECT PREPARACAO_LOTE.CD_PRE_MED, 
        CD_ITPRE_MED, CD_PREPARACAO_LOTE, PREPARACAO_LOTE.CD_SOLSAI_PRO CD_SOLICITACAO, 
        TP_STATUS, DS_STATUS, PREPARACAO_LOTE.CD_USUARIO_LIB, PREPARACAO_LOTE.DT_CONCLUSAO 
FROM DBAMV.PREPARACAO_LOTE,
DBAMV.SOLSAI_PRO
WHERE PREPARACAO_LOTE.TP_STATUS IN ('C','L')
AND PREPARACAO_LOTE.CD_SOLSAI_PRO = SOLSAI_PRO.CD_SOLSAI_PRO) PREPARACAO_LOTE
WHERE ITPRE_MED.CD_ITPRE_MED || SOLSAI_PRO.CD_SOLSAI_PRO in :CAMPO_CD_ITPRE_MED
AND PRE_MED.CD_PRE_MED = ITPRE_MED.CD_PRE_MED
AND PRE_MED.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO 
AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE 
AND PRE_MED.CD_PRE_MED = SOLSAI_PRO.CD_PRE_MED
AND ITPRE_MED.CD_TIP_PRESC = TIP_PRESC.CD_TIP_PRESC
AND TIP_PRESC.CD_PRODUTO = PRODUTO.CD_PRODUTO
AND PRODUTO.SN_ONCOLOGIA = 'S' 
AND TIP_PRESC.CD_PRODUTO = PRODUTO.CD_PRODUTO (+)
AND PREPARACAO_LOTE.CD_ITPRE_MED(+) = ITPRE_MED.CD_ITPRE_MED
UNION ALL
SELECT ITPRE_MED.CD_PRE_MED,
ITPRE_MED.CD_ITPRE_MED,
PACIENTE.CD_PACIENTE,
PACIENTE.NM_PACIENTE,
PRE_MED.CD_ATENDIMENTO,
'SEM PREPARO' DS_STATUS,   
NULL CD_SOLSAI_PRO,          
To_CHAR(DT_PRE_MED,'DD/MM/YYYY') DT_PRE_MED, 
To_CHAR(HR_PRE_MED,'HH24') HR_PRE_MED,
NULL DT_CONCLUSAO,
NULL CD_USUARIO_LIB
FROM DBAMV.PRE_MED,
DBAMV.ITPRE_MED,
DBAMV.TIP_PRESC,
DBAMV.PRODUTO,
DBAMV.ATENDIME,
DBAMV.PACIENTE
WHERE ITPRE_MED.CD_ITPRE_MED || '111' IN :CAMPO_CD_ITPRE_MED 
AND PRE_MED.CD_PRE_MED = ITPRE_MED.CD_PRE_MED
AND PRE_MED.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO 
AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE   
AND ITPRE_MED.CD_TIP_PRESC = TIP_PRESC.CD_TIP_PRESC
AND TIP_PRESC.CD_PRODUTO = PRODUTO.CD_PRODUTO
AND PRODUTO.SN_ONCOLOGIA = 'S' 
AND TIP_PRESC.CD_PRODUTO = PRODUTO.CD_PRODUTO



/*----------------------------------------------------------------------------*/
--R_MAPA_PRODUTO_PRODUTO
SELECT PRODUTO.CD_PRODUTO,
       PRODUTO.DS_PRODUTO || ' (ONCOLOGICO)' PRODUTO,
       ITPRE_MED.QT_ITPRE_MED QUANTIDADE,
       UNI_PRO.DS_UNIDADE UNIDADE,
       UNI_PRO.CD_UNI_PRO
  FROM DBAMV.PRE_MED,
       DBAMV.ITPRE_MED,
       DBAMV.TIP_PRESC,
       DBAMV.PRODUTO,
       DBAMV.UNI_PRO
 WHERE PRE_MED.CD_PRE_MED = ITPRE_MED.CD_PRE_MED
   AND ITPRE_MED.CD_TIP_PRESC = TIP_PRESC.CD_TIP_PRESC
   AND Nvl(TIP_PRESC.CD_PRODUTO, ITPRE_MED.CD_PRODUTO) = PRODUTO.CD_PRODUTO
   AND ITPRE_MED.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO
   AND ITPRE_MED.CD_ITPRE_MED = :CAMPO_CD_ITPRE_MED 
UNION ALL
SELECT PRODUTO.CD_PRODUTO,
			 PRODUTO.DS_PRODUTO PRODUTO,
       CITPRE_MED.QT_COMPONENTE QUANTIDADE,
			 UNI_PRO.DS_UNIDADE UNIDADE,
			 UNI_PRO.CD_UNI_PRO
  FROM DBAMV.PRE_MED,
       DBAMV.ITPRE_MED,
       DBAMV.CITPRE_MED,
       DBAMV.PRODUTO,
       DBAMV.UNI_PRO
 WHERE PRE_MED.CD_PRE_MED = ITPRE_MED.CD_PRE_MED
   AND ITPRE_MED.CD_ITPRE_MED = CITPRE_MED.CD_ITPRE_MED
   AND CITPRE_MED.CD_PRODUTO = PRODUTO.CD_PRODUTO
   AND CITPRE_MED.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO
   AND ITPRE_MED.CD_ITPRE_MED  = :CAMPO_CD_ITPRE_MED



/*----------------------------------------------------------------------------*/
--R_MAPA_PRODUTO_RESUMO_PRODUTO
SELECT PRODUTO.CD_PRODUTO,
       PRODUTO.DS_PRODUTO,
		   UNI_PRO.DS_UNIDADE,
		   1 ORDEM,
       Sum(ITPRE_MED.QT_ITPRE_MED) QT_SOLICITADO
  FROM DBAMV.PRE_MED,
       DBAMV.ITPRE_MED,
       DBAMV.TIP_PRESC,
       DBAMV.PRODUTO,
       DBAMV.UNI_PRO
 WHERE PRE_MED.CD_PRE_MED = ITPRE_MED.CD_PRE_MED
   AND ITPRE_MED.CD_TIP_PRESC = TIP_PRESC.CD_TIP_PRESC
   AND Nvl(TIP_PRESC.CD_PRODUTO, ITPRE_MED.CD_PRODUTO) = PRODUTO.CD_PRODUTO
   AND ITPRE_MED.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO
   AND ITPRE_MED.CD_ITPRE_MED IN :CAMPO_ITPRE_MED
 GROUP
    BY PRODUTO.CD_PRODUTO,
       PRODUTO.DS_PRODUTO,
		   UNI_PRO.DS_UNIDADE
UNION ALL
SELECT PRODUTO.CD_PRODUTO,
       PRODUTO.DS_PRODUTO,
		   UNI_PRO.DS_UNIDADE,
		   2 ORDEM,
       Sum(CITPRE_MED.QT_COMPONENTE) QT_SOLICITADO
  FROM DBAMV.PRE_MED,
       DBAMV.ITPRE_MED,
       DBAMV.CITPRE_MED,
       DBAMV.PRODUTO,
       DBAMV.UNI_PRO
 WHERE PRE_MED.CD_PRE_MED = ITPRE_MED.CD_PRE_MED
   AND ITPRE_MED.CD_ITPRE_MED = CITPRE_MED.CD_ITPRE_MED
   AND CITPRE_MED.CD_PRODUTO = PRODUTO.CD_PRODUTO
   AND CITPRE_MED.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO
   AND ITPRE_MED.CD_ITPRE_MED  IN   :CAMPO_ITPRE_MED
 GROUP
    BY PRODUTO.CD_PRODUTO,
       PRODUTO.DS_PRODUTO,
		   UNI_PRO.DS_UNIDADE
ORDER BY ORDEM, DS_PRODUTO
