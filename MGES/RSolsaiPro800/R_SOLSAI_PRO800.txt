select solsai_pro.cd_solsai_pro
      ,solsai_pro.cd_pre_med
      ,solsai_pro.cd_estoque
      ,estoque.ds_estoque
      ,solsai_pro.cd_estoque_solicitante
      ,estoque_solicitante.ds_estoque ds_estoque_solicitante
      ,solsai_pro.cd_atendimento
      ,solsai_pro.cd_unid_int
      ,unid_int.ds_unid_int
      ,solsai_pro.cd_setor
      ,setor.nm_setor
      ,area.cd_area
      ,area.ds_area
      ,atendime.cd_paciente
      ,paciente.nm_paciente
      ,atendime.cd_convenio
      ,convenio.nm_convenio
      ,atendime.cd_con_pla
      ,con_pla.ds_con_pla
      ,solsai_pro.cd_prestador
      ,prestador.nm_prestador
      ,solsai_pro.dt_solsai_pro
      ,solsai_pro.hr_solsai_pro
      ,solsai_pro.cd_usuario
      ,solsai_pro.sn_urgente
      ,solsai_pro.ds_obs
      ,est_pro.cd_localizacao
      ,localizacao.ds_localizacao
      ,itsolsai_pro.cd_produto
      ,produto.ds_produto
      ,uni_pro.ds_unidade
      ,est_pro.qt_estoque_atual
      ,pre_med.dh_impressao dt_data_prescricao
      ,sum(itsolsai_pro.qt_solicitado) qt_solicitado
  from dbamv.solsai_pro
      ,dbamv.itsolsai_pro
      ,dbamv.estoque
      ,dbamv.atendime
      ,dbamv.paciente
      ,dbamv.convenio
      ,dbamv.con_pla
      ,dbamv.prestador
      ,dbamv.est_pro
      ,dbamv.localizacao
      ,dbamv.produto
      ,dbamv.uni_pro
      ,dbamv.setor
      ,dbamv.unid_int
      ,dbamv.estoque estoque_solicitante
      ,dbamv.atendime_hcare
      ,dbamv.area
      ,dbamv.pre_med
 where itsolsai_pro.cd_solsai_pro        = solsai_pro.cd_solsai_pro
   and estoque.cd_estoque                = solsai_pro.cd_estoque
   and estoque_solicitante.cd_estoque(+) = solsai_pro.cd_estoque_solicitante
   and atendime.cd_atendimento(+)        = solsai_pro.cd_atendimento
   and paciente.cd_paciente(+)           = atendime.cd_paciente
   and setor.cd_setor(+)                 = solsai_pro.cd_setor
   and unid_int.cd_unid_int(+)           = solsai_pro.cd_unid_int
   and convenio.cd_convenio(+)           = atendime.cd_convenio
   and con_pla.cd_convenio(+)            = atendime.cd_convenio
   and con_pla.cd_con_pla(+)             = atendime.cd_con_pla
   and prestador.cd_prestador(+)         = atendime.cd_prestador
   and est_pro.cd_estoque                = solsai_pro.cd_estoque
   and est_pro.cd_produto                = itsolsai_pro.cd_produto
   and localizacao.cd_localizacao(+)     = est_pro.cd_localizacao
   and produto.cd_produto                = itsolsai_pro.cd_produto
   and uni_Pro.cd_uni_pro                = itsolsai_pro.cd_uni_pro
   and atendime_hcare.cd_atendimento(+)  = atendime.cd_atendimento
   and area.cd_area(+)                   = atendime_hcare.cd_area
   and pre_med.cd_pre_med(+)             = solsai_pro.cd_pre_med
   and solsai_pro.cd_solsai_pro          = :CAMPO_CD_SOLSAI_PRO
group by solsai_pro.cd_solsai_pro
      ,solsai_pro.cd_pre_med
      ,solsai_pro.cd_estoque
      ,estoque.ds_estoque
      ,solsai_pro.cd_estoque_solicitante
      ,estoque_solicitante.ds_estoque
      ,solsai_pro.cd_atendimento
      ,solsai_pro.cd_unid_int
      ,unid_int.ds_unid_int
      ,solsai_pro.cd_setor
      ,setor.nm_setor
      ,area.cd_area
      ,area.ds_area
      ,atendime.cd_paciente
      ,paciente.nm_paciente
      ,atendime.cd_convenio
      ,convenio.nm_convenio
      ,atendime.cd_con_pla
      ,con_pla.ds_con_pla
      ,solsai_pro.cd_prestador
      ,prestador.nm_prestador
      ,solsai_pro.dt_solsai_pro
      ,solsai_pro.hr_solsai_pro
      ,solsai_pro.cd_usuario
      ,solsai_pro.sn_urgente
      ,solsai_pro.ds_obs
      ,est_pro.cd_localizacao
      ,localizacao.ds_localizacao
      ,itsolsai_pro.cd_produto
      ,produto.ds_produto
      ,uni_pro.ds_unidade
      ,est_pro.qt_estoque_atual
      ,pre_med.nm_usuario
      ,pre_med.dh_impressao
ORDER BY est_pro.cd_localizacao, localizacao.ds_localizacao
