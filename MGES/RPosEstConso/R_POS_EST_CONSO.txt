--R_POS_EST_CONSO
Select *
  From (Select CD_PRODUTO,
               DS_PRODUTO,
               CD_SUB_CLA,
               DS_SUB_CLA,
               CD_CLASSE,
               DS_CLASSE,
               CD_ESPECIE,
               DS_ESPECIE,
               QT_ESTOQUE_INICIAL / DBAMV.Verif_Vl_Fator_Prod(CD_PRODUTO) As QT_ESTOQUE_INICIAL,
               VL_ESTOQUE_INICIAL,
               QT_ESTOQUE_FINAL / DBAMV.Verif_Vl_Fator_Prod(CD_PRODUTO) As QT_ESTOQUE_FINAL,
               QT_ESTOQUE_FINAL / DBAMV.Verif_Vl_Fator_Prod(CD_PRODUTO) As QT_ESTOQUE_FINAL_SEM_TRANSF,
               VL_ESTOQUE_FINAL,
               VL_ESTOQUE_FINAL As VL_ESTOQUE_FINAL_SEM_TRANSF,
               CD_MES,
               CD_ANO,
               CD_ESTOQUE,
               DS_UNIDADE,
               ENTRADAS / DBAMV.Verif_Vl_Fator_Prod(CD_PRODUTO) As ENTRADAS,
               VL_ENTRADAS,
               SAIDAS / DBAMV.Verif_Vl_Fator_Prod(CD_PRODUTO) As SAIDAS,
               VL_SAIDAS
          From (Select C_CONEST.CD_PRODUTO,
                       PRODUTO.DS_PRODUTO,
                       Decode(:CAMPO_QUEBRA,
                              'S',
                              PRODUTO.CD_SUB_CLA,
                              Null) As CD_SUB_CLA,
                       Decode(:CAMPO_QUEBRA,
                              'S',
                              SUB_CLAS.DS_SUB_CLA,
                              Null) As DS_SUB_CLA,
                       Decode(:CAMPO_QUEBRA,
                              'E',
                              Null,
                              'P',
                              Null,
                              PRODUTO.CD_CLASSE) As CD_CLASSE,
                       Decode(:CAMPO_QUEBRA,
                              'E',
                              Null,
                              'P',
                              Null,
                              CLASSE.DS_CLASSE) As DS_CLASSE,
                       Decode(:CAMPO_QUEBRA,
                              'P',
                              Null,
                              PRODUTO.CD_ESPECIE) As CD_ESPECIE,
                       Decode(:CAMPO_QUEBRA,
                              'P',
                              Null,
                              ESPECIE.DS_ESPECIE) As DS_ESPECIE,
                       C_CONEST.CD_MES,
                       C_CONEST.CD_ANO,
                       1 As CD_ESTOQUE,
                       DBAMV.Verif_Ds_Unid_Prod(PRODUTO.CD_PRODUTO) As DS_UNIDADE,
                       Sum(Decode(PRODUTO.SN_CONSIGNADO,
                                  'S',
                                  C_CONEST.QT_ESTOQUE_INICIAL_CONSIG,
                                  'C',
                                  C_CONEST.QT_ESTOQUE_INICIAL_CONSIG + C_CONEST.QT_ESTOQUE_INICIAL,
                                  C_CONEST.QT_ESTOQUE_INICIAL)) As QT_ESTOQUE_INICIAL,
                       Sum(Case
                             When :CAMPO_SN_CONS_EST_CORP = 'S' Or :CAMPO_SN_C_M_R_CONSO = 'S' Then
                               Case
                                 When PRODUTO.SN_CONSIGNADO = 'S' Then
                                   C_CONEST.QT_ESTOQUE_INICIAL_CONSIG * DBAMV.Verif_Vl_Custo_Medio(PRODUTO.CD_PRODUTO, Last_Day(Add_Months(To_Date(LPad(To_Char(C_CONEST.CD_MES), 2, '0') || To_Char(C_CONEST.CD_ANO), 'MMYYYY'), -1)), 'H', Null, Last_Day(Add_Months(To_Date(LPad(To_Char(C_CONEST.CD_MES), 2, '0') || To_Char(C_CONEST.CD_ANO), 'MMYYYY'), -1)) + 0.99999)
                                 When PRODUTO.SN_CONSIGNADO = 'C' Then
                                   (C_CONEST.QT_ESTOQUE_INICIAL_CONSIG + C_CONEST.QT_ESTOQUE_INICIAL) * DBAMV.Verif_Vl_Custo_Medio(PRODUTO.CD_PRODUTO, Last_Day(Add_Months(To_Date(LPad(To_Char(C_CONEST.CD_MES), 2, '0') || To_Char(C_CONEST.CD_ANO), 'MMYYYY'), -1)), 'H', Null, Last_Day(Add_Months(To_Date(LPad(To_Char(C_CONEST.CD_MES), 2, '0') || To_Char(C_CONEST.CD_ANO), 'MMYYYY'), -1)) + 0.99999)
                                 Else
                                   C_CONEST.QT_ESTOQUE_INICIAL * DBAMV.Verif_Vl_Custo_Medio(PRODUTO.CD_PRODUTO, Last_Day(Add_Months(To_Date(LPad(To_Char(C_CONEST.CD_MES), 2, '0') || To_Char(C_CONEST.CD_ANO), 'MMYYYY'), -1)), 'H', Null, Last_Day(Add_Months(To_Date(LPad(To_Char(C_CONEST.CD_MES), 2, '0') || To_Char(C_CONEST.CD_ANO), 'MMYYYY'), -1)) + 0.99999)
                               End
                             Else
                               Decode(PRODUTO.SN_CONSIGNADO,
                                      'S',
                                      C_CONEST.VL_ESTOQUE_INICIAL_CONSIG,
                                      'C',
                                      C_CONEST.VL_ESTOQUE_INICIAL_CONSIG + C_CONEST.VL_ESTOQUE_INICIAL,
                                      C_CONEST.VL_ESTOQUE_INICIAL)
                           End) As VL_ESTOQUE_INICIAL,
                       Sum(Decode(PRODUTO.SN_CONSIGNADO,
                                  'S',
                                  C_CONEST.QT_ESTOQUE_FINAL_CONSIG,
                                  'C',
                                  C_CONEST.QT_ESTOQUE_FINAL_CONSIG + C_CONEST.QT_ESTOQUE_FINAL,
                                  C_CONEST.QT_ESTOQUE_FINAL)) As QT_ESTOQUE_FINAL,
                       Sum(Case
                             When :CAMPO_SN_CONS_EST_CORP = 'S' Or :CAMPO_SN_C_M_R_CONSO = 'S' Then
                               Case
                                 When PRODUTO.SN_CONSIGNADO = 'S' Then
                                   C_CONEST.QT_ESTOQUE_FINAL_CONSIG * DBAMV.Verif_Vl_Custo_Medio(PRODUTO.CD_PRODUTO, Last_Day(To_Date(LPad(To_Char(C_CONEST.CD_MES), 2, '0') || To_Char(C_CONEST.CD_ANO), 'MMYYYY')), 'H', Null, Last_Day(To_Date(LPad(To_Char(C_CONEST.CD_MES), 2, '0') || To_Char(C_CONEST.CD_ANO), 'MMYYYY')) + 0.99999)
                                 When PRODUTO.SN_CONSIGNADO = 'C' Then
                                   (C_CONEST.QT_ESTOQUE_FINAL_CONSIG + C_CONEST.QT_ESTOQUE_FINAL) * DBAMV.Verif_Vl_Custo_Medio(PRODUTO.CD_PRODUTO, Last_Day(To_Date(LPad(To_Char(C_CONEST.CD_MES), 2, '0') || To_Char(C_CONEST.CD_ANO), 'MMYYYY')), 'H', Null, Last_Day(To_Date(LPad(To_Char(C_CONEST.CD_MES), 2, '0') || To_Char(C_CONEST.CD_ANO), 'MMYYYY')) + 0.99999)
                                 Else
                                   C_CONEST.QT_ESTOQUE_FINAL * DBAMV.Verif_Vl_Custo_Medio(PRODUTO.CD_PRODUTO, Last_Day(To_Date(LPad(To_Char(C_CONEST.CD_MES), 2, '0') || To_Char(C_CONEST.CD_ANO), 'MMYYYY')), 'H', Null, Last_Day(To_Date(LPad(To_Char(C_CONEST.CD_MES), 2, '0') || To_Char(C_CONEST.CD_ANO), 'MMYYYY')) + 0.99999)
                               End
                             Else
                               Decode(PRODUTO.SN_CONSIGNADO,
                                      'S',
                                      C_CONEST.VL_ESTOQUE_FINAL_CONSIG,
                                      'C',
                                      C_CONEST.VL_ESTOQUE_FINAL_CONSIG + C_CONEST.VL_ESTOQUE_FINAL,
                                      C_CONEST.VL_ESTOQUE_FINAL)
                           End) As VL_ESTOQUE_FINAL,
                       Sum(Nvl(C_CONEST.QT_ENTRADA,
                               0) + Decode(PRODUTO.SN_CONSIGNADO,
                                           'S',
                                           0,
                                           Nvl(C_CONEST.QT_ENTRADA_TRANSFERIDO,
                                               0)) + Decode(PRODUTO.SN_CONSIGNADO,
                                                            'S',
                                                            0,
                                                            Nvl(C_CONEST.QT_DEVOLUCAO_VENDA,
                                                                0)) + Decode(PRODUTO.SN_CONSIGNADO,
                                                                             'S',
                                                                             0,
                                                                             Nvl(C_CONEST.QT_ENTRADA_EMPRESTIMO_RECEBIDO,
                                                                                 0) + Nvl(C_CONEST.QT_ENTRADA_PAG_EMPRESTIMO,
                                                                                          0)) + Nvl(C_CONEST.QT_DEVOLUCAO_SETOR,
                                                                                                    0) + Nvl(C_CONEST.QT_DEVOLUCAO_PACIENTE,
                                                                                                             0) + Nvl(C_CONEST.QT_DOACAO,
                                                                                                                      0) + Nvl(C_CONEST.QT_NOTA_FISCAL_NAO_COBRADA,
                                                                                                                               0) + Decode(PRODUTO.SN_CONSIGNADO,
                                                                                                                                           'S',
                                                                                                                                           0,
                                                                                                                                           Nvl(C_CONEST.QT_MANIPULADA_ENTRADA,
                                                                                                                                               0)) + Decode(PRODUTO.SN_CONSIGNADO,
                                                                                                                                                            'S',
                                                                                                                                                            0,
                                                                                                                                                            Nvl(C_CONEST.QT_CONTAGEM_ENTRADA,
                                                                                                                                                                0))) As ENTRADAS,
                       Sum(Nvl(C_CONEST.VL_ENTRADA,
                               0) + Decode(PRODUTO.SN_CONSIGNADO,
                                           'S',
                                           0,
                                           Nvl(C_CONEST.VL_ENTRADA_TRANSFERIDO,
                                               0)) + Decode(PRODUTO.SN_CONSIGNADO,
                                                            'S',
                                                            0,
                                                            Nvl(C_CONEST.VL_DEVOLUCAO_VENDA,
                                                                0)) + Decode(PRODUTO.SN_CONSIGNADO,
                                                                             'S',
                                                                             0,
                                                                             Nvl(C_CONEST.VL_ENTRADA_EMPRESTIMO_RECEBIDO,
                                                                                 0) + Nvl(C_CONEST.VL_ENTRADA_PAG_EMPRESTIMO,
                                                                                          0)) + Nvl(C_CONEST.VL_DEVOLUCAO_SETOR,
                                                                                                    0) + Nvl(C_CONEST.VL_DEVOLUCAO_PACIENTE,
                                                                                                             0) + Nvl(C_CONEST.VL_DOACAO,
                                                                                                                      0) + Nvl(C_CONEST.VL_NOTA_FISCAL_NAO_COBRADA,
                                                                                                                               0) + Decode(PRODUTO.SN_CONSIGNADO,
                                                                                                                                           'S',
                                                                                                                                           0,
                                                                                                                                           Nvl(C_CONEST.VL_MANIPULADA_ENTRADA,
                                                                                                                                               0)) + Decode(PRODUTO.SN_CONSIGNADO,
                                                                                                                                                            'S',
                                                                                                                                                            0,
                                                                                                                                                            Nvl(C_CONEST.VL_CONTAGEM_ENTRADA,
                                                                                                                                                                0)) + Nvl(C_CONEST.VL_ENTRADA_AJUSTE_CONTABIL,
                                                                                                                                                                          0)) As VL_ENTRADAS,
                       Sum(Nvl(C_CONEST.QT_SAIDA_SETOR,
                               0) + Decode(PRODUTO.SN_CONSIGNADO,
                                           'S',
                                           0,
                                           Nvl(C_CONEST.QT_VENDA,
                                               0)) + Decode(PRODUTO.SN_CONSIGNADO,
                                                            'S',
                                                            0,
                                                            Nvl(C_CONEST.QT_BAIXA,
                                                                0)) + Decode(PRODUTO.SN_CONSIGNADO,
                                                                             'S',
                                                                             0,
                                                                             Nvl(C_CONEST.QT_SAIDA_EMPRESTIMO_CONCEDIDO,
                                                                                 0) + Nvl(C_CONEST.QT_SAIDA_PAG_EMPRESTIMO,
                                                                                          0)) + Nvl(C_CONEST.QT_SAIDA_PACIENTE,
                                                                                                    0) + Nvl(C_CONEST.QT_DEVOLVIDA_FORNEC,
                                                                                                             0) + Decode(PRODUTO.SN_CONSIGNADO,
                                                                                                                         'S',
                                                                                                                         0,
                                                                                                                         Nvl(C_CONEST.QT_MANIPULADA_SAIDA,
                                                                                                                             0)) + Decode(PRODUTO.SN_CONSIGNADO,
                                                                                                                                          'S',
                                                                                                                                          0,
                                                                                                                                          Nvl(C_CONEST.QT_TRANSFERENCIA_SAIDA,
                                                                                                                                              0)) + Decode(PRODUTO.SN_CONSIGNADO,
                                                                                                                                                           'S',
                                                                                                                                                           0,
                                                                                                                                                           Nvl(C_CONEST.QT_CONTAGEM_SAIDA,
                                                                                                                                                               0))) As SAIDAS,
                       Sum(Nvl(C_CONEST.VL_SAIDA_SETOR,
                               0) + Decode(PRODUTO.SN_CONSIGNADO,
                                           'S',
                                           0,
                                           Nvl(C_CONEST.VL_VENDA,
                                               0)) + Decode(PRODUTO.SN_CONSIGNADO,
                                                            'S',
                                                            0,
                                                            Nvl(C_CONEST.VL_BAIXA,
                                                                0)) + Decode(PRODUTO.SN_CONSIGNADO,
                                                                             'S',
                                                                             0,
                                                                             Nvl(C_CONEST.VL_SAIDA_EMPRESTIMO_CONCEDIDO,
                                                                                 0) + Nvl(C_CONEST.VL_SAIDA_PAG_EMPRESTIMO,
                                                                                          0)) + Nvl(C_CONEST.VL_SAIDA_PACIENTE,
                                                                                                    0) + Nvl(C_CONEST.VL_DEVOLVIDA_FORNEC,
                                                                                                             0) + Decode(PRODUTO.SN_CONSIGNADO,
                                                                                                                         'S',
                                                                                                                         0,
                                                                                                                         Nvl(C_CONEST.VL_MANIPULADA_SAIDA,
                                                                                                                             0)) + Decode(PRODUTO.SN_CONSIGNADO,
                                                                                                                                          'S',
                                                                                                                                          0,
                                                                                                                                          Nvl(C_CONEST.VL_TRANSFERENCIA_SAIDA,
                                                                                                                                              0)) + Decode(PRODUTO.SN_CONSIGNADO,
                                                                                                                                                           'S',
                                                                                                                                                           0,
                                                                                                                                                           Nvl(C_CONEST.VL_CONTAGEM_SAIDA,
                                                                                                                                                               0)) + Nvl(C_CONEST.VL_SAIDA_AJUSTE_CONTABIL,
                                                                                                                                                                         0)) As VL_SAIDAS
                  From DBAMV.C_CONEST
            Inner Join DBAMV.PRODUTO On
                       PRODUTO.CD_PRODUTO = C_CONEST.CD_PRODUTO
                   And PRODUTO.SN_MESTRE = 'N'
            Inner Join DBAMV.ESTOQUE On
                       ESTOQUE.CD_ESTOQUE = C_CONEST.CD_ESTOQUE
            Inner Join DBAMV.ESPECIE On
                       ESPECIE.CD_ESPECIE = PRODUTO.CD_ESPECIE
            Inner Join DBAMV.CLASSE On
                       CLASSE.CD_ESPECIE = PRODUTO.CD_ESPECIE
                   And CLASSE.CD_CLASSE = PRODUTO.CD_CLASSE
            Inner Join DBAMV.SUB_CLAS On
                       SUB_CLAS.CD_ESPECIE = PRODUTO.CD_ESPECIE
                   And SUB_CLAS.CD_CLASSE = PRODUTO.CD_CLASSE
                   And SUB_CLAS.CD_SUB_CLA = PRODUTO.CD_SUB_CLA
                 Where C_CONEST.CD_MES = To_Number(:CAMPO_DATA_INI)
                   And C_CONEST.CD_ANO = To_Number(:CAMPO_DATA_FIM)
              Group By C_CONEST.CD_PRODUTO,
                       PRODUTO.DS_PRODUTO,
                       PRODUTO.CD_SUB_CLA,
                       SUB_CLAS.DS_SUB_CLA,
                       PRODUTO.CD_CLASSE,
                       CLASSE.DS_CLASSE,
                       PRODUTO.CD_ESPECIE,
                       ESPECIE.DS_ESPECIE,
                       C_CONEST.CD_MES,
                       C_CONEST.CD_ANO,
                       DBAMV.Verif_Ds_Unid_Prod(PRODUTO.CD_PRODUTO)) TAB
      Order By 2,
               1)
Order By 7,
       5,
       3,
       2,
       1
