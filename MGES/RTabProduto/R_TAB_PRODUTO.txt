--R_TAB_PRODUTO
SELECT * FROM (
Select  especie.cd_especie         cd_especie
     ,  especie.ds_especie         ds_especie
     ,  classe.cd_classe           cd_classe
     ,  classe.ds_classe           ds_classe
     ,  sub_clas.cd_sub_cla        cd_sub_cla
     ,  sub_clas.ds_sub_cla        ds_sub_cla
     ,  produto.cd_produto         cd_produto
     ,  produto.ds_produto         ds_produto
     ,  produto.dt_cadastro        dt_cadastro
     ,  produto.sn_mestre          sn_mestre
     ,  uni_pro.cd_unidade         cd_unidade
     ,  uni_pro.ds_unidade         ds_unidade
  From  dbamv.especie              especie
     ,  dbamv.classe               classe
     ,  dbamv.sub_clas             sub_clas
     ,  dbamv.produto              produto
     ,  dbamv.uni_pro              uni_pro
 Where  especie.cd_especie      like :CAMPO_ESPECIE
   and  classe.cd_classe             like :CAMPO_CLASSE
   and  sub_clas.cd_sub_cla       like :CAMPO_SUB_CLASSE
   and  classe.cd_especie        = especie.cd_especie
   and  sub_clas.cd_especie      = especie.cd_especie
   and  sub_clas.cd_classe       = classe.cd_classe
   and  produto.cd_especie       = sub_clas.cd_especie
   and  produto.cd_classe        = sub_clas.cd_classe
   and  produto.cd_sub_cla       = sub_clas.cd_sub_cla
   and  produto.cd_produto       = uni_pro.cd_produto
   And Exists ( Select Empresa_Produto.Cd_Produto
                     From Dbamv.Empresa_Produto
                   Where Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
                        and Empresa_Produto.Cd_Produto = Produto.Cd_Produto )
)
ORDER BY DS_ESPECIE,CD_ESPECIE,
DS_CLASSE,CD_CLASSE,
DS_SUB_CLA,CD_SUB_CLA,
DS_PRODUTO,CD_PRODUTO,DT_CADASTRO,SN_MESTRE
