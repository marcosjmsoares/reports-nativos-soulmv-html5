SELECT  cd_mvto_estoque,
        tp_mvto_estoque,
        cd_estoque,
        ds_estoque,
        dt_mvto_estoque,
        hr_mvto_estoque,
        cd_atendimento,
        nm_paciente,
        cd_setor,
        nm_setor,
        qt_produzida,
        cd_estoque_destino,
        ds_estoque_destino,
        cd_solsai_pro,
        cd_aviso_cirurgia,
        cd_multi_empresa,
        ds_multi_empresa,
        cd_produto,
        cd_ium,
        ds_produto,
        tp_ativo,
        Sum(qt_movimentacao) qt_movimentacao,
        cd_lote,
        dt_validade,
        dh_mvto_estoque,
        cd_fornecedor,
        nr_cgc_cpf,
        nm_fornecedor,
        nm_fornecedor_dest,
        cd_mot_bai,
        ds_mot_bai,
	data_hora_mvto  data_hora_mvto
  FROM (SELECT  a.cd_mvto_estoque       cd_mvto_estoque,
                tp_mvto_estoque         tp_mvto_estoque,
                a.cd_estoque            cd_estoque,
                e.ds_estoque            ds_estoque,
                dt_mvto_estoque         dt_mvto_estoque,
                hr_mvto_estoque         hr_mvto_estoque,
                a.cd_atendimento        cd_atendimento,
                i.nm_paciente           nm_paciente,
                a.cd_setor              cd_setor,
                j.nm_setor              nm_setor,
                qt_produzida            qt_produzida,
                Nvl(a.cd_estoque_destino, a.cd_estoque) cd_estoque_destino,
                f.ds_estoque            ds_estoque_destino,
                cd_solsai_pro           cd_solsai_pro,
                cd_aviso_cirurgia       cd_aviso_cirurgia,
                a.cd_multi_empresa      cd_multi_empresa,
                g.ds_multi_empresa      ds_multi_empresa,
                cd_itmvto_estoque       cd_itmvto_estoque,
                b.cd_produto            cd_produto,
                lp.cd_ium               cd_ium,
                d.ds_produto            ds_produto,
                Decode(d.tp_ativo,'S','Ativo','Inativo')       tp_ativo,
                (b.qt_movimentacao*u.vl_fator) / dbamv.verif_vl_fator_prod(b.cd_produto)  qt_movimentacao,
                b.cd_lote                 cd_lote,
                to_char(b.dt_validade, 'DD/MM/YYYY')           dt_validade,
                to_char(dh_mvto_estoque, 'DD/MM/YYYY hh24') dh_mvto_estoque,
                b.cd_fornecedor           cd_fornecedor,
                vl_unitario               vl_unitario,
                c.nr_cgc_cpf              nr_cgc_cpf,
                c.nm_fornecedor           nm_fornecedor,
                l.nm_fornecedor           nm_fornecedor_dest,
                a.cd_mot_bai              cd_mot_bai,
                m.ds_mot_bai              ds_mot_bai,
		To_Char(DT_MVTO_ESTOQUE,'DD/MM/YYYY') || '  ' ||  To_Char (HR_MVTO_ESTOQUE,'hh24')  DATA_HORA_MVTO
          FROM dbamv.mvto_estoque a
             , dbamv.itmvto_estoque b
             , dbamv.fornecedor c
             , dbamv.produto d
             , dbamv.estoque e
             , dbamv.estoque f
             , dbamv.multi_empresas g
             , dbamv.atendime h
             , dbamv.paciente i
             , dbamv.setor j
             , dbamv.fornecedor l
             , dbamv.mot_bai m
             , dbamv.uni_pro u
             , dbamv.lot_pro lp
         Where a.cd_mvto_estoque            = b.cd_mvto_estoque
           and b.cd_uni_pro                 = u.cd_uni_pro
           and b.cd_fornecedor              = c.cd_fornecedor(+)
           and b.cd_produto                 = d.cd_produto
           and a.cd_estoque                 = e.cd_estoque
           and Nvl(lp.cd_lote, '*.*') 		= Nvl(b.cd_lote,'*.*')            
		   and Nvl(lp.dt_validade, SYSDATE) = Nvl(b.dt_validade, SYSDATE)
           and lp.cd_produto                = d.cd_produto
           and lp.cd_estoque                = a.cd_estoque
           and a.cd_estoque_destino         = f.cd_estoque (+)
           and a.cd_multi_empresa           = g.cd_multi_empresa (+)
           and a.cd_atendimento             = h.cd_atendimento (+)
           and h.cd_paciente                = i.cd_paciente (+)
           and a.cd_setor                   = j.cd_setor (+)
           and a.cd_fornecedor              = l.cd_fornecedor (+)
           and a.cd_mot_bai                 = m.cd_mot_bai (+)
           :CAMPO_CF_LOTE
) GROUP BY  cd_mvto_estoque,
            tp_mvto_estoque,
            cd_estoque,
            ds_estoque,
            dt_mvto_estoque,
            hr_mvto_estoque,
            cd_atendimento,
            nm_paciente,
            cd_setor,
            nm_setor,
            qt_produzida,
            cd_estoque_destino,
            ds_estoque_destino,
            cd_solsai_pro,
            cd_aviso_cirurgia,
            cd_multi_empresa,
            ds_multi_empresa,
            cd_produto,
            cd_ium,
            ds_produto,
            tp_ativo,
            cd_lote,
            dt_validade,
            dh_mvto_estoque,
            cd_fornecedor,
            nr_cgc_cpf,
            nm_fornecedor,
            nm_fornecedor_dest,
            cd_mot_bai,
            ds_mot_bai,
	    data_hora_mvto
ORDER BY CD_PRODUTO
       , CD_ESTOQUE
       , dt_mvto_estoque
       , dh_mvto_estoque
