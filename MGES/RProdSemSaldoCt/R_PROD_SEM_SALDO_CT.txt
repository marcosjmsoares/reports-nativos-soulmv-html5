Select Cota_Setor.Cd_Setor cd_setor
          ,Cota_Setor.Cd_Produto cd_produto
          ,Cota_Setor.Cd_Uni_Pro cd_uni_pro
          ,Cota_Setor.Qt_Cota  qt_cota
         , nvl(est_pro.qt_estoque_atual,0) qt_estoque_atual
          ,Cota_Setor.Nr_Dias nr_dias
          ,cota_setor.cd_estoque cd_Estoque_destino
          ,cota_setor.sn_bloqueia sn_bloqueia
          ,dt_vigencia dt_vigencia
          ,est_pro.cd_estoque cd_estoque_origem
          ,est_pro.cd_localizacao
          ,produto.ds_produto ds_produto
  From Dbamv.COTA_SETOR
          ,dbamv.est_pro
          ,dbamv.produto
  where est_pro.cd_produto = cota_setor.cd_produto
    and produto.cd_produto = cota_setor.cd_produto
    and  nvl(est_pro.qt_estoque_atual,0)  < nvl(Cota_Setor.Qt_Cota,0)
    and cota_setor.dt_vigencia <= :CAMPO_DT_VIGENCIA
    and est_pro.cd_estoque = cota_setor.cd_estoque
order by cd_estoque_origem, cd_produto
