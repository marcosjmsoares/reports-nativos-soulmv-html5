--R_PROD_FOR
SELECT PRODUTO.CD_PRODUTO CODIGO
               ,PRODUTO.DS_PRODUTO        DESCRICAO
               ,VERIF_DS_UNID_PROD(PRODUTO.CD_PRODUTO) UNIDADE
               ,FORNECEDOR.CD_FORNECEDOR
               ,FORNECEDOR.NM_FORNECEDOR
               ,EMPRESA_PRODUTO.DT_ULTIMA_ENTRADA DT_ULTIMA_ENTRADA
               ,EMPRESA_PRODUTO.VL_ULTIMA_ENTRADA *  VERIF_VL_FATOR_PROD( PRODUTO.CD_PRODUTO ) VL_ULTIMA_ENTRADA
               ,verif_vl_custo_medio ( produto.cd_produto,
                                    SYSDATE,
                                    'H',
                                    produto.vl_custo_medio,
                                    SYSDATE ) VL_CUSTO_MEDIO
FROM DBAMV.PRODUTO
           ,DBAMV.FORNECEDOR
           ,DBAMV.EMPRESA_PRODUTO
WHERE FORNECEDOR.TP_CLIENTE_FORN IN ( 'F', 'A', 'R', 'T' )
      AND PRODUTO.CD_PRODUTO = EMPRESA_PRODUTO.CD_PRODUTO
      AND EMPRESA_PRODUTO.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
      AND PRODUTO.CD_ULTIMO_FORNECEDOR = FORNECEDOR.CD_FORNECEDOR
      AND DBAMV.MGES.F_CONFIGURACAO_ESTOQUE(:CAMPO_ESTOQUE
                                              ,PRODUTO.CD_ESPECIE
                                              ,PRODUTO.CD_CLASSE
                                              ,PRODUTO.CD_SUB_CLA
                                              ,PRODUTO.CD_PRODUTO) > 0
      AND PRODUTO.SN_MESTRE =  'N'
