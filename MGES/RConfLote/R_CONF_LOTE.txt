--R_CONF_LOTE
SELECT  DISTINCT
        PRODUTO.CD_ESPECIE CD_ESPECIE,
        PRODUTO.CD_CLASSE CD_CLASSE,
        PRODUTO.CD_SUB_CLA CD_SUB_CLA,
        PRODUTO.CD_PRODUTO CD_PRODUTO,
        PRODUTO.DS_PRODUTO DS_PRODUTO,
        Sum(LOT_PRO.QT_ESTOQUE_ATUAL) QT_ESTOQUE_ATUAL
FROM    DBAMV.PRODUTO,
        DBAMV.EST_PRO,
        DBAMV.ESTOQUE,
        DBAMV.LOT_PRO
WHERE   EST_PRO.CD_PRODUTO = PRODUTO.CD_PRODUTO
AND     EST_PRO.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
AND     LOT_PRO.CD_PRODUTO = PRODUTO.CD_PRODUTO
AND     LOT_PRO.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
AND     ESTOQUE.CD_MULTI_EMPRESA = Dbamv.Pkg_mv2000.le_empresa
GROUP BY
        PRODUTO.CD_ESPECIE,
        PRODUTO.CD_CLASSE,
        PRODUTO.CD_SUB_CLA,
        PRODUTO.CD_PRODUTO,
        PRODUTO.DS_PRODUTO
