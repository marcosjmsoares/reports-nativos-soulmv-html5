--R_CONS_ESTOQUE
SELECT Produto.CD_PRODUTO CD_PRODUTO
      ,Produto.SN_MESTRE SN_MESTRE
      ,Produto_filho.CD_PRODUTO_TEM CD_PRODUTO_TEM
      ,Initcap ( Produto.Ds_Produto ) Ds_Produto
      ,Initcap ( Verif_Ds_Unid_Prod(Produto.cd_produto)) Ds_unidade
      ,Produto_filho.DATA DATA
      ,((SUM(Produto_filho.QT_MVTO)) / Verif_vl_fator_prod(Produto.cd_produto)) QT_MVTO
      ,SUM(Produto_filho.VL_MVTO) VL_MVTO
      , Dbamv.Verif_vl_fator_prod(Produto.cd_produto) fatorNovo
FROM ( Select Produto.Cd_Produto Cd_Produto
             ,produto.sn_mestre SN_MESTRE
             ,PRODUTO.CD_PRODUTO_TEM CD_PRODUTO_TEM
             ,Initcap ( Produto.Ds_Produto ) Ds_Produto
             ,Initcap ( Verif_Ds_Unid_Prod( Produto.Cd_Produto)) Ds_unidade
             ,To_Date(To_Char(C_Conest.Cd_Mes) || '/' || To_Char(C_Conest.Cd_Ano), 'mm/yyyy') Data
             ,(Sum((decode( '', 'S', Nvl(C_Conest.qt_saida_setor, 0), 0) +
                    decode( '', 'S', Nvl(C_Conest.qt_saida_paciente, 0), 0) +
                    decode( '', 'S', Nvl(C_Conest.qt_venda, 0), 0) +
                    decode( '', 'S', NVL( DECODE(ESTOQUE.SN_CONSIDERA_TRANSF_CONSUMO, 'N', 0, DECODE(ESTOQUE.TP_ESTOQUE, 'D', c_ConEst.qt_transferencia_saida,0)) , 0 ), 0)) -
                   (DECODE( '', 'S', Nvl( C_Conest.qt_devolucao_setor, 0), 0) +
                    decode( '', 'S', Nvl(C_Conest.qt_devolucao_paciente, 0), 0) +
                    decode( '', 'S', Nvl(C_Conest.qt_devolucao_venda, 0), 0) +
                    decode( '' , 'S', NVL( DECODE(ESTOQUE.SN_CONSIDERA_TRANSF_CONSUMO, 'N', 0, DECODE(ESTOQUE.TP_ESTOQUE, 'D', c_ConEst.qt_entrada_transferido,0)), 0 ), 0)))) qt_mvto
             ,Sum((decode( '', 'S', Nvl(C_Conest.vl_saida_setor, 0), 0) +
                   decode( '', 'S', Nvl(C_Conest.vl_saida_paciente, 0), 0) +
                   decode( '', 'S', Nvl(C_Conest.qt_venda, 0), 0) +
                   decode( '', 'S', NVL( DECODE(ESTOQUE.SN_CONSIDERA_TRANSF_CONSUMO, 'N', 0, DECODE(ESTOQUE.TP_ESTOQUE, 'D', C_Conest.vl_transferencia_saida,0)) , 0 ), 0)) -
                  (decode( '', 'S', Nvl(C_Conest.vl_devolucao_setor, 0), 0) +
                   decode( '', 'S', Nvl(C_Conest.vl_devolucao_paciente, 0), 0) +
                   decode( '', 'S', Nvl(C_Conest.qt_devolucao_venda, 0), 0) +
                   decode( '', 'S', NVL( DECODE(ESTOQUE.SN_CONSIDERA_TRANSF_CONSUMO, 'N', 0, decode(ESTOQUE.TP_ESTOQUE, 'D',C_Conest.vl_entrada_transferido, 0)), 0 ),0))) vl_mvto
       From   Dbamv.Estoque
             ,Dbamv.Produto
             ,Dbamv.C_Conest
       Where Estoque.Cd_multi_empresa =  :CAMPO_CD_MULTI_EMPRESA
         And To_Date(To_Char(C_Conest.Cd_Mes) || '/' ||To_Char(C_Conest.Cd_Ano), 'mm/yyyy')between :CAMPO_DT_INIC
         And 
         And C_conest.cd_estoque		= Estoque.Cd_estoque
         And C_CONEST.CD_PRODUTO = PRODUTO.CD_PRODUTO
         And PRODUTO.SN_MESTRE = 'N'
        And     (decode( '', 'S', Nvl(C_Conest.qt_saida_setor, 0), 0) +
                 decode( '', 'S', Nvl(C_Conest.qt_saida_paciente, 0), 0) +
                 decode( '', 'S', Nvl(C_Conest.qt_venda, 0), 0) +
                 decode( '', 'S', NVL( DECODE(ESTOQUE.SN_CONSIDERA_TRANSF_CONSUMO, 'N', 0, DECODE(ESTOQUE.TP_ESTOQUE, 'D', c_ConEst.qt_transferencia_saida,0)) , 0 ), 0)) -
                (DECODE( '', 'S', Nvl( C_Conest.qt_devolucao_setor, 0), 0) +
                 decode( '', 'S', Nvl(C_Conest.qt_devolucao_paciente, 0), 0) +
                 decode( '', 'S', Nvl(C_Conest.qt_devolucao_venda, 0), 0) +
                 decode( '' , 'S', NVL( DECODE(ESTOQUE.SN_CONSIDERA_TRANSF_CONSUMO, 'N', 0, DECODE(ESTOQUE.TP_ESTOQUE, 'D', c_ConEst.qt_entrada_transferido,0)), 0 ), 0)) <> 0
       Group By Produto.Cd_Produto,
                PRODUTO.CD_PRODUTO_TEM,
                PRODUTO.SN_MESTRE,
                Produto.Ds_Produto,
                C_Conest.Cd_Mes,
                C_Conest.Cd_Ano) Produto_filho,
     DBAMV.PRODUTO
GROUP BY Produto.CD_PRODUTO
      ,Produto.SN_MESTRE
      ,Produto_filho.CD_PRODUTO_TEM
      ,Initcap ( Produto.Ds_Produto )
      ,Initcap ( Verif_Ds_Unid_Prod( Produto.Cd_Produto ))
      ,Produto_filho.DATA
UNION ALL
Select        Produto.Cd_Produto Cd_Produto
             ,produto.sn_mestre SN_MESTRE
             ,PRODUTO.CD_PRODUTO_TEM CD_PRODUTO_TEM
             ,Initcap ( Produto.Ds_Produto ) Ds_Produto
             ,Initcap ( Verif_Ds_Unid_Prod( Produto.Cd_Produto)) Ds_unidade
             ,To_Date(To_Char(C_Conest.Cd_Mes) || '/' ||
                      To_Char(C_Conest.Cd_Ano), 'mm/yyyy') Data
             ,(Sum((decode( '', 'S', Nvl(C_Conest.qt_saida_setor, 0), 0) +
                    decode(  '', 'S', Nvl(C_Conest.qt_saida_paciente, 0), 0) +
                    decode(  '', 'S', Nvl(C_Conest.qt_venda, 0), 0) +
                    decode( '', 'S', NVL( DECODE(ESTOQUE.SN_CONSIDERA_TRANSF_CONSUMO, 'N', 0, DECODE(ESTOQUE.TP_ESTOQUE, 'D', c_ConEst.qt_transferencia_saida,0)) , 0 ), 0)) -
                   (DECODE(  '', 'S', Nvl( C_Conest.qt_devolucao_setor, 0), 0) +
                    decode(  '', 'S', Nvl(C_Conest.qt_devolucao_paciente, 0), 0) +
                    decode( '', 'S', Nvl(C_Conest.qt_devolucao_venda, 0), 0) +
                    decode('' , 'S', NVL( DECODE(ESTOQUE.SN_CONSIDERA_TRANSF_CONSUMO, 'N', 0, DECODE(ESTOQUE.TP_ESTOQUE, 'D', c_ConEst.qt_entrada_transferido,0)), 0 ), 0)))) /
                    Verif_vl_fator_prod( produto.cd_produto ) qt_mvto
             ,Sum((decode('', 'S', Nvl(C_Conest.vl_saida_setor, 0), 0) +
                   decode(  '', 'S', Nvl(C_Conest.vl_saida_paciente, 0), 0) +
                   decode(  '', 'S', Nvl(C_Conest.qt_venda, 0), 0) +
                   decode( '', 'S', NVL( DECODE(ESTOQUE.SN_CONSIDERA_TRANSF_CONSUMO, 'N', 0, DECODE(ESTOQUE.TP_ESTOQUE, 'D', C_Conest.vl_transferencia_saida,0)) , 0 ), 0)) -
                  (decode(  '', 'S', Nvl(C_Conest.vl_devolucao_setor, 0), 0) +
                   decode(  '', 'S', Nvl(C_Conest.vl_devolucao_paciente, 0), 0) +
                   decode( '', 'S', Nvl(C_Conest.qt_devolucao_venda, 0), 0) +
                   decode( '', 'S', NVL( DECODE(ESTOQUE.SN_CONSIDERA_TRANSF_CONSUMO, 'N', 0, decode(ESTOQUE.TP_ESTOQUE, 'D',C_Conest.vl_entrada_transferido, 0)), 0 ),0))) vl_mvto
             , Dbamv.Verif_vl_fator_prod(Produto.cd_produto) fatorNovo
From       Dbamv.Estoque
          ,Dbamv.Produto
          ,Dbamv.C_Conest
Where    Estoque.Cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
And      To_Date(To_Char(C_Conest.Cd_Mes) || '/' || To_Char(C_Conest.Cd_Ano), 'mm/yyyy') between :CAMPO_DT_INIC
and      
And      C_conest.cd_estoque		= Estoque.Cd_estoque
AND      C_CONEST.CD_PRODUTO = PRODUTO.CD_PRODUTO
AND      PRODUTO.SN_MESTRE = 'N'
AND      PRODUTO.CD_PRODUTO_TEM IS NULL
        And     (decode( '', 'S', Nvl(C_Conest.qt_saida_setor, 0), 0) +
				 decode( '', 'S', Nvl(C_Conest.qt_saida_paciente, 0), 0) +
				 decode(  '', 'S', Nvl(C_Conest.qt_venda, 0), 0) +
				 decode( '', 'S', NVL( DECODE(ESTOQUE.SN_CONSIDERA_TRANSF_CONSUMO, 'N', 0, DECODE(ESTOQUE.TP_ESTOQUE, 'D', c_ConEst.qt_transferencia_saida,0)) , 0 ), 0)) -
			    (DECODE(  '', 'S', Nvl( C_Conest.qt_devolucao_setor, 0), 0) +
				 decode(  '', 'S', Nvl(C_Conest.qt_devolucao_paciente, 0), 0) +
				 decode( '', 'S', Nvl(C_Conest.qt_devolucao_venda, 0), 0) +
				 decode('' , 'S', NVL( DECODE(ESTOQUE.SN_CONSIDERA_TRANSF_CONSUMO, 'N', 0, DECODE(ESTOQUE.TP_ESTOQUE, 'D', c_ConEst.qt_entrada_transferido,0)), 0 ), 0)) <> 0
Group By        Produto.Cd_Produto,
                PRODUTO.CD_PRODUTO_TEM,
                PRODUTO.SN_MESTRE,
                Produto.Ds_Produto,
                C_Conest.Cd_Mes,
                C_Conest.Cd_Ano
order by DATA,ds_produto
