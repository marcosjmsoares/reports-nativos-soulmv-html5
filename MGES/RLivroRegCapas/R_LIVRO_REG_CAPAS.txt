--R_LIVRO_REG_CAPAS
SELECT * FROM (
select  t.CD_TERMO_PORTARIA 				CD_TERMO_PORTARIA
 ,t.NR_PAG_INICIAL					NR_PAG_INICIAL
 ,t.NR_PAG_FINAL						NR_PAG_FINAL
 ,t.QT_PAG_EXTENSO					QT_PAG_EXTENSO
 ,t.DH_IMPRESSAO					DH_IMPRESSAO
 ,t.DT_COMPETENCIA					DT_COMPETENCIA
 ,t.CD_PORTARIA						CD_PORTARIA
 ,i.cd_lista_codigo_medicamento				cd_lista_codigo_medicamento
 ,l.ds_lista_codigo_medicamento				ds_lista_codigo_medicamento
 ,'TERMO DE ENCERRAMENTO' 				titulo
 ,nr_livro							nr_livro
 ,p.ds_portaria                                                                                          ds_portaria
from  	dbamv.termo_portaria t,
	dbamv.it_termo_portaria i,
 	dbamv.lista_codigo_medicamento l,
                dbamv.portaria p
where  t.cd_termo_portaria = :CAMPO_CD_TERMO_PORTARIA and
t.cd_termo_portaria = i.cd_termo_portaria
and l.cd_lista_codigo_medicamento = i.cd_lista_codigo_medicamento
and t.cd_portaria = p.cd_portaria
UNION
select  t.CD_TERMO_PORTARIA 				CD_TERMO_PORTARIA
 ,t.NR_PAG_INICIAL					NR_PAG_INICIAL
 ,t.NR_PAG_FINAL						NR_PAG_FINAL
 ,t.QT_PAG_EXTENSO					QT_PAG_EXTENSO
 ,t.DH_IMPRESSAO					DH_IMPRESSAO
 ,t.DT_COMPETENCIA					DT_COMPETENCIA
 ,t.CD_PORTARIA						CD_PORTARIA
 ,i.cd_lista_codigo_medicamento				cd_lista_codigo_medicamento
 ,l.ds_lista_codigo_medicamento				ds_lista_codigo_medicamento
 ,'TERMO DE ABERTURA' 					titulo
 ,nr_livro							nr_livro
 ,p.ds_portaria                                                                                          ds_portaria
from  	dbamv.termo_portaria t,
	dbamv.it_termo_portaria i,
 	dbamv.lista_codigo_medicamento l,
                dbamv.portaria p
where  t.cd_termo_portaria = :CAMPO_CD_TERMO_PORTARIA and
t.cd_termo_portaria = i.cd_termo_portaria
and l.cd_lista_codigo_medicamento = i.cd_lista_codigo_medicamento
and t.cd_portaria = p.cd_portaria
)
ORDER BY CD_TERMO_PORTARIA,NR_PAG_INICIAL,NR_PAG_FINAL,QT_PAG_EXTENSO,DH_IMPRESSAO,DT_COMPETENCIA,CD_PORTARIA,TITULO,DS_PORTARIA,NR_LIVRO
