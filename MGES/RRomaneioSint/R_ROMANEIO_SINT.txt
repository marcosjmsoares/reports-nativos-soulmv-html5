--R_ROMANEIO_SINT
SELECT   CD_ROMANEIO||' - '||DS_ROMANEIO DS_ROMANEIO,
         NM_USUARIO_LIBERACAO,
         DT_LIBERACAO,
         NM_MOTORISTA,
         NM_RAZAO_SOCIAL,
         DS_PLACA,
         DS_DESTINO,
         TP_SITUACAO,
         Count(1) QTD
FROM (  SELECT 	R.CD_ROMANEIO,
	    	        R.DS_ROMANEIO,
               R.TP_SITUACAO,
		            (SELECT U2.NM_USUARIO FROM DBASGU.USUARIOS U2 WHERE R.CD_USUARIO_LIBERACAO = U2.CD_USUARIO) NM_USUARIO_LIBERACAO,
		            TO_CHAR(R.DT_LIBERACAO,'DD/MM/YYYY') DT_LIBERACAO,
                MR.NM_MOTORISTA,
	              TR.NM_RAZAO_SOCIAL,
		            VR.DS_PLACA,
                RS.CD_DESTINO ||' - '||DR.DS_DESTINO DS_DESTINO
        FROM  	DBAMV.ROMANEIO R,
                DBAMV.TRANSPORTADORA_ROMANEIO TR,
                DBAMV.MOTORISTA_ROMANEIO MR,
                DBAMV.VEICULO_ROMANEIO VR,
                DBASGU.USUARIOS U,
                DBAMV.ROMANEIO_SOLICITACAO RS,
                DBAMV.DESTINO_ROMANEIO DR
        WHERE   R.CD_TRANSPORTADORA = TR.CD_TRANSPORTADORA(+)
        AND     R.CD_VEICULO = VR.CD_VEICULO
        AND     R.CD_MOTORISTA = MR.CD_MOTORISTA
        AND     R.CD_USUARIO = U.CD_USUARIO
        AND     RS.CD_DESTINO = DR.CD_DESTINO(+)
        AND     RS.CD_ROMANEIO(+) = R.CD_ROMANEIO
:CAMPO_CF_SITUACAO
)
GROUP BY CD_ROMANEIO,
         DS_ROMANEIO,
         NM_USUARIO_LIBERACAO,
         DT_LIBERACAO,
         NM_MOTORISTA,
         NM_RAZAO_SOCIAL,
         DS_PLACA,
         DS_DESTINO,
         TP_SITUACAO
ORDER BY 1
