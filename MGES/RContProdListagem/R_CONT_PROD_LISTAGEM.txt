--R_CONT_PROD_LISTAGEM
SELECT * FROM (
Select distinct
               especie.cd_especie cd_especie,
               especie.ds_especie ds_especie,
               produto.cd_produto cd_produto,
               produto.ds_produto ds_produto,
               estoque.cd_estoque cd_estoque,
               localizacao.Ds_Localizacao Ds_Localizacao_Prateleira
From  dbamv.especie,
      dbamv.produto,
      dbamv.itcontagem_da_abertura it,
      dbamv.contagem cont,
      dbamv.estoque ,
      dbamv.est_pro,
     dbamv.localizacao
Where especie.cd_especie = produto.cd_especie
  And it.cd_produto = produto.cd_produto
  And cont.cd_contagem = it.cd_contagem
  And cont.cd_estoque = estoque.cd_estoque
  And produto.cd_produto_tem is not null
  And Est_Pro.Cd_Produto =  Produto.Cd_Produto
  And Est_Pro.Cd_Estoque = estoque.Cd_Estoque
  And Localizacao.cd_estoque = Est_Pro.Cd_Estoque
  And Localizacao.cd_localizacao = Est_Pro.Cd_Localizacao
:CAMPO_CF_ORDEM
)
