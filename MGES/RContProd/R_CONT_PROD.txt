--R_CONT_PROD
SELECT   CONTAGEM.CD_CONTAGEM CD_CONTAGEM,
         CONTAGEM.CD_ESTOQUE  CD_ESTOQUE,
         ESTOQUE.DS_ESTOQUE DS_ESTOQUE,
         CONTAGEM.DT_CONTAGEM  DT_CONTAGEM,
         CONTAGEM.TP_CONTAGEM  TP_CONTAGEM,
         CONTAGEM.SN_ABERTO SN_ABERTO,
         ITCONTAGEM.CD_PRODUTO  CODIGO,
         PRODUTO.DS_PRODUTO  DESCRICAO,
         ITCONTAGEM.CD_LOTE CD_LOTE,
         ITCONTAGEM.DT_VALIDADE DT_VALIDADE,
         ITCONTAGEM.CD_UNI_PRO CD_UNI_PRO,
         UNI_PRO.DS_UNIDADE DS_UNIDADE,
         Sum(ITCONTAGEM.QT_ESTOQUE) QT_ESTOQUE,
         sum(itcontagem.qt_kit) qt_kit,
         Sum(ITCONTAGEM.QT_ESTOQUE_DOADO) QT_ESTOQUE_DOADO,
         UNI_PRO.VL_FATOR VL_FATOR,
         DECODE(CONTAGEM.SN_ABERTO, 'N', dbamv.verif_vl_custo_medio(itcontagem.cd_produto,NVL(contagem.dt_geracao,TRUNC(SYSDATE)) , 'H', Produto.Vl_Custo_Medio,                                                                             NVL(contagem.hr_geracao, to_date(to_char(sysdate,'hh24'), 'hh24')) )* Uni_Pro.Vl_Fator, ITCONTAGEM.VL_CUSTO_MEDIO)    VL_CUSTO_MEDIO,
         ESPECIE.CD_ESPECIE CD_ESPECIE,
         ESPECIE.DS_ESPECIE DS_ESPECIE
FROM  DBAMV.CONTAGEM,
            DBAMV.ESTOQUE,
            DBAMV.ITCONTAGEM,
            DBAMV.PRODUTO,
            DBAMV.UNI_PRO,
            DBAMV.ESPECIE
WHERE  Estoque.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA 
AND        CONTAGEM.CD_CONTAGEM              = ITCONTAGEM.CD_CONTAGEM
AND        CONTAGEM.CD_ESTOQUE                 = ESTOQUE.CD_ESTOQUE
AND        ITCONTAGEM.CD_PRODUTO             = PRODUTO.CD_PRODUTO
AND        ITCONTAGEM.CD_UNI_PRO               = UNI_PRO.CD_UNI_PRO
AND        PRODUTO.CD_ESPECIE                      = ESPECIE.CD_ESPECIE
Group By
CONTAGEM.CD_CONTAGEM,
         CONTAGEM.CD_ESTOQUE,
         ESTOQUE.DS_ESTOQUE,
         CONTAGEM.DT_CONTAGEM,
         CONTAGEM.TP_CONTAGEM,
         CONTAGEM.SN_ABERTO,
         ITCONTAGEM.CD_PRODUTO,
         PRODUTO.DS_PRODUTO ,
         ITCONTAGEM.CD_LOTE,
         ITCONTAGEM.DT_VALIDADE,
         ITCONTAGEM.CD_UNI_PRO,
         UNI_PRO.DS_UNIDADE,
         UNI_PRO.VL_FATOR,
         DECODE(CONTAGEM.SN_ABERTO, 'N', dbamv.verif_vl_custo_medio(itcontagem.cd_produto,NVL(contagem.dt_geracao,TRUNC(SYSDATE)) , 'H', Produto.Vl_Custo_Medio,                                                                             NVL(contagem.hr_geracao, to_date(to_char(sysdate,'hh24'), 'hh24')) )* Uni_Pro.Vl_Fator, ITCONTAGEM.VL_CUSTO_MEDIO)  ,
         ESPECIE.CD_ESPECIE ,
         ESPECIE.DS_ESPECIE
