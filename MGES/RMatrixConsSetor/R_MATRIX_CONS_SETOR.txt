--R_MATRIX_CONS_SETOR
SELECT * FROM (
Select Mvto_Estoque.Cd_Atendimento
	,Paciente.Nm_Paciente
	,Paciente.Nm_Social_Paciente
	,Paciente.Dt_Nascimento
  From Dbamv.Setor
	,Dbamv.Especie
	,Dbamv.Uni_Pro
	,Dbamv.Produto
	,Dbamv.ItMvto_Estoque
	,Dbamv.Paciente
	,Dbamv.Atendime
	,Dbamv.Mvto_Estoque
	,Dbamv.Estoque
 Where Mvto_Estoque.Dt_Mvto_Estoque	Between :CAMPO_DTINI and 
      AND MVTO_ESTOQUE.CD_ESTOQUE  = ESTOQUE.CD_ESTOQUE
      AND ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
      and Mvto_Estoque.Tp_Mvto_Estoque	In ( 'P', 'C' )
      AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
      and Atendime.Cd_Atendimento		= Mvto_Estoque.Cd_Atendimento
      and Paciente.Cd_Paciente		= Atendime.Cd_Paciente
      and ItMvto_Estoque.Cd_Mvto_Estoque	= Mvto_Estoque.Cd_Mvto_Estoque
      and Produto.Cd_Produto			= ItMvto_Estoque.Cd_Produto
      and Uni_Pro.Cd_Uni_Pro			= ItMvto_Estoque.Cd_Uni_Pro
      and Especie.Cd_Especie			= Produto.Cd_Especie
      and Setor.Cd_Setor			= Mvto_Estoque.Cd_Setor
 Group By Mvto_Estoque.Cd_Atendimento
	,Paciente.Nm_Paciente
	,Paciente.Nm_Social_Paciente
	,Paciente.Dt_Nascimento
) order by cd_atendimento , nm_paciente, nm_social_paciente, dt_nascimento

--R_MATRIX_CONS_SETOR_X_MATRIX
SELECT * FROM (
Select Mvto_Estoque.Cd_Atendimento cd_atendimento_sub
	,Paciente.Nm_Paciente 		nm_paciente_sub
	,InitCap( Especie.Ds_Especie) Ds_Especie_sub
	,InitCap( Setor.Nm_Setor ) Nm_Setor_sub
	,Sum( To_Number(Decode( Mvto_Estoque.Tp_Mvto_Estoque, 'P', 1, 'C', -1 ))
		* ItMvto_Estoque.Qt_Movimentacao * Uni_Pro.Vl_Fator
		* Verif_Vl_Custo_Medio( Produto.Cd_Produto, Mvto_Estoque.Dt_Mvto_Estoque, 'R', Produto.Vl_Custo_Medio, Mvto_Estoque.Hr_Mvto_Estoque ) )	Vl_Consumo_sub
  From Dbamv.Setor
	,Dbamv.Especie
	,Dbamv.Uni_Pro
	,Dbamv.Produto
	,Dbamv.ItMvto_Estoque
	,Dbamv.Paciente
	,Dbamv.Atendime
	,Dbamv.Mvto_Estoque
	,Dbamv.Estoque
 Where Mvto_Estoque.Dt_Mvto_Estoque	Between :CAMPO_DTINI and 
      AND MVTO_ESTOQUE.CD_ESTOQUE  	= ESTOQUE.CD_ESTOQUE
      AND ESTOQUE.CD_MULTI_EMPRESA 	= :CAMPO_CD_MULTI_EMPRESA
      and Mvto_Estoque.Tp_Mvto_Estoque	In ( 'P', 'C' )
      AND ATENDIME.CD_MULTI_EMPRESA 	= :CAMPO_CD_MULTI_EMPRESA
      and Atendime.Cd_Atendimento		= Mvto_Estoque.Cd_Atendimento
      and Paciente.Cd_Paciente		        = Atendime.Cd_Paciente
      and ItMvto_Estoque.Cd_Mvto_Estoque	= Mvto_Estoque.Cd_Mvto_Estoque
      and Produto.Cd_Produto			= ItMvto_Estoque.Cd_Produto
      and Uni_Pro.Cd_Uni_Pro			= ItMvto_Estoque.Cd_Uni_Pro
      and Especie.Cd_Especie			= Produto.Cd_Especie
      and Setor.Cd_Setor			= Mvto_Estoque.Cd_Setor
      and atendime.cd_atendimento		= :CAMPO_CD_ATENDIMENTO
 Group By Mvto_Estoque.Cd_Atendimento
	,Paciente.Nm_Paciente
	,Especie.Ds_Especie
	,Setor.Nm_Setor
) order by cd_atendimento_sub , nm_paciente_sub , ds_especie_sub, nm_setor_sub
