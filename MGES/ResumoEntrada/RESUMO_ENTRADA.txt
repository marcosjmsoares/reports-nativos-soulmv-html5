--RESUMO_ENTRADA
SELECT ent_pro.cd_ent_pro,
               ent_pro.dt_entrada,
               ent_pro.cd_fornecedor,
               fornecedor.nm_fornecedor,
               ent_pro.nr_documento,
               tip_doc.ds_tip_doc,
               itent_pro.cd_produto,
               produto.ds_produto,
               itent_pro.cd_uni_pro,
               uni_pro.ds_unidade,
               nvl(itlot_ent.qt_ent_pro,itent_pro.qt_entrada) qt_entrada,
                itent_pro.vl_unitario  vl_unitario,
               nvl(itlot_ent.qt_ent_pro,itent_pro.qt_entrada) * itent_pro.vl_unitario vl_total,
               itlot_ent.cd_lote
FROM   dbamv.tip_doc,
             dbamv.fornecedor,
             dbamv.produto,
             dbamv.uni_pro,
             dbamv.ent_pro,
             dbamv.itent_pro,
             dbamv.itlot_ent,
             dbamv.estoque
WHERE  ent_pro.cd_ent_pro    = itent_pro.cd_ent_pro
AND        ent_pro.cd_fornecedor = fornecedor.cd_fornecedor
AND        ent_pro.cd_estoque = estoque.cd_estoque
AND        ent_pro.cd_tip_doc  = tip_doc.cd_tip_doc
AND        itent_pro.cd_produto  = produto.cd_produto
AND        itent_pro.cd_uni_pro  = uni_pro.cd_uni_pro
AND        itent_pro.cd_itent_pro = itlot_ent.cd_itent_pro(+)
AND        ent_pro.dt_entrada BETWEEN :CAMPO_DTINIC  AND 
AND        estoque.cd_multi_empresa =  :CAMPO_CD_MULTI_EMPRESA
