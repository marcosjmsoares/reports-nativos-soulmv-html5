--R_CONSUMO_SETOR
SELECT * FROM (
Select    setor.cd_setor                                           cd_setor_produto
              ,setor.nm_setor                                          nm_setor_produto
              ,c_conset.cd_produto                               cd_produto_produto
              ,produto.ds_produto                                   ds_produto_produto  
              ,sum( ( ( nvl( c_conset.qt_saida_paciente, 0 )        + nvl( c_conset.qt_saida_setor  ,0 ) ) - 
                         ( nvl( c_conset.qt_devolvida_paciente, 0 ) + nvl( c_conset.qt_devolvida_setor,0 ) ) ) / verif_vl_fator_prod( c_conset.cd_produto )   )   qt_movimento_produto
              ,sum( ( nvl( c_conset.vl_saida_paciente, 0 )          + nvl( c_conset.vl_saida_setor  ,0 ) ) - 
                       ( nvl( c_conset.vl_devolvida_paciente, 0 )   + nvl( c_conset.vl_devolvida_setor,0 ) ) ) vl_total_produto
   From   dbamv.c_conset
        , dbamv.setor
        , dbamv.produto
        , dbamv.especie
	, dbamv.estoque
Where   c_conset.cd_mes    = TO_NUMBER( TO_CHAR( :CAMPO_DTINIC, 'MM' ) ) 
   and  c_conset.cd_ano    = TO_NUMBER( TO_CHAR( :CAMPO_DTINIC, 'YYYY' ) ) 
   and  produto.cd_especie = especie.cd_especie
   and  c_conset.cd_setor  = setor.cd_setor 
   and  produto.cd_produto = c_conset.cd_produto  
   and  estoque.cd_estoque = c_conset.cd_estoque
   and  estoque.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
group by  setor.cd_setor
              ,setor.nm_setor
              ,c_conset.cd_produto
              ,produto.ds_produto
    Order by  setor.cd_setor
              ,setor.nm_setor
              ,produto.ds_produto
)
ORDER BY CD_SETOR_PRODUTO,NM_SETOR_PRODUTO

              ,produto.ds_produto 				ds_produto
              ,sum( ( ( nvl( c_conset.qt_saida_paciente, 0 )        + nvl( c_conset.qt_saida_setor  ,0 ) ) -

              ,sum( ( nvl( c_conset.vl_saida_paciente, 0 )          + nvl( c_conset.vl_saida_setor  ,0 ) ) - 
                       ( nvl( c_conset.vl_devolvida_paciente, 0 )   + nvl( c_conset.vl_devolvida_setor,0 ) ) ) vl_total
   From   dbamv.c_conset

	, dbamv.estoque
Where   c_conset.cd_mes    = TO_NUMBER( TO_CHAR( :CAMPO_DTINIC, 'MM' ) ) 
   and  c_conset.cd_ano    = TO_NUMBER( TO_CHAR( :CAMPO_DTINIC, 'YYYY' ) )

   and  produto.cd_produto = c_conset.cd_produto

   and  estoque.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
group by setor.cd_setor
              ,setor.nm_setor
              ,verif_ds_unid_prod(c_conset.cd_produto)
              ,especie.cd_especie
              ,especie.ds_especie
              ,c_conset.cd_produto
              ,produto.ds_produto
              ,produto.vl_custo_medio
Order by  setor.nm_setor, especie.ds_especie
)
WHERE CD_SETOR_ESPECIE = :CAMPO_CD_SETOR_PRODUTO
)
ORDER BY CD_SETOR_ESPECIE,NM_SETOR_ESPECIE,CD_ESPECIE,DS_ESPECIE
