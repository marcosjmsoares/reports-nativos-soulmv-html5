select distinct
           a.cd_estoque,
           a.ds_estoque,
           b.cd_especie, b.ds_especie,
           c.cd_reduzido, c.cd_contabil, c.ds_conta,
           d.cd_reduzido cd_reduzido_despesas , d.cd_contabil cd_contabil_despesas, d.ds_conta ds_conta_despesas,
           e.cd_reduzido cd_reduzido_receita, e.cd_contabil cd_contabil_receita, e.ds_conta ds_conta_receita,
           f.cd_setor, f.nm_setor
           from Dbamv.estoque a,
                  Dbamv.especie b,
                  Dbamv.plano_contas c,
                  Dbamv.plano_contas d,
                  Dbamv.plano_contas e,
                  Dbamv.setor f,
                  Dbamv.estoque_pl_contas g,
                  Dbamv.multi_empresas h
           where a.cd_estoque = g.cd_estoque
               and a.Cd_Multi_Empresa = Pkg_Mv2000.Le_Empresa
               and b.cd_especie = g.cd_especie
               and c.cd_reduzido = g.cd_reduzido
               and d.cd_reduzido = g.cd_reduzido_despesa
               and e.cd_reduzido = g.cd_reduzido_receita
               and f.cd_setor = g.cd_setor
                order by ds_estoque, ds_especie
