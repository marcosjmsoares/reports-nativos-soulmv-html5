--R_CONTAGEM_DIVERGENCIA
SELECT distinct
  contagem.cd_contagem,
  contagem.dt_geracao,
  contagem.cd_usuario_abertura,
  contagem.dt_abertura,
  contagem.cd_usuario_finaliza,
  contagem.dt_finaliza,
  contagem.cd_usuario_aprova,
  contagem.dt_aprovacao,
  item_contagem_aberta.cd_produto,
  produto.ds_produto,
  item_contagem_aberta.cd_uni_pro,
  uni_pro.ds_unidade,
  produto.cd_produto,
  ITCONTAGEM_DA_ABERTURA.cd_sequencial
FROM
  dbamv.contagem,
  dbamv.item_contagem_aberta,
  DBAMV.ITCONTAGEM_DA_ABERTURA,
  dbamv.produto,
  dbamv.uni_pro
WHERE contagem.cd_contagem = item_contagem_aberta.cd_contagem
AND   contagem.cd_contagem = ITCONTAGEM_DA_ABERTURA.cd_contagem
AND   item_contagem_aberta.cd_produto = produto.cd_produto
AND   item_contagem_aberta.cd_produto = ITCONTAGEM_DA_ABERTURA.cd_produto
AND   item_contagem_aberta.cd_uni_pro = uni_pro.cd_uni_pro
AND   item_contagem_aberta.nr_sequencial = 0
AND   item_contagem_aberta.cd_contagem = :CAMPO_CD_CONTAGEM
AND   ((Nvl(:CAMPO_SN_ETAPAS,'N') = 'N' AND Nvl(Nvl(Nvl(Item_contagem_aberta.qt_contagem3,item_contagem_aberta.qt_contagem2),item_contagem_aberta.qt_contagem1),0) <>   Nvl(item_contagem_aberta.qt_saldo_atual,0)) or
       (Nvl(:CAMPO_SN_ETAPAS,'N') = 'S' AND NVL(item_contagem_aberta.QT_CONTAGEM1,0) != NVL(item_contagem_aberta.QT_CONTAGEM2,-1)))
ORDER BY ITCONTAGEM_DA_ABERTURA.cd_sequencial
