--R_DEV_FOR_ESPEC
SELECT * FROM (
SELECT itdev_for.cd_produto,
               produto.ds_produto,
               especie.cd_especie,
               especie.ds_especie,
               verif_ds_unid_prod(produto.cd_produto) ds_unidade,
               sum( itdev_for.qt_devolvida * uni_pro.vl_fator / verif_vl_fator_prod(produto.cd_produto)) QT_DEVOLVIDA,
               sum(itent_pro.vl_custo_real * itdev_for.qt_devolvida) vl_total
   FROM dbamv.dev_for,
              dbamv.itdev_for,
              dbamv.tip_doc,
              dbamv.fornecedor,
              dbamv.produto,
              dbamv.uni_pro,
              dbamv.ent_pro,
              dbamv.itent_pro,
              dbamv.estoque,
              dbamv.especie
WHERE ent_pro.cd_ent_pro      = itent_pro.cd_ent_pro
      AND dev_for.cd_ent_pro      = ent_pro.cd_ent_pro
      AND itdev_for.cd_devolucao = dev_for.cd_devolucao
      AND itdev_for.cd_itent_pro  = itent_pro.cd_itent_pro
      AND ent_pro.cd_fornecedor = fornecedor.cd_fornecedor
      AND ent_pro.cd_estoque     = estoque.cd_estoque
      AND dev_for.cd_tip_doc      = tip_doc.cd_tip_doc
      AND itdev_for.cd_produto    = produto.cd_produto
      AND especie.cd_especie     = produto.cd_especie
      AND itdev_for.cd_uni_pro    = uni_pro.cd_uni_pro
      AND trunc(dev_for.dt_devolucao) BETWEEN :CAMPO_DTINIC
                                                                  AND 
      AND ent_pro.cd_fornecedor IS NOT NULL
      AND Estoque.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
GROUP BY  itdev_for.cd_produto,
                     produto.ds_produto,
                     especie.cd_especie,
                     especie.ds_especie,
                     verif_ds_unid_prod(produto.cd_produto)
ORDER BY  ds_produto
)
ORDER BY CD_ESPECIE,DS_ESPECIE,ds_produto
