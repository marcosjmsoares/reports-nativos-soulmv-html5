SELECT  MVTO_ESTOQUE.CD_MVTO_ESTOQUE CD_MVTO_ESTOQUE
       ,MVTO_ESTOQUE.CD_ATENDIMENTO CD_ATENDIMENTO
       ,PACIENTE.CD_PACIENTE CD_PACIENTE1
       ,PACIENTE.NM_PACIENTE NM_PACIENTE
       ,MVTO_ESTOQUE.CD_PRESTADOR CD_PRESTADOR
       ,PRESTADOR.NM_PRESTADOR NM_PRESTADOR
       ,ATENDIME.CD_LEITO
       ,LEITO.DS_LEITO
       ,LEITO.CD_UNID_INT
       ,UNID_INT.DS_UNID_INT
FROM    dbamv.MVTO_ESTOQUE MVTO_ESTOQUE
       ,dbamv.ATENDIME ATENDIME
       ,dbamv.PACIENTE PACIENTE
       ,dbamv.PRESTADOR PRESTADOR
       ,dbamv.LEITO
       ,dbamv.UNID_INT
       ,dbamv.ESTOQUE 
WHERE ATENDIME.CD_ATENDIMENTO = MVTO_ESTOQUE.CD_ATENDIMENTO
AND   PACIENTE.CD_PACIENTE = ATENDIME.CD_PACIENTE
AND   PRESTADOR.CD_PRESTADOR = MVTO_ESTOQUE.CD_PRESTADOR
AND   ATENDIME.CD_LEITO = LEITO.CD_LEITO
AND   LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT
AND  MVTO_ESTOQUE.CD_MVTO_ESTOQUE = :CAMPO_CD_MVTO_ESTOQUE
AND MVTO_ESTOQUE.CD_ESTOQUE  = ESTOQUE.CD_ESTOQUE
AND ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
