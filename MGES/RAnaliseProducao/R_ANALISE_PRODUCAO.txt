--R_ANALISE_PRODUCAO
SELECT * FROM (SELECT S.CD_SOLSAI_PRO,
                      S.CD_PRE_MED,
		      PAC.NM_PACIENTE,
		      to_char(PM.DT_PRE_MED,'DD/MM/YYYY') DT_PRE_MED, 
					                  to_char(PM.HR_PRE_MED,'HH24') HR_PRE_MED,
		      P.CD_PRODUTO,
                      P.DS_PRODUTO,
                      Nvl(I.QT_SOLICITADO, DECODE(PM.TP_AGENDAMENTO,'FAR',IPM.QTD_TOTAL_DISPENSAR, IPM.QT_ITPRE_MED)) QT_SOLICITADO,
		      (SELECT DS_FOR_APL FROM DBAMV.FOR_APL WHERE CD_FOR_APL = IPM.CD_FOR_APL) DS_FOR_APL, 
		      (SELECT DS_TIP_FRE FROM DBAMV.TIP_FRE WHERE CD_TIP_FRE = IPM.CD_TIP_FRE) DS_TIP_FRE,
		      (SELECT DS_UNIDADE FROM DBAMV.UNI_PRO WHERE CD_UNI_PRO = IPM.CD_UNI_PRO ) DS_UNIDADE,
		      I.CD_ORDEM_PRODUCAO,
		      I.CD_ITENT_PRO_PRODUCAO,
                      Nvl((SELECT QT_ESTOQUE_ATUAL 
                       FROM   DBAMV.EST_PRO
                       WHERE  CD_PRODUTO = P.CD_PRODUTO AND CD_ESTOQUE = S.CD_ESTOQUE), 0) QT_ESTOQUE_ATUAL,
		      S.CD_ESTOQUE,
		      PAC.CD_PACIENTE,
		      PM.CD_SETOR,
		      PM.CD_UNID_INT
             FROM     DBAMV.SOLSAI_PRO S,
                      DBAMV.ITSOLSAI_PRO I,
                      DBAMV.PRE_MED PM,
                      DBAMV.ITPRE_MED IPM,
                      DBAMV.PRODUTO P,
                      DBAMV.PACIENTE PAC,
                      DBAMV.ATENDIME A
             WHERE    I.CD_SOLSAI_PRO = S.CD_SOLSAI_PRO
             AND      PM.CD_PRE_MED = S.CD_PRE_MED
             AND      PM.CD_PRE_MED = IPM.CD_PRE_MED
             AND      I.CD_PRODUTO = IPM.CD_PRODUTO
             AND      I.CD_PRODUTO = P.CD_PRODUTO
             AND     S.CD_ATENDIMENTO = A.CD_ATENDIMENTO
             AND     A.CD_PACIENTE = PAC.CD_PACIENTE
	     AND     P.SN_CONSIGNADO = 'R' 
       UNION
       SELECT	     S.CD_SOLSAI_PRO,
		     S.CD_PRE_MED,
                     PAC.NM_PACIENTE, 
		     to_char(PM.DT_PRE_MED,'DD/MM/YYYY') DT_PRE_MED, 
		     to_char(PM.HR_PRE_MED,'HH24') HR_PRE_MED, 
                     P.CD_PRODUTO,
                     P.DS_PRODUTO,
		     Nvl(I.QT_SOLICITADO, DECODE(PM.TP_AGENDAMENTO,'FAR',IPM.QTD_TOTAL_DISPENSAR, IPM.QT_ITPRE_MED)) QT_SOLICITADO,
		     (SELECT DS_FOR_APL FROM DBAMV.FOR_APL WHERE CD_FOR_APL = IPM.CD_FOR_APL) DS_FOR_APL, 
		     (SELECT DS_TIP_FRE FROM DBAMV.TIP_FRE WHERE CD_TIP_FRE = IPM.CD_TIP_FRE) DS_TIP_FRE,
		     (SELECT DS_UNIDADE FROM DBAMV.UNI_PRO WHERE CD_UNI_PRO = Nvl(IPM.CD_UNI_PRO, I.CD_UNI_PRO) ) DS_UNIDADE,
		     I.CD_ORDEM_PRODUCAO,  
		     I.CD_ITENT_PRO_PRODUCAO,
		     Nvl((SELECT QT_ESTOQUE_ATUAL FROM   DBAMV.EST_PRO
			  WHERE  CD_PRODUTO = P.CD_PRODUTO AND CD_ESTOQUE = S.CD_ESTOQUE), 0) QT_ESTOQUE_ATUAL,
			  S.CD_ESTOQUE,
		      PAC.CD_PACIENTE,	
			PM.CD_SETOR,
			PM.CD_UNID_INT
             FROM     DBAMV.SOLSAI_PRO S,
                      DBAMV.ITSOLSAI_PRO I,
                      DBAMV.PRE_MED PM,
                      DBAMV.ITPRE_MED IPM,
                      DBAMV.PRODUTO P,
                      DBAMV.PACIENTE PAC,
                      DBAMV.ATENDIME A
             WHERE    I.CD_SOLSAI_PRO = S.CD_SOLSAI_PRO
             AND      PM.CD_PRE_MED = S.CD_PRE_MED
             AND      PM.CD_PRE_MED = IPM.CD_PRE_MED
             AND      I.DS_NPADRONIZADO = IPM.DS_NPADRONIZADO
             AND      I.CD_PRODUTO = P.CD_PRODUTO
             AND     S.CD_ATENDIMENTO = A.CD_ATENDIMENTO
             AND     A.CD_PACIENTE = PAC.CD_PACIENTE
	     AND     P.SN_CONSIGNADO = 'R' 
       UNION
       SELECT        S.CD_SOLSAI_PRO,
                     S.CD_PRE_MED,
		     PAC.NM_PACIENTE,
		     to_char(PM.DT_PRE_MED,'DD/MM/YYYY') DT_PRE_MED, 
		     to_char(PM.HR_PRE_MED,'HH24') HR_PRE_MED, 
		     NULL CD_PRODUTO,
		     I.DS_NPADRONIZADO DS_PRODUTO,
		     Nvl(I.QT_SOLICITADO, DECODE(PM.TP_AGENDAMENTO,'FAR',IPM.QTD_TOTAL_DISPENSAR, IPM.QT_ITPRE_MED)) QT_SOLICITADO,
		     (SELECT DS_FOR_APL FROM DBAMV.FOR_APL WHERE CD_FOR_APL = IPM.CD_FOR_APL) DS_FOR_APL, 
		     (SELECT DS_TIP_FRE FROM DBAMV.TIP_FRE WHERE CD_TIP_FRE = IPM.CD_TIP_FRE) DS_TIP_FRE,
		     (SELECT DS_UNIDADE_PRESC FROM DBAMV.UNI_PRESC WHERE CD_UNI_PRESC = IPM.CD_UNI_PRESC ) DS_UNIDADE ,
		     I.CD_ORDEM_PRODUCAO,
		     I.CD_ITENT_PRO_PRODUCAO,
		     0 QT_ESTOQUE_ATUAL,
		     S.CD_ESTOQUE,
			PAC.CD_PACIENTE,
			PM.CD_SETOR,
			PM.CD_UNID_INT
        FROM     DBAMV.SOLSAI_PRO S,
                      DBAMV.ITSOLSAI_PRO I,
                      DBAMV.PRE_MED PM,
                      DBAMV.ITPRE_MED IPM,
                      DBAMV.PACIENTE PAC,
                      DBAMV.ATENDIME A
             WHERE    I.CD_SOLSAI_PRO = S.CD_SOLSAI_PRO
             AND      PM.CD_PRE_MED = S.CD_PRE_MED
             AND      PM.CD_PRE_MED = IPM.CD_PRE_MED
	      AND      I.DS_NPADRONIZADO = IPM.DS_NPADRONIZADO
             AND     S.CD_ATENDIMENTO = A.CD_ATENDIMENTO
             AND     A.CD_PACIENTE = PAC.CD_PACIENTE
             AND     I.CD_PRODUTO IS NULL) X
WHERE 1 = 1 
ORDER BY  CD_SOLSAI_PRO ASC
