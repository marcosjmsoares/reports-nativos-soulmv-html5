--R_BAIPRO
SELECT * FROM (
SELECT
       MVTO_ESTOQUE.CD_USUARIO,
       MVTO_ESTOQUE.CD_MVTO_ESTOQUE,
       MVTO_ESTOQUE.CD_ESTOQUE,
       ESTOQUE.DS_ESTOQUE,
       MVTO_ESTOQUE.CD_MOT_BAI,
       MOT_BAI.DS_MOT_BAI,
       MVTO_ESTOQUE.DT_MVTO_ESTOQUE,
       MVTO_ESTOQUE.HR_MVTO_ESTOQUE,
       MVTO_ESTOQUE.DT_IMPRESSAO,
       MVTO_ESTOQUE.CD_SETOR,
       SETOR.NM_SETOR,
       ITMVTO_ESTOQUE.CD_PRODUTO,
       PRODUTO.DS_PRODUTO,
       ITMVTO_ESTOQUE.CD_PRODUTO_KIT              CD_PRODUTO_KIT,
       PRODUTO_KIT.DS_PRODUTO                             DS_PRODUTO_KIT,
       UNI_PRO.DS_UNIDADE,
       ITMVTO_ESTOQUE.CD_LOTE,
       ITMVTO_ESTOQUE.DT_VALIDADE,
       ITMVTO_ESTOQUE.DS_ENDERECO,
       ITMVTO_ESTOQUE.QT_MOVIMENTACAO,
       VERIF_VL_CUSTO_MEDIO(ITMVTO_ESTOQUE.CD_PRODUTO,MVTO_ESTOQUE.DT_MVTO_ESTOQUE,'H',NULL,MVTO_ESTOQUE.HR_MVTO_ESTOQUE) * UNI_PRO.VL_FATOR VL_UNITARIO
FROM
       DBAMV.MVTO_ESTOQUE,
       DBAMV.ITMVTO_ESTOQUE,
       DBAMV.ESTOQUE,
       DBAMV.UNI_PRO,
       DBAMV.PRODUTO,
       DBAMV.PRODUTO PRODUTO_KIT,
       DBAMV.MOT_BAI,
       DBAMV.SETOR
WHERE  MVTO_ESTOQUE.TP_MVTO_ESTOQUE  =  'X'
AND    MVTO_ESTOQUE.CD_MOT_BAI       =  MOT_BAI.CD_MOT_BAI
AND    MVTO_ESTOQUE.CD_MVTO_ESTOQUE  =  ITMVTO_ESTOQUE.CD_MVTO_ESTOQUE
AND    MVTO_ESTOQUE.CD_ESTOQUE       =  ESTOQUE.CD_ESTOQUE
AND    ITMVTO_ESTOQUE.CD_UNI_PRO     =  UNI_PRO.CD_UNI_PRO(+)
AND    ITMVTO_ESTOQUE.CD_PRODUTO     =  PRODUTO.CD_PRODUTO(+)
AND    MVTO_ESTOQUE.CD_MVTO_ESTOQUE  =  :CAMPO_CD_MVTO_ESTOQUE
AND    ITMVTO_ESTOQUE.CD_PRODUTO_KIT =  PRODUTO_KIT.CD_PRODUTO(+)
AND    ESTOQUE.CD_MULTI_EMPRESA      =  :CAMPO_CD_MULTI_EMPRESA
AND    SETOR.CD_SETOR                =  MVTO_ESTOQUE.CD_SETOR
ORDER BY MVTO_ESTOQUE.CD_MVTO_ESTOQUE,
         PRODUTO.DS_PRODUTO
)
ORDER BY CD_MVTO_ESTOQUE,CD_ESTOQUE,DS_ESTOQUE,DT_MVTO_ESTOQUE,CD_USUARIO,HR_MVTO_ESTOQUE,CD_MOT_BAI,DS_MOT_BAI,DT_IMPRESSAO, DS_PRODUTO_KIT, CD_PRODUTO_KIT, DS_PRODUTO
