--R_DANFE
SELECT l2.ds_item_log_danfe
  FROM dbamv.log_danfe l1
     , dbamv.item_log_danfe l2
WHERE l1.cd_log_danfe = l2.cd_log_danfe
  AND l2.cd_log_danfe IN ( SELECT Max(l3.cd_log_danfe)
                             FROM dbamv.log_danfe l3
                            WHERE l3.nr_ide_nfe = :CAMPO_NR_IDE_NFE )
  AND l2.TP_ERRO = 'W'
ORDER BY l2.CD_ITEM_LOG_DANFE ASC
