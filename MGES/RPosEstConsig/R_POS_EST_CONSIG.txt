--R_POS_EST_CONSIG
SELECT * FROM (
select   decode(:CAMPO_ESTOQUE,'%','Todos', est_consig_forn.cd_estoque )     Estoque
           ,estoque.ds_estoque                  Ds_Estoque
           ,produto.cd_especie                   Especie
           ,produto.cd_classe                     Classe
           ,produto.cd_sub_cla                   SubClasse
           ,fornecedor.cd_fornecedor         Codigo_Fornecedor
           ,fornecedor.nm_fornecedor         Desc_Fornecedor
           ,produto.cd_produto                    Codigo_Produto
           ,produto.ds_produto                    Desc_Produto
           ,dbamv.verif_ds_unid_prod( produto.cd_produto, 'C' ) Unidade
           ,((nvl( est_consig_forn.qt_estoque_consignado, 0 ) +
             nvl( est_consig_forn.qt_estoque_transferencia, 0 )) -
            (Nvl(est_consig_forn.qt_estoque_consumido,0) + Nvl(est_consig_forn.qt_baixada, 0) )) /
            dbamv.verif_vl_fator_prod( produto.cd_produto, 'C' ) Qt_Total_Fornec
   from  dbamv.est_consig_forn
           ,dbamv.fornecedor
           ,dbamv.produto
           ,dbamv.estoque
where  est_consig_forn.cd_fornecedor = fornecedor.cd_fornecedor
   and  produto.cd_produto    = est_consig_forn.cd_produto
   and  estoque.cd_estoque    = est_consig_forn.cd_estoque
   and  estoque.cd_multi_empresa = Pkg_Mv2000.Le_Empresa
order by fornecedor.cd_fornecedor
            ,produto.cd_produto
            ,estoque.cd_estoque
)
ORDER BY CODIGO_FORNECEDOR,DESC_FORNECEDOR
