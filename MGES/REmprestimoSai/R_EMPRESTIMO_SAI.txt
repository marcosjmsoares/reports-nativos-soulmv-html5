--R_EMPRESTIMO_SAI
Select Distinct MVTO_ESTOQUE.CD_MVTO_ESTOQUE As SAIDA,
                MVTO_ESTOQUE.CD_ESTOQUE As ESTOQUE,
                ESTOQUE.DS_ESTOQUE,
                MVTO_ESTOQUE.CD_FORNECEDOR As FORNECEDOR,
                FORNECEDOR.NM_FORNECEDOR,
                To_Char(MVTO_ESTOQUE.DT_MVTO_ESTOQUE,
                        'DD/MM/YYYY') || ' ' || To_Char(MVTO_ESTOQUE.HR_MVTO_ESTOQUE,
                                                        'HH24') As DT_ENTRADA,
                MVTO_ESTOQUE.DS_OBS_EMPRESTIMO As OBSERVACAO,
                ITMVTO_ESTOQUE.DSP_CODIGO_DE_BARRAS As BARRAS,
                ITMVTO_ESTOQUE.CD_PRODUTO As PRODUTO,
                ITMVTO_ESTOQUE.CD_ITMVTO_ESTOQUE,
                PRODUTO.DS_PRODUTO,
                ITMVTO_ESTOQUE.CD_LOTE As LOTE,
                To_Char(ITMVTO_ESTOQUE.DT_VALIDADE,
                        'DD/MM/YYYY') As DT_VALIDADE,
                UNI_PRO.DS_UNIDADE As UNIDADE,
                ITMVTO_ESTOQUE.QT_MOVIMENTACAO,
                Nvl(ITMVTO_ESTOQUE.VL_UNITARIO_EMPRESTIMO,
                    DBAMV.Verif_Vl_Custo_Medio(PRODUTO.CD_PRODUTO, ITMVTO_ESTOQUE.DH_MVTO_ESTOQUE, 'H', Null, ITMVTO_ESTOQUE.DH_MVTO_ESTOQUE, :CAMPO_CD_MULTI_EMPRESA) * UNI_PRO.VL_FATOR) As VL_UNITARIO,
                UNI_PRO.VL_FATOR,
                ITMVTO_ESTOQUE.QT_MOVIMENTACAO * Nvl(ITMVTO_ESTOQUE.VL_UNITARIO_EMPRESTIMO,
                                                     DBAMV.Verif_Vl_Custo_Medio(PRODUTO.CD_PRODUTO, ITMVTO_ESTOQUE.DH_MVTO_ESTOQUE, 'H', Null, ITMVTO_ESTOQUE.DH_MVTO_ESTOQUE, :CAMPO_CD_MULTI_EMPRESA) * UNI_PRO.VL_FATOR) As VL_TOTAL
  From DBAMV.MVTO_ESTOQUE
Inner Join DBAMV.ITMVTO_ESTOQUE On
       ITMVTO_ESTOQUE.CD_MVTO_ESTOQUE = MVTO_ESTOQUE.CD_MVTO_ESTOQUE
Inner Join DBAMV.ESTOQUE On
       ESTOQUE.CD_ESTOQUE = MVTO_ESTOQUE.CD_ESTOQUE
   And ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
Inner Join DBAMV.PRODUTO On
       PRODUTO.CD_PRODUTO = ITMVTO_ESTOQUE.CD_PRODUTO
Inner Join DBAMV.UNI_PRO On
       UNI_PRO.CD_UNI_PRO = ITMVTO_ESTOQUE.CD_UNI_PRO
Left Join DBAMV.ITMVTO_ENT On
       ITMVTO_ENT.CD_ITMVTO_ESTOQUE = ITMVTO_ESTOQUE.CD_ITMVTO_ESTOQUE
Left Join DBAMV.FORNECEDOR On
       FORNECEDOR.CD_FORNECEDOR = MVTO_ESTOQUE.CD_FORNECEDOR
 Where 1 = 1
Order By MVTO_ESTOQUE.CD_MVTO_ESTOQUE,
       ITMVTO_ESTOQUE.CD_PRODUTO
