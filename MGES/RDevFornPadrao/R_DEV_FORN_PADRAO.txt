--R_DEV_FORN_PADRAO
SELECT * FROM (
select dev_for.cd_devolucao cd_devolucao
      ,dev_for.cd_ent_pro cd_ent_pro
      ,dev_for.cd_mot_dev cd_mot_dev
      ,mot_dev.ds_mot_dev ds_mot_dev
      ,dev_for.dt_devolucao dt_devolucao
      ,dev_for.hr_devolucao hr_devolucao
      ,dev_for.nr_documento nr_documento_dev
      ,dev_for.nr_serie nr_serie_dev
      ,dev_for.dt_emissao dt_emissao
      ,dev_for.vl_total vl_total
      ,dev_for.tp_devolucao tp_devolucao
      ,itdev_for.cd_produto cd_produto
      ,produto.ds_produto
      ,itdev_for.qt_devolvida qt_devolvida
      ,itdev_for.dt_validade dt_validade
      ,itdev_for.cd_lote cd_lote
      ,uni_pro.ds_unidade ds_unidade
      ,ent_pro.dt_entrada dt_entrada
      ,ent_pro.cd_ent_pro cd_ent_pro1
      ,ent_pro.nr_serie nr_serie_entrada
      ,ent_pro.nr_documento nr_documento_ent
      ,fornecedor.cd_fornecedor cd_fornecedor
      ,fornecedor.nm_fantasia nm_fantasia
      ,fornecedor.nm_fornecedor nm_fornecedor
      ,fornecedor.ds_endereco ds_endereco
      ,fornecedor.nr_cgc_cpf nr_cgc_cpf
      ,fornecedor.ds_endereco_email ds_endereco_email
      ,fornecedor.nr_cep nr_cep
      ,cidade.nm_cidade nm_cidade
      ,cidade.cd_uf cd_uf
      ,itdev_for.cd_itent_pro
  from dbamv.dev_for
      ,dbamv.ent_pro
      ,dbamv.itdev_for
      ,dbamv.produto
      ,dbamv.mot_dev
      ,dbamv.uni_pro
      ,dbamv.fornecedor
      ,dbamv.cidade
 where dev_for.cd_devolucao     = itdev_for.cd_devolucao
   and produto.cd_produto       = itdev_for.cd_produto
   and mot_dev.cd_mot_dev       = dev_for.cd_mot_dev
   and uni_pro.cd_uni_pro       = itdev_for.cd_uni_pro
   and ent_pro.cd_ent_pro       = dev_for.cd_ent_pro
   and fornecedor.cd_fornecedor = ent_pro.cd_fornecedor
   and fornecedor.cd_cidade     = cidade.cd_cidade
   and dev_for.cd_devolucao     = :CAMPO_CD_DEVOLUCAO
   And Exists ( Select Empresa_Produto.Cd_Produto
                     From Dbamv.Empresa_Produto
                   Where Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
                        and Empresa_Produto.Cd_Produto = Produto.Cd_Produto )
order by dev_for.cd_ent_pro, itdev_for.cd_produto
)
ORDER BY CD_DEVOLUCAO,CD_MOT_DEV,DS_MOT_DEV,CD_ENT_PRO1,DT_DEVOLUCAO,CD_ENT_PRO,VL_TOTAL,NR_CGC_CPF,HR_DEVOLUCAO,NR_SERIE_ENTRADA,TP_DEVOLUCAO,DT_ENTRADA,DS_ENDERECO_EMAIL,DS_ENDERECO,CD_FORNECEDOR,NR_DOCUMENTO_ENT,NM_FANTASIA,NR_CEP,NM_FORNECEDOR,NR_DOCUMENTO_DEV,NM_CIDADE,NR_SERIE_DEV,DT_EMISSAO,CD_UF
