--R_COMP_NOTA_CREDITO_PRODUTO
SELECT HOSPITAL.DS_RAZAO,
       HOSPITAL.CD_IDENTIFICADOR_HOSPITAL,
       FORNECEDOR.NM_FORNECEDOR,
       FORNECEDOR.CD_IDENTIFICADOR_PESSOA,
       NOTA_CRE_DEB_SUPRI.CD_NOTA_CRE_DEB,
       Decode(NOTA_CRE_DEB_SUPRI.TP_CRE_DEB, 'C', :CAMPO_TXT_CREDITO, 'D', :CAMPO_TXT_DEBITO) TP_CRE_DEB,
       (NOTA_CRE_DEB_SUPRI.CD_OPERACAO || Decode(NOTA_CRE_DEB_SUPRI.CD_OPERACAO, NULL, '', ' - ') || OPERACAO_NT_CRE_DEB_SUPRI.DS_OPERACAO) CD_OPERACAO,
       (NOTA_CRE_DEB_SUPRI.CD_ESTOQUE || Decode(NOTA_CRE_DEB_SUPRI.CD_ESTOQUE, NULL, '', ' - ') || ESTOQUE.DS_ESTOQUE) CD_ESTOQUE,
       (NOTA_CRE_DEB_SUPRI.CD_MOTIVO || Decode(NOTA_CRE_DEB_SUPRI.CD_MOTIVO, NULL, '', ' - ') || MOTIVO_LANC_NT_CRE_DEB_SUPRI.DS_MOTIVO_LANC) CD_MOTIVO,
       NOTA_CRE_DEB_SUPRI.DS_OBSERVACAO,
       NOTA_CRE_DEB_SUPRI.CD_MOEDA_LANCAMENTO,
       NOTA_CRE_DEB_SUPRI.VL_CAMBIO_LANCAMENTO,
       NOTA_CRE_DEB_SUPRI.NR_DOCUMENTO,
       NOTA_CRE_DEB_SUPRI.NR_SERIE,
       NOTA_CRE_DEB_SUPRI.NR_NCF,
       To_Char(NOTA_CRE_DEB_SUPRI.DT_EMISSAO, 'DD/MM/YYYY') DT_EMISSAO,
       To_Char(NOTA_CRE_DEB_SUPRI.DT_VENCIMENTO, 'DD/MM/YYYY') DT_VENCIMENTO,
       Nvl(NOTA_CRE_DEB_SUPRI.VL_DESCONTO_LANCAMENTO, 0) VL_DESCONTO_PADRAO_NOTA,
       Nvl(NOTA_CRE_DEB_SUPRI.VL_OUTROS_LANCAMENTO, 0) VL_OUTROS_PADRAO,
       Nvl(NOTA_CRE_DEB_SUPRI.VL_IMPOSTO_LANCAMENTO, 0) VL_IMPOSTO_PADRAO_NOTA,
       Nvl(NOTA_CRE_DEB_SUPRI.VL_TOTAL_NOTA_LANCAMENTO, 0) VL_TOTAL_NOTA_PADRAO,
       DECODE(NOTA_CRE_DEB_SUPRI.TP_SITUACAO, 'R', :CAMPO_TXT_REGISTRADA
                                            , 'S', :CAMPO_TXT_SOLICITADA
                                            , 'F', :CAMPO_TXT_CONFIRMADA
                                            , 'C', :CAMPO_TXT_CANCELADA) TP_SITUACAO,
       DECODE(NOTA_CRE_DEB_SUPRI.TP_SITUACAO, 'R', To_Char(NOTA_CRE_DEB_SUPRI.DT_REGISTRO, 'DD/MM/YYYY')
                                            , 'S', To_Char(NOTA_CRE_DEB_SUPRI.DT_SOLICITACAO, 'DD/MM/YYYY')
                                            , 'F', To_Char(NOTA_CRE_DEB_SUPRI.DT_CONFIRMACAO, 'DD/MM/YYYY')
                                            , 'C', To_Char(NOTA_CRE_DEB_SUPRI.DT_CANCELAMENTO, 'DD/MM/YYYY')) DT_SITUACAO,
       DECODE(NOTA_CRE_DEB_SUPRI.TP_SITUACAO, 'R', NOTA_CRE_DEB_SUPRI.CD_USU_REGISTRO
                                            , 'S', NOTA_CRE_DEB_SUPRI.CD_USU_SOLICITACAO
                                            , 'F', NOTA_CRE_DEB_SUPRI.CD_USU_CONFIRMACAO
                                            , 'C', NOTA_CRE_DEB_SUPRI.CD_USU_CANCELAMENTO) USU_SITUACAO,
       IT_NOTA_CRE_DEB_SUPRI.CD_ENT_PRO,
       Decode(IT_NOTA_CRE_DEB_SUPRI.CD_PRODUTO, NULL, IT_NOTA_CRE_DEB_SUPRI.DS_DESCONTO_BONUS_OUTROS, (IT_NOTA_CRE_DEB_SUPRI.CD_PRODUTO || Decode(IT_NOTA_CRE_DEB_SUPRI.CD_PRODUTO, NULL, '', ' - ') || PRODUTO.DS_PRODUTO)) CD_PRODUTO,
       IT_NOTA_CRE_DEB_SUPRI.CD_LOTE,
       To_Char(IT_NOTA_CRE_DEB_SUPRI.DT_VALIDADE_LOTE, 'DD/MM/YYYY') DT_VALIDADE_LOTE,
       (IT_NOTA_CRE_DEB_SUPRI.CD_UNI_PRO || Decode(IT_NOTA_CRE_DEB_SUPRI.CD_UNI_PRO, NULL, '', ' - ')|| UNI_PRO.DS_UNIDADE) CD_UNI_PRO,
       IT_NOTA_CRE_DEB_SUPRI.QT_CREDITO,
       Nvl(IT_NOTA_CRE_DEB_SUPRI.VL_UNITARIO_LANCAMENTO, 0) VL_UNITARIO_PADRAO,
       Nvl(IT_NOTA_CRE_DEB_SUPRI.VL_TOTAL_LANCAMENTO, 0) VL_TOTAL_PADRAO,
       Nvl(IT_NOTA_CRE_DEB_SUPRI.VL_PERC_DESCONTO, 0) VL_PERC_DESCONTO,
       Nvl(IT_NOTA_CRE_DEB_SUPRI.VL_DESCONTO_LANCAMENTO, 0) VL_DESCONTO_PADRAO,
       Nvl(IT_NOTA_CRE_DEB_SUPRI.VL_PERC_IMPOSTO, 0) VL_PERC_IMPOSTO,
       Nvl(IT_NOTA_CRE_DEB_SUPRI.VL_IMPOSTO_LANCAMENTO, 0) VL_IMPOSTO_PADRAO
  FROM DBAMV.NOTA_CRE_DEB_SUPRI,
       DBAMV.IT_NOTA_CRE_DEB_SUPRI,
       DBAMV.OPERACAO_NT_CRE_DEB_SUPRI,
       DBAMV.ESTOQUE,
       DBAMV.MOTIVO_LANC_NT_CRE_DEB_SUPRI,
       DBAMV.ENT_PRO,
       DBAMV.PRODUTO,
       DBAMV.LOTE,
       DBAMV.UNI_PRO,
       DBAMV.HOSPITAL,
       DBAMV.FORNECEDOR
 WHERE HOSPITAL.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
   AND NOTA_CRE_DEB_SUPRI.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
   AND NOTA_CRE_DEB_SUPRI.TP_PROD_SERV = 'P'
   AND NOTA_CRE_DEB_SUPRI.CD_FORNECEDOR = FORNECEDOR.CD_FORNECEDOR
   AND NOTA_CRE_DEB_SUPRI.CD_NOTA_CRE_DEB = IT_NOTA_CRE_DEB_SUPRI.CD_NOTA_CRE_DEB
   AND NOTA_CRE_DEB_SUPRI.CD_OPERACAO = OPERACAO_NT_CRE_DEB_SUPRI.CD_OPERACAO (+)
   AND NOTA_CRE_DEB_SUPRI.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE (+)
   AND NOTA_CRE_DEB_SUPRI.CD_MOTIVO = MOTIVO_LANC_NT_CRE_DEB_SUPRI.CD_MOTIVO_LANC (+)
   AND IT_NOTA_CRE_DEB_SUPRI.CD_ENT_PRO = ENT_PRO.CD_ENT_PRO (+)
   AND IT_NOTA_CRE_DEB_SUPRI.CD_PRODUTO = PRODUTO.CD_PRODUTO (+)
   AND IT_NOTA_CRE_DEB_SUPRI.CD_LOTE = LOTE.CD_LOTE (+)
   AND IT_NOTA_CRE_DEB_SUPRI.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO (+)
   AND NOTA_CRE_DEB_SUPRI.CD_NOTA_CRE_DEB = :CAMPO_CD_NOTA_CRE_DEB
