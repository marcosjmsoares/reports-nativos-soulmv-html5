--R_REQUISICAO_PRODUTO_SOLIC
select solsai_pro.cd_solsai_pro cd_solsai_pro
      ,solsai_pro.cd_requisicao cd_requisicao
      ,To_Char(solsai_pro.dt_solsai_pro, 'DD/MM/YYYY') dt_solsai_pro
      ,To_Char(solsai_pro.hr_solsai_pro, 'HH24') hr_solsai_pro
      ,solsai_pro.cd_estoque cd_estoque_solic
      ,estoque.ds_estoque ds_estoque_solic
      ,solsai_pro.cd_estoque_solicitante cd_estoque_dest
      ,estoque_solic.ds_estoque ds_estoque_dest
      ,solsai_pro.cd_setor cd_setor_dest
      ,setor.nm_setor nm_setor_dest
      ,solsai_pro.cd_usuario cd_usuario
  from dbamv.solsai_pro
      ,dbamv.estoque
      ,dbamv.estoque estoque_solic
      ,dbamv.setor
 where solsai_pro.cd_estoque = estoque.cd_estoque (+)
   AND solsai_pro.cd_estoque_solicitante = estoque_solic.cd_estoque (+)
   AND solsai_pro.cd_setor = setor.cd_setor(+)
   AND ESTOQUE.CD_MULTI_EMPRESA = Dbamv.Pkg_mv2000.le_empresa
   AND cd_requisicao
