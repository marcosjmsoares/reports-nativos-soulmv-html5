Select MVTO_ESTOQUE.CD_USUARIO,
       MVTO_ESTOQUE.CD_MVTO_ESTOQUE,
       MVTO_ESTOQUE.CD_SOLSAI_PRO,
       MVTO_ESTOQUE.CD_ESTOQUE As CD_ESTOQUE_ORIGEM,
       ESTOQUE_ORIGEM.DS_ESTOQUE As DS_ESTOQUE_ORIGEM,
       MVTO_ESTOQUE.CD_ESTOQUE_DESTINO,
       ESTOQUE_DESTINO.DS_ESTOQUE As DS_ESTOQUE_DESTINO,
       MVTO_ESTOQUE.DT_MVTO_ESTOQUE,
       MVTO_ESTOQUE.HR_MVTO_ESTOQUE,
       MVTO_ESTOQUE.DT_IMPRESSAO,
       MVTO_ESTOQUE.NR_DOCUMENTO,
       MVTO_ESTOQUE.DS_OBS_EMPRESTIMO As DS_OBSERVACAO,
       (Select MOTIVO_DIVERG_ATEND.DS_MOTIVO_DIVERG_ATEND
          From DBAMV.ITSOLSAI_PRO
    Inner Join DBAMV.MOTIVO_DIVERG_ATEND On
               MOTIVO_DIVERG_ATEND.CD_MOTIVO_DIVERG_ATEND = ITSOLSAI_PRO.CD_MOTIVO_DIVERG_ATEND
         Where ITSOLSAI_PRO.CD_ITSOLSAI_PRO = ITMVTO_ESTOQUE.CD_ITSOLSAI_PRO) As DS_MOTIVO_DIVERG_ATEND,
       ITMVTO_ESTOQUE.dsp_codigo_de_barras,
       ITMVTO_ESTOQUE.CD_PRODUTO,
       PRODUTO.DS_PRODUTO,
       ITMVTO_ESTOQUE.CD_PRODUTO_KIT As CD_PRODUTO_KIT,
       PRODUTO_KIT.DS_PRODUTO As DS_PRODUTO_KIT,
       UNI_PRO.DS_UNIDADE,
       ITMVTO_ESTOQUE.CD_LOTE,
       ITMVTO_ESTOQUE.DT_VALIDADE,
       Sum(ITMVTO_ESTOQUE.QT_MOVIMENTACAO) As QT_MOVIMENTACAO1,
       DBAMV.Verif_Vl_Custo_Medio(ITMVTO_ESTOQUE.CD_PRODUTO, MVTO_ESTOQUE.DT_MVTO_ESTOQUE, 'H', Null, MVTO_ESTOQUE.HR_MVTO_ESTOQUE) * UNI_PRO.VL_FATOR As VL_CUSTO_MEDIO
  From DBAMV.MVTO_ESTOQUE
Inner Join DBAMV.ITMVTO_ESTOQUE On
       ITMVTO_ESTOQUE.CD_MVTO_ESTOQUE = MVTO_ESTOQUE.CD_MVTO_ESTOQUE
Inner Join DBAMV.ESTOQUE ESTOQUE_ORIGEM On
       ESTOQUE_ORIGEM.CD_ESTOQUE = MVTO_ESTOQUE.CD_ESTOQUE
Inner Join DBAMV.ESTOQUE ESTOQUE_DESTINO On
       ESTOQUE_DESTINO.CD_ESTOQUE = MVTO_ESTOQUE.CD_ESTOQUE_DESTINO
Inner Join DBAMV.PRODUTO On
       PRODUTO.CD_PRODUTO = ITMVTO_ESTOQUE.CD_PRODUTO
Inner Join DBAMV.UNI_PRO On
       UNI_PRO.CD_UNI_PRO = ITMVTO_ESTOQUE.CD_UNI_PRO
Left Join DBAMV.PRODUTO PRODUTO_KIT On
       PRODUTO_KIT.CD_PRODUTO = ITMVTO_ESTOQUE.CD_PRODUTO_KIT
 Where MVTO_ESTOQUE.CD_MVTO_ESTOQUE = :CAMPO_CD_MVTO_ESTOQUE
   And MVTO_ESTOQUE.TP_MVTO_ESTOQUE = 'T'
Group By MVTO_ESTOQUE.CD_USUARIO,
       MVTO_ESTOQUE.CD_MVTO_ESTOQUE,
       MVTO_ESTOQUE.CD_SOLSAI_PRO,
       MVTO_ESTOQUE.CD_ESTOQUE,
       ESTOQUE_ORIGEM.DS_ESTOQUE,
       MVTO_ESTOQUE.CD_ESTOQUE_DESTINO,
       ESTOQUE_DESTINO.DS_ESTOQUE,
       MVTO_ESTOQUE.DT_MVTO_ESTOQUE,
       MVTO_ESTOQUE.HR_MVTO_ESTOQUE,
       MVTO_ESTOQUE.DT_IMPRESSAO,
       MVTO_ESTOQUE.NR_DOCUMENTO,
       MVTO_ESTOQUE.DS_OBS_EMPRESTIMO,
       ITMVTO_ESTOQUE.CD_ITSOLSAI_PRO,
       ITMVTO_ESTOQUE.dsp_codigo_de_barras,
       ITMVTO_ESTOQUE.CD_PRODUTO,
       PRODUTO.DS_PRODUTO,
       ITMVTO_ESTOQUE.CD_PRODUTO_KIT,
       PRODUTO_KIT.DS_PRODUTO,
       UNI_PRO.DS_UNIDADE,
       ITMVTO_ESTOQUE.CD_LOTE,
       ITMVTO_ESTOQUE.DT_VALIDADE,
       DBAMV.Verif_Vl_Custo_Medio(ITMVTO_ESTOQUE.CD_PRODUTO, MVTO_ESTOQUE.DT_MVTO_ESTOQUE, 'H', Null, MVTO_ESTOQUE.HR_MVTO_ESTOQUE) * UNI_PRO.VL_FATOR
Order By 2,
       4,
       5,
       7,
       6,
       1,
       8,
       9,
       3,
       10,
       11,
       18,
       16,
       15

--R_TRAPRO_R_3_Q_1
Select ISSP.CD_PRODUTO,
       P.DS_PRODUTO,
       UP.DS_UNIDADE,
       ISSP.QT_SOLICITADO As QTD_SOLICITADA,
       Nvl(ISSP.QT_ATENDIDA,
           0) As QTD_ATENDIDA,
       ISSP.QT_SOLICITADO - Nvl(ISSP.QT_ATENDIDA,
                                0) As QTD_DIFERENCA,
       P.CD_PRODUTO_TEM As DSP_CD_PRODUTO_TEM,
       P.SN_MESTRE As DSP_SN_MESTRE,
       ISSP.SN_CONF_DETERM_USU As CONF_USU,
       MDA.DS_MOTIVO_DIVERG_ATEND
  From DBAMV.ITSOLSAI_PRO ISSP
Inner Join DBAMV.PRODUTO P On
       ISSP.CD_PRODUTO = P.CD_PRODUTO
   And P.SN_KIT <> 'C'
Inner Join DBAMV.UNI_PRO UP On
       ISSP.CD_UNI_PRO = UP.CD_UNI_PRO
Inner Join DBAMV.SUB_CLAS SC On
       P.CD_ESPECIE = SC.CD_ESPECIE
   And P.CD_CLASSE = SC.CD_CLASSE
   And P.CD_SUB_CLA = SC.CD_SUB_CLA
   And SC.SN_PROD_PERMANENTE <> 'L'
Left Join DBAMV.MOTIVO_DIVERG_ATEND MDA On
       ISSP.CD_MOTIVO_DIVERG_ATEND = MDA.CD_MOTIVO_DIVERG_ATEND
 Where ISSP.CD_SOLSAI_PRO = :CAMPO_CD_SOLSAI_PRO
   And ISSP.QT_SOLICITADO <> Nvl(ISSP.QT_ATENDIDA,
                                 0)
   And Not Exists (Select 1
                     From DBAMV.ITMVTO_ESTOQUE IME
               Inner Join DBAMV.MVTO_ESTOQUE ME On
                          IME.CD_MVTO_ESTOQUE = ME.CD_MVTO_ESTOQUE
               Inner Join DBAMV.FORMULA F On
                          IME.CD_PRODUTO_KIT = F.CD_PRODUTO
                      And IME.CD_FORMULA_KIT = F.CD_FORMULA
                    Where IME.CD_PRODUTO_KIT = ISSP.CD_PRODUTO
                      And ME.CD_SOLSAI_PRO = ISSP.CD_SOLSAI_PRO)
Union All
Select IME.CD_PRODUTO,
       P.DS_PRODUTO,
       UP.DS_UNIDADE,
       0 As QTD_SOLICITADA,
       IME.QT_MOVIMENTACAO As QTD_ATENDIDA,
       -1 * IME.QT_MOVIMENTACAO As QTD_DIFERENCA,
       P.CD_PRODUTO_TEM As DSP_CD_PRODUTO_TEM,
       P.SN_MESTRE As DSP_SN_MESTRE,
       'N' As CONF_USU,
       Null As DS_MOTIVO_DIVERG_ATEND
  From DBAMV.ITMVTO_ESTOQUE IME
Inner Join DBAMV.MVTO_ESTOQUE ME On
       IME.CD_MVTO_ESTOQUE = ME.CD_MVTO_ESTOQUE
   And ME.CD_SOLSAI_PRO = :CAMPO_CD_SOLSAI_PRO
Inner Join DBAMV.PRODUTO P On
       IME.CD_PRODUTO = P.CD_PRODUTO
Inner Join DBAMV.UNI_PRO UP On
       IME.CD_UNI_PRO = UP.CD_UNI_PRO
 Where IME.CD_PRODUTO Not In (Select ISSP.CD_PRODUTO
                                From DBAMV.ITSOLSAI_PRO ISSP
                               Where ISSP.CD_SOLSAI_PRO = ME.CD_SOLSAI_PRO
                           Union All
                              Select P.CD_PRODUTO
                                From DBAMV.ITSOLSAI_PRO ISSP
                          Inner Join DBAMV.PRODUTO P On
                                     ISSP.CD_PRODUTO = P.CD_PRODUTO_TEM
                               Where ISSP.CD_SOLSAI_PRO = ME.CD_SOLSAI_PRO)
   And Not Exists (Select 1
                     From DBAMV.ITSOLSAI_PRO ISSP
                    Where ISSP.CD_PRODUTO = IME.CD_PRODUTO
                      And ISSP.CD_SOLSAI_PRO = ME.CD_SOLSAI_PRO)
Order By 1

--R_TRAPRO_R_PRODUTO_KIT_Q_MOV_ESTOQUE
SELECT
      ITMVTO_ESTOQUE_TRANSF.QT_MOVIMENTACAO_ATENDIDA,
       MVTO_ESTOQUE.CD_USUARIO,
       MVTO_ESTOQUE.CD_MVTO_ESTOQUE,
       MVTO_ESTOQUE.CD_SOLSAI_PRO,
       MVTO_ESTOQUE.CD_ESTOQUE       CD_ESTOQUE_ORIGEM,
       ESTOQUE_ORIGEM.DS_ESTOQUE  DS_ESTOQUE_ORIGEM ,
       MVTO_ESTOQUE.CD_ESTOQUE_DESTINO,
       ESTOQUE_DESTINO.DS_ESTOQUE   DS_ESTOQUE_DESTINO,
       MVTO_ESTOQUE.DT_MVTO_ESTOQUE,
       MVTO_ESTOQUE.HR_MVTO_ESTOQUE,
       MVTO_ESTOQUE.DT_IMPRESSAO,
       MVTO_ESTOQUE.NR_DOCUMENTO,
       MOTIVO_DIVERG_ATEND.DS_MOTIVO_DIVERG_ATEND,       
	   ITMVTO_ESTOQUE.dsp_codigo_de_barras,
       ITMVTO_ESTOQUE.CD_PRODUTO,
       PRODUTO.DS_PRODUTO,
       ITMVTO_ESTOQUE.CD_PRODUTO_KIT              CD_PRODUTO_KIT,
       PRODUTO_KIT.DS_PRODUTO                             DS_PRODUTO_KIT,
       UNI_PRO.DS_UNIDADE,
       ITMVTO_ESTOQUE.CD_LOTE,
       ITMVTO_ESTOQUE.DT_VALIDADE,
       ITMVTO_ESTOQUE.DS_ENDERECO,
       SUM(ITMVTO_ESTOQUE.QT_MOVIMENTACAO) QT_MOVIMENTACAO1,
       VERIF_VL_CUSTO_MEDIO(ITMVTO_ESTOQUE.CD_PRODUTO,MVTO_ESTOQUE.DT_MVTO_ESTOQUE,'H',NULL,MVTO_ESTOQUE.HR_MVTO_ESTOQUE) * UNI_PRO.VL_FATOR VL_CUSTO_MEDIO
FROM
       DBAMV.ITMVTO_ESTOQUE_TRANSF,
       DBAMV.MVTO_ESTOQUE    MVTO_ESTOQUE,
       DBAMV.ITMVTO_ESTOQUE  ITMVTO_ESTOQUE,
       DBAMV.ESTOQUE         ESTOQUE_ORIGEM,
       DBAMV.UNI_PRO         UNI_PRO,
       DBAMV.PRODUTO         PRODUTO,
       DBAMV.PRODUTO         PRODUTO_KIT,
       DBAMV.ESTOQUE         ESTOQUE_DESTINO,
       DBAMV.ITSOLSAI_PRO    ITSOLSAI_PRO, 
       DBAMV.MOTIVO_DIVERG_ATEND MOTIVO_DIVERG_ATEND 
WHERE
       MVTO_ESTOQUE.TP_MVTO_ESTOQUE    =  'T'
AND    MVTO_ESTOQUE.CD_MVTO_ESTOQUE    =  ITMVTO_ESTOQUE.CD_MVTO_ESTOQUE
AND    MVTO_ESTOQUE.CD_ESTOQUE         =  ESTOQUE_ORIGEM.CD_ESTOQUE
AND    MVTO_ESTOQUE.CD_ESTOQUE_DESTINO =  ESTOQUE_DESTINO.CD_ESTOQUE
AND    ITMVTO_ESTOQUE.CD_ITSOLSAI_PRO  =  ITSOLSAI_PRO.CD_ITSOLSAI_PRO(+) 
AND    ITSOLSAI_PRO.CD_MOTIVO_DIVERG_ATEND = MOTIVO_DIVERG_ATEND.CD_MOTIVO_DIVERG_ATEND(+)   
AND    ITMVTO_ESTOQUE.CD_UNI_PRO       =  UNI_PRO.CD_UNI_PRO(+)
AND    ITMVTO_ESTOQUE.CD_PRODUTO       =  PRODUTO.CD_PRODUTO(+)
AND    MVTO_ESTOQUE.CD_MVTO_ESTOQUE    =  :CAMPO_CD_MVTO_ESTOQUE 
AND    ITMVTO_ESTOQUE.CD_PRODUTO_KIT   =  PRODUTO_KIT.CD_PRODUTO(+)
AND    ITMVTO_ESTOQUE.CD_MVTO_ESTOQUE    =  ITMVTO_ESTOQUE_TRANSF.CD_MVTO_ESTOQUE(+)
AND    ITMVTO_ESTOQUE.CD_ITMVTO_ESTOQUE    =  ITMVTO_ESTOQUE_TRANSF.CD_ITMVTO_ESTOQUE(+)    
GROUP BY
       ITMVTO_ESTOQUE_TRANSF.QT_MOVIMENTACAO_ATENDIDA,
       MVTO_ESTOQUE.CD_USUARIO,
       MVTO_ESTOQUE.CD_MVTO_ESTOQUE,
       MVTO_ESTOQUE.CD_SOLSAI_PRO,                                                                  
       MVTO_ESTOQUE.CD_ESTOQUE,
       ESTOQUE_ORIGEM.DS_ESTOQUE,
       MVTO_ESTOQUE.CD_ESTOQUE_DESTINO,
       ESTOQUE_DESTINO.DS_ESTOQUE,
       MVTO_ESTOQUE.DT_MVTO_ESTOQUE,
       MVTO_ESTOQUE.HR_MVTO_ESTOQUE,
       MVTO_ESTOQUE.DT_IMPRESSAO,
       MVTO_ESTOQUE.NR_DOCUMENTO,
       MOTIVO_DIVERG_ATEND.DS_MOTIVO_DIVERG_ATEND,   
       ITMVTO_ESTOQUE.dsp_codigo_de_barras,
       ITMVTO_ESTOQUE.CD_PRODUTO,
       PRODUTO.DS_PRODUTO,
       ITMVTO_ESTOQUE.CD_PRODUTO_KIT,
       PRODUTO_KIT.DS_PRODUTO,
       UNI_PRO.DS_UNIDADE,
       ITMVTO_ESTOQUE.CD_LOTE,
       ITMVTO_ESTOQUE.DT_VALIDADE,
       ITMVTO_ESTOQUE.DS_ENDERECO,
       VERIF_VL_CUSTO_MEDIO( ITMVTO_ESTOQUE.CD_PRODUTO
                           , MVTO_ESTOQUE.DT_MVTO_ESTOQUE, 'H', NULL
                           , MVTO_ESTOQUE.HR_MVTO_ESTOQUE) * UNI_PRO.VL_FATOR
ORDER BY DS_ENDERECO, 
         CD_MVTO_ESTOQUE,
         CD_ESTOQUE_ORIGEM,
         DS_ESTOQUE_ORIGEM,
         DS_ESTOQUE_DESTINO,
         CD_ESTOQUE_DESTINO,
         CD_USUARIO,
         DT_MVTO_ESTOQUE,
         HR_MVTO_ESTOQUE,
         CD_SOLSAI_PRO,
         DT_IMPRESSAO,
         NR_DOCUMENTO,
         DS_PRODUTO,
         CD_PRODUTO
