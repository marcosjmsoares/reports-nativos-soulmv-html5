--R_LAB_X_DISTR
SELECT * FROM (
Select laborator.cd_laborator
	,laborator.nm_laborator
	,produto.cd_produto
	,produto.ds_produto
	,ent_pro.dt_emissao
	,fornecedor.cd_fornecedor
	,fornecedor.nm_fornecedor
	,ent_pro.cd_tip_doc
	,ent_pro.nr_documento
	,itent_pro.qt_entrada
	,itent_pro.vl_unitario
	,itent_pro.vl_desconto
	,itent_pro.vl_total
	,tip_doc.ds_tip_doc
	,Lab_Pro.Pc_Desc
	,Uni_Pro.Ds_Unidade
From   dbamv.ent_pro
	,dbamv.itent_pro
	,dbamv.fornecedor
	,dbamv.produto
	,dbamv.laborator
	,dbamv.lab_pro
	,dbamv.tip_doc
	,Dbamv.Uni_Pro
Where  Ent_Pro.Dt_Emissao Between :CAMPO_DTINIC and 
     and ent_pro.cd_ent_pro	= itent_pro.cd_ent_pro
     and produto.cd_produto	= itent_pro.cd_produto
     and lab_pro.cd_produto	= produto.cd_produto
     and lab_pro.cd_laborator	= laborator.cd_laborator
     and ent_pro.cd_fornecedor= fornecedor.cd_fornecedor
     and ent_pro.cd_tip_doc	= tip_doc.cd_tip_doc
     and ( :CAMPO_FORLAB = 'S' or Laborator.Cd_Fornecedor <> Ent_Pro.Cd_Fornecedor )
     and Uni_Pro.Cd_Uni_Pro	= Itent_Pro.Cd_Uni_Pro
     And Exists ( Select Empresa_Produto.Cd_Produto
                     From Dbamv.Empresa_Produto
                   Where Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
                        and Empresa_Produto.Cd_Produto = Produto.Cd_Produto )
     :CAMPO_CF_P_DETALHES
)
ORDER BY CD_LABORATOR,NM_LABORATOR
