      ,pro_fat.ds_pro_fat

      , 0 vl_saida
      , 0 qt_faturada
      , 0 vl_faturada
      , 0 qt_consumo_lcto
      , 0 vl_consumo_lcto
  from dbamv.pro_fat,
       dbamv.produto,
       dbamv.uni_pro,
       dbamv.itent_pro,
       dbamv.ent_pro,
       dbamv.estoque
 where   trunc(ent_pro.dt_entrada) between trunc(:CAMPO_DATA_INI) and Trunc(:CAMPO_DATA_FIM )
   AND        ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
   and pro_fat.cd_pro_fat    = produto.cd_pro_fat
   and itent_pro.cd_uni_pro  = uni_pro.cd_uni_pro
   and itent_pro.cd_ent_pro  = ent_pro.cd_ent_pro
   and ent_pro.cd_estoque  = estoque.cd_estoque
   and itent_pro.cd_produto = produto.cd_produto
) Tab
group by Tab.cd_setor
      ,Tab.cd_pro_fat
      ,Tab.ds_pro_fat
:CAMPO_CF_ORDEM
)
