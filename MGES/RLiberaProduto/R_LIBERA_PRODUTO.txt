--R_LIBERA_PRODUTO
SELECT
  LIBERACAO_PRODUTO.CD_LIBERACAO_PRODUTO,
  LIBERACAO_PRODUTO.USUARIO_LIBERACAO_PRODUTO,
  LIBERACAO_PRODUTO.DT_LIBERACAO_PRODUTO,
  Decode(TP_LIBERACAO_PRODUTO,'V','Vincular os produtos novos aos produtos antigos','Realizar a liberação ds produtos novos sem vincular aos produtos antigos') TIPO,
  ITLIBERACAO_PRODUTO.CD_PRODUTO_ANTIGO,
  (SELECT DS_PRODUTO FROM PRODUTO WHERE PRODUTO.CD_PRODUTO = ITLIBERACAO_PRODUTO.CD_PRODUTO_ANTIGO) DS_PRODUTO_ANTIGO,
  ITLIBERACAO_PRODUTO.CD_UNI_PRO_ANTIGO,
  (SELECT DS_UNIDADE FROM UNI_PRO WHERE UNI_PRO.CD_UNI_PRO = ITLIBERACAO_PRODUTO.CD_UNI_PRO_ANTIGO) DS_UNI_PRO_ANTIGO,
  ITLIBERACAO_PRODUTO.CD_PRODUTO,
  (SELECT DS_PRODUTO FROM PRODUTO WHERE PRODUTO.CD_PRODUTO = ITLIBERACAO_PRODUTO.CD_PRODUTO) DS_PRODUTO_NOVO,
  ITLIBERACAO_PRODUTO.CD_UNI_PRO,
  (SELECT DS_UNIDADE FROM UNI_PRO WHERE UNI_PRO.CD_UNI_PRO = ITLIBERACAO_PRODUTO.CD_UNI_PRO) DS_UNI_PRO_NOVO
FROM
  DBAMV.LIBERACAO_PRODUTO,
  DBAMV.ITLIBERACAO_PRODUTO
WHERE LIBERACAO_PRODUTO.CD_LIBERACAO_PRODUTO = ITLIBERACAO_PRODUTO.CD_LIBERACAO_PRODUTO
AND LIBERACAO_PRODUTO.CD_LIBERACAO_PRODUTO = :CAMPO_CD_LIBERACAO_PRODUTO
AND ITLIBERACAO_PRODUTO.SN_ATIVADO = 'S'
AND ITLIBERACAO_PRODUTO.SN_IMPRESSO = 'N'
ORDER BY ITLIBERACAO_PRODUTO.CD_PRODUTO_ANTIGO
