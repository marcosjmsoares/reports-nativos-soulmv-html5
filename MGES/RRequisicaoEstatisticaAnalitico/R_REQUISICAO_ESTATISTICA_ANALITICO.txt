--R_REQUISICAO_ESTATISTICA_ANALITICO
SELECT req.cd_estoque_dest cd_estoque_dest,
       est.ds_estoque ds_estoque,
       sol.cd_solsai_pro cd_solicitacao,
       to_char(Max(sol.hr_solsai_pro), 'DD/MM/YYYY HH24') dt_solicitacao,
       to_char(Max(tes.hr_mvto_estoque), 'DD/MM/YYYY HH24') dt_movimentacao,
       to_char(Max(tem.hr_mvto_estoque), 'DD/MM/YYYY HH24') dt_transferencia,
       to_char(Max(ent.hr_entrada), 'DD/MM/YYYY HH24') dt_entrada
FROM dbamv.requisicao_produto Req   
    ,dbamv.solsai_pro         Sol   
    ,dbamv.mvto_estoque       Tes   
    ,dbamv.mvto_estoque       Tem   
    ,dbamv.ent_pro            Ent   
    ,dbamv.estoque            Est   
WHERE req.cd_requisicao   = sol.cd_requisicao
  AND sol.cd_solsai_pro   = tes.cd_solsai_pro(+)
  AND tes.cd_mvto_estoque = tem.cd_movimentacao(+)
  AND tem.cd_ent_pro      = ent.cd_ent_pro(+)
  AND est.cd_estoque      = req.cd_estoque_dest
GROUP BY req.cd_estoque_dest,
         est.ds_estoque,
         sol.cd_solsai_pro
ORDER BY req.cd_estoque_dest,
         sol.cd_solsai_pro
