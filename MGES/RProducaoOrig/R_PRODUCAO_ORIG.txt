--R_PRODUCAO_ORIG
SELECT * FROM (
SELECT
         ENT_PRO.NR_DOCUMENTO                                            NR_DOCUMENTO_DE,
         ENT_PRO.DT_EMISSAO                                                    DT_EMISSAO_DE,
         NVL(ENT_PRO.VL_TOTAL, 0.00)                                      VL_TOTAL_DE,
         ENT_PRO.TP_DOCUMENTO_ENTRADA                         TP_DOCUMENTO_ENTRADA_DE,
         ENT_PRO.CD_ENT_PRO                                                 CD_ENT_PRO_EP,
         ENT_PRO.CD_ESTOQUE                                                 CD_ESTOQUE_EP,
         ESTOQUE.DS_ESTOQUE                                                 DS_ESTOQUE_EP,
         ENT_PRO.CD_TIP_DOC                                                    CD_TIP_ENT_EP,
         TIP_DOC.DS_TIP_DOC                                                     DS_TIP_ENT,
         ENT_PRO.DT_ENTRADA                                                 DT_ENTRADA_EP,
         ENT_PRO.HR_ENTRADA                                                HR_ENTRADA_EP,
         ITENT_PRO.CD_PRODUTO                                              CD_PRODUTO_IP,
         PRODUTO.DS_PRODUTO                                                 DS_PRODUTO_IP,
         ITENT_PRO.QT_ENTRADA                                              QT_ENTRADA_IP,
         ITENT_PRO.CD_UNI_PRO                                                CD_UNI_PRO_IP,
         UNI_PRO.DS_UNIDADE                                                    DS_UNIDADE_IP,
         ITENT_PRO.VL_TOTAL                                                     VL_TOTAL_IP,
         ITENT_PRO.VL_UNITARIO                                                VL_UNITARIO_IP,
        SETOR.NM_SETOR                                                              NM_SETOR,
        ITMVTO.CD_PRODUTO				CD_PRODUTO_ITMVTO,
        PROD_ITMVTO.DS_PRODUTO			DS_PRODUTO_ITMVTO,
        ITMVTO.CD_UNI_PRO				CD_UNI_PRO_ITMVTO,
        UNI_PRO_ITMVTO.DS_UNIDADE		DS_UNIDADE_ITMVTO,
        ITMVTO.QT_MOVIMENTACAO			QT_MOVIMENTACAO_ITMVTO,
        ITMVTO.CD_LOTE				CD_LOTE_ITMVTO,
        ITMVTO.DT_VALIDADE				DT_VALIDADE_ITMVTO
FROM	dbamv.ESTOQUE			ESTOQUE,
	dbamv.ENT_PRO			ENT_PRO,
	dbamv.ITENT_PRO		ITENT_PRO,
	dbamv.UNI_PRO			UNI_PRO,
	dbamv.PRODUTO			PRODUTO,
	dbamv.TIP_DOC			TIP_DOC,
	dbamv.SETOR			SETOR,
	dbamv.ITMVTO_ESTOQUE		ITMVTO,
	dbamv.UNI_PRO			UNI_PRO_ITMVTO,
	dbamv.PRODUTO			PROD_ITMVTO
WHERE   ENT_PRO.CD_TIP_DOC             = TIP_DOC.CD_TIP_DOC
AND        ESTOQUE.CD_MULTI_EMPRESA         =     :CAMPO_CD_MULTI_EMPRESA
AND        ENT_PRO.CD_SETOR_APL_MVTO_ESTOQUE  =  SETOR.CD_SETOR(+)
AND        ESTOQUE.CD_ESTOQUE		=     ENT_PRO.CD_ESTOQUE
AND        ITENT_PRO.CD_ENT_PRO		=     ENT_PRO.CD_ENT_PRO
AND        ITENT_PRO.CD_UNI_PRO		=     UNI_PRO.CD_UNI_PRO
AND        ITENT_PRO.CD_PRODUTO		=     PRODUTO.CD_PRODUTO
AND        ITMVTO.CD_ITENT_PRO		(+)=  ITENT_PRO.CD_ITENT_PRO
AND	UNI_PRO_ITMVTO.CD_UNI_PRO	(+)=  ITMVTO.CD_UNI_PRO
AND	PROD_ITMVTO.CD_PRODUTO	(+)=  ITMVTO.CD_PRODUTO
ORDER BY ENT_PRO.CD_ENT_PRO
)
ORDER BY CD_ESTOQUE_EP,DS_ESTOQUE_EP,DT_ENTRADA_EP,HR_ENTRADA_EP,CD_ENT_PRO_EP,VL_TOTAL_DE,TP_DOCUMENTO_ENTRADA_DE,NR_DOCUMENTO_DE,DS_TIP_ENT,CD_TIP_ENT_EP,DT_EMISSAO_DE,NM_SETOR,
CD_PRODUTO_IP,DS_PRODUTO_IP,CD_UNI_PRO_IP,DS_UNIDADE_IP,VL_TOTAL_IP,QT_ENTRADA_IP,VL_UNITARIO_IP
