--R_CONS_PAC_RESUMIDO
Select
           atendime.cd_atendimento
         , paciente.cd_paciente
         , paciente.nm_paciente
         , prestador.nm_prestador
         , mvto_estoque.dt_mvto_estoque
         , unid_int.ds_unid_int
         , itmvto_estoque.cd_produto
         , produto.ds_produto
         , uni_pro.ds_unidade
        , sum(itmvto_estoque.qt_movimentacao) qt_movimentacao
        , Avg( uni_pro.vl_fator * Verif_Vl_Custo_Medio( ItMvto_Estoque.Cd_Produto, Mvto_Estoque.Dt_Mvto_Estoque, :CAMPO_TP_CUSTO, produto.vl_custo_medio, Mvto_Estoque.Hr_Mvto_Estoque )) vl_unitario
	,solsai_pro.cd_solsai_pro
	,solsai_pro.cd_pre_med
 From
       dbamv.mvto_estoque
     , dbamv.itmvto_estoque
     , dbamv.paciente
     , dbamv.produto
     , dbamv.uni_pro
     , dbamv.atendime
     , dbamv.prestador
     , dbamv.unid_int
     , dbamv.estoque
     , dbamv.solsai_pro
 Where
   mvto_estoque.tp_mvto_estoque = 'P'
   AND mvto_estoque.dt_mvto_estoque BETWEEN  :CAMPO_DTINIC    AND  
   and estoque.cd_estoque = mvto_estoque.cd_estoque
   AND estoque.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
   and itmvto_estoque.cd_mvto_estoque = mvto_estoque.cd_mvto_estoque
   and prestador.cd_prestador         = mvto_estoque.cd_prestador
   and unid_int.cd_unid_int(+)           = mvto_estoque.cd_unid_int
   and produto.cd_produto             = itmvto_estoque.cd_produto
   and uni_pro.cd_uni_pro             = itmvto_estoque.cd_uni_pro
   AND atendime.cd_atendimento        = mvto_estoque.cd_atendimento
   AND paciente.cd_paciente           = atendime.cd_paciente
   and atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
   AND solsai_pro.cd_solsai_pro (+)   = mvto_estoque.cd_solsai_pro
group by
           atendime.cd_atendimento
         , paciente.cd_paciente
         , paciente.nm_paciente
         , prestador.nm_prestador
         , mvto_estoque.dt_mvto_estoque
         , unid_int.ds_unid_int
         , itmvto_estoque.cd_produto
         , produto.ds_produto
         , uni_pro.ds_unidade
         ,solsai_pro.cd_solsai_pro
	       ,solsai_pro.cd_pre_med
