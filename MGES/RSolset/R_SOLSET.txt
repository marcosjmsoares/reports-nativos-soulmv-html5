--R_SOLSET
Select Hospital.Ds_Razao, Hospital.Ds_Rodape_Reports,
 'N' Sn_Imprime_Logotipo, Hospital.Logotipo
  From Dbamv.Hospital
 Where Hospital.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA

--R_SOLSET_X_G_MATRIX
Select A.Cd_Setor
	,A.Nm_Setor
	,A.Horario
	,Count( A.Cd_Setor ) Qtd_Solicitacao
  From (
   Select SolSai_Pro.Cd_Setor
	,Setor.Nm_Setor
	,Trunc( Trunc( (SolSai_Pro.Hr_SolSai_Pro - Trunc( SolSai_Pro.Hr_SolSai_Pro )) * 24 ) /  :CAMPO_QTHORAS ) Horario
      From Dbamv.Setor
	,Dbamv.SolSai_Pro
    Where SolSai_Pro.Dt_SolSai_Pro Between :CAMPO_DTINI and 
        and Trunc( (SolSai_Pro.Hr_SolSai_Pro - Trunc( SolSai_Pro.Hr_SolSai_Pro )) * 24 ) Between :CAMPO_HRINI and 
        and Setor.Cd_Setor = SolSai_Pro.Cd_Setor
        and setor.cd_multi_empresa = pkg_mv2000.le_empresa
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 0 / :CAMPO_QTHORAS )  Horario From Dual
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 1 / :CAMPO_QTHORAS )  Horario From Dual
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 2 / :CAMPO_QTHORAS )  Horario From Dual
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 3 / :CAMPO_QTHORAS )  Horario From Dual
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 4 / :CAMPO_QTHORAS )  Horario From Dual
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 5 / :CAMPO_QTHORAS )  Horario From Dual
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 6 / :CAMPO_QTHORAS )  Horario From Dual
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 7 / :CAMPO_QTHORAS )  Horario From Dual
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 8 / :CAMPO_QTHORAS )  Horario From Dual
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 9 / :CAMPO_QTHORAS )  Horario From Dual
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 10 / :CAMPO_QTHORAS ) Horario From Dual
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 11 / :CAMPO_QTHORAS ) Horario From Dual
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 12 / :CAMPO_QTHORAS ) Horario From Dual
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 13 / :CAMPO_QTHORAS ) Horario From Dual
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 14 / :CAMPO_QTHORAS ) Horario From Dual
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 15 / :CAMPO_QTHORAS ) Horario From Dual
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 16 / :CAMPO_QTHORAS ) Horario From Dual
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 17 / :CAMPO_QTHORAS ) Horario From Dual
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 18 / :CAMPO_QTHORAS ) Horario From Dual
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 19 / :CAMPO_QTHORAS ) Horario From Dual
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 20 / :CAMPO_QTHORAS ) Horario From Dual
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 21 / :CAMPO_QTHORAS ) Horario From Dual
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 22 / :CAMPO_QTHORAS ) Horario From Dual
    Union All Select To_Number(Null) Cd_Setor, Null Nm_Setor, trunc( 23 / :CAMPO_QTHORAS ) Horario From Dual
) A
Group By A.Horario
	,A.Cd_Setor
	,A.Nm_Setor
ORDER BY A.HORARIO,
A.NM_SETOR,A.CD_SETOR
