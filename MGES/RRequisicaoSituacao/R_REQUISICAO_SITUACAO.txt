--R_REQUISICAO_SITUACAO
SELECT DISTINCT
      requisicao.cd_estoque_dest cd_estoque_dest,
      estoque.ds_estoque ds_estoque,
      requisicao.cd_requisicao cd_requisicao,
      To_Char(requisicao.dt_requisicao, 'DD/MM/YYYY HH24') dt_requisicao,
      Decode(requisicao.tp_status, 'A','ABERTA',
                        'L', 'LIBERADA',
                        'C', 'CANCELADA',
                        'T', 'ABERTA PARCIALMENTE ATENDIDA',
                        'P', 'LIBERADA PARCIAL',
                        'E', 'ATENDIDA',
                        'R', 'ATENDIDA PARCIAL',
                        'N', 'ENVIADA PARA REGIONAL',
                        'V', 'ENVIADA PARCIALMENTE',
                        'S', 'RECEBIDA NA REGIONAL',
                        'B', 'RECEBIDA PARCIALMENTE') tp_status
FROM dbamv.requisicao_produto requisicao,
     dbamv.estoque estoque
WHERE requisicao.cd_estoque_dest = estoque.cd_estoque
ORDER BY requisicao.cd_estoque_dest, requisicao.cd_requisicao
