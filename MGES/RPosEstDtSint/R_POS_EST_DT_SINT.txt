--R_POS_EST_DT_SINT
SELECT * FROM (
SELECT
               estoque.cd_estoque cd_estoque,
               estoque.ds_estoque ds_estoque,
               produto.cd_produto cd_produto,
               produto.ds_produto ds_produto,
               especie.cd_especie cd_especie,
               especie.ds_especie ds_especie,
               Verif_Ds_Unid_Prod( Produto.Cd_Produto, 'G' ) Ds_Unidade,
               Verif_Estoque( Estoque.Cd_Estoque, Produto.Cd_Produto, :CAMPO_DATA, :CAMPO_DATA )
		/ verif_vl_fator_prod(produto.cd_produto)  qt_estoque,
               verif_vl_custo_medio(produto.cd_produto, :CAMPO_DATA, :CAMPO_TP_CUSTO, produto.vl_custo_medio, :CAMPO_DATA )
		* verif_vl_fator_prod ( produto.cd_produto ) vl_unitario
  FROM	dbamv.estoque	estoque,
	dbamv.est_pro	est_pro,
	dbamv.especie	especie,
	dbamv.produto	produto
WHERE  estoque.cd_estoque	= est_pro.cd_estoque
      AND  produto.cd_produto		= est_pro.cd_produto
      AND  Produto.Sn_Mestre		= 'N'
      AND  Especie.Cd_especie	= Produto.Cd_Especie
      AND  estoque.cd_multi_empresa	= :CAMPO_CD_MULTI_EMPRESA
:CAMPO_CF_P_DETALHES
)
ORDER BY CD_ESTOQUE,DS_ESTOQUE,
CD_ESPECIE,DS_ESPECIE
