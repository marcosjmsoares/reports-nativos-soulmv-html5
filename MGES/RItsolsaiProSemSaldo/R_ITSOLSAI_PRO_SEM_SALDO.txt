--R_ITSOLSAI_PRO_SEM_SALDO
SELECT * FROM (
select  estoque.cd_estoque,
           estoque.ds_estoque,
           to_date (to_char (solsai_pro.dt_solsai_pro, 'dd/mm/yyyy') || to_char(solsai_pro.hr_solsai_pro, 'hh24'), 'dd/mm/yyyy hh24') dt_solsai_pro,
           solsai_pro.cd_setor,
           solsai_pro.cd_estoque_solicitante,
           produto.ds_produto,
           uni_pro.ds_unidade,
           solsai_pro.cd_usuario,
           itsolsai_pro_sem_saldo.cd_solsai_pro,
           itsolsai_pro_sem_saldo.cd_produto,
           itsolsai_pro_sem_saldo.qt_estoque,
           itsolsai_pro_sem_saldo.qt_cota,
           setor.nm_setor,
           produto.sn_mestre,
           produto.cd_produto_tem,
           solsai_pro.cd_unid_int,
           unid_int.ds_unid_int
   from dbamv.itsolsai_pro_sem_saldo,
           dbamv.solsai_pro,
           dbamv.estoque,
           dbamv.produto,
           dbamv.uni_pro,
           dbamv.setor,
           dbamv.unid_int
   where itsolsai_pro_sem_saldo.cd_solsai_pro = :CAMPO_CD_SOLSAI_PRO
      and solsai_pro.cd_estoque   = estoque.cd_estoque
      and solsai_pro.cd_solsai_pro = itsolsai_pro_sem_saldo.cd_solsai_pro
      and estoque.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
      and produto.cd_produto = itsolsai_pro_sem_saldo.cd_produto
      and itsolsai_pro_sem_saldo.cd_uni_pro = uni_pro.cd_uni_pro
      and solsai_pro.cd_setor = setor.cd_setor (+)
      and solsai_pro.cd_unid_int = unid_int.cd_unid_int (+)
)
ORDER BY CD_ESTOQUE,DS_ESTOQUE,CD_SETOR,NM_SETOR,CD_SOLSAI_PRO,DT_SOLSAI_PRO,CD_USUARIO,CD_ESTOQUE_SOLICITANTE,CD_UNID_INT,DS_UNID_INT
