--R_CLAS_PROD_ESPECIE
SELECT * FROM (
SELECT especie.cd_especie,
       especie.ds_especie,
       classe.cd_classe,
       classe.ds_classe,
       sub_clas.cd_sub_cla,
       sub_clas.ds_sub_cla,
       decode (:CAMPO_ORDEM, '1', produto.cd_produto, null ) cd_ordena_cd,
       decode (:CAMPO_ORDEM, '2', produto.ds_produto, null ) cd_ordena_des,
       produto.cd_produto codigo,
       produto.ds_produto descricao,
       produto.dt_cadastro,
       produto.tp_ativo,
       produto.dt_ultima_entrada,
       produto.qt_ultima_entrada,
       produto.vl_ultima_entrada,
       DECODE(NVL(DBAMV.PKT_CONFIGEST.RETORNA_CAMPO('SN_LOTE_VAL_PRD_EMP'),'N'),'S',produto.sn_controle_validade, produto.sn_controle_validade) sn_controle_validade,
       produto.sn_pscotropico,
       verif_ds_unid_prod( produto.cd_produto ) ds_unidade_pai,
       produto.vl_custo_medio * verif_vl_fator_prod( produto.cd_produto ) vl_custo_medio_pai,
       produto.vl_margem_lucro,
       produto.vl_preco_de_venda,
       produto.vl_tempo_de_reposicao,
       produto.sn_mestre,
       produto.sn_movimentacao,
       produto.hr_ultima_entrada,
       produto.sn_medicamento,
       produto.tp_classificacao_xyz,
       produto.sn_bloqueio_de_compra,
       produto.cd_tip_ativ,
       DECODE(NVL(DBAMV.PKT_CONFIGEST.RETORNA_CAMPO('SN_LOTE_VAL_PRD_EMP'),'N'),'S',produto.SN_LOTE, produto.SN_LOTE) sn_lote,
       produto.sn_imprime_etiqueta,
       produto_filho.cd_produto            codigo_filho,
       produto_filho.ds_produto            descricao_filho,
       produto_filho.dt_cadastro           dt_cadastro_filho,
       produto_filho.tp_ativo              tp_ativo_filho,
       produto_filho.dt_ultima_entrada     dt_ultima_entrada_filho,
       produto_filho.qt_ultima_entrada     qt_ultima_entrada_filho,
       produto_filho.vl_ultima_entrada     vl_ultima_entrada_filho,
       DECODE(NVL(DBAMV.PKT_CONFIGEST.RETORNA_CAMPO('SN_LOTE_VAL_PRD_EMP'),'N'),'S',produto_filho.sn_controle_validade, produto_filho.sn_controle_validade) sn_controle_validade_filho,
       produto_filho.sn_pscotropico        sn_pscotropico_filho,
       produto_filho.vl_custo_medio * verif_vl_fator_prod( produto_filho.cd_produto )       vl_custo_medio_filho,
       produto_filho.vl_margem_lucro       vl_margem_lucro_filho,
       produto_filho.vl_preco_de_venda     vl_preco_de_venda_filho,
       produto_filho.vl_tempo_de_reposicao vl_tempo_de_reposicao_filho,
       produto_filho.sn_mestre             sn_mestre_filho,
       produto_filho.sn_movimentacao       sn_movimentacao_filho,
       produto_filho.hr_ultima_entrada     sn_ultima_entrada_filho,
       produto_filho.sn_medicamento        sn_medicamento_filho,
       produto_filho.tp_classificacao_xyz  tp_classificacao_xyz_filho,
       produto_filho.sn_bloqueio_de_compra sn_bloqueio_de_compra_filho,
       produto_filho.cd_tip_ativ           cd_tip_ativ_filho,
       DECODE(NVL(DBAMV.PKT_CONFIGEST.RETORNA_CAMPO('SN_LOTE_VAL_PRD_EMP'),'N'),'S',produto_filho.SN_LOTE, produto_filho.SN_LOTE) sn_lote_filho,
       produto_filho.sn_imprime_etiqueta   sn_imprime_etiqueta_filho,
       produto.sn_padronizado              sn_padronizado,
       produto.sn_admitido 	           sn_admitido,
       produto.sn_consignado		   sn_consignado,
       produto.cd_pro_fat		   cd_pro_fat
FROM   dbamv.especie,
       dbamv.classe,
       dbamv.sub_clas,
       (select
			p.cd_especie, p.cd_classe, p.cd_sub_cla, p.cd_produto,p.sn_mestre, p.dt_cadastro,
			p.sn_imprime_etiqueta, p.sn_bloqueio_de_compra, p.sn_movimentacao, p.ds_produto, p.tp_ativo,
			p.sn_pscotropico, p.sn_lote,p.cd_produto_tem, p.sn_medicamento,p.dt_ultima_entrada,
			p.qt_ultima_entrada,p.vl_ultima_entrada,p.vl_custo_medio,p.vl_margem_lucro,p.vl_preco_de_venda,
			p.vl_tempo_de_reposicao,p.hr_ultima_entrada,p.tp_classificacao_xyz,p.cd_tip_ativ,p.sn_controle_validade,p.sn_kit,p.sn_padronizado,p.sn_admitido,p.sn_consignado,p.cd_pro_fat
		from
		dbamv.produto p,
		dbamv.empresa_produto ep
		where
		p.cd_produto = ep.cd_produto) produto,
       (select * from (
	    select
			p.cd_especie, p.cd_classe, p.cd_sub_cla, p.cd_produto,p.sn_mestre, p.dt_cadastro,
			p.sn_imprime_etiqueta, p.sn_bloqueio_de_compra, p.sn_movimentacao, p.ds_produto, p.tp_ativo,
			p.sn_pscotropico, p.sn_lote,p.cd_produto_tem, p.sn_medicamento,p.dt_ultima_entrada,
			p.qt_ultima_entrada,p.vl_ultima_entrada,p.vl_custo_medio,p.vl_margem_lucro,p.vl_preco_de_venda,
			p.vl_tempo_de_reposicao,p.hr_ultima_entrada,p.tp_classificacao_xyz,p.cd_tip_ativ,p.sn_controle_validade,p.sn_kit,p.sn_padronizado,p.sn_admitido,p.sn_consignado,p.cd_pro_fat
		from
		dbamv.produto p,
		dbamv.empresa_produto ep
		where
		p.cd_produto = ep.cd_produto
		AND    ep.cd_multi_empresa = pkg_mv2000.le_empresa) produto where 1=1 :CAMPO_CF_P_DETALHES) produto_filho
WHERE  especie.cd_especie      = classe.cd_especie
AND    classe.cd_especie       = sub_clas.cd_especie
AND    classe.cd_classe        = sub_clas.cd_classe
AND    sub_clas.cd_especie     = produto.cd_especie
AND    sub_clas.cd_classe      = produto.cd_classe
AND    sub_clas.cd_sub_cla     = produto.cd_sub_cla
AND    produto.cd_produto      = produto_filho.cd_produto_tem(+)
AND    produto.sn_kit not in ('C' , 'E')
:CAMPO_CF_ORDEM
)
ORDER BY CD_ESPECIE,DS_ESPECIE,
CD_CLASSE,DS_CLASSE,
CD_SUB_CLA,DS_SUB_CLA,
CD_ORDENA_CD,CD_ORDENA_DES,CODIGO,DESCRICAO,DT_CADASTRO,DT_ULTIMA_ENTRADA,QT_ULTIMA_ENTRADA,VL_ULTIMA_ENTRADA,HR_ULTIMA_ENTRADA,VL_MARGEM_LUCRO,VL_PRECO_DE_VENDA,VL_TEMPO_DE_REPOSICAO,TP_CLASSIFICACAO_XYZ,SN_BLOQUEIO_DE_COMPRA,SN_PSCOTROPICO,SN_MESTRE,SN_MEDICAMENTO,SN_IMPRIME_ETIQUETA,CD_TIP_ATIV,TP_ATIVO,DS_UNIDADE_PAI,VL_CUSTO_MEDIO_PAI,SN_MOVIMENTACAO
