--R_BALANCO_CONTABIL_PROD
SELECT * FROM (
SELECT    CD_MULTI_EMPRESA,
DS_ESPECIE,
     CD_PRODUTO,
     DS_PRODUTO,
     SUM( nvl(QT_ESTOQUE_INICIAL,0) )			QT_ESTOQUE_INICIAL,
     SUM( nvl(QT_ESTOQUE_INICIAL_SEM_TRANSF,0) )	QT_ESTOQUE_INICIAL_SEM_TRANSF,
     SUM( nvl(QT_ESTOQUE_FINAL,0) )			QT_ESTOQUE_FINAL,
     SUM( nvl(QT_ESTOQUE_FINAL_SEM_TRANSF,0) )		QT_ESTOQUE_FINAL_SEM_TRANSF,
     SUM( nvl(QT_ENTRADA,0) )				QT_ENTRADA,
     SUM( nvl(QT_ENTRADA_TRANSFERIDO,0) )		QT_ENTRADA_TRANSFERIDO,
     SUM( nvl(QT_SAIDA_SETOR,0) )			QT_SAIDA_SETOR,
     SUM( nvl(QT_SAIDA_PACIENTE,0) )			QT_SAIDA_PACIENTE,
     SUM( nvl(QT_BAIXA,0) )				QT_BAIXA,
     SUM( nvl(QT_VENDA,0) )				QT_VENDA,
     SUM( nvl(QT_EMPRESTIMO,0) )			QT_EMPRESTIMO,
     SUM( nvl(QT_EMPRESTIMO_entrada,0) )			QT_EMPRESTIMO_entrada,
     SUM(nvl(QT_ENT_EMPRESTIMO_RECEB,0) )		QT_ENT_EMPRESTIMO_RECEB,
     SUM(nvl(QT_SAI_PAG_EMPRESTIMO,0) )		QT_SAI_PAG_EMPRESTIMO,
     SUM(nvl(QT_ENT_PAG_EMPRESTIMO,0) )		QT_ENT_PAG_EMPRESTIMO,
     SUM( nvl(QT_SAI_EMPRESTIMO_CON,0) )		QT_SAI_EMPRESTIMO_CON,
     SUM( nvl(QT_DEVOLUCAO_SETOR,0) )			QT_DEVOLUCAO_SETOR,
     SUM( nvl(QT_DEVOLUCAO_PACIENTE,0) )		QT_DEVOLUCAO_PACIENTE,
     SUM( nvl(QT_DEVOLUCAO_VENDA,0) )			QT_DEVOLUCAO_VENDA,
     SUM( nvl(QT_DEMANDA,0) )				QT_DEMANDA,
     SUM( nvl(QT_TRANSFERENCIA_SAIDA,0) )		QT_TRANSFERENCIA_SAIDA,
     SUM( nvl(QT_DOACAO,0) )				QT_DOACAO,
     SUM( nvl(QT_BONIFICACAO,0) )			QT_BONIFICACAO,
     SUM( nvl(QT_MANIPULADA_ENTRADA,0) )		QT_MANIPULADA_ENTRADA,
     SUM( nvl(QT_MANIPULADA_SAIDA,0) )			QT_MANIPULADA_SAIDA,
     SUM( Decode('','M',qt_contagem_entrada,Decode( Sign( QT_CONTAGEM_ENTRADA - QT_CONTAGEM_SAIDA ), 1, QT_CONTAGEM_ENTRADA - QT_CONTAGEM_SAIDA, 0 ) ))			QT_CONTAGEM_ENTRADA,
     SUM(Decode('','M',qt_contagem_saida, Decode( Sign( QT_CONTAGEM_SAIDA - QT_CONTAGEM_ENTRADA ), 1, QT_CONTAGEM_SAIDA - QT_CONTAGEM_ENTRADA, 0 ) ))			QT_CONTAGEM_SAIDA,
     SUM( nvl(QT_DEVOLVIDA_FORNEC,0) )			QT_DEVOLVIDA_FORNEC,
     SUM( nvl(QT_VALE,0) )				QT_VALE,
     SUM( nvl(QT_TRANSF_EMP_SAI,0) )			QT_TRANSF_EMP_SAI,
     SUM( nvl(QT_TRANSF_EMP_ENT,0) )			QT_TRANSF_EMP_ENT,
     SUM( nvl(QT_NF_NAO_COBRADA,0))                                        QT_NF_NAO_COBRADA,
     SUM( Decode( Sign( nvl(QT_ENTRADA_AJUSTE,0) - nvl(QT_SAIDA_AJUSTE,0) ), 1, nvl(QT_ENTRADA_AJUSTE,0) - nvl(QT_SAIDA_AJUSTE,0), 0 ) ) QT_ENTRADA_AJUSTE,
     SUM( Decode( Sign( nvl(QT_SAIDA_AJUSTE,0) - nvl(QT_ENTRADA_AJUSTE,0) ), 1, nvl(QT_SAIDA_AJUSTE,0) - nvl(QT_ENTRADA_AJUSTE,0), 0 ) ) QT_SAIDA_AJUSTE,
	 SUM( nvl(QT_ACERTO_DE_ESTOQUE,0))                                        QT_ACERTO_DE_ESTOQUE
FROM(
SELECT   ESTOQUE.CD_MULTI_EMPRESA,
     C_CONEST.CD_PRODUTO,ESPECIE.DS_ESPECIE,
     PRODUTO.DS_PRODUTO,
     SUM(decode(produto.sn_consignado, 'S', C_CONEST.vl_ESTOQUE_INICIAL_consig, C_CONEST.vl_ESTOQUE_INICIAL))                                 QT_ESTOQUE_INICIAL,
     SUM(decode(produto.sn_consignado, 'S', C_CONEST.vl_ESTOQUE_INICIAL_consig, C_CONEST.vl_ESTOQUE_INICIAL_SEM_TRANSF))       QT_ESTOQUE_INICIAL_SEM_TRANSF,
     SUM(decode(produto.sn_consignado, 'S', C_CONEST.vl_ESTOQUE_FINAL_consig, C_CONEST.vl_ESTOQUE_FINAL))                                      QT_ESTOQUE_FINAL,
     SUM(decode(produto.sn_consignado, 'S', C_CONEST.vl_ESTOQUE_FINAL_consig, C_CONEST.vl_ESTOQUE_FINAL_SEM_TRANSF))	          QT_ESTOQUE_FINAL_SEM_TRANSF,
     SUM(C_CONEST.vl_ENTRADA)		 	                                                                                                                                          QT_ENTRADA,
     SUM(decode(produto.sn_consignado, 'S', 0, C_CONEST.vl_ENTRADA_TRANSFERIDO))	                                                                          QT_ENTRADA_TRANSFERIDO,
     SUM(C_CONEST.vl_SAIDA_SETOR)		                                                                                                                                          QT_SAIDA_SETOR,
     SUM(C_CONEST.vl_SAIDA_PACIENTE)		                                                                                                                                          QT_SAIDA_PACIENTE,
     SUM(decode(produto.sn_consignado, 'S', 0, C_CONEST.vl_BAIXA)) 			                                                                          QT_BAIXA,
     SUM(decode(produto.sn_consignado, 'S', 0, C_CONEST.vl_VENDA))			                                                                          QT_VENDA,
     SUM(decode(produto.sn_consignado, 'S', 0, C_CONEST.vl_EMPRESTIMO))			                                                                          QT_EMPRESTIMO,
     SUM(decode(produto.sn_consignado, 'S', 0, C_CONEST.vl_EMPRESTIMO_entrada))                	                                                                          QT_EMPRESTIMO_entrada,
     SUM(decode(produto.sn_consignado, 'S', 0, C_CONEST.vl_entrada_emprestimo_recebido))                                                                                     QT_ENT_EMPRESTIMO_RECEB,
     SUM(decode(produto.sn_consignado, 'S', 0, C_CONEST.vl_saida_pag_emprestimo))                                                                                                 QT_SAI_PAG_EMPRESTIMO,
     SUM(decode(produto.sn_consignado, 'S', 0, C_CONEST.vl_entrada_pag_emprestimo))                                                                                             QT_ENT_PAG_EMPRESTIMO,
     SUM(decode(produto.sn_consignado, 'S', 0, C_CONEST.vl_saida_emprestimo_concedido))                                                                                      QT_SAI_EMPRESTIMO_CON,
     SUM(C_CONEST.vl_DEVOLUCAO_SETOR)		                                                                                                                                          QT_DEVOLUCAO_SETOR,
     SUM(C_CONEST.vl_DEVOLUCAO_PACIENTE)	                                                                                                                                          QT_DEVOLUCAO_PACIENTE,
     SUM(decode(produto.sn_consignado, 'S', 0, C_CONEST.vl_DEVOLUCAO_venda))	                                                                                          QT_DEVOLUCAO_VENDA,
     SUM(decode(produto.sn_consignado, 'S', 0, C_CONEST.vl_DEMANDA))			                                                                          QT_DEMANDA,
     SUM(decode(produto.sn_consignado, 'S', 0, C_CONEST.vl_TRANSFERENCIA_SAIDA))	                                                                          QT_TRANSFERENCIA_SAIDA,
     SUM(nvl(C_CONEST.vl_DOACAO,0))			                                                                          QT_DOACAO,
     SUM(NVL(vl_BONIFICACAO,0))			                                                                          QT_BONIFICACAO,
     SUM(decode(produto.sn_consignado, 'S', 0, C_CONEST.vl_MANIPULADA_ENTRADA))	                                                                          QT_MANIPULADA_ENTRADA,
     SUM(decode(produto.sn_consignado, 'S', 0, C_CONEST.vl_MANIPULADA_SAIDA))		                                                                          QT_MANIPULADA_SAIDA,
     NVL(SUM(decode(produto.sn_consignado, 'S', 0, C_CONEST.vl_CONTAGEM_ENTRADA)),0)		                                                                          QT_CONTAGEM_ENTRADA,
     NVL(SUM(decode(produto.sn_consignado, 'S', 0, C_CONEST.vl_CONTAGEM_SAIDA)),0)		                                                                          QT_CONTAGEM_SAIDA,
     SUM(decode(produto.sn_consignado, 'S', 0, NVL(c_conest.vl_nota_fiscal_nao_cobrada,0)))                                                                                   QT_NF_NAO_COBRADA,
    Sum(Nvl(C_CONEST.VL_ACERTO_ENTRADA, 0)) As QT_ACERTO_DE_ESTOQUE,
     SUM(C_CONEST.vl_DEVOLVIDA_FORNEC)	 	                                                                                                                                          QT_DEVOLVIDA_FORNEC,
     SUM(nvl(C_CONEST.vl_VALE,0))		                                                                                                                                                          QT_VALE,
     SUM(nvl(c_conest.vl_transf_emp_sai,0))                                                                                                                                                                     QT_TRANSF_EMP_SAI,
     SUM(nvl(c_conest.vl_transf_emp_ent,0))                                                                                                                                                                    QT_TRANSF_EMP_ENT,
   SUM(decode ( dbamv.fnc_mges_retorna_processo ( to_date(C_CONEST.CD_MES
      ||'/'
      || :CAMPO_DT_ANO ,'MM/YYYY') ),'S', NVL(C_CONEST.VL_entrada_AJUSTE_CONTABIL,0),
               decode(:CAMPO_ESTOQUE,'%',nvl(c_conest.vl_entrada_ajuste_contabil_sem,0), NVL(C_CONEST.VL_ENTRADA_AJUSTE_CONTABIL,0)))	 	                                 )                                                QT_ENTRADA_AJUSTE,
     SUM( decode ( dbamv.fnc_mges_retorna_processo ( to_date(C_CONEST.CD_MES
      ||'/'
      || :CAMPO_DT_ANO ,'MM/YYYY') ),'S',NVL(C_CONEST.VL_SAIDA_AJUSTE_CONTABIL,0),
              decode(:CAMPO_ESTOQUE,'%',nvl(c_conest.vl_saida_ajuste_contabil_sem,0), NVL(C_CONEST.VL_SAIDA_AJUSTE_CONTABIL,0)))	) 	                                                                                                                          QT_SAIDA_AJUSTE
FROM DBAMV.C_CONEST,
     DBAMV.PRODUTO,
     DBAMV.ESPECIE,
     DBAMV.ESTOQUE
     ,DBAMV.TIP_ATIV
WHERE C_CONEST.CD_PRODUTO = PRODUTO.CD_PRODUTO
  AND ESPECIE.CD_ESPECIE  = PRODUTO.CD_ESPECIE
  AND  PRODUTO.SN_MESTRE           = 'N'
  AND ESTOQUE.CD_ESTOQUE   = C_CONEST.CD_ESTOQUE
  AND C_CONEST.CD_MES     BETWEEN :CAMPO_DT_INICIO AND 
  AND C_CONEST.CD_ANO     = :CAMPO_DT_ANO
  AND PRODUTO.CD_TIP_ATIV = TIP_ATIV.CD_TIP_ATIV(+)
GROUP BY ESTOQUE.CD_MULTI_EMPRESA,C_CONEST.CD_PRODUTO, ESPECIE.DS_ESPECIE, DS_PRODUTO
)
Group By CD_MULTI_EMPRESA, DS_ESPECIE,      CD_PRODUTO,
     DS_PRODUTO
ORDER BY DS_ESPECIE
)
ORDER BY CD_MULTI_EMPRESA, DS_ESPECIE,
CD_PRODUTO,DS_PRODUTO,QT_ESTOQUE_INICIAL,QT_ESTOQUE_INICIAL_SEM_TRANSF,QT_ESTOQUE_FINAL,QT_ESTOQUE_FINAL_SEM_TRANSF
