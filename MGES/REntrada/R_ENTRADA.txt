Select Nvl(ENT_PRO.NR_DOCUMENTO,
           0) As NR_DOCUMENTO1,
       ENT_PRO.DT_ENTRADA,
       ESPECIE.DS_ESPECIE,
       Sum(Nvl(DBAMV.converte_moeda_entrada(Nvl(:CAMPO_CONVERTE,
                                                '0'),
                                            Nvl(ITENT_PRO.VL_TOTAL_CUSTO_REAL,
                                                0),
                                            Decode(MULTI_EMPRESAS.SN_UTILIZA_MOEDA,
                                                   'S',
                                                   Nvl(ITENT_PRO.VL_TOTAL_CUSTO_REAL,
                                                       0) / Nvl(ENT_PRO_MOEDA.VL_INDICE_ENT_PRO,
                                                                1),
                                                   Nvl(ITENT_PRO.VL_TOTAL_CUSTO_REAL,
                                                       0)),
                                            :CAMPO_CD_MOEDA_IMPRESSAO,
                                            ENT_PRO_MOEDA.CD_MOEDA_INDICE,
                                            ENT_PRO_MOEDA.SN_CAMBIO_SUGERIDO,
                                            Nvl(ENT_PRO_MOEDA.DT_INDICE_MOEDA,
                                                ENT_PRO.DT_EMISSAO),
                                            ENT_PRO_MOEDA.VL_INDICE_ENT_PRO),
               0)) As TOT_ESPECIE
  From DBAMV.ENT_PRO
Inner Join DBAMV.ITENT_PRO On
       ITENT_PRO.CD_ENT_PRO = ENT_PRO.CD_ENT_PRO
Inner Join DBAMV.PRODUTO On
       PRODUTO.CD_PRODUTO = ITENT_PRO.CD_PRODUTO
Inner Join DBAMV.ESPECIE On
       ESPECIE.CD_ESPECIE = PRODUTO.CD_ESPECIE
Inner Join DBAMV.ESTOQUE On
       ESTOQUE.CD_ESTOQUE = ENT_PRO.CD_ESTOQUE
Inner Join DBAMV.MULTI_EMPRESAS On
       MULTI_EMPRESAS.CD_MULTI_EMPRESA = ESTOQUE.CD_MULTI_EMPRESA
Left Join DBAMV.ENT_PRO_MOEDA On
       ENT_PRO_MOEDA.CD_ENT_PRO = ENT_PRO.CD_ENT_PRO
Left Join DBAMV.ENT_PRO_ACERTO On
       ENT_PRO_ACERTO.CD_ENT_PRO = ENT_PRO.CD_ENT_PRO
 Where ENT_PRO.CD_ESTOQUE In (Select ESTOQUE.CD_ESTOQUE
                                From DBAMV.ESTOQUE
                               :CAMPO_CF_EMPRESAS_SUB_SELECTS)
   And ENT_PRO.DT_ENTRADA = :CAMPO_DT_ENTRADA2
   And (Nvl(ENT_PRO.NR_DOCUMENTO,
            '*.*') = Nvl(:CAMPO_NR_DOCUMENTO,
                         '*.*')
     Or Nvl(To_Char(ENT_PRO_ACERTO.CD_ENT_PRO_ACERTO),
            '*.*') =  Nvl(:CAMPO_NR_DOCUMENTO,
                          '*.*'))
   And ENT_PRO.CD_FORNECEDOR = :CAMPO_CD_FORNECEDOR
   And Nvl(ENT_PRO.NR_SERIE,
           '*.*') = Nvl(:CAMPO_NR_SERIE,
                        '*.*')
Group By ENT_PRO.NR_DOCUMENTO,
       ENT_PRO.DT_ENTRADA,
       ESPECIE.DS_ESPECIE
