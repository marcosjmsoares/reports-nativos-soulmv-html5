--R_HIST_PRODUTO_SETOR
Select *
From (Select CD_PRODUTO, DS_PRODUTO, CD_SETOR, NM_SETOR, DBAMV.Verif_Ds_Unid_Prod(CD_PRODUTO) As DS_UNIDADE,
      DT_COMPETENCIA As DT_COMPETENCIA_PRODUTO, Sum(QT_ENTRADA) / DBAMV.Verif_Vl_Fator_Prod(CD_PRODUTO) As QT_ENTRADA_PRODUTO,
      Sum(VL_ENTRADA) As VL_ENTRADA_PRODUTO, Sum(QT_SAIDA) / DBAMV.Verif_Vl_Fator_Prod(CD_PRODUTO) As QT_SAIDA_PRODUTO,
      Sum(VL_SAIDA) As VL_SAIDA_PRODUTO
      From (Select PRODUTO.CD_PRODUTO, PRODUTO.DS_PRODUTO, SETOR.CD_SETOR,
            Nvl(SETOR.NM_SETOR, '') As NM_SETOR, To_Date(To_Char(MVTO_ESTOQUE.DT_MVTO_ESTOQUE,
            'MM/YYYY'), 'MM/YYYY') As DT_COMPETENCIA, 0 As QT_ENTRADA, 0 As VL_ENTRADA, Decode(MVTO_ESTOQUE.TP_MVTO_ESTOQUE, 'D', -1,
            'C', -1, 'N', -1, 1) * ITMVTO_ESTOQUE.QT_MOVIMENTACAO * UNI_PRO.VL_FATOR As QT_SAIDA,
            Decode(MVTO_ESTOQUE.TP_MVTO_ESTOQUE, 'D', -1, 'C', -1, 'N', -1, 1) * ITMVTO_ESTOQUE.QT_MOVIMENTACAO * UNI_PRO.VL_FATOR *
            DBAMV.Verif_Vl_Custo_Medio(ITMVTO_ESTOQUE.CD_PRODUTO, MVTO_ESTOQUE.DT_MVTO_ESTOQUE, 'R', Null,
            MVTO_ESTOQUE.HR_MVTO_ESTOQUE) As VL_SAIDA
            From (Select EMPRESA_PRODUTO.CD_MULTI_EMPRESA, Decode(DBAMV.PKT_CONFIGEST.Retorna_Campo('SN_PADRONIZADO_EMP'), 'S',
                  EMPRESA_PRODUTO.SN_PADRONIZADO, PRODUTO.SN_PADRONIZADO) As SN_PADRONIZADO, PRODUTO.CD_PRO_FAT,
                  Decode(DBAMV.PKT_CONFIGEST.RetornarControleLoteValEmpPrd(), 'S', EMPRESA_PRODUTO.SN_MOVIMENTACAO,
                  PRODUTO.SN_MOVIMENTACAO) As SN_MOVIMENTACAO, PRODUTO.SN_CONSIGNADO,
                  Decode(DBAMV.PKT_CONFIGEST.RetornarControleLoteValEmpPrd(), 'S', EMPRESA_PRODUTO.SN_BLOQUEIO_DE_COMPRA,
                  PRODUTO.SN_BLOQUEIO_DE_COMPRA) As SN_BLOQUEIO_DE_COMPRA, PRODUTO.SN_ADMITIDO, PRODUTO.CD_PRODUTO, PRODUTO.DS_PRODUTO
                  From DBAMV.EMPRESA_PRODUTO, DBAMV.PRODUTO
                  Where EMPRESA_PRODUTO.CD_PRODUTO = PRODUTO.CD_PRODUTO And EMPRESA_PRODUTO.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
                     :CAMPO_CF_SUB_CLASSE) PRODUTO,
            DBAMV.UNI_PRO, DBAMV.ITMVTO_ESTOQUE, DBAMV.MVTO_ESTOQUE, DBAMV.ESTOQUE, DBAMV.SETOR
            Where MVTO_ESTOQUE.TP_MVTO_ESTOQUE In ('S', 'P', 'D', 'C', 'N') And ITMVTO_ESTOQUE.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO And
            ITMVTO_ESTOQUE.CD_MVTO_ESTOQUE = MVTO_ESTOQUE.CD_MVTO_ESTOQUE And ESTOQUE.CD_ESTOQUE = MVTO_ESTOQUE.CD_ESTOQUE And
            MVTO_ESTOQUE.DT_MVTO_ESTOQUE Between :CAMPO_DATA_INI And Last_Day(:CAMPO_DATA_FIM) + 0.99999 And
            ESTOQUE.CD_MULTI_EMPRESA = PRODUTO.CD_MULTI_EMPRESA And ITMVTO_ESTOQUE.CD_PRODUTO = PRODUTO.CD_PRODUTO And
            MVTO_ESTOQUE.CD_SETOR = SETOR.CD_SETOR(+) And (:CAMPO_SN_CONSID_TOD_MOV_SAI_CNSG = 'S' Or
            (:CAMPO_SN_CONSID_TOD_MOV_SAI_CNSG = 'N' And (PRODUTO.SN_CONSIGNADO In ('N', 'R') Or
            :CAMPO_S_TIPO_PROCESSO_CONSIGNADO = 'F')))   
            Union All
            Select LOG_MVTO_CONSIGNADO.CD_PRODUTO, PRODUTO.DS_PRODUTO, LOG_MVTO_CONSIGNADO.CD_SETOR, Nvl(SETOR.NM_SETOR,
            '') As NM_SETOR, To_Date(To_Char(LMCP.DT_ENTRADA, 'MM/YYYY'), 'MM/YYYY') As DT_COMPETENCIA,
            0 As QT_ENTRADA, 0 As VL_ENTRADA, LOG_MVTO_CONSIGNADO.QT_MOVIMENTO As QT_SAIDA, LOG_MVTO_CONSIGNADO.QT_MOVIMENTO *
            LOG_MVTO_CONSIGNADO.VL_UNITARIO As VL_SAIDA
            From (Select EMPRESA_PRODUTO.CD_MULTI_EMPRESA, Decode(DBAMV.PKT_CONFIGEST.Retorna_Campo('SN_PADRONIZADO_EMP'), 'S',
                  EMPRESA_PRODUTO.SN_PADRONIZADO, PRODUTO.SN_PADRONIZADO) As SN_PADRONIZADO, PRODUTO.CD_PRO_FAT,
                  Decode(DBAMV.PKT_CONFIGEST.RetornarControleLoteValEmpPrd(), 'S', EMPRESA_PRODUTO.SN_MOVIMENTACAO,
                  PRODUTO.SN_MOVIMENTACAO) As SN_MOVIMENTACAO, PRODUTO.SN_CONSIGNADO,
                  Decode(DBAMV.PKT_CONFIGEST.RetornarControleLoteValEmpPrd(), 'S', EMPRESA_PRODUTO.SN_BLOQUEIO_DE_COMPRA,
                  PRODUTO.SN_BLOQUEIO_DE_COMPRA) As SN_BLOQUEIO_DE_COMPRA, PRODUTO.SN_ADMITIDO, PRODUTO.CD_PRODUTO, PRODUTO.DS_PRODUTO
                  From DBAMV.EMPRESA_PRODUTO, DBAMV.PRODUTO
                  Where EMPRESA_PRODUTO.CD_PRODUTO = PRODUTO.CD_PRODUTO And EMPRESA_PRODUTO.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
                     :CAMPO_CF_SUB_CLASSE) PRODUTO,
            DBAMV.SETOR, DBAMV.ESTOQUE,
                 (Select LMC2.CD_LOG_MVTO_CONSIGNADO, EP.DT_ENTRADA From DBAMV.ENT_PRO EP, DBAMV.LOG_MVTO_CONSIGNADO LMC2 Where
                  LMC2.CD_ENT_PRO = EP.CD_ENT_PRO And LMC2.TP_MOVIMENTO = 'E' And EP.DT_ENTRADA Between :CAMPO_DATA_INI And
                  Last_Day(:CAMPO_DATA_FIM) + 0.99999) LMCP,
            DBAMV.LOG_MVTO_CONSIGNADO
            Where LOG_MVTO_CONSIGNADO.CD_PRODUTO = PRODUTO.CD_PRODUTO And LOG_MVTO_CONSIGNADO.CD_SETOR = SETOR.CD_SETOR(+) And
            LOG_MVTO_CONSIGNADO.CD_LOG_MVTO_CONSIGNADO_TEM = LMCP.CD_LOG_MVTO_CONSIGNADO And LOG_MVTO_CONSIGNADO.TP_MOVIMENTO = 'S' And
            LOG_MVTO_CONSIGNADO.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE And ESTOQUE.CD_MULTI_EMPRESA = PRODUTO.CD_MULTI_EMPRESA And
            :CAMPO_SN_CONSID_TOD_MOV_SAI_CNSG = 'N' And :CAMPO_S_TIPO_PROCESSO_CONSIGNADO = 'N'
            Union All
            Select PRODUTO.CD_PRODUTO, PRODUTO.DS_PRODUTO, SETOR.CD_SETOR, SETOR.NM_SETOR, To_Date(To_Char(ENT_PRO.DT_ENTRADA,
            'MM/YYYY'), 'MM/YYYY') As DT_COMPETENCIA, ITENT_PRO.QT_ENTRADA * UNI_PRO.VL_FATOR As QT_ENTRADA, ITENT_PRO.QT_ENTRADA *
            ITENT_PRO.VL_CUSTO_REAL As VL_ENTRADA, 0 As QT_SAIDA, 0 As VL_SAIDA
            From (Select EMPRESA_PRODUTO.CD_MULTI_EMPRESA, Decode(DBAMV.PKT_CONFIGEST.Retorna_Campo('SN_PADRONIZADO_EMP'), 'S',
                  EMPRESA_PRODUTO.SN_PADRONIZADO, PRODUTO.SN_PADRONIZADO) As SN_PADRONIZADO, PRODUTO.CD_PRO_FAT,
                  Decode(DBAMV.PKT_CONFIGEST.RetornarControleLoteValEmpPrd(), 'S', EMPRESA_PRODUTO.SN_MOVIMENTACAO,
                  PRODUTO.SN_MOVIMENTACAO) As SN_MOVIMENTACAO, PRODUTO.SN_CONSIGNADO,
                  Decode(DBAMV.PKT_CONFIGEST.RetornarControleLoteValEmpPrd(), 'S', EMPRESA_PRODUTO.SN_BLOQUEIO_DE_COMPRA,
                  PRODUTO.SN_BLOQUEIO_DE_COMPRA) As SN_BLOQUEIO_DE_COMPRA, PRODUTO.SN_ADMITIDO, PRODUTO.CD_PRODUTO, PRODUTO.DS_PRODUTO
                  From DBAMV.EMPRESA_PRODUTO, DBAMV.PRODUTO
                  Where EMPRESA_PRODUTO.CD_PRODUTO = PRODUTO.CD_PRODUTO And EMPRESA_PRODUTO.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
                     :CAMPO_CF_SUB_CLASSE) PRODUTO,
            DBAMV.UNI_PRO, DBAMV.ITENT_PRO, DBAMV.ENT_PRO, DBAMV.ESTOQUE, DBAMV.SETOR
            Where ITENT_PRO.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO And ITENT_PRO.CD_ENT_PRO = ENT_PRO.CD_ENT_PRO And
            ENT_PRO.DT_ENTRADA Between :CAMPO_DATA_INI And Last_Day(:CAMPO_DATA_FIM) + 0.99999 And
            ESTOQUE.CD_MULTI_EMPRESA = PRODUTO.CD_MULTI_EMPRESA And ITENT_PRO.CD_PRODUTO = PRODUTO.CD_PRODUTO And
            ENT_PRO.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE And ESTOQUE.CD_SETOR = SETOR.CD_SETOR And ENT_PRO.TP_DOCUMENTO_ENTRADA <> 'C'
              :CAMPO_CF_P_DETALHES)
      Group By CD_PRODUTO, DS_PRODUTO, CD_SETOR, NM_SETOR, DBAMV.Verif_Ds_Unid_Prod(CD_PRODUTO), DT_COMPETENCIA)
Order By DS_PRODUTO, NM_SETOR, DT_COMPETENCIA_PRODUTO
