--R_ENT_SEM_OC_JUSTIFICA
SELECT * FROM (
select justifica.cd_justifica,
       justifica.ds_justifica,
       ent_pro.cd_fornecedor,
       fornecedor.nm_fornecedor,
       tip_doc.ds_tip_doc,
       ent_pro.nr_documento,
       ent_pro.nr_serie,
       ent_pro.dt_emissao,
       ent_pro.vl_total,
       ent_pro.dt_entrada
from dbamv.fornecedor,
       dbamv.tip_doc,
       dbamv.ent_pro,
       dbamv.justifica
where  fornecedor.tp_cliente_forn in ( 'F', 'A', 'R', 'T' )
  and  ent_pro.cd_ord_com is null
  and  ent_pro.cd_justifica = justifica.cd_justifica
  and  justifica.tp_justifica in ('A', 'E')
  and  ent_pro.cd_fornecedor = fornecedor.cd_fornecedor
  and  ent_pro.cd_tip_doc = tip_doc.cd_tip_doc
  AND  TRUNC(ent_pro.dt_entrada)  >=  TRUNC(:CAMPO_DATA_INI)
  AND  TRUNC(ent_pro.dt_entrada)  <= TRUNC(:CAMPO_DATA_FIM)
  and Ent_Pro.Cd_Estoque in ( Select Estoque.Cd_Estoque
			            From Dbamv.Estoque
                             Where Estoque.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA )
:CAMPO_CF_FORNECEDOR
)
ORDER BY DS_JUSTIFICA,CD_JUSTIFICA,
DT_ENTRADA
