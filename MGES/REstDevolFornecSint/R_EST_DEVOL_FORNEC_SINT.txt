--R_EST_DEVOL_FORNEC_SINT
SELECT * FROM (
select fornecedor.cd_fornecedor,
       fornecedor.nm_fornecedor,
       tip_doc.cd_tip_doc,
       tip_doc.ds_tip_doc,
       dev_for.nr_documento,
       dev_for.nr_serie,
       dev_for.dt_emissao,
       dev_for.vl_total
from dbamv.fornecedor,
       dbamv.ent_pro,
       dbamv.tip_doc,
       dbamv.dev_for
where Fornecedor.Tp_Cliente_Forn In  ( 'F', 'A', 'R', 'T' )
  and  ent_pro.cd_fornecedor = fornecedor.cd_fornecedor
  and dev_for.cd_ent_pro = ent_pro.cd_ent_pro
  and ent_pro.cd_tip_doc = tip_doc.cd_tip_doc
  And Ent_Pro.Cd_Estoque in ( Select Estoque.Cd_Estoque
			            From Dbamv.Estoque
                            where estoque.cd_multi_empresa = Pkg_Mv2000.le_empresa)
:CAMPO_CF_TP_ORDEM
)
ORDER BY CD_FORNECEDOR,NM_FORNECEDOR
