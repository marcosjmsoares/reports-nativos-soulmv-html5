--R_SALDO_KITS
Select Est.Cd_estoque    	     Cd_Estoque
          ,Est.Ds_Estoque                 Ds_Estoque
          ,Est_Pro.Cd_Produto          Cd_Produto
          ,Prd.Ds_Produto                 Ds_Produto
          ,Nvl(Est_Pro.qt_estoque_atual,0) / dbamv.verif_vl_fator_prod(lot_pro.cd_produto)    Qt_estoque_atual
          ,Sum(Nvl(Lot_pro.qt_kit,0) / dbamv.verif_vl_fator_prod(lot_pro.cd_produto))              Qt_kit
          ,Nvl(Est_Pro.qt_estoque_reservado,0) / dbamv.verif_vl_fator_prod(lot_pro.cd_produto) Qt_estoque_reservado
          ,((Nvl(Est_Pro.qt_estoque_atual,0) / dbamv.verif_vl_fator_prod(lot_pro.cd_produto) - Sum(Nvl(Lot_pro.qt_kit,0)/dbamv.verif_vl_fator_prod(lot_pro.cd_produto)) ) - Nvl(Est_Pro.qt_estoque_reservado,0)/dbamv.verif_vl_fator_prod(lot_pro.cd_produto) ) Qtd_disponivel
          ,dbamv.verif_ds_unid_prod(lot_pro.cd_produto) ds_unidade
 From dbamv.Estoque Est
        ,dbamv.Est_Pro Est_Pro
        ,dbamv.Produto Prd
       ,dbamv.Lot_Pro Lot_Pro
Where Est.cd_Multi_Empresa  = dbamv.pkg_mv2000.le_empresa
     And Est.cd_estoque           = Est_pro.cd_estoque
     And Lot_Pro.Cd_Produto    = Prd.Cd_Produto
     And Lot_Pro.cd_produto	= Est_Pro.cd_produto
     And Lot_Pro.cd_estoque   = Est_Pro.cd_estoque
		 and lot_pro.qt_kit > 0
Group by Est.Cd_estoque
              ,Est.Ds_Estoque
              ,Est_Pro.Cd_Produto
              ,Prd.Ds_Produto
              ,Nvl(Est_Pro.qt_estoque_atual,0)/dbamv.verif_vl_fator_prod(lot_pro.cd_produto)
              ,Nvl(Est_Pro.qt_estoque_reservado,0)/dbamv.verif_vl_fator_prod(lot_pro.cd_produto)
              ,dbamv.verif_ds_unid_prod(lot_pro.cd_produto)
order by   Cd_Produto, Ds_Produto
