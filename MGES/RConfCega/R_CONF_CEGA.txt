--R_CONF_CEGA
SELECT ce.cd_conferencia_entrada                   AS cd_conferencia_entrada
     , ce.nr_serie                                 AS nr_serie
     , ce.nr_nota_fiscal                           AS nr_nota_fiscal
     , ce.dt_emissao                               AS dt_emissao
     , ce.dh_entrada                               AS dh_entrada
     , ce.cd_estoque                               AS cd_estoque
     , e.ds_estoque                                AS ds_estoque
     , ce.cd_fornecedor                            AS cd_fornecedor
     , f.nm_fornecedor                             AS nm_fornecedor
     , it.cd_produto                               AS cd_produto
     , p.sn_lote                                   AS sn_lote
     , p.sn_controle_validade                      AS sn_controle_validade
     , p.ds_produto                                AS ds_produto
     , it.cd_uni_pro                               AS cd_uni_pro
     , u.ds_unidade                                AS ds_unidade
     , it.cd_lote                                  AS cd_lote
     , it.dt_validade                              AS dt_validade
     , it.ds_marca                                 AS ds_marca
  FROM dbamv.conferencia_entrada ce
     , dbamv.item_conferencia_entrada it
     , dbamv.estoque e
     , dbamv.fornecedor f
     , dbamv.produto p
     , dbamv.uni_pro u
 WHERE ce.cd_conferencia_entrada = it.cd_conferencia_entrada
   AND   ce.cd_estoque = e.cd_estoque
   AND   f.cd_fornecedor = ce.cd_fornecedor
   AND   p.cd_produto = it.cd_produto
   AND   u.cd_uni_pro = it.cd_uni_pro
 GROUP BY ce.cd_conferencia_entrada
        , ce.nr_serie
        , ce.nr_nota_fiscal
        , ce.dt_emissao
        , ce.dh_entrada
        , ce.cd_estoque
        , e.ds_estoque
        , ce.cd_fornecedor
        , f.nm_fornecedor
        , it.cd_produto
        , p.sn_lote
        , p.sn_controle_validade
        , p.ds_produto
        , it.cd_uni_pro
        , u.ds_unidade
        , it.cd_lote
        , it.dt_validade
        , it.ds_marca
 ORDER BY ce.cd_conferencia_entrada
        , ce.nr_serie
        , ce.nr_nota_fiscal
        , ce.dt_emissao
        , ce.dh_entrada
        , ce.cd_estoque
        , ce.cd_fornecedor
        , it.cd_produto
        , it.cd_uni_pro
        , it.cd_lote
        , it.dt_validade asc
