--R_DEVFOR
SELECT * FROM (
select rownum
      ,dev_for.cd_devolucao cd_devolucao
      ,dev_for.cd_ent_pro cd_ent_pro
      ,dev_for.cd_mot_dev cd_mot_dev
      ,mot_dev.ds_mot_dev ds_mot_dev
      ,dev_for.dt_devolucao dt_devolucao
      ,dev_for.hr_devolucao hr_devolucao
      ,dev_for.nr_documento nr_documento
      ,dev_for.nr_serie nr_serie
      ,dev_for.dt_emissao dt_emissao
      ,dev_for.vl_total vl_total
      ,dev_for.tp_devolucao tp_devolucao
      ,itdev_for.cd_produto cd_produto
      ,produto.ds_produto ||' - '|| nvl(itdev_for.cd_lote,'Não Controla Lote') || ' - ' || to_char( itdev_for.dt_validade , 'DD/MM/YYYY') ds_prod
      ,itdev_for.qt_devolvida qt_devolvida
      ,itdev_for.cd_lote cd_lote
      ,uni_pro.ds_unidade ds_unidade
      ,ent_pro.dt_entrada dt_entrada
      ,ent_pro.cd_ent_pro cd_ent_pro2
      ,ent_pro.nr_serie nr_serie2
      ,ent_pro.nr_documento nr_documento2
  ,itent_pro.vl_custo_real
  ,(itent_pro.vl_custo_real * itdev_for.qt_devolvida)  vl_total_item
      ,fornecedor.cd_fornecedor cd_fornecedor
      ,fornecedor.nm_fantasia nm_fantasia
      ,fornecedor.nm_fornecedor nm_fornecedor
      ,fornecedor.ds_endereco ds_endereco
      ,fornecedor.nr_cgc_cpf nr_cgc_cpf
      ,fornecedor.ds_endereco_email ds_endereco_email
      ,fornecedor.nr_cep nr_cep
      ,cidade.nm_cidade nm_cidade
      ,cidade.cd_uf cd_uf
  from dbamv.dev_for
      ,dbamv.ent_pro
      ,dbamv.itdev_for
      ,dbamv.produto
      ,dbamv.mot_dev
      ,dbamv.uni_pro
      ,dbamv.fornecedor
      ,dbamv.cidade,
  dbamv.itent_pro
 where dev_for.cd_devolucao     = itdev_for.cd_devolucao
   and produto.cd_produto       = itdev_for.cd_produto
   and mot_dev.cd_mot_dev       = dev_for.cd_mot_dev
   and uni_pro.cd_uni_pro       = itdev_for.cd_uni_pro
   and ent_pro.cd_ent_pro       = dev_for.cd_ent_pro
and itent_pro.cd_itent_pro   = itdev_for.cd_itent_pro
and itent_pro.cd_produto = itdev_for.cd_produto
and itent_pro.cd_ent_pro = dev_for.cd_ent_pro
   and fornecedor.cd_fornecedor = ent_pro.cd_fornecedor
   and fornecedor.cd_cidade     = cidade.cd_cidade
   and dev_for.cd_devolucao     = :CAMPO_CD_DEVOLUCAO
   and Ent_Pro.Cd_Estoque in ( Select Estoque.Cd_Estoque
			            From Dbamv.Estoque
                             Where Estoque.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA )
)
ORDER BY rownum,CD_DEVOLUCAO,CD_MOT_DEV,DS_MOT_DEV,CD_ENT_PRO2,DT_DEVOLUCAO,CD_ENT_PRO,VL_TOTAL,NR_CGC_CPF,HR_DEVOLUCAO,NR_SERIE2,TP_DEVOLUCAO,DT_ENTRADA,DS_ENDERECO_EMAIL,DS_ENDERECO,CD_FORNECEDOR,NR_DOCUMENTO2,NM_FANTASIA,NR_CEP,NM_FORNECEDOR,NR_DOCUMENTO,NM_CIDADE,NR_SERIE,DT_EMISSAO,CD_UF
