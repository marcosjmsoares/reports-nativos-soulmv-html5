--R_CONSIGNADO_ENT
SELECT * FROM (
Select Ent_pro.cd_ent_pro                                                               cd_ent_pro,
           Ent_Pro.nr_documento                                                          nr_documento,
	   USUARIOS.NM_USUARIO                                                          NM_USUARIO,
           Ent_Pro.nr_serie                                                                     nr_serie,
           Ent_pro.cd_estoque                                                      cd_estoque,
           Estoque.ds_estoque                                                               ds_estoque,
           Ent_pro.cd_setor_apl_mvto_estoque                         cd_setor_apl_mvto_estoque,
           Setor.nm_setor                                                                        nm_setor,
           Ent_pro.cd_fornecedor                                                cd_fornecedor,
           Fornecedor.nm_fornecedor                                                     nm_fornecedor,
           to_char(Ent_pro.dt_entrada, 'dd/mm/yyyy') || ' ' || to_char(Ent_pro.hr_entrada,'hh24') data_entrada,
           Itent_pro.cd_produto                                                              cd_produto,
           Produto.ds_produto                                                                ds_produto,
           Uni_pro.ds_unidade                                                                ds_unidade,
           Nvl(Itlot_ent.qt_ent_pro,Itent_pro.qt_entrada)       qt_entrada,
           Itent_pro.vl_unitario                                                                 vl_unitario,
          Nvl(itlot_ent.qt_ent_pro* Itent_pro.vl_unitario,  Itent_pro.vl_total ) vl_total,
           Itlot_ent.cd_lote                                                                       cd_lote,
           Itlot_ent.dt_validade                                                                dt_validade,
           Itlot_ent.qt_ent_pro                                                                  qt_ent_pro,
           ENT_PRO.ds_processo																	ds_processo,             
       	   ENT_PRO.ds_empenho 																	 ds_empenho,               
           ENT_PRO.qt_parcelas_licitacao 											  qt_parcelas_licitacao	
  From Dbamv.Ent_pro                                                                        Ent_pro,
           Dbamv.Estoque                                                                       Estoque,
           Dbamv.Setor                                                                            Setor,
           Dbamv.Fornecedor                                                                  Fornecedor,
           Dbamv.Itent_pro                                                                      Itent_pro,
           Dbamv.Produto                                                                        Produto,
           Dbamv.Uni_pro                                                                        Uni_pro,
           Dbamv.Itlot_ent                                                                        Itlot_ent, DBASGU.USUARIOS
 Where
            Estoque.Cd_Multi_Empresa                                                  = :CAMPO_CD_MULTI_EMPRESA
     And Ent_pro.cd_estoque                                                             = Estoque.cd_estoque
     And Ent_pro.cd_setor_apl_mvto_estoque                                   = Setor.cd_setor(+)
     And Ent_pro.cd_fornecedor                                                         = Fornecedor.cd_fornecedor(+)
     And Ent_pro.cd_ent_pro                                                              = Itent_pro.cd_ent_pro
     And Itent_pro.cd_produto                                                            = Produto.cd_produto
     And Itent_pro.cd_uni_pro                                                            = Uni_pro.cd_uni_pro
     And Itent_pro.cd_itent_pro                                                          = Itlot_ent.cd_itent_pro(+)
     AND USUARIOS.CD_USUARIO                                                        = ENT_PRO.CD_USUARIO
Order by Ent_pro.cd_estoque,
              Estoque.ds_estoque,
              Produto.ds_produto,
              Itlot_ent.cd_lote
)
ORDER BY CD_ENT_PRO,NR_DOCUMENTO,NR_SERIE,CD_ESTOQUE,DS_ESTOQUE,CD_SETOR_APL_MVTO_ESTOQUE,NM_SETOR,CD_FORNECEDOR,NM_FORNECEDOR,DATA_ENTRADA, DS_PRODUTO
