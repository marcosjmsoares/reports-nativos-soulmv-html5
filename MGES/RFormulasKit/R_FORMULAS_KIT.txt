--R_FORMULAS_KIT
SELECT * FROM (
			Select formula.cd_formula                                               formula,
			           formula.tp_formula                                                tp_formula,
			           formula.cd_cirurgia                                               cirurgia,
			           cirurgia.ds_cirurgia                                               ds_cirurgia,
			           formula.cd_prestador                                            prestador,
			           prestador.nm_prestador                                        nm_prestador,
			           formula.cd_produto                                               prod_form,
			           prod_form.ds_produto                                           ds_prod_form,
			           prod_form.ds_especificacao                                 ds_especificacao,
			           uni_pro_form.ds_unidade                                      ds_unidade_form,
			           formula.qt_padrao                                                  qt_padrao_form,
			           formula.nr_dias_valido                                           nr_dias_valido_form,
			           formula.ds_indicacao                                             ds_indicacao_form,
			           itformula.cd_produto_tem                                       cd_produto_tem_itform,
			           prod_itform.ds_produto                                           ds_produto_itform,
			           uni_pro_itform.ds_unidade                                      ds_unidade_itform,
			           decode (prod_itform.sn_consignado, 'N', null, 'S', 'Consg') sn_consignado,
			           itformula.qt_padrao                                                 qt_padrao_itform,
			           nvl(formula.vl_administrativo,0) vl_administrativo,
			           Decode(itformula.cd_produto_tem,null,0,1) cont
			   from dbamv.formula                          formula,
			           dbamv.itformula                     itformula,
			           dbamv.prestador                     prestador,
			           dbamv.cirurgia                      cirurgia,
			           dbamv.produto                       prod_form,
			           dbamv.produto                       prod_itform,
			           dbamv.uni_pro                       uni_pro_form,
			           dbamv.uni_pro                       uni_pro_itform,
					   dbamv.Empresa_Produto			   Empresa_Produto
			where formula.cd_formula                 = itformula.cd_formula(+)
			    and formula.cd_prestador              = prestador.cd_prestador(+)
			    and formula.cd_cirurgia                  = cirurgia.cd_cirurgia(+)
			    and formula.cd_produto                 = prod_form.cd_produto
			    and itformula.cd_produto_tem        = prod_itform.cd_produto(+)
			    and formula.cd_uni_pro                 = uni_pro_form.cd_uni_pro(+)
			    and itformula.cd_uni_pro               = uni_pro_itform.cd_uni_pro(+)
			    and itformula.cd_produto_TEM     = uni_pro_itform.cd_produto(+)
			    and DECODE(DBAMV.PKT_CONFIGEST.RETORNA_CAMPO('SN_LOTE_VAL_PRD_EMP'), 'S',EMPRESA_PRODUTO.SN_MOVIMENTACAO,prod_form.SN_MOVIMENTACAO) = 'S'
			    and Empresa_Produto.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
			    and Empresa_Produto.Cd_Produto = Prod_Form.Cd_Produto
			Order by formula.cd_formula,
			         formula.cd_produto,
			         itformula.cd_produto_tem
			)
			ORDER BY formula,TP_FORMULA,VL_ADMINISTRATIVO,
			         prod_form,
			         cd_produto_tem_itform,
			PROD_FORM,CIRURGIA,DS_CIRURGIA,PRESTADOR,NM_PRESTADOR,DS_PROD_FORM,NR_DIAS_VALIDO_FORM,QT_PADRAO_FORM,DS_UNIDADE_FORM,DS_INDICACAO_FORM
