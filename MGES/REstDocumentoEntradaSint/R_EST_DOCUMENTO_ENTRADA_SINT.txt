--R_EST_DOCUMENTO_ENTRADA_SINT
SELECT * FROM (
select doc_ent.cd_fornecedor,
          fornecedor.nm_fornecedor,
          tip_doc.ds_tip_doc,
          doc_ent.nr_documento,
          doc_ent.nr_serie,
          doc_ent.dt_emissao,
          dup.dt_vencimento,
          dup.vl_parcela,
          doc_ent.vl_total,
          doc_ent.dt_entrada
from   dbamv.documento_entrada doc_ent,
          dbamv.fornecedor,
          dbamv.tip_doc,
         dbamv.duplicata dup
where doc_ent.cd_fornecedor = fornecedor.cd_fornecedor
   AND DOC_ENT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   and  fornecedor.Tp_Cliente_Forn In ( 'F', 'A', 'R', 'T' )
   and  doc_ent.cd_tip_doc = tip_doc.cd_tip_doc
   and  doc_ent.cd_fornecedor is not null
   and doc_ent.cd_fornecedor = dup.cd_fornecedor
   and doc_ent.nr_documento = dup.nr_documento
   and doc_ent.nr_serie = dup.nr_serie
   :CAMPO_CF_TP_DATA
)
ORDER BY CD_FORNECEDOR,NM_FORNECEDOR,
NR_DOCUMENTO,NR_SERIE,DT_EMISSAO,DT_ENTRADA,VL_TOTAL,DS_TIP_DOC, :CAMPO_CF_TP_ORDEM
