--R_VERIFICACAO_ROTATIVA
SELECT * FROM (
Select estoque.cd_estoque,
           estoque.ds_estoque,
           item_verificacao_rotativa_sald.cd_produto,
           produto.ds_produto,
           verificacao_rotativa_saldo.cd_verificacao_rotativa_saldo,
           verificacao_rotativa_saldo.dh_verificacao,
           decode(verificacao_rotativa_saldo.tp_status,
                        'A', 'Aberta',
                        '1', '1ª Verificacao',
                        '2', '2ª Verificacao',
                        '3', '3ª Verificacao',
                        'F', 'Concluída') tp_status,
           decode(verificacao_rotativa_saldo.tp_verificacao,
                       'T', 'Todo os Produtos',
                       'I',  'Ignorar Kits')tp_verificacao,
           uni_pro.ds_unidade,
           item_verificacao_rotativa_sald.qt_verificacao1,
           item_verificacao_rotativa_sald.qt_verificacao2,
           item_verificacao_rotativa_sald.qt_verificacao3,
           item_verificacao_rotativa_sald.qt_saldo_atual,
           item_verificacao_rotativa_sald.qt_kit
  from dbamv.estoque,
           dbamv.produto,
           dbamv.uni_pro,
           dbamv.verificacao_rotativa_saldo,
           dbamv.item_verificacao_rotativa_sald
where item_verificacao_rotativa_sald.cd_produto = produto.cd_produto
   and uni_pro.cd_uni_pro  = item_verificacao_rotativa_sald.cd_uni_pro
   and verificacao_rotativa_saldo.cd_verificacao_rotativa_saldo = item_verificacao_rotativa_sald.cd_verificacao_rotativa_saldo
   and TRUNC(verificacao_rotativa_saldo.dh_verificacao) Between :CAMPO_DT_INI and 
   and  verificacao_rotativa_saldo.cd_estoque = estoque.cd_estoque
    and estoque.cd_multi_empresa = Pkg_Mv2000.le_empresa
    Order by verificacao_rotativa_saldo.dh_verificacao, produto.ds_produto
)
ORDER BY CD_ESTOQUE,DS_ESTOQUE,DS_PRODUTO,
CD_VERIFICACAO_ROTATIVA_SALDO,DH_VERIFICACAO,TP_VERIFICACAO,TP_STATUS
