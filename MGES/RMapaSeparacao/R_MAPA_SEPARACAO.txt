--R_MAPA_SEPARACAO
SELECT  solsai_pro.cd_solsai_pro
       ,solsai_pro.dt_solsai_pro
       ,solsai_pro.ds_obs
       ,solsai_pro.cd_estoque_solicitante
       ,solsai_pro.cd_estoque cd_estoque
       ,solsai_pro.cd_usuario
       ,Decode(solsai_pro.sn_urgente,'S', 'Urgente', 'Normal') sn_urgente
       ,solsai_pro.cd_setor
       ,setor.nm_setor
       ,separacao.dt_separacao
       ,separacao.dt_separacao_validacao
       ,separacao.cd_separacao
       ,user1.nm_usuario nm_usuario
       ,user2.nm_usuario nm_usuario_validacao
       ,solsai_pro.cd_estoque_solicitante cd_estoque_solicitante
       ,est1.ds_estoque ds_estoque
       ,est2.ds_estoque ds_estoque_solicitante
FROM  dbamv.solsai_pro solsai_pro
     ,dbamv.separacao  separacao
     ,dbasgu.usuarios  user1
     ,dbasgu.usuarios  user2
     ,dbamv.estoque    est1
     ,dbamv.estoque     est2
     ,dbamv.setor setor
WHERE  solsai_pro.cd_solsai_pro = separacao.cd_solsai_pro
       AND  user1.cd_usuario = separacao.cd_usuario
       AND  user2.cd_usuario(+) = separacao.cd_usuario_validacao
       AND  est1.cd_estoque = solsai_pro.cd_estoque
       AND  est2.cd_estoque(+) = solsai_pro.cd_estoque_solicitante
       AND  setor.cd_setor(+) = solsai_pro.cd_setor
