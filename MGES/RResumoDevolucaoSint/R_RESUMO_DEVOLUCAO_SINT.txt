--R_RESUMO_DEVOLUCAO_SINT
SELECT itdev_for.cd_produto,
               produto.ds_produto,
               uni_pro.ds_unidade ds_unidade,
               sum(itdev_for.qt_devolvida) qt_entrada,
               sum(itent_pro.vl_custo_real * itdev_for.qt_devolvida) vl_total
FROM   dbamv.tip_doc,
             dbamv.fornecedor,
             dbamv.produto,
             dbamv.uni_pro,
             dbamv.ent_pro,
             dbamv.itent_pro,
             dbamv.estoque,
             dbamv.itdev_for,
             dbamv.dev_for
WHERE  ent_pro.cd_ent_pro    = itent_pro.cd_ent_pro
     AND   ent_pro.cd_fornecedor = fornecedor.cd_fornecedor
     AND   ent_pro.cd_estoque = estoque.cd_estoque
     AND   ent_pro.cd_tip_doc  = tip_doc.cd_tip_doc
     AND   itent_pro.cd_produto  = produto.cd_produto
     AND   itent_pro.cd_uni_pro  = uni_pro.cd_uni_pro
     AND   trunc(dev_for.dt_emissao) BETWEEN :CAMPO_DTINIC  AND 
     AND   ent_pro.cd_fornecedor is not null
     AND   Fornecedor.Tp_Cliente_Forn In  ( 'F', 'A', 'R', 'T' )
     AND   Itdev_for.cd_itent_pro = Itent_pro.cd_itent_pro
     AND   ent_pro.cd_fornecedor = fornecedor.cd_fornecedor
     AND   dev_for.cd_ent_pro = ent_pro.cd_ent_pro
     AND   dev_for.cd_devolucao = itdev_for.cd_devolucao
     AND   Estoque.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
group by  itdev_for.cd_produto,
               produto.ds_produto,
               verif_ds_unid_prod(produto.cd_produto),
               uni_pro.ds_unidade
order by ds_produto
