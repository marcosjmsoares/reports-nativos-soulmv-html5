--R_EST_SAIDA_PRODUTO_ESP
SELECT CD_ESPECIE,
       DS_ESPECIE,
       CD_PRODUTO,
       DS_PRODUTO,
       Sum(QT_MOVIMENTO /  VERIF_VL_FATOR_PROD(CD_PRODUTO)) QT_MOVIMENTO,
       CD_UNIDADE ,
       Sum(QT_MOVIMENTO * NVL(dbamv.verif_vl_custo_medio(cd_produto,dt_mvto_estoque, :CAMPO_TP_CUSTO, 0, dt_mvto_estoque ),0)) AS VL_TOTAL
FROM (
SELECT ESPECIE.CD_ESPECIE
	,ESPECIE.DS_ESPECIE
	,ITMVTO_ESTOQUE.CD_PRODUTO
	,PRODUTO.DS_PRODUTO
	,SUM(ITMVTO_ESTOQUE.QT_MOVIMENTACAO * DECODE( MVTO_ESTOQUE.TP_MVTO_ESTOQUE, 'D', -1, 'C', -1, 'N',-1,1 ) *  UNI_PRO.VL_FATOR) QT_MOVIMENTO                                                                						
	,VERIF_DS_UNID_PROD(ITMVTO_ESTOQUE.CD_PRODUTO)    CD_UNIDADE
  ,Trunc(MVTO_ESTOQUE.DT_MVTO_ESTOQUE,'MM') AS DT_MVTO_ESTOQUE
FROM
	DBAMV.ITMVTO_ESTOQUE
	,DBAMV.MVTO_ESTOQUE
	,DBAMV.PRODUTO
	,DBAMV.UNI_PRO
	,DBAMV.ESPECIE
	,DBAMV.ESTOQUE
WHERE	MVTO_ESTOQUE.TP_MVTO_ESTOQUE IN ( 'S', 'P', 'D', 'C','N','V' )
AND        MVTO_ESTOQUE.DT_MVTO_ESTOQUE BETWEEN :CAMPO_DATA_INI AND 
AND        MVTO_ESTOQUE.CD_MVTO_ESTOQUE = ITMVTO_ESTOQUE.CD_MVTO_ESTOQUE
AND        MVTO_ESTOQUE.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
AND        ITMVTO_ESTOQUE.CD_PRODUTO = PRODUTO.CD_PRODUTO
AND        ITMVTO_ESTOQUE.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO
AND        PRODUTO.CD_ESPECIE = ESPECIE.CD_ESPECIE
AND        ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA 
GROUP BY ESPECIE.CD_ESPECIE
	,ESPECIE.DS_ESPECIE
	,ITMVTO_ESTOQUE.CD_PRODUTO
	,PRODUTO.DS_PRODUTO
  ,Trunc(MVTO_ESTOQUE.DT_MVTO_ESTOQUE,'MM')
)
GROUP BY CD_ESPECIE,DS_ESPECIE, CD_PRODUTO, DS_PRODUTO , CD_UNIDADE
ORDER BY DS_ESPECIE,CD_ESPECIE, :CAMPO_CF_TP_SAIDA
