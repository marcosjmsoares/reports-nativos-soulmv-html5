--R_NFSR_INUTILIZA
SELECT INUTILIZA_FORMULARIO_NF.NR_SERIE
                  ,INUTILIZA_FORMULARIO_NF.NR_INICIAL
                  ,INUTILIZA_FORMULARIO_NF.NR_FINAL
                  ,Decode(FORMULARIO_NF_SIMPLES_REMESSA.SN_ATIVO,'S','SIM','N','NÃO') SN_ATIVO
                 ,INUTILIZA_FORMULARIO_NF.CD_PROTOCOLO
                 ,INUTILIZA_FORMULARIO_NF.DS_MOTIVO
                 ,INUTILIZA_FORMULARIO_NF.CD_USUARIO
                 ,INUTILIZA_FORMULARIO_NF.DT_INUTILIZA
                 ,MULTI_EMPRESAS.CD_MULTI_EMPRESA
                 ,MULTI_EMPRESAS.DS_MULTI_EMPRESA
      FROM DBAMV.INUTILIZA_FORMULARIO_NF
                ,DBAMV.MULTI_EMPRESAS
                ,DBAMV.FORMULARIO_NF_SIMPLES_REMESSA
  WHERE INUTILIZA_FORMULARIO_NF.CD_MULTI_EMPRESA                  = MULTI_EMPRESAS.CD_MULTI_EMPRESA
       AND FORMULARIO_NF_SIMPLES_REMESSA.CD_MULTI_EMPRESA  = MULTI_EMPRESAS.CD_MULTI_EMPRESA
       AND INUTILIZA_FORMULARIO_NF.CD_FORMULARIO                          = FORMULARIO_NF_SIMPLES_REMESSA.CD_FORMULARIO
