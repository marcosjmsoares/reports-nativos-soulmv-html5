--R_FORA_BRASIND
SELECT * FROM (
select  produto.cd_produto    codigo
         , produto.ds_produto    descricao
         , especie.cd_especie   cd_especie
         , especie.ds_especie   ds_especie
from   dbamv.produto
        , dbamv.especie
where  produto.cd_pro_fat not in (select imp_bra.cd_pro_fat from dbamv.imp_bra)
  and   especie.cd_especie = produto.cd_especie
  And Exists ( Select Empresa_Produto.Cd_Produto
                     From Dbamv.Empresa_Produto
                   Where Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
                        and Empresa_Produto.Cd_Produto = Produto.Cd_Produto )
:CAMPO_CF_P_DETALHES
)
ORDER BY DS_ESPECIE,CD_ESPECIE, :CAMPO_ORDEM
