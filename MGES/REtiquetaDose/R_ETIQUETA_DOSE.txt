SELECT PACIENTE.NM_PACIENTE PACIENTE,
       PACIENTE.DT_NASCIMENTO NASCIMENTO,
       PRE_MED.CD_PRE_MED PRESCRICAO,
       PACIENTE.NM_MAE,
       ITPRE_MED.CD_ITPRE_MED,       
	     PRODUTO.DS_PRODUTO,
	     PRODUTO_ESTABILIDADE.QT_HORAS_ESTABILIDADE,
       (SELECT PRESTADOR.NM_PRESTADOR
          FROM DBAMV.SOLSAI_PRO,
               DBAMV.PRESTADOR
         WHERE SOLSAI_PRO.CD_PRE_MED = PRE_MED.CD_PRE_MED
           AND PRESTADOR.CD_PRESTADOR = SOLSAI_PRO.CD_PRESTADOR) NM_PRESTADOR,
       (SELECT HR_PREPARACAO
          FROM PREPARACAO_LOTE
         WHERE PREPARACAO_LOTE.CD_ITPRE_MED = ITPRE_MED.CD_ITPRE_MED
           AND PREPARACAO_LOTE.TP_STATUS IN ('C','L')) HR_PREPARACAO,
       (SELECT CD_SOLSAI_PRO
          FROM DBAMV.SOLSAI_PRO
         WHERE SOLSAI_PRO.CD_PRE_MED = PRE_MED.CD_PRE_MED) CD_SOLSAI_PRO    
  FROM DBAMV.PRE_MED,
       DBAMV.ITPRE_MED,
       DBAMV.ATENDIME,
       DBAMV.PACIENTE,
       DBAMV.TIP_PRESC,
	     DBAMV.PRODUTO,
	     DBAMV.UNI_PRO,
       DBAMV.PRODUTO_ESTABILIDADE
 WHERE PRE_MED.CD_PRE_MED = ITPRE_MED.CD_PRE_MED
   AND PRE_MED.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
   AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
   AND ITPRE_MED.CD_TIP_PRESC = TIP_PRESC.CD_TIP_PRESC
   AND Nvl(ITPRE_MED.CD_PRODUTO, TIP_PRESC.CD_PRODUTO) = PRODUTO.CD_PRODUTO
   AND ITPRE_MED.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO
   AND PRODUTO.CD_PRODUTO = PRODUTO_ESTABILIDADE.CD_PRODUTO (+)
   AND ITPRE_MED.CD_ITPRE_MED in  :CAMPO_CD_ITPRE_MED
