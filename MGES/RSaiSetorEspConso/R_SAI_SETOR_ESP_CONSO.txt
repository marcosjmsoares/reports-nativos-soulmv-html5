--R_SAI_SETOR_ESP_CONSO
SELECT * FROM (
Select setor.cd_setor
          ,setor.nm_setor
          ,especie.cd_especie
          ,especie.ds_especie
          ,to_date(to_char(c_conset.cd_mes,'09') || '/'||c_conset.cd_ano, 'MM/YYYY') dt_mvto_estoque      
          ,null cd_mvto_estoque
          ,c_conset.cd_produto
          ,c_conset.cd_produto||' - '||produto.ds_produto ds_produto
          ,null nr_documento
          ,dbamv.verif_ds_unid_prod(c_conset.cd_produto)  ds_unidade
          ,dbamv.verif_vl_fator_prod(c_conset.cd_produto)  vl_fator
          ,0 vl_unitario
          ,null  tp_mvto_estoque
          ,0 qt_movimento
          , sum(qt_saida_setor) qt_saida_setor
          , sum(vl_saida_setor)  vl_saida_setor
          , sum(qt_devolvida_setor) qt_devolvida_setor
          , sum(vl_devolvida_setor) vl_devolvida_setor
          , sum(qt_saida_paciente) qt_saida_paciente
          , sum(vl_saida_paciente) vl_saida_paciente
          , sum(qt_devolvida_paciente) qt_devolvida_paciente
          , sum(vl_devolvida_paciente) vl_devolvida_paciente
From   dbamv.c_conset
         , dbamv.setor
         , dbamv.produto
         , dbamv.especie
	 , dbamv.estoque 
Where
          c_conset.cd_setor                = setor.cd_setor
AND       c_conset.CD_ESTOQUE  = ESTOQUE.CD_ESTOQUE
AND       ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
and       produto.cd_produto                      = c_conset.cd_produto
and       produto.cd_especie                      = especie.cd_especie
and c_conset.cd_ano|| trim(to_char(c_conset.cd_mes,'09')) between to_char( :CAMPO_DTINIC,'YYYYMM') AND to_char(:CAMPO_DTFIM ,'YYYYMM')
group by setor.cd_setor
          ,setor.nm_setor
          ,especie.cd_especie
          ,especie.ds_especie
          ,to_date(to_char(c_conset.cd_mes,'09') || '/'||c_conset.cd_ano, 'MM/YYYY')
          ,c_conset.cd_produto
          ,c_conset.cd_produto
          ,produto.ds_produto
          ,dbamv.verif_ds_unid_prod(c_conset.cd_produto)
          ,dbamv.verif_vl_fator_prod(c_conset.cd_produto)
Order by dt_mvto_estoque, setor.cd_setor, especie.cd_especie, ds_produto
)
ORDER BY CD_SETOR,NM_SETOR,
CD_ESPECIE,DS_ESPECIE,
DT_MVTO_ESTOQUE, DS_PRODUTO
