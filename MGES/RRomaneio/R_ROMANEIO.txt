--R_ROMANEIO
SELECT 		R.CD_ROMANEIO,
		R.DS_ROMANEIO,
        	U.NM_USUARIO,
		TO_CHAR(R.DT_CRIACAO,'DD/MM/YYYY') DT_CRIACAO,
		R.TP_SITUACAO,
        	(SELECT U2.NM_USUARIO FROM DBASGU.USUARIOS U2 WHERE R.CD_USUARIO_LIBERACAO = U2.CD_USUARIO) NM_USUARIO_LIBERACAO,
		TO_CHAR(R.DT_LIBERACAO,'DD/MM/YYYY') DT_LIBERACAO,
        	MR.NM_MOTORISTA,
	    	MR.NR_HABILITACAO,
		TO_CHAR(MR.DT_VENCIMENTO_HABILITACAO,'DD/MM/YYYY') DT_VENCIMENTO_HABILITACAO,
		MR.NR_TELEFONE_1||' - '||MR.NR_TELEFONE_2 AS CONTATO_MOT,
        	TR.NM_RAZAO_SOCIAL NM_RAZAO_SOCIAL,
		TR.NM_FANTASIA,
		TR.NR_CGC_CPF,
		TR.NR_TELEFONE_1 ||' - '||TR.NR_TELEFONE_2 AS CONTATO_TRANS,
    		VR.DS_MODELO DS_MODELO,
		VR.DS_PLACA,
		VR.DS_COR,
		VR.DS_CAPACIDADE_CARGA,
		R.DS_OBSERVACAO
FROM  	DBAMV.ROMANEIO R,
        DBAMV.TRANSPORTADORA_ROMANEIO TR,
        DBAMV.MOTORISTA_ROMANEIO MR,
        DBAMV.VEICULO_ROMANEIO VR,
        DBASGU.USUARIOS U
WHERE   R.CD_TRANSPORTADORA = TR.CD_TRANSPORTADORA(+)
AND     R.CD_VEICULO = VR.CD_VEICULO
AND     R.CD_MOTORISTA = MR.CD_MOTORISTA
AND     R.CD_USUARIO = U.CD_USUARIO
ORDER BY 1
