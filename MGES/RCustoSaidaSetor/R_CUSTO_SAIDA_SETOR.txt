--R_CUSTO_SAIDA_SETOR
Select sysdate from dual


/*----------------------------------------------------------------------------*/
--R_CUSTO_SAIDA_SETOR_R_GRUPO_DE_CUSTO1_Q_SAI_SETOR
SELECT * FROM (
SELECT SETOR.CD_SETOR
      ,SETOR.NM_SETOR
      ,GRUPO_DE_CUSTO.CD_GRUPO_DE_CUSTO
      ,GRUPO_DE_CUSTO.DS_GRUPO_DE_CUSTO
      ,SUB_CLAS.CD_SUB_CLA
      ,SUB_CLAS.DS_SUB_CLA
      ,COUNT(ITMVTO_ESTOQUE.CD_PRODUTO) QTD_ITENS_ATENDIDO
      ,SUM( ITMVTO_ESTOQUE.QT_MOVIMENTACAO * decode( mvto_estoque.tp_mvto_estoque, 'D', -1, 'C', -1,'N',-1, 1 )
      * (NVL( verif_vl_custo_medio(itmvto_estoque.cd_produto,mvto_estoque.dt_mvto_estoque, :CAMPO_TP_CUSTO, Produto.Vl_Custo_Medio, Mvto_Estoque.Hr_Mvto_Estoque ),0) * uni_pro.vl_fator ) )   VL_ATUAL
  FROM DBAMV.MVTO_ESTOQUE
      ,DBAMV.SETOR
      ,DBAMV.ESTOQUE
      ,DBAMV.ITMVTO_ESTOQUE
      ,DBAMV.UNI_PRO
      ,DBAMV.PRODUTO
      ,DBAMV.SUB_CLAS
      ,DBAMV.GRUPO_DE_CUSTO
 WHERE MVTO_ESTOQUE.DT_MVTO_ESTOQUE BETWEEN :CAMPO_DATA_INI AND 
   AND MVTO_ESTOQUE.CD_SETOR        = SETOR.CD_SETOR
   AND ITMVTO_ESTOQUE.CD_UNI_PRO    = UNI_PRO.CD_UNI_PRO
   AND MVTO_ESTOQUE.CD_MVTO_ESTOQUE = ITMVTO_ESTOQUE.CD_MVTO_ESTOQUE
   AND ITMVTO_ESTOQUE.CD_PRODUTO    = PRODUTO.CD_PRODUTO
   AND PRODUTO.CD_SUB_CLA           = SUB_CLAS.CD_SUB_CLA
   AND PRODUTO.CD_CLASSE            = SUB_CLAS.CD_CLASSE
   AND PRODUTO.CD_ESPECIE           = SUB_CLAS.CD_ESPECIE
   AND MVTO_ESTOQUE.CD_ESTOQUE      = ESTOQUE.CD_ESTOQUE
   AND SETOR.CD_GRUPO_DE_CUSTO      = GRUPO_DE_CUSTO.CD_GRUPO_DE_CUSTO(+)
   AND ESTOQUE.CD_MULTI_EMPRESA     = :CAMPO_CD_MULTI_EMPRESA
GROUP BY GRUPO_DE_CUSTO.DS_GRUPO_DE_CUSTO
        ,GRUPO_DE_CUSTO.CD_GRUPO_DE_CUSTO
        ,SETOR.CD_SETOR
        ,SETOR.NM_SETOR
        ,SUB_CLAS.CD_SUB_CLA
        ,SUB_CLAS.DS_SUB_CLA
ORDER BY DS_GRUPO_DE_CUSTO
        ,NM_SETOR
        ,DS_SUB_CLA
)
ORDER BY CD_GRUPO_DE_CUSTO, DS_GRUPO_DE_CUSTO,
        CD_SETOR,NM_SETOR, DS_SUB_CLA
