--R_CUST_SAISET_ESP_PCONS
SELECT GRUPO_DE_CUSTO.CD_GRUPO_DE_CUSTO            cd_grupo_de_custo
      ,GRUPO_DE_CUSTO.DS_GRUPO_DE_CUSTO            ds_grupo_de_custo
      ,SETOR.CD_SETOR                              cd_Setor_especie
      ,SETOR.NM_SETOR                              nm_setor_especie
      ,ESPECIE.CD_ESPECIE                          cd_especie
      ,ESPECIE.DS_ESPECIE                          ds_especie
      ,Sum(Nvl(C_ConSet.vl_saida_setor, 0))        vl_saida_setor
      ,Sum(Nvl(C_ConSet.vl_saida_paciente, 0))     vl_saida_paciente
      ,Sum(Nvl(C_ConSet.vl_baixa, 0))              vl_baixa
      ,Sum(Nvl(C_ConSet.vl_devolvida_setor, 0))    vl_devolvida_setor
      ,Sum(Nvl(C_ConSet.vl_devolvida_paciente, 0)) vl_devolvida_paciente
      ,Sum(Decode(Sign(Nvl(C_ConSet.vl_contagem_entrada,0) - Nvl(C_ConSet.vl_contagem_saida,0)), 1, (Nvl(C_ConSet.vl_contagem_entrada,0) - Nvl(C_ConSet.vl_contagem_saida,0)), 0)) vl_contagem_entrada
      ,Sum(Decode(Sign(Nvl(C_ConSet.vl_contagem_saida,0) - Nvl(C_ConSet.vl_contagem_entrada,0)), 1, (Nvl(C_ConSet.vl_contagem_saida,0) - Nvl(C_ConSet.vl_contagem_entrada,0)), 0)) vl_contagem_saida
FROM   DBAMV.C_CONSET
      ,DBAMV.SETOR
      ,DBAMV.ESTOQUE
      ,DBAMV.PRODUTO
      ,DBAMV.ESPECIE
      ,DBAMV.GRUPO_DE_CUSTO
WHERE  ((to_char(c_conset.cd_ano) || to_char(c_conset.cd_mes,'fm09'))
        between to_char(:CAMPO_DATA_INI,'YYYYMM')
            and to_char(:CAMPO_DATA_FIM ,'YYYYMM'))
AND    C_CONSET.CD_PRODUTO      = PRODUTO.CD_PRODUTO
AND    C_CONSET.CD_SETOR        = SETOR.CD_SETOR
AND    C_CONSET.CD_ESTOQUE      = ESTOQUE.CD_ESTOQUE
AND    PRODUTO.CD_ESPECIE       = ESPECIE.CD_ESPECIE
AND    SETOR.CD_GRUPO_DE_CUSTO  = GRUPO_DE_CUSTO.CD_GRUPO_DE_CUSTO
GROUP BY GRUPO_DE_CUSTO.CD_GRUPO_DE_CUSTO
        ,GRUPO_DE_CUSTO.DS_GRUPO_DE_CUSTO
        ,SETOR.CD_SETOR
        ,SETOR.NM_SETOR
        ,ESPECIE.CD_ESPECIE
        ,ESPECIE.DS_ESPECIE
ORDER BY CD_GRUPO_DE_CUSTO,
         CD_SETOR_ESPECIE,
         CD_ESPECIE
