--R_EST_ENTRADA_FORN
SELECT   FORNECEDOR.CD_FORNECEDOR
                ,FORNECEDOR.NM_FORNECEDOR
                ,FORNECEDOR.NR_CGC_CPF
                ,SUM(  NVL(ENT_PRO.VL_TOTAL,0)  )  VL_TOTAL
FROM      DBAMV.ENT_PRO
               ,DBAMV.FORNECEDOR
               ,DBAMV.TIP_DOC
WHERE	ENT_PRO.CD_FORNECEDOR = FORNECEDOR.CD_FORNECEDOR
AND	ENT_PRO.DT_ENTRADA BETWEEN :CAMPO_DATA_INI AND 
AND	ENT_PRO.CD_TIP_DOC = TIP_DOC.CD_TIP_DOC
And Ent_Pro.Cd_Estoque in ( Select Estoque.Cd_Estoque
			            From Dbamv.Estoque
                             Where Estoque.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA )
GROUP BY FORNECEDOR.CD_FORNECEDOR
                   ,FORNECEDOR.NM_FORNECEDOR
                   ,FORNECEDOR.NR_CGC_CPF
