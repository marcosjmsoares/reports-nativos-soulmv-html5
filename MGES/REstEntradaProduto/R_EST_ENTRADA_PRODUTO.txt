--R_EST_ENTRADA_PRODUTO
SELECT ITENT_PRO.CD_PRODUTO
              ,PRODUTO.DS_PRODUTO
              ,SUM(ITENT_PRO.VL_DESCONTO     / UNI_PRO.VL_FATOR)  VL_DESCONTO
              ,SUM(ITENT_PRO.VL_IMPOSTO         / UNI_PRO.VL_FATOR)  VL_IMPOSTO
              ,SUM(ITENT_PRO.VL_CUSTO_REAL / UNI_PRO.VL_FATOR)   VL_CUSTO_REAL
              ,SUM(ITENT_PRO.QT_ENTRADA * UNI_PRO.VL_FATOR)/VERIF_VL_FATOR_PROD(ITENT_PRO.CD_PRODUTO)   QT_MOVIMENTO
              ,VERIF_DS_UNID_PROD(ITENT_PRO.CD_PRODUTO)    CD_UNIDADE
              ,SUM(  NVL( ITENT_PRO.VL_CUSTO_REAL * ITENT_PRO.QT_ENTRADA,0)  )  VL_TOTAL
FROM      DBAMV.ENT_PRO
               ,DBAMV.ESTOQUE
               ,DBAMV.ITENT_PRO
               ,DBAMV.PRODUTO
               ,DBAMV.UNI_PRO
WHERE  ENT_PRO.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
AND        ENT_PRO.CD_ENT_PRO = ITENT_PRO.CD_ENT_PRO
AND	ENT_PRO.CD_FORNECEDOR IS NOT NULL
AND        ITENT_PRO.CD_PRODUTO = PRODUTO.CD_PRODUTO
AND        ITENT_PRO.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO
AND        ENT_PRO.DT_ENTRADA BETWEEN :CAMPO_DATA_INI AND 
AND        ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY ITENT_PRO.CD_PRODUTO
                   ,PRODUTO.DS_PRODUTO
