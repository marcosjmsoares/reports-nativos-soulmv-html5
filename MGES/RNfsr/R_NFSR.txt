SELECT MVTO_ESTOQUE_NFE.CD_MVTO_ESTOQUE            CD_MVTO
      ,MVTO_ESTOQUE_NFE.DT_MVTO_ESTOQUE            DT_MVTO
      ,MVTO_ESTOQUE_NFE.NR_DOCUMENTO               NR_DOC
      ,MVTO_ESTOQUE_NFE.NR_SERIE                   NR_SERIE
      ,MVTO_ESTOQUE_NFE.DT_AUTORIZACAO_NFE         DT_EMISSAO_NFE
      ,MVTO_ESTOQUE_NFE.DT_NFE_CANCELAMENTO        DT_CANC_NFE
      ,MVTO_ESTOQUE_NFE.PROTOCOLO_NFE              PROT_NFE
      ,MVTO_ESTOQUE_NFE.PROTOCOLO_NFE_CANCELAMENTO PROT_NFE_CANC
      ,ESTOQUE.CD_MULTI_EMPRESA                    CD_EMP_ORIGEM
      ,MVTO_ESTOQUE_NFE.CD_MULTI_EMPRESA           CD_EMP_DESTINO
      ,MULTI_EMPRESAS.DS_MULTI_EMPRESA             DS_EMP_DESTINO
      ,MVTO_ESTOQUE_NFE.CD_FORMULARIO_NFE     NR_FORMULARIO
      ,MVTO_ESTOQUE_NFE.CD_NUMERO_FORMULARIO_NFE  NR_SEQ_NF
      ,MVTO_ESTOQUE_NFE.dt_retorno_sapiens_emissao DT_SAPIENS_EMISSAO
      ,MVTO_ESTOQUE_NFE.dt_retorno_sapiens_cancela DT_SAPIENS_CANCELA
      ,MVTO_ESTOQUE_NFE.Vl_Total
      , 1 Linha 
FROM DBAMV.MVTO_ESTOQUE_NFE
    ,DBAMV.ESTOQUE
    ,DBAMV.MULTI_EMPRESAS
WHERE MVTO_ESTOQUE_NFE.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
  AND MULTI_EMPRESAS.CD_MULTI_EMPRESA = MVTO_ESTOQUE_NFE.CD_MULTI_EMPRESA
AND Decode(:CAMPO_TP_PERIODO ,'S',Trunc(MVTO_ESTOQUE_NFE.DT_MVTO_ESTOQUE) 
                                             ,'E',Trunc(MVTO_ESTOQUE_NFE.DT_AUTORIZACAO_NFE)
                                             ,'C',Trunc(MVTO_ESTOQUE_NFE.DT_NFE_CANCELAMENTO)) BETWEEN  :CAMPO_DT_INICIO ORDER BY ESTOQUE.CD_MULTI_EMPRESA
                 , MVTO_ESTOQUE_NFE.CD_MULTI_EMPRESA 
                 , MVTO_ESTOQUE_NFE.CD_FORMULARIO_NFE
                 , MVTO_ESTOQUE_NFE.CD_NUMERO_FORMULARIO_NFE
