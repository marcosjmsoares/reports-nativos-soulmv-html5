SELECT * FROM (
Select
           especie.cd_especie
          ,especie.ds_especie
          ,itmvto_estoque.cd_produto
          ,produto.ds_produto ds_produto
          ,verif_ds_unid_prod(itmvto_estoque.cd_produto)  ds_unidade
          ,SUM(decode(TP_MVTO_ESTOQUE, 'S', ((itmvto_estoque.qt_movimentacao * uni_pro.vl_fator) / verif_vl_fator_prod(itmvto_estoque.cd_produto)) ,
                                                                       'P', ((itmvto_estoque.qt_movimentacao * uni_pro.vl_fator) / verif_vl_fator_prod(itmvto_estoque.cd_produto)),
                                                                       'V', ((itmvto_estoque.qt_movimentacao * uni_pro.vl_fator) / verif_vl_fator_prod(itmvto_estoque.cd_produto)),  0 ))  qt_saida
          ,SUM(decode(TP_MVTO_ESTOQUE, 'D', ((itmvto_estoque.qt_movimentacao * uni_pro.vl_fator) / verif_vl_fator_prod(itmvto_estoque.cd_produto)),
                                                                       'C', ((itmvto_estoque.qt_movimentacao * uni_pro.vl_fator) / verif_vl_fator_prod(itmvto_estoque.cd_produto)),
                                                                       'N', ((itmvto_estoque.qt_movimentacao * uni_pro.vl_fator) / verif_vl_fator_prod(itmvto_estoque.cd_produto)), 0 ))  qt_devolvida
         ,SUM(ITMVTO_ESTOQUE.QT_MOVIMENTACAO * DECODE( MVTO_ESTOQUE.TP_MVTO_ESTOQUE, 'D', -1, 'C', -1, 'N',-1,1 ) *
	(NVL(  verif_vl_custo_medio(itmvto_estoque.cd_produto,mvto_estoque.dt_mvto_estoque, :CAMPO_TP_CUSTO, Produto.Vl_Custo_Medio, Mvto_Estoque.Hr_Mvto_Estoque ),0) * uni_pro.vl_fator ) )   VL_TOTAL
          ,estoque.cd_estoque
           ,estoque.ds_estoque
          From
           dbamv.mvto_estoque
         , dbamv.itmvto_estoque
         , dbamv.setor
         , dbamv.produto
         , dbamv.uni_pro
         , dbamv.especie
         , dbamv.estoque
Where
             mvto_estoque.dt_mvto_estoque BETWEEN :CAMPO_DTINIC  AND 
and       mvto_estoque.cd_mvto_estoque = itmvto_estoque.cd_mvto_estoque
and       mvto_estoque.cd_setor                = setor.cd_setor
and       produto.cd_produto                      = itmvto_estoque.cd_produto
and       produto.cd_especie                      = especie.cd_especie
and       uni_pro.cd_uni_pro                       = itmvto_estoque.cd_uni_pro
and mvto_estoque.cd_Estoque = estoque.cd_estoque
and mvto_estoque.tp_mvto_estoque in ( 'S', 'P', 'D','C','V','N')
and       estoque.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
group by
               especie.cd_especie
              ,especie.ds_especie
              ,itmvto_estoque.cd_produto
              ,produto.ds_produto
          ,estoque.cd_estoque
           ,estoque.ds_estoque
order by
  produto.ds_produto
)
ORDER BY CD_ESTOQUE,DS_ESTOQUE,
CD_ESPECIE,DS_ESPECIE, DS_PRODUTO
