--R_FICHA_VERIFICACAO_ROTATIVA
SELECT * FROM (
Select estoque.cd_estoque,
           estoque.ds_estoque,
           item_verificacao_rotativa_sald.cd_produto,
           produto.ds_produto,
           uni_pro.ds_unidade,
           verificacao_rotativa_saldo.cd_verificacao_rotativa_saldo,
           verificacao_rotativa_saldo.dh_verificacao,
           verificacao_rotativa_saldo.tp_status status,
           decode(verificacao_rotativa_saldo.tp_status,
                        'A', 'Aberta',
                        '1', '1ª Verificacao',
                        '2', '2ª Verificacao',
                        '3', '3ª Verificacao',
                        'F', 'Concluída') tp_status,
           decode(verificacao_rotativa_saldo.tp_verificacao,
                       'T', 'Todo os Produtos',
                       'I',  'Ignorar Kits')tp_verificacao
  from dbamv.estoque,
          dbamv.verificacao_rotativa_saldo,
          dbamv.item_verificacao_rotativa_sald,
          dbamv.uni_pro,
          dbamv.produto
where estoque.cd_estoque = verificacao_rotativa_saldo.cd_estoque
    and verificacao_rotativa_saldo.cd_verificacao_rotativa_saldo = item_verificacao_rotativa_sald.cd_verificacao_rotativa_saldo
    and item_verificacao_rotativa_sald.cd_uni_pro = uni_pro.cd_uni_pro
    and item_verificacao_rotativa_sald.cd_produto = produto.cd_produto
    and ( (item_verificacao_rotativa_sald.sn_bloqueado =  'S' And Verificacao_rotativa_saldo.Tp_status <> 'A') Or (Verificacao_rotativa_saldo.Tp_status = 'A'))
    and verificacao_rotativa_saldo.cd_verificacao_rotativa_saldo = :CAMPO_VERIFICACAO_ROTATIVA
)
ORDER BY CD_VERIFICACAO_ROTATIVA_SALDO,DH_VERIFICACAO,CD_ESTOQUE,DS_ESTOQUE,TP_STATUS,STATUS,TP_VERIFICACAO,cd_produto
