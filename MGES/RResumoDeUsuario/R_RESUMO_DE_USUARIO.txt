--R_RESUMO_DE_USUARIO
SELECT * FROM (
Select Mvto_estoque.cd_usuario
         ,estoque.cd_Estoque
         ,estoque.ds_Estoque
         ,Sum( Decode( Mvto_estoque.tp_mvto_estoque, 'C', 1, 'D', 1, 'N',1, 0 ) ) Doc_dev
         ,Sum( Decode( Mvto_estoque.tp_mvto_estoque, 'S', 1, 'P', 1, 'V',1, 0 ) ) Doc_Sai
         ,Sum( Decode( Mvto_estoque.tp_mvto_estoque, 'T', 1, 0 ) ) Doc_Tra
         ,Sum( Decode( Mvto_estoque.tp_mvto_estoque, 'X', 1, 0 ) ) Doc_Bai
         ,Count( Mvto_estoque.cd_mvto_estoque ) Doc_Tot
         ,( (:CAMPO_DTFINAL -  :CAMPO_DTINIC )+1) tot_dias
   From Dbamv.mvto_estoque
           ,Dbamv.estoque
    where mvto_estoque.dt_mvto_estoque <= :CAMPO_DTFINAL
        and mvto_Estoque.dt_mvto_estoque >= :CAMPO_DTINIC
        and Mvto_Estoque.Tp_Mvto_Estoque In ( 'S', 'P', 'D', 'C', 'T', 'X','N','V' )
        and estoque.cd_Estoque = mvto_Estoque.cd_Estoque
        and estoque.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
 group by mvto_estoque.cd_usuario
                        ,estoque.cd_Estoque
                       ,estoque.ds_Estoque
)
ORDER BY CD_ESTOQUE,DS_ESTOQUE
