--R_MVTO_KIT_MVTO
SELECT * FROM (
select  mvto_estoque.cd_setor
       ,setor.nm_setor
       ,mvto_estoque.dt_mvto_estoque
       ,mvto_estoque.cd_mvto_estoque
       ,decode(mvto_estoque.tp_mvto_estoque,'S','Saída Setor'
                                           ,'P','Saída Paciente'
                                           ,'D','Dev. Setor'
                                           ,'C','Dev. Paciente'
                                           ,'T','Transferência'
                                           ,'M','Manipulação'
                                           ,'O','Doação'
                                           ,'V','Venda'
                                           ,'N','Devolucao de vendas'
                                           ,'X','Baixa'
                                           ,'E','Empréstimo') tp_mvto
       ,formula.cd_produto
       ,Decode(produto.ds_produto_resumido,null,produto.ds_produto,produto.ds_produto_resumido) ds_produto
       ,nvl(itmvto_kits.qt_kit,1) qt_kit 
from    dbamv.mvto_estoque
       ,dbamv.itmvto_kits
       ,dbamv.formula
       ,dbamv.produto
       ,dbamv.setor
       ,dbamv.estoque 
where   mvto_estoque.cd_mvto_estoque = itmvto_kits.cd_mvto_estoque
and     formula.cd_formula           = itmvto_kits.cd_formula
and     produto.cd_produto           = formula.cd_produto
and     setor.cd_setor               = mvto_estoque.cd_setor
and     mvto_estoque.dt_mvto_estoque between :CAMPO_DATA_INI and 
AND MVTO_ESTOQUE.CD_ESTOQUE  = ESTOQUE.CD_ESTOQUE
AND ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
order by  setor.cd_setor
         ,mvto_estoque.dt_mvto_estoque
         ,mvto_estoque.cd_mvto_estoque
         ,formula.cd_produto
)
ORDER BY NM_SETOR,CD_SETOR,
DT_MVTO_ESTOQUE,
CD_MVTO_ESTOQUE,TP_MVTO
