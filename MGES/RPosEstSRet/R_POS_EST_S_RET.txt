--R_POS_EST_S_RET
Select CD_ESPECIE,
       DS_ESPECIE,
       CD_PRODUTO,
       DS_PRODUTO,
       VL_CUSTO_MEDIO,
       QT_ESTOQUE,
       QT_ESTOQUE_DOADO
  From (Select CD_ESPECIE,
               DS_ESPECIE,
               CD_PRODUTO,
               DS_PRODUTO,
               VL_CUSTO_MEDIO,
               Sum(QT_ESTOQUE) As QT_ESTOQUE,
               Sum(QT_ESTOQUE_DOADO) As QT_ESTOQUE_DOADO
          From (Select *
                  From Table(DBAMV.PKG_MGES_R_POS_EST_S.ProcessarRelatorio(:CAMPO_DH_RETROATIVO, :CAMPO_MESTRE, :CAMPO_PS_CD_LOCALIZACAO, :CAMPO_CF_PRODUTO, :CAMPO_CF_CLASSE, :CAMPO_CF_SUB_CLA, :CAMPO_CF_ACONDICIONADO, :CAMPO_CF_KIT, :CAMPO_CF_P_CONTROLADO, :CAMPO_CF_ESPECIE, :CAMPO_CF_ESTOQUE, :CAMPO_CF_P_DETALHES)))
      Group By CD_ESPECIE,
               DS_ESPECIE,
               CD_PRODUTO,
               DS_PRODUTO,
               VL_CUSTO_MEDIO)
 Where :CAMPO_QT_ZERO = 'A'
    Or (:CAMPO_QT_ZERO = 'Z'
    And QT_ESTOQUE = 0)
    Or (:CAMPO_QT_ZERO = 'D'
    And QT_ESTOQUE > 0)
Order By 1,
       4
