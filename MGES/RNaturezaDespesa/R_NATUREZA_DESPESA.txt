/*----------------------------------------------------------------------------*/
--R_NATUREZA_DESPESA
SELECT  CD_ENT_SERV
      , DT_ENTRADA
      , CD_SERVICO
      , NM_SERVICO
      , VL_SERVICO
      , CD_MULTI_EMPRESA
      , CD_ESPECIE
      , DS_ESPECIE
      , CD_CODIGO_NATUREZA_DESPESA
      , DS_NATUREZA_DESPESA
      , VL_ENT_NOTA_FISCAL
      , VL_ENT_VALE
      , VL_ENT_DOACAO
      , VL_ENT_NOTA_FICAL_N_COBRADA
      , VL_ENT_ACERTO_ESTOQUE
      , VL_DEVOL_FORNECEDOR
      , VL_ENT_CONTAGEM
      , VL_SAI_CONTAGEM
      , VL_SAI_SETOR
      , VL_SAI_PACIENTE
      , VL_SAI_DOACAO
      , VL_SAI_VENDA
      , VL_SAI_BAIXA
      , VL_SAI_EMPRESTIMO
      , VL_ENT_TRANSF_EMP
      , VL_SAI_TRANSF_EMP
      , VL_SAI_TOMBAMENTO
      , VL_DEVOL_SETOR
      , VL_DEVOL_PACIENTE
      , VL_DEVOL_VENDA
FROM (
SELECT  NULL CD_ENT_SERV
      , NULL DT_ENTRADA
      , NULL CD_SERVICO
      , NULL NM_SERVICO
      , 0 VL_SERVICO
      , ESTOQUE.CD_MULTI_EMPRESA                                CD_MULTI_EMPRESA
      , V.CD_ESPECIE                                            CD_ESPECIE
      , V.DS_ESPECIE                                            DS_ESPECIE
      , V.CD_CODIGO_NATUREZA_DESPESA                            CD_CODIGO_NATUREZA_DESPESA
      , V.DS_NATUREZA_DESPESA                                   DS_NATUREZA_DESPESA
      , SUM(V.VL_ENT_NOTA_FISCAL)                               VL_ENT_NOTA_FISCAL
      , SUM(V.VL_ENT_VALE)                                      VL_ENT_VALE
      , SUM(V.VL_ENT_DOACAO)                                    VL_ENT_DOACAO
      , SUM(V.VL_ENT_NOTA_FICAL_N_COBRADA)                      VL_ENT_NOTA_FICAL_N_COBRADA
      , SUM(V.VL_ENT_ACERTO_ESTOQUE)                            VL_ENT_ACERTO_ESTOQUE
      , SUM(V.VL_DEVOL_FORNECEDOR)                              VL_DEVOL_FORNECEDOR
      , SUM(V.QT_ENT_CONTAGEM              * V.VL_CUSTO_MEDIO)  VL_ENT_CONTAGEM
      , SUM(V.QT_SAI_CONTAGEM              * V.VL_CUSTO_MEDIO)  VL_SAI_CONTAGEM
      , SUM(V.QT_SAI_SETOR                 * V.VL_CUSTO_MEDIO)  VL_SAI_SETOR
      , SUM(V.QT_SAI_PACIENTE              * V.VL_CUSTO_MEDIO)  VL_SAI_PACIENTE
      , SUM(V.QT_SAI_DOACAO                * V.VL_CUSTO_MEDIO)  VL_SAI_DOACAO
      , SUM(V.QT_SAI_VENDA                 * V.VL_CUSTO_MEDIO)  VL_SAI_VENDA
      , SUM(V.QT_SAI_BAIXA                 * V.VL_CUSTO_MEDIO)  VL_SAI_BAIXA
      , SUM(V.QT_SAI_EMPRESTIMO            * V.VL_CUSTO_MEDIO)  VL_SAI_EMPRESTIMO
      , SUM(V.VL_ENT_TRANSF_EMP)                                VL_ENT_TRANSF_EMP
      , SUM(V.QT_SAI_TRANSF_EMP            * V.VL_CUSTO_MEDIO)  VL_SAI_TRANSF_EMP
      , SUM(V.QT_SAI_TOMBAMENTO            * V.VL_CUSTO_MEDIO)  VL_SAI_TOMBAMENTO
      , SUM(V.QT_DEVOL_SETOR               * V.VL_CUSTO_MEDIO)  VL_DEVOL_SETOR
      , SUM(V.QT_DEVOL_PACIENTE            * V.VL_CUSTO_MEDIO)  VL_DEVOL_PACIENTE
      , SUM(V.QT_DEVOL_VENDA               * V.VL_CUSTO_MEDIO)  VL_DEVOL_VENDA
FROM (
       SELECT VDIC_RELATORIO_NATUREZA_DESP.CD_ESTOQUE
            , PRODUTO.CD_PRODUTO
            , DECODE (:CAMPO_TP_QUEBRA, 'N', NULL , PRODUTO.CD_ESPECIE)  CD_ESPECIE
            , DECODE (:CAMPO_TP_QUEBRA, 'N', NULL , ESPECIE.DS_ESPECIE)  DS_ESPECIE
            , DECODE (:CAMPO_TP_QUEBRA, 'E', '@#$%0099' , VDIC_RELATORIO_NATUREZA_DESP.CD_CODIGO_NATUREZA_DESPESA)  CD_CODIGO_NATUREZA_DESPESA
            , DECODE (:CAMPO_TP_QUEBRA, 'E', '@#$%0099' , (SELECT ND.DS_NATUREZA_DESPESA
                                          FROM NATUREZA_DESPESA ND
                                         WHERE ND.CD_CODIGO_NATUREZA_DESPESA =  VDIC_RELATORIO_NATUREZA_DESP.CD_CODIGO_NATUREZA_DESPESA
                                           AND ND.CD_PRODUTO = PRODUTO.CD_PRODUTO)
                     ) DS_NATUREZA_DESPESA
            ,  DBAMV.VERIF_VL_CUSTO_MEDIO( PRODUTO.CD_PRODUTO, To_Date('01/'||To_Char(:CAMPO_DT_INICIO||'/'||:CAMPO_DT_ANO), 'DD/MM/YYYY'), 'H', NULL, To_Date('01/'||To_Char(:CAMPO_DT_INICIO||'/'||:CAMPO_DT_ANO), 'DD/MM/YYYY')) VL_CUSTO_MEDIO
	          , SUM(VDIC_RELATORIO_NATUREZA_DESP.VL_ENT_NOTA_FISCAL) 			     VL_ENT_NOTA_FISCAL
            , SUM(VDIC_RELATORIO_NATUREZA_DESP.VL_ENT_VALE)                  VL_ENT_VALE
            , SUM(VDIC_RELATORIO_NATUREZA_DESP.VL_ENT_DOACAO)                VL_ENT_DOACAO
            , SUM(VDIC_RELATORIO_NATUREZA_DESP.VL_ENT_NOTA_FICAL_N_COBRADA)  VL_ENT_NOTA_FICAL_N_COBRADA
            , SUM(VDIC_RELATORIO_NATUREZA_DESP.VL_ENT_ACERTO_ESTOQUE)        VL_ENT_ACERTO_ESTOQUE
            , SUM(VDIC_RELATORIO_NATUREZA_DESP.VL_DEVOL_FORNECEDOR )         VL_DEVOL_FORNECEDOR
            , SUM(VDIC_RELATORIO_NATUREZA_DESP.QT_ENT_CONTAGEM )             QT_ENT_CONTAGEM
            , SUM(VDIC_RELATORIO_NATUREZA_DESP.QT_SAI_CONTAGEM )             QT_SAI_CONTAGEM
            , SUM(VDIC_RELATORIO_NATUREZA_DESP.QT_SAI_SETOR)                 QT_SAI_SETOR
            , SUM(VDIC_RELATORIO_NATUREZA_DESP.QT_SAI_PACIENTE)              QT_SAI_PACIENTE
            , SUM(VDIC_RELATORIO_NATUREZA_DESP.QT_SAI_DOACAO)                QT_SAI_DOACAO
            , SUM(VDIC_RELATORIO_NATUREZA_DESP.QT_SAI_VENDA)                 QT_SAI_VENDA
            , SUM(VDIC_RELATORIO_NATUREZA_DESP.QT_SAI_BAIXA)                 QT_SAI_BAIXA
            , SUM(VDIC_RELATORIO_NATUREZA_DESP.QT_SAI_EMPRESTIMO)            QT_SAI_EMPRESTIMO
            , SUM(VDIC_RELATORIO_NATUREZA_DESP.VL_ENT_TRANSF_EMP)            VL_ENT_TRANSF_EMP
            , SUM(VDIC_RELATORIO_NATUREZA_DESP.QT_SAI_TRANSF_EMP)            QT_SAI_TRANSF_EMP
            , SUM(VDIC_RELATORIO_NATUREZA_DESP.QT_SAI_TOMBAMENTO)            QT_SAI_TOMBAMENTO
            , SUM(VDIC_RELATORIO_NATUREZA_DESP.QT_DEVOL_SETOR)               QT_DEVOL_SETOR
            , SUM(VDIC_RELATORIO_NATUREZA_DESP.QT_DEVOL_PACIENTE)            QT_DEVOL_PACIENTE
            , SUM(VDIC_RELATORIO_NATUREZA_DESP.QT_DEVOL_VENDA)               QT_DEVOL_VENDA
          FROM DBAMV.VDIC_RELATORIO_NATUREZA_DESP
             , DBAMV.PRODUTO
             , DBAMV.ESPECIE
        WHERE VDIC_RELATORIO_NATUREZA_DESP.CD_PRODUTO = PRODUTO.CD_PRODUTO
          AND PRODUTO.SN_MESTRE                       = 'N'
          AND PRODUTO.CD_ESPECIE                      = ESPECIE.CD_ESPECIE
          AND 'P' = :CAMPO_TP_NATUREZA
          AND TRUNC(DT_MOVIMENTACAO) BETWEEN Trunc(To_Date('01/'||To_Char(:CAMPO_DT_INICIO||'/'||:CAMPO_DT_ANO), 'DD/MM/YYYY'))
                                         AND Trunc(LAST_DAY(To_Date('01/'||To_Char(:CAMPO_DT_INICIO||'/'||:CAMPO_DT_ANO), 'DD/MM/YYYY')))
        GROUP BY  VDIC_RELATORIO_NATUREZA_DESP.CD_ESTOQUE
                , PRODUTO.CD_PRODUTO
                , PRODUTO.CD_ESPECIE
                , ESPECIE.DS_ESPECIE
                , VDIC_RELATORIO_NATUREZA_DESP.CD_CODIGO_NATUREZA_DESPESA
        ORDER BY 1,2,3
        ) V
        ,DBAMV.ESTOQUE
WHERE  V.CD_ESTOQUE = ESTOQUE.CD_ESTOQUE
GROUP BY  ESTOQUE.CD_MULTI_EMPRESA
        , V.CD_ESPECIE
        , V.DS_ESPECIE
        , V.CD_CODIGO_NATUREZA_DESPESA
        , V.DS_NATUREZA_DESPESA
UNION ALL
SELECT SERV.CD_ENT_SERV
      ,SERV.DT_ENTRADA
      ,SERV.CD_SERVICO
      ,SERV.NM_SERVICO
      ,SUM(SERV.VL_SERVICO)  VL_SERVICO
      , NULL CD_MULTI_EMPRESA
      , NULL CD_ESPECIE
      , NULL DS_ESPECIE
      , SERV.CD_CODIGO_NATUREZA_DESPESA
      , SERV.DS_NATUREZA_DESPESA
      , NULL VL_ENT_NOTA_FISCAL
      , NULL VL_ENT_VALE
      , NULL VL_ENT_DOACAO
      , NULL VL_ENT_NOTA_FICAL_N_COBRADA
      , NULL VL_ENT_ACERTO_ESTOQUE
      , NULL VL_DEVOL_FORNECEDOR
      , NULL VL_ENT_CONTAGEM
      , NULL VL_SAI_CONTAGEM
      , NULL VL_SAI_SETOR
      , NULL VL_SAI_PACIENTE
      , NULL VL_SAI_DOACAO
      , NULL VL_SAI_VENDA
      , NULL VL_SAI_BAIXA
      , NULL VL_SAI_EMPRESTIMO
      , NULL VL_ENT_TRANSF_EMP
      , NULL VL_SAI_TRANSF_EMP
      , NULL VL_SAI_TOMBAMENTO
      , NULL VL_DEVOL_SETOR
      , NULL VL_DEVOL_PACIENTE
      , NULL VL_DEVOL_VENDA
FROM (
SELECT ITENT_SERV.CD_CODIGO_NATUREZA_DESPESA
      , (SELECT ND.DS_NATUREZA_DESPESA
           FROM NATUREZA_DESPESA ND
          WHERE ND.CD_CODIGO_NATUREZA_DESPESA =  ITENT_SERV.CD_CODIGO_NATUREZA_DESPESA
            AND ND.CD_SERVICO = ITENT_SERV.CD_SERVICO
      ) DS_NATUREZA_DESPESA
      ,DECODE(:CAMPO_SN_AGRUPA_SERVICO, 'S', NULL,'N', ENT_SERV.CD_ENT_SERV ) CD_ENT_SERV
      ,ENT_SERV.DT_ENTRADA
      ,ITENT_SERV.CD_SERVICO
      ,MANU_SERV.NM_SERVICO
      ,SUM(NVL(ITENT_SERV.VL_TOTAL,0) - NVL(ITENT_SERV.VL_DESCONTO,0) + NVL(ITENT_SERV.VL_IMPOSTO,0)) VL_SERVICO
  FROM DBAMV.ENT_SERV
     , DBAMV.ITENT_SERV
     , DBAMV.MANU_SERV
 WHERE ENT_SERV.CD_ENT_SERV = ITENT_SERV.CD_ENT_SERV
   AND ITENT_SERV.CD_SERVICO =  MANU_SERV.CD_SERVICO(+)
   AND TRUNC(DT_ENTRADA) BETWEEN Trunc(To_Date('01/'||To_Char(:CAMPO_DT_INICIO||'/'||:CAMPO_DT_ANO), 'DD/MM/YYYY'))
                                         AND Trunc(LAST_DAY(To_Date('01/'||To_Char(:CAMPO_DT_INICIO||'/'||:CAMPO_DT_ANO), 'DD/MM/YYYY')))
   AND 'S' = :CAMPO_TP_NATUREZA
 GROUP BY  ITENT_SERV.CD_CODIGO_NATUREZA_DESPESA
      , ENT_SERV.DT_ENTRADA
      , ENT_SERV.CD_ENT_SERV
      , ITENT_SERV.CD_SERVICO
      , MANU_SERV.NM_SERVICO
      , ITENT_SERV.VL_TOTAL
) SERV
GROUP BY SERV.CD_CODIGO_NATUREZA_DESPESA
      , SERV.DS_NATUREZA_DESPESA
      , SERV.CD_ENT_SERV
      , SERV.DT_ENTRADA
      , SERV.CD_SERVICO
      , SERV.NM_SERVICO
)
WHERE CD_CODIGO_NATUREZA_DESPESA IS NOT NULL
ORDER BY CD_MULTI_EMPRESA, CD_ESPECIE, CD_CODIGO_NATUREZA_DESPESA, DT_ENTRADA,  CD_SERVICO
