--R_CON_PROD
Select produto.cd_produto
          ,produto.ds_produto
          ,Pro_fat.cd_pro_fat
          ,pro_fat.ds_pro_fat
From dbamv.produto produto
       , dbamv.pro_fat pro_fat
Where Produto.cd_pro_fat = pro_fat.cd_pro_fat
AND Exists ( SELECT Empresa_Produto.Cd_Produto FROM Dbamv.Empresa_Produto WHERE Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA AND Empresa_Produto.Cd_Produto = Produto.Cd_Produto )
and produto.cd_especie like :CAMPO_ESPECIE
and pro_fat.cd_gru_pro like  :CAMPO_GRU_PRO
and produto.ds_produto <> pro_fat.ds_pro_fat
