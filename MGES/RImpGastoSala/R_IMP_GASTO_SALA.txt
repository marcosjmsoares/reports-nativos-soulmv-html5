--R_IMP_GASTO_SALA
SELECT * FROM (
Select imp_gasto_sala.cd_gru_pro,
       imp_gasto_sala.cd_pro_fat,
       imp_gasto_sala.ds_impressora,
       estoque.cd_mvto_estoque cd_mvto_estoque,
       estoque.cd_aviso_cirurgia cd_aviso_cirurgia,
       estoque.cd_estoque cd_estoque,
       estoque.cd_atendimento cd_atendimento,
       estoque.cd_prestador cd_prestador,
       estoque.cd_usuario cd_usuario,
       estoque.dt_mvto_estoque dt_saida,
      estoque.hr_mvto_estoque hr_saida,
       est.ds_estoque ds_estoque,
       itestoque.cd_produto cd_produto,
       itestoque.cd_lote cd_lote,
       itestoque.dt_validade dt_validade,
       itestoque.qt_movimentacao qt_movimentacao,
       produto.ds_produto ds_produto,
       nvl( paciente.nm_paciente, AVISO_CIRURGIA.nm_paciente ) nm_paciente,
       prestador.nm_prestador nm_prestador,
       convenio.cd_convenio cd_convenio,
       convenio.nm_convenio nm_convenio,
       uni_pro.ds_unidade ds_unidade,
AVISO_CIRURGIA.DT_REALIZACAO,
CARTEIRA.NR_CARTEIRA
from dbamv.itmvto_Estoque itestoque,
       dbamv.mvto_estoque estoque,
       dbamv.produto,
       dbamv.atendime,
       dbamv.paciente,
       dbamv.prestador,
       dbamv.convenio,
       dbamv.uni_pro,
       dbamv.estoque est,
       dbamv.pro_fat,
       dbamv.gru_pro,
       dbamv.imp_gasto_sala,
DBAMV.AVISO_CIRURGIA,
DBAMV.CARTEIRA
where produto.cd_produto = itestoque.cd_produto
and est.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
and itestoque.cd_mvto_estoque = estoque.cd_mvto_estoque
and estoque.cd_atendimento = atendime.cd_atendimento (+)
and atendime.cd_paciente = paciente.cd_paciente (+)
and atendime.cd_prestador = prestador.cd_prestador (+)
and atendime.cd_convenio = convenio.cd_convenio (+)
and produto.cd_produto = uni_pro.cd_produto
and estoque.cd_estoque = est.cd_estoque
and itestoque.cd_uni_pro = uni_pro.cd_uni_pro
and pro_fat.cd_pro_fat = produto.cd_pro_fat
and pro_fat.cd_gru_pro = gru_pro.cd_gru_pro
and imp_gasto_sala.cd_gru_pro = gru_pro.cd_gru_pro
and estoque.cd_aviso_cirurgia =     :CAMPO_CD_AVISO_CIRURGIA
and estoque.cd_mvto_estoque =    :CAMPO_CD_MVTO_ESTOQUE
and imp_gasto_sala.cd_pro_fat = pro_fat.cd_pro_fat
and imp_gasto_sala.ds_impressora =  :CAMPO_DS_IMPRESSORA
and atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA (+) =  ESTOQUE.CD_AVISO_CIRURGIA
AND CARTEIRA.CD_PACIENTE (+)  = ATENDIME.CD_PACIENTE
union all
Select imp_gasto_sala.cd_gru_pro,
       imp_gasto_sala.cd_pro_fat,
       imp_gasto_sala.ds_impressora,
       estoque.cd_mvto_estoque cd_mvto_estoque,
       estoque.cd_aviso_cirurgia cd_aviso_cirurgia,
       estoque.cd_estoque cd_estoque,
       estoque.cd_atendimento cd_atendimento,
       estoque.cd_prestador cd_prestador,
       estoque.cd_usuario cd_usuario,
       estoque.dt_mvto_estoque dt_saida,
      estoque.hr_mvto_estoque hr_saida,
       est.ds_estoque ds_estoque,
       itestoque.cd_produto cd_produto,
       itestoque.cd_lote cd_lote,
       itestoque.dt_validade dt_validade,
       itestoque.qt_movimentacao qt_movimentacao,
       produto.ds_produto ds_produto,
       nvl( paciente.nm_paciente, AVISO_CIRURGIA.nm_paciente ) nm_paciente,
       prestador.nm_prestador nm_prestador,
       convenio.cd_convenio cd_convenio,
       convenio.nm_convenio nm_convenio,
       uni_pro.ds_unidade ds_unidade,
AVISO_CIRURGIA.DT_REALIZACAO,
CARTEIRA.NR_CARTEIRA
from dbamv.itmvto_Estoque itestoque,
       dbamv.mvto_estoque estoque,
       dbamv.produto,
       dbamv.atendime,
       dbamv.paciente,
       dbamv.prestador,
       dbamv.convenio,
       dbamv.uni_pro,
       dbamv.estoque est,
       dbamv.pro_fat,
       dbamv.gru_pro,
       dbamv.imp_gasto_sala,
DBAMV.AVISO_CIRURGIA,
DBAMV.CARTEIRA
where produto.cd_produto = itestoque.cd_produto
and est.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
and itestoque.cd_mvto_estoque = estoque.cd_mvto_estoque
and estoque.cd_atendimento = atendime.cd_atendimento (+)
and atendime.cd_paciente = paciente.cd_paciente (+)
and atendime.cd_prestador = prestador.cd_prestador (+)
and atendime.cd_convenio = convenio.cd_convenio (+)
and produto.cd_produto = uni_pro.cd_produto
and estoque.cd_estoque = est.cd_estoque
and itestoque.cd_uni_pro = uni_pro.cd_uni_pro
and pro_fat.cd_pro_fat = produto.cd_pro_fat
and pro_fat.cd_gru_pro = gru_pro.cd_gru_pro
and imp_gasto_sala.cd_gru_pro = gru_pro.cd_gru_pro
and estoque.cd_aviso_cirurgia =  :CAMPO_CD_AVISO_CIRURGIA
and estoque.cd_mvto_estoque =  :CAMPO_CD_MVTO_ESTOQUE
and imp_gasto_sala.ds_impressora =   :CAMPO_DS_IMPRESSORA
and atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA (+) =  ESTOQUE.CD_AVISO_CIRURGIA
AND CARTEIRA.CD_PACIENTE (+)  = ATENDIME.CD_PACIENTE
and imp_gasto_sala.cd_pro_fat is null
and pro_fat.cd_pro_fat not in ( Select igs.cd_pro_fat
                                  from dbamv.imp_gasto_sala igs
                                 Where igs.cd_pro_fat is not null)
)
ORDER BY CD_MVTO_ESTOQUE,CD_AVISO_CIRURGIA,CD_ESTOQUE,CD_ATENDIMENTO,CD_PRESTADOR,NM_CONVENIO,NM_PRESTADOR,CD_CONVENIO,NM_PACIENTE,DS_ESTOQUE,HR_SAIDA,DT_SAIDA,CD_USUARIO,NR_CARTEIRA,DT_REALIZACAO
