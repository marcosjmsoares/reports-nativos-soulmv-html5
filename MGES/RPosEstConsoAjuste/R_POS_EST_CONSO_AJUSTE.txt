--R_POS_EST_CONSO_AJUSTE
SELECT CD_PRODUTO,
       DS_PRODUTO,
       CD_SUB_CLA,
       DS_SUB_CLA,
       CD_CLASSE,
       DS_CLASSE,
       CD_ESPECIE,
       DS_ESPECIE,
       CD_ANO,
       CD_MES,
       DS_UNIDADE,
       CD_ESTOQUE,
       Nvl(QT_ESTOQUE_FINAL,0) AS QT_ESTOQUE_FINAL,
       Nvl(VL_ESTOQUE_FINAL,0) AS VL_ESTOQUE_FINAL,
       Nvl(VL_UNITARIO,0) AS  VL_UNITARIO,
       Nvl(QT_ESTOQUE_FINAL_SEM_TRANSF,0) AS QT_ESTOQUE_FINAL_SEM_TRANSF,
       Nvl(VL_ESTOQUE_FINAL_SEM_TRANSF,0) AS VL_ESTOQUE_FINAL_SEM_TRANSF,
       Nvl(QT_ESTOQUE_FINAL_REAL,0) AS QT_ESTOQUE_FINAL_REAL,
       Nvl(VL_ESTOQUE_FINAL_REAL,0) AS VL_ESTOQUE_FINAL_REAL,
       Nvl(QT_ENTRADA_AJUSTE,0) AS QT_ENTRADA_AJUSTE,
       Nvl(VL_ENTRADA_AJUSTE,0) AS VL_ENTRADA_AJUSTE,
       Nvl(QT_SAIDA_AJUSTE,0) AS QT_SAIDA_AJUSTE,
       Nvl(VL_SAIDA_AJUSTE,0) AS VL_SAIDA_AJUSTE,
       Nvl(SN_QUANTIDADE_NEGATIVA,'N') SN_QUANTIDADE_NEGATIVA,
       Nvl(SN_VALOR_NEGATIVO,'N') AS SN_VALOR_NEGATIVO
 FROM TABLE(DBAMV.PKG_MGES_M_CONSOLIDACAO.FNC_RETORNA_RELATORIO(To_Date('01/'||:CAMPO_DATA_INI||'/'||:CAMPO_DATA_FIM,'DD/MM/YYYY'),'S',:CAMPO_CD_PRODUTO,:CAMPO_ESPECIE,:CAMPO_CLASSE,:CAMPO_SUB_CLAS,:CAMPO_SN_GERA_AJUSTE_VL_NEGATIVO))
ORDER BY
CD_ESPECIE,
CD_CLASSE,
CD_SUB_CLA, DS_PRODUTO, CD_PRODUTO,CD_ESTOQUE
