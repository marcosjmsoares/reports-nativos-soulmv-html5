--R_IDENTIFICADOR_ETIQUETA
SELECT DISTINCT IDENTIFICADOR.* FROM (
select  ie.cd_produto
, nvl(mvto.dsp_cd_barras ,ie.cd_identificador ) cd_identificador
, prod.ds_produto
, mvto.sn_kit_armazenado sn_ativo
, mvto.cd_mvto_estoque
, nvl(mvto.cd_estoque_destino,mvto.cd_estoque) cd_estoque
, est.ds_estoque
 from dbamv.identificador_etiqueta ie, dbamv.produto prod
 ,dbamv.estoque est
 , dbamv.mvto_kit_produzido mvto
where ie.cd_produto = prod.cd_produto
   and prod.sn_kit = 'S'
   and fnc_mges_valida_cod_barra(mvto.dsp_cd_barras) = ie.CD_IDENTIFICADOR
   and mvto.cd_estoque = est.cd_estoque (+)
   and est.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
   AND ie.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
order by 1,2
) IDENTIFICADOR
ORDER BY CD_PRODUTO, CD_IDENTIFICADOR
