--R_TRAPRO_KIT
SELECT * FROM (
SELECT
       MVTO_ESTOQUE.CD_USUARIO,
       MVTO_ESTOQUE.CD_MVTO_ESTOQUE,
       MVTO_ESTOQUE.CD_SOLSAI_PRO,
       MVTO_ESTOQUE.CD_ESTOQUE       CD_ESTOQUE_ORIGEM,
       ESTOQUE_ORIGEM.DS_ESTOQUE  DS_ESTOQUE_ORIGEM ,
       MVTO_ESTOQUE.CD_ESTOQUE_DESTINO,
       ESTOQUE_DESTINO.DS_ESTOQUE   DS_ESTOQUE_DESTINO,
       MVTO_ESTOQUE.DT_MVTO_ESTOQUE,
       MVTO_ESTOQUE.HR_MVTO_ESTOQUE,
       MVTO_ESTOQUE.DT_IMPRESSAO,
       MVTO_ESTOQUE.cd_Kit,
       MVTO_ESTOQUE.NR_DOCUMENTO,
       MVTO_ESTOQUE.DT_VALIDADE   VALIDADE_KIT,
       ITMVTO_ESTOQUE.CD_PRODUTO,
       PRODUTO.DS_PRODUTO,
       ITMVTO_ESTOQUE.CD_PRODUTO_KIT              CD_PRODUTO_KIT,
       PRODUTO_KIT.DS_PRODUTO                             DS_PRODUTO_KIT,
       UNI_PRO.DS_UNIDADE,
       ITMVTO_ESTOQUE.CD_LOTE,
       ITMVTO_ESTOQUE.DT_VALIDADE,
       SUM(ITMVTO_ESTOQUE.QT_MOVIMENTACAO) QT_MOVIMENTACAO,
       VERIF_VL_CUSTO_MEDIO(ITMVTO_ESTOQUE.CD_PRODUTO,MVTO_ESTOQUE.DT_MVTO_ESTOQUE,'H',NULL,MVTO_ESTOQUE.HR_MVTO_ESTOQUE) * UNI_PRO.VL_FATOR VL_CUSTO_MEDIO
FROM
       DBAMV.MVTO_KIT_PRODUZIDO    MVTO_ESTOQUE,
       DBAMV.ITMVTO_KIT_PRODUZIDO  ITMVTO_ESTOQUE,
       DBAMV.ESTOQUE         ESTOQUE_ORIGEM,
       DBAMV.UNI_PRO         UNI_PRO,
       DBAMV.PRODUTO         PRODUTO,
       DBAMV.PRODUTO         PRODUTO_KIT,
       DBAMV.ESTOQUE         ESTOQUE_DESTINO
WHERE
MVTO_ESTOQUE.CD_MVTO_ESTOQUE    =  ITMVTO_ESTOQUE.CD_MVTO_ESTOQUE
AND    MVTO_ESTOQUE.CD_ESTOQUE         =  ESTOQUE_ORIGEM.CD_ESTOQUE
AND    MVTO_ESTOQUE.CD_ESTOQUE =  ESTOQUE_DESTINO.CD_ESTOQUE
AND    ITMVTO_ESTOQUE.CD_UNI_PRO       =  UNI_PRO.CD_UNI_PRO(+)
AND    ITMVTO_ESTOQUE.CD_PRODUTO       =  PRODUTO.CD_PRODUTO(+)
AND    MVTO_ESTOQUE.CD_MVTO_ESTOQUE    =  :CAMPO_CD_MVTO_ESTOQUE
AND        ITMVTO_ESTOQUE.CD_PRODUTO_KIT =   PRODUTO_KIT.CD_PRODUTO(+)
AND    ESTOQUE_ORIGEM.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY
       MVTO_ESTOQUE.CD_USUARIO,
       MVTO_ESTOQUE.CD_MVTO_ESTOQUE,
       MVTO_ESTOQUE.CD_SOLSAI_PRO,
       MVTO_ESTOQUE.CD_ESTOQUE,
       ESTOQUE_ORIGEM.DS_ESTOQUE,
       MVTO_ESTOQUE.CD_ESTOQUE_DESTINO,
       ESTOQUE_DESTINO.DS_ESTOQUE,
       MVTO_ESTOQUE.DT_MVTO_ESTOQUE,
       MVTO_ESTOQUE.HR_MVTO_ESTOQUE,
       MVTO_ESTOQUE.DT_IMPRESSAO,
       MVTO_ESTOQUE.cd_Kit,
       MVTO_ESTOQUE.NR_DOCUMENTO,
       MVTO_ESTOQUE.DT_VALIDADE,
       ITMVTO_ESTOQUE.CD_PRODUTO,
       PRODUTO.DS_PRODUTO,
       ITMVTO_ESTOQUE.CD_PRODUTO_KIT,
       PRODUTO_KIT.DS_PRODUTO,
       UNI_PRO.DS_UNIDADE,
       ITMVTO_ESTOQUE.CD_LOTE,
       ITMVTO_ESTOQUE.DT_VALIDADE,
VERIF_VL_CUSTO_MEDIO(ITMVTO_ESTOQUE.CD_PRODUTO,MVTO_ESTOQUE.DT_MVTO_ESTOQUE,'H',NULL,MVTO_ESTOQUE.HR_MVTO_ESTOQUE) * UNI_PRO.VL_FATOR
ORDER BY MVTO_ESTOQUE.CD_MVTO_ESTOQUE,
         PRODUTO.DS_PRODUTO
)
ORDER BY CD_MVTO_ESTOQUE,CD_ESTOQUE_ORIGEM,DS_ESTOQUE_ORIGEM,DS_ESTOQUE_DESTINO,CD_ESTOQUE_DESTINO,CD_USUARIO,DT_MVTO_ESTOQUE,HR_MVTO_ESTOQUE,CD_SOLSAI_PRO,DT_IMPRESSAO,NR_DOCUMENTO,CD_KIT,VALIDADE_KIT,
DS_PRODUTO_KIT,CD_PRODUTO_KIT

--R_TRAPRO_KIT_R_3_Q_1
SELECT ITSOLSAI_PRO.CD_PRODUTO               Cd_Produto,
         PRODUTO.DS_PRODUTO                            Ds_Produto,
         UNI_PRO.DS_UNIDADE                                Ds_Unidade,
         Nvl(ITSOLSAI_PRO.QT_SOLICITADO,0)	 Qtd_Solicitada,
         Nvl(ITSOLSAI_PRO.QT_ATENDIDA,0)	 Qtd_Atendida,
         (Nvl(ITSOLSAI_PRO.QT_SOLICITADO,0) - Nvl(ITSOLSAI_PRO.QT_ATENDIDA,0)) Qtd_Diferenca,
         PRODUTO.CD_PRODUTO_TEM                  Dsp_cd_produto_tem,
        PRODUTO.SN_MESTRE                                Dsp_sn_mestre,
        ITSOLSAI_PRO.SN_CONF_DETERM_USU   Conf_Usu
    FROM DBAMV.ITSOLSAI_PRO,
         DBAMV.PRODUTO,
         DBAMV.UNI_PRO,
         DBAMV.SUB_CLAS
   WHERE ITSOLSAI_PRO.CD_PRODUTO = PRODUTO.CD_PRODUTO
     AND ITSOLSAI_PRO.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO
     AND PRODUTO.CD_ESPECIE      = SUB_CLAS.CD_ESPECIE
     AND PRODUTO.CD_CLASSE       = SUB_CLAS.CD_CLASSE
     AND PRODUTO.CD_SUB_CLA      = SUB_CLAS.CD_SUB_CLA
     AND PRODUTO.SN_KIT <> 'C'
     AND SUB_CLAS.SN_PROD_PERMANENTE <> 'L'
     AND ITSOLSAI_PRO.CD_SOLSAI_PRO = :CAMPO_CD_SOLSAI_PRO
     And Exists ( Select Empresa_Produto.Cd_Produto
                     From Dbamv.Empresa_Produto
                   Where Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
                        and Empresa_Produto.Cd_Produto = Produto.Cd_Produto )
Union All
SELECT ItMvto_Estoque.Cd_Produto                Cd_Produto
       ,Produto.Ds_Produto                                   Ds_Produto
       ,Uni_Pro.Ds_Unidade                                  Ds_Unidade
       ,0	                                                            Qtd_Solicitada
       ,Nvl(ItMvto_Estoque.Qt_Movimentacao,0)   Qtd_Atendida
       ,- Nvl(ItMvto_Estoque.Qt_Movimentacao,0) Qtd_Diferenca
       ,Produto.Cd_Produto_Tem                            Dsp_cd_produto_tem
       ,Produto.Sn_Mestre                                       Dsp_sn_mestre
       ,'N'                                                                 Conf_Usu
 FROM Dbamv.mvto_kit_produzido  Mvto_Estoque
      ,Dbamv.Itmvto_kit_produzido  ItMvto_Estoque
      ,Dbamv.Produto
      ,Dbamv.Uni_Pro
WHERE Mvto_Estoque.Cd_Mvto_Estoque = ItMvto_Estoque.Cd_Mvto_Estoque
  AND Produto.Cd_Produto = Itmvto_Estoque.Cd_Produto
  AND Uni_Pro.Cd_Uni_Pro = Itmvto_Estoque.Cd_Uni_Pro
  AND Mvto_Estoque.Cd_SolSai_Pro = :CAMPO_CD_SOLSAI_PRO
  AND ItMvto_Estoque.Cd_Produto Not in ( Select Cd_Produto
                                           From Dbamv.ItSolSai_Pro
                                          Where Cd_SolSai_Pro = :CAMPO_CD_SOLSAI_PRO)
 And Exists ( Select Empresa_Produto.Cd_Produto
                     From Dbamv.Empresa_Produto
                   Where Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
                        and Empresa_Produto.Cd_Produto = Produto.Cd_Produto )
ORDER BY Cd_Produto
