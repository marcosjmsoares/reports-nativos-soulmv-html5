   AND ESTOQUE.CD_MULTI_EMPRESA                = DBAMV.PKG_MV2000.LE_EMPRESA()
   AND SolSai_Pro.Cd_Estoque_Solicitante       = Estoque_Solicitante.Cd_Estoque(+)
   AND SolSai_Pro.Cd_Atendimento               = Atendime.Cd_Atendimento(+)
   And SolSai_Pro.Cd_Setor                     = Setor.Cd_Setor(+)
   AND NVL(SolSai_Pro.Cd_Setor,'99999999')    = NVL(Decode(:CAMPO_CD_SETOR, '%',  SolSai_Pro.Cd_Setor, :CAMPO_CD_SETOR),'99999999')
   AND SolSai_Pro.Cd_Unid_Int                  = Unid_Int.Cd_Unid_Int(+)
   AND NVL(SolSai_Pro.Cd_Unid_Int,'99999999') = NVL(Decode(:CAMPO_UNID_INT, '%',  SolSai_Pro.Cd_Unid_Int, :CAMPO_UNID_INT),'99999999')
   And SolSai_Pro.Cd_Prestador                 = Prestador.Cd_Prestador(+)
   And SolSai_Pro.Cd_SolSai_Pro                = ItSolSai_Pro.Cd_SolSai_Pro
   And Atendime.Cd_Paciente                    = Paciente.Cd_Paciente(+)
   And Atendime.Cd_Convenio                    = Convenio.Cd_Convenio(+)
   And Atendime.Cd_Leito                       = Leito.Cd_Leito(+)
   And ItSolSai_Pro.Cd_Produto                 = Produto.Cd_Produto(+)

   And ItPre_Med.Cd_Tip_Fre                    = Tip_Fre.Cd_Tip_Fre(+)
   And ItPre_Med.Cd_Tip_Presc                  = Tip_Presc.Cd_Tip_Presc(+)
   And ItPre_Med.Cd_Unidade                    = Unidade.Cd_Unidade(+)
   And ItPre_Med.Cd_Pre_Med                    = Pre_Med.Cd_Pre_Med(+)
   And atendime.Cd_Multi_Empresa(+)            = DBAMV.PKG_MV2000.LE_EMPRESA()
   AND ITPRE_MED.CD_TIP_ESQ                    = TIP_ESQ.CD_TIP_ESQ(+)
   AND ITPRE_MED.CD_UNI_PRO_INF                = UNI_PRO_INF.CD_UNI_PRO(+)
   And Est_Pro.Cd_Produto                      =  Produto.Cd_Produto
   And Est_Pro.Cd_Estoque                      = SolSai_Pro.Cd_Estoque
   And Est_Pro.Cd_Estoque                      = Localizacao.Cd_Estoque(+)
   And Est_Pro.Cd_Localizacao                  = Localizacao.Cd_Localizacao(+)
   AND SolSai_Pro.Tp_SolSai_Pro                = Decode(:CAMPO_TP_SOLICITACAO, '%',  SolSai_Pro.Tp_SolSai_Pro  , :CAMPO_TP_SOLICITACAO)
   AND ItPre_Med.CD_FOR_APL                    = for_apl.CD_FOR_APL(+)
   AND SOLSAI_PRO.DT_SOLSAI_PRO + (SOLSAI_PRO.HR_SOLSAI_PRO - Trunc(SOLSAI_PRO.HR_SOLSAI_PRO)) BETWEEN :CAMPO_DT_INICIO order by SolSai_Pro.Cd_Estoque, SolSai_Pro.Cd_Setor, Atendime.Ds_Atendimento, Produto.Ds_Produto
