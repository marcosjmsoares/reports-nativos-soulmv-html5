SELECT ent_transf.CD_PRODUTO,
       ent_transf.CD_ENT_PRO,
       ent_transf.DT_ENTRADA,
       ent_transf.cd_fornecedor,
       fornecedor.nm_fornecedor,
       tipo_doc.ds_tip_doc,
       produto.ds_produto,
       ent_transf.cd_tip_doc,
       ent_transf.nr_documento,
       ent_transf.ds_especie,
       ent_transf.vl_ent_transf,
       ent_transf.vl_cmm,
       ent_transf.cd_multi_empresa_entrada,
       (SELECT ds_multi_empresa FROM multi_empresas WHERE multi_empresas.cd_multi_empresa = ent_transf.cd_multi_empresa_entrada) ds_multi_empresa_entrada,
       sai_transf.cd_multi_empresa_saida,
       (SELECT ds_multi_empresa FROM multi_empresas WHERE multi_empresas.cd_multi_empresa = sai_transf.cd_multi_empresa_saida) ds_multi_empresa_saida,
       sai_transf.dt_saida,
       sai_transf.cd_mvto_estoque,
       sai_transf.vl_saida_transf,
       sai_transf.cd_estoque
FROM dbamv.v_entrada_transf_emp ent_transf,
     dbamv.tip_doc tipo_doc,
     dbamv.fornecedor,
     dbamv.produto,
     dbamv.v_saida_transf_emp sai_transf
WHERE ent_transf.cd_ent_pro = sai_transf.cd_ent_pro
AND tipo_doc.cd_tip_doc = ent_transf.cd_tip_doc
AND produto.cd_produto = ent_transf.CD_PRODUTO
AND fornecedor.cd_fornecedor = ent_transf.cd_fornecedor
AND sai_transf.cd_multi_empresa_saida = pkg_mv2000.le_empresa
AND sai_transf.dt_saida between :CAMPO_DTINIC ORDER BY  sai_transf.cd_mvto_estoque, ent_transf.CD_ENT_PRO, ent_transf.cd_fornecedor,ent_transf.ds_especie,ent_transf.CD_PRODUTO
