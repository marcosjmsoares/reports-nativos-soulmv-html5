--R_CLAS_PROD_ESP
SELECT * FROM (
SELECT especie.cd_especie,
       especie.ds_especie,
       classe.cd_classe,
       classe.ds_classe,
       sub_clas.cd_sub_cla,
       sub_clas.ds_sub_cla,
       produto.cd_produto codigo,
       produto.ds_produto descricao,
       produto.dt_cadastro,
       produto.tp_ativo,
       produto.ds_especificacao,
       produto.dt_ultima_entrada,
       DECODE(NVL(DBAMV.PKT_CONFIGEST.RETORNA_CAMPO('SN_LOTE_VAL_PRD_EMP'),'N'),'S',EMP.sn_controle_validade, PRODUTO.sn_controle_validade) sn_controle_validade,
       produto.sn_pscotropico,
       verif_ds_unid_prod( produto.cd_produto ) ds_unidade,
       verif_vl_fator_prod( produto.cd_produto ) vl_fator,
       produto.vl_custo_medio,
       produto.vl_margem_lucro,
       produto.vl_preco_de_venda,
       produto.vl_tempo_de_reposicao,
       produto.sn_mestre,
       produto.sn_movimentacao,
       produto.hr_ultima_entrada,
       produto.sn_medicamento,
       produto.tp_classificacao_xyz,
       produto.sn_bloqueio_de_compra,
       produto.cd_tip_ativ,
       DECODE(NVL(DBAMV.PKT_CONFIGEST.RETORNA_CAMPO('SN_LOTE_VAL_PRD_EMP'),'N'),'S',EMP.SN_LOTE, PRODUTO.SN_LOTE) sn_lote,
       produto.sn_imprime_etiqueta,
       produto_filho.cd_produto            codigo_filho,
       produto_filho.ds_produto            descricao_filho,
       produto_filho.dt_cadastro           dt_cadastro_filho,
       produto_filho.tp_ativo              tp_ativo_filho,
       produto_filho.ds_especificacao      ds_especificacao_filho,
       produto_filho.dt_ultima_entrada     dt_ultima_entrada_filho,
       DECODE(NVL(DBAMV.PKT_CONFIGEST.RETORNA_CAMPO('SN_LOTE_VAL_PRD_EMP'),'N'),'S',EMP_FILHO.sn_controle_validade, produto_filho.sn_controle_validade)  sn_controle_validade_filho,
       produto_filho.sn_pscotropico        sn_pscotropico_filho,
       produto_filho.vl_custo_medio        vl_custo_medio_filho,
       produto_filho.vl_margem_lucro       vl_margem_lucro_filho,
       produto_filho.vl_preco_de_venda     vl_preco_de_venda_filho,
       produto_filho.vl_tempo_de_reposicao vl_tempo_de_reposicao_filho,
       verif_ds_unid_prod( produto_filho.cd_produto ) ds_unidade_filho,
       verif_vl_fator_prod( produto_filho.cd_produto ) vl_fator_filho,
       produto_filho.sn_mestre             sn_mestre_filho,
       produto_filho.sn_movimentacao       sn_movimentacao_filho,
       produto_filho.hr_ultima_entrada     sn_ultima_entrada_filho,
       produto_filho.sn_medicamento        sn_medicamento_filho,
       produto_filho.tp_classificacao_xyz  tp_classificacao_xyz_filho,
       produto_filho.sn_bloqueio_de_compra sn_bloqueio_de_compra_filho,
       produto_filho.cd_tip_ativ           cd_tip_ativ_filho,
       DECODE(NVL(DBAMV.PKT_CONFIGEST.RETORNA_CAMPO('SN_LOTE_VAL_PRD_EMP'),'N'),'S',EMP_FILHO.SN_LOTE, produto_filho.SN_LOTE) sn_lote_filho,
       produto_filho.sn_imprime_etiqueta   sn_imprime_etiqueta_filho
FROM   dbamv.especie,
       dbamv.classe,
       dbamv.sub_clas,
       dbamv.produto ,
       dbamv.produto produto_filho,
	   Dbamv.Empresa_Produto EMP,
	   Dbamv.Empresa_Produto EMP_FILHO
WHERE  especie.cd_especie      = classe.cd_especie
AND    EMP.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND    EMP.Cd_Produto = Produto.Cd_Produto
AND    EMP_FILHO.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND    EMP_FILHO.Cd_Produto = produto_filho.Cd_Produto
AND    classe.cd_especie       = sub_clas.cd_especie
AND    classe.cd_classe        = sub_clas.cd_classe
AND    sub_clas.cd_especie     = produto.cd_especie
AND    sub_clas.cd_classe      = produto.cd_classe
AND    sub_clas.cd_sub_cla     = produto.cd_sub_cla
AND    produto.cd_produto      = produto_filho.cd_produto_tem(+)
AND    produto.cd_produto_tem is null
AND    produto.cd_especie LIKE   :CAMPO_ESPECIE
AND    produto.cd_classe LIKE     :CAMPO_CLASSE
AND    produto.cd_sub_cla LIKE   :CAMPO_SUB_CLASSE
ORDER BY especie.ds_especie,
                    produto.ds_produto,
                    produto_filho.ds_produto
)
ORDER BY DS_ESPECIE,CD_ESPECIE,
DESCRICAO,CD_CLASSE,DS_CLASSE,CD_SUB_CLA,DS_SUB_CLA,CODIGO,DT_CADASTRO,VL_CUSTO_MEDIO,DT_ULTIMA_ENTRADA,HR_ULTIMA_ENTRADA,VL_MARGEM_LUCRO,VL_PRECO_DE_VENDA,VL_TEMPO_DE_REPOSICAO,TP_CLASSIFICACAO_XYZ,SN_BLOQUEIO_DE_COMPRA,VL_FATOR,DS_UNIDADE,SN_PSCOTROPICO,SN_MOVIMENTACAO,SN_MESTRE,SN_MEDICAMENTO,SN_IMPRIME_ETIQUETA,CD_TIP_ATIV,TP_ATIVO
