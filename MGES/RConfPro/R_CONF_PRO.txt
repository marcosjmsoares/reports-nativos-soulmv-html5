--R_CONF_PRO
SELECT * FROM (
SELECT   PRODUTO.CD_PRODUTO  CD_PRODUTO
            ,   PRODUTO.DS_PRODUTO   DS_PRODUTO
            ,   PRODUTO.CD_PRO_FAT    CD_PRO_FAT
            ,   PRO_FAT.DS_PRO_FAT      DS_PRO_FAT
            ,   PRO_FAT.DS_UNIDADE      DS_UNIDADE
            ,   verif_ds_unid_prod (produto.cd_produto) unid
            ,   verif_vl_fator_prod(produto.cd_produto) VL_FATOR
FROM     DBAMV.PRODUTO                PRODUTO
           ,   DBAMV.PRO_FAT                  PRO_FAT
Where Exists ( SELECT Empresa_Produto.Cd_Produto FROM Dbamv.Empresa_Produto WHERE Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA AND Empresa_Produto.Cd_Produto =
Produto.Cd_Produto )
:CAMPO_CF_CD_ESPECIE
)
ORDER BY CD_PRODUTO,DS_PRODUTO,UNID, VL_FATOR
