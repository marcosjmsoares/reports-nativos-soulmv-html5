--R_BAL_EST_CONF
Select * from
( Select  bal_est.cd_bal_est
     ,  bal_est.cd_estoque
     ,  estoque.ds_estoque
     ,  bal_est.nm_realizador
     ,  bal_est.nm_responsavel_conferencia    nm_resp_confer
     ,  bal_est.dt_bal_est
     ,  bal_est.ds_bal_est
     ,  itbal_est.cd_produto
     ,  produto.ds_produto
     ,  especie.cd_especie
     ,  especie.ds_especie
     , Nvl( Uni_Pro.Ds_Unidade, verif_ds_unid_prod( itbal_Est.cd_produto ) ) ds_unidade
     ,  itbal_est.qt_estoque qt_estoque
     ,  ( verif_vl_custo_medio(produto.cd_produto,dt_bal_est,'H',null,hr_bal_est)  * itbal_est.qt_estoque * Nvl( Uni_Pro.Vl_Fator, 1 ) )        vl_custo_medio
     ,  verif_vl_custo_medio(produto.cd_produto,dt_bal_est,'H',null,hr_bal_Est) * Nvl( Uni_Pro.Vl_Fator, verif_vl_fator_prod( itbal_Est.cd_produto ) ) vl_unitario
     ,  itbal_est.qt_contagem qt_contagem
  From  dbamv.bal_est
     ,  dbamv.estoque
     ,  dbamv.itbal_est
     ,  dbamv.produto
     ,  dbamv.uni_pro
     ,  dbamv.especie
where bal_est.cd_estoque    =  estoque.cd_estoque
   and  bal_est.cd_bal_est    =  itbal_est.cd_bal_est
   and  itbal_est.cd_produto  =  produto.cd_produto
   and uni_pro.cd_uni_pro	(+)= itbal_est.cd_uni_pro
   and  especie.cd_especie    =  produto.cd_especie
AND    ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
   :CAMPO_CF_SUB_CLA
)
ORDER BY CD_ESPECIE desc , :CAMPO_CF_ORDEM , NM_RESP_CONFER,DT_BAL_EST,DS_BAL_EST,DS_ESTOQUE,NM_REALIZADOR,CD_ESTOQUE,CD_BAL_EST,
DS_ESPECIE,CD_ESPECIE
