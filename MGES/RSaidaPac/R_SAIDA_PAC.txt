--R_SAIDA_PAC_R_5_Q_1
select * from (
SELECT ITSOLSAI_PRO.CD_PRODUTO               Cd_Produto,
         PRODUTO.DS_PRODUTO                            Ds_Produto,
         UNI_PRO.DS_UNIDADE                                Ds_Unidade,
         sum(Nvl(ITSOLSAI_PRO.QT_SOLICITADO,0))	 Qtd_Solicitada,
         sum(DECODE(NVL(PRODUTO.SN_MESTRE,'N'), 'S', 0,
                    decode(( select distinct cd_produto_atendido
                              from dbamv.itsolsai_pro_atendido
                             where cd_itsolsai_pro = itsolsai_pro.cd_itsolsai_pro
                               and cd_produto_atendido = itsolsai_pro.cd_produto), itsolsai_pro.cd_produto, Nvl(ITSOLSAI_PRO.QT_ATENDIDA,0), 0)))	 Qtd_Atendida,
         sum((Nvl(ITSOLSAI_PRO.QT_SOLICITADO,0) - DECODE(NVL(PRODUTO.SN_MESTRE,'N'), 'S', 0,
                    decode(( select distinct cd_produto_atendido
                              from dbamv.itsolsai_pro_atendido
                             where cd_itsolsai_pro = itsolsai_pro.cd_itsolsai_pro
                               and cd_produto_atendido = itsolsai_pro.cd_produto), itsolsai_pro.cd_produto, Nvl(ITSOLSAI_PRO.QT_ATENDIDA,0), 0)))) Qtd_Diferenca,
         PRODUTO.CD_PRODUTO_TEM                  Dsp_cd_produto_tem,
         PRODUTO.SN_MESTRE                                Dsp_sn_mestre,
         ITSOLSAI_PRO.SN_CONF_DETERM_USU  Conf_Usu
         ,MOTIVO_DIVERG_ATEND.DS_MOTIVO_DIVERG_ATEND
    FROM DBAMV.ITSOLSAI_PRO,
         DBAMV.PRODUTO,
         DBAMV.UNI_PRO,
         DBAMV.SUB_CLAS,
         DBAMV.MOTIVO_DIVERG_ATEND MOTIVO_DIVERG_ATEND
   WHERE ITSOLSAI_PRO.CD_PRODUTO = PRODUTO.CD_PRODUTO
     AND ITSOLSAI_PRO.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO
     AND PRODUTO.CD_ESPECIE      = SUB_CLAS.CD_ESPECIE
     AND PRODUTO.CD_CLASSE       = SUB_CLAS.CD_CLASSE
     AND PRODUTO.CD_SUB_CLA      = SUB_CLAS.CD_SUB_CLA
     AND PRODUTO.SN_KIT <> 'C'
     AND SUB_CLAS.SN_PROD_PERMANENTE <> 'L'
     AND ITSOLSAI_PRO.CD_SOLSAI_PRO = :CAMPO_CD_SOLSAI_PRO
     AND ITSOLSAI_PRO.CD_MOTIVO_DIVERG_ATEND = MOTIVO_DIVERG_ATEND.CD_MOTIVO_DIVERG_ATEND(+)
     And Exists ( Select Empresa_Produto.Cd_Produto
                          From Dbamv.Empresa_Produto
                       Where Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
                           and Empresa_Produto.Cd_Produto = Produto.Cd_Produto )
     AND itsolsai_pro.qt_solicitado <> itsolsai_pro.qt_atendida
     AND NOT EXISTS (SELECT 'X'
                      FROM itmvto_estoque it, mvto_estoque m, formula f
                     WHERE it.cd_produto_kit = itsolsai_pro.cd_produto
                       AND it.cd_mvto_estoque = m.cd_mvto_estoque
                       AND m.cd_solsai_pro = ITSOLSAI_PRO.CD_SOLSAI_PRO
                       AND f.cd_produto = it.cd_produto_kit
                       AND f.cd_formula = it.cd_formula_kit
                     )
GROUP BY  ITSOLSAI_PRO.CD_PRODUTO
        , PRODUTO.DS_PRODUTO
        , UNI_PRO.DS_UNIDADE
        , PRODUTO.CD_PRODUTO_TEM
        , PRODUTO.SN_MESTRE
        , ITSOLSAI_PRO.SN_CONF_DETERM_USU
        , MOTIVO_DIVERG_ATEND.DS_MOTIVO_DIVERG_ATEND
Union All
SELECT ItMvto_Estoque.Cd_Produto                Cd_Produto
       ,Produto.Ds_Produto                                   Ds_Produto
       ,Uni_Pro.Ds_Unidade                                  Ds_Unidade
       ,0	                                                            Qtd_Solicitada
       ,sum(DECODE(NVL(PRODUTO.SN_MESTRE,'N'),'N',Nvl(ItMvto_Estoque.Qt_Movimentacao,0) ,'S',0))  Qtd_Atendida
       ,sum(- Nvl(ItMvto_Estoque.Qt_Movimentacao,0)) Qtd_Diferenca
       ,Produto.Cd_Produto_Tem                           Dsp_cd_produto_tem
       ,Produto.Sn_Mestre                                      Dsp_sn_mestre
       ,'N'                                                                Conf_Usu
       ,'' DS_MOTIVO_DIVERG_ATEND
 FROM Dbamv.Mvto_Estoque
      ,Dbamv.ItMvto_Estoque
      ,Dbamv.Produto
      ,Dbamv.Uni_Pro
      ,Dbamv.Estoque
WHERE Mvto_Estoque.Cd_Mvto_Estoque = ItMvto_Estoque.Cd_Mvto_Estoque
  AND MVTO_ESTOQUE.CD_ESTOQUE  = ESTOQUE.CD_ESTOQUE
  AND ESTOQUE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
  AND Produto.Cd_Produto = Itmvto_Estoque.Cd_Produto
  AND Uni_Pro.Cd_Uni_Pro = Itmvto_Estoque.Cd_Uni_Pro
  AND Mvto_Estoque.Cd_SolSai_Pro = :CAMPO_CD_SOLSAI_PRO
  AND ItMvto_Estoque.Cd_Produto Not in ( Select Cd_Produto
                                                                       From Dbamv.ItSolSai_Pro
																	   Where Cd_SolSai_Pro = :CAMPO_CD_SOLSAI_PRO
Union All
                                          Select P.Cd_Produto
                                                                       From Dbamv.ItSolSai_Pro I,
                                                                            dbamv.produto p
                                                                    Where p.cd_produto_tem = i.cd_produto
                                                                      AND I.Cd_SolSai_Pro = :CAMPO_CD_SOLSAI_PRO)
  AND  NOT exists ( SELECT 'X'
                      FROM itsolsai_pro it
                     WHERE itmvto_estoque.cd_produto_kit = it.cd_produto
                       AND it.cd_solsai_pro = :CAMPO_CD_SOLSAI_PRO )
GROUP BY ItMvto_Estoque.Cd_Produto
       , Produto.Ds_Produto
       , Uni_Pro.Ds_Unidade
       , 0
       , Produto.Cd_Produto_Tem
       , Produto.Sn_Mestre
       , 'N'
ORDER BY Cd_Produto
) ORDER BY CD_PRODUTO
