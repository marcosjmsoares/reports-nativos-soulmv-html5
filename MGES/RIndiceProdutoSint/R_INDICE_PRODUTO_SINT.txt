--R_INDICE_PRODUTO_SINT
SELECT
estoque.cd_estoque
,estoque.ds_estoque
,Count(ent_pro.cd_ent_pro) QT_ENTRADA
			FROM
				dbamv.estoque estoque ,
				dbamv.ent_pro ent_pro,
				dbamv.fornecedor fornecedor,
				dbamv.ent_pro_conclusao conclusao,
				dbamv.tip_doc
			WHERE estoque.cd_estoque = ent_pro.cd_estoque
			AND ent_pro.cd_fornecedor = fornecedor.cd_fornecedor
			AND ent_pro.cd_ent_pro = conclusao.cd_ent_pro
			AND ent_pro.cd_tip_doc = tip_doc.cd_tip_doc
			AND conclusao.dt_conclusao_entrada IS NOT NULL
GROUP BY
		estoque.cd_estoque
		 ,estoque.ds_estoque
order by estoque.cd_estoque
