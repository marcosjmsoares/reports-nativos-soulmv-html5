--R_NOTA_DEVOLUCAO
SELECT *
FROM (SELECT ENT_PRO.cd_ent_pro CD_ENTRADA,
	ENT_PRO.DT_ENTRADA DATA_ENTRADA,
	ENT_PRO.DT_EMISSAO DATA_EMISSAO,
	ENT_PRO.NR_DOCUMENTO,
	ENT_PRO.NR_SERIE,
	ENT_PRO.VL_TOTAL VALOR_TOTAL,
	TIP_DOC.DS_TIP_DOC,
	FORNECEDOR.CD_FORNECEDOR,
	FORNECEDOR.NM_FORNECEDOR,
	FORNECEDOR.nr_cgc_cpf
   FROM dbamv.ENT_PRO,
	      		dbamv.TIP_DOC,
      	   		dbamv.FORNECEDOR
   WHERE  
   			   ENT_PRO.CD_TIP_DOC = TIP_DOC.CD_TIP_DOC 
	 AND ENT_PRO.CD_FORNECEDOR = FORNECEDOR.CD_FORNECEDOR 
	 AND ENT_PRO.CD_ENT_PRO  =      :CAMPO_CD_ENT_PRO 
)

--R_NOTA_DEVOLUCAO_subreport1
select * from (
 SELECT DEV_FOR.CD_DEVOLUCAO ,
                    DEV_FOR.NR_DOCUMENTO ,
                    DEV_FOR.NR_SERIE ,
                    MOT_DEV.CD_MOT_DEV ,
                    MOT_DEV.DS_MOT_DEV ,
                    DEV_FOR.DT_DEVOLUCAO ,
                    DEV_FOR.HR_DEVOLUCAO,
                    DEV_FOR.VL_TOTAL ,
                    ITDEV_FOR.CD_PRODUTO ,
                    ITDEV_FOR.CD_LOTE  ,
                    ITDEV_FOR.dt_validade ,
                    UNI_PRO.ds_unidade ds_unidade_dev,
                    Nvl(CFOP.DS_CFOP,' ')  DS_CFOP,
                    ITDEV_FOR.QT_DEVOLVIDA,
                    ITENT_PRO.VL_UNITARIO
              FROM  dbamv.DEV_FOR ,
                    dbamv.MOT_DEV ,
                    dbamv.ITDEV_FOR ,
                   dbamv.UNI_PRO ,
                    dbamv.ITENT_PRO ,
                    dbamv.CFOP ,
                    dbamv.ENT_PRO
              WHERE DEV_FOR.CD_MOT_DEV = MOT_DEV.CD_MOT_DEV
                AND DEV_FOR.CD_DEVOLUCAO = ITDEV_FOR.CD_DEVOLUCAO
                AND DEV_FOR.CD_ENT_PRO = ENT_PRO.CD_ENT_PRO
                AND ITDEV_FOR.CD_UNI_PRO = UNI_PRO.CD_UNI_PRO (+)
                AND ITDEV_FOR.CD_CFOP = CFOP.CD_CFOP (+)
                AND ITDEV_FOR.CD_ITENT_PRO = ITENT_PRO.CD_ITENT_PRO
			    AND DEV_FOR.CD_ENT_PRO  =    :CAMPO_CD_ENT_PRO 
   )
