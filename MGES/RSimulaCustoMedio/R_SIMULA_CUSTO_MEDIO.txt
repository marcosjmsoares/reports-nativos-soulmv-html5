--R_SIMULA_CUSTO_MEDIO
SELECT * FROM (
		select custo_medio_mensal.cd_produto
      ,produto.ds_produto
      ,to_char(custo_medio_mensal.dh_custo_medio,'MM/YYYY') competencia_simulacao
      ,to_char(add_months(custo_medio_mensal.dh_custo_medio,-1),'MM/YYYY') competencia_antes
	  ,verif_vl_custo_medio (custo_medio_mensal.cd_produto
                           ,last_day(add_months(custo_medio_mensal.dh_custo_medio,-1)) + 0.99999
                           ,'H'
                           ,null
                           ,last_day(add_months(custo_medio_mensal.dh_custo_medio,-1)) + 0.99999
                           ,dbamv.pkg_mv2000.le_empresa) vl_custo_medio_antes
      ,custo_medio_mensal.vl_custo_medio vl_custo_medio_simulado
from dbamv.custo_medio_mensal
    ,dbamv.produto
where custo_medio_mensal.cd_produto   = produto.cd_produto
  and custo_medio_mensal.sn_simulacao = 'S'
order by produto.ds_produto
)
ORDER BY COMPETENCIA_SIMULACAO, ds_produto
