--R_MAPA_SOL_800
select solsai_pro.cd_solsai_pro
      ,solsai_pro.cd_tp_solicitacao
      ,tipo_solicitacao.ds_tp_solicitacao
      ,solsai_pro.cd_atendimento
      ,atendime_hcare.cd_area
      ,area.ds_area
      ,to_char(solsai_pro.dt_solsai_pro, 'dd/mm/yyyy') dt_solsai_pro
      ,to_char(solsai_pro.hr_solsai_pro,'hh24') hr_solsai_pro
      ,solsai_pro.cd_agrupamento
      ,solsai_pro.cd_usuario cd_usuario_SOL
      ,solsai_pro.ds_obs
      ,convenio.cd_convenio
      ,convenio.nm_convenio
      ,paciente.cd_paciente
      ,paciente.nm_paciente
      ,to_char(mvto_estoque.dt_mvto_estoque, 'dd/mm/yyyy') dt_mvto_estoque
      ,to_char(mvto_estoque.hr_mvto_estoque, 'hh24') hr_mvto_estoque
      ,mvto_estoque.cd_mvto_estoque
      ,usuarios.cd_usuario
      ,to_char(nvl(solsai_pro.dt_consumo_inicial, mvto_estoque.dt_consumo_inicial), 'dd/mm/yyyy') dt_consumo_inicial
      ,to_char(nvl(solsai_pro.dt_consumo_final, mvto_estoque.dt_consumo_final), 'dd/mm/yyyy') dt_consumo_final
      ,itmvto_estoque.cd_produto
      ,produto.ds_produto
      ,itmvto_estoque.cd_lote
      ,to_char(itmvto_estoque.dt_validade, 'dd/mm/yyyy') dt_validade
      ,uni_pro.ds_unidade
      ,solsai_pro.cd_pre_med
      ,pre_med.dh_impressao
      ,sum(itmvto_estoque.qt_movimentacao ) qt_movimentacao
      ,ori_ate.ds_ori_ate
  from dbamv.mvto_estoque
      ,dbamv.itmvto_estoque
      ,dbamv.produto
      ,dbamv.uni_pro
      ,dbamv.solsai_pro
      ,dbamv.atendime
      ,dbamv.paciente
      ,dbamv.convenio
      ,dbasgu.usuarios
      ,dbamv.atendime_hcare
      ,dbamv.area
      ,dbamv.tipo_solicitacao
      ,dbamv.pre_med
      ,dbamv.ori_ate
 where itmvto_estoque.cd_mvto_estoque = mvto_estoque.cd_mvto_estoque
   and produto.cd_produto             = itmvto_estoque.cd_produto
   and uni_pro.cd_uni_pro             = itmvto_estoque.cd_uni_pro
   and solsai_pro.cd_solsai_pro       = mvto_estoque.cd_solsai_pro
   and atendime.cd_atendimento        = mvto_estoque.cd_atendimento
   and paciente.cd_paciente           = atendime.cd_paciente
   and convenio.cd_convenio           = atendime.cd_convenio
   and usuarios.cd_usuario            = mvto_estoque.cd_usuario
   and atendime_hcare.cd_atendimento(+) = solsai_pro.cd_atendimento
   and area.cd_area(+)                = atendime_hcare.cd_area
   and tipo_solicitacao.cd_tp_solicitacao(+) = solsai_pro.cd_tp_solicitacao
   and pre_med.cd_pre_med(+) = solsai_pro.cd_pre_med
   and atendime.cd_multi_empresa      = dbamv.pkg_mv2000.le_empresa
   and mvto_estoque.cd_solsai_pro     = :CAMPO_CD_SOLSAI_PRO
   and atendime.cd_ori_ate = ori_ate.cd_ori_ate
   and mvto_estoque.cd_estoque in ( select estoque.cd_estoque
                                      from dbamv.estoque
                                     where estoque.cd_multi_empresa = dbamv.pkg_mv2000.le_empresa)
group by solsai_pro.cd_solsai_pro
      ,solsai_pro.cd_tp_solicitacao
      ,tipo_solicitacao.ds_tp_solicitacao
      ,solsai_pro.cd_atendimento
      ,atendime_hcare.cd_area
      ,area.ds_area
      ,solsai_pro.dt_solsai_pro
      ,solsai_pro.hr_solsai_pro
      ,solsai_pro.cd_agrupamento
      ,solsai_pro.cd_usuario
      ,solsai_pro.ds_obs
      ,convenio.cd_convenio
      ,convenio.nm_convenio
      ,paciente.cd_paciente
      ,paciente.nm_paciente
      ,mvto_estoque.dt_mvto_estoque
      ,mvto_estoque.hr_mvto_estoque
      ,mvto_estoque.cd_mvto_estoque
      ,usuarios.cd_usuario
      ,nvl(solsai_pro.dt_consumo_inicial, mvto_estoque.dt_consumo_inicial)
      ,nvl(solsai_pro.dt_consumo_final, mvto_estoque.dt_consumo_final)
      ,itmvto_estoque.cd_produto
      ,produto.ds_produto
      ,itmvto_estoque.cd_lote
      ,itmvto_estoque.dt_validade
      ,uni_pro.ds_unidade
      ,solsai_pro.cd_pre_med
      ,pre_med.dh_impressao
      ,ori_ate.ds_ori_ate
order by mvto_estoque.dt_mvto_estoque
            ,mvto_estoque.hr_mvto_estoque
            ,mvto_estoque.cd_mvto_estoque
            ,produto.ds_produto
