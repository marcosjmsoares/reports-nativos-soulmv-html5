--R_SOLIC_AGENDAMENTO
SELECT * FROM (
Select atendime.cd_atendimento
      ,atendime.cd_paciente
      ,(SELECT PA.NM_SOCIAL_PACIENTE
        FROM DBAMV.PACIENTE PA
        WHERE PA.CD_PACIENTE = atendime.CD_PACIENTE
          AND NVL(PA.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE
      ,atendime.dt_atendimento dt_atendimento
      ,atendime.cd_cid
      ,atendime.cd_convenio
      ,atendime.cd_prestador
      ,paciente.nm_paciente
      ,paciente.dt_nascimento
      ,paciente.nm_mae
      ,cid.ds_cid
      ,convenio.nm_convenio
      ,servico.ds_servico
      ,prestador.nm_prestador
      ,solic_agendamento.cd_item_agendamento
      ,item_agendamento.ds_item_agendamento
      ,solic_agendamento.dt_solic_agendamento
      ,solic_agendamento.nr_ciclo
      ,solic_agendamento.nr_sessao
      ,solic_agendamento.nr_dia
      ,solic_agendamento.cd_tratamento
      ,solic_agendamento.ds_observacao
      ,ItPre_Med.cd_pre_med
  From Dbamv.Solic_Agendamento
     , Dbamv.ItPre_Med
     , Dbamv.Atendime
     , Dbamv.Paciente
     , Dbamv.Cid
     , Dbamv.Convenio
     , Dbamv.servico
     , Dbamv.Prestador
     ,Dbamv.item_agendamento
 Where Solic_Agendamento.Cd_Atendimento      = Atendime.Cd_Atendimento
   And atendime.cd_paciente                  = paciente.cd_paciente
   and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
   and atendime.cd_cid                       = cid.cd_cid(+)
   and atendime.cd_convenio                  = convenio.cd_convenio(+)
   and atendime.cd_servico                   = servico.cd_servico(+)
   and atendime.cd_prestador                 = prestador.cd_prestador
   and solic_agendamento.cd_item_agendamento = item_agendamento.cd_item_agendamento
   And solic_agendamento.Cd_ItPre_Med        = ItPre_Med.Cd_ItPre_Med(+)
   AND atendime.dt_atendimento BETWEEN nvl(:CAMPO_DT_ATENDIMENTO_INI, atendime.dt_atendimento) AND nvl(:CAMPO_DT_ATENDIMENTO_FIM, atendime.dt_atendimento)
order by solic_agendamento.nr_dia
)
ORDER BY CD_PACIENTE,
         DT_ATENDIMENTO,
         DT_NASCIMENTO,
         CD_ATENDIMENTO,
         NM_PACIENTE,
         NM_MAE,
         CD_CONVENIO,
         NM_CONVENIO,
         DS_SERVICO,
         CD_CID,
         DS_CID,
         CD_PRESTADOR,
         NM_PRESTADOR,
         DS_ITEM_AGENDAMENTO,
         CD_PRE_MED,
         CD_ITEM_AGENDAMENTO,
         NR_CICLO,
         NR_SESSAO,
         NR_DIA,
         DS_OBSERVACAO,
         CD_TRATAMENTO
