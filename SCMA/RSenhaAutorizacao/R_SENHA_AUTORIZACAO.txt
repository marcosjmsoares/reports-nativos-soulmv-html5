--R_SENHA_AUTORIZACAO
SELECT * FROM(SELECT DISTINCT CD_AGENDA_CENTRAL AS CODIGO
	,PACIENTE.CD_PACIENTE
	,PACIENTE.NM_PACIENTE
	,SENHA_AUTORIZACAO.CD_ESPECIALID CD_ESPECIALID
	,IT_AGENDA_CENTRAL.CD_ESPECIALID CD_ESPEC
	,ESPECIALID_PAD.DS_ESPECIALID    DS_ESPECIALID
	,SENHA_AUTORIZACAO.CD_SER_DIS    CD_SER_DIS
	,IT_AGENDA_CENTRAL.CD_SER_DIS    CD_SERDIS
	,SER_DIS_PAD.DS_SER_DIS          DS_SER_DIS
	,SENHA_AUTORIZACAO.CD_SENHA_AUTORIZACAO
	,SENHA_AUTORIZACAO.DS_SENHA
	,ORIGEM_SENHA_AUT.NM_ORIGEM_SENHA_AUT
	,SENHA_AUTORIZACAO.DT_CRIACAO
	,SENHA_AUTORIZACAO.DT_ULTIMA_ATIVACAO
	,SENHA_AUTORIZACAO.DT_ULTIMA_DESATIVACAO
	,SENHA_AUTORIZACAO.DT_EXPIRACAO
	,'AGEND' AS AGENDAMENTO
	,'AMB' AS TIPO 
	,CASE 
	WHEN SENHA_AUTORIZACAO.DT_EXPIRACAO <= SYSDATE THEN 'EXPIRADA'
	WHEN SENHA_AUTORIZACAO.DT_ULTIMA_ATIVACAO < SENHA_AUTORIZACAO.DT_ULTIMA_DESATIVACAO THEN 'INATIVA'
	WHEN SENHA_AUTORIZACAO.DT_ULTIMA_DESATIVACAO  < (DT_ULTIMA_ATIVACAO) THEN 'ATIVA'
	WHEN SENHA_AUTORIZACAO.DT_ULTIMA_DESATIVACAO IS NULL THEN 'ATIVA' 
	END AS SITUACAO
FROM DBAMV.SENHA_AUTORIZACAO
	,DBAMV.ORIGEM_SENHA_AUT
	,DBAMV.PACIENTE
	,DBAMV.CONVENIO
	,DBAMV.IT_AGENDA_CENTRAL
	,DBAMV.SER_DIS       SER_DIS_PAD
	,DBAMV.SER_DIS       SER_DIS_UTL
	,DBAMV.ESPECIALID    ESPECIALID_PAD
	,DBAMV.ESPECIALID    ESPECIALID_UTL
 WHERE SENHA_AUTORIZACAO.CD_PACIENTE = PACIENTE.CD_PACIENTE
	AND SENHA_AUTORIZACAO.CD_ORIGEM_SENHA_AUT = ORIGEM_SENHA_AUT.CD_ORIGEM_SENHA_AUT
	AND IT_AGENDA_CENTRAL.CD_SENHA_AUTORIZACAO = SENHA_AUTORIZACAO.CD_SENHA_AUTORIZACAO
	AND IT_AGENDA_CENTRAL.CD_IT_AGENDA_PAI IS NULL
	AND IT_AGENDA_CENTRAL.CD_PACIENTE = PACIENTE.CD_PACIENTE
	AND IT_AGENDA_CENTRAL.CD_CONVENIO = CONVENIO.CD_CONVENIO(+) 
	AND SENHA_AUTORIZACAO.CD_ESPECIALID = ESPECIALID_PAD.CD_ESPECIALID(+)
	AND SENHA_AUTORIZACAO.CD_SER_DIS = SER_DIS_PAD.CD_SER_DIS(+)
	AND IT_AGENDA_CENTRAL.CD_ESPECIALID = ESPECIALID_UTL.CD_ESPECIALID(+)
	AND IT_AGENDA_CENTRAL.CD_SER_DIS = SER_DIS_UTL.CD_SER_DIS(+)
	AND SENHA_AUTORIZACAO.DT_CRIACAO BETWEEN :CAMPO_DT_INICIAL AND 
	AND 'S' = :CAMPO_SN_IMPRIME_AGENDAMENTOS
UNION ALL
SELECT DISTINCT ATENDIME.CD_ATENDIMENTO AS CODIGO
	,PACIENTE.CD_PACIENTE
	,PACIENTE.NM_PACIENTE
	,SENHA_AUTORIZACAO.CD_ESPECIALID CD_ESPECIALID
	,ATENDIME.CD_ESPECIALID          CD_ESPEC
	,ESPECIALID_PAD.DS_ESPECIALID    DS_ESPECIALID
	,SENHA_AUTORIZACAO.CD_SER_DIS    CD_SER_DIS
	,ATENDIME.CD_SER_DIS             CD_SERDIS
	,SER_DIS_PAD.DS_SER_DIS          DS_SER_DIS
	,SENHA_AUTORIZACAO.CD_SENHA_AUTORIZACAO
	,SENHA_AUTORIZACAO.DS_SENHA
	,ORIGEM_SENHA_AUT.NM_ORIGEM_SENHA_AUT
	,SENHA_AUTORIZACAO.DT_CRIACAO
	,SENHA_AUTORIZACAO.DT_ULTIMA_ATIVACAO
	,SENHA_AUTORIZACAO.DT_ULTIMA_DESATIVACAO
	,SENHA_AUTORIZACAO.DT_EXPIRACAO
	,'ATEND'
	,CASE 
	WHEN ATENDIME.TP_ATENDIMENTO = 'A' THEN 'AMB'
	WHEN ATENDIME.TP_ATENDIMENTO = 'E' THEN 'EXT'
	WHEN ATENDIME.TP_ATENDIMENTO = 'I' THEN 'INT' 
	END AS TIPO
	,CASE 
	WHEN SENHA_AUTORIZACAO.DT_EXPIRACAO <= SYSDATE THEN 'EXPIRADA'
	WHEN SENHA_AUTORIZACAO.DT_ULTIMA_ATIVACAO < SENHA_AUTORIZACAO.DT_ULTIMA_DESATIVACAO THEN 'INATIVA'
	WHEN SENHA_AUTORIZACAO.DT_ULTIMA_DESATIVACAO < (DT_ULTIMA_ATIVACAO) THEN 'ATIVA'
	WHEN SENHA_AUTORIZACAO.DT_ULTIMA_DESATIVACAO IS NULL THEN 'ATIVA' 
	END AS SITUACAO
FROM DBAMV.SENHA_AUTORIZACAO
	,DBAMV.ORIGEM_SENHA_AUT
	,DBAMV.PACIENTE
	,DBAMV.CONVENIO        
	,DBAMV.ATENDIME
	,DBAMV.SER_DIS      SER_DIS_PAD
	,DBAMV.ESPECIALID   ESPECIALID_PAD
	,DBAMV.SER_DIS      SER_DIS_UTL 
	,DBAMV.ESPECIALID   ESPECIALID_UTL
WHERE SENHA_AUTORIZACAO.CD_PACIENTE = PACIENTE.CD_PACIENTE
	AND SENHA_AUTORIZACAO.CD_ORIGEM_SENHA_AUT = ORIGEM_SENHA_AUT.CD_ORIGEM_SENHA_AUT
	AND ATENDIME.CD_SENHA_AUTORIZACAO = SENHA_AUTORIZACAO.CD_SENHA_AUTORIZACAO
	AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
	AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO(+)
	AND SENHA_AUTORIZACAO.CD_ESPECIALID = ESPECIALID_PAD.CD_ESPECIALID(+)
	AND SENHA_AUTORIZACAO.CD_SER_DIS = SER_DIS_PAD.CD_SER_DIS(+)
	AND ATENDIME.CD_ESPECIALID = ESPECIALID_UTL.CD_ESPECIALID(+)
	AND ATENDIME.CD_SER_DIS = SER_DIS_UTL.CD_SER_DIS(+)
	AND SENHA_AUTORIZACAO.DT_CRIACAO BETWEEN :CAMPO_DT_INICIAL UNION ALL
SELECT DISTINCT AGE_CIR.CD_AGE_CIR AS CODIGO      
	,PACIENTE.CD_PACIENTE
	,PACIENTE.NM_PACIENTE
	,SENHA_AUTORIZACAO.CD_ESPECIALID CD_ESPECIALID
	,CIRURGIA_AVISO.CD_ESPECIALID    CD_ESPEC
	,ESPECIALID_PAD.DS_ESPECIALID    DS_ESPECIALID
	,SENHA_AUTORIZACAO.CD_SER_DIS    CD_SER_DIS
	,AGE_CIR.CD_SER_DIS              CD_SERDIS
	,SER_DIS_PAD.DS_SER_DIS          DS_SER_DIS    
	,SENHA_AUTORIZACAO.CD_SENHA_AUTORIZACAO
	,SENHA_AUTORIZACAO.DS_SENHA
	,ORIGEM_SENHA_AUT.NM_ORIGEM_SENHA_AUT
	,SENHA_AUTORIZACAO.DT_CRIACAO
	,SENHA_AUTORIZACAO.DT_ULTIMA_ATIVACAO
	,SENHA_AUTORIZACAO.DT_ULTIMA_DESATIVACAO
	,SENHA_AUTORIZACAO.DT_EXPIRACAO
	,'AGEND' AS AGENDAMENTO
	,'CIR' AS TIPO
	,CASE 
	WHEN SENHA_AUTORIZACAO.DT_EXPIRACAO <= SYSDATE THEN 'EXPIRADA'
	WHEN SENHA_AUTORIZACAO.DT_ULTIMA_ATIVACAO < SENHA_AUTORIZACAO.DT_ULTIMA_DESATIVACAO THEN 'INATIVA'
	WHEN SENHA_AUTORIZACAO.DT_ULTIMA_DESATIVACAO  < SENHA_AUTORIZACAO.DT_ULTIMA_ATIVACAO THEN 'ATIVA'
	WHEN SENHA_AUTORIZACAO.DT_ULTIMA_DESATIVACAO IS NULL THEN 'ATIVA' 
	END AS SITUACAO
FROM DBAMV.SENHA_AUTORIZACAO
	,DBAMV.ORIGEM_SENHA_AUT
	,DBAMV.PACIENTE
	,DBAMV.AVISO_CIRURGIA
	,DBAMV.CIRURGIA_AVISO
	,DBAMV.CONVENIO
	,DBAMV.AGE_CIR
	,DBAMV.SER_DIS      SER_DIS_PAD
	,DBAMV.ESPECIALID   ESPECIALID_PAD
	,DBAMV.SER_DIS      SER_DIS_UTL
	,DBAMV.ESPECIALID   ESPECIALID_UTL
WHERE SENHA_AUTORIZACAO.CD_PACIENTE = PACIENTE.CD_PACIENTE
	AND AVISO_CIRURGIA.CD_PACIENTE = PACIENTE.CD_PACIENTE
	AND SENHA_AUTORIZACAO.CD_ORIGEM_SENHA_AUT = ORIGEM_SENHA_AUT.CD_ORIGEM_SENHA_AUT
	AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
	AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = AGE_CIR.CD_AVISO_CIRURGIA
	AND SENHA_AUTORIZACAO.CD_SENHA_AUTORIZACAO = CIRURGIA_AVISO.CD_SENHA_AUTORIZACAO
	AND CIRURGIA_AVISO.CD_CONVENIO = CONVENIO.CD_CONVENIO(+)
	AND SENHA_AUTORIZACAO.CD_ESPECIALID = ESPECIALID_PAD.CD_ESPECIALID(+)
	AND SENHA_AUTORIZACAO.CD_SER_DIS = SER_DIS_PAD.CD_SER_DIS(+)
	AND CIRURGIA_AVISO.CD_ESPECIALID = ESPECIALID_UTL.CD_ESPECIALID(+)
	AND AGE_CIR.CD_SER_DIS = SER_DIS_UTL.CD_SER_DIS(+)
	AND SENHA_AUTORIZACAO.DT_CRIACAO BETWEEN :CAMPO_DT_INICIAL AND 
	AND 'S' = :CAMPO_SN_IMPRIME_AGENDAMENTOS
UNION ALL
SELECT DISTINCT AGE_CIR.CD_AGE_CIR AS CODIGO      
	,PACIENTE.CD_PACIENTE
	,PACIENTE.NM_PACIENTE
	,SENHA_AUTORIZACAO.CD_ESPECIALID CD_ESPECIALID
	,CIRURGIA_AVISO.CD_ESPECIALID    CD_ESPEC
	,ESPECIALID_PAD.DS_ESPECIALID    DS_ESPECIALID
	,SENHA_AUTORIZACAO.CD_SER_DIS    CD_SER_DIS
	,AGE_CIR.CD_SER_DIS              CD_SERDIS
	,SER_DIS_PAD.DS_SER_DIS          DS_SER_DIS    
	,SENHA_AUTORIZACAO.CD_SENHA_AUTORIZACAO
	,SENHA_AUTORIZACAO.DS_SENHA
	,ORIGEM_SENHA_AUT.NM_ORIGEM_SENHA_AUT
	,SENHA_AUTORIZACAO.DT_CRIACAO
	,SENHA_AUTORIZACAO.DT_ULTIMA_ATIVACAO
	,SENHA_AUTORIZACAO.DT_ULTIMA_DESATIVACAO
	,SENHA_AUTORIZACAO.DT_EXPIRACAO
	,'AGEND' AS AGENDAMENTO
	,'CIR' AS TIPO
	,CASE 
	WHEN SENHA_AUTORIZACAO.DT_EXPIRACAO <= SYSDATE THEN 'EXPIRADA'
	WHEN SENHA_AUTORIZACAO.DT_ULTIMA_ATIVACAO < SENHA_AUTORIZACAO.DT_ULTIMA_DESATIVACAO THEN 'INATIVA'
	WHEN SENHA_AUTORIZACAO.DT_ULTIMA_DESATIVACAO  < SENHA_AUTORIZACAO.DT_ULTIMA_ATIVACAO THEN 'ATIVA'
	WHEN SENHA_AUTORIZACAO.DT_ULTIMA_DESATIVACAO IS NULL THEN 'ATIVA' 
	END AS SITUACAO
FROM DBAMV.SENHA_AUTORIZACAO
	,DBAMV.ORIGEM_SENHA_AUT
	,DBAMV.PACIENTE
	,DBAMV.AVISO_CIRURGIA
	,DBAMV.CIRURGIA_AVISO
	,DBAMV.CONVENIO
	,DBAMV.AGE_CIR
	,DBAMV.SER_DIS      SER_DIS_PAD
	,DBAMV.ESPECIALID   ESPECIALID_PAD
	,DBAMV.SER_DIS      SER_DIS_UTL
	,DBAMV.ESPECIALID   ESPECIALID_UTL
WHERE SENHA_AUTORIZACAO.CD_PACIENTE = PACIENTE.CD_PACIENTE
	AND AVISO_CIRURGIA.CD_PACIENTE = PACIENTE.CD_PACIENTE
	AND SENHA_AUTORIZACAO.CD_ORIGEM_SENHA_AUT = ORIGEM_SENHA_AUT.CD_ORIGEM_SENHA_AUT
	AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
	AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = AGE_CIR.CD_AVISO_CIRURGIA
	AND SENHA_AUTORIZACAO.CD_SENHA_AUTORIZACAO = AGE_CIR.CD_SENHA_AUTORIZACAO
	AND CIRURGIA_AVISO.CD_CONVENIO = CONVENIO.CD_CONVENIO(+)
	AND SENHA_AUTORIZACAO.CD_ESPECIALID = ESPECIALID_PAD.CD_ESPECIALID(+)
	AND SENHA_AUTORIZACAO.CD_SER_DIS = SER_DIS_PAD.CD_SER_DIS(+)
	AND CIRURGIA_AVISO.CD_ESPECIALID = ESPECIALID_UTL.CD_ESPECIALID(+)
	AND AGE_CIR.CD_SER_DIS = SER_DIS_UTL.CD_SER_DIS(+)
	AND SENHA_AUTORIZACAO.DT_CRIACAO BETWEEN :CAMPO_DT_INICIAL AND 
	AND 'S' = :CAMPO_SN_IMPRIME_AGENDAMENTOS	) SENHAS
ORDER BY SENHAS.NM_PACIENTE, SENHAS.DS_SENHA
