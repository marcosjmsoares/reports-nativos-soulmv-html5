--R_PAC_AGENDADOS_CONVENIO
SELECT * FROM (
select vdic.data_agenda
      ,vdic.hora_agenda
      ,vdic.cod_horario_agenda
      ,vdic.cod_paciente
      ,vdic.nm_paciente
      ,vdic.cod_grupo_agendamento
      ,vdic.ds_grupo_agendamento
      ,dbamv.pack_internamento.retorna_idade(vdic.data_nascimento,sysdate) idade_paciente
      ,vdic.peso_paciente
      ,vdic.cod_convenio
      ,vdic.nm_convenio
      ,vdic.cod_prestador
      ,vdic.nm_prestador
      ,vdic.cod_recurso
      ,vdic.ds_recurso
      ,vdic.cod_setor
      ,vdic.nm_setor
      ,vdic.fone_paciente
      ,(vdic.observacao || chr(10) || vdic.observacao_geral) observacao
      ,vdic.cod_unidade_atendimento
      ,vdic.ds_unidade_atendimento
      ,vdic.cod_servico
      ,vdic.ds_servico
      ,vdic.cod_usuario_agendamento
      ,vdic.nm_usuario_agendamento
      ,vdic.cod_item_agendamento
      ,vdic.ds_item_agendamento
      ,vdic.cod_atendimento
from dbamv.vdic_horarios_agenda_scma vdic
where nvl(vdic.nm_paciente,vdic.cod_grupo_agendamento) is not null
and nvl(vdic.tipo_situacao,'M') not in ('C','A')
and nvl(vdic.sn_agenda_ativa,'N') = 'S'
and vdic.data_agenda between :CAMPO_DT_INICIAL order by nm_convenio, trunc(vdic.data_agenda), to_date(to_char(vdic.hora_agenda,'hh24'),'hh24')
)
ORDER BY NM_CONVENIO,COD_CONVENIO,
DATA_AGENDA
