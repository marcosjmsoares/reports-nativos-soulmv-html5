--R_ITEM_RECURSO
SELECT * FROM (
select item_agendamento.cd_item_agendamento
      ,item_agendamento.ds_item_agendamento
      ,recurso_central.cd_recurso_central
      ,recurso_central.ds_recurso_central
      ,decode(recurso_central.tp_recurso,'E','EQUIPAMENTO'
                                        ,'S','SALA'
                                        ,'P','PRESTADOR'
                                        ,'O','OUTROS') tp_recurso
from dbamv.item_agendamento
    ,dbamv.recurso_central
    ,dbamv.item_agendamento_recurso
where recurso_central.cd_recurso_central = item_agendamento_recurso.cd_recurso_central
  and item_agendamento.cd_item_agendamento = item_agendamento_recurso.cd_item_agendamento
order by item_agendamento.ds_item_agendamento
            ,recurso_central.ds_recurso_central
)
ORDER BY DS_ITEM_AGENDAMENTO,CD_ITEM_AGENDAMENTO
