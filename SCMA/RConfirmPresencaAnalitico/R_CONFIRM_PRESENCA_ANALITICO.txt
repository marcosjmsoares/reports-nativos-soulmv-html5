--R_CONFIRM_PRESENCA_ANALITICO
SELECT AGENDA_CENTRAL.CD_AGENDA_CENTRAL
     , AGENDA_CENTRAL.CD_RECURSO_CENTRAL
     , (SELECT DS_RECURSO_CENTRAL FROM DBAMV.RECURSO_CENTRAL WHERE CD_RECURSO_CENTRAL = AGENDA_CENTRAL.CD_RECURSO_CENTRAL) DS_RECURSO_CENTRAL
     , AGENDA_CENTRAL.CD_PRESTADOR
     , (SELECT NM_PRESTADOR FROM DBAMV.PRESTADOR WHERE CD_PRESTADOR = AGENDA_CENTRAL.CD_PRESTADOR) NM_PRESTADOR
     , AGENDA_CENTRAL.CD_SETOR
     , (SELECT NM_SETOR FROM DBAMV.SETOR WHERE CD_SETOR = AGENDA_CENTRAL.CD_SETOR) NM_SETOR
     , AGENDA_CENTRAL.CD_UNIDADE_ATENDIMENTO
     , (SELECT DS_UNIDADE_ATENDIMENTO FROM DBAMV.UNIDADE_ATENDIMENTO WHERE CD_UNIDADE_ATENDIMENTO = AGENDA_CENTRAL.CD_UNIDADE_ATENDIMENTO) DS_UNIDADE_ATENDIMENTO
     , IT_AGENDA_CENTRAL.CD_SER_DIS
     , (SELECT DS_SER_DIS FROM DBAMV.SER_DIS WHERE CD_SER_DIS = IT_AGENDA_CENTRAL.CD_SER_DIS) DS_SER_DIS
     , IT_AGENDA_CENTRAL.CD_PACIENTE
     , IT_AGENDA_CENTRAL.NM_PACIENTE
     , (SELECT PA.NM_SOCIAL_PACIENTE
        FROM DBAMV.PACIENTE PA
        WHERE PA.CD_PACIENTE = IT_AGENDA_CENTRAL.CD_PACIENTE
          AND NVL(PA.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE
     , AGENDA_CENTRAL.DT_AGENDA
     , AGENDA_CENTRAL.HR_INICIO
     , AGENDA_CENTRAL.HR_FIM
     , IT_AGENDA_CENTRAL.CD_ITEM_AGENDAMENTO
     , (SELECT DS_ITEM_AGENDAMENTO FROM DBAMV.ITEM_AGENDAMENTO WHERE CD_ITEM_AGENDAMENTO = IT_AGENDA_CENTRAL.CD_ITEM_AGENDAMENTO) DS_ITEM_AGENDAMENTO
     , Decode(IT_AGENDA_CENTRAL.TP_SITUACAO, 'M', :CAMPO_MSG_MARCADO, 'A', :CAMPO_MSG_AGUARDANDO, 'E', :CAMPO_MSG_ATENDIDO, 'C', :CAMPO_MSG_CANCELADO) TP_SITUACAO
     , Decode(IT_AGENDA_CENTRAL.TP_PRESENCA_FALTA, 'P', :CAMPO_MSG_PRESENCA, 'F', :CAMPO_MSG_FALTA, (CASE WHEN IT_AGENDA_CENTRAL.TP_SITUACAO = 'C' THEN  END)) TP_PRESENCA
     , IT_AGENDA_CENTRAL.TP_PRESENCA_FALTA
FROM DBAMV.AGENDA_CENTRAL
	 , DBAMV.IT_AGENDA_CENTRAL
WHERE AGENDA_CENTRAL.CD_AGENDA_CENTRAL         = IT_AGENDA_CENTRAL.CD_AGENDA_CENTRAL
  AND AGENDA_CENTRAL.SN_FALTA = 'N'
  AND IT_AGENDA_CENTRAL.SN_BLOQUEADO = 'N'
  AND (IT_AGENDA_CENTRAL.cd_grupo_agenda is not null or IT_AGENDA_CENTRAL.Nm_Paciente is not null or IT_AGENDA_CENTRAL.cd_paciente is not null)
GROUP BY AGENDA_CENTRAL.CD_AGENDA_CENTRAL
       , AGENDA_CENTRAL.CD_RECURSO_CENTRAL
       , AGENDA_CENTRAL.CD_PRESTADOR
       , AGENDA_CENTRAL.CD_SETOR
       , AGENDA_CENTRAL.CD_UNIDADE_ATENDIMENTO
       , IT_AGENDA_CENTRAL.CD_SER_DIS
       , IT_AGENDA_CENTRAL.CD_PACIENTE
       , IT_AGENDA_CENTRAL.NM_PACIENTE
       , AGENDA_CENTRAL.DT_AGENDA
       , AGENDA_CENTRAL.HR_INICIO
       , AGENDA_CENTRAL.HR_FIM
       , IT_AGENDA_CENTRAL.CD_ITEM_AGENDAMENTO
       , IT_AGENDA_CENTRAL.TP_SITUACAO
       , IT_AGENDA_CENTRAL.TP_PRESENCA_FALTA
ORDER BY 1
