--R_PERCENTUAL_ATENDIMENTOS
SELECT * FROM (
SELECT  distinct
       codigo_servico
      ,descricao_servico
      ,codigo_prestador
      ,nome_prestador
      ,codigo_agenda
      ,data_agenda
      ,horario_inicial
      ,horario_final
      ,capacidade_agenda
      ,codigo_setor
  FROM (
         SELECT  agenda_central_ser_tipo.cd_ser_dis  codigo_servico
                ,ser_dis.ds_ser_dis                  descricao_servico
                ,agenda_central.cd_prestador         codigo_prestador
                ,prestador.nm_prestador              nome_prestador
                ,agenda_central.cd_agenda_central    codigo_agenda
                ,agenda_central.dt_agenda            data_agenda
                ,agenda_central.hr_inicio            horario_inicial
                ,agenda_central.hr_fim               horario_final
                ,agenda_central.qt_atendimento       capacidade_agenda
                ,agenda_central.cd_setor             codigo_setor
           FROM  dbamv.agenda_central
                ,dbamv.prestador
                ,dbamv.agenda_central_ser_tipo
                ,dbamv.ser_dis
           WHERE agenda_central.cd_prestador     = prestador.cd_prestador
             AND agenda_central.cd_agenda_central = agenda_central_ser_tipo.cd_agenda_central
             AND agenda_central_ser_tipo.cd_ser_dis = ser_dis.cd_ser_dis
             AND dt_agenda BETWEEN Trunc(:CAMPO_DT_INICIAL) AND Trunc(:CAMPO_DT_FINAL)
        )
where 1=1
:CAMPO_CF_COND_CD_SER_DIS
)
ORDER BY CODIGO_SERVICO,DESCRICAO_SERVICO,
CODIGO_PRESTADOR,NOME_PRESTADOR
