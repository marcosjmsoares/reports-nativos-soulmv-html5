--R_COMPROVANTE_AGENDA_TURMA
Select Hospital.Ds_Razao, Hospital.Ds_Rodape_Reports,
 'N' Sn_Imprime_Logotipo, Hospital.Logotipo
  From Dbamv.Hospital
 Where Hospital.Cd_Multi_Empresa = nvl(:CAMPO_CD_MULTI_EMPRESA,1)


/*----------------------------------------------------------------------------*/
--R_COMPROVANTE_AGENDA_TURMA_R_G_CD_PACIENTE_Q_AGENDAMENTO
SELECT * FROM (
Select     agenda_turma.Cd_Agenda_turma                                     Cd_Agenda_turma,
           agenda_turma.Cd_Unidade_Atendimento                              Cd_Unidade_Atendimento,
           turma.cd_turma                                                   cd_turma,
           agenda_turma_prestador.Cd_Prestador                              Cd_Prestador,
           agenda_turma.Cd_Setor                                            Cd_Setor,
           agenda_turma.Dt_Agenda                                           Dt_Agenda,
           Prestador.nm_prestador                                           nm_prestador,
           turma.ds_turma                                                   ds_turma1,
           unidade_atendimento.ds_unidade_atendimento                       ds_unidade_atendimento,
           setor.nm_setor                                                   nm_setor,
           agenda_turma.dt_horario_inicial                                  dt_horario_inicial,
           agenda_turma.dt_horario_final                                    dt_horario_final,
           agenda_turma_paciente.cd_paciente                                cd_paciente,
           CASE WHEN dbamv.agenda_turma_paciente.cd_paciente IS NULL THEN dbamv.agenda_turma_paciente.nm_paciente
            WHEN dbamv.agenda_turma_paciente.cd_paciente IS NOT NULL THEN (SELECT nm_paciente FROM dbamv.paciente WHERE cd_paciente = dbamv.agenda_turma_paciente.cd_paciente)
       		  END AS nm_paciente,
           (SELECT PA.NM_SOCIAL_PACIENTE
              FROM DBAMV.PACIENTE PA
              WHERE PA.CD_PACIENTE = paciente.CD_PACIENTE
                AND NVL(PA.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE,
           Nvl(agenda_turma_paciente.nr_telefone_contato, paciente.nr_fone) Nr_fone,
           Nvl(agenda_turma_paciente.Dt_Nascimento, Paciente.dt_nascimento) dt_nascimento,
           Nvl(Paciente.email, agenda_turma_paciente.ds_email)              ds_email,
           Nvl(agenda_turma_paciente.Tp_Sexo, Paciente.tp_sexo)             tp_sexo,
           agenda_turma_paciente.cd_convenio                                cd_convenio,
           convenio.nm_convenio                                             nm_convenio,
           convenio.tp_convenio                                             tp_convenio,
           agenda_turma_paciente.cd_con_pla                                 cd_con_pla,
           con_pla.ds_con_pla                                               ds_con_pla,
           agenda_turma.Cd_Ser_Dis                                          Cd_Ser_Dis,
           agenda_turma.Cd_Tip_Mar                                          Cd_Tip_Mar,
           trunc(agenda_turma_paciente.dt_agendamento)          Dt_Gravacao,
           ser_dis.ds_ser_dis                                               ds_ser_dis,
           tip_mar.ds_tip_mar                                               ds_tip_mar,
           usuarios.nm_usuario                                              Usuario_agendamento
   from    dbamv.agenda_turma_paciente,
           dbamv.paciente,
           dbamv.ser_dis,
           dbamv.tip_mar,
           dbamv.convenio,
           dbamv.con_pla,
           dbasgu.usuarios,
           dbamv.agenda_turma,
           dbamv.prestador,
           dbamv.agenda_turma_prestador,
           dbamv.Unidade_atendimento,
           dbamv.Setor,
           dbamv.turma
Where   turma.cd_turma                         = agenda_turma.cd_turma
  and   agenda_turma.cd_agenda_turma           = agenda_turma_prestador.cd_agenda_turma
  and   agenda_turma_prestador.cd_prestador    = prestador.cd_prestador (+)
  and   agenda_turma.cd_agenda_turma           = agenda_turma_paciente.cd_agenda_turma
  and   agenda_turma.cd_ser_dis                = ser_dis.cd_ser_dis (+)
  and   agenda_turma.cd_tip_mar                = tip_mar.cd_tip_mar (+)
  and   agenda_turma.cd_unidade_atendimento    = unidade_atendimento.cd_unidade_atendimento (+)
  and   agenda_turma.cd_setor                  = setor.cd_setor
  and   agenda_turma_paciente.cd_paciente      = paciente.cd_paciente (+)
  and   agenda_turma_paciente.cd_convenio      = convenio.cd_convenio
  and   agenda_turma_paciente.cd_con_pla       = con_pla.cd_con_pla
  and   con_pla.cd_convenio                    = convenio.cd_convenio
  and   agenda_turma_paciente.cd_usuario       = usuarios.cd_usuario
  and   agenda_turma_prestador.sn_principal    = 'S'
:CAMPO_CF_WHERE_AGENDA_PAC
)
ORDER BY CD_PACIENTE,NM_PACIENTE,NR_FONE,DT_NASCIMENTO,DS_EMAIL,TP_SEXO,DT_GRAVACAO,USUARIO_AGENDAMENTO,
DT_AGENDA
