--R_POSICAO_FILA_ESPERA
SELECT F.NM_PACIENTE NM_PACIENTE,
		(SELECT PA.NM_SOCIAL_PACIENTE
        FROM DBAMV.PACIENTE PA
        WHERE PA.CD_PACIENTE = F.CD_PACIENTE
          AND NVL(PA.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE, 
		S.NM_SETOR NM_SETOR, 
		P.DS_PRIORIDADE DS_PRIORID, 
		R.DS_RECURSO_CENTRAL DS_RECURSO_CENTRAL,
		PR.NM_PRESTADOR NM_PRESTADOR,
		I.DS_ITEM_AGENDAMENTO DS_ITEM_AGENDAMENTO,
		U.DS_UNIDADE_ATENDIMENTO DS_UNIDADE_ATENDIMENTO,
		S.DS_SERVICO,
		Decode(F.TP_STATUS,'G','Agendado', 'C', 'Cancelado', 'A', 'Aguardando') TP_STATUS
 FROM DBAMV.AGENDA_FILA_ESPERA F
     ,DBAMV.PRIORIDADE P
     ,DBAMV.SETOR  S
     ,RECURSO_CENTRAL R
     ,PRESTADOR PR
     ,ITEM_AGENDAMENTO I
     ,UNIDADE_ATENDIMENTO U
     ,DBAMV.servico S
WHERE F.CD_PRIORIDADE = P.CD_PRIORIDADE(+)
  AND F.CD_SETOR   = S.CD_SETOR(+)
  AND F.CD_RECURSO_CENTRAL   = R.CD_RECURSO_CENTRAL(+)
  AND F.CD_PRESTADOR   = PR.CD_PRESTADOR(+)
  AND F.CD_ITEM_AGENDAMENTO   = I.CD_ITEM_AGENDAMENTO(+)
  AND F.CD_UNIDADE_ATENDIMENTO   = U.CD_UNIDADE_ATENDIMENTO(+)
  AND F.CD_SER_DIS = S.CD_SERVICO(+)
