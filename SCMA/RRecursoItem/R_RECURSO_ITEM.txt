--R_RECURSO_ITEM
SELECT * FROM (
select recurso_central.cd_recurso_central
      ,recurso_central.ds_recurso_central
      ,item_agendamento.cd_item_agendamento
      ,item_agendamento.ds_item_agendamento
      ,decode(item_agendamento.tp_item,'L', :CAMPO_TRADUCAO_LABORATORIAL
                                        ,'A', :CAMPO_TRADUCAO_AMBULATORIAL
                                        ,'I', :CAMPO_TRADUCAO_IMAGEM) tp_item
from dbamv.item_agendamento
    ,dbamv.recurso_central
    ,dbamv.item_agendamento_recurso
where recurso_central.cd_recurso_central = item_agendamento_recurso.cd_recurso_central
  and item_agendamento.cd_item_agendamento = item_agendamento_recurso.cd_item_agendamento
order by recurso_central.ds_recurso_central
           , item_agendamento.ds_item_agendamento
)
ORDER BY DS_RECURSO_CENTRAL,CD_RECURSO_CENTRAL,DS_ITEM_AGENDAMENTO,CD_ITEM_AGENDAMENTO,TP_ITEM
