--R_AGENDA_CANC_PRESTADOR
SELECT * FROM (
select vdic.data_agenda
      ,vdic.hora_agenda
      ,vdic.cod_paciente
      ,vdic.nm_paciente
      ,(SELECT PA.NM_SOCIAL_PACIENTE
        FROM DBAMV.PACIENTE PA
        WHERE PA.CD_PACIENTE = vdic.COD_PACIENTE
          AND NVL(PA.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE
      ,vdic.cod_grupo_agendamento
      ,vdic.ds_grupo_agendamento
      ,vdic.cod_prestador
      ,vdic.nm_prestador
      ,vdic.cod_recurso
      ,vdic.ds_recurso
      ,vdic.cod_setor
      ,vdic.nm_setor
      ,vdic.cod_servico
      ,vdic.ds_servico
      ,vdic.cod_item_agendamento
      ,vdic.ds_item_agendamento
      ,vdic.cod_horario_agenda
from dbamv.vdic_horarios_agenda_scma vdic
where vdic.tipo_situacao = 'C'
and data_agenda between :CAMPO_DT_INICIAL order by nm_prestador, ds_recurso, trunc(vdic.data_agenda), to_date(to_char(vdic.hora_agenda,'hh24'),'hh24')
)
ORDER BY NM_PRESTADOR,COD_PRESTADOR,COD_RECURSO,DS_RECURSO
