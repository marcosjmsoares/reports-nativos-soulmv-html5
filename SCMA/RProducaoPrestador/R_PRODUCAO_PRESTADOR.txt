--R_PRODUCAO_PRESTADOR
SELECT * FROM (
select vdic.data_agenda
      ,vdic.hora_agenda
      ,vdic.cod_paciente
      ,vdic.nm_paciente
      ,(SELECT PA.NM_SOCIAL_PACIENTE
        FROM DBAMV.PACIENTE PA
        WHERE PA.CD_PACIENTE = vdic.cod_paciente
          AND NVL(PA.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE
      ,vdic.cod_convenio
      ,vdic.nm_convenio
      ,vdic.cod_prestador
      ,vdic.nm_prestador
      ,vdic.cod_grupo_agendamento
      ,vdic.ds_grupo_agendamento
      ,vdic.cod_recurso
      ,vdic.ds_recurso
      ,vdic.cod_setor
      ,vdic.nm_setor
      ,Nvl(vdic.observacao, vdic.observacao_geral) observacao
      ,vdic.cod_item_agendamento
      ,vdic.ds_item_agendamento
from dbamv.vdic_horarios_agenda_scma vdic
where (nvl(vdic.tipo_situacao,'M') not in ('C','A') or vdic.cod_atendimento is not null)
and vdic.data_agenda between :CAMPO_DT_INICIAL and 
:CAMPO_CF_COND_CD_RECURSO
)
ORDER BY  NM_PRESTADOR, COD_PRESTADOR, trunc(data_agenda), to_date(to_char(hora_agenda,'hh24'),'hh24'),COD_RECURSO,DS_RECURSO
