--R_ITEM_AGENDAMENTO
select item_agendamento.cd_item_agendamento
      ,item_agendamento.ds_item_agendamento
      ,decode(item_agendamento.tp_item,'A',:CAMPO_TRADUCAO_AMBULATORIAL
                                      ,'I',:CAMPO_TRADUCAO_IMAGEM
                                      ,'L',:CAMPO_TRADUCAO_LABORATORIAL) tp_item
      ,item_agendamento.cd_exa_lab
      ,exa_lab.nm_exa_lab
      ,item_agendamento.cd_exa_rx
      ,exa_rx.ds_exa_rx
      ,nvl(to_char(item_agendamento.hr_realizacao, 'hh24'),'00') hr_realizacao
      ,decode(item_agendamento.sn_ativo,'S', :CAMPO_TRADUCAO_SIM
                                       ,'N',  :CAMPO_TRADUCAO_NAO) sn_ativo
      ,decode(item_agendamento.sn_checa_guia,'S', :CAMPO_TRADUCAO_SIM
                                            ,'N', :CAMPO_TRADUCAO_NAO) sn_checa_guia
from dbamv.item_agendamento
    ,dbamv.exa_lab
    ,dbamv.exa_rx
where item_agendamento.cd_exa_lab = exa_lab.cd_exa_lab(+)
  and item_agendamento.cd_exa_rx = exa_rx.cd_exa_rx(+)
