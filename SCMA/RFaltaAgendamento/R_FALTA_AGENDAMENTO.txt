--R_FALTA_AGENDAMENTO
SELECT
cd_paciente COD_PACIENTE1
, nm_paciente NM_PACIENTE1
, prestador.nm_prestador NM_PRESTADOR1
, recurso_central.ds_recurso_central DS_RECURSO_CENTRAL
, ds_item_agendamento CD_IT_AGENDA_CENTRAL
, dh_presenca_falta HORA_AGENDA1
, dt_gravacao DATA_AGENDA1
, agenda_central.cd_prestador COD_PRESTADOR1
FROM recurso_central, agenda_central, it_agenda_central, prestador, item_agendamento 
WHERE it_agenda_central.cd_it_agenda_pai IS NULL 
AND it_agenda_central.tp_presenca_falta = 'F'
AND it_agenda_central.cd_agenda_central = agenda_central.cd_agenda_central
AND agenda_central.cd_prestador = prestador.cd_prestador
AND agenda_central.cd_recurso_central = recurso_central.cd_recurso_central(+)
AND it_agenda_central.cd_item_agendamento = item_agendamento.cd_item_agendamento(+)
AND TRUNC(dh_presenca_falta) between :CAMPO_DT_INICIAL ORDER BY dh_presenca_falta desc
