--R_COMPROVANTE_ONCOLOGICO_Q_1
SELECT AGENDAMENTO_ONCOLOGICO.CD_AGENDAMENTO_ONCOLOGICO,
       AGENDAMENTO_ONCOLOGICO.DH_INICIO_AGENDAMENTO_ONC AS HR_INICIO,
       AGENDAMENTO_ONCOLOGICO.DH_FINAL_AGENDAMENTO_ONC AS HR_FIM,
       PACIENTE.DT_NASCIMENTO,
       PACIENTE.NR_DDD_FONE,
       PACIENTE.NR_DDI_FONE,
       PACIENTE.NR_FONE,
       PACIENTE.EMAIL AS DS_EMAIL,
       DECODE(PACIENTE.TP_SEXO, 'M', 'MASC.', 'F', 'FEM.', 'I', 'IND.', 'G', 'IGN.', 'NP') AS TP_SEXO,
       DECODE(PACIENTE.SN_UTIL_IDENTIDADE_GENERO, 'S', DECODE(
                                                                (SELECT MAX(1)
                                                                 FROM DBASGU.CONFIG_ITEM_MODULO
                                                                 WHERE CD_MODULO = 'CAD_PAC'
                                                                   AND SN_VISIBLE = 'S'
                                                                   AND CD_ITEM = 'DSP_DS_IDENTIDADE_GENERO'
                                                                   AND CD_BLOCO = 'PACIENTE') ,1, IDENTIDADE_GENERO.DS_IDENTIDADE_GENERO , NULL), NULL) AS IDENTIDADE_GENERO ,
       DECODE(PACIENTE.SN_UTIL_ORIENTACAO_SEXUAL, 'S', DECODE(
                                                                (SELECT MAX(1)
                                                                 FROM DBASGU.CONFIG_ITEM_MODULO
                                                                 WHERE CD_MODULO = 'CAD_PAC'
                                                                   AND SN_VISIBLE = 'S'
                                                                   AND CD_ITEM = 'DSP_DS_ORIENTACAO_SEXUAL'
                                                                   AND CD_BLOCO = 'PACIENTE' ),1, ORIENTACAO_SEXUAL.DS_ORIENTACAO_SEXUAL, NULL), NULL) AS ORIENTACAO_SEXUAL,
       AGENDAMENTO_ONCOLOGICO.NR_DIA NR_DIA,
       AGENDAMENTO_ONCOLOGICO.NR_SESSAO NR_SESSAO,
       AGENDAMENTO_ONCOLOGICO.NR_CICLO NR_CICLO,
       AGENDAMENTO_ONCOLOGICO.TP_STATUS TP_SITUACAO,
       DECODE(AGENDAMENTO_ONCOLOGICO.TP_STATUS, 'A', 'Aguardando autorização', 'G', 'Agendado', 'R', 'Reagendado', 'C', 'Cancelado') TP_SITUACAO_DESC,
       AGENDAMENTO_ONCOLOGICO.CD_PACIENTE CD_PACIENTE,
       DECODE(PACIENTE.SN_UTILIZA_NOME_SOCIAL, 'S', NM_SOCIAL_PACIENTE, NM_PACIENTE) NM_PACIENTE,
       'Sessão: ' || AGENDAMENTO_ONCOLOGICO.NR_SESSAO || ' de ' || NVL(
                                                                         (SELECT MAX(AGDO.NR_SESSAO)
                                                                          FROM DBAMV.AGENDAMENTO_ONCOLOGICO AGDO
                                                                          WHERE AGDO.NR_CICLO = AGENDAMENTO_ONCOLOGICO.NR_CICLO
                                                                            AND (AGDO.CD_AGENDAMENTO_ONCOLOGICO_PAI = AGENDAMENTO_ONCOLOGICO.CD_AGENDAMENTO_ONCOLOGICO
                                                                                 OR AGDO.CD_AGENDAMENTO_ONCOLOGICO in
                                                                                   (SELECT CD_AGENDAMENTO_ONCOLOGICO
                                                                                    FROM DBAMV.AGENDAMENTO_ONCOLOGICO AGDO2
                                                                                    WHERE AGDO2.CD_AGENDAMENTO_ONCOLOGICO_PAI = AGENDAMENTO_ONCOLOGICO.CD_AGENDAMENTO_ONCOLOGICO_PAI))), AGENDAMENTO_ONCOLOGICO.NR_SESSAO) || ' (Ciclo ' || AGENDAMENTO_ONCOLOGICO.NR_CICLO || ')' DS_SESSAO,
       CONVENIO.NM_CONVENIO NM_CONVENIO,
       RECURSO_ONCOLOGICO.DS_RECURSO_ONCOLOGICO AS DS_RECURSO,
       AGENDAMENTO_ONCOLOGICO.CD_PROTOCOLO CD_PROTOCOLO,
       PROTOCOLO.DS_PROTOCOLO NM_PROTOCOLO,
       NVL(
             (SELECT NVL(CTACP.QT_DIAS_AUTORIZACAO, CTAC.QT_DIAS_AUTORIZACAO)
              FROM DBAMV.PW_CONFIG_TEMPO_AUTZ_CONVENIO CTAC, DBAMV.PW_CONFIG_TEMPO_AUTZ_CONV_PROT CTACP
              WHERE CTAC.CD_CONFIG_TEMPO_AUTZ_CONVENIO = CTACP.CD_CONFIG_TEMPO_AUTZ_CONVENIO(+)
                AND CTAC.CD_CONVENIO = CONVENIO.CD_CONVENIO
                AND CTACP.CD_PROTOCOLO (+) = PROTOCOLO.CD_PROTOCOLO), 0) AS QT_DIAS_AUTORIZACAO,
       DECODE(AGENDAMENTO_ONCOLOGICO.TP_STATUS, 'A', 'Atendido', 'C', 'Cancelado', 'G', 'Agendado', 'F', 'Falta') TP_STATUS_DESC,
       AGENDAMENTO_ONCOLOGICO.TP_STATUS TP_STATUS,
       AGENDAMENTO_ONCOLOGICO.DS_OBSERVACAO DS_OBSERVACAO_GERAL,
       ITEM_AGENDAMENTO.DS_ITEM_AGENDAMENTO,
       CONVENIO.NM_CONVENIO,
       (SELECT MAX (NM_PRESTADOR) 
       FROM DBAMV.PRESTADOR 
       WHERE PRESTADOR.CD_PRESTADOR = SOLIC_AGENDAMENTO.CD_PRESTADOR ) AS NM_PRESTADOR
FROM DBAMV.AGENDAMENTO_ONCOLOGICO,
     DBAMV.RECURSO_ONCOLOGICO,
     DBAMV.PACIENTE,
     DBAMV.CONVENIO,
     DBAMV.PROTOCOLO,
     DBAMV.ITEM_AGENDAMENTO,
     DBAMV.ORIENTACAO_SEXUAL,
     DBAMV.IDENTIDADE_GENERO,
     DBAMV.SOLIC_AGENDAMENTO
WHERE AGENDAMENTO_ONCOLOGICO.CD_RECURSO_ONCOLOGICO = RECURSO_ONCOLOGICO.CD_RECURSO_ONCOLOGICO(+)
  AND AGENDAMENTO_ONCOLOGICO.CD_PACIENTE = PACIENTE.CD_PACIENTE
  AND AGENDAMENTO_ONCOLOGICO.CD_CONVENIO = CONVENIO.CD_CONVENIO (+)
  AND AGENDAMENTO_ONCOLOGICO.CD_PROTOCOLO = PROTOCOLO.CD_PROTOCOLO (+)
  AND RECURSO_ONCOLOGICO.CD_MULTI_EMPRESA=:CAMPO_CD_MULTI_EMPRESA
  AND AGENDAMENTO_ONCOLOGICO.TP_STATUS = 'G'
  AND AGENDAMENTO_ONCOLOGICO.CD_ITEM_AGENDAMENTO = ITEM_AGENDAMENTO.CD_ITEM_AGENDAMENTO(+)
  AND PACIENTE.CD_ORIENTACAO_SEXUAL = ORIENTACAO_SEXUAL.CD_ORIENTACAO_SEXUAL(+)
  AND PACIENTE.CD_IDENTIDADE_GENERO = IDENTIDADE_GENERO.CD_IDENTIDADE_GENERO (+)
  AND AGENDAMENTO_ONCOLOGICO.CD_SOLIC_AGENDAMENTO = SOLIC_AGENDAMENTO.CD_SOLIC_AGENDAMENTO(+)
 AND AGENDAMENTO_ONCOLOGICO.CD_AGENDAMENTO_ONCOLOGICO IN  (      (   SELECT   REGEXP_SUBSTR(:CAMPO_CD_AGENDAMENTO_ONCOLOGICO  , '[^,]+', 1, LEVEL) FROM DUAL CONNECT BY REGEXP_SUBSTR(:CAMPO_CD_AGENDAMENTO_ONCOLOGICO, '[^,]+', 1, LEVEL) IS NOT NULL     )   )
ORDER BY PACIENTE.CD_PACIENTE,CD_AGENDAMENTO_ONCOLOGICO
