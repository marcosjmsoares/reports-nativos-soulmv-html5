SELECT LOG_OPERA_AGENDA_CENTRAL.CD_PACIENTE,
	LOG_OPERA_AGENDA_CENTRAL.NM_PACIENTE,
	(SELECT PA.NM_SOCIAL_PACIENTE
        FROM DBAMV.PACIENTE PA
        WHERE PA.CD_PACIENTE = LOG_OPERA_AGENDA_CENTRAL.CD_PACIENTE
          AND NVL(PA.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE,
	NVL((SELECT P.NR_FONE FROM PACIENTE P WHERE P.CD_PACIENTE = LOG_OPERA_AGENDA_CENTRAL.CD_PACIENTE),'') NR_FONE,
	NVL((SELECT PR.NM_PRESTADOR FROM PRESTADOR PR WHERE PR.CD_PRESTADOR = LOG_OPERA_AGENDA_CENTRAL.CD_PRESTADOR),'') NM_PRESTADOR,
	NVL((SELECT R.DS_RECURSO_CENTRAL FROM RECURSO_CENTRAL R WHERE R.CD_RECURSO_CENTRAL = LOG_OPERA_AGENDA_CENTRAL.CD_RECURSO_CENTRAL),'') DS_RECURSO_CENTRAL,
	NVL((SELECT C.NM_CONVENIO FROM CONVENIO C WHERE C.CD_CONVENIO = IT_AGENDA_CENTRAL.CD_CONVENIO),'') NM_CONVENIO,
	NVL((SELECT T.DS_TIP_MAR FROM TIP_MAR T WHERE T.CD_TIP_MAR = IT_AGENDA_CENTRAL.CD_TIP_MAR),'') DS_TIP_MAR,
	NVL((SELECT s.DS_SER_DIS FROM SER_DIS S WHERE S.CD_SER_DIS = IT_AGENDA_CENTRAL.CD_SER_DIS),'') DS_SER_DIS,
	ITEM_AGENDAMENTO.DS_ITEM_AGENDAMENTO,
	LOG_OPERA_AGENDA_CENTRAL.TP_OPERACAO,
	LOG_OPERA_AGENDA_CENTRAL.CD_AGENDA_CENTRAL,
	LOG_OPERA_AGENDA_CENTRAL.CD_AGENDA_CENTRAL_ANTERIOR,
	TO_CHAR(LOG_OPERA_AGENDA_CENTRAL.DT_AGENDA, 'DD/MM/YYYY HH24') DT_AGENDA,
	TO_CHAR(LOG_OPERA_AGENDA_CENTRAL.DT_ANTERIOR, 'DD/MM/YYYY HH24') DT_ANTERIOR,
	LOG_OPERA_AGENDA_CENTRAL.DT_OPERA_AGENDA,
	LOG_OPERA_AGENDA_CENTRAL.CD_USUARIO
FROM DBAMV.LOG_OPERA_AGENDA_CENTRAL
	INNER JOIN DBAMV.ITEM_AGENDAMENTO ON (ITEM_AGENDAMENTO.CD_ITEM_AGENDAMENTO = LOG_OPERA_AGENDA_CENTRAL.CD_ITEM_AGENDAMENTO)
	LEFT OUTER JOIN DBAMV.IT_AGENDA_CENTRAL ON (IT_AGENDA_CENTRAL.CD_IT_AGENDA_CENTRAL = LOG_OPERA_AGENDA_CENTRAL.CD_IT_AGENDA_CENTRAL)
WHERE LOG_OPERA_AGENDA_CENTRAL.DT_OPERA_AGENDA BETWEEN TO_DATE(TO_CHAR(trunc(:CAMPO_DT_INICIAL), 'DD/MM/YYYY') || ' 00' ,'DD/MM/YYYY  hh24') AND TO_DATE(TO_CHAR(trunc(:CAMPO_DT_FINAL), 'DD/MM/YYYY') || ' 23','DD/MM/YYYY hh24')
	AND LOG_OPERA_AGENDA_CENTRAL.TP_OPERACAO = 'T'
ORDER BY LOG_OPERA_AGENDA_CENTRAL.NM_PACIENTE
