--R_AGENDA_PRESTADOR_SCMA
SELECT * FROM (
select data_agenda
      ,cod_prestador
      ,nm_prestador
      ,cod_recurso
      ,ds_recurso
      ,cod_setor
      ,ds_setor
      ,tipo_agenda
      ,hora_inicio
      ,hora_fim
      ,(to_date(to_char(hora_fim,'hh24'),'hh24') - to_date(to_char(hora_inicio,'hh24'),'hh24')) horas_trabalhadas
      ,cod_agenda
      ,tempo_medio
      ,(to_date(to_char(hora_fim,'hh24'),'hh24') - to_date(to_char(hora_inicio,'hh24'),'hh24')) * 24  horas_calculadas
from dbamv.vdic_agenda_scma
where nvl(agenda_ativa,'NAO') = 'SIM'
and (cod_prestador is not null or cod_recurso is not null)
and data_agenda between :CAMPO_DT_INICIAL and 
:CAMPO_CF_COND_CD_RECURSO
)
ORDER BY DATA_AGENDA, NM_PRESTADOR
