Select
    Paciente.Nm_Paciente,
    Ped_Lab.Cd_Ped_Lab Codigo_Pedido,
    Ender_Coleta_Entrega.Vl_Servico,
    dbamv.pack_fncp.extensos(Ender_Coleta_Entrega.Vl_Servico) Vl_Servico_extenso,
    To_Char(Sysdate,'dd/mm/yyyy hh24') Data_Hora,
    USER USUARIO
  From
    Dbamv.Ped_Lab,
    Dbamv.Atendime,
    Dbamv.Paciente,
    Dbamv.Ender_Coleta_Entrega
 Where
    Ped_Lab.Cd_Atendimento = Atendime.Cd_Atendimento And
    Atendime.Cd_Paciente   = Paciente.Cd_Paciente    And
    Ped_Lab.Cd_Ped_Lab     = Ender_Coleta_Entrega.Cd_Ped_Lab And
    Ender_Coleta_Entrega.Tp_Servico = 'C' And
    Ped_Lab.Cd_Ped_lab     =  :CAMPO_PCD_PED_LAB
    and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
