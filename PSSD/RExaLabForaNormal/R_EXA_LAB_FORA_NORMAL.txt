--R_EXA_LAB_FORA_NORMAL
SELECT * FROM
  (SELECT 
          PACIENTE.NM_PACIENTE NM_PACIENTE,
     (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
          PACIENTE.DS_ENDERECO DS_ENDERECO,
          PACIENTE.NR_FONE NR_FONE,
          PACIENTE.TP_SEXO TP_SEXO,
          PACIENTE.DT_NASCIMENTO DT_NASCIMENTO,
          PED_LAB.CD_PED_LAB CD_PED_LAB,
          ITPED_LAB.CD_ITPED_LAB CD_ITPED_LAB,
          EXA_LAB.NM_EXA_LAB NM_EXA_LAB,
          EXA_LAB.TP_RESULTADO TP_RESULTADO,
          RES_EXA.NM_CAMPO NM_CAMPO,
          RES_EXA.DS_RESULTADO DS_RESULTADO,
          VAL_EXA.TP_CAMPO TP_CAMPO,
          VAL_EXA.CD_VERSAO CD_VERSAO,
          VAL_REFERENCIA.VL_REFERENCIA_INICIAL VL_REFERENCIA_INICIAL,
          VAL_REFERENCIA.VL_REFERENCIA_FINAL VL_REFERENCIA_FINAL
   FROM DBAMV.PED_LAB,
        DBAMV.ITPED_LAB,
        DBAMV.EXA_LAB,
        DBAMV.RES_EXA,
        DBAMV.VAL_EXA,
        DBAMV.VAL_REFERENCIA,
        DBAMV.ATENDIME,
        DBAMV.PACIENTE
   WHERE PED_LAB.CD_PED_LAB = ITPED_LAB.CD_PED_LAB
     AND ITPED_LAB.CD_EXA_LAB = EXA_LAB.CD_EXA_LAB
     AND ITPED_LAB.CD_VERSAO = RES_EXA.CD_VERSAO
     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
     AND ITPED_LAB.CD_ITPED_LAB = RES_EXA.CD_ITPED_LAB
     AND ITPED_LAB.CD_EXA_LAB = RES_EXA.CD_EXA_LAB
     AND RES_EXA.CD_VERSAO = VAL_EXA.CD_VERSAO
     AND RES_EXA.NM_CAMPO = VAL_EXA.NM_CAMPO
     AND VAL_EXA.CD_VERSAO = VAL_REFERENCIA.CD_VERSAO
     AND VAL_EXA.NM_CAMPO = VAL_REFERENCIA.NM_CAMPO
     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
     AND ATENDIME.CD_ATENDIMENTO = PED_LAB.CD_ATENDIMENTO
     AND ITPED_LAB.SN_REALIZADO = 'S'
     AND EXA_LAB.TP_RESULTADO = 'F'
     AND (TRUNC(SYSDATE - PACIENTE.DT_NASCIMENTO) BETWEEN (NVL(VAL_REFERENCIA.NR_ANO_INICIAL, 0) * 365) + (NVL(VAL_REFERENCIA.NR_MES_INICIAL, 0) * 30) + (NVL(VAL_REFERENCIA.NR_DIA_INICIAL, 0)) AND (NVL(VAL_REFERENCIA.NR_ANO_FINAL, 0) * 365) + (NVL(VAL_REFERENCIA.NR_MES_FINAL, 0) * 30) + (NVL(VAL_REFERENCIA.NR_DIA_FINAL, 0))
            AND VAL_REFERENCIA.TP_SEXO IN ('A', PACIENTE.TP_SEXO))
     AND DBAMV.VALIDA_VAL_REF(RES_EXA.DS_RESULTADO, VAL_REFERENCIA.VL_REFERENCIA_INICIAL, VAL_REFERENCIA.VL_REFERENCIA_FINAL) = 'F' 
   UNION ALL 
   SELECT  
          PACIENTE.NM_PACIENTE NM_PACIENTE,
     (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
          PACIENTE.DS_ENDERECO DS_ENDERECO,
          PACIENTE.NR_FONE NR_FONE,
          PACIENTE.TP_SEXO TP_SEXO,
          PACIENTE.DT_NASCIMENTO DT_NASCIMENTO,
          PED_LAB.CD_PED_LAB CD_PED_LAB,
          ITPED_LAB.CD_ITPED_LAB CD_ITPED_LAB,
          EXA_LAB.NM_EXA_LAB NM_EXA_LAB,
          EXA_LAB.TP_RESULTADO TP_RESULTADO,
          RES_EXA.NM_CAMPO NM_CAMPO,
          ITPED_LAB.DS_RESULTADO DS_RESULTADO,
          VAL_EXA.TP_CAMPO TP_CAMPO,
          VAL_EXA.CD_VERSAO CD_VERSAO,
          VAL_REFERENCIA.VL_REFERENCIA_INICIAL VL_REFERENCIA_INICIAL,
          VAL_REFERENCIA.VL_REFERENCIA_FINAL VL_REFERENCIA_FINAL
   FROM DBAMV.PED_LAB,
        DBAMV.ITPED_LAB,
        DBAMV.EXA_LAB,
        DBAMV.RES_EXA,
        DBAMV.VAL_EXA,
        DBAMV.VAL_REFERENCIA,
        DBAMV.ATENDIME,
        DBAMV.PACIENTE
   WHERE PED_LAB.CD_PED_LAB = ITPED_LAB.CD_PED_LAB
     AND ITPED_LAB.CD_EXA_LAB = EXA_LAB.CD_EXA_LAB
     AND ITPED_LAB.CD_VERSAO = RES_EXA.CD_VERSAO
     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
     AND ITPED_LAB.CD_ITPED_LAB = RES_EXA.CD_ITPED_LAB
     AND ITPED_LAB.CD_EXA_LAB = RES_EXA.CD_EXA_LAB
     AND RES_EXA.CD_VERSAO = VAL_EXA.CD_VERSAO
     AND RES_EXA.NM_CAMPO = VAL_EXA.NM_CAMPO
     AND VAL_EXA.CD_VERSAO = VAL_REFERENCIA.CD_VERSAO
     AND VAL_EXA.NM_CAMPO = VAL_REFERENCIA.NM_CAMPO
     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
     AND ATENDIME.CD_ATENDIMENTO = PED_LAB.CD_ATENDIMENTO
     AND ITPED_LAB.SN_REALIZADO = 'S'
     AND EXA_LAB.TP_RESULTADO = 'R'
     AND (TRUNC(SYSDATE - PACIENTE.DT_NASCIMENTO) BETWEEN (NVL(VAL_REFERENCIA.NR_ANO_INICIAL, 0) * 365) + (NVL(VAL_REFERENCIA.NR_MES_INICIAL, 0) * 30) + (NVL(VAL_REFERENCIA.NR_DIA_INICIAL, 0)) AND (NVL(VAL_REFERENCIA.NR_ANO_FINAL, 0) * 365) + (NVL(VAL_REFERENCIA.NR_MES_FINAL, 0) * 30) + (NVL(VAL_REFERENCIA.NR_DIA_FINAL, 0))
          AND VAL_REFERENCIA.TP_SEXO IN ('A', PACIENTE.TP_SEXO))
     AND DBAMV.VALIDA_VAL_REF(ITPED_LAB.DS_RESULTADO, VAL_REFERENCIA.VL_REFERENCIA_INICIAL, VAL_REFERENCIA.VL_REFERENCIA_FINAL) = 'F'  
   UNION ALL 
   SELECT  
          PACIENTE.NM_PACIENTE NM_PACIENTE,
     (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
          PACIENTE.DS_ENDERECO DS_ENDERECO,
          PACIENTE.NR_FONE NR_FONE,
          PACIENTE.TP_SEXO TP_SEXO,
          PACIENTE.DT_NASCIMENTO DT_NASCIMENTO,
          ITPED_LAB.CD_PED_LAB CD_PED_LAB,
          ITPED_LAB.CD_ITPED_LAB CD_ITPED_LAB,
          EXA_LAB.NM_EXA_LAB NM_EXA_LAB,
          EXA_LAB.TP_RESULTADO TP_RESULTADO,
          'Cultura' NM_CAMPO,
          'Positivo' DS_RESULTADO,
          'C' TP_CAMPO,
          ITPED_LAB.CD_VERSAO CD_VERSAO,
          TO_CHAR(NULL) VL_REFERENCIA_INICIAL,
          TO_CHAR(NULL) VL_REFERENCIA_FINAL
   FROM DBAMV.EXA_LAB,
        DBAMV.CULTURA,
        DBAMV.MATERIAL,
        DBAMV.ITPED_LAB,
        DBAMV.ATENDIME,
        DBAMV.PACIENTE,
        DBAMV.PED_LAB
   WHERE CULTURA.CD_EXA_LAB = EXA_LAB.CD_EXA_LAB
     AND CULTURA.CD_ITPED_LAB = ITPED_LAB.CD_ITPED_LAB
     AND MATERIAL.CD_MATERIAL = CULTURA.CD_MATERIAL
     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
     AND ATENDIME.CD_ATENDIMENTO = PED_LAB.CD_ATENDIMENTO
     AND PED_LAB.CD_PED_LAB = ITPED_LAB.CD_PED_LAB
     AND CULTURA.TP_RESULTADO <> 'N' 
     :CAMPO_CF_COND_PERIODO )
ORDER BY NM_PACIENTE,
         DS_ENDERECO,
         NR_FONE,
         TP_SEXO,
         DT_NASCIMENTO,
         CD_PED_LAB,
         CD_ITPED_LAB,
         NM_EXA_LAB,
         TP_RESULTADO,
         DS_RESULTADO,
         TP_CAMPO,
         NM_CAMPO,
         CD_VERSAO,
         VL_REFERENCIA_INICIAL,
         VL_REFERENCIA_FINAL
