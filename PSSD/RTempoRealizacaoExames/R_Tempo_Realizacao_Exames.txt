--R_Tempo_Realizacao_Exames
SELECT CD_EXA_LAB,
         NM_MNEMONICO,
         NM_EXA_LAB,
         Avg(Tempo_confimacao_setor) TEMPO_CONFIRMACAO_SETOR,
         Avg(Tempo_confimacao_Laboratorio) TEMPO_CONFIRMACAO_LABORATORIO,
         Avg(Tempo_assinado) TEMPO_ASSINADO,
         Nvl(Avg(Tempo_confimacao_setor),0) + Avg(Tempo_confimacao_Laboratorio) + Avg(Tempo_assinado)   TOTAL
      FROM (SELECT EXA_LAB.CD_EXA_LAB
       , EXA_LAB.NM_MNEMONICO
       , EXA_LAB.NM_EXA_LAB
       , CASE
           WHEN  AMOSTRA.DT_COLETA_SETOR IS NOT NULL  then  round (
                  (
                    (
                      (Trunc(AMOSTRA.DT_COLETA_SETOR) + (HR_COLETA_SETOR - TRUNC(HR_COLETA_SETOR)))
                      -   (Trunc(PED_LAB.DT_PEDIDO) + (HR_PED_LAB - TRUNC(HR_PED_LAB)))
                     ) * 1440
                  )
                ) ELSE To_number(NULL) end  Tempo_confimacao_setor
        ,   round (
                  (
                    (
                      (Trunc(AMOSTRA.DT_CONFIRMA_LAB) + (HR_CONFIRMA_LAB - TRUNC(HR_CONFIRMA_LAB)))
                      -  CASE WHEN AMOSTRA.DT_COLETA_SETOR IS NOT NULL THEN  (Trunc(AMOSTRA.DT_COLETA_SETOR) +
(TRUNC(HR_COLETA_SETOR,'MI') - TRUNC(HR_COLETA_SETOR))) ELSE  (Trunc(PED_LAB.DT_PEDIDO) + (HR_PED_LAB -
TRUNC(HR_PED_LAB))) END
                     ) * 1440
                  ))
                Tempo_confimacao_Laboratorio
       ,   round (
                  (
                    (
                      ITPED_LAB.DT_ASSINADO
                      -  (Trunc(AMOSTRA.DT_CONFIRMA_LAB) + (HR_CONFIRMA_LAB - TRUNC(HR_CONFIRMA_LAB)))
                     ) * 1440
                  ))
            Tempo_assinado
    FROM DBAMV.PED_LAB,
         DBAMV.ATENDIME,
         DBAMV.ITPED_LAB,
         DBAMV.EXA_LAB,
         DBAMV.AMOSTRA,
         DBAMV.AMOSTRA_EXA_LAB
  WHERE PED_LAB.CD_PED_LAB  = ITPED_LAB.CD_PED_LAB
      AND PED_LAB.CD_ATENDIMENTO  = ATENDIME.CD_ATENDIMENTO
    AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
    AND AMOSTRA_EXA_LAB.CD_ITPED_LAB  = ITPED_LAB.CD_ITPED_LAB
    AND AMOSTRA_EXA_LAB.CD_AMOSTRA = AMOSTRA.CD_AMOSTRA
    AND EXA_LAB.CD_EXA_LAB  = ITPED_LAB.CD_EXA_LAB
    AND SN_REALIZADO ='S'
    AND SN_ASSINADO='S'
    AND ITPED_LAB.DT_ASSINADO IS NOT NULL
:CAMPO_COND_PERIODO
)
GROUP BY  CD_EXA_LAB,
         NM_MNEMONICO,
         NM_EXA_LAB
