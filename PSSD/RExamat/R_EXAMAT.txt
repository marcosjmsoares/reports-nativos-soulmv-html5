--R_EXAMAT
SELECT * FROM (
SELECT  EXA_LAB_MATERIAL.CD_EXA_LAB CD_EXA_LAB2
       ,EXA_LAB.CD_EXA_LAB CODIGO
       ,EXA_LAB.NM_EXA_LAB DESCRICAO
       ,MATERIAL.CD_MATERIAL CD_MATERIAL
       ,MATERIAL.DS_MATERIAL DS_MATERIAL
FROM    dbamv.exa_lab_material EXA_LAB_MATERIAL
       ,dbamv.EXA_LAB EXA_LAB
       ,dbamv.MATERIAL MATERIAL
WHERE EXA_LAB.CD_EXA_LAB = EXA_LAB_MATERIAL.CD_EXA_LAB
AND   MATERIAL.CD_MATERIAL = EXA_LAB_MATERIAL.CD_MATERIAL
ORDER BY :CAMPO_ORDEM
)
ORDER BY CD_EXA_LAB2,CODIGO,DESCRICAO
