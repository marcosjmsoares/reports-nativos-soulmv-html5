--R_Media_Exames
SELECT CD_EXA_LAB,
         NM_MNEMONICO,
         NM_EXA_LAB,
         SUM(TEMPO_15) TEMPO_15,
         SUM(TEMPO_1630) TEMPO_1630,
         SUM(TEMPO_3160) TEMPO_3160,
         SUM(TEMPO_6190) TEMPO_6190,
         SUM(TEMPO_91120) TEMPO_91120,
         SUM(TEMPO_MAIOR120) TEMPO_MAIOR120
FROM (
SELECT CD_EXA_LAB,
         NM_MNEMONICO,
         NM_EXA_LAB,
         CASE WHEN Tempo_min < 15 THEN 1 ELSE 0 END TEMPO_15,
         CASE WHEN Tempo_min > 15 AND Tempo_min < 31  THEN 1 ELSE 0 END TEMPO_1630,
         CASE WHEN Tempo_min > 30 AND Tempo_min < 61  THEN 1 ELSE 0 END TEMPO_3160,
         CASE WHEN Tempo_min > 60 AND Tempo_min < 91  THEN 1 ELSE 0 END TEMPO_6190,
         CASE WHEN Tempo_min > 90 AND Tempo_min < 121  THEN 1 ELSE 0 END TEMPO_91120,
         CASE WHEN Tempo_min > 120  THEN 1 ELSE 0 END TEMPO_MAIOR120
    FROM (
  SELECT EXA_LAB.CD_EXA_LAB
       , EXA_LAB.NM_MNEMONICO
       , EXA_LAB.NM_EXA_LAB
 , round ((((Trunc(ITPED_LAB.DT_LAUDO) + (Trunc(ITPED_LAB.HR_LAUDO,'MI') - Trunc(ITPED_LAB.HR_LAUDO))) -
(Trunc(AMOSTRA.DT_CONFIRMA_LAB) + (Trunc(hr_confirma_lab,'MI') - Trunc(hr_confirma_lab)))) * 1440))
Tempo_min
    FROM DBAMV.PED_LAB,
         DBAMV.ITPED_LAB,
         DBAMV.EXA_LAB,
         DBAMV.AMOSTRA,
         DBAMV.AMOSTRA_EXA_LAB
  WHERE  PED_LAB.CD_PED_LAB  = ITPED_LAB.CD_PED_LAB
    AND  AMOSTRA_EXA_LAB.CD_ITPED_LAB  = ITPED_LAB.CD_ITPED_LAB
    AND  AMOSTRA_EXA_LAB.CD_AMOSTRA = AMOSTRA.CD_AMOSTRA
    AND EXA_LAB.CD_EXA_LAB  = ITPED_LAB.CD_EXA_LAB
    AND SN_REALIZADO ='S'
:CAMPO_COND_CD_SETOR
	)
  )
GROUP BY
        CD_EXA_LAB,
         NM_MNEMONICO,
         NM_EXA_LAB
