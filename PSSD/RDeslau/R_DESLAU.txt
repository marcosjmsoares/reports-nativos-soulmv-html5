--R_DESLAU
SELECT * FROM (
SELECT   DESENHO.DS_LINHA_DE_DESENHO   DS_LINHA_DE_DESENHO1
                ,VERSAO.CD_EXA_LAB                           CD_EXA_LAB
                ,EXA_LAB.NM_EXA_LAB                         NM_EXA_LAB
                ,VERSAO.CD_VERSAO                            CD_VERSAO
                , VERSAO.NM_VERSAO                          NM_VERSAO
FROM      dbamv.DESENHO DESENHO
               ,dbamv.VERSAO VERSAO
               ,dbamv.EXA_LAB EXA_LAB
WHERE  VERSAO.CD_VERSAO = DESENHO.CD_VERSAO(+)
AND        EXA_LAB.CD_EXA_LAB = VERSAO.CD_EXA_LAB
AND        EXA_LAB.CD_EXA_LAB LIKE :CAMPO_EXAME
ORDER BY CD_LINHA
)
ORDER BY CD_EXA_LAB,NM_EXA_LAB,CD_VERSAO,NM_VERSAO
