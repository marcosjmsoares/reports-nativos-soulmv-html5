--R_PED_S_CONF_COLETA
SELECT * FROM (
SELECT atendime.cd_atendimento,
       ped_lab.cd_ped_lab,
       paciente.cd_paciente,
       paciente.nm_paciente,
       (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE,
       convenio.nm_convenio,
       setor.nm_setor,
       amostra.cd_amostra,
       exa_lab.cd_exa_lab,
       exa_lab.nm_exa_lab
FROM   dbamv.atendime,
       dbamv.paciente,
       dbamv.ped_lab,
       dbamv.itped_lab,
       dbamv.convenio,
       dbamv.exa_lab,
       dbamv.amostra,
       dbamv.coleta_material,
       dbamv.amostra_exa_lab,
       dbamv.setor,
       dbamv.exa_lab_set_exa
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And atendime.cd_paciente = paciente.cd_paciente
AND    ped_lab.cd_atendimento = atendime.cd_atendimento
AND    ped_lab.cd_ped_lab = itped_lab.cd_ped_lab
AND    atendime.cd_convenio = convenio.cd_convenio
AND    atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND    itped_lab.cd_exa_lab = exa_lab.cd_exa_lab
AND    ped_lab.cd_ped_lab = coleta_material.cd_ped_lab
AND    ped_lab.cd_setor = setor.cd_setor
AND    coleta_material.cd_coleta_material = amostra.cd_coleta_material
AND    amostra.cd_amostra = amostra_exa_lab.cd_amostra
AND    itped_lab.cd_itped_lab = amostra_exa_lab.cd_itped_lab(+)
AND    exa_lab.cd_exa_lab = exa_lab_set_exa.cd_exa_lab
AND    itped_lab.cd_set_exa = exa_lab_set_exa.cd_set_exa
AND   NVL(amostra.sn_coleta,'N') = 'N'
:CAMPO_COND_BANCADA
)
ORDER BY CD_ATENDIMENTO,CD_PED_LAB,CD_PACIENTE,NM_PACIENTE,NM_SETOR,NM_CONVENIO,
CD_AMOSTRA
