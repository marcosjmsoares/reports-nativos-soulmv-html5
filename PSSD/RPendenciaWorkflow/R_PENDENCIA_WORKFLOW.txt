SELECT AMOSTRA.CD_AMOSTRA,
       material.ds_material,
       PED_LAB.CD_ATENDIMENTO cd_atendimento,
       PED_LAB.CD_PRESTADOR cd_prestador,
       ITPED_LAB.CD_EXA_LAB cd_exa_lab,
       To_Char(PED_LAB.DT_PEDIDO, 'dd/mm/yyyy') DT_PEDIDO,
       To_Char(PED_LAB.hr_ped_lab, 'hh24') hr_ped_lab,
  (SELECT nm_convenio
   FROM dbamv.convenio
   WHERE ped_lab.cd_convenio = convenio.cd_convenio) nm_convenio,
       setor.nm_SETOR,
       setor.cd_setor cd_setor,
       setor.nr_ramal,
  (SELECT PACIENTE.NM_PACIENTE
   FROM DBAMV.ATENDIME,
        DBAMV.PACIENTE
   WHERE PACIENTE.CD_PACIENTE = ATENDIME.CD_PACIENTE
     AND ATENDIME.CD_ATENDIMENTO = PED_LAB.CD_ATENDIMENTO) NM_PACIENTE,
  (SELECT PACIENTE.CD_PACIENTE
   FROM DBAMV.ATENDIME,
        DBAMV.PACIENTE
   WHERE PACIENTE.CD_PACIENTE = ATENDIME.CD_PACIENTE
     AND ATENDIME.CD_ATENDIMENTO = PED_LAB.CD_ATENDIMENTO) CD_PACIENTE,
  (SELECT P.NM_SOCIAL_PACIENTE
   FROM DBAMV.PACIENTE P,
        DBAMV.ATENDIME
   WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S'
      AND P.CD_PACIENTE = ATENDIME.CD_PACIENTE
      AND ATENDIME.CD_ATENDIMENTO = PED_LAB.CD_ATENDIMENTO) NM_SOCIAL_PACIENTE,
       itped_lab.DT_ENTREGA,
       itped_lab.CD_ITPED_LAB cd_itped_lab,
       ped_lab.CD_SET_EXA,
       amostra.DT_COLETA,
       ped_lab.cd_ped_lab cd_ped_lab,
  (SELECT NR_SEQ_RACK
   FROM DBAMV.RACK_AMOSTRA
   WHERE CD_ITPED_LAB = ITPED_LAB.CD_ITPED_LAB
     AND RACK_AMOSTRA.CD_AMOSTRA = AMOSTRA.CD_AMOSTRA) NR_SEQ_RACK
FROM DBAMV.PED_LAB,
     DBAMV.SETOR,
     DBAMV.ITPED_LAB,
     DBAMV.MATERIAL,
     DBAMV.AMOSTRA,
     DBAMV.AMOSTRA_EXA_LAB 
WHERE ITPED_LAB.CD_ITPED_LAB = AMOSTRA_EXA_LAB.CD_ITPED_LAB
  AND AMOSTRA_EXA_LAB.CD_AMOSTRA = AMOSTRA.CD_AMOSTRA
  AND AMOSTRA.CD_MATERIAL = MATERIAL.CD_MATERIAL (+)
