--R_Quantidade_Recoleta
SELECT itped_lab.cd_ped_lab,
       exa_lab.nm_exa_lab,
       exa_lab.NM_MNEMONICO,
       motivo_recoleta.ds_motivo_recoleta ds_motivo,
       Count(1) QUANTIDADE
FROM dbamv.amostra,
     dbamv.amostra_exa_lab,
     dbamv.itped_lab,
     dbamv.exa_lab,
     dbamv.ped_lab,
     dbamv.motivo_recoleta
WHERE amostra_exa_lab.cd_amostra = amostra.cd_amostra
  AND amostra_exa_lab.cd_itped_lab = itped_lab.cd_itped_lab
  AND itped_lab.cd_exa_lab = exa_lab.cd_exa_lab
  AND PED_LAB.CD_PED_LAB = ITPED_LAB.CD_PED_LAB
  AND motivo_recoleta.ds_motivo_recoleta IS NOT NULL
  AND amostra.cd_motivo_recoleta = motivo_recoleta.cd_motivo_recoleta (+)
  GROUP BY itped_lab.cd_ped_lab,
	 exa_lab.nm_exa_lab,
	 motivo_recoleta.ds_motivo_recoleta,
	 exa_lab.NM_MNEMONICO
