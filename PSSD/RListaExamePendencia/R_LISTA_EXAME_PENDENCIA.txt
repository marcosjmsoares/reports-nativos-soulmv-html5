--R_LISTA_EXAME_PENDENCIA
SELECT * FROM (
SELECT   DECODE(:CAMPO_SN_QUEBRA_ENTREGA, 'S', TRUNC(ITPED_LAB.DT_ENTREGA), NULL)       DT_ENTREGA
        ,ATENDIME.CD_ATENDIMENTO    CD_ATENDIMENTO
        ,PED_LAB.CD_PED_LAB         CD_PED_LAB
        ,TO_CHAR(PED_LAB.DT_PEDIDO,'DD/MM/YYYY') || ' ' || TO_CHAR(PED_LAB.HR_PED_LAB,'HH24')  DT_PEDIDO
        ,PACIENTE.CD_PACIENTE       CD_PACIENTE
        ,PACIENTE.NM_PACIENTE       NM_PACIENTE
        ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE
        ,SETOR.NM_SETOR             NM_SETOR
        ,CONVENIO.NM_CONVENIO       NM_CONVENIO
        ,EXA_LAB.NM_MNEMONICO        NM_MNEMONICO
        ,EXA_LAB.NM_EXA_LAB         NM_EXA_LAB
        ,TO_CHAR(ITPED_LAB.DT_LAUDO,'DD/MM/YYYY') || ' ' || TO_CHAR(ITPED_LAB.HR_LAUDO,'HH24')  DT_REALIZACAO
        ,ITPED_LAB.DT_REVISADO      DT_REVISAO
        ,ITPED_LAB.DT_ASSINADO      DT_ASSINATURA
        ,ITPED_LAB.DT_EMITIDO       DT_IMPRESSAO
FROM     DBAMV.ATENDIME
        ,DBAMV.PACIENTE
        ,DBAMV.PED_LAB
        ,DBAMV.CONVENIO
        ,DBAMV.SETOR
		,DBAMV.ITPED_LAB
        ,DBAMV.EXA_LAB
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And ATENDIME.CD_PACIENTE  =  PACIENTE.CD_PACIENTE
AND    ATENDIME.CD_ATENDIMENTO  =  PED_LAB.CD_ATENDIMENTO
AND    PED_LAB.CD_CONVENIO  =  CONVENIO.CD_CONVENIO
AND    ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND    PED_LAB.CD_SETOR  =  SETOR.CD_SETOR
AND    PED_LAB.CD_PED_LAB  =  ITPED_LAB.CD_PED_LAB
AND    ITPED_LAB.CD_EXA_LAB  =  EXA_LAB.CD_EXA_LAB
AND    TRUNC(ITPED_LAB.DT_ENTREGA)  BETWEEN :CAMPO_DT_INICIAL ORDER BY ITPED_LAB.CD_ITPED_LAB
)
ORDER BY DT_ENTREGA,
CD_PED_LAB,CD_ATENDIMENTO,DT_PEDIDO,CD_PACIENTE,NM_PACIENTE,NM_SETOR,NM_CONVENIO
