SELECT SESC_DIA.CD_ESCALA_DIA  CD_ESCALA_DIA
              ,SESC_DIA.DT_ESCALA_DIA DT_ESCALA_DIA
              ,TO_CHAR(SESC_DIA.DT_ESCALA_DIA, 'D' )               NM_DIA
              ,TO_CHAR(ITSESC_DIA.HR_AGENDA, 'HH24')    HR_AGENDA
              ,SRECURSO.CD_RECURSO CD_RECURSO
              ,SRECURSO.NM_RECURSO NM_RECURSO
              ,SRECURSO.CD_SET_EXA CD_SET_EXA
              ,SET_EXA.NM_SET_EXA NM_SET_EXA
              ,ITSESC_DIA.CD_PACIENTE CD_PACIENTE
              ,ITSESC_DIA.CD_ATENDIMENTO CD_ATENDIMENTO
              ,ITSESC_DIA.NM_PACIENTE NM_PACIENTE
              ,ITSESC_DIA.CD_CONVENIO CD_CONVENIO
              ,CONVENIO.NM_CONVENIO
              ,ITSESC_DIA.NR_TELEFONE
              ,ITSESC_DIA.NM_USUARIO
FROM  dbamv.ITSESC_DIA
           ,dbamv.SESC_DIA
           ,dbamv.SRECURSO
           ,dbamv.SET_EXA
           ,dbamv.CONVENIO
Where
TO_CHAR(SESC_DIA.CD_ESCALA_DIA)  =  :CAMPO_ESCALA_DIA
AND       TO_CHAR(ITSESC_DIA.HR_AGENDA, 'HH24')  =  :CAMPO_HR_AGENDA
AND       SESC_DIA.CD_ESCALA_DIA  =  ITSESC_DIA.CD_ESCALA_DIA
AND       CONVENIO.CD_CONVENIO(+)  =  ITSESC_DIA.CD_CONVENIO
AND       SRECURSO.CD_RECURSO(+)  =  SESC_DIA.CD_RECURSO
AND       SET_EXA.CD_SET_EXA(+)  =  SRECURSO.CD_SET_EXA
AND       SET_EXA.CD_SET_EXA IN (SELECT CD_SET_EXA
                                                              FROM DBAMV.USUARIOS_SET_EXA
                                                              WHERE NM_USUARIO = USER)
ORDER BY  SESC_DIA.CD_ESCALA_DIA
                    ,ITSESC_DIA.HR_AGENDA
