--R_RELACAO_PACIENTES_SD
SELECT * FROM (
SELECT Count(DISTINCT(PED_LAB.CD_PED_LAB))             TOT_PACIENTES
FROM   DBAMV.ATENDIME
            ,DBAMV.PACIENTE
            ,DBAMV.ORI_ATE
            ,DBAMV.PED_LAB
            ,DBAMV.CONVENIO
            ,DBAMV.ITPED_LAB
            ,DBAMV.EXA_LAB
            ,DBAMV.PRESTADOR
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND        ATENDIME.CD_ORI_ATE = ORI_ATE.CD_ORI_ATE
AND        ATENDIME.CD_ATENDIMENTO = PED_LAB.CD_ATENDIMENTO
AND        PED_LAB.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND        ATENDIME.CD_CONVENIO  = CONVENIO.CD_CONVENIO
AND        PED_LAB.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
AND        PED_LAB.CD_PED_LAB = ITPED_LAB.CD_PED_LAB
AND        ITPED_LAB.CD_EXA_LAB = EXA_LAB.CD_EXA_LAB
AND EXISTS  (SELECT 1
                FROM  DBAMV.USUARIOS_SET_EXA
                WHERE USUARIOS_SET_EXA.CD_SET_EXA = ITPED_LAB.CD_SET_EXA
                AND   USUARIOS_SET_EXA.NM_USUARIO = USER)
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
:CAMPO_CF_COND_SET_EXA
),
(SELECT Count(EXA_LAB.NM_EXA_LAB) TOT_EXAMES
FROM   DBAMV.ATENDIME
            ,DBAMV.PACIENTE
            ,DBAMV.ORI_ATE
            ,DBAMV.PED_LAB
            ,DBAMV.CONVENIO
            ,DBAMV.ITPED_LAB
            ,DBAMV.EXA_LAB
            ,DBAMV.PRESTADOR
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND        ATENDIME.CD_ORI_ATE = ORI_ATE.CD_ORI_ATE
AND        ATENDIME.CD_ATENDIMENTO = PED_LAB.CD_ATENDIMENTO
AND        PED_LAB.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND        ATENDIME.CD_CONVENIO  = CONVENIO.CD_CONVENIO
AND        PED_LAB.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
AND        PED_LAB.CD_PED_LAB = ITPED_LAB.CD_PED_LAB
AND        ITPED_LAB.CD_EXA_LAB = EXA_LAB.CD_EXA_LAB
AND EXISTS  (SELECT 1
                FROM  DBAMV.USUARIOS_SET_EXA
                WHERE USUARIOS_SET_EXA.CD_SET_EXA = ITPED_LAB.CD_SET_EXA
                AND   USUARIOS_SET_EXA.NM_USUARIO = USER)
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
:CAMPO_CF_COND_SET_EXA
)

--R_RELACAO_PACIENTES_SD_R_PEDIDO_Q_RELACAO
SELECT * FROM
  (SELECT PED_LAB.CD_PED_LAB CD_PED_LAB ,
          PED_LAB.DT_PEDIDO DT_PEDIDO ,
          PACIENTE.NM_PACIENTE NM_PACIENTE ,
      (SELECT P.NM_SOCIAL_PACIENTE 
       FROM DBAMV.PACIENTE P 
       WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' 
        AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
          (PACIENTE.DS_ENDERECO || DECODE(PACIENTE.NR_ENDERECO, NULL, '', ';') || PACIENTE.NR_ENDERECO) DS_ENDERECO ,
          CONVENIO.NM_CONVENIO NM_CONVENIO ,
          ORI_ATE.DS_ORI_ATE DS_ORI_ATE ,
          EXA_LAB.NM_EXA_LAB NM_EXA_LAB
   FROM DBAMV.ATENDIME ,
        DBAMV.PACIENTE ,
        DBAMV.ORI_ATE ,
        DBAMV.PED_LAB ,
        DBAMV.CONVENIO ,
        DBAMV.ITPED_LAB ,
        DBAMV.EXA_LAB ,
        DBAMV.PRESTADOR,
        Dbamv.Empresa_Convenio
   WHERE Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
     AND Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
     AND ATENDIME.CD_ORI_ATE = ORI_ATE.CD_ORI_ATE
     AND ATENDIME.CD_ATENDIMENTO = PED_LAB.CD_ATENDIMENTO
     AND TRUNC(PED_LAB.DT_PEDIDO) BETWEEN :CAMPO_DT_INIC AND 
     AND PED_LAB.CD_CONVENIO = CONVENIO.CD_CONVENIO
     AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
     AND PED_LAB.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
     AND PED_LAB.CD_PED_LAB = ITPED_LAB.CD_PED_LAB
     AND ITPED_LAB.CD_EXA_LAB = EXA_LAB.CD_EXA_LAB
     AND EXISTS
       (SELECT 1
        FROM DBAMV.USUARIOS_SET_EXA
        WHERE USUARIOS_SET_EXA.CD_SET_EXA = ITPED_LAB.CD_SET_EXA
          AND USUARIOS_SET_EXA.NM_USUARIO = USER)
     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
     :CAMPO_CF_COND_SET_EXA )
ORDER BY DT_PEDIDO,
         NM_PACIENTE,
         DS_ENDERECO,
         NM_CONVENIO,
         DS_ORI_ATE,
         CD_PED_LAB
