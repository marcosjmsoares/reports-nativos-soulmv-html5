--R_EXAPEN
SELECT * FROM (
SELECT  ITPED_LAB.CD_LABORATORIO            CD_LABORATORIO
       	,LABORATORIO.DS_LABORATORIO     DS_LABORATORIO
       	,PED_LAB_FILTRADA.CD_PRESTADOR                  CD_SOLICITA
               ,DECODE(PED_LAB_FILTRADA.CD_PRESTADOR, NULL, PED_LAB_FILTRADA.NM_PRESTADOR, PRESTADOR.NM_PRESTADOR)   SOLICITANTE
               ,PED_LAB_FILTRADA.DT_PEDIDO                           DT_PEDIDO
               ,PED_LAB_FILTRADA.HR_PED_LAB                        HR_PED_LAB
               ,PED_LAB_FILTRADA.CD_PED_LAB                        CD_PED_LAB2
               ,PED_LAB_FILTRADA.CD_ATENDIMENTO              CD_ATENDIMENTO
               ,PED_LAB_FILTRADA.CD_SETOR                            CD_SETOR
               ,SETOR.NM_SETOR                               NM_SETOR
               ,ATENDIME.CD_PACIENTE                    CD_PACIENTE
               ,PACIENTE.NM_PACIENTE                    NM_PACIENTE
               ,ATENDIME.CD_CONVENIO                   CD_CONVENIO
               ,CONVENIO.NM_CONVENIO                  NM_CONVENIO
               ,ATENDIME.CD_ORI_ATE                      ORIGEM
               ,ORI_ATE.DS_ORI_ATE                         DS_ORI_ATE
FROM  ( SELECT PED_LAB.CD_PRESTADOR
                 ,PED_LAB.NM_PRESTADOR
                 ,PED_LAB.DT_PEDIDO
                 ,PED_LAB.HR_PED_LAB
                 ,PED_LAB.CD_PED_LAB
                 ,PED_LAB.CD_ATENDIMENTO
                 ,PED_LAB.CD_SETOR
             FROM DBAMV.PED_LAB
           WHERE ped_lab.dt_pedido >=  :CAMPO_CF_DT_INI
         )  PED_LAB_FILTRADA 
           ,DBAMV.PRESTADOR
           ,DBAMV.ATENDIME
           ,DBAMV.CONVENIO
           ,DBAMV.PACIENTE
           ,DBAMV.ORI_ATE
           ,DBAMV.ITPED_LAB
           ,DBAMV.LABORATORIO
           ,DBAMV.SETOR
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And PRESTADOR.CD_PRESTADOR(+)   =  PED_LAB_FILTRADA.CD_PRESTADOR
AND     PED_LAB_FILTRADA.CD_ATENDIMENTO             =  ATENDIME.CD_ATENDIMENTO
AND     ATENDIME.CD_PACIENTE                   =  PACIENTE.CD_PACIENTE
AND     ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND     ATENDIME.CD_CONVENIO                  =  CONVENIO.CD_CONVENIO
AND     ATENDIME.CD_ORI_ATE                     =  ORI_ATE.CD_ORI_ATE
AND     PED_LAB_FILTRADA.CD_PED_LAB                       =  ITPED_LAB.CD_PED_LAB
AND     NVL(ITPED_LAB.SN_REALIZADO,'N')  = 'N'    
AND     ITPED_LAB.CD_LABORATORIO          =  LABORATORIO.CD_LABORATORIO
AND     PED_LAB_FILTRADA.CD_SETOR                           =  SETOR.CD_SETOR
AND       EXISTS  (SELECT 1
                  FROM  DBAMV.USUARIOS_SET_EXA
                  WHERE USUARIOS_SET_EXA.CD_SET_EXA  = ITPED_LAB.CD_SET_EXA   
                        AND USUARIOS_SET_EXA.NM_USUARIO = USER)
GROUP BY ITPED_LAB.CD_LABORATORIO
                    ,LABORATORIO.DS_LABORATORIO
                    ,PED_LAB_FILTRADA.CD_PRESTADOR
                    ,DECODE(PED_LAB_FILTRADA.CD_PRESTADOR, NULL, PED_LAB_FILTRADA.NM_PRESTADOR, PRESTADOR.NM_PRESTADOR)
                    ,PED_LAB_FILTRADA.DT_PEDIDO
                    ,PED_LAB_FILTRADA.HR_PED_LAB
                    ,PED_LAB_FILTRADA.CD_PED_LAB
                    ,PED_LAB_FILTRADA.CD_ATENDIMENTO
                    ,PED_LAB_FILTRADA.CD_SETOR
                    ,SETOR.NM_SETOR
                    ,ATENDIME.CD_PACIENTE
                    ,PACIENTE.NM_PACIENTE
                    ,ATENDIME.CD_CONVENIO
                   ,CONVENIO.NM_CONVENIO
                   ,ATENDIME.CD_ORI_ATE
                   ,ORI_ATE.DS_ORI_ATE
)
ORDER BY CD_LABORATORIO,DS_LABORATORIO,
CD_PED_LAB2
