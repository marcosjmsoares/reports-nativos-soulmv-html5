--R_Retificacao_Laudo
SELECT ITPED_LAB.CD_PED_LAB
     , ITPED_LAB.CD_EXA_LAB
     , EXA_LAB.NM_EXA_LAB
     , EXA_LAB.NM_MNEMONICO
     , Count(1) QTD_RETIFICADA
  FROM DBAMV.ITPED_LAB
     , DBAMV.EXA_LAB
     , DBAMV.PED_LAB
  WHERE ITPED_LAB.CD_EXA_LAB = EXA_LAB.CD_EXA_LAB
    AND SN_LAUDO_RETIFICADO = 'S'
    AND PED_LAB.CD_PED_LAB = ITPED_LAB.CD_PED_LAB
  GROUP BY ITPED_LAB.CD_PED_LAB
     , ITPED_LAB.CD_EXA_LAB
     , EXA_LAB.NM_EXA_LAB
     , EXA_LAB.NM_MNEMONICO
