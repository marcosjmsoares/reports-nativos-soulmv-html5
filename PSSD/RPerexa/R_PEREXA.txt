--R_PEREXA
SELECT * FROM (
SELECT PER_EXALAB.CD_PERGUNTA CODIGO
              ,PERGUNTA.DS_PERGUNTA    DESCRICAO
              ,PER_EXALAB.CD_EXA_LAB CD_EXA_LAB
              ,EXA_LAB.NM_EXA_LAB NM_EXA_LAB
FROM   dbamv.PER_EXALAB PER_EXALAB
            ,dbamv.PERGUNTA PERGUNTA
            ,dbamv.EXA_LAB EXA_LAB
WHERE PERGUNTA.CD_PERGUNTA = PER_EXALAB.CD_PERGUNTA
AND       EXA_LAB.CD_EXA_LAB = PER_EXALAB.CD_EXA_LAB
AND       PERGUNTA.CD_PERGUNTA LIKE :CAMPO_PERGUNTA
ORDER  BY PER_EXALAB.CD_PERGUNTA
              ,PERGUNTA.DS_PERGUNTA
)
ORDER BY CODIGO,DESCRICAO
