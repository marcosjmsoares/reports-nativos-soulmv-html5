--R_MATEXA
SELECT * FROM (
SELECT MATERIAL.CD_MATERIAL CODIGO
              ,MATERIAL.DS_MATERIAL    DESCRICAO
              ,EXA_LAB_MATERIAL.CD_EXA_LAB CD_EXA_LAB
              ,EXA_LAB.NM_EXA_LAB NM_EXA_LAB
FROM   dbamv.EXA_LAB_MATERIAL EXA_LAB_MATERIAL
            ,dbamv.MATERIAL MATERIAL
            ,dbamv.EXA_LAB EXA_LAB
WHERE MATERIAL.CD_MATERIAL = EXA_LAB_MATERIAL.CD_MATERIAL(+)
AND       EXA_LAB.CD_EXA_LAB(+) = EXA_LAB_MATERIAL.CD_EXA_LAB
AND       MATERIAL.CD_MATERIAL LIKE :CAMPO_MATERIAL
ORDER  BY EXA_LAB_MATERIAL.CD_MATERIAL
              ,MATERIAL.DS_MATERIAL
)
ORDER BY CODIGO,DESCRICAO
