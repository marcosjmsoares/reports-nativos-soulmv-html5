--R_EXAME_SOL_SET_EXA
SELECT * FROM (
SELECT ITPED_LAB.CD_SET_EXA                 CD_SETOR
             ,SET_EXA.NM_SET_EXA                    NM_SETOR
             ,PED_LAB.DT_PEDIDO                       DT_PEDIDO
             ,ATENDIME.CD_ORI_ATE                   CD_ORI_ATE
             ,ORI_ATE.DS_ORI_ATE                      DS_ORI_ATE
             ,PACIENTE.CD_PACIENTE                 CD_PACIENTE
             ,PACIENTE.NM_PACIENTE                NM_PACIENTE
             ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE
             ,ATENDIME.CD_CONVENIO               CD_CONVENIO
             ,CONVENIO.NM_CONVENIO              NM_CONVENIO
             ,CONVENIO.TP_CONVENIO               TP_CONVENIO
             ,PED_LAB.CD_PRESTADOR              CD_PRESTADOR
             ,DECODE(PED_LAB.CD_PRESTADOR, NULL, PED_LAB.NM_PRESTADOR, PRESTADOR.NM_PRESTADOR)    NM_PRESTADOR
             ,PED_LAB.CD_PED_LAB                    CD_PED_LAB
             ,ITPED_LAB.CD_EXA_LAB                 CD_EXA_LAB
             ,PED_LAB.CD_ATENDIMENTO          CD_ATENDIMENTO
             ,DECODE(EXA_LAB.NM_MNEMONICO, NULL, EXA_LAB.NM_EXA_LAB
             ,EXA_LAB.NM_MNEMONICO || ' - ' || EXA_LAB.NM_EXA_LAB)  NM_EXA_LAB
             ,DECODE(CONVENIO.TP_CONVENIO, 'A', EXA_LAB.CD_PROCEDIMENTO_SIA, 'H', EXA_LAB.CD_PROCEDIMENTO_SIH, EXA_LAB.CD_PRO_FAT)  CD_PRO_FAT
FROM DBAMV.PED_LAB
          ,DBAMV.SET_EXA
          ,DBAMV.ATENDIME
          ,DBAMV.ORI_ATE
          ,DBAMV.PACIENTE
          ,DBAMV.CONVENIO
          ,DBAMV.PRESTADOR
          ,DBAMV.ITPED_LAB
          ,DBAMV.EXA_LAB
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And SET_EXA.CD_SET_EXA                   =  ITPED_LAB.CD_SET_EXA
AND        ATENDIME.CD_ATENDIMENTO      = PED_LAB.CD_ATENDIMENTO
AND        ORI_ATE.CD_ORI_ATE                    = ATENDIME.CD_ORI_ATE
AND        ATENDIME.CD_PACIENTE              = PACIENTE.CD_PACIENTE
AND        CONVENIO.CD_CONVENIO             = ATENDIME.CD_CONVENIO
AND        PRESTADOR.CD_PRESTADOR(+)  = PED_LAB.CD_PRESTADOR
AND        ITPED_LAB.CD_PED_LAB               = PED_LAB.CD_PED_LAB
AND        EXA_LAB.CD_EXA_LAB                   = ITPED_LAB.CD_EXA_LAB
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND EXISTS  (SELECT 1 
                FROM DBAMV.USUARIOS_SET_EXA
                WHERE  USUARIOS_SET_EXA.CD_SET_EXA = ITPED_LAB.CD_SET_EXA
                AND USUARIOS_SET_EXA.NM_USUARIO = USER)
:CAMPO_COND_CON_PLA
)
ORDER BY CD_SETOR,NM_SETOR,
DT_PEDIDO,CD_ORI_ATE,DS_ORI_ATE,CD_PACIENTE,NM_PACIENTE,CD_CONVENIO,NM_CONVENIO,TP_CONVENIO,CD_PRESTADOR,NM_PRESTADOR,CD_PED_LAB,CD_ATENDIMENTO
