--R_LISTA_EXAME_LOTE
SELECT * FROM (
SELECT  TRUNC(PED_LAB.DT_PEDIDO) DT_PEDIDO
       ,PED_LAB.CD_PED_LAB
       ,ATENDIME.CD_PACIENTE
       ,PACIENTE.NM_PACIENTE
       ,(SELECT Pac.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE PAC WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND atendime.CD_PACIENTE = PAC.CD_PACIENTE) NM_SOCIAL_PACIENTE
       ,SETOR.NM_SETOR
       ,PED_LAB.TP_SOLICITACAO
       ,EXA_LAB.CD_EXA_LAB
       ,EXA_LAB.NM_EXA_LAB
       ,AMOSTRA_EXA_LAB.CD_AMOSTRA
  FROM  DBAMV.ATENDIME
       ,DBAMV.PACIENTE
       ,DBAMV.PED_LAB
       ,DBAMV.ITPED_LAB
       ,DBAMV.EXA_LAB
       ,DBAMV.SETOR
       ,DBAMV.LOTE_BANCADA
       ,DBAMV.COLETA_MATERIAL
       ,DBAMV.AMOSTRA_EXA_LAB
 WHERE  ATENDIME.CD_PACIENTE               = PACIENTE.CD_PACIENTE
    AND SETOR.CD_SETOR                               = PED_LAB.CD_SETOR
    AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
    AND ATENDIME.CD_ATENDIMENTO           = PED_LAB.CD_ATENDIMENTO
    AND PED_LAB.CD_PED_LAB                        = ITPED_LAB.CD_PED_LAB
    AND PED_LAB.CD_PED_LAB                        = COLETA_MATERIAL.CD_PED_LAB(+)
    AND ITPED_LAB.CD_EXA_LAB                     = EXA_LAB.CD_EXA_LAB
    AND ITPED_LAB.CD_LOTE_BANCADA        = LOTE_BANCADA.CD_LOTE_BANCADA
    AND ITPED_LAB.CD_ITPED_LAB                 = AMOSTRA_EXA_LAB.CD_ITPED_LAB(+)
ORDER BY PED_LAB.CD_PED_LAB
)
ORDER BY DT_PEDIDO,CD_PED_LAB,CD_PACIENTE,NM_PACIENTE,NM_SETOR,TP_SOLICITACAO,
CD_EXA_LAB,NM_EXA_LAB
