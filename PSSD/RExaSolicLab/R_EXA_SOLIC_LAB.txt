--R_EXA_SOLIC_LAB
SELECT
      PED_LAB.DT_PEDIDO
      ,ITPED_LAB.CD_LABORATORIO                               CD_LABORATORIO
       ,LABORATORIO.DS_LABORATORIO                                 DS_LABORATORIO
       ,TRUNC ( PED_LAB.DT_PEDIDO )                                     DT_PEDIDO
       ,TO_CHAR(PED_LAB.HR_PED_LAB,'HH24')                HORA
       ,ATENDIME.CD_PACIENTE                                               CD_PACIENTE
       ,ITPED_LAB.CD_EXA_LAB                                                CD_EXAME
       ,EXA_LAB.NM_EXA_LAB                                                   NOME_EXAME
       ,PED_LAB.CD_SETOR                                                       CD_SETOR_SOLICIT
       ,SETOR.NM_SETOR                                                          NM_SETOR_SOLICIT
       ,PACIENTE.NM_PACIENTE                                               NM_PACIENTE
       ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE
       ,ATENDIME.CD_CONVENIO                                              CD_CONVENIO
       ,CONVENIO.NM_CONVENIO                                             NM_CONVENIO
       ,AMOSTRA_EXA_LAB.CD_AMOSTRA                             AMOSTRA
       ,BANCADA.DS_BANCADA                                                BANCADA
FROM  DBAMV.PED_LAB
     ,DBAMV.SET_EXA
     ,DBAMV.SETOR
     ,DBAMV.ATENDIME
     ,DBAMV.CONVENIO
     ,DBAMV.PACIENTE
     ,DBAMV.ITPED_LAB
     ,DBAMV.LABORATORIO
     ,DBAMV.BANCADA
     ,DBAMV.AMOSTRA_EXA_LAB
     ,DBAMV.EXA_LAB
Where trunc(PED_LAB.DT_PEDIDO)   BETWEEN :CAMPO_DTINI  AND 
  AND  PED_LAB.CD_ATENDIMENTO             = ATENDIME.CD_ATENDIMENTO
  AND ATENDIME.CD_MULTI_EMPRESA        = :CAMPO_CD_MULTI_EMPRESA
  AND ATENDIME.CD_PACIENTE                     = PACIENTE.CD_PACIENTE
  AND ATENDIME.CD_CONVENIO                    = CONVENIO.CD_CONVENIO
  AND PED_LAB.CD_PED_LAB                         = ITPED_LAB.CD_PED_LAB
  AND ITPED_LAB.CD_LABORATORIO            = LABORATORIO.CD_LABORATORIO
  AND ITPED_LAB.CD_EXA_LAB                      = EXA_LAB.CD_EXA_LAB
  AND SETOR.CD_SETOR                                = PED_LAB.CD_SETOR
  AND SET_EXA.CD_SET_EXA                         = ITPED_LAB.CD_SET_EXA
  AND AMOSTRA_EXA_LAB.CD_ITPED_LAB  = ITPED_LAB.CD_ITPED_LAB
  AND AMOSTRA_EXA_LAB.CD_BANCADA    = BANCADA.CD_BANCADA
ORDER BY ITPED_LAB.CD_LABORATORIO
        ,LABORATORIO.DS_LABORATORIO
        ,TRUNC(PED_LAB.DT_PEDIDO)
        ,TO_CHAR(PED_LAB.HR_PED_LAB,'HH24')
        , ATENDIME.CD_PACIENTE
