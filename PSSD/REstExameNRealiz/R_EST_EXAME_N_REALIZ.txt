--R_EST_EXAME_N_REALIZ
SELECT     CODIGO
              ,:CAMPO_DS_COLUNA    DESCRICAO
              ,COUNT(*)                QT_EXAMES
FROM   (SELECT EXA_LAB.CD_EXA_LAB  
      ,EXA_LAB.NM_EXA_LAB
      ,DECODE(CONVENIO.TP_CONVENIO, 'C', EXA_LAB.CD_PRO_FAT
                                  , 'P', EXA_LAB.CD_PRO_FAT
                                  , 'H', EXA_LAB.CD_PROCEDIMENTO_SIH
                                  , 'A', EXA_LAB.CD_PROCEDIMENTO_SIA)  CD_PRO_FAT
                                  ,PRO_FAT.DS_PRO_FAT
                                  ,DT_EXAME
FROM   DBAMV.PED_LAB
      ,DBAMV.CONVENIO
      ,DBAMV.ITPED_LAB
      ,DBAMV.EXA_LAB
      ,DBAMV.PRO_FAT
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
    And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
    And PED_LAB.CD_CONVENIO = CONVENIO.CD_CONVENIO
    And PED_LAB.CD_PED_LAB = ITPED_LAB.CD_PED_LAB
    And NVL(ITPED_LAB.SN_REALIZADO,'N') = 'N'
    And ITPED_LAB.CD_EXA_LAB = EXA_LAB.CD_EXA_LAB
AND   EXA_LAB.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT(+)) V
GROUP BY  :CAMPO_CD_COLUNA
                    ,:CAMPO_DS_COLUNA
ORDER BY QT_EXAMES DESC
