--R_ETIQUETA_AMOSTRA
SELECT CD_CIDADE,
       '1' A
FROM DBAMV.CIDADE WHERE ROWNUM <= :CAMPO_QUANT


/*----------------------------------------------------------------------------*/
--R_ETIQUETA_AMOSTRA_RF_ETIQ1_Q_ATENDIMENTO
SELECT * FROM (
SELECT '1'   A1
              ,ATENDIME.CD_ATENDIMENTO
              ,PED_LAB.CD_PED_LAB
              ,('*'  ||  TO_CHAR(PED_LAB.CD_PED_LAB)  ||  '*')   CD_PEDIDO_BAR
              ,PED_LAB.DT_PEDIDO
              ,PED_LAB.HR_PED_LAB
              ,AMOSTRA.CD_AMOSTRA
              ,('*'  ||  TO_CHAR(AMOSTRA.CD_AMOSTRA)  ||  '*')    CD_AMOSTRA_BAR
              ,ATENDIME.CD_PACIENTE
              ,PACIENTE.NM_PACIENTE
              ,(UNID_INT.DS_UNID_INT || ' - '  || LEITO.DS_RESUMO)   DS_LEITO
              ,LEITO.DS_RESUMO
              ,UNID_INT.DS_UNID_INT
              ,Tubo_Coleta.Ds_Tubo_Coleta
              ,Material.Ds_Material
              ,Amostra.Dt_Coleta
FROM    DBAMV.ATENDIME
              ,DBAMV.PACIENTE
              ,DBAMV.LEITO
              ,DBAMV.PED_LAB
              ,DBAMV.COLETA_MATERIAL
              ,DBAMV.AMOSTRA
              ,DBAMV.TUBO_COLETA
              ,DBAMV.MATERIAL
              ,DBAMV.UNID_INT
WHERE  ATENDIME.CD_PACIENTE  =  PACIENTE.CD_PACIENTE
AND        ATENDIME.CD_LEITO  =  LEITO.CD_LEITO(+)
AND        ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND        ATENDIME.CD_ATENDIMENTO  =  PED_LAB.CD_ATENDIMENTO
And         Leito.Cd_Unid_Int = Unid_Int.Cd_Unid_Int(+)
AND        PED_LAB.CD_PED_LAB = COLETA_MATERIAL.CD_PED_LAB
AND        AMOSTRA.CD_COLETA_MATERIAL = COLETA_MATERIAL.CD_COLETA_MATERIAL
And         Amostra.Cd_Tubo_Coleta = Tubo_Coleta.Cd_Tubo_Coleta(+)
And         Amostra.Cd_Material = Material.Cd_Material(+)
and   ATENDIME.CD_ATENDIMENTO  = 283406
:CAMPO_CF_WHERE_AMOSTRA
)
WHERE A1 = :CAMPO_A
