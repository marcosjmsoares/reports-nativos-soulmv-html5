--R_FATURA_POR_EXAME_R_BANCADA_Q_EXAMES
SELECT *
  FROM (
        SELECT BANCADA.CD_BANCADA CD_BANCADA,
                BANCADA.DS_BANCADA DS_BANCADA,
                EXA_LAB.NM_MNEMONICO NM_MNEMONICO,
                EXA_LAB.CD_PRO_FAT CD_PROCEDIMENTO,
                EXA_LAB.NM_EXA_LAB NM_EXA_LAB,
                DBAMV.CALC_VL_PROC_UNIT(CONVENIO.TP_CONVENIO,
                                        EXA_LAB.CD_PRO_FAT,
                                        PED_LAB.DT_PEDIDO,
                                        PED_LAB.HR_PED_LAB,
                                        CONVENIO.CD_CONVENIO,
                                        NVL(PED_LAB.CD_CON_PLA, 1),
                                        ATENDIME.TP_ATENDIMENTO,
                                        ATENDIME.CD_TIP_ACOM,
                                        'U') VL_UNITARIO,
                COUNT(*) QT_EXAME,
                ATENDIME.CD_PACIENTE CD_PACIENTE,
                ITPED_LAB.cd_itped_lab
          FROM DBAMV.ATENDIME,
                DBAMV.PED_LAB,
                DBAMV.CONVENIO,
                DBAMV.ITPED_LAB,
                DBAMV.EXA_LAB,
                DBAMV.EXA_LAB_SET_EXA,
                DBAMV.BANCADA,
                Dbamv.Empresa_Convenio
         Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
           And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
           And ATENDIME.CD_ATENDIMENTO = PED_LAB.CD_ATENDIMENTO
           AND PED_LAB.CD_CONVENIO = CONVENIO.CD_CONVENIO
           AND CONVENIO.TP_CONVENIO IN ('C', 'P')
           AND ITPED_LAB.CD_SET_EXA IN
               (SELECT USUARIOS_SET_EXA.CD_SET_EXA
                  FROM DBAMV.USUARIOS_SET_EXA
                 WHERE USUARIOS_SET_EXA.NM_USUARIO = USER)
           AND PED_LAB.CD_PED_LAB = ITPED_LAB.CD_PED_LAB
           AND ITPED_LAB.CD_EXA_LAB = EXA_LAB.CD_EXA_LAB
           AND ITPED_LAB.CD_EXA_LAB = EXA_LAB_SET_EXA.CD_EXA_LAB(+)
           AND ITPED_LAB.CD_SET_EXA = EXA_LAB_SET_EXA.CD_SET_EXA(+)
           AND EXA_LAB_SET_EXA.CD_BANCADA = BANCADA.CD_BANCADA(+)
           AND EXA_LAB_SET_EXA.CD_SET_EXA = BANCADA.CD_SET_EXA(+)
           AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
         GROUP BY BANCADA.CD_BANCADA,
                   BANCADA.DS_BANCADA,
                   EXA_LAB.NM_MNEMONICO,
                   EXA_LAB.CD_PRO_FAT,
                   EXA_LAB.NM_EXA_LAB,
                   DBAMV.CALC_VL_PROC_UNIT(CONVENIO.TP_CONVENIO,
                                           EXA_LAB.CD_PRO_FAT,
                                           PED_LAB.DT_PEDIDO,
                                           PED_LAB.HR_PED_LAB,
                                           CONVENIO.CD_CONVENIO,
                                           NVL(PED_LAB.CD_CON_PLA, 1),
                                           ATENDIME.TP_ATENDIMENTO,
                                           ATENDIME.CD_TIP_ACOM,
                                           'U'),
                   ATENDIME.CD_PACIENTE,
                   ITPED_LAB.cd_itped_lab
        UNION ALL
        SELECT BANCADA.CD_BANCADA CD_BANCADA,
               BANCADA.DS_BANCADA DS_BANCADA,
               EXA_LAB.NM_MNEMONICO NM_MNEMONICO,
               EXA_LAB.CD_PRO_FAT CD_PROCEDIMENTO,
               EXA_LAB.NM_EXA_LAB NM_EXA_LAB,
               DBAMV.CALC_VL_PROC_UNIT(CONVENIO.TP_CONVENIO,
                                       EXA_LAB.CD_PRO_FAT,
                                       PED_LAB.DT_PEDIDO,
                                       PED_LAB.HR_PED_LAB,
                                       CONVENIO.CD_CONVENIO,
                                       NVL(PED_LAB.CD_CON_PLA, 1),
                                       ATENDIME.TP_ATENDIMENTO,
                                       ATENDIME.CD_TIP_ACOM,
                                       'U') VL_UNITARIO,
               COUNT(*) QT_EXAME,
               ATENDIME.CD_PACIENTE CD_PACIENTE,
               GERMES_CULTURA.cd_itped_lab
          FROM DBAMV.ATENDIME,
               DBAMV.PED_LAB,
               DBAMV.CONVENIO,
               DBAMV.ITPED_LAB,
               DBAMV.EXA_LAB,
               DBAMV.EXA_LAB_SET_EXA,
               DBAMV.BANCADA,
               Dbamv.Empresa_Convenio,
               DBAMV.GERMES_CULTURA
         Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
           And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
           And ATENDIME.CD_ATENDIMENTO = PED_LAB.CD_ATENDIMENTO
           AND PED_LAB.CD_CONVENIO = CONVENIO.CD_CONVENIO
           AND CONVENIO.TP_CONVENIO IN ('C', 'P')
           AND ITPED_LAB.CD_SET_EXA IN
               (SELECT USUARIOS_SET_EXA.CD_SET_EXA
                  FROM DBAMV.USUARIOS_SET_EXA
                 WHERE USUARIOS_SET_EXA.NM_USUARIO = USER)
           AND PED_LAB.CD_PED_LAB = ITPED_LAB.CD_PED_LAB
           AND GERMES_CULTURA.CD_EXA_LAB_FATURA = EXA_LAB.CD_EXA_LAB
           AND GERMES_CULTURA.CD_EXA_LAB_FATURA = EXA_LAB_SET_EXA.CD_EXA_LAB
           AND GERMES_CULTURA.CD_EXA_LAB = ITPED_LAB.CD_EXA_LAB
           AND GERMES_CULTURA.CD_ITPED_LAB = ITPED_LAB.CD_ITPED_LAB
           AND ITPED_LAB.CD_SET_EXA = EXA_LAB_SET_EXA.CD_SET_EXA(+)
           AND EXA_LAB_SET_EXA.CD_BANCADA = BANCADA.CD_BANCADA(+)
           AND EXA_LAB_SET_EXA.CD_SET_EXA = BANCADA.CD_SET_EXA(+)
           AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
         GROUP BY BANCADA.CD_BANCADA,
                  BANCADA.DS_BANCADA,
                  EXA_LAB.NM_MNEMONICO,
                  EXA_LAB.CD_PRO_FAT,
                  EXA_LAB.NM_EXA_LAB,
                  DBAMV.CALC_VL_PROC_UNIT(CONVENIO.TP_CONVENIO,
                                          EXA_LAB.CD_PRO_FAT,
                                          PED_LAB.DT_PEDIDO,
                                          PED_LAB.HR_PED_LAB,
                                          CONVENIO.CD_CONVENIO,
                                          NVL(PED_LAB.CD_CON_PLA, 1),
                                          ATENDIME.TP_ATENDIMENTO,
                                          ATENDIME.CD_TIP_ACOM,
                                          'U'),
                  ATENDIME.CD_PACIENTE,
                  GERMES_CULTURA.cd_itped_lab
        UNION ALL
        SELECT BANCADA.CD_BANCADA CD_BANCADA,
               BANCADA.DS_BANCADA DS_BANCADA,
               EXA_LAB.NM_MNEMONICO NM_MNEMONICO,
               DECODE(CONVENIO.TP_CONVENIO,
                      'H',
                      EXA_LAB.cd_procedimento_sih,
                      'A',
                      EXA_LAB.cd_procedimento_sia,
                      EXA_LAB.CD_PRO_FAT) CD_PROCEDIMENTO,
               EXA_LAB.NM_EXA_LAB NM_EXA_LAB,
               DBAMV.CALC_VL_PROC_UNIT(CONVENIO.TP_CONVENIO,
                                       decode(convenio.tp_convenio,
                                              'A',
                                              exa_lab.cd_procedimento_sia,
                                              'H',
                                              exa_lab.cd_procedimento_sih,
                                              'C',
                                              exa_lab.cd_pro_fat,
                                              'P',
                                              exa_lab.cd_pro_fat),
                                       DECODE('',
                                              'P',
                                              PED_LAB.DT_PEDIDO,
                                              ITPED_LAB.DT_LAUDO),
                                       PED_LAB.HR_PED_LAB,
                                       ATENDIME.CD_CONVENIO,
                                       ATENDIME.CD_CON_PLA,
                                       ATENDIME.TP_ATENDIMENTO,
                                       ATENDIME.CD_TIP_ACOM,
                                       'U') VL_UNITARIO,
               COUNT(*) QT_EXAME,
               ATENDIME.CD_PACIENTE CD_PACIENTE,
               ITPED_LAB.cd_itped_lab
          FROM DBAMV.ATENDIME,
               DBAMV.PED_LAB,
               DBAMV.CONVENIO,
               DBAMV.ITPED_LAB,
               DBAMV.EXA_LAB,
               DBAMV.EXA_LAB_SET_EXA,
               DBAMV.BANCADA,
               Dbamv.Empresa_Convenio
         Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
           And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
           AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
           And ATENDIME.CD_ATENDIMENTO = PED_LAB.CD_ATENDIMENTO
           AND PED_LAB.CD_CONVENIO = CONVENIO.CD_CONVENIO
           AND CONVENIO.TP_CONVENIO IN ('A', 'H')
           AND ITPED_LAB.CD_SET_EXA IN
               (SELECT USUARIOS_SET_EXA.CD_SET_EXA
                  FROM DBAMV.USUARIOS_SET_EXA
                 WHERE USUARIOS_SET_EXA.NM_USUARIO = USER)
           AND PED_LAB.CD_PED_LAB = ITPED_LAB.CD_PED_LAB
           AND ITPED_LAB.CD_EXA_LAB = EXA_LAB.CD_EXA_LAB
           AND ITPED_LAB.CD_EXA_LAB = EXA_LAB_SET_EXA.CD_EXA_LAB(+)
           AND ITPED_LAB.CD_SET_EXA = EXA_LAB_SET_EXA.CD_SET_EXA(+)
           AND EXA_LAB_SET_EXA.CD_BANCADA = BANCADA.CD_BANCADA(+)
           AND EXA_LAB_SET_EXA.CD_SET_EXA = BANCADA.CD_SET_EXA(+)
         GROUP BY BANCADA.CD_BANCADA,
                  BANCADA.DS_BANCADA,
                  EXA_LAB.NM_MNEMONICO,
                  DECODE(CONVENIO.TP_CONVENIO,
                         'H',
                         EXA_LAB.cd_procedimento_sih,
                         'A',
                         EXA_LAB.cd_procedimento_sia,
                         EXA_LAB.CD_PRO_FAT),
                  EXA_LAB.NM_EXA_LAB,
                  ATENDIME.CD_PACIENTE,
                  ITPED_LAB.cd_itped_lab,
                  DBAMV.CALC_VL_PROC_UNIT(CONVENIO.TP_CONVENIO,
                                          decode(convenio.tp_convenio,
                                                 'A',
                                                 exa_lab.cd_procedimento_sia,
                                                 'H',
                                                 exa_lab.cd_procedimento_sih,
                                                 'C',
                                                 exa_lab.cd_pro_fat,
                                                 'P',
                                                 exa_lab.cd_pro_fat),
                                          DECODE('',
                                                 'P',
                                                 PED_LAB.DT_PEDIDO,
                                                 ITPED_LAB.DT_LAUDO),
                                          PED_LAB.HR_PED_LAB,
                                          ATENDIME.CD_CONVENIO,
                                          ATENDIME.CD_CON_PLA,
                                          ATENDIME.TP_ATENDIMENTO,
                                          ATENDIME.CD_TIP_ACOM,
                                          'U')
        UNION ALL
        SELECT BANCADA.CD_BANCADA CD_BANCADA,
               BANCADA.DS_BANCADA DS_BANCADA,
               EXA_LAB.NM_MNEMONICO NM_MNEMONICO,
               DECODE(CONVENIO.TP_CONVENIO,
                      'H',
                      EXA_LAB.cd_procedimento_sih,
                      'A',
                      EXA_LAB.cd_procedimento_sia,
                      EXA_LAB.CD_PRO_FAT) CD_PROCEDIMENTO,
               EXA_LAB.NM_EXA_LAB NM_EXA_LAB,
               DBAMV.CALC_VL_PROC_UNIT(CONVENIO.TP_CONVENIO,
                                       decode(convenio.tp_convenio,
                                              'A',
                                              exa_lab.cd_procedimento_sia,
                                              'H',
                                              exa_lab.cd_procedimento_sih,
                                              'C',
                                              exa_lab.cd_pro_fat,
                                              'P',
                                              exa_lab.cd_pro_fat),
                                       DECODE('',
                                              'P',
                                              PED_LAB.DT_PEDIDO,
                                              ITPED_LAB.DT_LAUDO),
                                       PED_LAB.HR_PED_LAB,
                                       ATENDIME.CD_CONVENIO,
                                       ATENDIME.CD_CON_PLA,
                                       ATENDIME.TP_ATENDIMENTO,
                                       ATENDIME.CD_TIP_ACOM,
                                       'U') VL_UNITARIO,
               COUNT(*) QT_EXAME,
               ATENDIME.CD_PACIENTE CD_PACIENTE,
               germes_cultura.cd_itped_lab
          FROM DBAMV.ATENDIME,
               DBAMV.PED_LAB,
               DBAMV.CONVENIO,
               DBAMV.ITPED_LAB,
               DBAMV.EXA_LAB,
               DBAMV.EXA_LAB_SET_EXA,
               DBAMV.BANCADA,
               DBAMV.EMPRESA_CONVENIO,
               DBAMV.GERMES_CULTURA
         WHERE EMPRESA_CONVENIO.CD_CONVENIO = CONVENIO.CD_CONVENIO
           AND EMPRESA_CONVENIO.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
           AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
           AND ATENDIME.CD_ATENDIMENTO = PED_LAB.CD_ATENDIMENTO
           AND PED_LAB.CD_CONVENIO = CONVENIO.CD_CONVENIO
           AND CONVENIO.TP_CONVENIO IN ('A', 'H')
           AND ITPED_LAB.CD_SET_EXA IN
               (SELECT USUARIOS_SET_EXA.CD_SET_EXA
                  FROM DBAMV.USUARIOS_SET_EXA
                 WHERE USUARIOS_SET_EXA.NM_USUARIO = USER)
           AND PED_LAB.CD_PED_LAB = ITPED_LAB.CD_PED_LAB
           AND GERMES_CULTURA.CD_EXA_LAB_FATURA = EXA_LAB.CD_EXA_LAB
           AND GERMES_CULTURA.CD_EXA_LAB_FATURA = EXA_LAB_SET_EXA.CD_EXA_LAB
           AND GERMES_CULTURA.CD_EXA_LAB = ITPED_LAB.CD_EXA_LAB
           AND GERMES_CULTURA.CD_ITPED_LAB = ITPED_LAB.CD_ITPED_LAB
           AND ITPED_LAB.CD_SET_EXA = EXA_LAB_SET_EXA.CD_SET_EXA(+)
           AND EXA_LAB_SET_EXA.CD_BANCADA = BANCADA.CD_BANCADA(+)
           AND EXA_LAB_SET_EXA.CD_SET_EXA = BANCADA.CD_SET_EXA(+)
         GROUP BY BANCADA.CD_BANCADA,
                  BANCADA.DS_BANCADA,
                  EXA_LAB.NM_MNEMONICO,
                  DECODE(CONVENIO.TP_CONVENIO,
                         'H',
                         EXA_LAB.CD_PROCEDIMENTO_SIH,
                         'A',
                         EXA_LAB.CD_PROCEDIMENTO_SIA,
                         EXA_LAB.CD_PRO_FAT),
                  EXA_LAB.NM_EXA_LAB,
                  ATENDIME.CD_PACIENTE,
                  GERMES_CULTURA.CD_ITPED_LAB,
                  DBAMV.CALC_VL_PROC_UNIT(CONVENIO.TP_CONVENIO,
                                          DECODE(CONVENIO.TP_CONVENIO,
                                                 'A',
                                                 EXA_LAB.CD_PROCEDIMENTO_SIA,
                                                 'H',
                                                 EXA_LAB.CD_PROCEDIMENTO_SIH,
                                                 'C',
                                                 EXA_LAB.CD_PRO_FAT,
                                                 'P',
                                                 EXA_LAB.CD_PRO_FAT),
                                          DECODE('',
                                                 'P',
                                                 PED_LAB.DT_PEDIDO,
                                                 ITPED_LAB.DT_LAUDO),
                                          PED_LAB.HR_PED_LAB,
                                          ATENDIME.CD_CONVENIO,
                                          ATENDIME.CD_CON_PLA,
                                          ATENDIME.TP_ATENDIMENTO,
                                          ATENDIME.CD_TIP_ACOM,
                                          'U'))
 ORDER BY CD_BANCADA,
          DS_BANCADA,
          NM_MNEMONICO,
          CD_PROCEDIMENTO,
          NM_EXA_LAB,
          VL_UNITARIO
