--R_EXAMED
SELECT * FROM (
SELECT   nvl(ped_lab.cd_prestador, 9999)    cd_prestador
               ,decode(ped_lab.cd_prestador, null, ped_lab.nm_prestador, prestador.nm_prestador)    nm_prestador
               ,itped_lab.cd_exa_lab                      cd_exa_lab
               ,exa_lab.nm_exa_lab                       nm_exa_lab
               ,count(*)           qtd_exame
FROM  dbamv.atendime
           ,dbamv.ped_lab
           ,dbamv.ori_ate
           ,dbamv.itped_lab
           ,dbamv.exa_lab
           ,dbamv.prestador
           ,dbamv.convenio
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And atendime.cd_ori_ate  = ori_ate.cd_ori_ate
and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
and          atendime.cd_atendimento  =  ped_lab.cd_atendimento
and          ped_lab.cd_ped_lab  =  itped_lab.cd_ped_lab
AND EXISTS (SELECT 1
                       FROM DBAMV.USUARIOS_SET_EXA
                       WHERE USUARIOS_SET_EXA.CD_SET_EXA = ITPED_LAB.CD_SET_EXA
                       AND   USUARIOS_SET_EXA.NM_USUARIO    = USER)
and          itped_lab.cd_exa_lab                = exa_lab.cd_exa_lab
and          ped_lab.cd_prestador               = prestador.cd_prestador(+)
and          atendime.cd_convenio             = convenio.cd_convenio
GROUP BY  ped_lab.cd_prestador
                   ,decode(ped_lab.cd_prestador, null, ped_lab.nm_prestador, prestador.nm_prestador)
                   ,itped_lab.cd_exa_lab
                   ,exa_lab.nm_exa_lab
)
ORDER BY CD_PRESTADOR,NM_PRESTADOR, cd_exa_lab

--R_EXAMED_R_CONVENIO1_Q_CONVENIO
SELECT * FROM (
SELECT  nvl(ped_lab.cd_prestador, 9999)   cd_prestador2
               ,itped_lab.cd_exa_lab                    cd_exa_lab2
               ,atendime.cd_convenio                 cd_convenio
               ,convenio.nm_convenio                nm_convenio
               ,count(atendime.cd_convenio)      qtd_convenio
FROM  dbamv.atendime
           ,dbamv.convenio
           ,dbamv.prestador
           ,dbamv.ped_lab
           ,dbamv.itped_lab
           ,dbamv.exa_lab
           ,dbamv.ori_ate
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And atendime.cd_ori_ate  =  ori_ate.cd_ori_ate
and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
and  atendime.cd_atendimento  =  ped_lab.cd_atendimento
AND EXISTS (SELECT 1
                       FROM DBAMV.USUARIOS_SET_EXA
                       WHERE USUARIOS_SET_EXA.CD_SET_EXA = ITPED_LAB.CD_SET_EXA
                       AND   USUARIOS_SET_EXA.NM_USUARIO = USER)
and          ped_lab.cd_ped_lab                =  itped_lab.cd_ped_lab
and          itped_lab.cd_exa_lab               =  exa_lab.cd_exa_lab
and          ped_lab.cd_prestador              =  prestador.cd_prestador(+)
and          atendime.cd_convenio            =  convenio.cd_convenio
GROUP BY ped_lab.cd_prestador
                   ,itped_lab.cd_exa_lab
                   ,atendime.cd_convenio
                   ,convenio.nm_convenio
)
WHERE CD_PRESTADOR2 = :CAMPO_CD_PRESTADOR
	AND cd_exa_lab2 = :CAMPO_CD_EXA_LAB

--R_EXAMED_R_ORIGEM1_Q_ORIGEM
SELECT * FROM (
SELECT nvl(ped_lab.cd_prestador, 9999)    cd_prestador1
              ,itped_lab.cd_exa_lab                     cd_exa_lab1
              ,atendime.cd_ori_ate                      cd_ori_ate
              ,ori_ate.ds_ori_ate                          ds_ori_ate
              ,count(atendime.cd_ori_ate)           qtd_origem
FROM  dbamv.atendime
           ,dbamv.ori_ate
           ,dbamv.prestador
           ,dbamv.ped_lab
           ,dbamv.itped_lab
           ,dbamv.convenio
           ,dbamv.exa_lab
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And atendime.cd_ori_ate  =  ori_ate.cd_ori_ate
and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
and         atendime.cd_atendimento  =  ped_lab.cd_atendimento
AND EXISTS (SELECT 1
                       FROM DBAMV.USUARIOS_SET_EXA
                        WHERE USUARIOS_SET_EXA.CD_SET_EXA = ITPED_LAB.CD_SET_EXA
                        AND   USUARIOS_SET_EXA.NM_USUARIO = USER)
and        ped_lab.cd_ped_lab                =  itped_lab.cd_ped_lab
and        itped_lab.cd_exa_lab              =  exa_lab.cd_exa_lab
and        ped_lab.cd_prestador             =  prestador.cd_prestador(+)
and        atendime.cd_convenio            =  convenio.cd_convenio
GROUP BY ped_lab.cd_prestador
                   ,itped_lab.cd_exa_lab
                   ,atendime.cd_ori_ate
                   ,ori_ate.ds_ori_ate
)
WHERE CD_PRESTADOR1 = :CAMPO_CD_PRESTADOR
	AND cd_exa_lab1 = :CAMPO_CD_EXA_LAB
