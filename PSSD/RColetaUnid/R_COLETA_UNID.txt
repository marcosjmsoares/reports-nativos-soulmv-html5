--R_COLETA_UNID
SELECT coleta_material.cd_coleta_material ,
       coleta_material.cd_unid_int ,
       unid_int.ds_unid_int ,
       atendime.cd_atendimento ,
       atendime.cd_paciente ,
       paciente.nm_paciente ,
    (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
       atendime.cd_convenio ,
       convenio.nm_convenio ,
       atendime.cd_ori_ate ,
       atendime.cd_prestador ,
       prestador.nm_prestador ,
       ped_lab.cd_ped_lab ,
       ped_lab.dt_pedido ,
       ori_ate.ds_ori_ate ,
       material.ds_material ,
       coleta_material.dt_solicitacao ,
       TO_CHAR(coleta_material.hr_solicitacao, 'hh') ,
       coleta_material.ds_observacao ,
       exa_lab.nm_exa_lab ,
       exa_lab.cd_exa_lab ,
       AMOSTRA.CD_MATERIAL ,
       AMOSTRA.CD_TUBO_COLETA ,
       TUBO_COLETA.DS_TUBO_COLETA ,
       AMOSTRA.CD_AMOSTRA
FROM dbamv.coleta_material ,
     dbamv.unid_int ,
     dbamv.ori_ate ,
     dbamv.material ,
     dbamv.convenio ,
     dbamv.prestador ,
     dbamv.atendime ,
     dbamv.paciente ,
     dbamv.ped_lab ,
     dbamv.itped_lab ,
     dbamv.exa_lab ,
     DBAMV.AMOSTRA ,
     DBAMV.AMOSTRA_EXA_LAB ,
     DBAMV.TUBO_COLETA,
     Dbamv.Empresa_Convenio
WHERE Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
  AND Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
  AND atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA 
  AND COLETA_MATERIAL.CD_COLETA_MATERIAL = AMOSTRA.CD_COLETA_MATERIAL
  AND AMOSTRA.CD_AMOSTRA = AMOSTRA_EXA_LAB.CD_AMOSTRA
  AND AMOSTRA.CD_MATERIAL = MATERIAL.CD_MATERIAL
  AND coleta_material.cd_unid_int = unid_int.cd_unid_int(+)
  AND ped_lab.cd_ped_lab = coleta_material.cd_ped_lab
  AND ped_lab.cd_atendimento = atendime.cd_atendimento
  AND itped_lab.cd_ped_lab = ped_lab.cd_ped_lab
  AND itped_lab.cd_exa_lab = exa_lab.cd_exa_lab
  AND Itped_lab.Cd_itped_lab = Amostra_exa_lab.Cd_itped_lab
  AND atendime.cd_paciente = paciente.cd_paciente
  AND atendime.cd_convenio = convenio.cd_convenio
  AND atendime.cd_ori_ate = ori_ate.cd_ori_ate
  AND atendime.cd_prestador = prestador.cd_prestador
  AND TUBO_COLETA.CD_TUBO_COLETA = AMOSTRA.CD_TUBO_COLETA
  AND NVL(AMOSTRA.SN_COLETA, 'N') = :CAMPO_CF_REALIZADAS 
  AND TRUNC(coleta_material.dt_solicitacao) BETWEEN :CAMPO_DTINIC ORDER BY coleta_material.cd_unid_int
