--R_EXA_RESTRICAO
SELECT * FROM (
select  ped_lab.cd_ped_lab numero_pedido
        ,ped_lab.dt_pedido data_pedido
       ,exa_lab.cd_exa_lab codigo_exame
       ,nm_exa_lab nome_exame
       ,itped_lab.cd_tecnico_exec codigo_responsavel
       ,prestador.nm_prestador nome_responsavel
       ,itped_lab.cd_set_exa codigo_setor
       ,set_exa.nm_set_exa nome_setor
       ,ds_restricao restricao
from dbamv.exa_lab
    ,dbamv.itped_lab
    ,dbamv.ped_lab
    ,dbamv.prestador
    ,dbamv.set_exa
where ped_lab.cd_ped_lab = itped_lab.cd_ped_lab
and itped_lab.cd_exa_lab = exa_lab.cd_exa_lab
and itped_lab.cd_tecnico_exec = prestador.cd_prestador(+)
and itped_lab.cd_set_exa = set_exa.cd_set_exa(+)
and ds_restricao is not null
)
ORDER BY NUMERO_PEDIDO,DATA_PEDIDO
