--R_LABEXA
SELECT * FROM (
SELECT  LABORATORIO.CD_LABORATORIO    CODIGO
               ,LABORATORIO.DS_LABORATORIO    DESCRICAO
               ,EXA_LAB.NM_EXA_LAB
               ,EXA_LAB.CD_EXA_LAB
FROM    dbamv.EXA_LAB_LABORATORIO
             ,dbamv.LABORATORIO
             ,dbamv.EXA_LAB
WHERE       LABORATORIO.CD_LABORATORIO = EXA_LAB_LABORATORIO.CD_LABORATORIO
AND             EXA_LAB_LABORATORIO.CD_EXA_LAB  =  EXA_LAB.CD_EXA_LAB
AND             LABORATORIO.CD_LABORATORIO LIKE :CAMPO_LAB
)
ORDER BY CODIGO,DESCRICAO
