--R_PACIENTE_BANCADA
SELECT * FROM
  (SELECT DISTINCT (amostra_exa_lab.cd_amostra) cd_amostra_b ,
                   ped_lab.cd_ped_lab cd_ped_lab_b ,
                   bancada.cd_bancada cd_bancada_b ,
                   bancada.ds_bancada ds_bancada_b ,
                   atendime.cd_atendimento cd_atendimento_b ,
                   ori_ate.ds_ori_ate ds_ori_ate_b ,
                   paciente.cd_paciente||'  '||'-'||'  '||paciente.nm_paciente nm_paciente_b ,
          (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE,
                   paciente.cd_paciente cd_paciente_b ,
                   DECODE (itped_lab.sn_realizado, 'S', itped_lab.dt_laudo, ped_lab.dt_pedido) Dt_Solicitado_Or_Realizado ,
                   ped_lab.hr_ped_lab hr_ped_lab_b ,
                   paciente.tp_sexo tp_sexo_b
   FROM dbamv.atendime ,
        dbamv.paciente ,
        dbamv.itped_lab ,
        dbamv.ped_lab ,
        dbamv.exa_lab ,
        dbamv.setor ,
        dbamv.bancada ,
        dbamv.amostra_exa_lab ,
        dbamv.ori_ate ,
     (SELECT amostra.cd_amostra,
             amostra.cd_material
      FROM dbamv.amostra
      WHERE NOT EXISTS
          (SELECT 1 FROM DBAMV.AMOSTRA AMOS 
           WHERE AMOS.cd_amostra_pai = amostra.cd_amostra)) amostra
   WHERE itped_lab.cd_itped_lab = amostra_exa_lab.cd_itped_lab
     AND amostra_exa_lab.cd_amostra = amostra.cd_amostra
     AND amostra_exa_lab.cd_bancada = bancada.cd_bancada
     AND itped_lab.cd_ped_lab = ped_lab.cd_ped_lab
     AND paciente.cd_paciente = atendime.cd_paciente
     AND atendime.cd_atendimento = ped_lab.cd_atendimento
     AND atendime.cd_ori_ate = ori_ate.cd_ori_ate
     AND atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
     AND EXISTS
       (SELECT 1 FROM dbamv.usuarios_set_exa
        WHERE itped_lab.cd_set_exa = usuarios_set_exa.cd_set_exa
          AND usuarios_set_exa.nm_usuario = USER)
     AND itped_lab.cd_exa_lab = exa_lab.cd_exa_lab
     AND ped_lab.cd_setor = setor.cd_setor 
     :CAMPO_CF_COND_PERIODO )
ORDER BY ds_bancada_b,
         cd_ped_lab_b,
         cd_amostra_b
