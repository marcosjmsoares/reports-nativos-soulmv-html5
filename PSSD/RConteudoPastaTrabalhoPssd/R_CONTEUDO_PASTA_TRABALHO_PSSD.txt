--R_CONTEUDO_PASTA_TRABALHO_PSSD
SELECT pasta_trabalho_conteudo.cd_lote_bancada, bancada.ds_bancada,
pasta_trabalho_conteudo.dt_lote_bancada, pasta_trabalho_conteudo.cd_ped_lab,
pasta_trabalho_conteudo.cd_exa_lab,exa_lab.nm_exa_lab
FROM dbamv.pasta_trabalho_conteudo
LEFT JOIN dbamv.lote_bancada ON (lote_bancada.cd_lote_bancada = pasta_trabalho_conteudo.cd_lote_bancada)
LEFT JOIN dbamv.bancada ON (lote_bancada.cd_bancada = bancada.cd_bancada)
LEFT JOIN dbamv.exa_lab ON (exa_lab.cd_exa_lab = pasta_trabalho_conteudo.cd_exa_lab)
WHERE pasta_trabalho_conteudo.CD_PASTA_TRABALHO_PSSD = :CAMPO_CD_PASTA_TRABALHO_PSSD
order by cd_lote_bancada, cd_ped_lab, cd_exa_lab, cd_itped_lab
