--R_COMPR_ENTREGA
SELECT * FROM
  (SELECT PROT.CD_ENTREGA_EXAME ,
          PROT.DT_ENTREGA_EXAME ,
          to_char(PROT.HR_ENTREGA_EXAME, 'hh24') HR_ENTREGA_EXAME ,
          PROT.NM_RESPONSAVEL ,
          PROT.DS_DOCUMENTO ,
          PROT.NR_CPF ,
          PROT.CD_USUARIO ,
          PROT.DT_PROTOCOLO ,
          ATENDIME.CD_ATENDIMENTO ,
          PED_LAB.CD_PED_LAB ,
          PACIENTE.CD_PACIENTE ,
          PACIENTE.NM_PACIENTE ,
      (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
          PACIENTE.CD_PACIENTE||' '||PACIENTE.NM_PACIENTE||' '||NM_SOCIAL_PACIENTE PACIENTE ,
          EXA_LAB.CD_EXA_LAB ,
          EXA_LAB.NM_EXA_LAB ,
          EXA_LAB.CD_EXA_LAB||' '||EXA_LAB.NM_EXA_LAB EXA_LAB
   FROM DBAMV.ENTREGA_EXAME_SADT PROT ,
        DBAMV.PED_LAB ,
        DBAMV.ATENDIME ,
        DBAMV.PACIENTE ,
        DBAMV.ITPED_LAB ,
        DBAMV.EXA_LAB
   WHERE PED_LAB.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
     AND PED_LAB.CD_PED_LAB = ITPED_LAB.CD_PED_LAB
     AND ITPED_LAB.CD_ENTREGA_EXAME = PROT.CD_ENTREGA_EXAME
     AND ITPED_LAB.CD_EXA_LAB = EXA_LAB.CD_EXA_LAB :CAMPO_CF_PROTOCOLO )
ORDER BY CD_ENTREGA_EXAME,
         DT_ENTREGA_EXAME,
         HR_ENTREGA_EXAME,
         NM_RESPONSAVEL,
         DS_DOCUMENTO,
         NR_CPF,
         CD_USUARIO,
         DT_PROTOCOLO,
         CD_ATENDIMENTO,
         CD_PED_LAB,
         CD_PACIENTE,
         NM_PACIENTE,
         PACIENTE
