--R_Motivo_Retificacao_Exame
SELECT DISTINCT ITPED_LAB.cd_motivo_retificacao_exame,
       motivo_retificacao_exame.ds_motivo_retificacao_exame
  FROM DBAMV.ITPED_LAB,
       DBAMV.EXA_LAB,
       DBAMV.motivo_retificacao_exame,
       DBAMV.PED_LAB
 WHERE ITPED_LAB.CD_motivo_retificacao_exame =  motivo_retificacao_exame.CD_motivo_retificacao_exame
   AND ITPED_LAB.CD_EXA_LAB = EXA_LAB.CD_EXA_LAB
   AND PED_LAB.CD_PED_LAB  =  ITPED_LAB.CD_PED_LAB
