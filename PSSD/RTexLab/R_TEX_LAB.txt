--R_TEX_LAB
SELECT * FROM (
SELECT T.CD_TEXTO
      ,T.DS_TEXTO
      ,T.DS_PADRAO
      ,NVL(E.CD_EXA_LAB,0) CD_EXA_LAB
      ,NVL(E.NM_EXA_LAB,'Exame Não Informado') NM_EXA_LAB
      ,NVL(S.CD_SET_EXA,0) CD_SET_EXA
      ,NVL(S.NM_SET_EXA,'Setor Não Informado') NM_SET_EXA
FROM
  DBAMV.TEXTOS_PAD_PSSD T,
  DBAMV.EXA_LAB E,
  DBAMV.SET_EXA S,
  DBAMV.EXA_LAB_SET_EXA,
  dbamv.exa_lab_textos_pad A                                     
WHERE A.CD_EXA_LAB = E.CD_EXA_LAB               
  AND T.CD_TEXTO   = A.CD_TEXTO                        
AND E.CD_EXA_LAB    = EXA_LAB_SET_EXA.cd_exa_lab(+)
AND S.CD_SET_EXA(+) = EXA_LAB_SET_EXA.cd_set_exa
:CAMPO_COND_TEXTO
)
ORDER BY CD_SET_EXA,NM_SET_EXA
