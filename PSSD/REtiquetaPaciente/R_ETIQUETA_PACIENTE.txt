--R_ETIQUETA_PACIENTE
SELECT CD_CIDADE,
       '1' A
FROM DBAMV.CIDADE WHERE ROWNUM <= :CAMPO_ETIQ


/*----------------------------------------------------------------------------*/
--R_ETIQUETA_PACIENTE_RF_ETIQ1_Q_ATENDIMENTO
SELECT * FROM (
SELECT '1' A1,
              PACIENTE.CD_PACIENTE
             ,PACIENTE.NM_PACIENTE
			 ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE
             ,PACIENTE.DT_NASCIMENTO
             ,PACIENTE.DT_CADASTRO
             ,DECODE(PACIENTE.TP_SEXO, 'M', 'MASCULINO', 'F', 'FEMININO', 'AMBOS')  DS_SEXO
             ,(PACIENTE.DS_ENDERECO || ' '  ||  PACIENTE.NR_ENDERECO || ' ' || PACIENTE.DS_COMPLEMENTO)   DS_ENDERECO
             ,PACIENTE.NM_BAIRRO
             ,(CIDADE.NM_CIDADE ||  '-'  ||  CIDADE.CD_UF)  NM_CIDADE
             ,PACIENTE.NR_CEP
             ,DECODE(PACIENTE.SN_UTIL_IDENTIDADE_GENERO,'S',DECODE((SELECT MAX(1) FROM DBASGU.CONFIG_ITEM_MODULO
             WHERE CD_MODULO = 'CAD_PAC'
             AND SN_VISIBLE = 'S'
             AND CD_ITEM = 'DSP_DS_IDENTIDADE_GENERO'
             AND CD_BLOCO = 'PACIENTE')
             ,1,IDEN.DS_IDENTIDADE_GENERO
             ,NULL),NULL) AS IDENTIDADE_GENERO
             ,DECODE(PACIENTE.SN_UTIL_ORIENTACAO_SEXUAL,'S',DECODE((SELECT MAX(1) FROM DBASGU.CONFIG_ITEM_MODULO
             WHERE CD_MODULO = 'CAD_PAC'
             AND SN_VISIBLE = 'S'
             AND CD_ITEM = 'DSP_DS_ORIENTACAO_SEXUAL'
             AND CD_BLOCO = 'PACIENTE'
             ),1,ORI.DS_ORIENTACAO_SEXUAL,NULL),NULL) AS ORIENTACAO_SEXUAL
FROM     DBAMV.PACIENTE
              ,DBAMV.CIDADE
              ,DBAMV.IDENTIDADE_GENERO IDEN
              ,DBAMV.ORIENTACAO_SEXUAL ORI
WHERE  PACIENTE.CD_CIDADE = CIDADE.CD_CIDADE(+)
AND PACIENTE.CD_IDENTIDADE_GENERO = IDEN.CD_IDENTIDADE_GENERO(+)
AND PACIENTE.CD_ORIENTACAO_SEXUAL = ORI.CD_ORIENTACAO_SEXUAL (+)
)
WHERE A1 = :CAMPO_A
