--R_EXA_ORIENTACAO
SELECT * FROM (
select distinct exa_lab.cd_exa_lab codigo,
       nm_exa_lab nome,
       ds_orientacao orientacao
  from dbamv.exa_lab,
         dbamv.itped_lab
 where itped_lab.cd_exa_lab = exa_lab.cd_exa_lab
    and ds_orientacao is not null
UNION
select distinct exa_lab.cd_exa_lab,
          nm_exa_lab,
          ds_orientacao
  from dbamv.exa_lab
where ds_orientacao is not null
:CAMPO_CF_EXA_LAB
)
ORDER BY CODIGO,NOME
