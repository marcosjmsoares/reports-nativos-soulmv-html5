--R_PEDEXACON
SELECT * FROM (
SELECT  CONVENIO.CD_CONVENIO              CD_CONVENIO
               ,CONVENIO.NM_CONVENIO             NM_CONVENIO
               ,PED_LAB.DT_PEDIDO                     DT_PEDIDO
               ,PED_LAB.CD_PED_LAB                   CD_PED_LAB3
               ,Count(ITPED_LAB.CD_ITPED_LAB) QTD_EXAMES
               ,PED_LAB.CD_PRESTADOR             CD_SOLICITA
               ,DECODE(PED_LAB.CD_PRESTADOR, NULL, PED_LAB.NM_PRESTADOR, PRESTADOR.NM_PRESTADOR)    SOLICITANTE
               ,PED_LAB.CD_ATENDIMENTO         CD_ATENDIMENTO
               ,PED_LAB.CD_SETOR                       CD_SETOR
               ,SETOR.NM_SETOR                          NM_SETOR
               ,ATENDIME.CD_PACIENTE               CD_PACIENTE
              ,PACIENTE.NM_PACIENTE                NM_PACIENTE
              ,ATENDIME.CD_ORI_ATE                  ORIGEM
              ,ORI_ATE.DS_ORI_ATE                     DS_ORI_ATE
FROM    DBAMV.PED_LAB
             ,DBAMV.PRESTADOR
             ,DBAMV.ATENDIME
             ,DBAMV.CONVENIO
             ,DBAMV.PACIENTE
             ,DBAMV.ORI_ATE
             ,DBAMV.ITPED_LAB
             ,DBAMV.LABORATORIO
             ,DBAMV.SETOR
Where      ATENDIME.CD_MULTI_EMPRESA =  :CAMPO_CD_MULTI_EMPRESA
AND         ATENDIME.CD_PACIENTE                 =  PACIENTE.CD_PACIENTE
AND         ATENDIME.CD_ORI_ATE                   =  ORI_ATE.CD_ORI_ATE
AND         ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND         ATENDIME.CD_ATENDIMENTO        =  PED_LAB.CD_ATENDIMENTO
AND         PED_LAB.CD_PRESTADOR              =  PRESTADOR.CD_PRESTADOR(+)
AND         PED_LAB.CD_CONVENIO                  =  CONVENIO.CD_CONVENIO
AND         PED_LAB.CD_SETOR                         =  SETOR.CD_SETOR
AND         PED_LAB.CD_PED_LAB                     =  ITPED_LAB.CD_PED_LAB
AND         ITPED_LAB.CD_LABORATORIO        = LABORATORIO.CD_LABORATORIO
   AND EXISTS  (SELECT 1 
                FROM DBAMV.USUARIOS_SET_EXA
                WHERE USUARIOS_SET_EXA.CD_SET_EXA = ITPED_LAB.CD_SET_EXA
                      AND USUARIOS_SET_EXA.NM_USUARIO = USER)
GROUP BY  CONVENIO.CD_CONVENIO
                    ,CONVENIO.NM_CONVENIO
                    ,PED_LAB.DT_PEDIDO
                    ,PED_LAB.CD_PED_LAB
                    ,PED_LAB.CD_PRESTADOR
                    ,DECODE(PED_LAB.CD_PRESTADOR, NULL, PED_LAB.NM_PRESTADOR, PRESTADOR.NM_PRESTADOR)
                    ,PED_LAB.CD_ATENDIMENTO
                    ,PED_LAB.CD_SETOR
                    ,SETOR.NM_SETOR
                    ,ATENDIME.CD_PACIENTE
                    ,PACIENTE.NM_PACIENTE
                    ,ATENDIME.CD_ORI_ATE
                   ,ORI_ATE.DS_ORI_ATE
)
ORDER BY CD_CONVENIO,
DT_PEDIDO,
CD_PED_LAB3,
CD_ATENDIMENTO
