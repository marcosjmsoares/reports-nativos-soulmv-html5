--R_EXAPER
SELECT * FROM (
SELECT  PER_EXALAB.CD_EXA_LAB CD_EXA_LAB2
       ,EXA_LAB.CD_EXA_LAB CODIGO
       ,EXA_LAB.NM_EXA_LAB DESCRICAO
       ,PERGUNTA.CD_PERGUNTA CD_PERGUNTA
       ,PERGUNTA.DS_PERGUNTA DS_PERGUNTA
FROM    dbamv.PER_EXALAB PER_EXALAB
       ,dbamv.EXA_LAB EXA_LAB
       ,dbamv.PERGUNTA PERGUNTA
WHERE EXA_LAB.CD_EXA_LAB = PER_EXALAB.CD_EXA_LAB
AND   PERGUNTA.CD_PERGUNTA = PER_EXALAB.CD_PERGUNTA
ORDER BY :CAMPO_ORDEM
)
