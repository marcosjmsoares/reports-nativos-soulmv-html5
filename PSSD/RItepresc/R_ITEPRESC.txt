--R_ITEPRESC
SELECT  tip_presc.cd_tip_presc   Codigo
               ,tip_presc.ds_tip_presc    Descricao
FROM    dbamv.tip_presc
             ,dbamv.tip_esq
WHERE  tip_presc.cd_tip_esq  =  tip_esq.cd_tip_esq
and          tip_esq.sn_exa_lab  =  'S'
and          nvl(tip_presc.cd_exa_lab,0) = 0
ORDER BY
