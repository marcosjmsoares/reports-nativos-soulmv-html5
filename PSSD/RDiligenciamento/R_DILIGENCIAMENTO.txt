--R_DILIGENCIAMENTO
SELECT solicitacao_diligencia_diagn.cd_solicitacao_diligencia,
       solicitacao_diligencia_diagn.cd_diligenciamento,
       solicitacao_diligencia_diagn.cd_ped_lab,
       solicitacao_diligencia_diagn.cd_amostra,
       solicitacao_diligencia_diagn.dt_solicitacao,
       solicitacao_diligencia_diagn.dt_conclusao_diligenciamento,
       solicitacao_diligencia_diagn.cd_usuario_conclusao_diligen,
       solicitacao_diligencia_diagn.dt_conclusao_diligenciamento,
       solicitacao_diligencia_diagn.ds_observacao,
       solicitacao_diligencia_diagn.tp_status,
       usuarios.nm_usuario,
       diligenciamento_diagn.DS_DILIGENCIAMENTO
FROM  dbamv.solicitacao_diligencia_diagn
INNER JOIN dbamv.diligenciamento_diagn ON (solicitacao_diligencia_diagn.CD_DILIGENCIAMENTO = diligenciamento_diagn.CD_DILIGENCIAMENTO)
INNER JOIN dbasgu.usuarios ON (usuarios.cd_usuario = solicitacao_diligencia_diagn.cd_usuario)
WHERE  solicitacao_diligencia_diagn.cd_solicitacao_diligencia = :CAMPO_CD_SOLIC_DILIGENCIA
