--R_SETEXA
SELECT SET_EXA.CD_SET_EXA CODIGO,
       SET_EXA.NM_SET_EXA DESCRICAO,
       SET_EXA.TP_SETOR TIPO,
       SET_EXA.NM_ATUAL_CONTROLE
  FROM dbamv.SET_EXA SET_EXA,
       dbamv.SETOR SETOR
  WHERE SET_EXA.CD_SETOR = SETOR.CD_SETOR
    AND SETOR.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
ORDER BY DESCRICAO
