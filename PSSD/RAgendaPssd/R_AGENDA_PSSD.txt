--R_AGENDA_PSSD
SELECT * FROM
  (SELECT ITSESC_DIA.CD_ESCALA_DIA CD_ESCALA_DIA ,
          SESC_DIA.DT_ESCALA_DIA DT_ESCALA_DIA ,
          SESC_DIA.CD_RECURSO CD_RECURSO ,
          SRECURSO.NM_RECURSO NM_RECURSO ,
          SET_EXA.NM_SET_EXA NM_SET_EXA ,
          SESC_DIA.SN_ATIVO SN_ATIVO ,
          ITSESC_DIA.CD_PACIENTE CD_PACIENTE ,
          ITSESC_DIA.NM_PACIENTE NM_PACIENTE ,
     (SELECT PAC.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE PAC WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ITSESC_DIA.CD_PACIENTE = PAC.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
          ITSESC_DIA.HR_AGENDA HR_AGENDA ,
          TO_CHAR(ITSESC_DIA.HR_AGENDA, 'HH24') HR_AGENDA2 ,
          ITSESC_DIA.NR_TELEFONE NR_TELEFONE ,
          CONVENIO.NM_CONVENIO NM_CONVENIO,
          DECODE(ITSESC_DIA.NR_IDADE, NULL, SUBSTR(PACK_INTERNAMENTO.RETORNA_IDADE(PACIENTE.DT_NASCIMENTO, SYSDATE), 1, 3), TO_CHAR(ITSESC_DIA.NR_IDADE)) NR_IDADE
   FROM dbamv.SESC_DIA ,
        dbamv.ITSESC_DIA ,
        dbamv.SRECURSO ,
        dbamv.PACIENTE ,
        dbamv.CONVENIO ,
        dbamv.SET_EXA ,
        dbamv.Empresa_Convenio
   WHERE Empresa_Convenio.Cd_Convenio (+) = Convenio.Cd_Convenio
     AND Empresa_Convenio.Cd_Multi_Empresa (+) = :CAMPO_CD_MULTI_EMPRESA
     AND TRUNC(SESC_DIA.DT_ESCALA_DIA) BETWEEN :CAMPO_DT_INIC AND 
     AND NVL(SESC_DIA.SN_ATIVO, 'S') = 'S'
     AND SESC_DIA.CD_ESCALA_DIA = ITSESC_DIA.CD_ESCALA_DIA
     AND NVL(ITSESC_DIA.TP_SITUACAO, 'A') <> 'C'
     AND ITSESC_DIA.CD_PACIENTE = PACIENTE.CD_PACIENTE(+)
     AND ITSESC_DIA.CD_CONVENIO = CONVENIO.CD_CONVENIO(+)
     AND SESC_DIA.CD_RECURSO = SRECURSO.CD_RECURSO
     AND SET_EXA.CD_SET_EXA = SRECURSO.CD_SET_EXA
     AND SET_EXA.CD_SET_EXA IN
       (SELECT cd_set_exa
        FROM DBAMV.usuarios_set_exa
        WHERE usuarios_set_exa.nm_usuario =
            (SELECT user_users.username
             FROM user_users)) 
   ORDER BY DT_ESCALA_DIA ,
            CD_RECURSO ,
            TO_CHAR(HR_AGENDA, 'HH24'))
ORDER BY NM_RECURSO ,
         CD_RECURSO ,
         DT_ESCALA_DIA ,
         TO_CHAR(HR_AGENDA, 'HH24')
