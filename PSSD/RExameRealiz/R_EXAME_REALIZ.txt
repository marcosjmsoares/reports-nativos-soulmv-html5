--R_EXAME_REALIZ
SELECT * FROM
  (SELECT ATENDIME.CD_ATENDIMENTO CD_ATENDIMENTO ,
          PACIENTE.CD_PACIENTE CD_PACIENTE ,
          PACIENTE.NM_PACIENTE NM_PACIENTE ,
     (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
          PED_LAB.CD_PED_LAB CD_PED_LAB ,
          PED_LAB.DT_PEDIDO DT_PEDIDO ,
          PED_LAB.NR_CONTROLE NR_CONTROLE ,
          CONVENIO.NM_CONVENIO NM_CONVENIO ,
          CON_PLA.DS_CON_PLA DS_CON_PLA ,
          SETOR.NM_SETOR NM_SETOR ,
          SET_EXA.NM_SET_EXA NM_SET_EXA ,
          NVL(PED_LAB.NM_PRESTADOR, PRESTADOR.NM_PRESTADOR) NM_PRESTADOR ,
          EXA_LAB.CD_EXA_LAB CD_EXA_LAB ,
          EXA_LAB.NM_EXA_LAB NM_EXA_LAB ,
          ITPED_LAB.DT_LAUDO DT_LAUDO ,
          ENTREGA_EXAME_SADT.DT_ENTREGA_EXAME DT_ENTREGA_EXAME ,
          ITPED_LAB.CD_ITPED_LAB CD_ITPED_LAB
   FROM DBAMV.ATENDIME ,
        DBAMV.PACIENTE ,
        DBAMV.PED_LAB ,
        DBAMV.CONVENIO ,
        DBAMV.CON_PLA ,
        DBAMV.SETOR ,
        DBAMV.SET_EXA ,
        DBAMV.PRESTADOR ,
        DBAMV.ITPED_LAB ,
        DBAMV.EXA_LAB ,
        DBAMV.ITENTREGA_EXAME_SADT ,
        DBAMV.ENTREGA_EXAME_SADT
   WHERE ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
     AND ATENDIME.CD_ATENDIMENTO = PED_LAB.CD_ATENDIMENTO
     AND PED_LAB.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
     AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
     AND ATENDIME.CD_CONVENIO = CON_PLA.CD_CONVENIO(+)
     AND ATENDIME.CD_CON_PLA = CON_PLA.CD_CON_PLA(+)
     AND PED_LAB.CD_SETOR = SETOR.CD_SETOR
     AND PED_LAB.CD_PED_LAB = ITPED_LAB.CD_PED_LAB
     AND ITPED_LAB.CD_SET_EXA = SET_EXA.CD_SET_EXA
     AND ITPED_LAB.CD_EXA_LAB = EXA_LAB.CD_EXA_LAB
     AND ITPED_LAB.CD_ITPED_LAB = ITENTREGA_EXAME_SADT.CD_ITPED_LAB(+)
     AND ITPED_LAB.CD_PED_LAB = ITENTREGA_EXAME_SADT.CD_PED_LAB(+)
     AND ITPED_LAB.CD_EXA_LAB = ITENTREGA_EXAME_SADT.CD_EXA_LAB(+)
     AND ITPED_LAB.CD_ENTREGA_EXAME = ITENTREGA_EXAME_SADT.CD_ENTREGA_EXAME(+)
     AND ITENTREGA_EXAME_SADT.CD_ENTREGA_EXAME = ENTREGA_EXAME_SADT.CD_ENTREGA_EXAME(+)
     AND EXISTS
       (SELECT 1
        FROM DBAMV.USUARIOS_SET_EXA
        WHERE USUARIOS_SET_EXA.CD_SET_EXA = ITPED_LAB.CD_SET_EXA
          AND USUARIOS_SET_EXA.NM_USUARIO = USER) 
:CAMPO_COND_EXAME )
ORDER BY CD_PED_LAB,
         DT_PEDIDO,
         NR_CONTROLE,
         CD_ATENDIMENTO,
         CD_PACIENTE,
         NM_PACIENTE,
         NM_CONVENIO,
         DS_CON_PLA,
         NM_PRESTADOR,
         NM_SETOR,
         DT_LAUDO
