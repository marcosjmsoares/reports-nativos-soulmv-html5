--R_PEDEXA_R_5_Q_SINTETICO
SELECT * FROM
  (SELECT TRUNC(PED_LAB.DT_PEDIDO) DT_PEDIDO ,
          PED_LAB.CD_PED_LAB ,
          ATENDIME.CD_PACIENTE ,
          PACIENTE.NM_PACIENTE ,
      (SELECT P.NM_SOCIAL_PACIENTE 
       FROM DBAMV.PACIENTE P 
       WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' 
        AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
          SETOR.NM_SETOR ,
          CONVENIO.NM_CONVENIO ,
          EXA_LAB.CD_EXA_LAB ,
          EXA_LAB.NM_EXA_LAB ,
          LABORATORIO.CD_LABORATORIO ,
          LABORATORIO.DS_LABORATORIO
   FROM DBAMV.ATENDIME ,
        DBAMV.PACIENTE ,
        DBAMV.PED_LAB ,
        DBAMV.ITPED_LAB ,
        DBAMV.EXA_LAB ,
        DBAMV.SETOR ,
        DBAMV.CONVENIO ,
        DBAMV.LABORATORIO ,
        DBAMV.ORI_ATE,
        Dbamv.Empresa_Convenio
   WHERE Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
     AND Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
     AND ATENDIME.CD_ATENDIMENTO = PED_LAB.CD_ATENDIMENTO
     AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
     AND ATENDIME.CD_ORI_ATE = ORI_ATE.CD_ORI_ATE
     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA 
     AND CONVENIO.CD_CONVENIO = PED_LAB.CD_CONVENIO
     AND SETOR.CD_SETOR = PED_LAB.CD_SETOR
     AND PED_LAB.CD_PED_LAB = ITPED_LAB.CD_PED_LAB
     AND ITPED_LAB.CD_EXA_LAB = EXA_LAB.CD_EXA_LAB
     AND ITPED_LAB.CD_LABORATORIO = LABORATORIO.CD_LABORATORIO
     AND TO_DATE(TO_CHAR(PED_LAB.DT_PEDIDO, 'DD/MM/YYYY'), 'DD/MM/YYYY') BETWEEN :CAMPO_DTINI ORDER BY PED_LAB.CD_PED_LAB)
ORDER BY DS_LABORATORIO,
         CD_LABORATORIO,
         NM_PACIENTE,
         NM_EXA_LAB

--R_PEDEXA_R_CG__G_PED_LAB1_Q_EXAME_SOL_LABORATORIO
SELECT * FROM
  (SELECT PED_LAB.CD_PRESTADOR CD_SOLICITA ,
          ITPED_LAB.CD_ITPED_LAB CD_ITPED_LAB ,
          DECODE(PED_LAB.CD_PRESTADOR, 
                 NULL, 
                 PED_LAB.NM_PRESTADOR, 
                 PRESTADOR.NM_PRESTADOR) SOLICITANTE ,
          PED_LAB.CD_SETOR CD_SETOR ,
          SETOR.NM_SETOR NM_SETOR ,
          EXA_LAB.CD_EXA_LAB CD_EXA_LAB ,
          DECODE(EXA_LAB.NM_MNEMONICO, 
                 NULL, 
                 EXA_LAB.NM_EXA_LAB, 
                 EXA_LAB.NM_MNEMONICO || ' - ' || EXA_LAB.NM_EXA_LAB) NM_EXA_LAB ,
          ATENDIME.CD_PACIENTE CD_PACIENTE ,
          PACIENTE.NM_PACIENTE NM_PACIENTE ,
      (SELECT P.NM_SOCIAL_PACIENTE 
       FROM DBAMV.PACIENTE P 
       WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' 
        AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
          ATENDIME.CD_CONVENIO CD_CONVENIO ,
          CONVENIO.NM_CONVENIO NM_CONVENIO ,
          ATENDIME.CD_ORI_ATE ORIGEM ,
          ORI_ATE.DS_ORI_ATE DS_ORI_ATE , 
          ATENDIME.NR_CARTEIRA NR_CARTEIRA ,
     (SELECT COUNT(*)
      FROM DBAMV.EXA_LAB,
           DBAMV.PED_LAB,
           DBAMV.ITPED_LAB,
           DBAMV.LABORATORIO
      WHERE EXA_LAB.CD_EXA_LAB = ITPED_LAB.CD_EXA_LAB
        AND ITPED_LAB.CD_PED_LAB = PED_LAB.CD_PED_LAB
        AND ITPED_LAB.CD_LABORATORIO = LABORATORIO.CD_LABORATORIO
        AND LABORATORIO.CD_LABORATORIO = :CAMPO_CD_LABORATORIO
        AND PED_LAB.DT_PEDIDO BETWEEN To_Date(To_Char(:CAMPO_DT_PEDIDO, 'dd/mm/yyyy')||' 00', 'dd/mm/yyyy HH24') 
        AND To_Date(To_Char(:CAMPO_DT_PEDIDO, 'dd/mm/yyyy')||' 23', 'dd/mm/yyyy HH24') ) QTD_EXAME_POR_DATA
   FROM DBAMV.PED_LAB ,
        DBAMV.PRESTADOR ,
        DBAMV.SETOR ,
        DBAMV.ATENDIME ,
        DBAMV.CONVENIO ,
        DBAMV.PACIENTE ,
        DBAMV.ORI_ATE ,
        DBAMV.ITPED_LAB ,
        DBAMV.LABORATORIO , 
        DBAMV.EXA_LAB,
        Dbamv.Empresa_Convenio
   WHERE Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
     AND EXA_LAB.CD_EXA_LAB = ITPED_LAB.CD_EXA_LAB
     AND LABORATORIO.CD_LABORATORIO = ITPED_LAB.CD_LABORATORIO
     AND Empresa_Convenio.Cd_Multi_Empresa = ATENDIME.CD_MULTI_EMPRESA
     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
     AND PED_LAB.DT_PEDIDO BETWEEN To_Date(To_Char(:CAMPO_DTINI, 'dd/mm/yyyy')||' 00', 'dd/mm/yyyy HH24') AND To_Date(To_Char(:CAMPO_DTFIM, 'dd/mm/yyyy')||' 23', 'dd/mm/yyyy HH24')
     AND PRESTADOR.CD_PRESTADOR(+) = PED_LAB.CD_PRESTADOR
     AND PED_LAB.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
     AND PED_LAB.CD_ATENDIMENTO = :CAMPO_CD_ATENDIMENTO
     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
     AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
     AND ATENDIME.CD_ORI_ATE = ORI_ATE.CD_ORI_ATE
     AND PED_LAB.CD_PED_LAB = ITPED_LAB.CD_PED_LAB
     AND PED_LAB.CD_PED_LAB = :CAMPO_CD_PED_LAB
     AND ITPED_LAB.CD_LABORATORIO = LABORATORIO.CD_LABORATORIO
     AND LABORATORIO.CD_LABORATORIO = :CAMPO_CD_LABORATORIO
     AND SETOR.CD_SETOR = PED_LAB.CD_SETOR 
   GROUP BY PED_LAB.CD_PRESTADOR ,
            DECODE(PED_LAB.CD_PRESTADOR, NULL, PED_LAB.NM_PRESTADOR, PRESTADOR.NM_PRESTADOR) ,
            PED_LAB.CD_SETOR ,
            SETOR.NM_SETOR ,
            ATENDIME.CD_PACIENTE ,
            EXA_LAB.CD_EXA_LAB ,
            DECODE(EXA_LAB.NM_MNEMONICO, NULL, EXA_LAB.NM_EXA_LAB, EXA_LAB.NM_MNEMONICO || ' - ' || EXA_LAB.NM_EXA_LAB) ,
            PACIENTE.NM_PACIENTE ,
            ATENDIME.CD_CONVENIO ,
            CONVENIO.NM_CONVENIO ,
            ATENDIME.CD_ORI_ATE ,
            ORI_ATE.DS_ORI_ATE , 
            ATENDIME.NR_CARTEIRA ,
            CD_ITPED_LAB)
ORDER BY CD_SOLICITA,
         SOLICITANTE,
         CD_SETOR,
         NM_SETOR,
         CD_EXA_LAB,
         NM_EXA_LAB,
         CD_PACIENTE,
         NM_PACIENTE,
         CD_CONVENIO,
         NM_CONVENIO,
         ORIGEM,
         DS_ORI_ATE,
         NR_CARTEIRA

--R_PEDEXA_R_LABORATORIO_Q_EXAME_SOL_LABORATORIO
SELECT * FROM (
SELECT  ITPED_LAB.CD_LABORATORIO             CD_LABORATORIO
              ,LABORATORIO.DS_LABORATORIO       DS_LABORATORIO
              ,TRUNC ( PED_LAB.DT_PEDIDO )           DT_PEDIDO
              ,PED_LAB.CD_PED_LAB                          CD_PED_LAB
              ,PED_LAB.CD_ATENDIMENTO                CD_ATENDIMENTO
	,count(ITPED_LAB.CD_itped_lab) QTD
FROM  DBAMV.PED_LAB
            ,DBAMV.PRESTADOR
            ,DBAMV.SETOR
            ,DBAMV.ATENDIME
            ,DBAMV.CONVENIO
            ,DBAMV.PACIENTE
            ,DBAMV.ORI_ATE
            ,DBAMV.ITPED_LAB
            ,DBAMV.LABORATORIO
            ,DBAMV.EXA_LAB
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
AND        EXA_LAB.CD_EXA_LAB  =  ITPED_LAB.CD_EXA_LAB
AND        LABORATORIO.CD_LABORATORIO  =  ITPED_LAB.CD_LABORATORIO
And        Empresa_Convenio.Cd_Multi_Empresa = ATENDIME.CD_MULTI_EMPRESA
and        ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
And        PED_LAB.DT_PEDIDO  BETWEEN  To_Date(To_Char(:CAMPO_DTINI, 'dd/mm/yyyy')||' 00', 'dd/mm/yyyy HH24')  AND  To_Date(To_Char(:CAMPO_DTFIM, 'dd/mm/yyyy')||' 23', 'dd/mm/yyyy HH24')
AND        PRESTADOR.CD_PRESTADOR(+)   = PED_LAB.CD_PRESTADOR
AND        PED_LAB.CD_ATENDIMENTO   = ATENDIME.CD_ATENDIMENTO
AND        ATENDIME.CD_PACIENTE     = PACIENTE.CD_PACIENTE
AND        ATENDIME.CD_CONVENIO     = CONVENIO.CD_CONVENIO
AND        ATENDIME.CD_ORI_ATE      = ORI_ATE.CD_ORI_ATE
AND        PED_LAB.CD_PED_LAB       = ITPED_LAB.CD_PED_LAB
AND        ITPED_LAB.CD_LABORATORIO = LABORATORIO.CD_LABORATORIO
AND        SETOR.CD_SETOR = PED_LAB.CD_SETOR
GROUP BY LABORATORIO.DS_LABORATORIO
		   ,ITPED_LAB.CD_LABORATORIO
		   ,PED_LAB.CD_ATENDIMENTO
                   ,PED_LAB.DT_PEDIDO
                   ,PED_LAB.CD_PED_LAB
)
ORDER BY DS_LABORATORIO,
DT_PEDIDO,
CD_ATENDIMENTO,CD_PED_LAB
