--R_EXALAB
SELECT DISTINCT
               EXA_LAB.CD_EXA_LAB
              ,EXA_LAB.NM_EXA_LAB
              ,EXA_LAB.SN_CULTURA
              ,EXA_LAB.SN_CONTAGIOSO
              ,EXA_LAB.TP_SEXO
              ,EXA_LAB.VL_CUSTO
              ,EXA_LAB.CD_VERSAO_ATUAL
              ,EXA_LAB.DS_ORIENTACAO
              ,EXA_LAB.NR_ORDEM_DE_IMPRESSAO
              ,EXA_LAB.NR_DIAS_PERIODO
              ,EXA_LAB.NR_SOLICITACOES_PERIODO    SOLIC_PERIODO
              ,EXA_LAB.NM_MNEMONICO NM_MNEMONICO
              ,EXA_LAB.CD_PRO_FAT CD_PRO_FAT
FROM    dbamv.EXA_LAB
             ,dbamv.EXA_LAB_SET_EXA
             ,dbamv.LABORATORIO
             ,dbamv.EXA_LAB_LABORATORIO
WHERE  EXA_LAB.CD_EXA_LAB  =  EXA_LAB_SET_EXA.CD_EXA_LAB(+)
AND        EXA_LAB.CD_EXA_LAB =  EXA_LAB_LABORATORIO.CD_EXA_LAB(+)
