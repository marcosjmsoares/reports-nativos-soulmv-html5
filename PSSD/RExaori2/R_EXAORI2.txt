--R_EXAORI2
SELECT
       atendime.cd_ori_ate   ,
       ori_ate.ds_ori_ate       ,
       itped_lab.cd_exa_lab ,
       exa_lab.nm_exa_lab  ,
       count( atendime.cd_ori_ate ) qtd_origem
FROM   dbamv.atendime  ,
             dbamv.ped_lab          ,
             dbamv.ori_ate            ,
             dbamv.itped_lab        ,
             dbamv.exa_lab
WHERE      atendime.cd_ori_ate     =  ori_ate.cd_ori_ate            AND
                    atendime.cd_atendimento =  ped_lab.cd_atendimento      AND
                    atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA      AND
                    ped_lab.cd_ped_lab         =   itped_lab.cd_ped_lab          AND
                    itped_lab.cd_exa_lab = exa_lab.cd_exa_lab                     AND
                   trunc(ped_lab.dt_pedido)  BETWEEN  :CAMPO_DTINIC  AND      AND
EXISTS (SELECT 1
               FROM DBAMV.USUARIOS_SET_EXA
               WHERE USUARIOS_SET_EXA.CD_SET_EXA = ITPED_LAB.CD_SET_EXA
               AND   USUARIOS_SET_EXA.NM_USUARIO = USER)
GROUP BY
       atendime.cd_ori_ate   ,
       ori_ate.ds_ori_ate       ,
       itped_lab.cd_exa_lab ,
       exa_lab.nm_exa_lab
ORDER BY  atendime.cd_ori_ate
