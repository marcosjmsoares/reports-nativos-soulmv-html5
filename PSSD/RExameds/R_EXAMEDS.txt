--R_EXAMEDS
SELECT   ped_lab.cd_prestador                  cd_prestador
                ,prestador.ds_codigo_conselho     ds_codigo_conselho
                ,decode(ped_lab.cd_prestador, null, ped_lab.nm_prestador, prestador.nm_prestador)   nm_prestador
                ,count(itped_lab.cd_exa_lab)        qtd_exame
FROM  dbamv.atendime
           ,dbamv.ped_lab
           ,dbamv.ori_ate
           ,dbamv.itped_lab
           ,dbamv.exa_lab
           ,dbamv.prestador
           ,dbamv.convenio
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And atendime.cd_ori_ate  =  ori_ate.cd_ori_ate
and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
AND atendime.cd_atendimento  =  ped_lab.cd_atendimento
AND EXISTS (SELECT 1
                       FROM DBAMV.USUARIOS_SET_EXA
                       WHERE USUARIOS_SET_EXA.CD_SET_EXA = ITPED_LAB.CD_SET_EXA
                       AND   USUARIOS_SET_EXA.NM_USUARIO = USER)
AND        ped_lab.cd_ped_lab  =  itped_lab.cd_ped_lab
AND        itped_lab.cd_exa_lab  =  exa_lab.cd_exa_lab
AND        ped_lab.cd_prestador  = prestador.cd_prestador(+)
AND        atendime.cd_convenio = convenio.cd_convenio
GROUP BY  ped_lab.cd_prestador
                   ,prestador.ds_codigo_conselho
                   ,decode(ped_lab.cd_prestador, null, ped_lab.nm_prestador, prestador.nm_prestador)
ORDER BY qtd_exame desc
