--R_Listagem_Bloqueio
SELECT CD_BLOQUEIO_EXAME
          ,NM_SET_EXA
          ,NM_EXA_LAB
          ,DECODE(BLOQUEIO_EXAME.TP_BLOQUEIO, 'C', 'Bloqueio Solocitação Exame', 'Bloqueio de Digitação de Resultado') TIPO_BLOQUEIO
          ,DS_MATERIAL
          ,DS_MOTIVO
        FROM DBAMV.BLOQUEIO_EXAME,
             DBAMV.SET_EXA,
             DBAMV.EXA_LAB,
             DBAMV.MATERIAL
      WHERE  BLOQUEIO_EXAME.CD_SET_EXA = SET_EXA.CD_SET_EXA
        AND  BLOQUEIO_EXAME.CD_EXA_LAB = EXA_LAB.CD_EXA_LAB
        AND  BLOQUEIO_EXAME.CD_MATERIAL = MATERIAL.CD_MATERIAL (+)
        AND  BLOQUEIO_EXAME.SN_ATIVO = :CAMPO_SN_ATIVO
