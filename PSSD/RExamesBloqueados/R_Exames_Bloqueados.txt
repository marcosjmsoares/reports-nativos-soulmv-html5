--R_Exames_Bloqueados
SELECT EXA_LAB.NM_MNEMONICO || ' - ' ||   EXA_LAB.NM_EXA_LAB EXAME,
      DECODE(TP_BLOQUEIO, 'C', 'Bloqueio Solocitação Exame', 'Bloqueio de Digitação de Resultado') TIPO_BLOQUEIO,
      ITPED_LAB.CD_PED_LAB,
      ITPED_LAB.CD_ITPED_LAB,
       (SELECT Sum(1)
         FROM DBAMV.ITPED_LAB,
         DBAMV.EXA_LAB EXA_LAB1,
         DBAMV.BLOQUEIO_EXAME,
         DBAMV.PED_LAB
        WHERE ITPED_LAB.CD_BLOQUEIO_EXAME =  BLOQUEIO_EXAME.CD_BLOQUEIO_EXAME
        AND ITPED_LAB.CD_EXA_LAB = EXA_LAB1.CD_EXA_LAB
        AND PED_LAB.CD_PED_LAB  =  ITPED_LAB.CD_PED_LAB
        AND EXA_LAB1.CD_EXA_LAB = EXA_LAB.CD_EXA_LAB
        :CAMPO_COND_CD_EXAME
        )   TOTAL
  FROM DBAMV.ITPED_LAB,
       DBAMV.EXA_LAB,
       DBAMV.BLOQUEIO_EXAME,
       DBAMV.PED_LAB
 WHERE ITPED_LAB.CD_BLOQUEIO_EXAME =  BLOQUEIO_EXAME.CD_BLOQUEIO_EXAME
   AND ITPED_LAB.CD_EXA_LAB = EXA_LAB.CD_EXA_LAB
   AND PED_LAB.CD_PED_LAB  =  ITPED_LAB.CD_PED_LAB
