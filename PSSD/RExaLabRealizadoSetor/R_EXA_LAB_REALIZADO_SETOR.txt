--R_EXA_LAB_REALIZADO_SETOR
SELECT Exame ,
       Pedido ,
       SetorSolicitante ,
       Atendimento ,
       Paciente ,
       NM_SOCIAL_PACIENTE ,
       DtRealizacao ,
       Convenio ,
       Plano ,
       Qtd ,
       CH ,
       Valor
FROM
  (SELECT Exa_Lab.Nm_Mnemonico ||'  '||Exa_Lab.Nm_Exa_Lab Exame ,
          Ped_Lab.Cd_Ped_Lab Pedido ,
          Ped_Lab.Cd_Setor||' - '||Setor.Nm_Setor SetorSolicitante ,
          Ped_Lab.Cd_Atendimento Atendimento ,
          Atendime.Cd_Paciente||' - '||Paciente.Nm_Paciente Paciente ,
      (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
          Itped_lab.Dt_Laudo DtRealizacao ,
          Convenio.Nm_Convenio Convenio ,
          Con_Pla.Ds_Con_Pla Plano ,
          Exa_Lab.Cd_Exa_Lab ,
          count(Exa_Lab.Cd_Exa_Lab) Qtd ,
          dbamv.calc_qt_ch(Decode(Convenio.Tp_Convenio, 
                              'A', Exa_Lab.CD_PROCEDIMENTO_SIA , 
                              'H', Exa_Lab.CD_PROCEDIMENTO_SIH , 
                              'C', Exa_Lab.Cd_Pro_Fat , 
                              'P', Exa_Lab.Cd_Pro_Fat) ,
                              Ped_Lab.Dt_Pedido ,
                              Ped_Lab.Hr_Ped_Lab ,
                              Convenio.Cd_Convenio ,
                              Nvl(Con_Pla.Cd_Con_Pla, 1) ,
                              Atendime.Tp_Atendimento , NULL , NULL , 100 , NULL) CH ,
          Count(Exa_Lab.Cd_Exa_Lab) * Dbamv.Calc_Vl_Proc_Unit(Convenio.Tp_Convenio , 
                Decode(Convenio.Tp_Convenio, 
                'A', Exa_Lab.CD_PROCEDIMENTO_SIA , 
                'H', Exa_Lab.CD_PROCEDIMENTO_SIH , 
                'C', Exa_Lab.Cd_Pro_Fat , 
                'P', Exa_Lab.Cd_Pro_Fat) , 
          Ped_Lab.Dt_Pedido , 
          Ped_Lab.Hr_Ped_Lab , 
          Ped_Lab.Cd_Convenio , 
          NVL(Ped_Lab.Cd_Con_Pla, 1) , 
          Atendime.Tp_Atendimento , 
          Atendime.Cd_Tip_Acom , 'U') Valor
   FROM Dbamv.Exa_Lab ,
        Dbamv.Itped_Lab ,
        Dbamv.Ped_Lab ,
        Dbamv.Setor ,
        Dbamv.Atendime ,
        Dbamv.Paciente ,
        Dbamv.Convenio ,
        Dbamv.Con_Pla ,
        Dbamv.Prestador ,
        Dbamv.Set_Exa,
        Dbamv.Empresa_Con_Pla
   WHERE Empresa_Con_Pla.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
     AND Empresa_Con_Pla.Cd_Convenio = Atendime.Cd_Convenio
     AND Empresa_Con_Pla.Cd_Con_Pla = NVL(Atendime.Cd_Con_pla, 1)
     AND Exa_Lab.Cd_Exa_Lab = Itped_Lab.Cd_Exa_Lab
     AND Ped_Lab.Cd_Ped_Lab = Itped_Lab.Cd_Ped_Lab
     AND Setor.Cd_Setor = Ped_Lab.Cd_Setor
     AND Atendime.Cd_Atendimento = Ped_Lab.Cd_Atendimento
     AND Atendime.Cd_Paciente = Paciente.Cd_Paciente
     AND Atendime.Cd_Convenio = Convenio.Cd_Convenio
     AND Con_Pla.Cd_Convenio = Convenio.Cd_Convenio
     AND NVL(Atendime.Cd_Con_Pla, 1) = Con_Pla.Cd_Con_Pla
     AND itped_lab.sn_realizado = 'S'
     AND Trunc(Itped_lab.Dt_Laudo) BETWEEN :CAMPO_DT_INICIAL AND 
     AND ItPed_Lab.Cd_Set_Exa = set_exa.cd_set_exa
     AND ItPed_Lab.Cd_Medico_Exec = Prestador.Cd_Prestador(+)
     AND atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
   GROUP BY Exa_Lab.Nm_Mnemonico ,
            Exa_Lab.Nm_Exa_Lab ,
            Ped_Lab.Cd_Ped_Lab ,
            Ped_Lab.Cd_Setor ,
            Setor.Nm_Setor ,
            Ped_Lab.Cd_Atendimento ,
            Atendime.Cd_Paciente ,
            Paciente.Nm_Paciente ,
            Paciente.NM_SOCIAL_PACIENTE ,
            Itped_lab.Dt_Laudo ,
            Convenio.Nm_Convenio ,
            Con_Pla.Ds_Con_Pla ,
            Convenio.Tp_Convenio ,
            Convenio.Cd_Convenio ,
            Con_Pla.Cd_Con_Pla ,
            Atendime.Tp_atendimento ,
            Atendime.Cd_Tip_acom ,
            Dbamv.Calc_Vl_Proc_Unit(Convenio.Tp_Convenio , 
                                    Decode(Convenio.Tp_Convenio, 
                                    'A', Exa_Lab.CD_PROCEDIMENTO_SIA , 
                                    'H', Exa_Lab.CD_PROCEDIMENTO_SIH , 
                                    'C', Exa_Lab.Cd_Pro_Fat , 
                                    'P', Exa_Lab.Cd_Pro_Fat) , 
                                    Ped_Lab.Dt_Pedido , 
                                    Ped_Lab.Hr_Ped_Lab , 
                                    Ped_Lab.Cd_Convenio , 
                                    NVL(Ped_Lab.Cd_Con_Pla, 1) , 
                                    Atendime.Tp_Atendimento , 
                                    Atendime.Cd_Tip_Acom , 'U') ,
            dbamv.calc_qt_ch(Decode(Convenio.Tp_Convenio, 
                                    'A', Exa_Lab.CD_PROCEDIMENTO_SIA , 
                                    'H', Exa_Lab.CD_PROCEDIMENTO_SIH , 
                                    'C', Exa_Lab.Cd_Pro_Fat , 
                                    'P', Exa_Lab.Cd_Pro_Fat) , 
                                    Ped_Lab.Dt_Pedido , 
                                    Ped_Lab.Hr_Ped_Lab , 
                                    Convenio.Cd_Convenio , 
                                    Nvl(Con_Pla.Cd_Con_Pla, 1) , 
                                    Atendime.Tp_atendimento ,
                                    NULL , 
                                    NULL , 
                                    0 , 
                                    NULL) ,
            Exa_Lab.Cd_Exa_Lab ,
            Exa_Lab.CD_PROCEDIMENTO_SIA ,
            Exa_Lab.CD_PROCEDIMENTO_SIH ,
            Exa_Lab.Cd_Pro_Fat ,
            Ped_Lab.Dt_Pedido ,
            Ped_Lab.Hr_Ped_Lab)
GROUP BY Exame ,
         Pedido ,
         SetorSolicitante ,
         Atendimento ,
         Paciente ,
         NM_SOCIAL_PACIENTE ,
         DtRealizacao ,
         Convenio ,
         Plano ,
         Qtd ,
         CH ,
         Valor
ORDER BY SetorSolicitante,
         Atendimento,
         Exame
