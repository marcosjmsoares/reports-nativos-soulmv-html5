--R_ENTREGA_POR_TIPO
SELECT * FROM
  (SELECT LOCAL_PADRAO.CD_LOCAL_PADRAO ,
          LOCAL_PADRAO.DS_LOCAL_PADRAO ,
          DECODE(LOCAL_PADRAO.CD_LOCAL_PADRAO, NULL, 'OUTROS LOCAIS', LOCAL_PADRAO.CD_LOCAL_PADRAO||' - '||LOCAL_PADRAO.DS_LOCAL_PADRAO) LOCAL_PADRAO ,
          ENDER_COLETA_ENTREGA.CD_TIPO_ENTREGA ,
          TIPO_ENTREGA.ds_tipo_entrega ,
          PED_LAB.CD_ATENDIMENTO ,
          PED_LAB.CD_PED_LAB ,
          PACIENTE.CD_PACIENTE ,
          PACIENTE.NM_PACIENTE ,
     (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
          PACIENTE.CD_PACIENTE||' - '||PACIENTE.NM_PACIENTE PACIENTE ,
          ORI_ATE.DS_ORI_ATE ,
          ITPED_LAB.CD_EXA_LAB ,
          ENDER_COLETA_ENTREGA.ds_logradouro ,
          ENDER_COLETA_ENTREGA.nr_logradouro ,
          ENDER_COLETA_ENTREGA.ds_complemento ,
          ENDER_COLETA_ENTREGA.nm_bairro ,
          ENDER_COLETA_ENTREGA.nr_cep ,
          ENDER_COLETA_ENTREGA.nr_telefone ,
          ENDER_COLETA_ENTREGA.nr_celular ,
          ENDER_COLETA_ENTREGA.nr_fax ,
          ENDER_COLETA_ENTREGA.ds_email ,
          ENDER_COLETA_ENTREGA.cd_cidade ,
          Cidade.Nm_Cidade
   FROM DBAMV.PED_LAB ,
        DBAMV.ATENDIME ,
        DBAMV.ITPED_LAB ,
        DBAMV.PACIENTE ,
        DBAMV.ORI_ATE ,
        DBAMV.ENDER_COLETA_ENTREGA ,
        DBAMV.TIPO_ENTREGA_RESULTADO TIPO_ENTREGA ,
        DBAMV.LOCAL_PADRAO_ENTREGA LOCAL_PADRAO ,
        Dbamv.Cidade
   WHERE ATENDIME.CD_ATENDIMENTO = PED_LAB.CD_ATENDIMENTO
     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
     AND ATENDIME.CD_ORI_ATE = ORI_ATE.CD_ORI_ATE
     AND PED_LAB.CD_PED_LAB = ENDER_COLETA_ENTREGA.CD_PED_LAB
     AND PED_LAB.CD_PED_LAB = ITPED_LAB.CD_PED_LAB
     AND ENDER_COLETA_ENTREGA.CD_Cidade = Cidade.Cd_Cidade
     AND ENDER_COLETA_ENTREGA.CD_TIPO_ENTREGA = TIPO_ENTREGA.CD_TIPO_ENTREGA
     AND ENDER_COLETA_ENTREGA.CD_LOCAL_PADRAO = LOCAL_PADRAO.CD_LOCAL_PADRAO(+)
     AND ENDER_COLETA_ENTREGA.TP_SERVICO = 'E'
     AND ENDER_COLETA_ENTREGA.DT_SERVICO BETWEEN :CAMPO_DTINIC ORDER BY ENDER_COLETA_ENTREGA.CD_TIPO_ENTREGA ,
            DS_LOCAL_PADRAO ,
            PED_LAB.CD_ATENDIMENTO)
ORDER BY DS_TIPO_ENTREGA,
         CD_TIPO_ENTREGA,
         CD_LOCAL_PADRAO,
         DS_LOCAL_PADRAO,
         LOCAL_PADRAO,
         CD_ATENDIMENTO,
         CD_PED_LAB,
         CD_PACIENTE,
         NM_PACIENTE,
         PACIENTE,
         DS_ORI_ATE,
         DS_LOGRADOURO,
         NR_LOGRADOURO,
         DS_COMPLEMENTO,
         NM_BAIRRO,
         NR_CEP,
         NR_TELEFONE,
         NR_CELULAR,
         NR_FAX,
         DS_EMAIL,
         CD_CIDADE,
         NM_CIDADE
