--R_REGISTRA_MOV_AMOSTRA
SELECT * FROM
  (SELECT DISTINCT AMOSTRA.CD_AMOSTRA ,
                   BANCADA.DS_BANCADA ,
                   TO_CHAR(MOV_AMOSTRA.HR_MOV_AMOSTRA, 'dd/mm/yyyy hh24') DH_MOV_AMOSTRA ,
                   MOV_AMOSTRA.CD_USUARIO_MOV_AMOSTRA ,
                   ATENDIME.CD_ATENDIMENTO ,
                   PACIENTE.NM_PACIENTE ,
            (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
                   AMOSTRA.CD_AMOSTRA_PAI
   FROM DBAMV.COLETA_MATERIAL ,
        DBAMV.AMOSTRA ,
        DBAMV.AMOSTRA_EXA_LAB ,
        DBAMV.ITPED_LAB ,
        DBAMV.PED_LAB ,
        DBAMV.ATENDIME ,
        DBAMV.PACIENTE ,
        dbamv.MOV_AMOSTRA ,
        DBAMV.BANCADA ,
        DBAMV.SET_EXA
   WHERE ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
     AND ATENDIME.CD_ATENDIMENTO = PED_LAB.CD_ATENDIMENTO
     AND PED_LAB.CD_PED_LAB = COLETA_MATERIAL.CD_PED_LAB
     AND COLETA_MATERIAL.CD_COLETA_MATERIAL = AMOSTRA.CD_COLETA_MATERIAL
     AND AMOSTRA.CD_AMOSTRA = AMOSTRA_EXA_LAB.CD_AMOSTRA
     AND AMOSTRA_EXA_LAB.CD_ITPED_LAB = ITPED_LAB.CD_ITPED_LAB
     AND PED_LAB.CD_PED_LAB = ITPED_LAB.CD_PED_LAB
     AND SET_EXA.CD_SET_EXA = BANCADA.CD_SET_EXA
     AND MOV_AMOSTRA.CD_COLETA_MATERIAL = COLETA_MATERIAL.CD_COLETA_MATERIAL
     AND MOV_AMOSTRA.CD_AMOSTRA = AMOSTRA.CD_AMOSTRA
     AND MOV_AMOSTRA.CD_BANCADA = BANCADA.CD_BANCADA  
     :CAMPO_COND_RECOLETA )
ORDER BY CD_AMOSTRA,
         DH_MOV_AMOSTRA ASC
