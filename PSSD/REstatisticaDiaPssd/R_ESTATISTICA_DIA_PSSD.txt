--R_ESTATISTICA_DIA_PSSD
SELECT  CD_EXA_LAB
              , NM_EXA_LAB
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '01',1))  DIA1
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '02',1))  DIA2
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '03',1))  DIA3
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '04',1))  DIA4
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '05',1))  DIA5
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '06',1))  DIA6
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '07',1))  DIA7
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '08',1))  DIA8
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '09',1))  DIA9
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '10',1))  DIA10
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '11',1))  DIA11
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '12',1))  DIA12
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '13',1))  DIA13
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '14',1))  DIA14
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '15',1))  DIA15
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '16',1))  DIA16
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '17',1))  DIA17
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '18',1))  DIA18
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '19',1))  DIA19
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '20',1))  DIA20
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '21',1))  DIA21
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '22',1))  DIA22
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '23',1))  DIA23
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '24',1))  DIA24
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '25',1))  DIA25
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '26',1))  DIA26
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '27',1))  DIA27
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '28',1))  DIA28
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '29',1))  DIA29
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '30',1))  DIA30
              ,SUM(DECODE(TO_CHAR(DT_LAUDO,'DD'), '31',1))  DIA31
              ,SUM(1)  TOT
FROM  ( SELECT
         EXA_LAB.CD_EXA_LAB              AS CD_EXA_LAB
        ,EXA_LAB.NM_EXA_LAB              AS NM_EXA_LAB
        ,ITPED_LAB.DT_LAUDO              AS DT_LAUDO
        ,TO_CHAR(ITPED_LAB.DT_LAUDO,'DD')AS DIA
FROM     DBAMV.PED_LAB
              ,DBAMV.ITPED_LAB
              ,DBAMV.EXA_LAB
WHERE  TO_CHAR(ITPED_LAB.DT_LAUDO,'MM/YYYY')  =  :CAMPO_MES
AND     ITPED_LAB.CD_PED_LAB  =  PED_LAB.CD_PED_LAB
AND     ITPED_LAB.CD_EXA_LAB  =  EXA_LAB.CD_EXA_LAB
AND NVL(ITPED_LAB.SN_REALIZADO,'N')  =  'S'
UNION ALL
select
         EXA_LAB.CD_EXA_LAB              AS CD_EXA_LAB
        ,EXA_LAB.NM_EXA_LAB              AS NM_EXA_LAB
        ,ITPED_LAB.DT_LAUDO              AS DT_LAUDO
        ,TO_CHAR(ITPED_LAB.DT_LAUDO,'DD')AS DIA
FROM   DBAMV.PED_LAB
      ,DBAMV.ITPED_LAB
      ,DBAMV.EXA_LAB
      ,DBAMV.GERMES_CULTURA
WHERE  TO_CHAR(ITPED_LAB.DT_LAUDO,'MM/YYYY')  =  :CAMPO_MES
  AND ITPED_LAB.CD_PED_LAB                 = PED_LAB.CD_PED_LAB
  AND GERMES_CULTURA.cd_exa_lab_fatura     = EXA_LAB.CD_EXA_LAB
  AND NVL(ITPED_LAB.SN_REALIZADO,'N')      = 'S'
  AND GERMES_CULTURA.cd_ped_lab            = PED_LAB.CD_PED_LAB
  AND GERMES_CULTURA.cd_exa_lab            = ITPED_LAB.CD_EXA_LAB
  AND GERMES_CULTURA.cd_itped_lab          = ITPED_LAB.cd_itped_lab
  AND GERMES_CULTURA.cd_exa_lab_fatura     is not null
:CAMPO_COND_SET_EXA
)
GROUP BY CD_EXA_LAB
                  ,NM_EXA_LAB
ORDER BY NM_EXA_LAB
