--R_COD_PRO
SELECT  DISTINCT COD_PRO.CD_CONVENIO
               ,CONVENIO.NM_CONVENIO
               ,COD_PRO.CD_PRO_FAT			CD_PRO_FAT
               ,PRO_FAT.DS_PRO_FAT                    		DS_PRO_FAT
               ,COD_PRO.DS_CODIGO_COBRANCA 		DS_CODIGO_COBRANCA
               ,COD_PRO.DS_NOME_COBRANCA    		DS_NOME_COBRANCA
FROM     DBAMV.COD_PRO
              ,DBAMV.CONVENIO
              ,DBAMV.PRO_FAT
	      ,dbamv.empresa_convenio
WHERE Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and CONVENIO.CD_CONVENIO  =  COD_PRO.CD_CONVENIO
AND       PRO_FAT.CD_PRO_FAT  =  COD_PRO.CD_PRO_FAT
and ( Cod_Pro.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA or Cod_Pro.Cd_Multi_Empresa Is Null )
ORDER BY NM_CONVENIO, CD_CONVENIO, :CAMPO_ORDEM
