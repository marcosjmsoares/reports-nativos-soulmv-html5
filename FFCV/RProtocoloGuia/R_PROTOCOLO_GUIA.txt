--R_PROTOCOLO_GUIA
SELECT * FROM (
SELECT PRESTADOR.CD_PRESTADOR
     , PRESTADOR.NM_PRESTADOR
     , CONVENIO.CD_CONVENIO
     , CONVENIO.NM_CONVENIO
     , REG_FAT.CD_REG_FAT
     , REG_FAT.CD_ATENDIMENTO
     , PACIENTE.CD_PACIENTE
     , PACIENTE.NM_PACIENTE
     , PRESTADORES_IT_GUIA.DT_EVENTO
     , GUIA.DT_SOLICITACAO
     , ATI_MED.DS_ATI_MED
     , PRO_FAT.CD_PRO_FAT
     , PRO_FAT.DS_PRO_FAT
     , GUIA.NR_GUIA
     , GUIA.DT_RETORNO_CONV
     , PRESTADORES_IT_GUIA.DT_EMISSAO_PROTOCOLO
     , PRESTADORES_IT_GUIA.CD_GUIA
     , PRESTADORES_IT_GUIA.CD_IT_GUIA
     , ATI_MED.CD_ATI_MED
  FROM DBAMV.GUIA
     , DBAMV.IT_GUIA
     , DBAMV.REG_FAT
     , DBAMV.PACIENTE
     , DBAMV.PRESTADORES_IT_GUIA
     , DBAMV.PRO_FAT
     , DBAMV.ATI_MED
     , DBAMV.PRESTADOR
     , DBAMV.CONVENIO
 WHERE GUIA.CD_GUIA                = IT_GUIA.CD_GUIA
   AND REG_FAT.CD_GUIA             = GUIA.CD_GUIA
   AND REG_FAT.CD_MULTI_EMPRESA    = :CAMPO_CD_MULTI_EMPRESA
   AND PACIENTE.CD_PACIENTE        = GUIA.CD_PACIENTE
   AND GUIA.CD_CONVENIO            = CONVENIO.CD_CONVENIO
   AND IT_GUIA.CD_IT_GUIA          = PRESTADORES_IT_GUIA.CD_IT_GUIA
   AND PRESTADORES_IT_GUIA.CD_GUIA = IT_GUIA.CD_GUIA
   AND PRO_FAT.CD_PRO_FAT          = IT_GUIA.CD_PRO_FAT
   AND ATI_MED.CD_ATI_MED          = PRESTADORES_IT_GUIA.CD_ATI_MED
   AND PRESTADOR.CD_PRESTADOR      = PRESTADORES_IT_GUIA.CD_PRESTADOR
   AND GUIA.DT_RETORNO_CONV        is not null
   AND to_char(GUIA.DT_RETORNO_CONV,'yyyymmdd')  <= to_char(:CAMPO_DT_REFERENCIA,'yyyymmdd')
   AND nvl(GUIA.TP_SITUACAO,'X')   = 'A'
 ORDER BY PRESTADOR.NM_PRESTADOR
        , CONVENIO.NM_CONVENIO
        , PACIENTE.NM_PACIENTE
)
ORDER BY NM_PRESTADOR,CD_PRESTADOR,
NM_CONVENIO,CD_CONVENIO
