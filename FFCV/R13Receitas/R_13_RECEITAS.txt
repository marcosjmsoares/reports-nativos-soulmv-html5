SELECT * FROM (
SELECT  NM_CONVENIO
       ,MES
       ,DS_CON_PLA
       ,SUM(VL_TOTAL_CONTA)  VALOR
FROM
(SELECT  CONVENIO.CD_CONVENIO          CD_CONVENIO
       ,CONVENIO.NM_CONVENIO                  NM_CONVENIO
       ,decode (:CAMPO_DT_REFERENCIA , 'Entrega', to_char(remessa.dt_entrega_da_fatura,'mm'), to_char(fatura.dt_competencia, 'mm')) mes
       ,NVL(ITREG_FAT.VL_TOTAL_CONTA,0)              VL_TOTAL_CONTA
       ,CON_PLA.CD_CON_PLA
       ,CON_PLA.DS_CON_PLA
FROM    DBAMV.FATURA

       ,DBAMV.CONVENIO
       ,DBAMV.CON_PLA

AND   REG_FAT.CD_CONVENIO    = CON_PLA.CD_CONVENIO
AND   REG_FAT.CD_CON_PLA     = CON_PLA.CD_CON_PLA
AND   CONVENIO.TP_CONVENIO <> 'H'

SELECT  CONVENIO.CD_CONVENIO                  CD_CONVENIO
       ,CONVENIO.NM_CONVENIO                  NM_CONVENIO
