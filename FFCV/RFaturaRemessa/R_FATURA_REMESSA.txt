--R_FATURA_REMESSA
SELECT  FATURA.CD_FATURA
               ,FATURA.DS_FATURA
               ,TO_CHAR(FATURA.DT_COMPETENCIA, 'MM/YYYY')   COMPETENCIA
               ,CONVENIO.NM_CONVENIO
               ,REMESSA.CD_REMESSA
               ,REMESSA.DT_ABERTURA
               ,REMESSA.DT_FECHAMENTO
               ,REMESSA.DT_ENTREGA_DA_FATURA
               ,REMESSA.DT_PREVISTA_PARA_PGTO
               ,REMESSA.SN_FECHADA
               ,REMESSA.SN_PAGA
FROM     DBAMV.FATURA
              ,DBAMV.CONVENIO
              ,DBAMV.REMESSA_FATURA   REMESSA
	      ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And FATURA.CD_CONVENIO  =  CONVENIO.CD_CONVENIO
AND        FATURA.CD_FATURA  =  REMESSA.CD_FATURA
AND        FATURA.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
