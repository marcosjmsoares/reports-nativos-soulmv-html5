--R_CONFAT_PORT
SELECT * FROM (
SELECT  CONVENIO.CD_CONVENIO CD_CONVENIO
       ,CONVENIO.NM_CONVENIO NM_CONVENIO
       ,con_pla.cd_con_pla
       ,con_pla.ds_con_pla
       ,regra.cd_regra
       ,itregra.cd_regra itregra
       ,tab_fat.cd_tab_fat
       ,TAB_FAT.DS_TAB_FAT DS_tab_FAT
       ,TAB_FAT.TP_TAB_FAT TP_tab_FAT
       , itregra.cd_gru_pro
       , GRU_PRO.dS_gru_pro
       ,VAL_PORTE.DT_VIGENCIA DT_VIGENCIA
       ,TO_CHAR( VAL_PORTE.DT_VIGENCIA, 'YYYY' )  ANO
       ,VAL_PORTE.VL_PORTE VL_PORTE
       ,VAL_PORTE.CD_POR_ANE CD_POR_ANE
       ,POR_ANE.CD_POR_ANE CD_POR_ANE1
       ,POR_ANE.DS_POR_ANE DS_POR_ANE
  FROM  dbamv.TAB_FAT TAB_FAT
       ,dbamv.VAL_PORTE VAL_PORTE
       ,dbamv.POR_ANE POR_ANE
       ,DBAMV.ITREGRA
       ,DBAMV.REGRA
       ,DBAMV.CON_PLA
       ,DBAMV.CONVENIO
       ,DBAMV.GRU_PRO
       ,Dbamv.Empresa_Con_Pla
Where Empresa_Con_Pla.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And Empresa_Con_Pla.Cd_Convenio = Con_Pla.Cd_Convenio
And Empresa_Con_Pla.Cd_Con_Pla = Con_Pla.Cd_Con_Pla
And VAL_PORTE.CD_TAB_FAT        = TAB_FAT.CD_TAB_FAT
   AND  POR_ANE.CD_POR_ANE             = VAL_PORTE.CD_POR_ANE
   AND  ITREGRA.CD_TAB_FAT             = TAB_FAT.CD_TAB_FAT
   AND  ITREGRA.CD_REGRA               = REGRA.CD_REGRA
   AND  CON_PLA.CD_REGRA               = REGRA.CD_REGRA
   AND  CON_PLA.CD_CONVENIO            = CONVENIO.CD_CONVENIO
   AND  ITREGRA.CD_GRU_PRO             = GRU_PRO.CD_GRU_PRO
ORDER BY CONVENIO.CD_CONVENIO
       ,con_pla.cd_con_pla
       ,gru_pro.cd_gru_pro
       ,tab_fat.cd_tab_fat
        ,VAL_PORTE.DT_VIGENCIA
        ,VAL_PORTE.CD_POR_ANE
)
