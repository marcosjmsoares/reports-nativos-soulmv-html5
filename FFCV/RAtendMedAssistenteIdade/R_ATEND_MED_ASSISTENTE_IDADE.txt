             NM_PRESTADOR,
             COUNT(DISTINCT CD_ATENDIMENTO) INTERNACOES,
             SUM(VL_TOTAL_CONTA) VL_TOTAL_CONTA

(
SELECT PRES.CD_PRESTADOR,
              PRES.NM_PRESTADOR,
              ATEN.CD_ATENDIMENTO,
              NVL(REG_FAT.VL_TOTAL_CONTA,0) VL_TOTAL_CONTA,

             DBAMV.SERVICO,
             DBAMV.ORI_ATE,

And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and TRUNC(ATEN.DT_ATENDIMENTO)  BETWEEN  :CAMPO_DATA_INI  AND  
AND        ATEN.CD_PRESTADOR               =  PRES.CD_PRESTADOR
AND       REG_FAT.CD_ATENDIMENTO(+) = ATEN.CD_ATENDIMENTO
AND       ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA

SELECT PRES.CD_PRESTADOR,

             DBAMV.PRO_FAT,

	     dbamv.empresa_convenio
WHERE  Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and TRUNC(ATEN.DT_ATENDIMENTO)  BETWEEN  :CAMPO_DATA_INI  AND  
AND       ATEN.CD_PRESTADOR                =     PRES.CD_PRESTADOR
AND       ITREG_AMB.CD_ATENDIMENTO =    ATEN.CD_ATENDIMENTO
AND       ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
