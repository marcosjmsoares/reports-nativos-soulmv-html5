--R_ANALISE_PACOTE_SINT
SELECT CONVENIO.CD_CONVENIO
             ,CONVENIO.NM_CONVENIO
             ,PRO_FAT.CD_PRO_FAT                                    CD_PRO_INT
             ,PRO_FAT.DS_PRO_FAT                                    DS_PRO_INT
             ,ATENDIME.CD_ATENDIMENTO
             ,PACIENTE.CD_PACIENTE
             ,PACIENTE.NM_PACIENTE
             ,(REG_FAT.DT_FINAL - REG_FAT.DT_INICIO)   PERMANENCIA
             ,SUM(ITREG_FAT.VL_TOTAL_CONTA)                  VL_PACOTE
             ,REG_FAT.CD_REMESSA                                   CD_REMESSA
             ,REG_FAT.CD_REG_FAT                                     CD_CONTA
FROM    DBAMV.ATENDIME
             ,DBAMV.PACIENTE
             ,DBAMV.REG_FAT
             ,DBAMV.CONVENIO
             ,DBAMV.ITREG_FAT
             ,DBAMV.PRO_FAT
             ,DBAMV.REMESSA_FATURA
             ,DBAMV.FATURA,
	     dbamv.empresa_convenio
WHERE  Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND        ATENDIME.CD_ATENDIMENTO = REG_FAT.CD_ATENDIMENTO
AND        REG_FAT.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND        REG_FAT.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
AND        ITREG_FAT.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
AND        PRO_FAT.SN_PACOTE = 'S'
AND        REG_FAT.CD_REMESSA = REMESSA_FATURA.CD_REMESSA
AND        REMESSA_FATURA.CD_FATURA = FATURA.CD_FATURA
AND        CONVENIO.TP_CONVENIO <> 'H'
and        atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
and  ( select count(it.cd_pro_fat)
            from dbamv.itreg_fat it, dbamv.pro_fat pr,
                 dbamv.reg_fat rg,   dbamv.remessa_fatura rm
           where it.cd_pro_fat = pr.cd_pro_fat
             and rg.cd_reg_fat = it.cd_reg_fat
             and pr.sn_pacote = 'S'
             and rg.cd_atendimento = ATENDIME.CD_ATENDIMENTO
             and rm.cd_remessa     = rg.cd_remessa
             and ( :CAMPO_REMESSA = '%' or rm.cd_remessa = REMESSA_FATURA.CD_REMESSA )
             and ( :CAMPO_PROCED   = '%' or it.cd_pro_fat     = ITREG_FAT.CD_PRO_FAT )                   ) = 1
and
( (ITREG_FAT.CD_CONTA_PACOTE IS NULL AND :CAMPO_PROTOCOLO_PACOTE='N')
  OR
  (ITREG_FAT.CD_CONTA_PACOTE IS NOT NULL  AND
  (select count(*)
            from dbamv.conta_pacote,
                   dbamv.pacote
        where conta_pacote.cd_reg_fat=REG_FAT.CD_REG_FAT
            and pacote.cd_pacote= conta_pacote.cd_pacote
            and pacote.sn_protocolo_de_pacote=:CAMPO_PROTOCOLO_PACOTE) > 0))
GROUP BY CONVENIO.CD_CONVENIO
                   ,CONVENIO.NM_CONVENIO
                   ,PRO_FAT.CD_PRO_FAT
                   ,PRO_FAT.DS_PRO_FAT
                   ,ATENDIME.CD_ATENDIMENTO
                   ,PACIENTE.CD_PACIENTE
                   ,PACIENTE.NM_PACIENTE
                   ,(REG_FAT.DT_FINAL - REG_FAT.DT_INICIO)
                   ,REG_FAT.CD_REMESSA
                   ,REG_FAT.CD_REG_FAT 
UNION ALL
SELECT CONVENIO.CD_CONVENIO
              ,CONVENIO.NM_CONVENIO
              ,PRO_FAT.CD_PRO_FAT                 CD_PRO_INT
              ,PRO_FAT.DS_PRO_FAT                 DS_PRO_INT
              ,ATENDIME.CD_ATENDIMENTO
              ,PACIENTE.CD_PACIENTE
              ,PACIENTE.NM_PACIENTE
              ,1                                                                  PERMANENCIA
              ,SUM(ITREG_AMB.VL_TOTAL_CONTA)    VL_PACOTE
              ,REG_AMB.CD_REMESSA                         CD_REMESSA
             , REG_AMB.CD_REG_AMB                         CD_CONTA
FROM    DBAMV.ATENDIME
             ,DBAMV.PACIENTE
             ,DBAMV.ITREG_AMB
             ,DBAMV.PRO_FAT
             ,DBAMV.CONVENIO
             ,DBAMV.REG_AMB
             ,DBAMV.REMESSA_FATURA
             ,DBAMV.FATURA,
	     dbamv.empresa_convenio
WHERE  Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND        ATENDIME.CD_ATENDIMENTO = ITREG_AMB.CD_ATENDIMENTO
AND        ITREG_AMB.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
AND        PRO_FAT.SN_PACOTE = 'S'
AND        ITREG_AMB.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND        ITREG_AMB.CD_REG_AMB = REG_AMB.CD_REG_AMB
AND        REG_AMB.CD_REMESSA = REMESSA_FATURA.CD_REMESSA
AND        REMESSA_FATURA.CD_FATURA = FATURA.CD_FATURA
AND        REG_AMB.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND        CONVENIO.TP_CONVENIO <> 'H'
and        atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
and  ( select count(it.cd_pro_fat)
            from dbamv.itreg_amb it, dbamv.pro_fat pr,
                 dbamv.reg_amb rg,   dbamv.remessa_fatura rm
           where it.cd_pro_fat = pr.cd_pro_fat
             and rg.cd_reg_amb = it.cd_reg_amb
             and pr.sn_pacote = 'S'
             and it.cd_atendimento = ATENDIME.CD_ATENDIMENTO
             and rm.cd_remessa     = rg.cd_remessa
             and ( :CAMPO_REMESSA = '%' or rm.cd_remessa = REMESSA_FATURA.CD_REMESSA )
             and ( :CAMPO_PROCED   = '%' or it.cd_pro_fat     = ITREG_AMB.CD_PRO_FAT )      ) = 1
and
( (ITREG_AMB.CD_CONTA_PACOTE IS NULL  AND :CAMPO_PROTOCOLO_PACOTE='N')
  OR
  (ITREG_AMB.CD_CONTA_PACOTE IS NOT NULL  AND
  (select count(*)
            from dbamv.conta_pacote,
                   dbamv.pacote
        where conta_pacote.cd_reg_amb=REG_AMB.CD_REG_AMB
            and pacote.cd_pacote= conta_pacote.cd_pacote
            and pacote.sn_protocolo_de_pacote=:CAMPO_PROTOCOLO_PACOTE) > 0))
GROUP BY CONVENIO.CD_CONVENIO
                   ,CONVENIO.NM_CONVENIO
                   ,PRO_FAT.CD_PRO_FAT
                   ,PRO_FAT.DS_PRO_FAT
                   ,ATENDIME.CD_ATENDIMENTO
                   ,PACIENTE.CD_PACIENTE
                   ,PACIENTE.NM_PACIENTE
                   ,1
                   ,REG_AMB.CD_REMESSA
                   ,REG_AMB.CD_REG_AMB
ORDER BY CD_CONVENIO, DS_PRO_INT, CD_PRO_INT, CD_CONTA
