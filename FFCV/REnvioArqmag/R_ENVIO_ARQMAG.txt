--R_ENVIO_ARQMAG_R_CAPA_REMESSA_Q_2
SELECT * FROM (
select log.cd_remessa,
       'Serviços Profissionais' tipo,
       '1' ordem,
       sum(log.vl_sp) total
  from dbamv.log_meio_mag log
 where ( log.cd_remessa       = :CAMPO_CD_REMESSA
         or log.cd_remessa in ( select r.cd_remessa from dbamv.remessa_fatura r
                                 where r.cd_remessa_pai = :CAMPO_CD_REMESSA ) )
 group by log.cd_remessa
union all
select log.cd_remessa,
       'Serviços Diagnósticos' tipo,
       '2' ordem,
       sum(log.vl_sd) total
  from dbamv.log_meio_mag log
 where ( log.cd_remessa       = :CAMPO_CD_REMESSA
         or log.cd_remessa in ( select r.cd_remessa from dbamv.remessa_fatura r
                                 where r.cd_remessa_pai = :CAMPO_CD_REMESSA ) )
 group by log.cd_remessa
union all
select log.cd_remessa,
       'Serviços Hospitalares' tipo,
       '3' ordem,
       sum(log.vl_sh) total
  from dbamv.log_meio_mag log
 where ( log.cd_remessa       = :CAMPO_CD_REMESSA
         or log.cd_remessa in ( select r.cd_remessa from dbamv.remessa_fatura r
                                 where r.cd_remessa_pai = :CAMPO_CD_REMESSA ) )
 group by log.cd_remessa
union all
select log.cd_remessa,
       'Materiais' tipo,
       '4' ordem,
       sum(log.vl_mt) total
  from dbamv.log_meio_mag log
 where ( log.cd_remessa       = :CAMPO_CD_REMESSA
         or log.cd_remessa in ( select r.cd_remessa from dbamv.remessa_fatura r
                                 where r.cd_remessa_pai = :CAMPO_CD_REMESSA ) )
 group by log.cd_remessa
union all
select log.cd_remessa,
       'Medicamentos' tipo,
       '5' ordem,
       sum(log.vl_md) total
  from dbamv.log_meio_mag log
 where ( log.cd_remessa       = :CAMPO_CD_REMESSA
         or log.cd_remessa in ( select r.cd_remessa from dbamv.remessa_fatura r
                                 where r.cd_remessa_pai = :CAMPO_CD_REMESSA ) )
 group by log.cd_remessa
union all
select log.cd_remessa,
       'Órteses e Próteses' tipo,
       '6' ordem,
       sum(log.vl_op) total
  from dbamv.log_meio_mag log
 where ( log.cd_remessa       = :CAMPO_CD_REMESSA
         or log.cd_remessa in ( select r.cd_remessa from dbamv.remessa_fatura r
                                 where r.cd_remessa_pai = :CAMPO_CD_REMESSA ) )
 group by log.cd_remessa
)
ORDER BY CD_REMESSA
