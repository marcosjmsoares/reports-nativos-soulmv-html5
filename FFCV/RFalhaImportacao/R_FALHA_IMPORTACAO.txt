--R_FALHA_IMPORTACAO
SELECT * FROM (
Select distinct log_fa_imp.cd_atendimento,
           log_fa_imp.cd_reg_fat,
           log_fa_imp.cd_mvto_falha,
           log_fa_imp.cd_pro_fat,
           log_fa_imp.cd_item_falha,
           log_fa_imp.dt_importacao,
           log_fa_imp.tp_importacao tp_importacao,
           log_fa_imp.nm_usuario_baixou,
           log_fa_imp.ds_msg_erro,
           log_fa_imp.tp_erro,
           pro_fat.ds_pro_fat ds_tuss
From  dbamv.log_falha_importacao log_fa_imp
          ,dbamv.convenio
          ,dbamv.atendime
	      ,Dbamv.Empresa_Convenio
          ,dbamv.pro_fat
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And trunc(log_fa_imp.dt_importacao) between trunc(:CAMPO_DATA_INICIAL)
and trunc(nvl(:CAMPO_DATA_FINAL, sysdate))
and ATENDIME.CD_ATENDIMENTO = log_fa_imp.CD_ATENDIMENTO
and CONVENIO.CD_CONVENIO = ATENDIME.CD_CONVENIO
and CONVENIO.TP_CONVENIO IN ('C','P')
and pro_fat.cd_pro_fat(+) = log_fa_imp.cd_pro_fat
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
:CAMPO_CF_COND_ORDEM
)
ORDER BY TP_IMPORTACAO, cd_atendimento
