SELECT * FROM (
SELECT DISTINCT ATENDIME.DT_ATENDIMENTO       DT_ATENDIMENTO
              ,PACIENTE.CD_PACIENTE                CD_PACIENTE
              ,PACIENTE.NM_PACIENTE               NM_PACIENTE
              ,PRO_FAT.CD_PRO_FAT                  CD_PRO_FAT
              ,PRO_FAT.DS_PRO_FAT                  DS_PRO_FAT
              ,PRO_FAT.DS_UNIDADE                  DS_UNIDADE
              ,GRU_PRO.CD_GRU_PRO               CD_GRU_PRO
              ,To_Number(Substr(PRESTADOR.CD_PRESTADOR,1,12)) CD_PRESTADOR
              ,PRESTADOR.NM_PRESTADOR     NM_PRESTADOR
              ,REG_FAT.CD_REG_FAT                  NR_CONTA
              ,ITREG_FAT.CD_LANCAMENTO      CD_LANCAMENTO
              ,ITREG_FAT.DT_LANCAMENTO      DT_LANCAMENTO
              ,sum(ITREG_FAT.QT_LANCAMENTO)       QT_LANCAMENTO
              ,SUM( DECODE(Nvl( ITREG_FAT.tp_pagamento, 'P'),'C',0,decode (Nvl(ITREG_FAT.SN_PERTENCE_PACOTE,'S'),'S',0,NVL( ITREG_FAT.VL_TOTAL_CONTA, 0 ))) ) VALOR_PAGAR
              ,'HOSPITALAR'                                  TIPO_CONTA
              ,ATENDIME.CD_ATENDIMENTO     CD_ATENDIMENTO
              ,GRU_FAT.CD_GRU_FAT                CD_GRU_FAT
              ,GRU_FAT.DS_GRU_FAT                DS_GRU_FAT
              ,REG_FAT.CD_CONVENIO            CD_CONVENIO
              ,REG_FAT.CD_CON_PLA              CD_CON_PLA
              ,NVL(ITREG_FAT.VL_ACRESCIMO,0)                         VL_ACRESCIMO
              ,NVL(ITREG_FAT.VL_DESCONTO,0)                          VL_DESCONTO
              ,NVL(ITREG_FAT.VL_PERCENTUAL_MULTIPLA,0)  VL_PERCENTUAL_MULTIPLA
              ,REG_FAT.CD_REGRA                  CD_REGRA
              ,REMESSA.CD_AGRUPAMENTO  CD_AGRUPA
              ,FATURA.DT_COMPETENCIA       DT_COMPETENCIA
             ,ATENDIME.TP_ATENDIMENTO
              ,EMPRESA_CON_PLA.CD_REGRA    CD_REGRA_CONV
              ,ATENDIME.CD_CONVENIO CD_CONVENIO_A
              ,ATENDIME.CD_CON_PLA   CD_CON_PLA_A
              ,ATENDIME.NR_CARTEIRA
FROM   DBAMV.ATENDIME                  ATENDIME
            ,DBAMV.PACIENTE                   PACIENTE
            ,DBAMV.REG_FAT                     REG_FAT
            ,DBAMV.ITREG_FAT                  ITREG_FAT
            ,DBAMV.PRO_FAT                     PRO_FAT
            ,DBAMV.GRU_PRO                    GRU_PRO
            ,DBAMV.PRESTADOR               PRESTADOR
            ,DBAMV.REMESSA_FATURA    REMESSA
            ,DBAMV.FATURA                       FATURA
            ,DBAMV.GRU_FAT           GRU_FAT
            ,DBAMV.EMPRESA_CON_PLA  EMPRESA_CON_PLA
WHERE  REG_FAT.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
AND REG_fat.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND        ITREG_FAT.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
AND        GRU_FAT.CD_GRU_FAT = ITREG_FAT.CD_GRU_FAT
AND        REG_FAT.CD_REMESSA = REMESSA.CD_REMESSA
AND        REMESSA.CD_FATURA = FATURA.CD_FATURA
AND        REG_FAT.SN_FECHADA = 'S'
AND        ITREG_FAT.SN_PERTENCE_PACOTE = 'N'
AND        NVL(ITREG_FAT.TP_PAGAMENTO,'X') <> 'C'
AND        REMESSA.CD_REMESSA = :CAMPO_CD_REMESSA
AND        ATENDIME.CD_ATENDIMENTO = REG_FAT.CD_ATENDIMENTO
AND        ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND        PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
AND        ITREG_FAT.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
AND        EMPRESA_CON_PLA.CD_CONVENIO = ATENDIME.CD_CONVENIO
AND        EMPRESA_CON_PLA.CD_CON_PLA    = REG_FAT.CD_CON_PLA
AND        EMPRESA_CON_PLA.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND ATENDIME.CD_MULTI_EMPRESA =  :CAMPO_CD_MULTI_EMPRESA
group by  ATENDIME.DT_ATENDIMENTO
              ,PACIENTE.CD_PACIENTE
              ,PACIENTE.NM_PACIENTE
              ,PRO_FAT.CD_PRO_FAT
              ,PRO_FAT.DS_PRO_FAT
              ,PRO_FAT.DS_UNIDADE
              ,GRU_PRO.CD_GRU_PRO
              ,To_Number(Substr(PRESTADOR.CD_PRESTADOR,1,12))
              ,PRESTADOR.NM_PRESTADOR
              ,REG_FAT.CD_REG_FAT
              ,ITREG_FAT.CD_LANCAMENTO
              ,ITREG_FAT.DT_LANCAMENTO
              ,'HOSPITALAR'
              ,ATENDIME.CD_ATENDIMENTO
              ,GRU_FAT.CD_GRU_FAT
              ,GRU_FAT.DS_GRU_FAT
              ,REG_FAT.CD_CONVENIO
              ,REG_FAT.CD_CON_PLA
              ,NVL(ITREG_FAT.VL_ACRESCIMO,0)
              ,NVL(ITREG_FAT.VL_DESCONTO,0)
              ,NVL(ITREG_FAT.VL_PERCENTUAL_MULTIPLA,0)
              ,REG_FAT.CD_REGRA
              ,REMESSA.CD_AGRUPAMENTO
              ,FATURA.DT_COMPETENCIA
              ,ATENDIME.TP_ATENDIMENTO
              ,EMPRESA_CON_PLA.CD_REGRA
              ,ATENDIME.CD_CONVENIO
              ,ATENDIME.CD_CON_PLA
              ,ATENDIME.NR_CARTEIRA
UNION ALL
SELECT DISTINCT ATENDIME.DT_ATENDIMENTO     DT_ATENDIMENTO
              ,PACIENTE.CD_PACIENTE              CD_PACIENTE
              ,PACIENTE.NM_PACIENTE              NM_PACIENTE
              ,PRO_FAT.CD_PRO_FAT                 CD_PRO_FAT
              ,PRO_FAT.DS_PRO_FAT                 DS_PRO_FAT
              ,PRO_FAT.DS_UNIDADE                 DS_UNIDADE
              ,GRU_PRO.CD_GRU_PRO              CD_GRU_PRO
              ,To_Number(Substr(PRESTADOR.CD_PRESTADOR,1,12)) CD_PRESTADOR
              ,PRESTADOR.NM_PRESTADOR     NM_PRESTADOR
              ,REG_AMB.CD_REG_AMB                NR_CONTA
              ,ITREG_AMB.CD_LANCAMENTO      CD_LANCAMENTO
              ,REG_AMB.DT_LANCAMENTO         DT_LANCAMENTO
              ,sum(ITREG_AMB.QT_LANCAMENTO)  QT_LANCAMENTO
              ,SUM( DECODE(Nvl( itreg_amb.tp_pagamento, 'P'),'C',0,decode (Nvl(ITREG_AMB.SN_PERTENCE_PACOTE,'S'),'S',0,NVL( ITREG_AMB.VL_TOTAL_CONTA, 0 ))) ) VALOR_PAGAR
              ,'AMBULATORIAL'                             TIPO_CONTA
              ,ATENDIME.CD_ATENDIMENTO     CD_ATENDIMENTO
              ,GRU_FAT.CD_GRU_FAT CD_GRU_FAT
              ,GRU_FAT.DS_GRU_FAT DS_GRU_FAT
              ,ITREG_AMB.CD_CONVENIO CD_CONVENIO
              ,NVL(ITREG_AMB.CD_CON_PLA,ATENDIME.CD_CON_PLA)    CD_CON_PLA
              ,NVL(ITREG_AMB.VL_ACRESCIMO,0)                          VL_ACRESCIMO
              ,NVL(ITREG_AMB.VL_DESCONTO,0)                           VL_DESCONTO
              ,NVL(ITREG_AMB.VL_PERCENTUAL_MULTIPLA,0)   VL_PERCENTUAL_MULTIPLA
              ,REG_AMB.CD_REGRA                 CD_REGRA
              ,REMESSA.CD_AGRUPAMENTO        CD_AGRUPA
              ,FATURA.DT_COMPETENCIA       DT_COMPETENCIA
              ,ATENDIME.TP_ATENDIMENTO
              ,EMPRESA_CON_PLA.CD_REGRA    CD_REGRA_CONV
              ,ATENDIME.CD_CONVENIO  CD_CONVENIO_A
              ,ATENDIME.CD_CON_PLA    CD_CON_PLA_A
              ,ATENDIME.NR_CARTEIRA
FROM   DBAMV.ATENDIME                   ATENDIME
            ,DBAMV.PACIENTE                    PACIENTE
            ,DBAMV.ITREG_AMB                  ITREG_AMB
            ,DBAMV.REG_AMB                     REG_AMB
            ,DBAMV.PRO_FAT                      PRO_FAT
            ,DBAMV.GRU_PRO                     GRU_PRO
            ,DBAMV.PRESTADOR                 PRESTADOR
            ,DBAMV.REMESSA_FATURA      REMESSA
            ,DBAMV.FATURA                          FATURA
            ,DBAMV.GRU_FAT           GRU_FAT
            ,DBAMV.EMPRESA_CON_PLA     EMPRESA_CON_PLA
WHERE  ITREG_AMB.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
AND REG_AMB.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND        ITREG_AMB.CD_REG_AMB = REG_AMB.CD_REG_AMB
AND        GRU_FAT.CD_GRU_FAT = ITREG_AMB.CD_GRU_FAT
AND        REG_AMB.CD_REMESSA = REMESSA.CD_REMESSA
AND        REMESSA.CD_FATURA = FATURA.CD_FATURA
AND        ITREG_AMB.SN_FECHADA = 'S'
AND        ITREG_AMB.SN_PERTENCE_PACOTE = 'N'
AND        NVL(ITREG_AMB.TP_PAGAMENTO,'X') <> 'C'
AND        REMESSA.CD_REMESSA = :CAMPO_CD_REMESSA
AND        ATENDIME.CD_ATENDIMENTO = ITREG_AMB.CD_ATENDIMENTO
AND        ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND        PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
AND        ITREG_AMB.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
AND        EMPRESA_CON_PLA.CD_CONVENIO = ATENDIME.CD_CONVENIO
AND        EMPRESA_CON_PLA.CD_CON_PLA   = NVL(ITREG_AMB.CD_CON_PLA,ATENDIME.CD_CON_PLA)
AND        EMPRESA_CON_PLA.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND ATENDIME.CD_MULTI_EMPRESA =  :CAMPO_CD_MULTI_EMPRESA
group BY ATENDIME.DT_ATENDIMENTO
              ,PACIENTE.CD_PACIENTE
              ,PACIENTE.NM_PACIENTE
              ,PRO_FAT.CD_PRO_FAT
              ,PRO_FAT.DS_PRO_FAT
              ,PRO_FAT.DS_UNIDADE
              ,GRU_PRO.CD_GRU_PRO
              ,To_Number(Substr(PRESTADOR.CD_PRESTADOR,1,12))
              ,PRESTADOR.NM_PRESTADOR
              ,REG_AMB.CD_REG_AMB
              ,ITREG_AMB.CD_LANCAMENTO
              ,REG_AMB.DT_LANCAMENTO
              ,'AMBULATORIAL'
              ,ATENDIME.CD_ATENDIMENTO
              ,GRU_FAT.CD_GRU_FAT
              ,GRU_FAT.DS_GRU_FAT
              ,ITREG_AMB.CD_CONVENIO
              ,NVL(ITREG_AMB.CD_CON_PLA,ATENDIME.CD_CON_PLA)
              ,NVL(ITREG_AMB.VL_ACRESCIMO,0)
              ,NVL(ITREG_AMB.VL_DESCONTO,0)
              ,NVL(ITREG_AMB.VL_PERCENTUAL_MULTIPLA,0)
              ,REG_AMB.CD_REGRA
              ,REMESSA.CD_AGRUPAMENTO
              ,FATURA.DT_COMPETENCIA
              ,ATENDIME.TP_ATENDIMENTO
              ,EMPRESA_CON_PLA.CD_REGRA
              ,ATENDIME.CD_CONVENIO
              ,ATENDIME.CD_CON_PLA
              ,ATENDIME.NR_CARTEIRA
)
