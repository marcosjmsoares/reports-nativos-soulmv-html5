select * from (
SELECT 1,G.DS_GRU_FAT  , I.vl_gru_fat
  FROM DBAMV.GRU_FAT G , DBAMV.ITNOTA_FISCAL I
 WHERE  G.CD_GRU_FAT  = I.CD_GRU_FAT
   AND  i.CD_NOTA_FISCAL        = :CAMPO_CD_NOTA_FISCAL
union all
SELECT 2 , ad.ds_desc_acres  , nf.vl_desconto_nota
FROM dbamv.con_rec cr  ,
     dbamv.descontos_acrescimos ad ,
     dbamv.nota_fiscal nf ,
     dbamv.convenio c
WHERE nf.cd_nota_fiscal = cr.cd_nota_fiscal
 AND ad.cd_desc_acres = cr.cd_desconto
 AND c.cd_convenio  = nf.cd_convenio
 AND  Nvl(nf.vl_desconto_nota,0) > 0
 AND c.tp_convenio = 'P'
 AND nf.cd_nota_fiscal   = :CAMPO_CD_NOTA_FISCAL
 UNION ALL
 SELECT 2 , nf.DS_VL_DESCONTO  , nf.vl_desconto_nota
FROM dbamv.nota_fiscal nf ,
     dbamv.convenio c
WHERE c.cd_convenio  = nf.cd_convenio
 AND c.tp_convenio = 'C'
 AND  Nvl(nf.vl_desconto_nota,0) > 0
 AND nf.cd_nota_fiscal   = :CAMPO_CD_NOTA_FISCAL
UNION ALL
 SELECT 2 , nf.DS_VL_ACRESCIMO  , nf.vl_acrescimo_nota
FROM dbamv.nota_fiscal nf ,
     dbamv.convenio c
WHERE c.cd_convenio  = nf.cd_convenio
 AND c.tp_convenio = 'C'
 AND  Nvl(nf.vl_acrescimo_nota,0) > 0
 AND nf.cd_nota_fiscal   = :CAMPO_CD_NOTA_FISCAL
   )
 ORDER BY 1,2
