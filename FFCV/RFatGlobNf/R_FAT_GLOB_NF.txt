                                                 )
               AND convenio.tp_convenio not in ( 'H', 'A' )
               AND itreg_amb.cd_atendimento = atendime.cd_atendimento
               AND atendime.cd_servico = servico.cd_servico(+)
               and atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
          AND (itreg_amb.cd_reg_amb, itreg_amb.cd_lancamento) IN 
		                  (SELECT cd_reg_amb, cd_lancamento_amb
			                 FROM dbamv.itfat_nota_fiscal
							WHERE cd_reg_fat IS null
							  AND cd_nota_fiscal IN (SELECT cd_nota_fiscal
							                           FROM dbamv.nota_fiscal
							  						  WHERE 1=1 :CAMPO_CF_COND_DATA ))
          GROUP BY reg_amb.cd_convenio
                  ,itreg_amb.cd_gru_fat
                  ,gru_fat.tp_gru_fat
          UNION ALL
          SELECT   reg_fat.cd_convenio cd_convenio
                  ,itreg_fat.cd_gru_fat cd_gru_fat

                  ,sum( nvl(itlan_med.vl_liquido, itreg_fat.vl_total_conta)) vl_tot_internacao
                  ,0 vl_tot_ambulatorial
                  ,sum( nvl(itlan_med.vl_liquido, itreg_fat.vl_total_conta)) vl_total_conta
              FROM dbamv.itreg_fat itreg_fat
                  ,dbamv.reg_fat reg_fat

                  ,dbamv.atendime atendime
                  ,dbamv.itlan_med
             WHERE  itreg_fat.cd_reg_fat = reg_fat.cd_reg_fat
               AND itreg_fat.cd_gru_fat = gru_fat.cd_gru_fat
               AND empresa_convenio.cd_convenio = convenio.cd_convenio
               AND empresa_convenio.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA

               AND remessa_fatura.cd_fatura = fatura.cd_fatura

               AND pro_fat.sn_pacote = 'N'
               and itlan_med.cd_reg_fat(+) = itreg_fat.cd_reg_fat
               and itlan_med.cd_lancamento(+) = itreg_fat.cd_lancamento
               and nvl( itlan_med.tp_pagamento, nvl( itreg_fat.tp_pagamento, 'P' ) ) <> 'C'
               AND NVL (reg_fat.sn_diagno, 'N') = 'N'
               AND reg_fat.cd_convenio = convenio.cd_convenio
               AND reg_fat.sn_fechada = DECODE (:CAMPO_CONTA
                                               ,'T', reg_fat.sn_fechada
                                               ,:CAMPO_CONTA
                                               )
               AND convenio.tp_convenio not in ( 'H', 'A' )
               AND reg_fat.cd_atendimento = atendime.cd_atendimento
               AND atendime.cd_servico = servico.cd_servico(+)
               and atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
          AND (itreg_fat.cd_reg_fat, itreg_fat.cd_lancamento) IN 
                      (SELECT cd_reg_fat, cd_lancamento_fat
			             FROM dbamv.itfat_nota_fiscal
						 WHERE cd_reg_amb IS null
					       AND cd_nota_fiscal IN (SELECT cd_nota_fiscal
						                            FROM dbamv.nota_fiscal
							   					   WHERE 1=1 :CAMPO_CF_COND_DATA ))

                  ,itreg_fat.cd_gru_fat
                  ,gru_fat.tp_gru_fat

          SELECT   reg_amb.cd_convenio cd_convenio
                  ,itreg_amb.cd_gru_fat cd_gru_fat
                  ,gru_fat.tp_gru_fat tp_gru_fat
                  ,0 vl_tot_internacao
                  ,SUM (itreg_amb.vl_total_conta) vl_tot_ambulatorial
                  ,SUM (itreg_amb.vl_total_conta) vl_total_conta
              FROM dbamv.itreg_amb itreg_amb
                  ,dbamv.reg_amb reg_amb
                  ,dbamv.gru_fat gru_fat
                  ,dbamv.remessa_fatura remessa_fatura
                  ,dbamv.fatura fatura
                  ,dbamv.pro_fat pro_fat
                  ,dbamv.convenio convenio
                  ,dbamv.empresa_convenio
                  ,dbamv.servico servico
                  ,dbamv.atendime atendime
             WHERE  itreg_amb.cd_reg_amb = reg_amb.cd_reg_amb
               AND itreg_amb.cd_gru_fat = gru_fat.cd_gru_fat
               AND reg_amb.cd_remessa = remessa_fatura.cd_remessa

               AND pro_fat.sn_pacote = 'N'
               AND NVL (itreg_amb.tp_pagamento, 'X') <> 'C'
               AND NVL (reg_amb.sn_diagno, 'N') = 'N'
               AND reg_amb.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
               AND itreg_amb.cd_convenio = convenio.cd_convenio
               AND itreg_amb.sn_fechada = DECODE (:CAMPO_CONTA
                                                 ,'T', itreg_amb.sn_fechada
                                                 ,:CAMPO_CONTA
                                                 )
               AND convenio.tp_convenio not in ( 'H', 'A' )
               AND itreg_amb.cd_atendimento = atendime.cd_atendimento
               AND atendime.cd_servico = servico.cd_servico(+)
               and atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
          AND (itreg_amb.cd_reg_amb, itreg_amb.cd_lancamento) IN

			                 FROM dbamv.itfat_nota_fiscal
							WHERE cd_reg_fat IS null
							  AND cd_nota_fiscal IN (SELECT cd_nota_fiscal
							                           FROM dbamv.nota_fiscal

          GROUP BY reg_amb.cd_convenio
                  ,itreg_amb.cd_gru_fat

         ) convenio
