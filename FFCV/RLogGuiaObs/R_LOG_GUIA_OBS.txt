--R_LOG_GUIA_OBS
SELECT g.nr_guia, g.cd_atendimento, g.cd_aviso_cirurgia, g.cd_res_lei,
       Decode(g.tp_guia, 'I', 'Internação', 'C', 'Consulta', 'R', 'Prorrogação',
                         'P', 'Procedimento', 'O', 'OPME', 'Q', 'Quimioterapia',
                         'D', 'Radioterapia', 'S', 'SADT', 'M', 'Medicamento', 'I', 'Materiais') tp_guia,
       g.tp_situacao,
       l.cd_log_guia, l.dt_movimento, l.cd_usu_movimento,
       Decode(l.tp_movimento, 'I', 'Inclusão', 'U', 'Alteração', 'D', 'Exclusão' ) tp_movimento,
       l.ds_observacao_ant,
       l.ds_observacao_atu
FROM dbamv.guia g, dbamv.log_guia_obs l
WHERE g.cd_guia = :CAMPO_CD_GUIA
   AND l.cd_guia = g.cd_guia
ORDER BY g.cd_guia, l.cd_log_guia
