--R_EXEC_CONPLA
SELECT * FROM (
SELECT  TAB_CONVENIO.CD_CONVENIO              CD_CONVENIO
               ,TAB_CONVENIO.CD_CON_PLA                CD_CON_PLA
               ,CONVENIO.NM_CONVENIO                      NM_CONVENIO
               ,CON_PLA.DS_CON_PLA                           DS_CON_PLA
               ,TAB_CONVENIO.CD_PRO_FAT                CD_PRO_FAT
               ,PRO_FAT.DS_PRO_FAT                           DS_PRO_FAT
               ,TAB_CONVENIO.DT_VIGENCIA                DT_VIGENCIA
               ,TAB_CONVENIO.SN_USAR_INDICE         SN_USAR_INDICE
               ,TAB_CONVENIO.VL_TAB_CONVENIO      VL_TAB_CONVENIO
               ,TAB_CONVENIO.CD_PRESTADOR         CD_PRESTADOR
               ,TAB_CONVENIO.CD_SETOR                   CD_SETOR
FROM     dbamv.TAB_CONVENIO   TAB_CONVENIO
              ,dbamv.CON_PLA CON_PLA
              ,dbamv.CONVENIO CONVENIO
              ,dbamv.PRO_FAT PRO_FAT
	      ,Dbamv.Empresa_Con_Pla
Where Empresa_Con_Pla.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And Empresa_Con_Pla.Cd_Convenio = Con_Pla.Cd_Convenio
And Empresa_Con_Pla.Cd_Con_Pla = Con_Pla.Cd_Con_Pla
And CON_PLA.CD_CONVENIO    =  TAB_CONVENIO.CD_CONVENIO
    AND   CON_PLA.CD_CON_PLA      =   TAB_CONVENIO.CD_CON_PLA
    AND   CONVENIO.CD_CONVENIO  =  CON_PLA.CD_CONVENIO
    AND   PRO_FAT.CD_PRO_FAT      =   TAB_CONVENIO.CD_PRO_FAT
    AND   CONVENIO.CD_CONVENIO  like  :CAMPO_CD_CONVENIO
    AND   CON_PLA.CD_CON_PLA       like  :CAMPO_CD_CON_PLA
    and     tab_convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
UNION ALL
SELECT  TAB_CONVENIO.CD_CONVENIO              CD_CONVENIO
               ,TAB_CONVENIO.CD_CON_PLA                CD_CON_PLA
               ,CONVENIO.NM_CONVENIO                      NM_CONVENIO
               ,'TODOS OS PLANOS'    DS_CON_PLA
               ,TAB_CONVENIO.CD_PRO_FAT                CD_PRO_FAT
               ,PRO_FAT.DS_PRO_FAT                           DS_PRO_FAT
               ,TAB_CONVENIO.DT_VIGENCIA                DT_VIGENCIA
               ,TAB_CONVENIO.SN_USAR_INDICE         SN_USAR_INDICE
               ,TAB_CONVENIO.VL_TAB_CONVENIO      VL_TAB_CONVENIO
               ,TAB_CONVENIO.CD_PRESTADOR         CD_PRESTADOR
               ,TAB_CONVENIO.CD_SETOR                   CD_SETOR
FROM     dbamv.TAB_CONVENIO   TAB_CONVENIO
              ,dbamv.CONVENIO CONVENIO
              ,dbamv.PRO_FAT PRO_FAT
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And CONVENIO.CD_CONVENIO    =  TAB_CONVENIO.CD_CONVENIO
    AND   PRO_FAT.CD_PRO_FAT      =   TAB_CONVENIO.CD_PRO_FAT
    AND   CONVENIO.CD_CONVENIO  like  :CAMPO_CD_CONVENIO
    AND   TAB_CONVENIO.CD_CON_PLA IS NULL
    and     tab_convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
ORDER BY   6,7
)
ORDER BY NM_CONVENIO,CD_CONVENIO,
DS_CON_PLA,CD_CON_PLA,DS_PRO_FAT
