--R_RECIBO_PAC_PADRAO
SELECT
             ATENDIME.CD_ATENDIMENTO
            ,ATENDIME.DT_ATENDIMENTO
            ,ATENDIME.DT_ALTA
            ,dbamv.pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(PACIENTE.CD_PACIENTE, 1) NM_PACIENTE
            ,PACIENTE.NR_DOCUMENTO
            ,LEITO.DS_LEITO
            ,REG_FAT.CD_REG_FAT           NU_CONTA
            ,NOTA_FISCAL.NR_ID_NOTA_FISCAL
            ,NOTA_FISCAL.DT_EMISSAO
            ,NOTA_FISCAL.NM_CLIENTE  NM_CLIENTE_NA_NOTA
            ,NOTA_FISCAL.NR_CGC_CPF  NR_CGC_CPF_NA_NOTA
            ,NOTA_FISCAL.NM_BAIRRO   NM_BAIRRO_NA_NOTA
            ,NOTA_FISCAL.DS_ENDERECO DS_ENDERECO_NA_NOTA
            ,NOTA_FISCAL.NM_CIDADE   NM_CIDADE_NA_NOTA
            ,NOTA_FISCAL.NM_UF       NM_UF_NA_NOTA
            ,NOTA_FISCAL.NR_FONE     NR_FONE_NA_NOTA
            ,NOTA_FISCAL.NR_CEP      NR_CEP_NA_NOTA
            ,NOTA_FISCAL.NR_INSCRICAO_ESTADUAL NR_INSCRICAO_ESTADUAL_NA_NOTA
            ,NOTA_FISCAL.DS_OBSERVACOES
            ,NOTA_FISCAL.DS_VL_ACRESCIMO
            ,NOTA_FISCAL.VL_ACRESCIMO_NOTA
            ,NOTA_FISCAL.DS_VL_DESCONTO
            ,NOTA_FISCAL.VL_DESCONTO_NOTA
            ,nvl(reccon_rec.vl_recebido,0)  VL_TOTAL_CONTA
            ,nvl(responsavel.nm_responsavel,dbamv.pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(PACIENTE.CD_PACIENTE, 1)) nm_responsavel
            ,PACIENTE.CD_PACIENTE
FROM DBAMV.ATENDIME
          ,DBAMV.PACIENTE
          ,DBAMV.LEITO
          ,DBAMV.REG_FAT
          ,DBAMV.NOTA_FISCAL
          ,dbamv.con_rec
          ,dbamv.itcon_rec
          ,dbamv.reccon_rec
          ,dbamv.responsavel
WHERE ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND       ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
AND       ATENDIME.CD_ATENDIMENTO = REG_FAT.CD_ATENDIMENTO
AND       ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND       REG_FAT.CD_REG_FAT = NOTA_FISCAL.CD_REG_FAT(+)
and       con_rec.cd_con_rec = itcon_rec.cd_con_rec
and       itcon_rec.cd_itcon_rec = reccon_rec.cd_itcon_rec
and       con_rec.cd_atendimento = atendime.cd_atendimento
and       reccon_rec.cd_reccon_rec = :CAMPO_CD_RECCON_REC
and       reccon_rec.cd_resposavel = responsavel.cd_resposavel(+)
AND       ATENDIME.CD_ATENDIMENTO = :CAMPO_CD_ATENDIMENTO
AND       REG_FAT.CD_REG_FAT = :CAMPO_CD_CONTA
AND   NVL( Nota_Fiscal.cd_multi_empresa, :CAMPO_CD_MULTI_EMPRESA ) = :CAMPO_CD_MULTI_EMPRESA
UNION
SELECT ATENDIME.CD_ATENDIMENTO
             ,ATENDIME.DT_ATENDIMENTO
             ,ATENDIME.DT_ATENDIMENTO  DT_ALTA
             ,dbamv.pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(PACIENTE.CD_PACIENTE, 1) NM_PACIENTE
             ,PACIENTE.NR_DOCUMENTO
             ,LEITO.DS_LEITO
             ,ITREG_AMB.CD_REG_AMB                    NU_CONTA
             ,NOTA_FISCAL.NR_ID_NOTA_FISCAL
             ,NOTA_FISCAL.DT_EMISSAO
            ,NOTA_FISCAL.NM_CLIENTE  NM_CLIENTE_NA_NOTA
            ,NOTA_FISCAL.NR_CGC_CPF  NR_CGC_CPF_NA_NOTA
            ,NOTA_FISCAL.NM_BAIRRO   NM_BAIRRO_NA_NOTA
            ,NOTA_FISCAL.DS_ENDERECO DS_ENDERECO_NA_NOTA
            ,NOTA_FISCAL.NM_CIDADE   NM_CIDADE_NA_NOTA
            ,NOTA_FISCAL.NM_UF       NM_UF_NA_NOTA
            ,NOTA_FISCAL.NR_FONE     NR_FONE_NA_NOTA
            ,NOTA_FISCAL.NR_CEP      NR_CEP_NA_NOTA
            ,NOTA_FISCAL.NR_INSCRICAO_ESTADUAL NR_INSCRICAO_ESTADUAL_NA_NOTA
            ,NOTA_FISCAL.DS_OBSERVACOES
            ,NOTA_FISCAL.DS_VL_ACRESCIMO
            ,NOTA_FISCAL.VL_ACRESCIMO_NOTA
            ,NOTA_FISCAL.DS_VL_DESCONTO
            ,NOTA_FISCAL.VL_DESCONTO_NOTA
            ,nvl(reccon_rec.vl_recebido,0)  VL_TOTAL_CONTA
            ,nvl(responsavel.nm_responsavel,dbamv.pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(PACIENTE.CD_PACIENTE, 1)) nm_responsavel
            ,PACIENTE.CD_PACIENTE
FROM DBAMV.ATENDIME
          ,DBAMV.PACIENTE
          ,DBAMV.LEITO
          ,DBAMV.ITREG_AMB
          ,DBAMV.NOTA_FISCAL
          ,dbamv.con_rec
          ,dbamv.itcon_rec
          ,dbamv.reccon_rec
          ,dbamv.responsavel
WHERE ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND       ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
AND       ATENDIME.CD_ATENDIMENTO = ITREG_AMB.CD_ATENDIMENTO
AND      ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND       ATENDIME.CD_ATENDIMENTO = NOTA_FISCAL.CD_ATENDIMENTO(+)
and       con_rec.cd_con_rec = itcon_rec.cd_con_rec
and       itcon_rec.cd_itcon_rec = reccon_rec.cd_itcon_rec
and       con_rec.cd_atendimento = atendime.cd_atendimento
and       reccon_rec.cd_reccon_rec = :CAMPO_CD_RECCON_REC
and       reccon_rec.cd_resposavel = responsavel.cd_resposavel(+)
AND       ITREG_AMB.SN_PERTENCE_PACOTE = 'N'
AND        NVL(ITREG_AMB.TP_PAGAMENTO,'X') <> 'C'
AND       ATENDIME.CD_ATENDIMENTO =  :CAMPO_CD_ATENDIMENTO
AND       ITREG_AMB.CD_REG_AMB =  :CAMPO_CD_CONTA
AND   NVL( Nota_Fiscal.cd_multi_empresa, :CAMPO_CD_MULTI_EMPRESA ) = :CAMPO_CD_MULTI_EMPRESA
