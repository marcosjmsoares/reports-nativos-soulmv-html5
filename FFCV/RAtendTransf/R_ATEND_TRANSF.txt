--R_ATEND_TRANSF
select log_transferencia.cd_log_transf,
       log_transferencia.cd_convenio,
       convenio.nm_convenio,
       log_transferencia.cd_prestador,
       log_transferencia.cd_prestador cd_prestador1,
       prestador.nm_prestador,
       log_transferencia.dt_transf,
       log_transferencia.cd_atendimento,
       log_transferencia.cd_paciente,
       paciente.nm_paciente,
       log_transferencia.cd_pro_fat,
       pro_fat.ds_pro_fat,
       log_transferencia.cd_ori_ate,
       log_transferencia.cd_motivo,
       log_transferencia.ds_observacao,
       log_transferencia.tp_transf,
       log_transferencia.cd_usuario_trasf,
       log_transferencia.cd_atendimento_origem,
       Decode(log_transferencia.tp_transf,'AH','Amb > Hosp','Hosp > Amb') tipo_transf
from dbamv.log_transferencia,
     dbamv.convenio,
     dbamv.prestador,
     dbamv.paciente,
     dbamv.pro_fat
where trunc(dt_transf) between to_date(:CAMPO_DT_INICIO,'dd/MM/yyyy') and to_date(:CAMPO_DT_FIM,'dd/MM/yyyy')
  AND log_transferencia.cd_convenio = convenio.cd_convenio(+)
  AND log_transferencia.cd_prestador = prestador.cd_prestador(+)
  AND log_transferencia.cd_paciente = paciente.cd_paciente(+)
  AND log_transferencia.cd_pro_fat = pro_fat.cd_pro_fat(+)
order by log_transferencia.dt_transf, log_transferencia.cd_convenio
