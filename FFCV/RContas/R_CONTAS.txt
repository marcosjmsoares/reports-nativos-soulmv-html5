--R_CONTAS_R_ITENS_Q_ITENS
SELECT * FROM (
select cd_REG_FAT,
           cd_lancamento,
           dt_lancamento,
           cd_gru_pro,
           ds_gru_pro,
           cd_pro_fat,
           ds_pro_fat,
           cd_prestador,
           nm_prestador,
           qt_lancamento,
           vl_unitario
from
   ( SELECT ITREG_FAT.cd_REG_FAT,
                    ITREG_FAT.cd_lancamento,
	    ITREG_FAT.dt_lancamento,
                    gru_pro.cd_gru_pro,
                    gru_pro.ds_gru_pro,
                    ITREG_FAT.cd_pro_fat,
                    PRO_FAT.ds_pro_fat,
                    ITREG_FAT.cd_prestador,
                    PRESTADOR.nm_prestador,
	    ITREG_FAT.qt_lancamento,
                    ITREG_FAT.vl_unitario
      FROM   DBAMV.ITREG_FAT,
                   DBAMV.REG_FAT,
                   DBAMV.PRO_FAT,
	   DBAMV.PRESTADOR,
                   DBAMV.GRU_PRO
    WHERE  ITREG_FAT.cd_pro_fat = PRO_FAT.CD_PRO_FAT
      AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
      and     ITREG_FAT.cd_prestador = PRESTADOR.cd_prestador (+)
      and     nvl(itreg_fat.tp_pagamento, 'S') <> 'C'
      and     itreg_fat.sn_pertence_pacote = 'N'
      and      gru_pro.cd_gru_pro = pro_fat.cd_gru_pro
   and itreg_fat.cd_setor in (select distinct cd_setor
                                       from dbamv.Usuario_unid_int
                                       where cd_id_usuario = user)
      and      itreg_fat.cd_reg_fat = reg_fat.cd_reg_fat
      and      reg_fat.CD_ATENDIMENTO = :CAMPO_CD_ATENDIMENTO
      and      itreg_fat.cd_reg_fat = :CAMPO_CD_REG_FAT ) Item
order by Item.dt_lancamento,
              item.cd_gru_pro
)
ORDER BY CD_GRU_PRO,DS_GRU_PRO,dt_lancamento

--R_CONTAS_R_SETOR_Q_RESUMO_SETOR
SELECT * FROM (
Select Resumo.cd_setor,
       Resumo.nm_setor,
       Resumo.cd_seq_resumo_setor,
       Resumo.tp_gru_pro,
       Resumo.cd_gru_pro,
       Resumo.ds_gru_pro,
       Resumo.tp_proced,
       Resumo.cd_seq_resumo_gru_pro,
       Resumo.vl_total_conta,
       Resumo.vl_acrescimo,
       Resumo.vl_desconto,
       Resumo.vl_filme
from (Select itreg_fat.cd_setor,
             setor.nm_setor,
             nvl(setor.cd_seq_resumo,99) cd_seq_resumo_setor,
             gru_pro.tp_gru_pro,
             gru_pro.cd_gru_pro,
             gru_pro.ds_gru_pro,
             Decode( gru_pro.tp_gru_pro, 'SP', 'Honorário Médico',
                                                          'SD', 'Serviço Diagnóstico',
                                                          'SH', 'Serviço Hospitalar',
                                                          'MT', 'Material',
                                                          'MD', 'Medicamento',
                                                          'OP', 'Órtese e Prótese' ) tp_proced,
             itfor_apre_gru_pro.cd_seq_resumo cd_seq_resumo_gru_pro,
             Sum( nvl((itlan_med.vl_ato * qt_lancamento),itreg_fat.vl_total_conta)) vl_total_conta,
             Sum( itreg_fat.vl_acrescimo ) vl_acrescimo,
             Sum( itreg_fat.vl_desconto ) vl_desconto,
             Sum( itreg_fat.vl_filme_unitario * itreg_fat.qt_lancamento ) vl_filme
        From dbamv.itreg_fat,
             dbamv.reg_fat,
       dbamv.gru_fat,
       dbamv.gru_pro,
       dbamv.pro_fat,
       dbamv.setor,
       dbamv.convenio,
       dbamv.for_apre,
       dbamv.itfor_apre_gru_pro,
       dbamv.itlan_med,
       Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And itreg_fat.cd_gru_fat = gru_fat.cd_gru_fat
   AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   and itreg_fat.cd_pro_fat = pro_fat.cd_pro_fat
   and gru_pro.cd_gru_pro = pro_fat.cd_gru_pro
   and itreg_fat.cd_setor = setor.cd_setor (+)
   and reg_fat.cd_reg_fat = itreg_fat.cd_reg_fat
   and convenio.cd_convenio = reg_fat.cd_convenio
   and for_apre.cd_for_apre = convenio.cd_for_apre
   and itfor_apre_gru_pro.cd_for_apre = for_apre.cd_for_apre
   and itfor_apre_gru_pro.cd_gru_pro = gru_pro.cd_gru_pro
   and Setor.cd_setor in (select distinct cd_setor
                                       from dbamv.Usuario_unid_int
                                       where cd_id_usuario = user)
   and itlan_med.cd_reg_fat(+) = itreg_fat.cd_reg_fat
   and itlan_med.cd_lancamento(+) = itreg_fat.cd_lancamento
   and nvl(itreg_fat.tp_pagamento, 'S') <> 'C'
   and itreg_fat.sn_pertence_pacote = 'N'
   and reg_fat.cd_atendimento = :CAMPO_CD_ATENDIMENTO
   and itreg_fat.cd_reg_fat = :CAMPO_CD_REG_FAT
Group by  itreg_fat.cd_setor,
                setor.nm_setor,
                nvl(setor.cd_seq_resumo,99),
                gru_pro.cd_gru_pro,
                gru_pro.ds_gru_pro,
                gru_pro.tp_gru_pro,
                itfor_apre_gru_pro.cd_seq_resumo) resumo
order by  resumo.cd_gru_pro
)
ORDER BY CD_SETOR,NM_SETOR,CD_SEQ_RESUMO_SETOR
