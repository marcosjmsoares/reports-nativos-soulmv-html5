--R_DIARIAS_AUTOMAT
SELECT DISTINCT
               LOG.CD_ATENDIMENTO
              ,LOG.CD_REG_FAT
              ,LOG.QT_IMPORTADA
              ,LOG.CD_PRO_FAT
              ,LOG.DS_ERRO_CALCULO
              ,PRO_FAT.DS_PRO_FAT
              ,LEITO.CD_LEITO
              ,LEITO.CD_UNID_INT
FROM DBAMV.LOG_DIARIA_AUTOMATICA  LOG
          ,DBAMV.PRO_FAT
          ,DBAMV.MOV_INT
          ,DBAMV.LEITO
WHERE LOG.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
AND       LOG.CD_ATENDIMENTO = MOV_INT.CD_ATENDIMENTO
AND       LOG.CD_MOV_INT = MOV_INT.CD_MOV_INT
AND       MOV_INT.CD_LEITO = LEITO.CD_LEITO
AND 	  LOG.CD_ATENDIMENTO IN (SELECT CD_ATENDIMENTO FROM DBAMV.ATENDIME WHERE CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA)
AND     TRUNC(LOG.DT_IMPORTACAO) = :CAMPO_DT_IMPORTA
ORDER BY LOG.CD_ATENDIMENTO
