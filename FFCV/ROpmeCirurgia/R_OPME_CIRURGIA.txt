--R_OPME_CIRURGIA
SELECT * FROM (
select   to_char(nvl(aviso_cirurgia.dt_inicio_cirurgia,age_cir.dt_inicio_age_cir),'hh24') inicio
           ,to_char(nvl(aviso_cirurgia.dt_fim_cirurgia,age_cir.dt_final_age_cir),'hh24') fim
           ,aviso_cirurgia.cd_aviso_cirurgia aviso
           ,aviso_cirurgia.Dt_Aviso_Cirurgia dt_aviso
           ,aviso_cirurgia.cd_atendimento atendimento
           ,dbamv.pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(aviso_cirurgia.cd_paciente, 1) paciente
           ,aviso_cirurgia.vl_idade idade
           ,decode(aviso_cirurgia.tp_sexo,'F','FEMININO','M','MASCULINO') sexo
           ,aviso_cirurgia.cd_sal_cir cd_sal_cir
           ,aviso_cirurgia.cd_cen_cir cd_cen_cir
           ,decode(aviso_cirurgia.tp_situacao,'A','AVISO','R','REALIZADA','C','CANCELADA','G','AGENDADA') situacao
           ,aviso_cirurgia.nr_telefone_contato contato
           ,aviso_cirurgia.cd_usuario  usuario_aviso
           ,cirurgia_aviso.cd_cirurgia cd_cirurgia
           ,cirurgia_aviso.cd_convenio cd_convenio
           ,convenio.nm_convenio nm_convenio
           ,cirurgia.ds_cirurgia cirurgia
           ,guia.cd_guia cd_guia
           ,prestador_aviso.cd_prestador cd_prestador
           ,prestador.nm_prestador prestador
           ,cen_cir.ds_cen_cir  centro_cirurgico
           ,sal_cir.ds_sal_cir sala
	from  dbamv.aviso_cirurgia aviso_cirurgia
	       ,dbamv.cirurgia_aviso cirurgia_aviso
	       ,dbamv.age_cir age_cir
	       ,dbamv.cirurgia cirurgia
	       ,dbamv.guia guia
                       ,dbamv.prestador_aviso prestador_aviso
	      ,dbamv.prestador prestador
	      ,dbamv.cen_cir cen_cir
	      ,dbamv.sal_cir sal_cir
                      ,dbamv.convenio convenio
 where  aviso_cirurgia.cd_aviso_cirurgia            = age_cir.cd_aviso_cirurgia (+)
   and  aviso_cirurgia.cd_aviso_cirurgia              = cirurgia_aviso.cd_aviso_cirurgia
   and  aviso_cirurgia.cd_cen_cir                        = cen_cir.cd_cen_cir
   and  aviso_cirurgia.cd_sal_cir                         = sal_cir.cd_sal_cir
   and  cirurgia_aviso.cd_cirurgia                        = cirurgia.cd_cirurgia
   and  guia.cd_aviso_cirurgia                            = aviso_cirurgia.cd_aviso_cirurgia
   and  guia.tp_guia                                            = 'O'
   and  prestador_aviso.sn_principal                   =   'S'
   and  prestador_aviso.cd_cirurgia_aviso          =   cirurgia_aviso.cd_cirurgia_aviso
   and  prestador_aviso.cd_aviso_cirurgia          =   aviso_cirurgia.cd_aviso_cirurgia
   and  prestador_aviso.cd_prestador                 =   prestador.cd_prestador
   and  cen_cir.cd_cen_cir                                 =   sal_cir.cd_cen_cir
   and  convenio.cd_convenio                           =   cirurgia_aviso.cd_convenio(+)
   and aviso_cirurgia.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
Order by  aviso_cirurgia.cd_aviso_cirurgia
         ,aviso_cirurgia.Dt_Aviso_Cirurgia
)
ORDER BY aviso, dt_aviso,cd_cirurgia,cd_sal_cir,sala
