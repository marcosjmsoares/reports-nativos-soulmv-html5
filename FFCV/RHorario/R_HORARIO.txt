--R_HORARIO
SELECT * FROM (
Select  horario_especial.cd_horario                                           cd_horario
        ,  horario_especial.ds_horario                                            ds_horario
        ,  ithorario_especial.cd_dia                                                cd_dia
        ,  TO_CHAR( ithorario_especial.HR_FIM   , 'HH24' )    HR_FIM
        ,  TO_CHAR( ithorario_especial.HR_INICIO, 'HH24' )  HR_INICIO
 From  dbamv.horario_especial
        ,  dbamv.ithorario_especial
Where     horario_especial.cd_horario = ithorario_especial.cd_horario
Order By  horario_especial.cd_horario
           ,    ithorario_especial.cd_dia
           ,   TO_CHAR( ithorario_especial.HR_FIM   , 'HH24' )
)
ORDER BY DS_HORARIO,CD_HORARIO,
CD_DIA
