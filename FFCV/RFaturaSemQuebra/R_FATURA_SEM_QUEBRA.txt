       Decode(gru_pro.tp_gru_pro,'OP','B','MT','D','MD', 'C','A') tipo,
        decode(cod_pro.ds_nome_cobranca,null, decode(itfor_apre_gru_pro.sn_imprime_cbhpm,'S',nvl(substr(hierarquizado.ds_pro_fat_hierarquizado,1,60),pro_fat.ds_pro_fat),
  decode(itfor_apre_gru_pro.sn_imprime_simpro,'S',nvl(substr(simpro.ds_simpro,1,60),pro_fat.ds_pro_fat),pro_fat.ds_pro_fat)),cod_pro.ds_nome_cobranca) ds_ordenacao,
       Decode(gru_pro.tp_gru_pro,'OP','Material e Medicamento por Nota Fiscal','MT','Materiais','MD', 'Medicamentos','Serviços' ) tp_proced,
       itreg_fat.cd_pro_fat cd_pro_fat_calc,
       decode(itfor_apre_gru_pro.sn_imprime_cbhpm,'S',NVL(hierarquizado.cd_pro_fat_hierarquizado,pro_fat.cd_pro_fat), pro_fat.cd_pro_fat) cd_pro_fat,
        decode(itfor_apre_gru_pro.sn_imprime_cbhpm,'S',NVL(substr(hierarquizado.ds_pro_fat_hierarquizado,1,60), pro_fat.ds_pro_fat), pro_fat.ds_pro_fat) ds_pro_fat,
       sum(itreg_fat.qt_lancamento) qt_lancamento,
  nvl((itlan_med.vl_ato * SUM(itreg_fat.qt_lancamento) + nvl(itlan_med.vl_desconto_conta,0)),
          decode(itfor_apre_gru_pro.sn_imprime_vl_filme,'N',itreg_fat.vl_unitario + NVL(itreg_fat.vl_filme_unitario,0),itreg_fat.vl_unitario)) vl_unitario,
       decode(itfor_apre_gru_pro.sn_imprime_vl_filme,'S',itreg_fat.vl_filme_unitario * Sum( itreg_fat.qt_lancamento ),0) vl_filme,
       DECODE(itlan_med.cd_reg_fat, NULL,
                      sum( itreg_fat.vl_total_conta),
                      sum(itlan_med.vl_liquido)) vl_total_conta,
       DECODE(itlan_med.cd_reg_fat, NULL,
                       Sum( itreg_fat.vl_total_conta ) + sum(nvl(itreg_fat.vl_desconto_conta,0)),
                       Sum(itlan_med.vl_liquido)  + Sum(nvl(itlan_med.vl_desconto_conta,0)))  vl_liquido,
       sum(nvl(itreg_fat.vl_desconto_conta,0)) vl_desconto_conta,
       max(pro_fat.ds_unidade) ds_unidade,
       max(itfor_apre_gru_pro.sn_imprime_ch) sn_imprime_ch,
       max(itfor_apre_gru_pro.sn_imprime_simpro) sn_imprime_simpro,
       max(itfor_apre_gru_pro.sn_imprime_brasindice) sn_imprime_brasindice,
       max(itfor_apre_gru_pro.sn_imprime_cbhpm) sn_imprime_cbhpm,
       max(nvl(itreg_fat.sn_horario_especial,'N')) sn_horario_especial,
       max(convenio.cd_convenio) cd_convenio,
       nvl(max(prestador_equipe.nm_mnemonico),max(prestador.nm_mnemonico)) prestador,
        decode(cod_pro.ds_nome_cobranca,null, decode(itfor_apre_gru_pro.sn_imprime_cbhpm,'S',nvl(substr(hierarquizado.ds_pro_fat_hierarquizado,1,60),pro_fat.ds_pro_fat),
  decode(itfor_apre_gru_pro.sn_imprime_simpro,'S',nvl(substr(simpro.ds_simpro,1,60),pro_fat.ds_pro_fat),pro_fat.ds_pro_fat)),cod_pro.ds_nome_cobranca)  func_desc_fat,
       nvl(cod_pro.ds_unidade_cobranca,pro_fat.ds_unidade) func_unid_fat,
       nvl(cod_pro.ds_codigo_cobranca,itreg_fat.cd_pro_fat) func_cd_fat,
       itlan_med.cd_ati_med cd_ati_med,
       itlan_med.vl_desconto_conta   vl_desc_conta ,
       itfor_apre_gru_pro.sn_imprime_vl_filme,
       max(imp_simpro.cd_tab_fat) cd_tab_fat
  From dbamv.itreg_fat,
       dbamv.reg_fat,
       dbamv.pro_fat,
       dbamv.gru_pro,
       dbamv.atendime,
       dbamv.for_apre,
       dbamv.itfor_apre_gru_pro,
       dbamv.itlan_med,
       dbamv.simpro,
       dbamv.convenio,
       dbamv.prestador,
       dbamv.prestador prestador_equipe,
      (select distinct cod_pro.cd_pro_fat,cod_pro.ds_nome_cobranca,cod_pro.ds_unidade_cobranca,cod_pro.ds_codigo_cobranca
       from   dbamv.reg_fat
             ,dbamv.cod_pro
             ,dbamv.atendime
       where ( Cod_Pro.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA or Cod_Pro.Cd_Multi_Empresa Is Null )
         and  reg_fat.cd_reg_fat  = :CAMPO_CD_REG_FAT
         and  reg_fat.cd_atendimento = :CAMPO_CD_ATENDIMENTO
         and  cod_pro.cd_convenio = reg_fat.cd_convenio
         and  atendime.cd_atendimento = reg_fat.cd_atendimento
         and  (atendime.tp_atendimento = cod_pro.tp_atendimento or cod_pro.tp_atendimento = 'T')) cod_pro,
    (select  DISTINCT PRO_HIE.cd_pro_fat_hierarquizado
                ,PRO_HIE.ds_pro_fat_hierarquizado
                ,PRO_HIE.cd_pro_fat
       from   dbamv.reg_fat
             ,dbamv.PRO_FAT_HIERARQUIZADO PRO_HIE
             ,dbamv.itreg_fat
       where  reg_fat.cd_reg_fat  = :CAMPO_CD_REG_FAT
         and  reg_fat.cd_atendimento = :CAMPO_CD_ATENDIMENTO
         and  itreg_fat.cd_reg_fat = reg_fat.cd_reg_fat
         and  itreg_fat.cd_pro_fat = PRO_HIE.cd_pro_fat) hierarquizado,
      (select cd_pro_fat,
              max(cd_tab_fat) cd_tab_fat
      from
      (select distinct pro_fat.cd_pro_fat,
                       itregra.cd_tab_fat
       from   dbamv.pro_fat,
              dbamv.itregra,
              dbamv.reg_fat,
              dbamv.imp_simpro
       where  reg_fat.cd_regra = itregra.cd_regra
         and  reg_fat.cd_reg_fat  = :CAMPO_CD_REG_FAT
         and  reg_fat.cd_atendimento = :CAMPO_CD_ATENDIMENTO
         and  itregra.cd_gru_pro = pro_fat.cd_gru_pro
         and  pro_fat.cd_pro_fat = imp_simpro.cd_pro_fat
         and  itregra.cd_tab_fat = imp_simpro.cd_tab_fat
       union
       select distinct imp_simpro.cd_pro_fat,
                        0 cd_tab_fat
       from   dbamv.imp_simpro
       where  imp_simpro.cd_tab_fat is null)
       group by cd_pro_fat)  simpro_tab,

    AND ITREG_FAT.CD_PRO_FAT = HIERARQUIZADO.CD_PRO_FAT(+)
   and imp_simpro.cd_simpro = simpro.cd_simpro (+)
