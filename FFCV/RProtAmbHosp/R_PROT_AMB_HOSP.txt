select log_meio_mag.nr_nota,
       paciente.cd_paciente,
       paciente.nm_paciente,
       log_meio_mag.cd_remessa,
       log_meio_mag.tp_nota,
       log_meio_mag.vl_sp,
       log_meio_mag.vl_sh,
       log_meio_mag.vl_sd,
       sum( Nvl( log_meio_mag.vl_sh , 0 )+ Nvl( log_meio_mag.vl_sp, 0 )+ Nvl( log_meio_mag.vl_sd, 0 ) ) vl_total
  from dbamv.paciente,
       dbamv.atendime,
       dbamv.log_meio_mag
 where log_meio_mag.cd_remessa = :CAMPO_REMESSA
   and log_meio_mag.tp_nota = :CAMPO_TP_NOTA
   and atendime.cd_atendimento = log_meio_mag.cd_atendimento
   and paciente.cd_paciente = atendime.cd_paciente
  and atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
Group by
    log_meio_mag.nr_nota,
    paciente.cd_paciente,
    paciente.nm_paciente,
    log_meio_mag.cd_remessa,
    log_meio_mag.tp_nota,
    log_meio_mag.vl_sp,
    log_meio_mag.vl_sh,
    log_meio_mag.vl_sd
order by paciente.nm_paciente
