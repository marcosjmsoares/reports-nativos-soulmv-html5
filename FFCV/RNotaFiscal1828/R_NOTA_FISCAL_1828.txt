--R_NOTA_FISCAL_1828
SELECT	itfat_nota_fiscal.cd_remessa, nota_fiscal.dt_emissao dt_emissao
		  ,nota_fiscal.nm_cliente nm_cliente
		  ,nota_fiscal.ds_endereco ds_endereco
		  ,nota_fiscal.nm_cidade nm_cidade
		  ,nota_fiscal.nm_uf nm_uf
		  ,'SERVICOS EXECUTADOS' descricao
		  ,nota_fiscal.vl_total_nota preco_total
		  ,SUM (itnota_fiscal.vl_gru_fat) vl_gru_fat
		  ,dbamv.pack_fncp.extensos (nota_fiscal.vl_total_nota) vl_extenso
		  ,nota_fiscal.cd_nota_fiscal cd_nota_fiscal
		  ,nota_fiscal.nr_cgc_cpf nr_cgc_cpf
		  ,nota_fiscal.nr_inscricao_municipal nr_inscricao_municipal
		  ,nota_fiscal.nr_inscricao_estadual nr_inscricao_estadual
		  ,nota_fiscal.nr_cep nr_cep
		  ,nota_fiscal.nm_bairro nm_bairro
        ,NVL(nota_fiscal.cd_reg_fat, nota_fiscal.cd_atendimento) conta
        ,nota_fiscal.nr_id_nota_fiscal nr_id_nota_fiscal
        ,DECODE(NOTA_FISCAL.DS_OBSERVACOES, NULL, DBAMV.FNC_FFCV_NF_TRIBUTO(nota_fiscal.cd_nota_fiscal), NOTA_FISCAL.DS_OBSERVACOES||CHR(10)||CHR(10)||DBAMV.FNC_FFCV_NF_TRIBUTO(nota_fiscal.cd_nota_fiscal)) DS_OBSERVACOES
        ,nota_fiscal.cd_convenio cd_convenio
        ,nota_fiscal.tp_nota_fiscal
        ,nota_fiscal.dt_pagamento
        ,nota_fiscal.vl_perc_iss
        ,nota_fiscal.vl_iss
        ,DECODE(NOTA_FISCAL.vl_iss, NULL, '', 'ISS') DS_ISS
	 FROM dbamv.nota_fiscal nota_fiscal
		  ,dbamv.itnota_fiscal itnota_fiscal
      , dbamv.itfat_nota_fiscal
	WHERE nota_fiscal.cd_nota_fiscal = itnota_fiscal.cd_nota_fiscal
  AND itfat_nota_fiscal.cd_nota_fiscal (+) = nota_fiscal.cd_nota_fiscal
     AND nota_fiscal.cd_nota_fiscal = :CAMPO_CD_NOTA_FISCAL
       AND Nota_Fiscal.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
GROUP BY itfat_nota_fiscal.cd_remessa, nota_fiscal.dt_emissao
		  ,nota_fiscal.nm_cliente
		  ,nota_fiscal.nm_cliente
		  ,nota_fiscal.ds_endereco
		  ,nota_fiscal.nm_cidade
		  ,nota_fiscal.nm_uf
		  ,nota_fiscal.vl_total_nota
		  ,dbamv.pack_fncp.extensos (nota_fiscal.vl_total_nota)
		  ,nota_fiscal.cd_nota_fiscal
		  ,nota_fiscal.nr_cgc_cpf
		  ,nota_fiscal.nr_inscricao_municipal
		  ,nota_fiscal.nr_inscricao_estadual
		  ,nota_fiscal.nr_cep
		  ,nota_fiscal.nm_bairro
        ,NVL(nota_fiscal.cd_reg_fat, nota_fiscal.cd_atendimento)
        ,nota_fiscal.nr_id_nota_fiscal
        ,DECODE(NOTA_FISCAL.DS_OBSERVACOES, NULL, DBAMV.FNC_FFCV_NF_TRIBUTO(nota_fiscal.cd_nota_fiscal), NOTA_FISCAL.DS_OBSERVACOES||CHR(10)||CHR(10)||DBAMV.FNC_FFCV_NF_TRIBUTO(nota_fiscal.cd_nota_fiscal))
        ,nota_fiscal.cd_convenio
        ,nota_fiscal.tp_nota_fiscal
        ,nota_fiscal.dt_pagamento
        ,nota_fiscal.vl_perc_iss
        ,nota_fiscal.vl_iss
        ,DECODE(NOTA_FISCAL.vl_iss, NULL, '', 'ISS')
