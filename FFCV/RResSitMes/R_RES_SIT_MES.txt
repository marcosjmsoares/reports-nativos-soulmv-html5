--R_RES_SIT_MES
SELECT * FROM (
SELECT CD_CONVENIO
                                     ,NM_CONVENIO
                                     ,CONTA
                                     ,CD_ATENDIMENTO
                                     ,NM_PACIENTE
                                     ,DS_ORI_ATE
                                     ,DT_FINAL
                                     ,DT_INICIO
                                     ,TIPO
                                     ,QT_N_FATURADO
FROM
(SELECT    CD_CONVENIO
                                     ,NM_CONVENIO
                                     ,CONTA
                                     ,CD_ATENDIMENTO
                                     ,NM_PACIENTE
                                     ,DS_ORI_ATE
                                     ,DT_FINAL
                                     ,DT_INICIO
                                     ,TIPO
                                     ,SUM(QT_N_FATURADO) QT_N_FATURADO
    FROM (
           SELECT   CONVENIO.CD_CONVENIO        CD_CONVENIO
	          ,CONVENIO.NM_CONVENIO        NM_CONVENIO
                          ,REG_FAT.CD_REG_FAT CONTA
                          ,ATENDIME.CD_ATENDIMENTO
                          ,PACIENTE.NM_PACIENTE
                          ,ORI_ATE.DS_ORI_ATE
                          ,REG_FAT.DT_FINAL                   DT_FINAL
                          ,REG_FAT.DT_INICIO                  DT_INICIO
                          ,'I' TIPO
  	          ,SUM( NVL(REG_FAT.VL_TOTAL_CONTA, 0))  QT_N_FATURADO
             FROM   DBAMV.REG_FAT
                         ,DBAMV.REMESSA_FATURA
                         ,DBAMV.FATURA
	         ,DBAMV.CONVENIO
                         ,DBAMV.ATENDIME
                         ,DBAMV.ORI_ATE
                         ,DBAMV.PACIENTE
           WHERE           ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
                          AND ATENDIME.CD_ORI_ATE   = ORI_ATE.CD_ORI_ATE
                          AND ATENDIME.CD_ATENDIMENTO = REG_FAT.CD_ATENDIMENTO
	          AND REMESSA_FATURA.CD_FATURA  = FATURA.CD_FATURA
                          AND REMESSA_FATURA.CD_REMESSA = REG_FAT.CD_REMESSA
                          AND REG_FAT.SN_FECHADA = 'S'
                          AND REMESSA_FATURA.SN_FECHADA = 'S'
	          AND REG_FAT.CD_CONVENIO = CONVENIO.CD_CONVENIO
   	         AND REG_FAT.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
                          AND   NVL(REG_FAT.sn_diagno, 'N') = 'N'
	          AND CONVENIO.TP_CONVENIO <> 'H'
                        AND ATENDIME.CD_MULTI_EMPRESA =   :CAMPO_CD_MULTI_EMPRESA
           GROUP BY CONVENIO.CD_CONVENIO
	          ,CONVENIO.NM_CONVENIO
                          ,REG_FAT.CD_REG_FAT
                          ,ATENDIME.CD_ATENDIMENTO
                          ,PACIENTE.NM_PACIENTE
                          ,ORI_ATE.DS_ORI_ATE
                          ,REG_FAT.DT_FINAL
                          ,REG_FAT.DT_INICIO
                          ,'I'
      UNION ALL
           SELECT CONVENIO.CD_CONVENIO             CD_CONVENIO
                         ,CONVENIO.NM_CONVENIO             NM_CONVENIO
                         ,ITREG_AMB.CD_REG_AMB             CONTA
                         ,ITREG_AMB.CD_ATENDIMENTO   CD_ATENDIMENTO
                         ,PACIENTE.NM_PACIENTE              NM_PACIENTE
                          ,ORI_ATE.DS_ORI_ATE
                         ,ATENDIME.DT_ATENDIMENTO      DT_FINAL
                         ,ATENDIME.DT_ATENDIMENTO      DT_INICIO
                         ,'A' TIPO
                         ,NVL(ITREG_AMB.VL_TOTAL_CONTA,0)   QT_N_FATURADO
             FROM   DBAMV.REG_AMB
                         ,DBAMV.ITREG_AMB
                         ,DBAMV.CONVENIO
                         ,DBAMV.REMESSA_FATURA
                         ,DBAMV.FATURA
                         ,DBAMV.ATENDIME
                         ,DBAMV.PACIENTE
                         ,DBAMV.ORI_ATE
           WHERE REG_AMB.CD_REG_AMB = ITREG_AMB.CD_REG_AMB
                          AND ITREG_AMB.SN_FECHADA = 'S'
                          AND ITREG_AMB.CD_CONVENIO = CONVENIO.CD_CONVENIO
                          AND REG_AMB.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
                          AND NVL(ITREG_AMB.SN_PERTENCE_PACOTE, 'N') = 'N'
                          AND NVL(ITREG_AMB.TP_PAGAMENTO,'X') <> 'C'
                          AND CONVENIO.TP_CONVENIO <> 'H'
	          AND REMESSA_FATURA.CD_FATURA  = FATURA.CD_FATURA
                          AND REMESSA_FATURA.CD_REMESSA = REG_AMB.CD_REMESSA
                          AND REMESSA_FATURA.SN_FECHADA = 'S'
                          AND   NVL(REG_AMB.sn_diagno, 'N') = 'N'
                          AND PACIENTE.CD_PACIENTE = ATENDIME.CD_PACIENTE
                          AND ORI_ATE.CD_ORI_ATE = ATENDIME.CD_ORI_ATE
                          AND ATENDIME.CD_ATENDIMENTO = ITREG_AMB.CD_ATENDIMENTO
                          AND ATENDIME.CD_MULTI_EMPRESA =  :CAMPO_CD_MULTI_EMPRESA
 			  :CAMPO_CF_PERIODO
              )
where nvl(qt_n_faturado,0) >= 0
GROUP BY CD_CONVENIO
                                     ,NM_CONVENIO
                                     ,CONTA
                                     ,CD_ATENDIMENTO
                                     ,NM_PACIENTE
                                     ,DS_ORI_ATE
                                     ,DT_FINAL
                                     ,DT_INICIO
                                     ,TIPO )
ORDER BY TIPO, CONTA
)
ORDER BY NM_CONVENIO,CD_CONVENIO,TIPO, NM_PACIENTE
