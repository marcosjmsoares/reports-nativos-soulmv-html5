--R_GUIA_ORCAMENTO_OPME_COPA_DOR
SELECT GUIA.CD_GUIA
     , PACIENTE.NM_PACIENTE
     , CARTEIRA.NR_MATRICULA_TITULAR
     , CONVENIO.NM_CONVENIO
     , To_Char(GUIA.DT_SOLICITACAO,'DD/MM/YYYY') DT_SOLICITACAO
     , '('||DBAMV.PKG_TISS_UTIL.FNC_TRADUZ_PROC(GUIA.CD_CONVENIO,'C',PROC_CIR.CD_PRO_FAT) ||') '
     || DBAMV.PKG_TISS_UTIL.FNC_TRADUZ_PROC(GUIA.CD_CONVENIO,'D',PROC_CIR.CD_PRO_FAT) DS_CIRURGIA
     , TO_CHAR(GUIA.DT_SOLICITACAO,'DD/MM/YYYY') DT_SOLICITACAO
     , Nvl(GUIA.CD_SENHA,'NO ATO') CD_SENHA
     , GUIA.CD_CONVENIO
     , DBAMV.PKG_TISS_UTIL.FNC_TRADUZ_PROC(GUIA.CD_CONVENIO,'C',IT_GUIA.CD_PRO_FAT) CD_PRO_FAT
     , DBAMV.PKG_TISS_UTIL.FNC_TRADUZ_PROC(GUIA.CD_CONVENIO,'D',IT_GUIA.CD_PRO_FAT) DS_PRO_FAT
     , FORNECEDOR.CD_FORNECEDOR
     , FORNECEDOR.NM_FORNECEDOR
     , IT_GUIA.QT_AUTORIZADO
     , Nvl(Nvl(DBAMV.FNC_FFCV_TRADUZ_TUSS(19,'CD_CONVENIO',GUIA.CD_CONVENIO,'CD_PRO_FAT',IT_GUIA.CD_PRO_FAT,NULL,NULL,NULL,NULL,NULL,NULL,'CD_REFERENCIA'),FORNECEDOR.CD_RMS),FORNECEDOR.NR_REGISTRO_ANVISA) NR_ANVISA
     , FORNECEDOR.VL_UNITARIO
     , (IT_GUIA.QT_AUTORIZADO * FORNECEDOR.VL_UNITARIO) VL_TOTAL_ITEM
     , IMP_SIMPRO.CD_SIMPRO
     , IT_GUIA.CD_PRO_FAT
  FROM DBAMV.GUIA
     , DBAMV.IT_GUIA
     , DBAMV.PACIENTE
     , DBAMV.CONVENIO
     , DBAMV.ATENDIME
     , DBAMV.IMP_SIMPRO
     , ( SELECT NR_MATRICULA_TITULAR, DT_VALIDADE, CD_CONVENIO, CD_CON_PLA
           FROM (SELECT NVL(CARTEIRA.NR_MATRICULA_TITULAR,CARTEIRA.NR_CARTEIRA) NR_MATRICULA_TITULAR, MAX(CARTEIRA.DT_VALIDADE) DT_VALIDADE,
                        CARTEIRA.CD_CONVENIO, CARTEIRA.CD_CON_PLA
                   FROM DBAMV.CARTEIRA, DBAMV.GUIA
                  WHERE GUIA.CD_GUIA = :CAMPO_CD_GUIA
                    AND CARTEIRA.CD_PACIENTE = GUIA.CD_PACIENTE
                    AND CARTEIRA.CD_CONVENIO = GUIA.CD_CONVENIO
                  GROUP BY NVL(CARTEIRA.NR_MATRICULA_TITULAR,CARTEIRA.NR_CARTEIRA), CARTEIRA.CD_CONVENIO, CARTEIRA.CD_CON_PLA )
          WHERE ROWNUM=1 ) CARTEIRA
     , ( SELECT PRO_FAT.DS_PRO_FAT
              , PRO_FAT.CD_PRO_FAT
              , GUIA.CD_GUIA
           FROM DBAMV.PRO_FAT
              ,  DBAMV.GUIA
              ,  DBAMV.CIRURGIA_AVISO AVISO
              ,  DBAMV.CIRURGIA
              ,  DBAMV.AVISO_CIRURGIA
           WHERE AVISO.CD_AVISO_CIRURGIA = GUIA.CD_AVISO_CIRURGIA
             AND AVISO.CD_CIRURGIA = CIRURGIA.CD_CIRURGIA
             AND GUIA.CD_GUIA = :CAMPO_CD_GUIA
             AND PRO_FAT.CD_PRO_FAT = CIRURGIA.CD_PRO_FAT
             AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = AVISO.CD_AVISO_CIRURGIA
             ) PROC_CIR
     ,  ( SELECT DISTINCT VAL_OPME.CD_FORNECEDOR
               , FORNECEDOR.NM_FORNECEDOR
               , Nvl(VAL_OPME.VL_PROPOSTO,VAL_OPME.VL_UNITARIO) VL_UNITARIO
               , IT_GUIA.CD_IT_GUIA
               , FORNECEDOR.NR_REGISTRO_ANVISA
               , VAL_OPME.CD_RMS
            FROM DBAMV.IT_GUIA
               ,  DBAMV.VAL_OPME_IT_GUIA VAL_OPME
               ,  DBAMV.FORNECEDOR
           WHERE IT_GUIA.CD_GUIA = :CAMPO_CD_GUIA
             AND VAL_OPME.CD_GUIA(+)           = IT_GUIA.CD_GUIA
             AND VAL_OPME.CD_IT_GUIA(+)        = IT_GUIA.CD_IT_GUIA
             AND FORNECEDOR.CD_FORNECEDOR(+)   = VAL_OPME.CD_FORNECEDOR
             AND VAL_OPME.SN_ORC_ENVIO = 'S' ) FORNECEDOR
 WHERE CARTEIRA.CD_CONVENIO(+)   = GUIA.CD_CONVENIO
   AND GUIA.CD_GUIA = :CAMPO_CD_GUIA
   AND GUIA.TP_GUIA = 'O'
   AND GUIA.CD_GUIA = IT_GUIA.CD_GUIA
   AND ATENDIME.CD_ATENDIMENTO = GUIA.CD_ATENDIMENTO(+)
   AND CONVENIO.CD_CONVENIO = GUIA.CD_CONVENIO
   AND PACIENTE.CD_PACIENTE = NVL(GUIA.CD_PACIENTE,ATENDIME.CD_PACIENTE)
   AND GUIA.CD_GUIA = PROC_CIR.CD_GUIA(+)
   AND IT_GUIA.CD_IT_GUIA = FORNECEDOR.CD_IT_GUIA
   AND IT_GUIA.CD_PRO_FAT = IMP_SIMPRO.CD_PRO_FAT(+)
