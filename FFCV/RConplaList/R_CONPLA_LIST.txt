--R_CONPLA_LIST
SELECT  CONVENIO.CD_CONVENIO                                CODIGO
              ,CONVENIO.NM_CONVENIO                                DESCRICAO
              ,CONVENIO.CD_FOR_APRE                                 CD_FOR_APRE
              ,CONVENIO.TP_CONVENIO                                 TP_CONVENIO
              ,CON_PLA.CD_CON_PLA                                     CD_CON_PLA1
              ,CON_PLA.DS_CON_PLA                                     DS_CON_PLA1
              ,CON_PLA.CD_REGRA                                        CD_REGRA
              ,CON_PLA.CD_INDICE                                         CD_INDICE
              ,FOR_APRE.DS_FOR_APRE                               DS_FOR_APRE
              ,REGRA.DS_REGRA                                           DS_REGRA
              ,INDICE.NM_INDICE                                            NM_INDICE
                                                                 DS_OBSERVACAO
FROM    dbamv.CONVENIO               CONVENIO
             ,dbamv.FOR_APRE               FOR_APRE
             ,dbamv.CON_PLA                  CON_PLA
             ,dbamv.REGRA                      REGRA
             ,dbamv.INDICE                       INDICE
,Dbamv.Empresa_Con_Pla
Where Empresa_Con_Pla.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And Empresa_Con_Pla.Cd_Convenio = Con_Pla.Cd_Convenio
And Empresa_Con_Pla.Cd_Con_Pla = Con_Pla.Cd_Con_Pla
And CONVENIO.CD_FOR_APRE = FOR_APRE.CD_FOR_APRE(+)
AND        CONVENIO.CD_CONVENIO =  CON_PLA.CD_CONVENIO
AND       CON_PLA.CD_REGRA = REGRA.CD_REGRA(+)
AND       CON_PLA.CD_INDICE = INDICE.CD_INDICE(+)
ORDER BY CONVENIO.NM_CONVENIO
                   ,CONVENIO.TP_CONVENIO
                   ,FOR_APRE.DS_FOR_APRE 
                   ,CD_CON_PLA1
