--R_STATUS_CONTA
SELECT CD_ATENDIMENTO
      ,FECHA_CIERRE
      ,CD_PACIENTE
      ,NM_PACIENTE
      ,FACTURA
      ,CD_CONVENIO||' - '||NM_CONVENIO CONVENIO
      ,CD_ORI_ATE||' - '||DS_ORI_ATE ORIGEN
      ,NM_USUARIO
      ,TOTAL_CUENTA
      ,TP_CONTA
      ,TIPO_MOVIMENTO
      ,CD_CONVENIO
      ,CD_ORI_ATE
      ,TP_QUITACAO
FROM (
  SELECT DISTINCT
        ITA.CD_ATENDIMENTO
        ,TO_DATE(ITA.DT_FECHAMENTO) FECHA_CIERRE
        ,ATE.CD_PACIENTE
        ,PA.NM_PACIENTE
        ,ITA.CD_REG_AMB FACTURA
        ,ITA.CD_CONVENIO
        ,CON.NM_CONVENIO
        ,ATE.CD_ORI_ATE
        ,ORI.DS_ORI_ATE
        ,ATE.NM_USUARIO
        ,SUM(NVL(ITA.VL_TOTAL_CONTA,0)) TOTAL_CUENTA
        ,:CAMPO_TRADUCAO_TP_CONTA_FECHADA TP_CONTA
        ,DECODE(ITA.TP_MVTO_DESCONTO ,'FDISTRIBUI' , :CAMPO_TRADUCAO_CONTA_DISTRIBUICAO,
                                      'FDISTRIBUI' , :CAMPO_TRADUCAO_INSTITUICOES_SEGUROS,
                                      :CAMPO_TRADUCAO_CONTA_PRINCIPAL) TIPO_MOVIMENTO
        ,:CAMPO_CF_TP_QUITACAO  TP_QUITACAO
  FROM ITREG_AMB  ITA
      ,ATENDIME   ATE
      ,CONVENIO   CON
      ,ORI_ATE    ORI
      ,PACIENTE   PA
  WHERE ATE.CD_PACIENTE = PA.CD_PACIENTE
    AND ITA.CD_ATENDIMENTO = ATE.CD_ATENDIMENTO
    AND ITA.CD_CONVENIO = CON.CD_CONVENIO
    AND ATE.CD_ORI_ATE = ORI.CD_ORI_ATE
    AND TRUNC(ATE.DT_ATENDIMENTO) BETWEEN  TO_DATE(:CAMPO_DT_ATENDIMENTO, 'DD/MM/YYYY HH24') AND TO_DATE(:CAMPO_DT_ATENDIMENTO_FINAL, 'DD/MM/YYYY HH24')
    AND NOT EXISTS ( SELECT CD_LANCAMENTO FROM DBAMV.ITREG_AMB WHERE ITREG_AMB.CD_REG_AMB = ITA.CD_REG_AMB AND ITREG_AMB.SN_FECHADA = 'N')
  GROUP BY ITA.CD_ATENDIMENTO
          ,TO_DATE(ITA.DT_FECHAMENTO)
          ,ATE.CD_PACIENTE
          ,PA.NM_PACIENTE
          ,ITA.CD_REG_AMB
          ,ITA.CD_CONVENIO
          ,CON.NM_CONVENIO
          ,ATE.CD_ORI_ATE
          ,ORI.DS_ORI_ATE
          ,ATE.NM_USUARIO
          ,ITA.TP_MVTO_DESCONTO
  UNION ALL
  SELECT DISTINCT
        ITA.CD_ATENDIMENTO
        ,TO_DATE(ITA.DT_FECHAMENTO) FECHA_CIERRE
        ,ATE.CD_PACIENTE
        ,PA.NM_PACIENTE
        ,ITA.CD_REG_AMB FACTURA
        ,ITA.CD_CONVENIO
        ,CON.NM_CONVENIO
        ,ATE.CD_ORI_ATE
        ,ORI.DS_ORI_ATE
        ,ATE.NM_USUARIO
        ,SUM(NVL(ITA.VL_TOTAL_CONTA,0)) TOTAL_CUENTA
        ,:CAMPO_TRADUCAO_TP_CONTA_ABERTA TP_CONTA
        ,DECODE(ITA.TP_MVTO_DESCONTO ,'FDISTRIBUI' , :CAMPO_TRADUCAO_CONTA_DISTRIBUICAO,
                                      'FDISTRIBUI' , :CAMPO_TRADUCAO_INSTITUICOES_SEGUROS,
                                      :CAMPO_TRADUCAO_CONTA_PRINCIPAL) TIPO_MOVIMENTO
        ,:CAMPO_CF_TP_QUITACAO  TP_QUITACAO
  FROM ITREG_AMB   ITA
      ,ATENDIME    ATE
      ,CONVENIO    CON
      ,ORI_ATE     ORI
      ,PACIENTE    PA
  WHERE  ATE.CD_PACIENTE = PA.CD_PACIENTE
    AND ITA.CD_ATENDIMENTO = ATE.CD_ATENDIMENTO
    AND ITA.CD_CONVENIO = CON.CD_CONVENIO
    AND ATE.CD_ORI_ATE = ORI.CD_ORI_ATE
    AND TRUNC(ATE.DT_ATENDIMENTO) BETWEEN  TO_DATE(:CAMPO_DT_ATENDIMENTO, 'DD/MM/YYYY HH24') AND TO_DATE(:CAMPO_DT_ATENDIMENTO_FINAL, 'DD/MM/YYYY HH24')
    AND  EXISTS ( SELECT CD_LANCAMENTO FROM DBAMV.ITREG_AMB WHERE ITREG_AMB.CD_REG_AMB = ITA.CD_REG_AMB AND ITREG_AMB.SN_FECHADA = 'N')
  GROUP BY ITA.CD_ATENDIMENTO
          ,TO_DATE(ITA.DT_FECHAMENTO)
          ,ATE.CD_PACIENTE
          ,PA.NM_PACIENTE
          ,ITA.CD_REG_AMB
          ,ITA.CD_CONVENIO
          ,CON.NM_CONVENIO
          ,ATE.CD_ORI_ATE
          ,ORI.DS_ORI_ATE
          ,ATE.NM_USUARIO
	,ITA.TP_MVTO_DESCONTO
  ORDER BY 1
)
WHERE CD_ATENDIMENTO IS NOT NULL
