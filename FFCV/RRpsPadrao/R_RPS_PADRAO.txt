SELECT VIA, NF.CD_NOTA_FISCAL, NF.CD_ITNOTA_FISCAL,
       (SELECT  SubStr(REPLACE(LISTAGG(CONCAT('| ',decode(DET.TP_DETALHAMENTO,'I','ISS','P','PIS','L','CSLL','J','IRPJ','C','COFINS','OUTROS')||' : '||TO_CHAR(NFT.VL_TRIBUTO, 'FM999G999G999D90', 'nls_numeric_characters='',.''') ) ) WITHIN GROUP (ORDER BY DET.TP_DETALHAMENTO),';',''),2)
       FROM DBAMV.ITNOTA_FISCAL_TRIBUTO NFT
       INNER JOIN DBAMV.TIP_DETALHE DET ON NFT.CD_DETALHAMENTO = DET.CD_DETALHAMENTO
       LEFT JOIN DBAMV.TIP_DETALHE_CST CST ON NFT.CD_CST = CST.CD_CST
       WHERE NFT.CD_NOTA_FISCAL = NF.CD_NOTA_FISCAL
		   AND NFT.CD_ITNOTA_FISCAL = NF.CD_ITNOTA_FISCAL)  DET_TRIBUTO_ITEM ,
       NM_CLIENTE , NR_CGC_CPF, DS_ENDERE, DS_BAIRRO, NM_CIDADE, NM_UF, NR_CEP,NR_INSCRICAO_ESTADUAL, NR_INSCRICAO_MUNICIPAL, EMAIL,
       NR_FONE, DT_EMISSAO, NR_ID_NOTA_FISCAL, VL_TOTAL_NOTA, DS_OBSERVACAO, CD_GRU_FAT,DS_GRU_FAT, VL_GRU_FAT, VL_DESCONTO_NOTA, VL_ACRESCIMO_NOTA,DS_DESCONTO, DS_ACRESCIMO,
       CD_SERIE, CD_CONVENIO, TP_CONVENIO, EMAIL_HOSPITAL, DS_MULTI_EMPRESA,  DS_ENDERECO_EMPRESA, NR_DDD_EMPRESA, NR_FONE_EMPRESA, NM_CIDADE_EMPRESA, NR_DDD_FAX_EMPRESA, NR_FAX_EMPRESA,
       NR_CGC, NR_INSCRICAO_MUNIC, NR_INSCRICAO_EST, DT_ALTA, HR_ALTA, DT_ATENDIMENTO,CD_ATENDIMENTO, HR_ATENDIMENTO, DS_ORI_ATE, NM_PACIENTE,DS_ENDERECO_COBRANCA DS_ENDERECO_COBRANCA,
       DS_BAIRRO_COBRANCA DS_BAIRRO_COBRANCA,DS_CIDADE_COBRANCA DS_CIDADE_COBRANCA, NR_CEP_COBRANCA NR_CEP_COBRANCA , DS_UF_COBRANCA , DADOS_INTERMEDIARIO,
       OBSERVACAO,DS_TRIBUTOS,SUB_TOTAL,HR_EMISSAO_NFE,NR_NOTA_FISCAL_NFE,CD_VERIFICACAO_NFE

       DECODE(FNC_FFCV_SN_INFORMA_INTER(CONVENIO.CD_CONVENIO),'S',NULL,NOTA_FISCAL.NR_CGC_CPF) NR_CGC_CPF,
       DECODE(FNC_FFCV_SN_INFORMA_INTER(CONVENIO.CD_CONVENIO),'S',NULL,TIPO_LOGRADOURO.DS_RESUMIDO || ' ' || NOTA_FISCAL.DS_ENDERECO || ',' || NOTA_FISCAL.NR_ENDERECO) DS_ENDERE,

       MULTI_EMPRESAS.NR_DDD_FAX  NR_DDD_FAX_EMPRESA,
       SUBSTR(NVL(EMPRESA_FORMULARIO.DS_NR_FONE_FAX, MULTI_EMPRESAS.NR_FAX),1,4) || '-' ||  SUBSTR(NVL(EMPRESA_FORMULARIO.DS_NR_FONE_FAX, MULTI_EMPRESAS.NR_FAX),6,4) NR_FAX_EMPRESA,
       SUBSTR(LPAD(NVL(EMPRESA_FORMULARIO.DS_CNPJ, MULTI_EMPRESAS.CD_CGC), 14, 0),1,2) || '.' ||
       SUBSTR(LPAD(NVL(EMPRESA_FORMULARIO.DS_CNPJ, MULTI_EMPRESAS.CD_CGC), 14, 0),3,3) || '.' ||
       SUBSTR(LPAD(NVL(EMPRESA_FORMULARIO.DS_CNPJ, MULTI_EMPRESAS.CD_CGC), 14, 0),6,3) || '/' ||
       SUBSTR(LPAD(NVL(EMPRESA_FORMULARIO.DS_CNPJ, MULTI_EMPRESAS.CD_CGC), 14, 0),9,4) || '-' ||
       SUBSTR(LPAD(NVL(EMPRESA_FORMULARIO.DS_CNPJ, MULTI_EMPRESAS.CD_CGC), 14, 0),13,2) NR_CGC,

       ATENDIME.HR_ALTA,
       ATENDIME.DT_ATENDIMENTO,

   AND MULTI_EMPRESAS.CD_MULTI_EMPRESA             = NOTA_FISCAL.CD_MULTI_EMPRESA
   AND NOTA_FISCAL.CD_NOTA_FISCAL                  = ITNOTA_FISCAL.CD_NOTA_FISCAL
   AND GRU_FAT.CD_GRU_FAT(+)                       = ITNOTA_FISCAL.CD_GRU_FAT
   AND SERVICO_NOTA_FISCAL.CD_SERVICO(+)           = ITNOTA_FISCAL.CD_SERVICO
   AND CONVENIO.CD_CONVENIO(+)                     = NOTA_FISCAL.CD_CONVENIO
   AND NOTA_FISCAL.CD_NOTA_FISCAL                  = :CAMPO_CD_NOTA_FISCAL
   AND NOTA_FISCAL.CD_FORMULARIO_NF                = FORMULARIO_NF.CD_FORMULARIO_NF
   AND CIDADE.CD_CIDADE                            = MULTI_EMPRESAS.CD_CIDADE
   AND NOTA_FISCAL.CD_ATENDIMENTO                  = ATENDIME.CD_ATENDIMENTO(+)
   AND ATENDIME.CD_PACIENTE                        = PACIENTE.CD_PACIENTE(+)
   AND ATENDIME.CD_ORI_ATE                         = ORI_ATE.CD_ORI_ATE(+)
   AND TIPO_LOGRADOURO.CD_TIPO_LOGRADOURO(+)       = NOTA_FISCAL.CD_TIPO_LOGRADOURO
   AND NOTA_FISCAL.TP_NOTA_FISCAL                  = 'N'
   AND EMPRESA_FORMULARIO.CD_EMPRESA_FORMULARIO(+) = NOTA_FISCAL.CD_FORMULARIO_NF
GROUP BY
       NOTA_FISCAL.CD_NOTA_FISCAL,
	   ITNOTA_FISCAL.CD_ITNOTA_FISCAL,
       DECODE(FNC_FFCV_SN_INFORMA_INTER(CONVENIO.CD_CONVENIO),'S',NULL,NOTA_FISCAL.NM_CLIENTE),
       DECODE(FNC_FFCV_SN_INFORMA_INTER(CONVENIO.CD_CONVENIO),'S',NULL,NOTA_FISCAL.NR_CGC_CPF),
       DECODE(FNC_FFCV_SN_INFORMA_INTER(CONVENIO.CD_CONVENIO),'S',NULL,TIPO_LOGRADOURO.DS_RESUMIDO || ' ' || NOTA_FISCAL.DS_ENDERECO || ',' || NOTA_FISCAL.NR_ENDERECO),

       DECODE(FNC_FFCV_SN_INFORMA_INTER(CONVENIO.CD_CONVENIO),'S',NULL,NOTA_FISCAL.NR_FONE),
       TO_CHAR(NOTA_FISCAL.DT_EMISSAO, 'DD/MM/YYYY' ),
       NOTA_FISCAL.NR_ID_NOTA_FISCAL,
       NOTA_FISCAL.VL_TOTAL_NOTA,
       NOTA_FISCAL.DS_OBSERVACOES,
        DECODE (
