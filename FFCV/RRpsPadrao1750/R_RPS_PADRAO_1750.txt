SELECT * FROM (
SELECT '1ª Via Cliente'                                                                                                                VIA,
       NOTA_FISCAL.CD_NOTA_FISCAL                                          CD_NOTA_FISCAL,
       NOTA_FISCAL.NM_CLIENTE || ' ' || NOTA_FISCAL.DS_ENDERECO || ' Nº do Contribuinte  ' || CONVENIO.NR_INSCRICAO_ESTADUAL  NM_CLIENTE,
       NOTA_FISCAL.NR_CGC_CPF                                                  NR_CGC_CPF,
       TIPO_LOGRADOURO.DS_RESUMIDO || ' ' || NOTA_FISCAL.DS_ENDERECO || ',' || NOTA_FISCAL.NR_ENDERECO                   DS_ENDERE,
       NOTA_FISCAL.NM_BAIRRO                  DS_BAIRRO,
       NOTA_FISCAL.NM_CIDADE NM_CIDADE ,
       NOTA_FISCAL.NM_UF NM_UF,
       NOTA_FISCAL.NR_CEP                                                          NR_CEP,
       NOTA_FISCAL.NR_INSCRICAO_ESTADUAL                            NR_INSCRICAO_ESTADUAL,
       NOTA_FISCAL.NR_INSCRICAO_MUNICIPAL                            NR_INSCRICAO_MUNICIPAL,
       NOTA_FISCAL.EMAIL                            EMAIL,
       NOTA_FISCAL.NR_FONE                                                        NR_FONE,
       to_char(NOTA_FISCAL.DT_EMISSAO, 'DD/MM/YYYY' )                          DT_EMISSAO,
       LPad(NR_ID_NOTA_FISCAL,6,'0') || '/' || TO_CHAR(NOTA_FISCAL.dt_emissao, 'yyyy') NR_ID_NOTA_FISCAL,
       NOTA_FISCAL.VL_TOTAL_NOTA                                            VL_TOTAL_NOTA,
       dbamv.FNC_FFCV_CD_REMESSA(NOTA_FISCAL.CD_NOTA_FISCAL)        CD_REMESSA,
       NOTA_FISCAL.DS_OBSERVACOES                                           DS_OBSERVACAO,
       NULL                                                                 CD_GRU_FAT,
       dbamv.FNC_FFCV_RETORNA_DS_CONF(ITNOTA_FISCAL.CD_GRU_FAT,GRU_FAT.DS_GRU_FAT,'R_RPS_GRUFAT_DESCR','FFCV',NOTA_FISCAL.Cd_Multi_Empresa) DS_GRU_FAT,
       sum(ITNOTA_FISCAL.VL_GRU_FAT)                                                VL_GRU_FAT,
       nvl(nota_fiscal.vl_desconto_nota,0)                               vl_desconto_nota,
       nvl(nota_fiscal.vl_acrescimo_nota,0)                             vl_acrescimo_nota,
       NVL( NOTA_FISCAL.DS_VL_DESCONTO, 'DESCONTO' )                          DS_DESCONTO,
       NVL( NOTA_FISCAL.DS_VL_ACRESCIMO, 'ACRESCIMO' )                       DS_ACRESCIMO,
       FORMULARIO_NF.CD_SERIE                                                    CD_SERIE,
       NOTA_FISCAL.CD_CONVENIO                                                  CD_CONVENIO,
       CONVENIO.TP_CONVENIO                                                        TP_CONVENIO,
       multi_empresas.ds_email_contato EMAIL_HOSPITAL,
       multi_empresas.ds_multi_empresa DS_MULTI_EMPRESA,
       multi_empresas.ds_endereco || '  ' || multi_empresas.nr_endereco  || '  ' || multi_empresas.NM_BAIRRO   DS_ENDERECO_EMPRESA,
       multi_empresas.NM_BAIRRO DS_BAIRRO_EMPRESA,
       multi_empresas.nr_ddd_empresa  NR_DDD_EMPRESA,
       substr(multi_empresas.nr_telefone_empresa,1,4) || '-' ||  substr(multi_empresas.nr_telefone_empresa,5,4)     NR_FONE_EMPRESA,
       cidade.nm_cidade || ' - ' || multi_empresas.cd_uf NM_CIDADE_EMPRESA,
       multi_empresas.nr_ddd_fax  NR_DDD_FAX_EMPRESA,
       substr(multi_empresas.nr_fax,1,4) || '-' ||  substr(multi_empresas.nr_fax,6,4) NR_FAX_EMPRESA,
       multi_empresas.cd_cgc NR_CGC,
       multi_empresas.cd_imun NR_INSCRICAO_MUNIC,
       multi_empresas.cd_iest NR_INSCRICAO_EST,
       ATENDIME.DT_ALTA,
       ATENDIME.HR_ALTA,
       ATENDIME.DT_ATENDIMENTO,
       ATENDIME.CD_ATENDIMENTO,
       ATENDIME.HR_ATENDIMENTO,
      ORI_ATE.DS_ORI_ATE,
      PACIENTE.NM_PACIENTE,
      convenio.ds_endereco_cobranca DS_ENDERECO_COBRANCA,
      convenio.ds_bairro_cobranca DS_BAIRRO_COBRANCA,
      convenio.ds_cidade_cobranca DS_CIDADE_COBRANCA,
      convenio.NR_CEP_COBRANCA NR_CEP_COBRANCA ,
      convenio.DS_UF_COBRANCA DS_UF_COBRANCA  ,
      Decode(Nvl(NOTA_FISCAL.nm_intermediario,'X'),'X',null,'Intermediário :'||NOTA_FISCAL.nm_intermediario||' CNPJ : '||NOTA_FISCAL.nr_cnpj_intermediario||Chr(10)||
      'Insc C.C.M.: '||NOTA_FISCAL.nr_inscr_munic_intermediario||' E-MAIL :'||NOTA_FISCAL.ds_email_intermediario) Dados_Intermediario,
       substr(dbamv.pkg_mv2000.le_configuracao('FFCV', 'DS_INTERMEDIARIO'), 1, 300) Observacao,
      dbamv.FNC_FFCV_NF_DT_VENCIMENTO(NOTA_FISCAL.CD_NOTA_FISCAL) DT_VENCIMENTO,
       dbamv.FNC_FFCV_NF_DESC_CONTA_PAGAMEN(NOTA_FISCAL.CD_MULTI_EMPRESA)                   DS_CONTAS_PAG,
       NOTA_FISCAL.TP_NOTA_FISCAL                               TP_NOTA_FISCAL,
       dbamv.FNC_FFCV_INDICE_USD(NOTA_FISCAL.DT_EMISSAO,NOTA_FISCAL.CD_MULTI_EMPRESA)        VL_CONVERSOR,
       dbamv.FNC_FFCV_QTD_CONTAS_REMESSA(NOTA_FISCAL.CD_NOTA_FISCAL)    QTD_CONTAS
FROM DBAMV.NOTA_FISCAL  ,
           DBAMV.ITNOTA_FISCAL,
           DBAMV.GRU_FAT    ,
           DBAMV.CONVENIO,
           DBAMV.FORMULARIO_NF,
           DBAMV.MULTI_EMPRESAS,
           DBAMV.CIDADE,
           DBAMV.EMPRESA_CONVENIO,
           dbamv.ATENDIME,
           DBAMV.PACIENTE,
           DBAMV.ORI_ATE,
           dbamv.tipo_logradouro,
           DBAMV.CON_REC
where Empresa_Convenio.Cd_Convenio      = Convenio.Cd_Convenio
   and NOTA_FISCAL.Cd_Multi_Empresa      = :CAMPO_CD_MULTI_EMPRESA
   and Empresa_Convenio.Cd_Multi_Empresa(+) = MULTI_EMPRESAS.CD_MULTI_EMPRESA
   and MULTI_EMPRESAS.CD_MULTI_EMPRESA   = NOTA_FISCAL.Cd_Multi_Empresa
   And NOTA_FISCAL.CD_NOTA_FISCAL        = ITNOTA_FISCAL.CD_NOTA_FISCAL
   AND GRU_FAT.CD_GRU_FAT                = ITNOTA_FISCAL.CD_GRU_FAT
   AND CONVENIO.CD_CONVENIO(+)           = NOTA_FISCAL.CD_CONVENIO
   AND NOTA_FISCAL.CD_NOTA_FISCAL        = :CAMPO_CD_NOTA_FISCAL
   AND NOTA_FISCAL.CD_FORMULARIO_NF      = FORMULARIO_NF.CD_FORMULARIO_NF
   AND CIDADE.CD_CIDADE                  = MULTI_EMPRESAS.CD_CIDADE
   AND NOTA_FISCAL.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO(+)
   AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE(+)
   AND ATENDIME.CD_ORI_ATE = ORI_ATE.CD_ORI_ATE(+)
   AND TIPO_LOGRADOURO.CD_TIPO_LOGRADOURO(+) = NOTA_FISCAL.CD_TIPO_LOGRADOURO
   AND NOTA_FISCAL.CD_NOTA_FISCAL        = CON_REC.CD_NOTA_FISCAL(+)
group by
       NOTA_FISCAL.CD_NOTA_FISCAL,
       NOTA_FISCAL.NM_CLIENTE || ' ' || NOTA_FISCAL.DS_ENDERECO || ' Nº do Contribuinte  ' || CONVENIO.NR_INSCRICAO_ESTADUAL,
       NOTA_FISCAL.NR_CGC_CPF,
       TIPO_LOGRADOURO.DS_RESUMIDO || ' ' || NOTA_FISCAL.DS_ENDERECO || ',' || NOTA_FISCAL.NR_ENDERECO,
       NOTA_FISCAL.NM_BAIRRO,
       NOTA_FISCAL.NM_CIDADE,
       NOTA_FISCAL.NM_UF,
       NOTA_FISCAL.NR_CEP,
       NOTA_FISCAL.NR_INSCRICAO_ESTADUAL,
       NOTA_FISCAL.NR_INSCRICAO_MUNICIPAL,
       NOTA_FISCAL.EMAIL,
       NOTA_FISCAL.NR_FONE,
       to_char(NOTA_FISCAL.DT_EMISSAO, 'DD/MM/YYYY' ),
       LPad(NR_ID_NOTA_FISCAL,6,'0') || '/' || TO_CHAR(NOTA_FISCAL.dt_emissao, 'yyyy'),
       NOTA_FISCAL.VL_TOTAL_NOTA,
       dbamv.FNC_FFCV_CD_REMESSA(NOTA_FISCAL.CD_NOTA_FISCAL),
       NOTA_FISCAL.DS_OBSERVACOES,
       dbamv.FNC_FFCV_RETORNA_DS_CONF(ITNOTA_FISCAL.CD_GRU_FAT,GRU_FAT.DS_GRU_FAT,'R_RPS_GRUFAT_DESCR','FFCV',NOTA_FISCAL.Cd_Multi_Empresa),
       nvl(nota_fiscal.vl_desconto_nota,0),
       nvl(nota_fiscal.vl_acrescimo_nota,0),
       multi_empresas.ds_email_contato,
       NVL( NOTA_FISCAL.DS_VL_DESCONTO, 'DESCONTO' ),
       NVL( NOTA_FISCAL.DS_VL_ACRESCIMO, 'ACRESCIMO' ),
       FORMULARIO_NF.CD_SERIE,
       NOTA_FISCAL.CD_CONVENIO,
       CONVENIO.TP_CONVENIO,
       ATENDIME.DT_ALTA,
       ATENDIME.HR_ALTA,
       ATENDIME.CD_ATENDIMENTO,
       ATENDIME.DT_ATENDIMENTO,
       ATENDIME.HR_ATENDIMENTO,
       ORI_ATE.DS_ORI_ATE,
       multi_empresas.ds_email_contato,
       multi_empresas.ds_multi_empresa,
       multi_empresas.ds_endereco || '  ' || multi_empresas.nr_endereco  || '  ' || multi_empresas.NM_BAIRRO  ,
       multi_empresas.NM_BAIRRO,
       multi_empresas.nr_ddd_empresa ,
       substr(multi_empresas.nr_telefone_empresa,1,4) || '-' ||  substr(multi_empresas.nr_telefone_empresa,5,4),
       cidade.nm_cidade || ' - ' || multi_empresas.cd_uf,
       multi_empresas.nr_ddd_fax,
       substr(multi_empresas.nr_fax,1,4) || '-' ||  substr(multi_empresas.nr_fax,6,4),
       multi_empresas.cd_cgc,
       multi_empresas.cd_imun,
       multi_empresas.cd_iest,
       PACIENTE.NM_PACIENTE,
       convenio.ds_endereco_cobranca ,
       convenio.ds_bairro_cobranca ,
       convenio.ds_cidade_cobranca ,
       convenio.NR_CEP_COBRANCA,
       convenio.DS_UF_COBRANCA ,
       Decode(Nvl(NOTA_FISCAL.nm_intermediario,'X'),'X',null,'Intermediário :'||NOTA_FISCAL.nm_intermediario||' CNPJ : '||NOTA_FISCAL.nr_cnpj_intermediario||Chr(10)||
      'Insc C.C.M.: '||NOTA_FISCAL.nr_inscr_munic_intermediario||' E-MAIL :'||NOTA_FISCAL.ds_email_intermediario),
       substr(dbamv.pkg_mv2000.le_configuracao('FFCV', 'DS_INTERMEDIARIO'), 1, 300),
      dbamv.FNC_FFCV_NF_DT_VENCIMENTO(NOTA_FISCAL.CD_NOTA_FISCAL) ,
       dbamv.FNC_FFCV_NF_DESC_CONTA_PAGAMEN(NOTA_FISCAL.CD_MULTI_EMPRESA),
       NOTA_FISCAL.TP_NOTA_FISCAL,
       dbamv.FNC_FFCV_INDICE_USD(NOTA_FISCAL.DT_EMISSAO,NOTA_FISCAL.CD_MULTI_EMPRESA),
       dbamv.FNC_FFCV_QTD_CONTAS_REMESSA(CON_REC.CD_REMESSA)
Order by ds_gru_fat
)
ORDER BY CD_NOTA_FISCAL, DS_GRU_FAT
