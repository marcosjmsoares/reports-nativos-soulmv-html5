--R_ATENDE_SEM_GUIA_CG__R_ATENDIME_Q_GUIAS_SEM_AUTORIZA
SELECT ATENDIME.CD_CONVENIO              CD_CONVENIO
             ,CONVENIO.NM_CONVENIO              NM_CONVENIO
             ,ATENDIME.CD_ATENDIMENTO       CD_ATENDIMENTO
             ,ATENDIME.CD_PACIENTE                CD_PACIENTE
             ,dbamv.pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(PACIENTE.CD_PACIENTE, 1)                NM_PACIENTE
             ,PRESTADOR.NM_MNEMONICO       NM_MNEMONICO
             ,ATENDIME.DT_ATENDIMENTO       DT_ATENDIMENTO
             ,ATENDIME.CD_LEITO                       CD_LEITO
             ,LEITO.DS_LEITO                               DS_LEITO
             ,ATENDIME.DT_ALTA                        DT_ALTA
             ,PRESTADOR.NM_PRESTADOR      NM_PRESTADOR
FROM   DBAMV.ATENDIME    ATENDIME
             ,DBAMV.PACIENTE    PACIENTE
             ,DBAMV.CONVENIO    CONVENIO
             ,DBAMV.LEITO       LEITO
             ,DBAMV.PRESTADOR   PRESTADOR
             ,dbamv.unid_int,
	     dbamv.empresa_convenio
WHERE  Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND        ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND        CONVENIO.TP_CONVENIO = 'C'
AND        ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
AND        ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
and         unid_int.cd_unid_int     =   leito.cd_unid_int
AND       ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
MINUS
SELECT GUIA.CD_CONVENIO                         CD_CONVENIO
              ,CONVENIO.NM_CONVENIO              NM_CONVENIO
              ,GUIA.CD_ATENDIMENTO                 CD_ATENDIMENTO
              ,GUIA.CD_PACIENTE                          CD_PACIENTE
              ,dbamv.pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(PACIENTE.CD_PACIENTE, 1)                NM_PACIENTE
              ,PRESTADOR.NM_MNEMONICO       NM_MNEMONICO
              ,ATENDIME.DT_ATENDIMENTO       DT_ATENDIMENTO
              ,LEITO.CD_LEITO                               CD_LEITO
              ,LEITO.DS_LEITO                               DS_LEITO
              ,ATENDIME.DT_ALTA                        DT_ALTA
              ,PRESTADOR.NM_PRESTADOR      NM_PRESTADOR
FROM    DBAMV.GUIA
             ,DBAMV.PACIENTE PACIENTE
             ,DBAMV.CONVENIO CONVENIO
             ,DBAMV.ATENDIME ATENDIME
             ,DBAMV.LEITO LEITO
             ,DBAMV.PRESTADOR PRESTADOR
            ,dbamv.unid_int,
	     dbamv.empresa_convenio
WHERE  Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and GUIA.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND        GUIA.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND        GUIA.DT_NEGACAO IS NULL
AND        GUIA.DT_SUSPENSAO IS NULL
AND        GUIA.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
AND        ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
AND        ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
AND        ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
and       unid_int.cd_unid_int     = leito.cd_unid_int
ORDER BY NM_CONVENIO, NM_PACIENTE, CD_ATENDIMENTO, NM_MNEMONICO
