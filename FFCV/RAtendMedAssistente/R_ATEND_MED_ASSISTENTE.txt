--R_ATEND_MED_ASSISTENTE
SELECT NVL(CD_PRESTADOR,0) CD_PRESTADOR,
             NM_PRESTADOR,
             COUNT(DISTINCT CD_ATENDIMENTO) INTERNACOES,
             SUM(VL_TOTAL_CONTA) VL_TOTAL_CONTA
FROM
(
SELECT PRES.CD_PRESTADOR,
              PRES.NM_PRESTADOR,
              ATEN.CD_ATENDIMENTO,
              NVL(REG_FAT.VL_TOTAL_CONTA,0) VL_TOTAL_CONTA,
              CONVENIO.CD_CONVENIO,
              CONVENIO.NM_CONVENIO
FROM   DBAMV.ATENDIME      ATEN,
             DBAMV.PRESTADOR   PRES,
             DBAMV.REG_FAT,
             DBAMV.CONVENIO,
             DBAMV.SERVICO,
             DBAMV.ORI_ATE,
	     dbamv.empresa_convenio
WHERE  Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and TRUNC(ATEN.DT_ATENDIMENTO)  BETWEEN  :CAMPO_DATA_INI  AND  
AND        ATEN.CD_PRESTADOR               =  PRES.CD_PRESTADOR(+)
AND       REG_FAT.CD_ATENDIMENTO(+) = ATEN.CD_ATENDIMENTO
AND       ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND       ATEN.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND       ATEN.CD_SERVICO = SERVICO.CD_SERVICO(+)
AND       ATEN.CD_ORI_ATE = ORI_ATE.CD_ORI_ATE
AND       CONVENIO.TP_CONVENIO <> 'H'
UNION ALL
SELECT PRES.CD_PRESTADOR,
              PRES.NM_PRESTADOR,
              ATEN.CD_ATENDIMENTO,
             NVL(ITREG_AMB.VL_TOTAL_CONTA,0) VL_TOTAL_CONTA,
             CONVENIO.CD_CONVENIO,
             CONVENIO.NM_CONVENIO
FROM   DBAMV.ATENDIME      ATEN,
             DBAMV.PRESTADOR   PRES,
             DBAMV.ITREG_AMB,
             DBAMV.CONVENIO,
             DBAMV.PRO_FAT,
             DBAMV.GRU_PRO,
             DBAMV.SERVICO,
             DBAMV.ORI_ATE	,
	     dbamv.empresa_convenio
WHERE  Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and TRUNC(ATEN.DT_ATENDIMENTO)  BETWEEN  :CAMPO_DATA_INI  AND  
AND       ATEN.CD_PRESTADOR                =     PRES.CD_PRESTADOR(+)
AND       ITREG_AMB.CD_ATENDIMENTO =    ATEN.CD_ATENDIMENTO
AND       ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND       ATEN.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND       ITREG_AMB.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
AND       PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
AND       ITREG_AMB.SN_PERTENCE_PACOTE = 'N'
AND       NVL(ITREG_AMB.TP_PAGAMENTO, 'X') <> 'C'
AND       CONVENIO.TP_CONVENIO <> 'H'
AND       ATEN.CD_SERVICO = SERVICO.CD_SERVICO(+)
AND       ATEN.CD_ORI_ATE = ORI_ATE.CD_ORI_ATE
:CAMPO_CF_CONTA_FECHADA_AMB
)
GROUP BY  NM_PRESTADOR,
                    CD_PRESTADOR
