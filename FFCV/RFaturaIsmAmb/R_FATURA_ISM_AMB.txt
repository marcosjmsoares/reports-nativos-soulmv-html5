--R_FATURA_ISM_AMB
SELECT ATENDIME.CD_ATENDIMENTO
      ,CARTEIRA.NM_TITULAR
      ,CARTEIRA.NR_CARTEIRA MATRICULA
      ,PACIENTE.NM_PACIENTE
      ,DECODE(PACIENTE.TP_SEXO,'M','(X) M','( ) M')  SEXO_MAS
      ,DECODE(PACIENTE.TP_SEXO,'F' ,'(X) F','( ) F')    SEXO_FEM
      ,PACIENTE.DT_NASCIMENTO
      ,DT_ATENDIMENTO DATA
      ,ATENDIME.DT_ALTA
      ,ATENDIME.CD_CID
      ,CID.DS_CID
      ,ITREG_AMB.CD_REG_AMB CD_REG_FAT
      ,HOSPITAL.DS_RAZAO
  FROM DBAMV.ATENDIME,
       DBAMV.PACIENTE,
       DBAMV.CARTEIRA,
       DBAMV.CID,
       DBAMV.ITREG_AMB,
       DBAMV.HOSPITAL
 WHERE ATENDIME.CD_PACIENTE               =   PACIENTE.CD_PACIENTE
   AND     ATENDIME.CD_PACIENTE                =   CARTEIRA.CD_PACIENTE(+)
   AND     ATENDIME.CD_CONVENIO              =   CARTEIRA.CD_CONVENIO(+)
   AND     ATENDIME.CD_CON_PLA                 =   CARTEIRA.CD_CON_PLA(+)
   AND     CID.CD_CID(+)                                   =   ATENDIME.CD_CID
   AND     ITREG_AMB.CD_ATENDIMENTO         =   ATENDIME.CD_ATENDIMENTO
   AND     ATENDIME.CD_ATENDIMENTO      =  :CAMPO_ATENDIME
   AND     ITREG_AMB.CD_REG_AMB                  =   TO_NUMBER( :CAMPO_CD_REG_FAT)
   and Hospital.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY ATENDIME.CD_ATENDIMENTO
      ,CARTEIRA.NM_TITULAR
      ,CARTEIRA.NR_CARTEIRA
      ,PACIENTE.NM_PACIENTE
      ,DECODE(PACIENTE.TP_SEXO,'M','(X) M','( ) M')
      ,DECODE(PACIENTE.TP_SEXO,'F' ,'(X) F','( ) F')
      ,PACIENTE.DT_NASCIMENTO
      ,DT_ATENDIMENTO
      ,ATENDIME.DT_ALTA
      ,ATENDIME.CD_CID
      ,CID.DS_CID
      ,ITREG_AMB.CD_REG_AMB
      ,HOSPITAL.DS_RAZAO

SELECT * FROM (

      ,ITREG_AMB.CD_REG_AMB  CD_REG_AMB
      ,DECODE(LANCMIN.CD_LANC, ITREG_AMB.CD_LANCAMENTO, '1', '2') ORDEM
      ,NVL( CIRURGIAO.CD_PRESTADOR, '0' ) CD_CIRURGIAO
      ,NVL( AUXILIAR1.CD_PRESTADOR, '0' ) CD_AUXILIAR1
      ,NVL( AUXILIAR2.CD_PRESTADOR, '0' ) CD_AUXILIAR2
      ,NVL( AUXILIAR3.CD_PRESTADOR, '0' ) CD_AUXILIAR3
      ,NVL( ANESTESIA.CD_PRESTADOR, '0' ) CD_ANESTESIA

      ,GUIA.NR_GUIA

      ,ITREG_AMB.cd_lancamento
      ,ITREG_AMB.CD_ATENDIMENTO CD_ATEND1

       DBAMV.ITREG_AMB,
       DBAMV.PRO_FAT,
       DBAMV.GUIA,
       DBAMV.GRU_FAT
      ,(SELECT MIN( I.CD_LANCAMENTO ) CD_LANC
          FROM  DBAMV.ITREG_AMB I,  DBAMV.GRU_FAT G
         WHERE G.CD_GRU_FAT(+)  =  I.CD_GRU_FAT
           AND NVL( G.TP_GRU_FAT, '  ' )  =   'SP'
           AND I.CD_ATENDIMENTO =  :CAMPO_ATENDIME
           AND I.CD_REG_AMB     =  :CAMPO_CD_REG_FAT ) LANCMIN
      ,(SELECT PR.DS_CODIGO_CONSELHO CD_PRESTADOR,
               IT.CD_LANCAMENTO
          FROM DBAMV.ITREG_AMB IT,
               DBAMV.PRESTADOR PR, DBAMV.ATI_MED AM
         WHERE IT.CD_REG_AMB = :CAMPO_CD_REG_FAT
	       AND PR.CD_PRESTADOR     = IT.CD_PRESTADOR
	       AND AM.CD_ATI_MED       = IT.CD_ATI_MED
           AND IT.CD_PRO_FAT      not in ('00020010','00010015')
           AND AM.CD_ATI_MED      IN ('01', '07')  ) CIRURGIAO
      ,(SELECT PR.DS_CODIGO_CONSELHO CD_PRESTADOR,
               IT.CD_LANCAMENTO
          FROM DBAMV.ITREG_AMB IT,
               DBAMV.PRESTADOR PR, DBAMV.ATI_MED AM
         WHERE IT.CD_REG_AMB = :CAMPO_CD_REG_FAT
		   AND PR.CD_PRESTADOR     = IT.CD_PRESTADOR
		   AND AM.CD_ATI_MED       = IT.CD_ATI_MED
           AND IT.CD_PRO_FAT      not in ('00020010','00010015')
           AND AM.CD_ATI_MED       = '02'  ) AUXILIAR1
      ,(SELECT PR.DS_CODIGO_CONSELHO CD_PRESTADOR,
               IT.CD_LANCAMENTO
          FROM DBAMV.ITREG_AMB IT,
               DBAMV.PRESTADOR PR, DBAMV.ATI_MED AM
         WHERE IT.CD_REG_AMB = :CAMPO_CD_REG_FAT
	       AND PR.CD_PRESTADOR     = IT.CD_PRESTADOR

               IT.CD_LANCAMENTO
          FROM DBAMV.ITREG_AMB IT,
               DBAMV.PRESTADOR PR, DBAMV.ATI_MED AM
         WHERE IT.CD_REG_AMB = :CAMPO_CD_REG_FAT
	       AND PR.CD_PRESTADOR     = IT.CD_PRESTADOR
	       AND AM.CD_ATI_MED       = IT.CD_ATI_MED

and  ITREG_AMB.CD_ATENDIMENTO       = ATENDIME.CD_ATENDIMENTO
    AND PRO_FAT.CD_PRO_FAT           = ITREG_AMB.CD_PRO_FAT
	AND GRU_FAT.CD_GRU_FAT(+)        = ITREG_AMB.CD_GRU_FAT
    AND NVL( GRU_FAT.TP_GRU_FAT, '  ' )  =  'SP'

      ,' ' CD_AUXILIAR3
      ,' ' CD_ANESTESIA
      ,' ' CD_PRO_FAT
      ,' ' DS_PRO_FAT
      ,' ' NR_GUIA
      ,' ' VIA_DE_ACESSO

      ,0   cd_lancamento

            DBAMV.ITREG_AMB

     AND ATENDIME.CD_ATENDIMENTO      = :CAMPO_ATENDIME
      AND ITREG_AMB.CD_REG_AMB           = TO_NUMBER( :CAMPO_CD_REG_FAT)
      AND NOT EXISTS ( SELECT IT.CD_GRU_FAT
                                          FROM DBAMV.ITREG_AMB IT,
                                                     DBAMV.GRU_FAT GR
                                      WHERE IT.CD_REG_AMB     = TO_NUMBER( :CAMPO_CD_REG_FAT)
                                            AND IT.CD_ATENDIMENTO = :CAMPO_ATENDIME
                                            AND IT.CD_GRU_FAT     = GR.CD_GRU_FAT
                                             AND GR.TP_GRU_FAT     = 'SP' )
 GROUP BY ATENDIME.CD_ATENDIMENTO
UNION
SELECT ITREG_AMB.CD_ATENDIMENTO
      ,ITREG_AMB.CD_REG_AMB
      ,DECODE(LANCMIN.CD_LANC, ITREG_AMB.CD_LANCAMENTO, '1', '2') ORDEM
      ,CIRURGIAO.CD_PRESTADOR CD_CIRURGIAO
      ,' ' CD_AUXILIAR1
      ,' ' CD_AUXILIAR2
      ,' ' CD_AUXILIAR3
      ,' ' CD_ANESTESIA
      ,PRO_FAT.CD_PRO_FAT
      ,PRO_FAT.DS_PRO_FAT
      ,GUIA.NR_GUIA
      ,DECODE( ITREG_AMB.VL_PERCENTUAL_MULTIPLA, 100, 'PRINCIPAL', 70, 'DVA', 50, 'MVA', '   ' ) VIA_DE_ACESSO
      ,SUM(ITREG_AMB.qt_lancamento)   QT_LANCAMENTO
      ,0 cd_lancamento
      ,ITREG_AMB.CD_ATENDIMENTO CD_ATEND1
 FROM DBAMV.ATENDIME,
      DBAMV.ITREG_AMB,
      DBAMV.PRO_FAT,
      DBAMV.GUIA,
      DBAMV.GRU_FAT
     ,(SELECT MIN( I.CD_LANCAMENTO ) CD_LANC
         FROM  DBAMV.ITREG_AMB I,  DBAMV.GRU_FAT G
        WHERE G.CD_GRU_FAT(+)  =  I.CD_GRU_FAT
          AND NVL( G.TP_GRU_FAT, '  ' )  =   'SP'
          AND I.CD_ATENDIMENTO =  :CAMPO_ATENDIME
          AND I.CD_REG_AMB     =  :CAMPO_CD_REG_FAT ) LANCMIN
     ,(SELECT PR.DS_CODIGO_CONSELHO CD_PRESTADOR,
              IT.CD_LANCAMENTO
         FROM DBAMV.ITREG_AMB IT
              ,DBAMV.PRESTADOR PR, DBAMV.ATI_MED AM
        WHERE IT.CD_REG_AMB = :CAMPO_CD_REG_FAT
	      AND PR.CD_PRESTADOR     = IT.CD_PRESTADOR
	      AND AM.CD_ATI_MED       = IT.CD_ATI_MED
          AND IT.CD_PRO_FAT  NOT in ('00010016', '00010065')
          AND AM.CD_ATI_MED      IN ('01', '07')  ) CIRURGIAO
where guia.cd_atendimento = atendime.cd_atendimento
and  ITREG_AMB.CD_ATENDIMENTO       = ATENDIME.CD_ATENDIMENTO
   AND PRO_FAT.CD_PRO_FAT           = ITREG_AMB.CD_PRO_FAT
   AND GRU_FAT.CD_GRU_FAT(+)        = ITREG_AMB.CD_GRU_FAT
   AND NVL( GRU_FAT.TP_GRU_FAT, '  ' )  =  'SP'
   AND ATENDIME.CD_ATENDIMENTO      = :CAMPO_ATENDIME
   AND ITREG_AMB.CD_REG_AMB           = TO_NUMBER( :CAMPO_CD_REG_FAT )
   and guia.cd_atendimento(+)       = atendime.cd_atendimento
   and guia.cd_paciente(+)          = atendime.cd_paciente
   and guia.cd_convenio(+)          = atendime.cd_convenio
   and nvl(guia.tp_guia,'C')       IN ('C','P')
   AND ITREG_AMB.CD_PRO_FAT        in ('00020010','00010015')
   AND ITREG_AMB.CD_LANCAMENTO      = CIRURGIAO.CD_LANCAMENTO(+)
 GROUP BY ITREG_AMB.CD_ATENDIMENTO
         ,ITREG_AMB.CD_REG_AMB
         ,DECODE(LANCMIN.CD_LANC, ITREG_AMB.CD_LANCAMENTO, '1', '2')
         ,CIRURGIAO.CD_PRESTADOR
         ,PRO_FAT.CD_PRO_FAT
         ,PRO_FAT.DS_PRO_FAT
         ,GUIA.NR_GUIA
         ,DECODE( ITREG_AMB.VL_PERCENTUAL_MULTIPLA, 100, 'PRINCIPAL', 70, 'DVA', 50, 'MVA', '   ' )
         ,ITREG_AMB.CD_ATENDIMENTO
 ORDER BY 1,2,3,4,5,6,7,8,12 desc,9
)
ORDER BY ORDEM,CD_ATENDIMENTO,CD_REG_AMB,
CD_CIRURGIAO,CD_AUXILIAR1,CD_AUXILIAR2,CD_AUXILIAR3,CD_ANESTESIA
