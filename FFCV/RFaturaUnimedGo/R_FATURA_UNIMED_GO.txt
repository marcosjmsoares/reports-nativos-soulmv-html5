SELECT DISTINCT ATENDIME.CD_ATENDIMENTO

       ,NVL( PRO_FAT.DS_PRO_FAT, 'NAO ENCONTRADO' )     DS_PRO_FAT
       ,NVL( PRES_CON.CD_PRESTADOR_CONVENIADO, NVL( PRESTADOR.DS_CODIGO_CONSELHO, '      ' ) ) DS_CODIGO_CONSELHO
       ,NVL( PRESTADOR.NM_PRESTADOR, 'NAO ENCONTRADO' ) NM_PRESTADOR
       ,NVL( TIP_ACOM.DS_TIP_ACOM, 'NAO INFORMADO' )    DS_TIP_ACOM
  FROM DBAMV.ATENDIME,
       DBAMV.PACIENTE,
       DBAMV.CARTEIRA,
       DBAMV.CID,
        ( SELECT DISTINCT PRES_CON.CD_CONVENIO, PRES_CON.CD_PRESTADOR, PRES_CON.SN_PAGA_PELO_CONVENIO, PRES_CON.CD_PRESTADOR_CONVENIADO
	      FROM 	DBAMV.PRES_CON, DBAMV.REG_FAT
				, DBAMV.ITREG_FAT 
	      WHERE (PRES_CON.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA OR PRES_CON.CD_MULTI_EMPRESA IS NULL  )
			AND REG_FAT.CD_REG_FAT      = :CAMPO_CD_REG_FAT
			AND REG_FAT.CD_ATENDIMENTO  = :CAMPO_ATENDIME
			AND PRES_CON.CD_CONVENIO	= REG_FAT.CD_CONVENIO
			AND REG_FAT.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
			AND ITREG_FAT.CD_PRESTADOR = PRES_CON.CD_PRESTADOR
			AND (PRES_CON.CD_CON_PLA = REG_FAT.CD_CON_PLA OR PRES_CON.CD_CON_PLA IS NULL)
			AND (PRES_CON.CD_REGRA = REG_FAT.CD_REGRA OR PRES_CON.CD_REGRA IS NULL)
			AND (PRES_CON.CD_GRU_PRO  =   (SELECT PRO_FAT.CD_GRU_PRO
										FROM DBAMV.PRO_FAT
									   WHERE PRO_FAT.CD_PRO_FAT = ITREG_FAT.CD_PRO_FAT )   OR PRES_CON.CD_GRU_PRO IS NULL)
			AND (PRES_CON.CD_PRO_FAT  = ITREG_FAT.CD_PRO_FAT OR PRES_CON.CD_PRO_FAT IS NULL)
		  )   PRES_CON
          ,
       DBAMV.REG_FAT,
       DBAMV.PRO_FAT,
       DBAMV.TIP_ACOM,
       DBAMV.PRESTADOR,
       DBAMV.HOSPITAL
 WHERE ATENDIME.CD_PACIENTE    = PACIENTE.CD_PACIENTE
   AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   AND ATENDIME.CD_PACIENTE    = CARTEIRA.CD_PACIENTE(+)
   AND ATENDIME.CD_CONVENIO    = CARTEIRA.CD_CONVENIO(+)
   AND ATENDIME.CD_CON_PLA     = CARTEIRA.CD_CON_PLA(+)
   AND CID.CD_CID(+)           = ATENDIME.CD_CID
   AND ATENDIME.CD_ATENDIMENTO = REG_FAT.CD_ATENDIMENTO
   AND PRO_FAT.CD_PRO_FAT(+)   = REG_FAT.CD_PRO_FAT_SOLICITADO
   AND PRESTADOR.CD_PRESTADOR(+) = ATENDIME.CD_PRESTADOR
   AND TIP_ACOM.CD_TIP_ACOM(+) = ATENDIME.CD_TIP_ACOM
   AND PRES_CON.CD_CONVENIO(+) = ATENDIME.CD_CONVENIO
   AND PRES_CON.CD_PRESTADOR(+)= ATENDIME.CD_PRESTADOR
   AND ATENDIME.CD_ATENDIMENTO = :CAMPO_ATENDIME
   AND REG_FAT.CD_REG_FAT = :CAMPO_CD_REG_FAT
   and Hospital.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA

       AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
       AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
       AND ATENDIME.CD_ATENDIMENTO     = :CAMPO_ATENDIME
       AND REG_FAT.CD_REG_FAT                 = :CAMPO_CD_REG_FAT
       AND REG_FAT.CD_REG_FAT                 = ITREG_FAT.CD_REG_FAT
       AND PRESTADOR.CD_PRESTADOR(+) = ITREG_FAT.CD_PRESTADOR
       AND PRES_CON.CD_PRESTADOR        = ITREG_FAT.CD_PRESTADOR
       AND PRES_CON.CD_CONVENIO            = ATENDIME.CD_CONVENIO
       AND PRO_FAT.CD_PRO_FAT                  = ITREG_FAT.CD_PRO_FAT
       AND GRU_PRO.CD_GRU_PRO               = PRO_FAT.CD_GRU_PRO
       AND GRU_PRO.TP_GRU_PRO               = 'SP'
       AND ITREG_FAT.CD_PRO_FAT             <> '00040010'    
       AND ITREG_FAT.CD_PRESTADOR        IS NOT NULL
GROUP BY ATENDIME.CD_ATENDIMENTO
                 ,REG_FAT.CD_REG_FAT
                 ,PRES_CON.CD_PRESTADOR_CONVENIADO
                 ,ITREG_FAT.CD_PRO_FAT
                 ,ITREG_FAT.CD_ATI_MED
UNION
SELECT ATENDIME.CD_ATENDIMENTO
             ,REG_FAT.CD_REG_FAT
             ,NVL( PRESTADOR.DS_CODIGO_CONSELHO, 'NAO INFORMADO' ) CD_PRESTADOR
             ,DECODE( ITREG_FAT.CD_PRO_FAT, '00010065', '00020010', '00010016', '00020010', ITREG_FAT.CD_PRO_FAT )     CD_PRO_FAT,
              ITREG_FAT.CD_PRO_FAT   PRO_FAT_ORIG
             ,ITREG_FAT.CD_ATI_MED
             ,SUM( ITREG_FAT.QT_LANCAMENTO ) QTDE
  FROM DBAMV.ATENDIME,
             DBAMV.REG_FAT,
             DBAMV.PRO_FAT,
             DBAMV.GRU_PRO,
             DBAMV.ITREG_FAT,
             DBAMV.PRESTADOR
 WHERE ATENDIME.CD_ATENDIMENTO     = REG_FAT.CD_ATENDIMENTO
       AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
       AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
       AND ATENDIME.CD_ATENDIMENTO     = :CAMPO_ATENDIME
       AND REG_FAT.CD_REG_FAT                 = :CAMPO_CD_REG_FAT
       AND REG_FAT.CD_REG_FAT                 = ITREG_FAT.CD_REG_FAT
       AND PRESTADOR.CD_PRESTADOR(+) = ITREG_FAT.CD_PRESTADOR
       AND NOT EXISTS (SELECT P.CD_PRESTADOR
                         FROM DBAMV.PRES_CON P
                        WHERE P.CD_CONVENIO = ATENDIME.CD_CONVENIO
                          and (p.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA or p.cd_multi_empresa is null  )  
                          AND P.CD_PRESTADOR = ITREG_FAT.CD_PRESTADOR)
       AND PRO_FAT.CD_PRO_FAT                  = ITREG_FAT.CD_PRO_FAT
       AND GRU_PRO.CD_GRU_PRO               = PRO_FAT.CD_GRU_PRO
       AND GRU_PRO.TP_GRU_PRO               = 'SP'
       AND ITREG_FAT.CD_PRO_FAT             <> '00040010'    
       AND ITREG_FAT.CD_PRESTADOR        IS NOT NULL
GROUP BY ATENDIME.CD_ATENDIMENTO
                 , REG_FAT.CD_REG_FAT
                 , NVL( PRESTADOR.DS_CODIGO_CONSELHO, 'NAO INFORMADO' )
                 ,ITREG_FAT.CD_PRO_FAT
                 ,ITREG_FAT.CD_ATI_MED
order by 5 desc, 7 desc

--R_FATURA_UNIMED_GO_R_TIPO_AMB1_Q_TIPO_AMB
SELECT * FROM (
SELECT  DS_GRU_FAT,
                DS_CODIGO_COBRANCA,
                QTD,
                CD_ATENDIMENTO,
                CD_REG_FAT
    FROM  (
SELECT DECODE( COD_PRO.DS_NOME_COBRANCA,
                                NULL, PRO_FAT.DS_PRO_FAT,
                                           COD_PRO.DS_NOME_COBRANCA )     DS_GRU_FAT
     , DECODE( COD_PRO.DS_CODIGO_COBRANCA,
                        NULL, PRO_FAT.CD_PRO_FAT,
                                   COD_PRO.DS_CODIGO_COBRANCA )  DS_CODIGO_COBRANCA
     , SUM( decode( itreg_fat.cd_pro_fat, '01000014', (itreg_fat.qt_lancamento * (itreg_fat.vl_percentual_multipla / 100)),
                                                                '01000013', (itreg_fat.qt_lancamento * (itreg_fat.vl_percentual_multipla / 100)),
                                                                '01000015', (itreg_fat.qt_lancamento * (itreg_fat.vl_percentual_multipla / 100)),
                                                                '01000017', (itreg_fat.qt_lancamento * (itreg_fat.vl_percentual_multipla / 100)),
                                                                '01000018', (itreg_fat.qt_lancamento * (itreg_fat.vl_percentual_multipla / 100)),
                                                                '01000020', (itreg_fat.qt_lancamento * (itreg_fat.vl_percentual_multipla / 100)),
                                                                '01000021', (itreg_fat.qt_lancamento * (itreg_fat.vl_percentual_multipla / 100)),
                                                                '01000165', (itreg_fat.qt_lancamento * (itreg_fat.vl_percentual_multipla / 100)),
                                                                '01000298', (itreg_fat.qt_lancamento * (itreg_fat.vl_percentual_multipla / 100)),
                                                                                    ITREG_FAT.QT_LANCAMENTO ) )  QTD     
     , MAX( REG_FAT.CD_ATENDIMENTO )   CD_ATENDIMENTO
     , MAX( REG_FAT.CD_REG_FAT )    CD_REG_FAT
  FROM DBAMV.REG_FAT
      ,DBAMV.ITREG_FAT
      ,DBAMV.PRO_FAT
      ,DBAMV.GRU_FAT
      ,(select distinct C.cd_pro_fat, C.ds_nome_cobranca, C.ds_codigo_cobranca
       from   dbamv.reg_fat R
             ,dbamv.cod_pro C
             ,dbamv.atendime A
       where ( C.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA or C.Cd_Multi_Empresa Is Null )
         and  R.cd_reg_fat  = :CAMPO_CD_REG_FAT
         and  R.cd_atendimento = :CAMPO_ATENDIME
         and  C.cd_convenio = R.cd_convenio
         and  A.cd_atendimento = R.cd_atendimento
         AND A.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
)   COD_PRO
 WHERE CD_ATENDIMENTO       = :CAMPO_ATENDIME
   AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   AND REG_FAT.CD_REG_FAT   = :CAMPO_CD_REG_FAT
   AND REG_FAT.CD_REG_FAT   = ITREG_FAT.CD_REG_FAT
   AND ITREG_FAT.CD_GRU_FAT = GRU_FAT.CD_GRU_FAT
   AND ITREG_FAT.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
   AND COD_PRO.cd_pro_fat (+)  = ITREG_FAT.cd_pro_fat
   AND ( GRU_FAT.TP_GRU_FAT = 'SD'    or    ITREG_FAT.CD_PRO_FAT = '00040010'  )  
GROUP BY DECODE( COD_PRO.DS_NOME_COBRANCA,
                    NULL, PRO_FAT.DS_PRO_FAT,
                   COD_PRO.DS_NOME_COBRANCA )
                 , DECODE( COD_PRO.DS_CODIGO_COBRANCA,
                   NULL, PRO_FAT.CD_PRO_FAT,
                   COD_PRO.DS_CODIGO_COBRANCA )
)
ORDER BY 2
)
WHERE CD_ATENDIMENTO = :CAMPO_CD_ATENDIMENTO
