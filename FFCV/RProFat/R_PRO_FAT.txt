--R_PRO_FAT
SELECT * FROM (
SELECT  PRO_FAT.CD_PRO_FAT  CODIGO
               ,UPPER(PRO_FAT.DS_PRO_FAT) DESCRICAO
               ,PRO_FAT.DS_UNIDADE
               ,PRO_FAT.CD_GRU_PRO
               ,GRU_PRO.DS_GRU_PRO
               ,PRO_FAT.TP_SEXO
               ,PRO_FAT.CD_POR_ANE
               ,POR_ANE.DS_POR_ANE
               ,PRO_FAT.NR_AUXILIAR
               ,PRO_FAT.NR_INCIDENCIAS
               ,PRO_FAT.VL_FILME
FROM  dbamv.PRO_FAT
           ,dbamv.GRU_PRO
           ,dbamv.POR_ANE
WHERE  GRU_PRO.CD_GRU_PRO  =  PRO_FAT.CD_GRU_PRO
      AND  PRO_FAT.CD_POR_ANE =  POR_ANE.CD_POR_ANE(+)
      AND  PRO_FAT.SN_ATIVO = 'S'
     AND   PRO_FAT.CD_SUS is null
:CAMPO_CF_COND_GRUPO
)
ORDER BY CD_GRU_PRO,DS_GRU_PRO, :CAMPO_ORDEM
