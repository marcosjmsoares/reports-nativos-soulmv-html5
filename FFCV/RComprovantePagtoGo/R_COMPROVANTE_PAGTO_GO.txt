--R_COMPROVANTE_PAGTO_GO
SELECT ATENDIME.CD_ATENDIMENTO
      ,CARTEIRA.NM_TITULAR
      ,CARTEIRA.NR_CARTEIRA MATRICULA
      ,PACIENTE.NM_PACIENTE
      ,DECODE(PACIENTE.TP_SEXO,'M','Masculino','Feminino') SEXO_MAS
      ,PACIENTE.DT_NASCIMENTO
      ,DT_ATENDIMENTO DATA
      ,ATENDIME.DT_ALTA
      ,NVL( GUIA.CD_GUIA, 0 )  CD_GUIA
      ,NVL( GUIA.NR_GUIA, 0 )  NR_GUIA
      ,PRO_FAT.CD_PRO_FAT
      ,PRO_FAT.DS_PRO_FAT
      ,ATI_MED.DS_ATI_MED||': CRM-' ||NVL(PRESTADOR.DS_CODIGO_CONSELHO,'N/I')||'  NOME-'||PRESTADOR.NM_PRESTADOR NM_PRESTADOR
      ,nvl(itlan_med.vl_ato*itreg_fat.qt_lancamento,ITREG_FAT.VL_TOTAL_CONTA) valor_perc
  FROM DBAMV.ATENDIME,
       DBAMV.PACIENTE,
       DBAMV.CARTEIRA,
       DBAMV.REG_FAT,
       DBAMV.ITREG_FAT,
       DBAMV.PRESTADOR,
       DBAMV.PRO_FAT,
       DBAMV.GRU_FAT,
       DBAMV.GUIA,
       DBAMV.ITLAN_MED,
       DBAMV.ATI_MED
 WHERE ATENDIME.CD_PACIENTE               = PACIENTE.CD_PACIENTE
   AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   AND     ATENDIME.CD_PACIENTE               = CARTEIRA.CD_PACIENTE(+)
   AND     ATENDIME.CD_CONVENIO              = CARTEIRA.CD_CONVENIO(+)
   AND     ATENDIME.CD_CON_PLA                = CARTEIRA.CD_CON_PLA(+)
   AND     REG_FAT.CD_ATENDIMENTO         = ATENDIME.CD_ATENDIMENTO
   AND     REG_FAT.CD_REG_FAT                   = ITREG_FAT.CD_REG_FAT
   AND     PRO_FAT.CD_PRO_FAT                   = ITREG_FAT.CD_PRO_FAT
   AND     GRU_FAT.CD_GRU_FAT(+)              = ITREG_FAT.CD_GRU_FAT
   AND     ITLAN_MED.CD_REG_FAT(+)           = ITREG_FAT.CD_REG_FAT
   AND     ITLAN_MED.CD_LANCAMENTO(+)  = ITREG_FAT.CD_LANCAMENTO
   AND     PRESTADOR.CD_PRESTADOR       = NVL(ITLAN_MED.CD_PRESTADOR,ITREG_FAT.CD_PRESTADOR)
   AND     NVL( GRU_FAT.TP_GRU_FAT, '  ' )  = 'SP'
   AND     ATI_MED.CD_ATI_MED                    = NVL(ITLAN_MED.CD_ATI_MED,ITREG_FAT.CD_ATI_MED)
   AND     ATENDIME.CD_ATENDIMENTO      = :CAMPO_ATENDIME
   AND     REG_FAT.CD_REG_FAT                = :CAMPO_CD_REG_FAT
   AND     GUIA.CD_ATENDIMENTO(+)          = ATENDIME.CD_ATENDIMENTO
   AND     GUIA.CD_CONVENIO(+)                  = ATENDIME.CD_CONVENIO
   AND     NVL( GUIA.TP_GUIA, 'I' )                = 'I'
  AND     nvl(itlan_med.tp_pagamento, NVL( ITREG_FAT.TP_PAGAMENTO, 'P' )) = 'C'
union
SELECT ATENDIME.CD_ATENDIMENTO
      ,CARTEIRA.NM_TITULAR
      ,CARTEIRA.NR_CARTEIRA MATRICULA
      ,PACIENTE.NM_PACIENTE
      ,DECODE(PACIENTE.TP_SEXO,'M','Masculino','Feminino') SEXO_MAS
      ,PACIENTE.DT_NASCIMENTO
      ,DT_ATENDIMENTO DATA
      ,ATENDIME.DT_ALTA
      ,NVL( GUIA.CD_GUIA, 0 )  CD_GUIA
      ,NVL( GUIA.NR_GUIA, 0 )  NR_GUIA
      ,PRO_FAT.CD_PRO_FAT
      ,PRO_FAT.DS_PRO_FAT
      ,ATI_MED.DS_ATI_MED||': CRM-' ||NVL(PRESTADOR.DS_CODIGO_CONSELHO,'N/I')||'  NOME-'||PRESTADOR.NM_PRESTADOR NM_PRESTADOR
      ,ITREG_AMB.VL_TOTAL_CONTA * nvl(vl_percentual_pago/100,1) valor_perc
  FROM DBAMV.ATENDIME,
       DBAMV.PACIENTE,
       DBAMV.CARTEIRA,
       DBAMV.ITREG_AMB,
       DBAMV.PRESTADOR,
       DBAMV.PRO_FAT,
       DBAMV.GRU_FAT,
       DBAMV.GUIA,
       DBAMV.ATI_MED
where guia.cd_atendimento = atendime.cd_atendimento
   and atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
   and  ATENDIME.CD_PACIENTE               = PACIENTE.CD_PACIENTE
   AND     ATENDIME.CD_PACIENTE               = CARTEIRA.CD_PACIENTE(+)
   AND     ATENDIME.CD_CONVENIO              = CARTEIRA.CD_CONVENIO(+)
   AND     ATENDIME.CD_CON_PLA                = CARTEIRA.CD_CON_PLA(+)
   AND     ITREG_AMB.CD_ATENDIMENTO         = ATENDIME.CD_ATENDIMENTO
   AND     PRO_FAT.CD_PRO_FAT                   = ITREG_AMB.CD_PRO_FAT
   AND     GRU_FAT.CD_GRU_FAT(+)              = ITREG_AMB.CD_GRU_FAT
   AND     PRESTADOR.CD_PRESTADOR(+)       = ITREG_AMB.CD_PRESTADOR
   AND     NVL( GRU_FAT.TP_GRU_FAT, '  ' )  = 'SP'
   AND     ATI_MED.CD_ATI_MED(+)                    = ITREG_AMB.CD_ATI_MED
   AND     ATENDIME.CD_ATENDIMENTO      = :CAMPO_ATENDIME
   AND     ITREG_AMB.CD_REG_AMB                = :CAMPO_CD_REG_FAT
   AND     GUIA.CD_ATENDIMENTO(+)          = ATENDIME.CD_ATENDIMENTO
   AND     GUIA.CD_CONVENIO(+)                  = ATENDIME.CD_CONVENIO
   AND     NVL( GUIA.TP_GUIA, 'C' )                IN ('C','P')
   AND     NVL( ITREG_AMB.TP_PAGAMENTO, 'P' ) = 'C'
