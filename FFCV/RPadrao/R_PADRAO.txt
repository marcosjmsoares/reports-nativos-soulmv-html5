--R_PADRAO
SELECT  PAD_COBR.CD_PADRAO
               ,PAD_COBR.DS_PADRAO
               ,REGRA.CD_REGRA
               ,REGRA.DS_REGRA
               ,PRO_FAT.CD_PRO_FAT
               ,PRO_FAT.DS_PRO_FAT
               ,PAD_COBR.TP_PADRAO
               ,PAD_COBR.NR_PERIODO
               ,PAD_COBR.DT_PRIMEIRO
FROM    DBAMV.PADRAO_COBRANCA  PAD_COBR
             ,DBAMV.REGRA REGRA
             ,DBAMV.PRO_FAT PRO_FAT
WHERE PAD_COBR.CD_REGRA  =  REGRA.CD_REGRA(+)
AND       PAD_COBR.CD_PRO_FAT  =  PRO_FAT.CD_PRO_FAT(+)
