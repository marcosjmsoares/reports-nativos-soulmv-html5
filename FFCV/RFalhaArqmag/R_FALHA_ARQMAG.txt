--R_FALHA_ARQMAG
Select cd_reg_fat,
           cd_atendimento,
            nm_paciente,
            tp_erro,
            tp_conta
From (
Select reg_fat.cd_reg_fat,
       reg_fat.cd_atendimento,
       paciente.nm_paciente,
       'Convenio sem Apresentação em Meio Magnético' tp_erro,
       'Hospitalar' tp_conta
  From dbamv.reg_fat,
       dbamv.atendime,
       dbamv.paciente,
       dbamv.convenio,
       Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And paciente.cd_paciente = atendime.cd_paciente
   AND REG_fat.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   and reg_fat.cd_atendimento = atendime.cd_atendimento
   and reg_fat.cd_convenio = convenio.cd_convenio
   and convenio.cd_apr_conta_meio_mag is null
   and reg_fat.cd_remessa = :CAMPO_CD_REMESSA
Union All
Select distinct itreg_amb.cd_reg_amb,
       itreg_amb.cd_atendimento,
       paciente.nm_paciente,
       'Convenio sem Apresentação em Meio Magnético' tp_erro,
       'Ambulatorial' tp_conta
  From dbamv.itreg_amb,
       dbamv.atendime,
       dbamv.paciente,
       dbamv.convenio,
       dbamv.reg_amb,
       Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And paciente.cd_paciente = atendime.cd_paciente
   AND REG_AMB.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   and itreg_amb.cd_atendimento = atendime.cd_atendimento
   and itreg_amb.cd_convenio = convenio.cd_convenio
   and convenio.cd_apr_conta_meio_mag is null
   and itreg_amb.cd_reg_amb = reg_amb.cd_reg_amb
   and reg_amb.cd_remessa = :CAMPO_CD_REMESSA
Union All
Select reg_fat.cd_reg_fat,
       reg_fat.cd_atendimento,
       paciente.nm_paciente,
       'Tipo de Acomodação não configurado' tp_erro,
       'Hospitalar' tp_conta
  From dbamv.reg_fat,
       dbamv.atendime,
       dbamv.paciente,
       dbamv.convenio
       ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And paciente.cd_paciente = atendime.cd_paciente
   AND REG_fat.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   and reg_fat.cd_atendimento = atendime.cd_atendimento
   and reg_fat.cd_convenio = convenio.cd_convenio
   and atendime.cd_tip_acom not in ( Select tip_aco_meio_mag.cd_tip_acom
                                       From dbamv.tip_aco_meio_mag
                                     Where tip_aco_meio_mag.cd_apr_conta_meio_mag = convenio.cd_apr_conta_meio_mag )
   and convenio.cd_apr_conta_meio_mag is not null
   and reg_fat.cd_remessa = :CAMPO_CD_REMESSA
Union All
Select distinct itreg_amb.cd_reg_amb,
       itreg_amb.cd_atendimento,
       paciente.nm_paciente,
       'Tipo de Acomodação não configurado' tp_erro,
       'Ambulatorial' tp_conta
  From dbamv.itreg_amb,
       dbamv.atendime,
       dbamv.paciente,
       dbamv.convenio,
       dbamv.reg_amb
       ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And paciente.cd_paciente = atendime.cd_paciente
   AND REG_AMB.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   and itreg_amb.cd_atendimento = atendime.cd_atendimento
   and itreg_amb.cd_convenio = convenio.cd_convenio
   and Nvl( atendime.cd_tip_acom, 0 ) not in ( Select Nvl( tip_aco_meio_mag.cd_tip_acom, 0 )
                                                From dbamv.tip_aco_meio_mag
                                                Where tip_aco_meio_mag.cd_apr_conta_meio_mag = convenio.cd_apr_conta_meio_mag )
   and convenio.cd_apr_conta_meio_mag is not null
   and reg_amb.cd_reg_amb = itreg_amb.cd_reg_amb
   and reg_amb.cd_remessa = :CAMPO_CD_REMESSA
Union All
Select reg_fat.cd_reg_fat,
       reg_fat.cd_atendimento,
       paciente.nm_paciente,
       'Tipo de Atendimento não configurado' tp_erro,
       'Hospitalar' tp_conta
  From dbamv.reg_fat,
       dbamv.atendime,
       dbamv.paciente,
       dbamv.convenio,
       Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And paciente.cd_paciente = atendime.cd_paciente
   AND REG_fat.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   and reg_fat.cd_atendimento = atendime.cd_atendimento
   and reg_fat.cd_convenio = convenio.cd_convenio
   and atendime.cd_tipo_internacao not in ( Select tip_int_meio_mag.cd_tipo_internacao
                                              From dbamv.tip_int_meio_mag
                                              Where tip_int_meio_mag.cd_apr_conta_meio_mag = convenio.cd_apr_conta_meio_mag )
   and convenio.cd_apr_conta_meio_mag is not null
   and reg_fat.cd_remessa = :CAMPO_CD_REMESSA
Union All
Select itreg_amb.cd_reg_amb,
       itreg_amb.cd_atendimento,
       paciente.nm_paciente,
       'Tipo de Atendimento não configurado' tp_erro,
       'Ambulatorial' tp_conta
  From dbamv.itreg_amb,
       dbamv.atendime,
       dbamv.paciente,
       dbamv.convenio,
       dbamv.reg_amb,
       Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And paciente.cd_paciente = atendime.cd_paciente
   AND REG_AMB.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   and itreg_amb.cd_atendimento = atendime.cd_atendimento
   and itreg_amb.cd_convenio = convenio.cd_convenio
   and atendime.cd_tipo_internacao not in ( Select tip_int_meio_mag.cd_tipo_internacao
                                              From dbamv.tip_int_meio_mag
                                              Where tip_int_meio_mag.cd_apr_conta_meio_mag = convenio.cd_apr_conta_meio_mag )

   and reg_amb.cd_remessa = :CAMPO_CD_REMESSA
   and Nvl( itreg_amb.cd_ati_med, 0 ) not in ( Select Nvl( ati_med_meio_mag.cd_ati_med, 0 )
