SELECT paciente.nm_paciente,
       reg_fat.cd_reg_fat,
       Nvl(guia.nr_guia,atendime.nr_guia) || ' / ' ||  guia.cd_senha  guia_senha,
       to_char(atendime.dt_atendimento,'dd/mm/yyyy') dt_atendimento,
       To_Char(reg_fat.dt_inicio, 'dd/mm/yyyy') dt_inicio,
       Nvl(To_Char(reg_fat.dt_final, 'dd/mm/yyyy'), SYSDATE) dt_final,
       convenio.cd_convenio || ' - ' || convenio.nm_razao_social ds_convenio,
       mov_int.cd_leito || ' - ' || leito.ds_leito ds_leito,
       con_pla.cd_con_pla || ' - ' || con_pla.ds_con_pla  ds_plano,
       Sum(itreg_fat_original.vl_total_conta) vl_total_conta_faturado,
       Sum(itreg_fat_original.vl_total_conta) vl_bruto_paciente
  FROM dbamv.paciente,
       dbamv.atendime,
       dbamv.reg_fat,
       dbamv.convenio,
       dbamv.con_pla,
       dbamv.leito,
       dbamv.itreg_fat_original,
       dbamv.empresa_convenio,
       dbamv.guia,
       dbamv.mov_int
WHERE atendime.cd_atendimento = :CAMPO_ATENDIMENTO
  AND reg_fat.cd_atendimento = atendime.cd_atendimento
  AND reg_fat.cd_reg_fat = :CAMPO_CONTA
  and nvl(itreg_fat_original.sn_paciente_paga, 'N') = 'N'
  and itreg_fat_original.sn_pertence_pacote = 'N'
  and nvl(itreg_fat_original.tp_pagamento, 'P') <> 'C'
  AND itreg_fat_original.cd_reg_fat = reg_fat.cd_reg_fat
  AND paciente.cd_paciente = atendime.cd_paciente
  AND convenio.cd_convenio = atendime.cd_convenio
  AND con_pla.cd_convenio = convenio.cd_convenio
  AND con_pla.cd_con_pla = atendime.cd_con_pla
  AND empresa_convenio.cd_convenio = convenio.cd_convenio
  AND empresA_convenio.cd_multi_empresa = nvl( (select r.cd_multi_empresa from dbamv.reg_fat r
                                                                                  where r.cd_reg_fat = reg_fat.cd_conta_pai), :CAMPO_CD_MULTI_EMPRESA )
  AND reg_fat.cd_guia = guia.cd_guia(+)
  AND mov_int.cD_atendimento(+) = :CAMPO_ATENDIMENTO
  AND mov_int.dt_mov_int = (SELECT Max(dt_mov_int) FROM dbamv.mov_int WHERE cd_atendimento = :CAMPO_ATENDIMENTO) 
  AND leito.cd_leito = mov_int.cd_leito
GROUP BY  paciente.nm_paciente,
       reg_fat.cd_reg_fat,
       Nvl(guia.nr_guia,atendime.nr_guia) || ' / ' ||  guia.cd_senha,
       atendime.dt_atendimento,
       To_Char(reg_fat.dt_inicio, 'dd/mm/yyyy'),
       To_Char(reg_fat.dt_final, 'dd/mm/yyyy'),
       convenio.cd_convenio || ' - ' ||  convenio.nm_razao_social,
       mov_int.cd_leito || ' - ' || leito.ds_leito,
       con_pla.cd_con_pla || ' - ' || con_pla.ds_con_pla
UNION ALL
SELECT paciente.nm_paciente,
       reg_amb.cd_reg_amb,
       Nvl(guia.nr_guia, atendime.nr_guia) || ' / ' ||  guia.cd_senha  guia_senha,
       to_char(atendime.dt_atendimento,'dd/mm/yyyy') dt_atendimento,
       To_Char(reg_amb.dt_lancamento, 'dd/mm/yyyy') dt_inicio,
       Nvl(To_Char(reg_amb.dt_lancamento_final, 'dd/mm/yyyy'), SYSDATE) dt_final,
       convenio.cd_convenio || ' - ' || convenio.nm_razao_social ds_convenio,
       leito.cd_leito || ' - ' || leito.ds_leito ds_leito,
       con_pla.cd_con_pla || ' - ' || con_pla.ds_con_pla  ds_plano,
       Sum(Nvl(itreg_amb_original.vl_total_conta,0)) vl_total_conta_faturado,
       Sum(itreG_amb_original.vl_total_conta - itreg_amb_original.vl_acrescimo + itreg_amb_original.vl_desconto_conta)  vl_bruto_paciente
  FROM dbamv.paciente,
       dbamv.atendime,
       dbamv.reg_amb,
       dbamv.convenio,
       dbamv.con_pla,
       dbamv.leito,
       dbamv.itreg_amb_original,
       dbamv.empresa_convenio,
       dbamv.guia
WHERE atendime.cd_atendimento = :CAMPO_ATENDIMENTO
  AND itreg_amb_original.cd_atendimento = atendime.cd_atendimento
  AND reg_amb.cd_reg_amb = :CAMPO_CONTA
  AND itreg_amb_original.cd_reg_amb = :CAMPO_CONTA
  AND paciente.cd_paciente = atendime.cd_paciente
  AND convenio.cd_convenio = atendime.cd_convenio
  AND leito.cd_leito (+) = atendime.cd_leito
  AND con_pla.cd_convenio = convenio.cd_convenio
  AND con_pla.cd_con_pla = atendime.cd_con_pla
  AND empresA_convenio.cd_convenio = convenio.cd_convenio
  AND empresa_convenio.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
  AND reg_amb.cd_guia = guia.cd_guia(+)
  and nvl(itreg_amb_original.sn_paciente_paga, 'N') = 'N'
  and itreg_amb_original.sn_pertence_pacote = 'N'
  and nvl(itreg_amb_original.tp_pagamento, 'P') <> 'C'
GROUP BY  paciente.nm_paciente,
       reg_amb.cd_reg_amb,
       Nvl(guia.nr_guia, atendime.nr_guia) || ' / ' ||  guia.cd_senha ,
       atendime.dt_atendimento,
       To_Char(reg_amb.dt_lancamento, 'dd/mm/yyyy'),
       To_Char(reg_amb.dt_lancamento_final, 'dd/mm/yyyy'),
       convenio.cd_convenio || ' - ' ||  convenio.nm_razao_social,
       leito.cd_leito || ' - ' || leito.ds_leito,
       con_pla.cd_con_pla || ' - ' || con_pla.ds_con_pla

select auxiliar.cd_pro_Fat,
        auxiliar.sn_horario_especial,
        auxiliar.qt_lancamento,
        auxiliar.cd_lancamento,
        sum(auxiliar.vl_total_conta) vl_total_conta,
        sum(auxiliar.vl_acrescimo) vl_acrescimo,
        Sum(auxiliar.vl_desconto_conta) vl_desconto_conta,
        pro_fat.cd_pro_fat || ' - ' || pro_fat.ds_pro_fat ds_pro_fat,
        auxiliar.cd_setor || ' - ' || setor.nm_setor   ds_setor
from (
        select conta_auditada.cd_pro_fat,
               conta_auditada.sn_horario_especial,
               conta_original.qt_lancamento,
               conta_original.cd_lancamento,
               conta_original.vl_total_conta,
               conta_original.vl_acrescimo,
               conta_original.vl_desconto_conta,
               conta_original.cd_setor
		          from (select cd_reg_fat, cd_lancamento, cd_pro_fat, qt_lancamento, vl_percentual_multipla
		                    , vl_unitario, nvl(vl_total_conta, 0) vl_total_conta, tp_mvto,  sn_horario_especial,
                        vl_desconto_conta, vl_acrescimo, cd_setor
		                  from dbamv.itreg_fat
		                where cd_reg_fat = :CAMPO_CONTA
		                  and exists (select 'X'
		                                from dbamv.auditoria_conta
		                                where cd_reg_fat = :CAMPO_CONTA
		                                  and dt_cancelou is null)) conta_auditada,
		              (select cd_reg_fat, cd_lancamento, cd_pro_fat, qt_lancamento, vl_percentual_multipla
		                    , vl_unitario, nvl(vl_total_conta, 0) vl_total_conta, tp_mvto, sn_horario_especial,
                        vl_desconto_conta, vl_acrescimo, cd_setor
		                  from dbamv.itreg_fat_original
		                where cd_reg_fat = :CAMPO_CONTA
		                  and exists (select 'X'
		                                from dbamv.auditoria_conta
		                                where cd_reg_fat = :CAMPO_CONTA
		                                  and dt_cancelou is null)) conta_original
		        where conta_auditada.cd_pro_fat = conta_original.cd_pro_fat
		          and conta_original.cd_lancamento = conta_auditada.cd_lancamento
		          and conta_original.cd_reg_fat = conta_auditada.cd_reg_fat
		          and ((conta_original.qt_lancamento > conta_auditada.qt_lancamento)
		            or (conta_original.vl_percentual_multipla > conta_auditada.vl_percentual_multipla)
		            or (conta_original.vl_unitario > conta_auditada.vl_unitario))
            union
		        select itreg_fat_original.cd_pro_fat,
                    itreg_fat_original.sn_horario_especial,
                    itreg_fat_original.qt_lancamento,
                    itreg_fat_original.cd_lancamento,
                    itreg_fat_original.vl_total_conta  vl_total_conta,
                    itreg_fat_original.vl_acrescimo,
                    itreg_fat_original.vl_desconto_conta,
                    itreg_fat_original.cd_setor
              from dbamv.itreg_fat_original
		         where itreg_fat_original.cd_reg_fat = :CAMPO_CONTA
		           and exists(select 'X'
		                        from dbamv.auditoria_conta ac
		                       where ac.dt_cancelou is null
		                         and ac.cd_reg_fat = itreg_fat_original.cd_reg_fat
		                         and ac.cd_lancamento_fat = itreg_fat_original.cd_lancamento
		                         and ac.cd_pro_fat = itreg_fat_original.cd_pro_fat)		                         
		           and not exists ( select 'X'
		                              from dbamv.itreg_fat ito
		                             where ito.cd_reg_fat = itreg_fat_original.cd_reg_fat
		                               and ito.cd_lancamento = itreg_fat_original.cd_lancamento
		                               and ito.cd_pro_fat = itreg_fat_original.cd_pro_fat)		                               
              UNION
              select conta_auditada.cd_pro_fat,
               conta_auditada.sn_horario_especial,
               conta_original.qt_lancamento,
               conta_original.cd_lancamento,
               conta_original.vl_total_conta,
               conta_original.vl_acrescimo,
               conta_original.vl_desconto_conta,
               conta_original.cd_setor
		          from (select cd_reg_amb, cd_lancamento, cd_pro_fat, qt_lancamento, vl_percentual_multipla
		                    , vl_unitario, nvl(vl_total_conta, 0) vl_total_conta, tp_mvto, sn_horario_especial,
                        vl_desconto_conta, vl_acrescimo, cd_setor
		                  from dbamv.itreg_amb
		                where cd_reg_amb = :CAMPO_CONTA
		                  and exists (select 'X'
		                                from dbamv.auditoria_conta
		                                where cd_reg_amb = :CAMPO_CONTA
		                                  and dt_cancelou is null)) conta_auditada,
		              (select cd_reg_amb, cd_lancamento, cd_pro_fat, qt_lancamento, vl_percentual_multipla
		                    , vl_unitario, nvl(vl_total_conta, 0) vl_total_conta, tp_mvto, sn_horario_especial,
                        vl_desconto_conta, vl_acrescimo,cd_setor
		                  from dbamv.itreg_amb_original
		                where cd_reg_amb = :CAMPO_CONTA
		                  and exists (select 'X'
		                                from dbamv.auditoria_conta
		                                where cd_reg_amb = :CAMPO_CONTA
		                                  and dt_cancelou is null)) conta_original
		        where conta_auditada.cd_pro_fat = conta_original.cd_pro_fat
		          and conta_original.cd_lancamento = conta_auditada.cd_lancamento
		          and conta_original.cd_reg_amb = conta_auditada.cd_reg_amb
              and ((conta_original.qt_lancamento > conta_auditada.qt_lancamento)
		            or (conta_original.vl_percentual_multipla > conta_auditada.vl_percentual_multipla)
		            or (conta_original.vl_unitario > conta_auditada.vl_unitario))
            union
		        SELECT  itreg_amb_original.cd_pro_fat,
                    itreg_amb_original.sn_horario_especial,
                    itreg_amb_original.qt_lancamento,
                    itreg_amb_original.cd_lancamento,
                    itreg_amb_original.vl_total_conta  vl_total_conta,
                    itreg_amb_original.vl_acrescimo,
                    itreg_amb_original.vl_desconto_conta,
                    itreg_amb_original.cd_setor                    
              from dbamv.itreg_amb_original
		         where itreg_amb_original.cd_reg_amb = :CAMPO_CONTA
		           and exists(select 'X'
		                        from dbamv.auditoria_conta ac
		                       where ac.dt_cancelou is null
		                         and ac.cd_reg_amb  = itreg_amb_original.cd_reg_amb
		                         and ac.cd_lancamento_amb = itreg_amb_original.cd_lancamento
		                         and ac.cd_pro_fat = itreg_amb_original.cd_pro_fat)		                         
		           and not exists ( select 'X'
		                              from dbamv.itreg_amb ito
		                             where ito.cd_reg_amb = itreg_amb_original.cd_reg_amb
		                               and ito.cd_lancamento = itreg_amb_original.cd_lancamento
		                               and ito.cd_pro_fat = itreg_amb_original.cd_Pro_fat)		                               
      ) auxiliar,
      dbamv.pro_fat,
      dbamv.setor
     WHERE pro_fat.cd_pro_fat = auxiliar.cd_pro_fat
       AND setor.cd_setor = auxiliar.cd_setor
  GROUP by auxiliar.qt_lancamento, auxiliar.cd_pro_fat, auxiliar.sn_horario_especial, pro_fat.cd_pro_fat || ' - ' || pro_fat.ds_pro_fat,
            auxiliar.cd_setor || ' - ' || setor.nm_setor, auxiliar.cd_lancamento
)
ORDER BY DS_SETOR
