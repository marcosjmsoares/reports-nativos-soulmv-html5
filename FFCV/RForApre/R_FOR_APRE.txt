--R_FOR_APRE
SELECT * FROM (
SELECT  FOR_APRE.CD_FOR_APRE CD_FOR_APRE
               ,FOR_APRE.DS_FOR_APRE DS_FOR_APRE
               ,ITFOR_APRE.CD_FOR_APRE CD_FOR_APRE1
               ,ITFOR_APRE.CD_GRU_FAT CD_GRU_FAT
               ,GRU_FAT.CD_GRU_FAT CD_GRU_FAT1
               ,GRU_FAT.DS_GRU_FAT DS_GRU_FAT
               ,ITFOR_APRE.CD_SEQUENCIA CD_SEQUENCIA
FROM    dbamv.FOR_APRE FOR_APRE
             ,dbamv.ITFOR_APRE ITFOR_APRE
             ,dbamv.GRU_FAT GRU_FAT
WHERE ITFOR_APRE.CD_FOR_APRE = FOR_APRE.CD_FOR_APRE
AND       GRU_FAT.CD_GRU_FAT = ITFOR_APRE.CD_GRU_FAT
)
ORDER BY DS_FOR_APRE,CD_FOR_APRE,:CAMPO_CF_ORDEM
