--R_CONPLA_CAD
SELECT  CONVENIO.CD_CONVENIO                                    CODIGO
              ,CONVENIO.NM_CONVENIO                                    DESCRICAO
              ,CONVENIO.CD_FOR_APRE                                    CD_FOR_APRE
              ,CONVENIO.TP_CONVENIO                                    TP_CONVENIO
,Decode(CONVENIO.TP_CONVENIO,'P','PARTICULAR','C','CONVENIO','A','SIA/SUS','H','SIH/SUS',null) NM_TP_CONVENIO
              ,CON_PLA.CD_CON_PLA                                        CD_CON_PLA1
              ,CON_PLA.DS_CON_PLA                                        DS_CON_PLA1
              ,empresa_CON_PLA.CD_REGRA                            CD_REGRA 
              ,empresa_CON_PLA.CD_INDICE                             CD_INDICE 
              ,nvl(empresa_con_pla.DS_OBSERVACAO,con_pla.DS_OBSERVACAO)  DS_OBSERVACAO    
              ,FOR_APRE.DS_FOR_APRE                                   DS_FOR_APRE
              ,REGRA.DS_REGRA                                                DS_REGRA
              ,INDICE.NM_INDICE                                                 NM_INDICE
              ,CONVENIO.SN_FATURA_AMBULATORIO_SUS   SN_FATURA_AMBULATORIO_SUS
              ,CONVENIO.SN_COBRA_INSUMOS                       SN_COBRA_INSUMOS
              ,CONVENIO.TP_CONVENIO                                    TP_CONVENIO
              ,CONVENIO.TP_FORMA_AGRUPAMENTO            TP_FORMA_AGRUPAMENTO
              ,CONVENIO.TP_IMPORTAR_MATMED                  TP_IMPORTAR_MATMED
              ,CONVENIO.DS_BAIRRO                                         DS_BAIRRO
              ,CONVENIO.DS_CIDADE                                         DS_CIDADE
              ,CONVENIO.DS_ENDERECO                                  DS_ENDERECO
              ,CONVENIO.DS_UF                                                 DS_UF
              ,CONVENIO.NR_CGC                                               NR_CGC
              ,CONVENIO.NR_FONE                                             NR_FONE
              ,CONVENIO.NR_CEP                                               NR_CEP
              ,CONVENIO.NM_RAZAO_SOCIAL                           NM_RAZAO_SOCIAL
              ,CONVENIO.SN_GUIA                                              SN_GUIA
              ,CONVENIO.SN_VALIDADE_CARTEIRA                 SN_VALIDADE_CARTEIRA
              ,CONVENIO.SN_ABATE_DEVOLUCAO                   SN_ABATE_DEVOLUCAO
              ,CONVENIO.SN_ATIVO                                            SN_ATIVO
              ,CONVENIO.NR_DIAS_RETORNO                          NR_DIAS_RETORNO
              ,CONVENIO.CD_HOSPITAL_NO_CONVENIO         CD_HOSPITAL_NO_CONVENIO
              ,CONVENIO.CD_CONVENIO_UNIFICADO               CD_CONVENIO_UNIFICADO
              ,CONVENIO.NR_SEQ_REMESSA                            CD_APR_CONTA_MEIO_MAG
              ,CONVENIO.VL_PERC_IR                                         VL_PERC_IR
              ,CONVENIO.VL_PERC_ISS                                       VL_PERC_ISS
              ,CONVENIO.NM_CONTATO                                     NM_CONTATO
              ,CONVENIO.ds_endereco_correspondencia
              ,CONVENIO.ds_bairro_correspondencia
              ,CONVENIO.ds_cidade_correspondencia
              ,CONVENIO.nr_cep_correspondencia
              ,CONVENIO.ds_uf_correspondencia
              ,CONVENIO.ds_endereco_cobranca
              ,CONVENIO.ds_bairro_cobranca
              ,CONVENIO.ds_cidade_cobranca
              ,CONVENIO.nr_cep_cobranca
              ,CONVENIO.ds_uf_cobranca
              ,CONVENIO.dt_inicio_contrato
FROM    dbamv.CONVENIO               CONVENIO
             ,dbamv.FOR_APRE               FOR_APRE
             ,dbamv.CON_PLA                  CON_PLA
             ,dbamv.REGRA                      REGRA
             ,dbamv.INDICE                       INDICE
             ,Dbamv.Empresa_Con_Pla
Where Empresa_Con_Pla.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And Empresa_Con_Pla.Cd_Convenio = Con_Pla.Cd_Convenio
And Empresa_Con_Pla.Cd_Con_Pla = Con_Pla.Cd_Con_Pla
And CONVENIO.CD_FOR_APRE = FOR_APRE.CD_FOR_APRE(+)
AND        CONVENIO.CD_CONVENIO =  CON_PLA.CD_CONVENIO
AND       empresa_CON_PLA.CD_REGRA = REGRA.CD_REGRA(+)  
AND       empresa_CON_PLA.CD_INDICE = INDICE.CD_INDICE(+)  
and         empresa_con_pla.cd_con_pla = con_pla.cd_con_pla
and         empresa_con_pla.cd_convenio = con_pla.cd_convenio
and         empresa_con_pla.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
ORDER BY CONVENIO.NM_CONVENIO
                   ,CONVENIO.TP_CONVENIO
                   ,FOR_APRE.DS_FOR_APRE
