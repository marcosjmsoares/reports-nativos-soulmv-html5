--R_PREVISAO_GUIA
SELECT SETOR.CD_SETOR,
       SETOR.NM_SETOR,
       MOV_INT.CD_ATENDIMENTO,
       ATENDIME.CD_PACIENTE,
       PACIENTE.NM_PACIENTE,
       ACO_PRO_FAT.QT_LANCAMENTO QT_LANCAMENTO,
       MOV_INT_ATEND.CD_REG_FAT,
       ATENDIME.DT_ATENDIMENTO DT_INICIO,
       ATENDIME.DT_ALTA DT_FINAL,
       REG_FAT.CD_CONVENIO,
       REG_FAT.CD_CON_PLA,
       LEITO.DS_LEITO
  FROM DBAMV.MOV_INT,
       DBAMV.SETOR,
       DBAMV.LEITO,
       DBAMV.UNID_INT,
       DBAMV.ATENDIME,
       DBAMV.PACIENTE,
       DBAMV.REG_FAT,
       DBAMV.CONVENIO,
       ( SELECT R.CD_ATENDIMENTO,SUM(I.QT_LANCAMENTO) QT_LANCAMENTO
           FROM DBAMV.ITREG_FAT I,
                DBAMV.REG_FAT R,
                DBAMV.ACOMOD_PRO_FAT A
          WHERE R.CD_REG_FAT  = I.CD_REG_FAT
            AND A.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
            AND R.CD_CONVENIO = A.CD_CONVENIO
            AND R.CD_CON_PLA  = A.CD_CON_PLA
            AND R.CD_TIP_ACOM = A.CD_TIP_ACOM
            AND I.CD_PRO_FAT  = A.CD_PRO_FAT
          GROUP BY R.CD_ATENDIMENTO              ) ACO_PRO_FAT,
       ( SELECT MAX( MOV_INT.CD_MOV_INT) CD_MOV_INT, MOV_INT.CD_ATENDIMENTO , MIN(REG_FAT.CD_REG_FAT) CD_REG_FAT
           FROM DBAMV.LEITO,
                DBAMV.UNID_INT,
                DBAMV.MOV_INT,
                DBAMV.REG_FAT,
                DBAMV.ATENDIME
          WHERE ATENDIME.CD_ATENDIMENTO  = MOV_INT.CD_ATENDIMENTO
            AND REG_FAT.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
            AND LEITO.CD_LEITO           = MOV_INT.CD_LEITO
            AND UNID_INT.CD_UNID_INT     = LEITO.CD_UNID_INT
            AND ATENDIME.TP_ATENDIMENTO  = 'I'
            AND ATENDIME.DT_ALTA         IS NULL
            AND REG_FAT.CD_ATENDIMENTO   = ATENDIME.CD_ATENDIMENTO
            AND REG_FAT.CD_CONVENIO      = ATENDIME.CD_CONVENIO
          GROUP BY MOV_INT.CD_ATENDIMENTO
          ) MOV_INT_ATEND
 WHERE MOV_INT.CD_MOV_INT       = MOV_INT_ATEND.CD_MOV_INT
   AND REG_FAT.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
   AND ATENDIME.CD_ATENDIMENTO  = MOV_INT.CD_ATENDIMENTO
   AND PACIENTE.CD_PACIENTE(+)  = ATENDIME.CD_PACIENTE
   AND REG_FAT.CD_ATENDIMENTO   = MOV_INT_ATEND.CD_ATENDIMENTO
   AND REG_FAT.CD_REG_FAT       = MOV_INT_ATEND.CD_REG_FAT
   AND LEITO.CD_LEITO           = MOV_INT.CD_LEITO
   AND UNID_INT.CD_UNID_INT     = LEITO.CD_UNID_INT
   AND CONVENIO.CD_CONVENIO     = REG_FAT.CD_CONVENIO
   AND CONVENIO.TP_CONVENIO     = 'C'
   AND ACO_PRO_FAT.CD_ATENDIMENTO  = REG_FAT.CD_ATENDIMENTO
   AND UNID_INT.CD_SETOR         = SETOR.CD_SETOR
   AND ATENDIME.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
ORDER BY NM_SETOR,NM_PACIENTE
