--R_TAB_PROCEDIMENTOS
SELECT * FROM (
SELECT TAB_FAT.CD_TAB_FAT                   CD_TAB_FAT1
              ,TAB_FAT.DS_TAB_FAT                   DS_TAB_FAT
              ,TAB_FAT.TP_TAB_FAT                   TP_TAB_FAT
              ,VAL_PRO.CD_PRO_FAT                  CODIGO
              ,VAL_PRO.CD_TAB_FAT                   CD_TAB_FAT
              ,TRUNC(VAL_PRO.DT_VIGENCIA)   DT_VIGENCIA
              ,DECODE (:CAMPO_CF_PERCENTUAL, NULL, VAL_PRO.VL_TOTAL,:CAMPO_CF_PERCENTUAL)   VL_TOTAL
              ,PRO_FAT.DS_PRO_FAT                   DESCRICAO
              ,PRO_FAT.CD_PRO_FAT                   CD_PRO_FAT2
              ,PRO_FAT.DS_UNIDADE                   DS_UNIDADE1
              ,PRO_FAT.NR_AUXILIAR                   NR_AUXILIAR1
              ,PRO_FAT.TP_SEXO                          TP_SEXO1
              ,0		                           VL_FILME1
              ,POR_ANE.CD_POR_ANE                  CD_POR_ANE1
              ,POR_ANE.DS_POR_ANE                  DS_POR_ANE
              ,GRU_PRO.CD_GRU_PRO                CD_GRU_PRO1
              ,GRU_PRO.DS_GRU_PRO                DS_GRU_PRO
              ,CONVENIO.CD_CONVENIO              CD_CONVENIO
              ,CONVENIO.NM_CONVENIO             NM_CONVENIO
              ,REGRA.CD_REGRA                          CD_REGRA
      FROM  dbamv.VAL_PRO        VAL_PRO
           ,dbamv.TAB_FAT        TAB_FAT
           ,dbamv.PRO_FAT        PRO_FAT
           ,dbamv.POR_ANE        POR_ANE
           ,dbamv.GRU_PRO        GRU_PRO
           ,dbamv.CONVENIO       CONVENIO
           ,dbamv.CON_PLA        CON_PLA
           ,dbamv.ITREGRA       ITREGRA
           ,dbamv.REGRA          REGRA
,Dbamv.Empresa_Con_Pla
Where Empresa_Con_Pla.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And Empresa_Con_Pla.Cd_Convenio = Con_Pla.Cd_Convenio
And Empresa_Con_Pla.Cd_Con_Pla = Con_Pla.Cd_Con_Pla
And TAB_FAT.CD_TAB_FAT      = VAL_PRO.CD_TAB_FAT
AND   PRO_FAT.CD_PRO_FAT         = VAL_PRO.CD_PRO_FAT
AND   PRO_FAT.SN_ATIVO               = 'S'
AND       NVL(VAL_PRO.SN_ATIVO, 'S') = decode(:CAMPO_SN_ATIVO, 'T', NVL(VAL_PRO.SN_ATIVO, 'S'), :CAMPO_SN_ATIVO)
AND   GRU_PRO.CD_GRU_PRO       = PRO_FAT.CD_GRU_PRO
AND   CONVENIO.CD_CONVENIO     = CON_PLA.CD_CONVENIO
AND   CON_PLA.CD_REGRA             = REGRA.CD_REGRA
AND   ITREGRA.CD_REGRA             = REGRA.CD_REGRA
AND   ITREGRA.CD_TAB_FAT          = TAB_FAT.CD_TAB_FAT
AND   ITREGRA.CD_GRU_PRO        = GRU_PRO.CD_GRU_PRO
AND   PRO_FAT.CD_POR_ANE        = POR_ANE.CD_POR_ANE (+)
AND   TRUNC(VAL_PRO.DT_VIGENCIA)  = (SELECT MAX(TRUNC(VAL_PRO_3.DT_VIGENCIA))
                                     FROM DBAMV.VAL_PRO    VAL_PRO_3
                                     WHERE VAL_PRO_3.CD_TAB_FAT = VAL_PRO.CD_TAB_FAT
                                           AND  NVL(VAL_PRO_3.SN_ATIVO, 'S') = decode(:CAMPO_SN_ATIVO, 'T', NVL(VAL_PRO_3.SN_ATIVO, 'S'), :CAMPO_SN_ATIVO)
                                     AND   VAL_PRO_3.CD_PRO_FAT = VAL_PRO.CD_PRO_FAT)
AND PRO_FAT.CD_PRO_FAT NOT IN ( SELECT PROIBICAO.CD_PRO_FAT
                                                                    FROM DBAMV.PROIBICAO
                                                                 WHERE Proibicao.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
								        AND PROIBICAO.CD_CONVENIO = CONVENIO.CD_CONVENIO
                                                                        AND ((PROIBICAO.TP_PROIBICAO = 'NA'
                                                                                  OR PROIBICAO.TP_PROIBICAO = 'FC')
                                                                                   AND PROIBICAO.TP_ATENDIMENTO = 'T'))
GROUP BY CONVENIO.CD_CONVENIO
              ,TAB_FAT.CD_TAB_FAT
              ,TAB_FAT.DS_TAB_FAT
              ,TAB_FAT.TP_TAB_FAT
              ,VAL_PRO.CD_PRO_FAT
              ,VAL_PRO.CD_TAB_FAT
              ,TRUNC(VAL_PRO.DT_VIGENCIA)
              ,DECODE (:CAMPO_CF_PERCENTUAL, NULL, VAL_PRO.VL_TOTAL,:CAMPO_CF_PERCENTUAL)
              ,PRO_FAT.DS_PRO_FAT
              ,PRO_FAT.CD_PRO_FAT
              ,PRO_FAT.DS_UNIDADE
              ,PRO_FAT.NR_AUXILIAR
              ,PRO_FAT.TP_SEXO
              ,0
              ,POR_ANE.CD_POR_ANE
              ,POR_ANE.DS_POR_ANE
              ,GRU_PRO.CD_GRU_PRO
              ,GRU_PRO.DS_GRU_PRO
              ,CONVENIO.NM_CONVENIO
              ,REGRA.CD_REGRA
ORDER BY :CAMPO_ORDEM
)
ORDER BY CD_CONVENIO,NM_CONVENIO,
TP_TAB_FAT,DS_TAB_FAT,CD_TAB_FAT,
CD_GRU_PRO1,DS_GRU_PRO
