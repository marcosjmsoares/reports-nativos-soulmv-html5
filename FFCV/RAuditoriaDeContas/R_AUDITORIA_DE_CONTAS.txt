  qt_saida               ,
  qt_devolvida           ,
  Qt_Consumo             ,
  qt_faturada            ,
  ds_unidade             ,
  qt_prescrita           ,
  cd_pro_fat_entrada     ,
  cd_pro_fat_alta        ,
  regra
   FROM
  (SELECT Cd_Setor                                                                                 ,
    Nm_Setor                                                                                       ,
    NVL( Cd_Atendimento_Pai, Cd_Atendimento ) Cd_Atendimento                                       ,
    NVL( Dt_Atendimento_Pai, Dt_Atendimento ) Dt_Atendimento                                       ,

  	                                                  and trunc(:CAMPO_DT_FIM)
  	                                                         or        trunc(atendime.dt_Alta)  between trunc(:CAMPO_DT_INI)
  	                                                  and trunc(:CAMPO_DT_FIM))

      0 Qt_Devolvida                                                                                                                                                                                   ,
      0 Qt_Consumo                                                                                                                                                                                     ,
      NVL( ItReg_Amb.Qt_Lancamento, 0 ) Qt_Faturada                                                                                                                                                    ,

      Pro_fat.Cd_gru_pro
       FROM Dbamv.ItReg_Amb      ,
      Dbamv.Atendime             ,
      Dbamv.Atendime Atendime_Pai,
      Dbamv.Paciente             ,
      Dbamv.Paciente Paciente_Pai,
      Dbamv.Setor                ,
      Dbamv.Convenio             ,
      Dbamv.Pro_Fat              ,
      Dbamv.Pro_Fat Pro_Fat_Int  ,
      Dbamv.Pro_Fat Pro_Fat_Alta
      WHERE ItReg_Amb.Cd_Pro_Fat                = Pro_Fat.Cd_Pro_Fat
    AND Atendime.Cd_Pro_Int                     = Pro_Fat_Int.Cd_Pro_Fat (+)
    AND Atendime.Cd_Pro_Int_Procedimento_Entrad = Pro_fat_Alta.Cd_Pro_Fat (+)
    AND ItReg_Amb.Cd_Atendimento                = Atendime.Cd_Atendimento

  	                                                  and trunc(:CAMPO_DT_FIM)
  	                                                         or        trunc(atendime.dt_Alta)  between trunc(:CAMPO_DT_INI)
  	                                                  and trunc(:CAMPO_DT_FIM))

      Pro_Fat_Alta.Cd_Pro_Fat Cd_Pro_Fat_Alta                                                    ,
      Pro_Fat_Alta.Ds_Pro_Fat Ds_Pro_Fat_Alta                                                    ,
      Atendime.Dt_Atendimento                                                                    ,
