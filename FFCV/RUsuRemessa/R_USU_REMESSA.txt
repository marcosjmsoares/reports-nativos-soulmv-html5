--R_USU_REMESSA
SELECT r.nm_usuario_fechou,
       r.cd_remessa,
       c.cd_convenio,
       c.nm_convenio,
       f.dt_competencia
  FROM dbamv.remessa_fatura r,
       dbamv.fatura f,
       dbamv.convenio c
 WHERE f.cd_fatura = r.cd_fatura
   AND f.cd_convenio = c.cd_convenio
   AND r.nm_usuario_fechou IS NOT NULL
 ORDER BY r.nm_usuario_fechou, c.cd_convenio
