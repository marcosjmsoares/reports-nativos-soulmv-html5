select NVL(SUM(TOTAL_GERAL),0) TOTAL_GERAL from (
SELECT  CD_CONVENIO CD_CONVENIO
       ,NM_CONVENIO NM_CONVENIO
       ,SUM(VL_INTERNACAO)   VL_INTERNACAO
       ,SUM(VL_AMBULATORIAL) VL_AMBULATORIAL
       ,SUM(VL_EXTERNO)      VL_EXTERNO
       ,SUM(VL_URGENCIA)     VL_URGENCIA
       ,SUM(QT_INTERNACAO)   QT_INTERNACAO
       ,SUM(QT_AMBULATORIAL) QT_AMBULATORIAL
       ,SUM(QT_EXTERNO)  QT_EXTERNO
       ,SUM(QT_URGENCIA) QT_URGENCIA
       ,SUM(VALOR)    VALOR
       ,COUNT(VALOR)  QTD
       ,SUM(VL_REPASSE)      VL_REPASSE
       ,(DECODE(SUM(VL_AMBULATORIAL), NULL, 0,SUM(VL_AMBULATORIAL)) + DECODE(SUM(VL_URGENCIA), NULL, 0,SUM(VL_URGENCIA)) + DECODE(SUM(VL_EXTERNO), NULL, 0,SUM(VL_EXTERNO)) + DECODE(SUM(VL_INTERNACAO), NULL, 0,SUM(VL_INTERNACAO)) ) TOTAL_GERAL
FROM (
SELECT  CD_CONVENIO
       ,NM_CONVENIO
       ,CD_ATENDIMENTO
       ,SUM(VL_INTERNACAO)     VL_INTERNACAO
       ,SUM(VL_AMBULATORIAL) VL_AMBULATORIAL
       ,SUM(VL_EXTERNO)           VL_EXTERNO
       ,SUM(VL_URGENCIA)          VL_URGENCIA
       ,DECODE(SUM(QT_INTERNACAO),0,0,SUM(QT_INTERNACAO))   QT_INTERNACAO
       ,DECODE(SUM(QT_AMBULATORIAL),0,0,SUM(QT_AMBULATORIAL))   QT_AMBULATORIAL
       ,DECODE(SUM(QT_EXTERNO),0,0,SUM(QT_EXTERNO))   QT_EXTERNO
       ,DECODE(SUM(QT_URGENCIA),0,0,SUM(QT_URGENCIA))   QT_URGENCIA

       ,COUNT(VL_TOTAL_CONTA)  QTD

(SELECT  CONVENIO.CD_CONVENIO                  CD_CONVENIO
       ,CONVENIO.NM_CONVENIO                     NM_CONVENIO
       ,ATENDIME.CD_ATENDIMENTO             CD_ATENDIMENTO
       ,DECODE(ATENDIME.TP_ATENDIMENTO, 'I', NVL(NVL(ITLAN_MED.VL_LIQUIDO, ITREG_FAT.VL_TOTAL_CONTA),0)
									  , 'H', NVL(NVL(ITLAN_MED.VL_LIQUIDO, ITREG_FAT.VL_TOTAL_CONTA),0), 0)	VL_INTERNACAO       
	   ,DECODE(ATENDIME.TP_ATENDIMENTO, 'A', NVL(ITREG_FAT.VL_TOTAL_CONTA, 0),0) VL_AMBULATORIAL
       ,DECODE(ATENDIME.TP_ATENDIMENTO, 'E', NVL(ITREG_FAT.VL_TOTAL_CONTA, 0),0) VL_EXTERNO
       ,DECODE(ATENDIME.TP_ATENDIMENTO, 'U', NVL(ITREG_FAT.VL_TOTAL_CONTA, 0),0) VL_URGENCIA
       ,DECODE(ATENDIME.TP_ATENDIMENTO, 'I', 1, 'H', 1 ,0)   QT_INTERNACAO
       ,DECODE(ATENDIME.TP_ATENDIMENTO, 'A', 1,0) QT_AMBULATORIAL
       ,DECODE(ATENDIME.TP_ATENDIMENTO, 'E', 1,0) QT_EXTERNO
       ,DECODE(ATENDIME.TP_ATENDIMENTO, 'U', 1,0) QT_URGENCIA
       ,NVL(ITREG_FAT.VL_TOTAL_CONTA,0)              VL_TOTAL_CONTA
       ,DECODE(GRU_FAT.TP_GRU_FAT,'SP',DECODE(NVL(ITREG_FAT.TP_PAGAMENTO,'P'),'P',ITREG_FAT.VL_TOTAL_CONTA,0),0)  VL_REPASSE
FROM    DBAMV.REG_FAT
       ,DBAMV.ITREG_FAT

       ,DBAMV.CON_PLA
       ,DBAMV.ATENDIME
       ,DBAMV.PRO_FAT
       ,DBAMV.SERVICO

       , DBAMV.ITLAN_MED
Where REG_FAT.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
AND   ITREG_FAT.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
AND   ITREG_FAT.CD_GRU_FAT = GRU_FAT.CD_GRU_FAT
          AND ITREG_FAT.CD_REG_FAT = ITLAN_MED.CD_REG_FAT(+)
          AND ITREG_FAT.CD_LANCAMENTO = ITLAN_MED.CD_LANCAMENTO(+)
          AND NVL(NVL(ITLAN_MED.TP_PAGAMENTO, ITREG_FAT.TP_PAGAMENTO),'X') <> 'C'
AND   REG_FAT.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
AND   REG_FAT.CD_CONVENIO    = CONVENIO.CD_CONVENIO
AND   REG_FAT.CD_CONVENIO    = CON_PLA.CD_CONVENIO
AND   REG_FAT.CD_CON_PLA     = CON_PLA.CD_CON_PLA
AND   CONVENIO.TP_CONVENIO IN ('C','P')
