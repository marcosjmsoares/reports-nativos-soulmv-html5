--R_ENVIA_PROTOCO_FAT
SELECT * FROM (
SELECT to_char( GUIA.DT_ENVIO_FAT, 'dd/MM/yyyy') DT_ENVIO_FAT
      ,to_char( GUIA.DT_ENVIO_FAT, 'hh24') HR_ENVIO_FAT
      ,GUIA.CD_ATENDIMENTO
      ,PACIENTE.NM_PACIENTE
      ,to_char( ATENDIME.DT_ATENDIMENTO, 'dd/MM/yyyy') DT_ATENDIMENTO
      ,to_char( ATENDIME.DT_ALTA, 'dd/MM/yyyy') DT_ALTA
      ,CONVENIO.CD_CONVENIO
      ,CONVENIO.NM_CONVENIO
      ,GUIA.CD_GUIA
      ,GUIA.CD_SENHA
      ,GUIA.NR_GUIA
      ,Decode( GUIA.TP_GUIA, 'I', 'Internação',
                                                'C', 'Consulta',
                                                'P', 'Procedimento',
                                                'R', 'Prorrogação') tp_guia
      ,GUIA.NR_DIAS_AUTORIZADOS
      ,GUIA.NR_HORAS_AUTORIZADAS
      ,GUIA.DT_VALIDADE
FROM DBAMV.GUIA
    ,DBAMV.PACIENTE
    ,DBAMV.CONVENIO
    ,DBAMV.ATENDIME
    ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And ATENDIME.CD_ATENDIMENTO = GUIA.CD_ATENDIMENTO
  AND PACIENTE.CD_PACIENTE = ATENDIME.CD_PACIENTE
  AND CONVENIO.CD_CONVENIO = GUIA.CD_CONVENIO
 :CAMPO_CF_GUIA
)
ORDER BY DT_ENVIO_FAT,HR_ENVIO_FAT,CD_ATENDIMENTO,NM_PACIENTE,DT_ATENDIMENTO,DT_ALTA,CD_CONVENIO,NM_CONVENIO,
CD_GUIA,CD_SENHA,NR_GUIA,TP_GUIA,NR_DIAS_AUTORIZADOS,NR_HORAS_AUTORIZADAS,DT_VALIDADE
