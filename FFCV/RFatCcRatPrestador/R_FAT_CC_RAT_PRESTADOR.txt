--R_FAT_CC_RAT_PRESTADOR
SELECT * FROM (
SELECT GRUPO NM_PRESTADOR, SETOR NM_SETOR, SUM(VALOR) VL_TOTAL
FROM
(SELECT NVL(S.GRUPO,N.GRUPO) GRUPO
      ,NVL(S.SETOR,N.SETOR) SETOR
      ,NVL(TRUNC((S.VALOR * N.VALOR) + 0.005,2) ,N.VALOR) VALOR
FROM
(SELECT          NM_PRESTADOR GRUPO
               ,NM_SETOR   SETOR
               ,SUM(VL_TOTAL) VALOR
               ,CONTA      CONTA
FROM  ( SELECT  PRESTADOR.NM_PRESTADOR       NM_PRESTADOR
                            ,DECODE(SETOR.NM_SETOR, NULL, ' Não Informado', SETOR.NM_SETOR)        NM_SETOR
                            ,SUM(nvl(ITLAN_MED.VL_LIQUIDO,ITREG_FAT.VL_TOTAL_CONTA) )       VL_TOTAL
                            ,REG_FAT.CD_REG_FAT   CONTA
               FROM   DBAMV.REG_FAT
                            ,DBAMV.ITREG_FAT
                            ,DBAMV.PRO_FAT
                            ,DBAMV.GRU_PRO
                            ,DBAMV.SETOR
                            ,DBAMV.REMESSA_FATURA     REMESSA
                            ,DBAMV.FATURA
                            ,DBAMV.CONVENIO
                            ,DBAMV.ATENDIME
                            ,DBAMV.PRESTADOR
                            ,dbamv.itlan_med
               WHERE  REG_FAT.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
               AND        REG_FAT.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
               AND        ITREG_FAT.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
               AND        ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
               AND        PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
               AND        ITREG_FAT.CD_SETOR = SETOR.CD_SETOR(+)
               AND        REG_FAT.CD_REMESSA = REMESSA.CD_REMESSA
               AND        REMESSA.CD_FATURA = FATURA.CD_FATURA
               AND        REG_FAT.CD_CONVENIO = CONVENIO.CD_CONVENIO
               AND        ITREG_FAT.SN_PERTENCE_PACOTE = 'N'
               AND        NVL(REG_FAT.SN_DIAGNO,'N' ) =  'N'
               AND ITLAN_MED.CD_REG_FAT(+) = ITREG_FAT.CD_REG_FAT
aND ITLAN_MED.CD_LANCAMENTO(+) = ITREG_FAT.CD_LANCAMENTO
and nvl(nvl(itlan_med.tp_pagamento,itreg_fat.tp_pagamento),'X') <> 'C'
AND CONVENIO.TP_CONVENIO not in ( 'H', 'A')
                AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
                GROUP BY  PRESTADOR.NM_PRESTADOR
                           ,SETOR.NM_SETOR
                           ,REG_FAT.CD_REG_FAT
               UNION ALL
               SELECT  PRESTADOR.NM_PRESTADOR  NM_PRESTADOR
                             ,DECODE(SETOR.NM_SETOR, NULL, ' Não Informado', SETOR.NM_SETOR)    NM_SETOR
                             ,SUM(ITREG_AMB.VL_TOTAL_CONTA)          VL_TOTAL
                             ,REG_AMB.CD_REG_AMB CONTA
                FROM   DBAMV.REG_AMB
                             ,DBAMV.ITREG_AMB
                             ,DBAMV.PRO_FAT
                             ,DBAMV.GRU_PRO
                             ,DBAMV.SETOR
                             ,DBAMV.REMESSA_FATURA  REMESSA
                             ,DBAMV.FATURA
                             ,DBAMV.CONVENIO
                             ,DBAMV.ATENDIME
                             ,DBAMV.PRESTADOR
                WHERE  REG_AMB.CD_REG_AMB = ITREG_AMB.CD_REG_AMB
                AND        ITREG_AMB.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
                AND        ITREG_AMB.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
                AND        ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
                AND        PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
                AND        ITREG_AMB.CD_SETOR = SETOR.CD_SETOR(+)
                AND        REG_AMB.CD_REMESSA = REMESSA.CD_REMESSA
                AND        REMESSA.CD_FATURA = FATURA.CD_FATURA
                AND        ITREG_AMB.CD_CONVENIO = CONVENIO.CD_CONVENIO
                AND        ITREG_AMB.SN_PERTENCE_PACOTE = 'N'
                AND        NVL(REG_AMB.SN_DIAGNO,'N' ) =  'N'
                AND        NVL(ITREG_AMB.TP_PAGAMENTO,'X') <> 'C'
                AND        REG_AMB.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
                AND        CONVENIO.TP_CONVENIO not in ( 'H', 'A')
                AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
               GROUP BY  PRESTADOR.NM_PRESTADOR
                         ,SETOR.NM_SETOR
                         ,REG_AMB.CD_REG_AMB )
WHERE VL_TOTAL IS NOT NULL
GROUP BY  NM_PRESTADOR
                    ,NM_SETOR
                    ,CONTA) N,
(SELECT VALOR.GRUPO , VALOR.SETOR , (VALOR.VALOR / Decode (TOTAL.VALOR,0,1,TOTAL.VALOR)) VALOR, VALOR.CONTA
FROM
(SELECT          NM_PRESTADOR GRUPO
               ,NM_SETOR   SETOR
               ,SUM(VL_TOTAL) VALOR
               ,CONTA      CONTA
FROM  ( SELECT  PRESTADOR.NM_PRESTADOR       NM_PRESTADOR
                            ,DECODE(SETOR.NM_SETOR, NULL, ' Não Informado', SETOR.NM_SETOR)        NM_SETOR
                            ,SUM(nvl(ITLAN_MED.VL_LIQUIDO,ITREG_FAT.VL_TOTAL_CONTA) )       VL_TOTAL
                            ,REG_FAT.CD_REG_FAT   CONTA
               FROM   DBAMV.REG_FAT
                            ,DBAMV.ITREG_FAT
                            ,DBAMV.PRO_FAT
                            ,DBAMV.GRU_PRO
                            ,DBAMV.SETOR
                            ,DBAMV.REMESSA_FATURA     REMESSA
                            ,DBAMV.FATURA
                            ,DBAMV.CONVENIO
                            ,DBAMV.ATENDIME
                            ,DBAMV.PRESTADOR
                            ,dbamv.itlan_med
               WHERE  REG_FAT.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
               AND        REG_FAT.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
               AND        ITREG_FAT.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
               AND        ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
               AND        PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
               AND        ITREG_FAT.CD_SETOR = SETOR.CD_SETOR(+)
               AND        REG_FAT.CD_REMESSA = REMESSA.CD_REMESSA
               AND        REMESSA.CD_FATURA = FATURA.CD_FATURA
               AND        REG_FAT.CD_CONVENIO = CONVENIO.CD_CONVENIO
               AND        PRO_FAT.SN_PACOTE = 'N'
               AND        NVL(REG_FAT.SN_DIAGNO,'N' ) =  'N'
               AND ITLAN_MED.CD_REG_FAT(+) = ITREG_FAT.CD_REG_FAT
aND ITLAN_MED.CD_LANCAMENTO(+) = ITREG_FAT.CD_LANCAMENTO
and nvl(nvl(itlan_med.tp_pagamento,itreg_fat.tp_pagamento),'X') <> 'C'
AND CONVENIO.TP_CONVENIO not in ( 'H', 'A')
               AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
               GROUP BY  PRESTADOR.NM_PRESTADOR
                         ,SETOR.NM_SETOR
                         ,REG_FAT.CD_REG_FAT
               UNION ALL
               SELECT  PRESTADOR.NM_PRESTADOR  NM_PRESTADOR
                             ,DECODE(SETOR.NM_SETOR, NULL, ' Não Informado', SETOR.NM_SETOR)    NM_SETOR
                             ,SUM(ITREG_AMB.VL_TOTAL_CONTA)          VL_TOTAL
                             ,REG_AMB.CD_REG_AMB CONTA
                FROM   DBAMV.REG_AMB
                             ,DBAMV.ITREG_AMB
                             ,DBAMV.PRO_FAT
                             ,DBAMV.GRU_PRO
                             ,DBAMV.SETOR
                             ,DBAMV.REMESSA_FATURA  REMESSA
                             ,DBAMV.FATURA
                             ,DBAMV.CONVENIO
                             ,DBAMV.ATENDIME
                             ,DBAMV.PRESTADOR
                WHERE  REG_AMB.CD_REG_AMB = ITREG_AMB.CD_REG_AMB
                AND        ITREG_AMB.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
                AND        ITREG_AMB.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
                AND        ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
                AND        PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
                AND        ITREG_AMB.CD_SETOR = SETOR.CD_SETOR(+)
                AND        REG_AMB.CD_REMESSA = REMESSA.CD_REMESSA
                AND        REMESSA.CD_FATURA = FATURA.CD_FATURA
                AND        ITREG_AMB.CD_CONVENIO = CONVENIO.CD_CONVENIO
                AND        PRO_FAT.SN_PACOTE = 'N'
                AND        NVL(REG_AMB.SN_DIAGNO,'N' ) =  'N'
                AND        NVL(ITREG_AMB.TP_PAGAMENTO,'X') <> 'C'
                AND        REG_AMB.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
                AND        CONVENIO.TP_CONVENIO  not in ( 'H', 'A')
                AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
                 GROUP BY  PRESTADOR.NM_PRESTADOR
                         ,SETOR.NM_SETOR
                         ,REG_AMB.CD_REG_AMB )
WHERE VL_TOTAL IS NOT NULL
GROUP BY  NM_PRESTADOR
                    ,NM_SETOR
                    ,CONTA) VALOR,
(SELECT         SUM(VL_TOTAL) VALOR
              ,CONTA      CONTA
FROM  ( SELECT  PRESTADOR.NM_PRESTADOR       NM_PRESTADOR
                            ,DECODE(SETOR.NM_SETOR, NULL, ' Não Informado', SETOR.NM_SETOR)        NM_SETOR
                            ,SUM(nvl(ITLAN_MED.VL_LIQUIDO,ITREG_FAT.VL_TOTAL_CONTA))       VL_TOTAL
                            ,REG_FAT.CD_REG_FAT   CONTA
               FROM   DBAMV.REG_FAT
                            ,DBAMV.ITREG_FAT
                            ,DBAMV.PRO_FAT
                            ,DBAMV.GRU_PRO
                            ,DBAMV.SETOR
                            ,DBAMV.REMESSA_FATURA     REMESSA
                            ,DBAMV.FATURA
                            ,DBAMV.CONVENIO
                            ,DBAMV.ATENDIME
                            ,DBAMV.PRESTADOR
                            ,dbamv.itlan_med
               WHERE  REG_FAT.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
               AND        REG_FAT.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
               AND        ITREG_FAT.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
               AND        ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
               AND        PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
               AND        ITREG_FAT.CD_SETOR = SETOR.CD_SETOR(+)
               AND        REG_FAT.CD_REMESSA = REMESSA.CD_REMESSA
               AND        REMESSA.CD_FATURA = FATURA.CD_FATURA
               AND        REG_FAT.CD_CONVENIO = CONVENIO.CD_CONVENIO
               AND        PRO_FAT.SN_PACOTE = 'N'
               AND        NVL(REG_FAT.SN_DIAGNO,'N' ) =  'N'
              AND ITLAN_MED.CD_REG_FAT(+) = ITREG_FAT.CD_REG_FAT
aND ITLAN_MED.CD_LANCAMENTO(+) = ITREG_FAT.CD_LANCAMENTO
and nvl(nvl(itlan_med.tp_pagamento,itreg_fat.tp_pagamento),'X') <> 'C'
AND CONVENIO.TP_CONVENIO not in ( 'H', 'A')
               AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
               GROUP BY  PRESTADOR.NM_PRESTADOR
                         ,SETOR.NM_SETOR
                         ,REG_FAT.CD_REG_FAT
               UNION ALL
               SELECT  PRESTADOR.NM_PRESTADOR  NM_PRESTADOR
                             ,DECODE(SETOR.NM_SETOR, NULL, ' Não Informado', SETOR.NM_SETOR)    NM_SETOR
                             ,SUM(ITREG_AMB.VL_TOTAL_CONTA)          VL_TOTAL
                             ,REG_AMB.CD_REG_AMB CONTA
                FROM   DBAMV.REG_AMB
                             ,DBAMV.ITREG_AMB
                             ,DBAMV.PRO_FAT
                             ,DBAMV.GRU_PRO
                             ,DBAMV.SETOR
                             ,DBAMV.REMESSA_FATURA  REMESSA
                             ,DBAMV.FATURA
                             ,DBAMV.CONVENIO
                             ,DBAMV.ATENDIME
                             ,DBAMV.PRESTADOR
                WHERE  REG_AMB.CD_REG_AMB = ITREG_AMB.CD_REG_AMB
                AND        ITREG_AMB.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
                AND        ITREG_AMB.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
                AND        ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
                AND        PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
                AND        ITREG_AMB.CD_SETOR = SETOR.CD_SETOR(+)
                AND        REG_AMB.CD_REMESSA = REMESSA.CD_REMESSA
                AND        REMESSA.CD_FATURA = FATURA.CD_FATURA
                AND        ITREG_AMB.CD_CONVENIO = CONVENIO.CD_CONVENIO
                AND        PRO_FAT.SN_PACOTE = 'N'
                AND        NVL(REG_AMB.SN_DIAGNO,'N' ) =  'N'
                AND        NVL(ITREG_AMB.TP_PAGAMENTO,'X') <> 'C'
                AND        REG_AMB.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
                AND        CONVENIO.TP_CONVENIO  not in ( 'H', 'A')
                AND        ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
                GROUP BY  PRESTADOR.NM_PRESTADOR
                         ,SETOR.NM_SETOR
                         ,REG_AMB.CD_REG_AMB )
WHERE VL_TOTAL IS NOT NULL
GROUP BY  CONTA ) TOTAL
WHERE TOTAL.CONTA = VALOR.CONTA) S
WHERE S.CONTA(+) = N.CONTA)
GROUP BY GRUPO,SETOR
)
ORDER BY NM_PRESTADOR,
NM_SETOR
