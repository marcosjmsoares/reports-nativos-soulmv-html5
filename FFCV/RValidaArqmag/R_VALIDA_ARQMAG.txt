--R_VALIDA_ARQMAG
select remessa_fatura.cd_remessa,
       fatura.cd_convenio,
       fatura.ds_fatura,
       convenio.nm_convenio,
       reg_fat.cd_reg_fat cd_conta,
       reg_fat.cd_atendimento,
       atendime.cd_paciente,
       paciente.nm_paciente,
       'Hospitalar' tipo
  from dbamv.reg_fat,
       dbamv.fatura,
       dbamv.convenio,
       dbamv.remessa_fatura,
       dbamv.atendime,
       dbamv.paciente
 where ( remessa_fatura.cd_remessa = :CAMPO_CD_REMESSA
          or remessa_fatura.cd_remessa_pai = :CAMPO_CD_REMESSA )
   and remessa_fatura.cd_fatura  = fatura.cd_fatura
   and reg_fat.cd_remessa        = remessa_fatura.cd_remessa
   and convenio.cd_convenio      = fatura.cd_convenio
   and reg_fat.cd_atendimento    = atendime.cd_atendimento
   and paciente.cd_paciente      = atendime.cd_paciente
   and reg_fat.sn_fechada        = 'N'
   and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
Union all
select distinct remessa_fatura.cd_remessa,
       fatura.cd_convenio,
       fatura.ds_fatura,
       convenio.nm_convenio,
       reg_amb.cd_reg_amb cd_conta,
       atendime.cd_atendimento,
       atendime.cd_paciente,
       paciente.nm_paciente,
       'Ambulatorial' tipo
  from dbamv.reg_amb,
       dbamv.itreg_amb,
       dbamv.fatura,
       dbamv.convenio,
       dbamv.remessa_fatura,
       dbamv.atendime,
       dbamv.paciente
 where ( remessa_fatura.cd_remessa = :CAMPO_CD_REMESSA
          or remessa_fatura.cd_remessa_pai = :CAMPO_CD_REMESSA )
   and remessa_fatura.cd_fatura  = fatura.cd_fatura
   and reg_amb.cd_remessa        = remessa_fatura.cd_remessa
   and convenio.cd_convenio      = fatura.cd_convenio
   and reg_amb.cd_reg_amb        = itreg_amb.cd_reg_amb
   and itreg_amb.cd_atendimento  = atendime.cd_atendimento
   and paciente.cd_paciente      = atendime.cd_paciente
   and itreg_amb.sn_fechada      = 'N'
   and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
 order by 1,8,5,6
