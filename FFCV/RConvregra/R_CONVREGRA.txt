--R_CONVREGRA
SELECT * FROM (
Select convenio.cd_convenio,
       convenio.nm_convenio,
       con_pla.cd_con_pla,
       con_pla.ds_con_pla,
       con_pla.cd_regra,
       regra.ds_regra,
       con_pla.cd_indice,
       indice.nm_indice,
       itregra.cd_horario,
       horario_especial.ds_horario,
       itregra.cd_gru_pro,
       gru_pro.ds_gru_pro,
       itregra.cd_tab_fat,
       tab_fat.ds_tab_fat
  From dbamv.regra,
       dbamv.convenio,
       dbamv.con_pla,
       dbamv.indice,
       dbamv.itregra,
       dbamv.tab_fat,
       dbamv.gru_pro,
       dbamv.horario_especial,
Dbamv.Empresa_Con_Pla
Where Empresa_Con_Pla.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And Empresa_Con_Pla.Cd_Convenio = Con_Pla.Cd_Convenio
And Empresa_Con_Pla.Cd_Con_Pla = Con_Pla.Cd_Con_Pla
And convenio.cd_convenio = con_pla.cd_convenio
   and con_pla.cd_regra = regra.cd_regra
   and con_pla.cd_indice = indice.cd_indice
   and regra.cd_regra = itregra.cd_regra
   and itregra.cd_horario = horario_especial.cd_horario (+)
   and itregra.cd_tab_fat = tab_fat.cd_tab_fat
   and itregra.cd_gru_pro = gru_pro.cd_gru_pro
   :CAMPO_CF_CONVENIO
)
ORDER BY NM_CONVENIO,CD_CONVENIO,
CD_CON_PLA,CD_REGRA,DS_REGRA,CD_INDICE,NM_INDICE,DS_CON_PLA
