--R_SUB_GRU_PRO_HIERARQUIZADO
SELECT * FROM (
SELECT  SUB_GRU.CD_SUB_GRU_PRO_HIERARQUIZADO  CODIGO
               ,SUB_GRU.DS_SUB_GRU_PRO_HIERARQUIZADO  DESCRICAO
               ,GRU_PRO.CD_GRU_PRO_HIERARQUIZADO
               ,GRU_PRO.DS_GRU_PRO_HIERARQUIZADO
FROM  dbamv.SUB_GRU_PRO_HIERARQUIZADO SUB_GRU
           ,dbamv.GRU_PRO_HIERARQUIZADO GRU_PRO
WHERE  GRU_PRO.CD_GRU_PRO_HIERARQUIZADO  =  SUB_GRU.CD_GRU_PRO_HIERARQUIZADO
:CAMPO_CF_COND_GRUPO
)
ORDER BY CD_GRU_PRO_HIERARQUIZADO,DS_GRU_PRO_HIERARQUIZADO,:CAMPO_ORDEM
