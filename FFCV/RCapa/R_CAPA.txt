SELECT DECODE('','F','',cd_remessa) cd_remessa
      ,cd_gru_fat
      ,ds_gru_fat
      ,tp_gru_fat
      ,cd_sequencia
      ,sum(vl_total_conta) vl_total_conta
  FROM (
Select  reg_fat.cd_remessa              cd_remessa
      , gru_fat.cd_gru_fat              cd_gru_fat
      , gru_fat.ds_gru_fat              ds_gru_fat
      , gru_fat.tp_gru_fat              tp_gru_fat
      , itfor_apre.cd_sequencia         cd_sequencia
      , sum( nvl(itlan_med.vl_liquido, nvl(itreg_fat.vl_total_conta, 0)) ) vl_total_conta
   From dbamv.reg_fat          reg_fat
      , dbamv.itreg_fat        itreg_fat
      , dbamv.gru_fat          gru_fat
      , dbamv.convenio         convenio
      , dbamv.for_apre         for_apre
      , dbamv.itfor_apre       itfor_apre
      , dbamv.empresa_convenio
      , dbamv.itlan_med
      , dbamv.remessa_fatura
      , dbamv.fatura
      , dbamv.atendime
  Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
    And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
    and reg_fat.cd_reg_fat  =  itreg_fat.cd_reg_fat
    and atendime.cd_atendimento	= reg_fat.cd_atendimento
    and itreg_fat.cd_gru_fat     = gru_Fat.cd_gru_fat
    and nvl(itreg_fat.sn_pertence_pacote, 'N') = 'N'
    and reg_fat.cd_convenio  =  convenio.cd_convenio
    and convenio.cd_for_apre  =  for_apre.cd_for_apre
    and for_apre.cd_for_apre  =  itfor_apre.cd_for_apre
    and gru_fat.cd_gru_fat = itfor_apre.cd_gru_fat
    and itlan_med.cd_reg_fat(+)    = itreg_fat.cd_reg_fat
    and itlan_med.cd_lancamento(+) = itreg_fat.cd_lancamento
    and nvl(itlan_med.tp_pagamento, nvl(itreg_fat.tp_pagamento,'X')) <> 'C'
    and fatura.cd_fatura   = remessa_fatura.cd_fatura
    and reg_fat.cd_remessa = remessa_fatura.cd_remessa
  Group by reg_fat.cd_remessa
      , gru_fat.cd_gru_fat
      , gru_fat.ds_gru_fat
      , gru_fat.tp_gru_fat
      , itfor_apre.cd_sequencia
  Union all
 Select reg_amb.cd_remessa               cd_remessa
      , gru_fat.cd_gru_fat               cd_gru_fat
      , gru_fat.ds_gru_fat               ds_gru_fat
      , gru_fat.tp_gru_fat               tp_gru_fat
      , itfor_apre.cd_sequencia          cd_sequencia
      , sum( nvl(itreg_amb.vl_total_conta, 0) )  vl_total_conta
   From dbamv.itreg_amb        itreg_amb
      , dbamv.reg_amb          reg_amb
      , dbamv.gru_fat          gru_fat
      , dbamv.convenio         convenio
      , dbamv.for_apre         for_apre
      , dbamv.itfor_apre       itfor_apre
      , dbamv.empresa_convenio
      , dbamv.remessa_fatura
      , dbamv.fatura
      , dbamv.atendime
  Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
    And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
    and itreg_amb.sn_pertence_pacote  =  'N'
    and nvl(itreg_amb.tp_pagamento,'X') <> 'C'
    and itreg_amb.cd_reg_amb  =  reg_amb.cd_reg_amb
	and itreg_amb.cd_atendimento = atendime.cd_atendimento
    AND REG_amb.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
    and itreg_amb.cd_gru_fat  =  gru_fat.cd_gru_fat
    and reg_amb.cd_convenio  =  convenio.cd_convenio
    and convenio.cd_for_apre = for_apre.cd_for_apre
    and for_apre.cd_for_apre  =  itfor_apre.cd_for_apre
    and gru_fat.cd_gru_fat  =  itfor_apre.cd_gru_fat
    and fatura.cd_fatura   = remessa_fatura.cd_fatura
    and reg_amb.cd_remessa = remessa_fatura.cd_remessa
  Group by  reg_amb.cd_remessa
          , gru_fat.cd_gru_fat
          , gru_fat.ds_gru_fat
          , gru_fat.tp_gru_fat
          , itfor_apre.cd_sequencia
  order by cd_sequencia
)
GROUP BY DECODE('','F','',cd_remessa)
      ,cd_gru_fat
      ,ds_gru_fat
      ,tp_gru_fat
      ,cd_sequencia
ORDER BY CD_GRU_FAT,DS_GRU_FAT,TP_GRU_FAT
