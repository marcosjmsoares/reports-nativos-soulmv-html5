--R_LIGACAO_PACIENTE
SELECT * FROM (
     select atendime.cd_atendimento cd_atendimento
          , paciente.nm_paciente nm_paciente
          , ligacao.nr_ramal nr_ramal
          , ligacao.nr_ddd nr_ddd
          , ligacao.nr_ddi nr_ddi
          , ligacao.nr_telefone nr_telefone
          , ligacao.dt_ligacao dt_ligacao
          , to_char(ligacao.qt_ligacao_minuto,'FM09')||':'||to_char(ligacao.qt_ligacao_segundo,'FM09') duracao
          , ligacao.cd_ligacao_paciente
          , itreg_fat.cd_reg_fat cd_reg_fat
          , sum(itreg_fat.vl_total_conta) valor
       from dbamv.ligacao_paciente ligacao
          , dbamv.atendime
          , dbamv.paciente
          , dbamv.itreg_fat
      where atendime.cd_atendimento = ligacao.cd_atendimento
        and paciente.cd_paciente = atendime.cd_paciente
        and itreg_fat.cd_mvto = ligacao.cd_ligacao_paciente
        and upper(itreg_fat.tp_mvto) = 'LIGAÇÃO'
        and trunc(ligacao.dt_ligacao) between trunc(:CAMPO_DTINIC) and trunc(:CAMPO_DTFIM)
        and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
     group by atendime.cd_atendimento
          , paciente.nm_paciente
          , ligacao.nr_ramal
          , ligacao.nr_ddd
          , ligacao.nr_ddi
          , ligacao.nr_telefone
          , ligacao.dt_ligacao
          , to_char(ligacao.qt_ligacao_minuto,'FM09')||':'||to_char(ligacao.qt_ligacao_segundo,'FM09')
          , ligacao.cd_ligacao_paciente
          , itreg_fat.cd_reg_fat
    ORDER BY ligacao.dt_ligacao
)
ORDER BY CD_ATENDIMENTO,NM_PACIENTE,
CD_REG_FAT,NR_DDD,NR_RAMAL,NR_DDI,DT_LIGACAO,NR_TELEFONE,DURACAO,CD_LIGACAO_PACIENTE,VALOR
