--R_FAT_BAIRRO_DT_LANC
SELECT *
   FROM
  (SELECT nm_bairro bairro_faixa                         ,
    cd_faixa_etaria_relatorio                            ,
    ds_faixa_etaria_relatorio                            ,
    SUM(DECODE( tp_sexo,'M',qt_paciente,0)) qt_paciente_M,
    SUM(DECODE( tp_sexo,'F',qt_paciente,0)) qt_paciente_F
     FROM
    (SELECT reg_fat.cd_convenio                                                            ,
      paciente.nm_bairro                                                                   ,
      paciente.tp_sexo                                                                     ,
      RETORNA_FAIXA_ETARIA_RELATORIO(paciente.dt_nascimento,'C') cd_faixa_etaria_relatorio ,
      ds_faixa_etaria_relatorio                                                            ,
      COUNT(DISTINCT atendime.cd_atendimento) qt_paciente
       FROM dbamv.itreg_fat        ,
      dbamv.reg_fat                ,
      dbamv.faixa_etaria_relatorio ,
      dbamv.convenio               ,
      dbamv.atendime               ,
      dbamv.paciente               ,
      Dbamv.Empresa_Convenio
      WHERE Empresa_Convenio.Cd_Convenio  = Convenio.Cd_Convenio
    AND Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
    AND itreg_fat.cd_reg_fat              = reg_fat.cd_reg_fat 
    AND itreg_fat.dt_lancamento BETWEEN to_date(:CAMPO_DT_INICIO,'dd/mm/yyyy') AND (to_date(:CAMPO_DT_FINAL,'dd/mm/yyyy') + 0.9)
    AND paciente.cd_paciente                             = atendime.cd_paciente
    AND faixa_etaria_relatorio.cd_faixa_etaria_relatorio = RETORNA_FAIXA_ETARIA_RELATORIO(paciente.dt_nascimento,'C')
    AND itreg_fat.sn_pertence_pacote                     = 'N'
    AND NVL(reg_fat.sn_diagno, 'N')                      = 'N'
    AND NVL(itreg_fat.tp_pagamento,'X')                 <> 'C'
    AND reg_fat.cd_convenio                              = convenio.cd_convenio
    AND convenio.tp_convenio                            <> 'H'
    AND reg_fat.cd_atendimento                           = atendime.cd_atendimento
    AND atendime.cd_multi_empresa                        = :CAMPO_CD_MULTI_EMPRESA 
   GROUP BY reg_fat.cd_convenio                                  ,
      paciente.nm_bairro                                         ,
      paciente.tp_sexo                                           ,
      RETORNA_FAIXA_ETARIA_RELATORIO(paciente.dt_nascimento,'C') ,
      ds_faixa_etaria_relatorio
  UNION ALL
     SELECT reg_amb.cd_convenio                                                            ,
      paciente.nm_bairro                                                                   ,
      paciente.tp_sexo                                                                     ,
      RETORNA_FAIXA_ETARIA_RELATORIO(paciente.dt_nascimento,'C') cd_faixa_etaria_relatorio ,
      ds_faixa_etaria_relatorio                                                            ,
      COUNT(DISTINCT atendime.cd_atendimento) qt_paciente
       FROM dbamv.itreg_amb itreg_amb                     ,
      dbamv.reg_amb reg_amb                               ,
      dbamv.faixa_etaria_relatorio faixa_etaria_relatorio ,
      dbamv.convenio convenio                             ,
      dbamv.atendime atendime                             ,
      dbamv.paciente paciente                             ,
      Dbamv.Empresa_Convenio
      WHERE Empresa_Convenio.Cd_Convenio  = Convenio.Cd_Convenio
    AND Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
    AND itreg_amb.cd_reg_amb              = reg_amb.cd_reg_amb 
    AND atendime.dt_atendimento BETWEEN to_date(:CAMPO_DT_INICIO,'dd/mm/yyyy') AND (to_date(:CAMPO_DT_FINAL,'dd/mm/yyyy') + 0.9)
    AND paciente.cd_paciente                             = atendime.cd_paciente
    AND itreg_amb.sn_pertence_pacote                     = 'N'
    AND faixa_etaria_relatorio.cd_faixa_etaria_relatorio = RETORNA_FAIXA_ETARIA_RELATORIO(paciente.dt_nascimento,'C')
    AND NVL(reg_amb.sn_diagno, 'N')                      = 'N'
    AND NVL(itreg_amb.tp_pagamento,'X')                 <> 'C'
    AND reg_amb.cd_multi_empresa                         = :CAMPO_CD_MULTI_EMPRESA
    AND reg_amb.cd_convenio                              = convenio.cd_convenio
    AND convenio.tp_convenio                            <> 'H'
    AND itreg_amb.cd_atendimento                         = atendime.cd_atendimento
    AND atendime.cd_multi_empresa                        = :CAMPO_CD_MULTI_EMPRESA 
   GROUP BY reg_amb.cd_convenio                                  ,
      paciente.nm_bairro                                         ,
      paciente.tp_sexo                                           ,
      RETORNA_FAIXA_ETARIA_RELATORIO(paciente.dt_nascimento,'C') ,
      ds_faixa_etaria_relatorio
  UNION ALL
     SELECT reg_fat.cd_convenio                                                            ,
      paciente.nm_bairro                                                                   ,
      paciente.tp_sexo                                                                     ,
      RETORNA_FAIXA_ETARIA_RELATORIO(paciente.dt_nascimento,'C') cd_faixa_etaria_relatorio ,
      ds_faixa_etaria_relatorio                                                            ,
      COUNT(DISTINCT atendime.cd_atendimento) qt_paciente
       FROM dbamv.itreg_fat        ,
      dbamv.reg_fat                ,
      dbamv.faixa_etaria_relatorio ,
      dbamv.convenio               ,
      dbamv.atendime               ,
      dbamv.paciente               ,
      Dbamv.Empresa_Convenio
      WHERE Empresa_Convenio.Cd_Convenio  = Convenio.Cd_Convenio
    AND Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
    AND itreg_fat.cd_reg_fat              = reg_Fat.cd_reg_fat 
    AND itreg_fat.dt_lancamento BETWEEN to_date(:CAMPO_DT_INICIO,'dd/mm/yyyy') AND (to_date(:CAMPO_DT_FINAL,'dd/mm/yyyy') + 0.9)
    AND paciente.cd_paciente                             = atendime.cd_paciente
    AND faixa_etaria_relatorio.cd_faixa_etaria_relatorio = RETORNA_FAIXA_ETARIA_RELATORIO(paciente.dt_nascimento,'C')
    AND itreg_fat.sn_pertence_pacote                     = 'N'
    AND NVL(reg_fat.sn_diagno, 'N')                      = 'N'
    AND reg_fat.cd_convenio                              = convenio.cd_convenio
    AND convenio.tp_convenio                            <> 'H'
    AND reg_fat.cd_atendimento                           = atendime.cd_atendimento
    AND atendime.cd_multi_empresa                        = :CAMPO_CD_MULTI_EMPRESA 
   GROUP BY reg_fat.cd_convenio                                  ,
      paciente.nm_bairro                                         ,
      paciente.tp_sexo                                           ,
      RETORNA_FAIXA_ETARIA_RELATORIO(paciente.dt_nascimento,'C') ,
      ds_faixa_etaria_relatorio
  UNION ALL
     SELECT reg_amb.cd_convenio                                                            ,
      paciente.nm_bairro                                                                   ,
      paciente.tp_sexo                                                                     ,
      RETORNA_FAIXA_ETARIA_RELATORIO(paciente.dt_nascimento,'C') cd_faixa_etaria_relatorio ,
      ds_faixa_etaria_relatorio                                                            ,
      COUNT(DISTINCT atendime.cd_atendimento) qt_paciente
       FROM dbamv.itreg_amb itreg_amb                     ,
      dbamv.reg_amb reg_amb                               ,
      dbamv.faixa_etaria_relatorio faixa_etaria_relatorio ,
      dbamv.convenio convenio                             ,
      dbamv.atendime atendime                             ,
      dbamv.paciente paciente                             ,
      Dbamv.Empresa_Convenio
      WHERE Empresa_Convenio.Cd_Convenio  = Convenio.Cd_Convenio
    AND Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
    AND itreg_amb.cd_reg_amb              = reg_amb.cd_reg_amb 
    AND atendime.dt_atendimento BETWEEN to_date(:CAMPO_DT_INICIO,'dd/mm/yyyy') AND (to_date(:CAMPO_DT_FINAL,'dd/mm/yyyy') + 0.9)
    AND paciente.cd_paciente                             = atendime.cd_paciente
    AND faixa_etaria_relatorio.cd_faixa_etaria_relatorio = RETORNA_FAIXA_ETARIA_RELATORIO(paciente.dt_nascimento,'C')
    AND itreg_amb.sn_pertence_pacote                     = 'N'
    AND NVL(reg_amb.sn_diagno, 'N')                      = 'N'
    AND reg_amb.cd_multi_empresa                         = :CAMPO_CD_MULTI_EMPRESA
    AND reg_amb.cd_convenio                              = convenio.cd_convenio
    AND convenio.tp_convenio                            <> 'H'
    AND itreg_amb.cd_atendimento                         = atendime.cd_atendimento
    AND atendime.cd_multi_empresa                        = :CAMPO_CD_MULTI_EMPRESA 
   GROUP BY reg_amb.cd_convenio                                  ,
      paciente.nm_bairro                                         ,
      paciente.tp_sexo                                           ,
      RETORNA_FAIXA_ETARIA_RELATORIO(paciente.dt_nascimento,'C') ,
      ds_faixa_etaria_relatorio
    ) convenio 
 GROUP BY nm_bairro          ,
    cd_faixa_etaria_relatorio,
    ds_faixa_etaria_relatorio
 ORDER BY nm_bairro,
    cd_faixa_etaria_relatorio
  )
ORDER BY BAIRRO_FAIXA,
  DS_FAIXA_ETARIA_RELATORIO

--R_FAT_BAIRRO_DT_LANC_R_VALORES1_Q_FAT_NM_BAIRRO
Select nm_bairro,
       SUM(decode(tp_atendimento,'H',qt_total_conta,'I',qt_total_conta,0)) qt_total_conta_INT,
       SUM(decode(tp_atendimento,'A',qt_total_conta,0)) qt_total_conta_AMB,
       SUM(decode(tp_atendimento,'E',qt_total_conta,0)) qt_total_conta_EME,
       SUM(decode(tp_atendimento,'U',qt_total_conta,0)) qt_total_conta_URG,
       SUM(decode(tp_atendimento,'H',vl_total_conta,'I',vl_total_conta,0)) vl_total_conta_INT,
       SUM(decode(tp_atendimento,'A',vl_total_conta,0)) vl_total_conta_AMB,
       SUM(decode(tp_atendimento,'E',vl_total_conta,0)) vl_total_conta_EME,
       SUM(decode(tp_atendimento,'U',vl_total_conta,0)) vl_total_conta_URG
FROM     ( SELECT reg_fat.cd_convenio
                              , paciente.nm_bairro
                              , atendime.tp_atendimento
                              , sum(NVL( itreg_fat.vl_total_conta,0) )  vl_total_conta
                              , count(distinct atendime.cd_atendimento)  qt_total_conta
                  FROM   dbamv.itreg_fat
                             , dbamv.reg_fat
                             , dbamv.convenio
                             , dbamv.atendime
                             , dbamv.paciente
			     ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And itreg_fat.cd_reg_fat     = reg_fat.cd_reg_fat
                  and         paciente.cd_paciente     = atendime.cd_paciente
                  and         itreg_fat.dt_lancamento between  to_date(:CAMPO_DT_INICIO,'dd/mm/yyyy')  and  (to_date(:CAMPO_DT_FINAL,'dd/mm/yyyy') + 0.9)
                  and         itreg_fat.sn_pertence_pacote = 'N'
                  and         NVL(reg_fat.sn_diagno, 'N') = 'N'
                  and         nvl(itreg_fat.tp_pagamento,'X') <> 'C'
                  and         reg_fat.cd_convenio = convenio.cd_convenio
                  and         convenio.tp_convenio <> 'H'
                  and         reg_fat.cd_atendimento = atendime.cd_atendimento
                  and         atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
                  GROUP BY reg_fat.cd_convenio
                                    , paciente.nm_bairro
                                    , atendime.tp_atendimento
               UNION ALL
               SELECT reg_amb.cd_convenio
                            , paciente.nm_bairro
                            , atendime.tp_atendimento
                            , sum(NVL( itreg_amb.vl_total_conta,0) )  vl_total_conta
                            , count(distinct atendime.cd_atendimento )  qt_total_conta
               FROM         dbamv.itreg_amb             itreg_amb
                          , dbamv.reg_amb               reg_amb
                           , dbamv.convenio              convenio
                          , dbamv.atendime              atendime
                          , dbamv.paciente              paciente
			  ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And itreg_amb.cd_reg_amb     = reg_amb.cd_reg_amb
               and    paciente.cd_paciente     = atendime.cd_paciente
               and   atendime.dt_atendimento between  to_date(:CAMPO_DT_INICIO,'dd/mm/yyyy')  and  (to_date(:CAMPO_DT_FINAL,'dd/mm/yyyy') + 0.9)
               and    itreg_amb.sn_pertence_pacote = 'N'
               and    NVL(reg_amb.sn_diagno, 'N') = 'N'
               and    nvl(itreg_amb.tp_pagamento,'X') <> 'C'
               and    reg_amb.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
               and    reg_amb.cd_convenio = convenio.cd_convenio
               and    convenio.tp_convenio <> 'H'
               and    itreg_amb.cd_atendimento = atendime.cd_atendimento
               and    atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
               GROUP BY  reg_amb.cd_convenio
                             , paciente.nm_bairro
                             , atendime.tp_atendimento
         UNION ALL
              SELECT  reg_fat.cd_convenio
                            , paciente.nm_bairro
                            , atendime.tp_atendimento
                            , sum(NVL( itreg_fat.vl_total_conta,0) )  vl_total_conta
                            , count(distinct atendime.cd_atendimento )  qt_total_conta
              FROM         dbamv.itreg_fat
                         , dbamv.reg_fat
                         , dbamv.convenio
                         , dbamv.atendime
                         , dbamv.paciente
			 ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And itreg_fat.cd_reg_fat     = reg_Fat.cd_reg_fat
              and         paciente.cd_paciente     = atendime.cd_paciente
              and         itreg_fat.dt_lancamento between  to_date(:CAMPO_DT_INICIO,'dd/mm/yyyy')  and  (to_date(:CAMPO_DT_FINAL,'dd/mm/yyyy') + 0.9)
              and          itreg_fat.sn_pertence_pacote = 'N'
              and         NVL(reg_fat.sn_diagno, 'N') = 'N'
              and         reg_fat.cd_convenio = convenio.cd_convenio
              and         convenio.tp_convenio <> 'H'
              and         reg_fat.cd_atendimento = atendime.cd_atendimento
              and         atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
              GROUP BY  reg_fat.cd_convenio
                                 , paciente.nm_bairro
                                 , atendime.tp_atendimento
              UNION ALL
               SELECT reg_amb.cd_convenio
                            , paciente.nm_bairro
                            , atendime.tp_atendimento
                            , sum(NVL( itreg_amb.vl_total_conta,0) )  vl_total_conta
                            , count(distinct atendime.cd_atendimento )  qt_total_conta
               FROM         dbamv.itreg_amb             itreg_amb
                          , dbamv.reg_amb               reg_amb
                          , dbamv.convenio              convenio
                          , dbamv.atendime              atendime
                          , dbamv.paciente              paciente
			  ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And itreg_amb.cd_reg_amb     = reg_amb.cd_reg_amb
               and   atendime.dt_atendimento between  to_date(:CAMPO_DT_INICIO,'dd/mm/yyyy')  and  (to_date(:CAMPO_DT_FINAL,'dd/mm/yyyy') + 0.9)
               and    paciente.cd_paciente     = atendime.cd_paciente
               and    itreg_amb.sn_pertence_pacote = 'N'
               and    NVL(reg_amb.sn_diagno, 'N') = 'N'
               and    reg_amb.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
               and    reg_amb.cd_convenio = convenio.cd_convenio
               and    convenio.tp_convenio <> 'H'
               and    itreg_amb.cd_atendimento = atendime.cd_atendimento
               and    atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
               GROUP BY         reg_amb.cd_convenio
                             , paciente.nm_bairro
                             , atendime.tp_atendimento  )   convenio
Group by  nm_bairro
Order By  nm_bairro
