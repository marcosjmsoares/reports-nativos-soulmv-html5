--R_HISTORICO_COMPETENCIA
select distinct
          to_char(dbamv.log_competencia_faturamento.dh_movimento, 'dd/mm/yyyy') data,
          log_competencia_faturamento.dt_competencia,
          convenio.nm_convenio,
          convenio.cd_convenio,
          log_competencia_faturamento.ds_movimento,
          to_char(dbamv.log_competencia_faturamento.dh_movimento, 'hh24') hora,
          usuarios.nm_usuario
 from dbamv.log_competencia_faturamento,
         dbamv.convenio,
         dbasgu.usuarios
where dbasgu.usuarios.cd_usuario = dbamv.log_competencia_faturamento.cd_usuario
  and dbamv.convenio.cd_convenio = dbamv.log_competencia_faturamento.cd_convenio
order by 1 desc, 2, 3
