--R_NOTA_FISC_PART_661
SELECT * FROM (
SELECT NOTA_FISCAL.CD_NOTA_FISCAL                        CD_NOTA_FISCAL,
               NOTA_FISCAL.CD_ATENDIMENTO                     CD_ATENDIMENTO,
               NOTA_FISCAL.NM_CLIENTE                                NM_CLIENTE,
               NOTA_FISCAL.NR_CGC_CPF                               NR_CGC_CPF,
               NOTA_FISCAL.NM_BAIRRO                                 NM_BAIRRO,
               NOTA_FISCAL.DS_ENDERECO                           DS_ENDERECO,
               NOTA_FISCAL.NM_CIDADE                                 NM_CIDADE,
               NOTA_FISCAL.NM_UF                                          NM_UF,
               NOTA_FISCAL.NR_INSCRICAO_ESTADUAL       NR_INSCRICAO_ESTADUAL,
               NOTA_FISCAL.DT_EMISSAO                                DT_EMISSAO,
               NOTA_FISCAL.NR_ID_NOTA_FISCAL                  NR_ID_NOTA_FISCAL,
               NOTA_FISCAL.DS_OBSERVACOES                     DS_OBSERVACOES,
               ITNOTA_FISCAL.CD_GRU_FAT                            CD_GRU_FAT,
               ITNOTA_FISCAL.VL_GRU_FAT                             VL_GRU_FAT,
               GRU_FAT.DS_GRU_FAT                                       DS_GRU_FAT,
               nota_fiscal.cd_reg_fat                                             CD_REG_FAT,
               nota_fiscal.cd_reg_amb                                           CD_REG_AMB,
               nota_fiscal.vl_total_nota                                           VL_TOTAL_NOTA
FROM   DBAMV.NOTA_FISCAL                                            NOTA_FISCAL,
             DBAMV.ITNOTA_FISCAL                                         ITNOTA_FISCAL,
             DBAMV.GRU_FAT                                                   GRU_FAT
WHERE  NOTA_FISCAL.CD_NOTA_FISCAL =   ITNOTA_FISCAL.CD_NOTA_FISCAL
AND        GRU_FAT.CD_GRU_FAT                =   ITNOTA_FISCAL.CD_GRU_FAT
AND        NOTA_FISCAL.CD_NOTA_FISCAL =   :CAMPO_CD_NOTA_FISCAL
AND    Nota_Fiscal.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
ORDER BY ITNOTA_FISCAL.CD_GRU_FAT
)
ORDER BY CD_NOTA_FISCAL,NM_CLIENTE,DS_ENDERECO,NM_BAIRRO,NM_CIDADE,NM_UF,NR_CGC_CPF,NR_INSCRICAO_ESTADUAL,DT_EMISSAO,NR_ID_NOTA_FISCAL,CD_REG_AMB,CD_REG_FAT,CD_ATENDIMENTO,VL_TOTAL_NOTA,DS_OBSERVACOES
