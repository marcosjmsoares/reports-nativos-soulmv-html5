SELECT distinct ATENDIME.CD_ATENDIMENTO
      ,PACIENTE.NM_PACIENTE
      ,DECODE(PACIENTE.TP_SEXO,'M','(X) M','( ) M')  SEXO_MAS
      ,DECODE(PACIENTE.TP_SEXO,'F' ,'(X) F','( ) F')    SEXO_FEM
      ,PACIENTE.DT_NASCIMENTO
      ,DT_ATENDIMENTO DATA
      ,ATENDIME.DT_ALTA
      ,ATENDIME.CD_CID
      ,CID.DS_CID
      ,REG_FAT.CD_REG_FAT
      ,HOSPITAL.DS_RAZAO
      ,ATENDIME.CD_PACIENTE
      ,ATENDIME.CD_CONVENIO
      ,ATENDIME.CD_CON_PLA
  FROM DBAMV.ATENDIME,
       DBAMV.PACIENTE,
       DBAMV.CID,
       DBAMV.REG_FAT,
       DBAMV.HOSPITAL
 WHERE ATENDIME.CD_PACIENTE               =   PACIENTE.CD_PACIENTE
   AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   AND     CID.CD_CID(+)                                   =   ATENDIME.CD_CID
   AND     REG_FAT.CD_ATENDIMENTO         =   ATENDIME.CD_ATENDIMENTO
   AND     ATENDIME.CD_ATENDIMENTO      =  :CAMPO_ATENDIME
   AND     REG_FAT.CD_REG_FAT                  =   TO_NUMBER( :CAMPO_CD_REG_FAT)
   and Hospital.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
