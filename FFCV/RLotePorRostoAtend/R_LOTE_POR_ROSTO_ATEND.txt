SELECT * FROM (
SELECT
       ATENDIME.CD_ORI_ATE,
       ATENDIME.CD_ATENDIMENTO CD_ATENDIMENTO,
       ATENDIME.CD_PACIENTE ,
       PACIENTE.NM_PACIENTE,
       GUIA.NR_GUIA,
       REG_FAT.DT_INICIO,
       REG_FAT.DT_FINAL,
       CONVENIO.CD_CONVENIO,
       CONVENIO.NM_CONVENIO,
       NVL( REG_FAT.VL_TOTAL_CONTA, 0)  VL_TOTAL_CONTA,
       'H'  tp_agrupamento,
       ITAGRUPAMENTO.CD_AGRUPAMENTO         CD_AGRUPAMENTO,
       AGRUPAMENTO.DS_AGRUPAMENTO            DS_AGRUPAMENTO,
       ORI_ATE.CD_ORI_ATE CD_ORI_ATE1,
       ORI_ATE.DS_ORI_ATE,
       FATURA.DT_COMPETENCIA,
       0                                   CD_LANCAMENTO,
       'I'                                 TIPO_GUIA,
       CONVENIO.CD_CONVENIO                CD_CONVENIO_GUIA,
       REMESSA_FATURA.DT_ENTREGA_DA_FATURA,
       ITREG_FAT.TP_PAGAMENTO
FROM   DBAMV.ATENDIME,
       DBAMV.PACIENTE,
       DBAMV.CONVENIO,
       DBAMV.REG_FAT,
       DBAMV.ITREG_FAT,
       DBAMV.ORI_ATE,
       DBAMV.REMESSA_FATURA,
       DBAMV.FATURA,
       DBAMV.ITAGRUPAMENTO                                ITAGRUPAMENTO,
       DBAMV.AGRUPAMENTO                                   AGRUPAMENTO,
       DBAMV.GUIA
WHERE  REMESSA_FATURA.CD_REMESSA         =  :CAMPO_CD_REMESSA
AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND        ATENDIME.CD_PACIENTE          = PACIENTE.CD_PACIENTE
AND        ATENDIME.CD_ATENDIMENTO       = REG_FAT.CD_ATENDIMENTO
AND       REG_FAT.CD_CONVENIO            = CONVENIO.CD_CONVENIO
AND       REG_FAT.CD_REG_FAT              = ITREG_FAT.CD_REG_FAT
AND       REG_FAT.CD_REMESSA             = REMESSA_FATURA.CD_REMESSA
AND       REMESSA_FATURA.CD_FATURA       = FATURA.CD_FATURA
AND       ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND        ATENDIME.CD_PACIENTE           = GUIA.CD_PACIENTE(+)
AND        ATENDIME.CD_CONVENIO           = GUIA.CD_CONVENIO(+)
AND       CONVENIO.CD_CONVENIO                       =       ITAGRUPAMENTO.CD_CONVENIO
and         atendime.cd_ori_Ate = ori_ate.cd_ori_ate
AND        ORI_ATE.CD_ORI_ATE                          =     ITAGRUPAMENTO.CD_ORI_ATE
AND       ITAGRUPAMENTO.CD_AGRUPAMENTO =      AGRUPAMENTO.CD_AGRUPAMENTO
AND        ITAGRUPAMENTO.TP_AGRUPAMENTO    =   'H'
AND        ATENDIME.CD_ATENDIMENTO = GUIA.CD_ATENDIMENTO(+)
and         remessa_fatura.cd_agrupamento is null
and         (guia.cd_guia is null or guia.cd_guia = (select min(cd_guia)
                                                    from dbamv.guia guia2
                                                    where cd_paciente = atendime.cd_paciente
                                                     and  cd_convenio = atendime.cd_convenio
                                                     and  cd_atendimento = atendime.cd_atendimento ) )
Union All
 SELECT
       ATENDIME.CD_ORI_ATE,
       ATENDIME.CD_ATENDIMENTO CD_ATENDIMENTO,
       ATENDIME.CD_PACIENTE,
       PACIENTE.NM_PACIENTE,
       GUIA.NR_GUIA,
       ATENDIME.dt_atendimento,
       atendime.dt_atendimento dt_alta,
       CONVENIO.CD_CONVENIO,
       CONVENIO.NM_CONVENIO,
       SUM( NVL( ITREG_AMB.VL_TOTAL_CONTA, 0 ) ) vl_total_conta,
       'A'  tp_agrupamento,
       ITAGRUPAMENTO.CD_AGRUPAMENTO,
       AGRUPAMENTO.DS_AGRUPAMENTO,
       ORI_ATE.CD_ORI_ATE CD_ORI_ATE1,
       ORI_ATE.DS_ORI_ATE,
       FATURA.DT_COMPETENCIA,
       0                                   CD_LANCAMENTO,
       'C'                                 TIPO_GUIA,
       CONVENIO.CD_CONVENIO                CD_CONVENIO_GUIA,
       REMESSA_FATURA.DT_ENTREGA_DA_FATURA,
       ITREG_AMB.TP_PAGAMENTO
FROM   DBAMV.ATENDIME,
       DBAMV.PACIENTE,
       DBAMV.CONVENIO,
       DBAMV.REG_AMB,
       DBAMV.ITREG_AMB,
       DBAMV.ORI_ATE,
       DBAMV.REMESSA_FATURA,
       DBAMV.FATURA,
       DBAMV.ITAGRUPAMENTO                                ITAGRUPAMENTO,
       DBAMV.AGRUPAMENTO                                   AGRUPAMENTO,
       DBAMV.GUIA
WHERE  REMESSA_FATURA.CD_REMESSA         =  :CAMPO_CD_REMESSA
AND REG_AMB.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND        ATENDIME.CD_PACIENTE          = PACIENTE.CD_PACIENTE
AND        ATENDIME.CD_ATENDIMENTO       = itreg_amb.CD_ATENDIMENTO
AND        ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND       REG_AMB.CD_CONVENIO            = CONVENIO.CD_CONVENIO
AND       REG_AMB.CD_REMESSA             = REMESSA_FATURA.CD_REMESSA
AND       REMESSA_FATURA.CD_FATURA       = FATURA.CD_FATURA
AND        ATENDIME.CD_PACIENTE           = GUIA.CD_PACIENTE(+)
AND        ATENDIME.CD_CONVENIO           = GUIA.CD_CONVENIO(+)
AND       CONVENIO.CD_CONVENIO                       =       ITAGRUPAMENTO.CD_CONVENIO
and         atendime.cd_ori_Ate = ori_ate.cd_ori_ate
AND        ORI_ATE.CD_ORI_ATE                          =     ITAGRUPAMENTO.CD_ORI_ATE
AND       ITAGRUPAMENTO.CD_AGRUPAMENTO =      AGRUPAMENTO.CD_AGRUPAMENTO
AND        ITAGRUPAMENTO.TP_AGRUPAMENTO    =   'A'
AND        ATENDIME.CD_ATENDIMENTO = GUIA.CD_ATENDIMENTO(+)
and       reg_amb.cd_reg_amb = itreg_amb.cd_reg_amb
and         remessa_fatura.cd_agrupamento is null
and      Nvl( itreg_amb.sn_pertence_pacote , 'N' ) = 'N'
and     Nvl( itreg_amb.tp_pagamento, 'P') <> 'C'
and         (guia.cd_guia is null or guia.cd_guia = (select min(guia2.cd_guia)
                                                    from dbamv.guia guia2
                                                    , dbamv.atendime
							where guia2.cd_atendimento = atendime.cd_atendimento
							and atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
							and  guia2.cd_paciente = atendime.cd_paciente
                                                     and  guia2.cd_convenio = atendime.cd_convenio
                                                     and  guia2.cd_atendimento = atendime.cd_atendimento ) )
GROUP BY     ATENDIME.CD_ATENDIMENTO,
                        ATENDIME.CD_ORI_ATE,
                        ATENDIME.CD_PACIENTE,
                       PACIENTE.NM_PACIENTE,
                       ATENDIME.DT_ATENDIMENTO,
                       ATENDIME.DT_ATENDIMENTO,
                      CONVENIO.CD_CONVENIO,
                      CONVENIO.NM_CONVENIO,
                      ORI_ATE.CD_ORI_ATE,
                      ORI_ATE.DS_ORI_ATE,
       ITAGRUPAMENTO.CD_AGRUPAMENTO,
       AGRUPAMENTO.DS_AGRUPAMENTO,
                      FATURA.DT_COMPETENCIA,
                      GUIA.NR_GUIA	,
                      REMESSA_FATURA.DT_ENTREGA_DA_FATURA,
                      ITREG_AMB.TP_PAGAMENTO
Union All
SELECT
       ATENDIME.CD_ORI_ATE                                  CD_ORI_ATE,
       ATENDIME.CD_ATENDIMENTO                       CD_ATENDIMENTO,
       ATENDIME.CD_PACIENTE                                CD_PACIENTE,
       PACIENTE.NM_PACIENTE                                NM_PACIENTE,
       GUIA.NR_GUIA                                                   NR_GUIA,
       REG_FAT.DT_INICIO                                         DT_ATENDIMENTO,
       REG_FAT.DT_FINAL                                          DT_ALTA,
       CONVENIO.CD_CONVENIO                               CD_CONVENIO,
       CONVENIO.NM_CONVENIO                              NM_CONVENIO,
       NVL( REG_FAT.VL_TOTAL_CONTA, 0)            VL_TOTAL_CONTA,
       ITAGRUPAMENTO.TP_AGRUPAMENTO         TP_AGRUPAMENTO,
       ITAGRUPAMENTO.CD_AGRUPAMENTO         CD_AGRUPAMENTO,
       AGRUPAMENTO.DS_AGRUPAMENTO            DS_AGRUPAMENTO,
       ORI_ATE.CD_ORI_ATE                                      CD_ORI_ATE1,
       ORI_ATE.DS_ORI_ATE                                      DS_ORI_ATE,
       FATURA.DT_COMPETENCIA		    DT_COMPETENCIA,
       0                                                                           CD_LANCAMENTO,
       'I'                                                                           TIPO_GUIA,
       CONVENIO.CD_CONVENIO                                CD_CONVENIO_GUIA,
       REMESSA_FATURA.DT_ENTREGA_DA_FATURA,
       ITREG_FAT.TP_PAGAMENTO
FROM   DBAMV.ATENDIME                                      ATENDIME,
       DBAMV.PACIENTE                                             PACIENTE,
       DBAMV.CONVENIO                                            CONVENIO,
       DBAMV.REG_FAT                                               REG_FAT,
       DBAMV.ITREG_FAT                                           ITREG_FAT,
       DBAMV.ITAGRUPAMENTO                                ITAGRUPAMENTO,
       DBAMV.AGRUPAMENTO                                   AGRUPAMENTO,
       DBAMV.ORI_ATE                                                ORI_ATE,
       DBAMV.REMESSA_FATURA                              REMESSA_FATURA,
       DBAMV.FATURA                                                 FATURA,
       DBAMV.GUIA				    GUIA
WHERE  REMESSA_FATURA.CD_REMESSA         =     :CAMPO_CD_REMESSA
AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND        ATENDIME.CD_PACIENTE                       =      PACIENTE.CD_PACIENTE
AND        ATENDIME.CD_ORI_ATE                         =      ITAGRUPAMENTO.CD_ORI_ATE
AND        ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND        ATENDIME.CD_ATENDIMENTO              =      REG_FAT.CD_ATENDIMENTO
AND       REG_FAT.CD_CONVENIO                         =      CONVENIO.CD_CONVENIO
AND      REG_FAT.CD_REG_FAT                             =     ITREG_FAT.CD_REG_FAT
AND       CONVENIO.CD_CONVENIO                       =       ITAGRUPAMENTO.CD_CONVENIO
AND       ORI_ATE.CD_ORI_ATE                              =      ITAGRUPAMENTO.CD_ORI_ATE
AND       ITAGRUPAMENTO.CD_AGRUPAMENTO =      AGRUPAMENTO.CD_AGRUPAMENTO
and       agrupamento.cd_agrupamento = remessa_fatura.cd_agrupamento(+)
AND       REG_FAT.CD_REMESSA                           =      REMESSA_FATURA.CD_REMESSA
AND       REMESSA_FATURA.CD_FATURA             =       FATURA.CD_FATURA
AND        ITAGRUPAMENTO.TP_AGRUPAMENTO    =   'H'
AND        ATENDIME.CD_PACIENTE = GUIA.CD_PACIENTE(+)
AND        ATENDIME.CD_CONVENIO = GUIA.CD_CONVENIO(+)
AND        ATENDIME.CD_ATENDIMENTO = GUIA.CD_ATENDIMENTO(+)
and         (guia.cd_guia is null or guia.cd_guia = (select min(cd_guia)
                                                    from dbamv.guia guia2
                                                    where cd_paciente = atendime.cd_paciente
                                                     and  cd_convenio = atendime.cd_convenio
                                                     and  cd_atendimento = atendime.cd_atendimento )
           )
UNION ALL
SELECT
       ATENDIME.CD_ORI_ATE                                          CD_ORI_ATE,
       ATENDIME.CD_ATENDIMENTO                               CD_ATENDIMENTO,
       ATENDIME.CD_PACIENTE                                        CD_PACIENTE,
       PACIENTE.NM_PACIENTE                                         NM_PACIENTE,
       GUIA.NR_GUIA                                                           NR_GUIA,
       ATENDIME.DT_ATENDIMENTO                                DT_ATENDIMENTO,
       ATENDIME.DT_ATENDIMENTO                                DT_ALTA,
       CONVENIO.CD_CONVENIO                                        CD_CONVENIO,
       CONVENIO.NM_CONVENIO                                       NM_CONVENIO,
       SUM( NVL( ITREG_AMB.VL_TOTAL_CONTA, 0 ) )    VL_TOTAL_CONTA,
       ITAGRUPAMENTO.TP_AGRUPAMENTO                  TP_AGRUPAMENTO,
       ITAGRUPAMENTO.CD_AGRUPAMENTO                  CD_AGRUPAMENTO,
       AGRUPAMENTO.DS_AGRUPAMENTO                     DS_AGRUPAMENTO,
       ORI_ATE.CD_ORI_ATE                                               CD_ORI_ATE1,
       ORI_ATE.DS_ORI_ATE                                               DS_ORI_ATE,
       FATURA.DT_COMPETENCIA		             DT_COMPETENCIA,
       MIN(ITREG_AMB.CD_LANCAMENTO)                        CD_LANCAMENTO,
       'C'                                                                                   TIPO_GUIA,
       CONVENIO.CD_CONVENIO                                         CD_CONVENIO_GUIA,
       REMESSA_FATURA.DT_ENTREGA_DA_FATURA ,
       ITREG_AMB.TP_PAGAMENTO
FROM   DBAMV.ATENDIME                           ATENDIME,
             DBAMV.PACIENTE                            PACIENTE,
             DBAMV.CONVENIO                           CONVENIO,
             DBAMV.REG_AMB                            REG_AMB,
             DBAMV.ITREG_AMB                         ITREG_AMB,
             DBAMV.ITAGRUPAMENTO              ITAGRUPAMENTO,
             DBAMV.AGRUPAMENTO                 AGRUPAMENTO,
             DBAMV.ORI_ATE                              ORI_ATE,
             DBAMV.REMESSA_FATURA            REMESSA_FATURA,
             DBAMV.FATURA                               FATURA,
             DBAMV.GUIA                                     GUIA
WHERE  REMESSA_FATURA.CD_REMESSA      =      :CAMPO_CD_REMESSA
AND REG_AMB.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND        ATENDIME.CD_PACIENTE                     =      PACIENTE.CD_PACIENTE
AND        ATENDIME.CD_ATENDIMENTO            =      ITREG_AMB.CD_ATENDIMENTO
AND        REG_AMB.CD_REG_AMB                      =      ITREG_AMB.CD_REG_AMB
AND        ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND        ATENDIME.CD_ORI_ATE                      =      ORI_ATE.CD_ORI_ATE
AND        ITREG_AMB.CD_CONVENIO                  =     ITAGRUPAMENTO.CD_CONVENIO
AND        CONVENIO.CD_CONVENIO                   =      ITAGRUPAMENTO.CD_CONVENIO
AND        ORI_ATE.CD_ORI_ATE                          =     ITAGRUPAMENTO.CD_ORI_ATE
AND        AGRUPAMENTO.CD_AGRUPAMENTO =     ITAGRUPAMENTO.CD_AGRUPAMENTO
and        agrupamento.cd_agrupamento = remessa_fatura.cd_agrupamento(+)
and        convenio.cd_convenio = reg_amb.cd_convenio
AND        REG_AMB.CD_REMESSA                      =     REMESSA_FATURA.CD_REMESSA
AND        REMESSA_FATURA.CD_FATURA         =     FATURA.CD_FATURA
AND        ITAGRUPAMENTO.TP_AGRUPAMENTO    =   'A'
AND        ITREG_AMB.SN_PERTENCE_PACOTE = 'N'
AND        NVL(ITREG_AMB.TP_PAGAMENTO,'X') <> 'C'
AND        ATENDIME.CD_PACIENTE = GUIA.CD_PACIENTE(+)
AND        ATENDIME.CD_CONVENIO = GUIA.CD_CONVENIO(+)
AND        ATENDIME.CD_ATENDIMENTO = GUIA.CD_ATENDIMENTO(+)
and          (guia.cd_guia is null or guia.cd_guia = (select min(cd_guia)
                                                    from dbamv.guia guia2
                                                    where guia2.cd_paciente = atendime.cd_paciente
                                                     and   guia2.cd_convenio = atendime.cd_convenio
                                                     and   guia2.cd_atendimento = atendime.cd_atendimento)
          )
GROUP BY     ATENDIME.CD_ATENDIMENTO ,
                        ATENDIME.CD_ORI_ATE ,
                        ATENDIME.CD_PACIENTE ,
                       PACIENTE.NM_PACIENTE ,
                       ATENDIME.DT_ATENDIMENTO ,
                       ATENDIME.DT_ATENDIMENTO ,
                      CONVENIO.CD_CONVENIO ,
                      CONVENIO.NM_CONVENIO ,
                      ITAGRUPAMENTO.TP_AGRUPAMENTO ,
                      ITAGRUPAMENTO.CD_AGRUPAMENTO ,
                      AGRUPAMENTO.DS_AGRUPAMENTO ,
                      ORI_ATE.CD_ORI_ATE ,
                      ORI_ATE.DS_ORI_ATE ,
                      FATURA.DT_COMPETENCIA ,
                      GUIA.NR_GUIA	 ,
                      REMESSA_FATURA.DT_ENTREGA_DA_FATURA,
                      ITREG_AMB.TP_PAGAMENTO
ORDER BY DS_AGRUPAMENTO, TP_AGRUPAMENTO, CD_CONVENIO     :CAMPO_CF_ORDEM
)
ORDER BY cd_agrupamento, dt_competencia, DS_AGRUPAMENTO, TP_AGRUPAMENTO, CD_CONVENIO
