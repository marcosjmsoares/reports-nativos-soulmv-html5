--R_FAT_CC_SERVICO
SELECT NM_SETOR
               ,DS_SERVICO
               ,SUM(VL_TOTAL) VL_TOTAL
FROM  ( SELECT DECODE(SETOR.NM_SETOR, NULL, ' Não Informado', SETOR.NM_SETOR) NM_SETOR
                            ,DECODE(SERVICO.DS_SERVICO, NULL, ' Não Informado', SERVICO.DS_SERVICO)       DS_SERVICO
                            ,Sum(nvl(itlan_med.vl_liquido,itreg_fat.vl_total_conta))       VL_TOTAL
               FROM   DBAMV.REG_FAT
                            ,DBAMV.ITREG_FAT
                            ,DBAMV.PRO_FAT
                            ,DBAMV.ITLAN_MED
                            ,DBAMV.GRU_PRO
                            ,DBAMV.SETOR
                            ,DBAMV.REMESSA_FATURA     REMESSA
                            ,DBAMV.FATURA
                            ,DBAMV.CONVENIO
                            ,DBAMV.ATENDIME
                            ,DBAMV.SERVICO
			    ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And REG_FAT.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
               AND        REG_FAT.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
               AND        ATENDIME.CD_SERVICO = SERVICO.CD_SERVICO(+)
               AND        ITREG_FAT.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
               AND        PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
               AND        REG_FAT.CD_REMESSA = REMESSA.CD_REMESSA
               AND        REMESSA.CD_FATURA = FATURA.CD_FATURA
               AND        REG_FAT.CD_CONVENIO = CONVENIO.CD_CONVENIO
               AND        nvl( ITREG_FAT.SN_PERTENCE_PACOTE, 'N' ) = 'N'
               AND        NVL(REG_FAT.SN_DIAGNO,'N' ) =  'N'
               AND        ITLAN_MED.CD_REG_FAT(+) = ITREG_FAT.CD_REG_FAT
               AND        ITLAN_MED.CD_LANCAMENTO(+) = ITREG_FAT.CD_LANCAMENTO
               AND        NVL( ITLAN_MED.TP_PAGAMENTO, NVL( ITREG_FAT.TP_PAGAMENTO, 'P' ) ) <> 'C'
               AND        REG_FAT.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
               AND        CONVENIO.TP_CONVENIO <> 'H'
                GROUP BY  SETOR.NM_SETOR
                                   ,SERVICO.DS_SERVICO
               UNION ALL
               SELECT DECODE(SETOR.NM_SETOR, NULL, ' Não Informado', SETOR.NM_SETOR) NM_SETOR
                             ,DECODE(SERVICO.DS_SERVICO, NULL, ' Não Informado', SERVICO.DS_SERVICO)    DS_SERVICO
                             ,SUM(ITREG_AMB.VL_TOTAL_CONTA)          VL_TOTAL
                FROM   DBAMV.REG_AMB
                             ,DBAMV.ITREG_AMB
                             ,DBAMV.PRO_FAT
                             ,DBAMV.GRU_PRO
                             ,DBAMV.SETOR
                             ,DBAMV.REMESSA_FATURA  REMESSA
                             ,DBAMV.FATURA
                             ,DBAMV.CONVENIO
                             ,DBAMV.ATENDIME
                             ,DBAMV.SERVICO
			     ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And REG_AMB.CD_REG_AMB = ITREG_AMB.CD_REG_AMB
                AND        ITREG_AMB.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
                AND        ATENDIME.CD_SERVICO = SERVICO.CD_SERVICO(+)
                AND        ITREG_AMB.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
                AND        PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
                AND        REG_AMB.CD_REMESSA = REMESSA.CD_REMESSA
                AND        REMESSA.CD_FATURA = FATURA.CD_FATURA
                AND        ITREG_AMB.CD_CONVENIO = CONVENIO.CD_CONVENIO
                AND        nvl( ITREG_AMB.SN_PERTENCE_PACOTE, 'N' ) = 'N'
                AND        NVL(REG_AMB.SN_DIAGNO,'N' ) =  'N'
                AND        NVL(ITREG_AMB.TP_PAGAMENTO,'X') <> 'C'
                AND        REG_AMB.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
                AND        CONVENIO.TP_CONVENIO <> 'H'
              GROUP BY  SETOR.NM_SETOR
                                   ,SERVICO.DS_SERVICO
               UNION ALL
               SELECT DECODE(SETOR.NM_SETOR, NULL, ' Não Informado', SETOR.NM_SETOR) NM_SETOR
                            ,DECODE(SERVICO.DS_SERVICO, NULL, ' Não Informado', SERVICO.DS_SERVICO)   DS_SERVICO
                            ,Sum(nvl(itlan_med.vl_liquido,itreg_fat.vl_total_conta))      VL_TOTAL
               FROM   DBAMV.REG_FAT
                            ,DBAMV.ITREG_FAT
                            ,DBAMV.PRO_FAT
                            ,DBAMV.GRU_PRO
                            ,DBAMV.ITLAN_MED
                            ,DBAMV.SETOR
                            ,DBAMV.REMESSA_FATURA     REMESSA
                            ,DBAMV.FATURA
                            ,DBAMV.CONVENIO
                            ,DBAMV.ATENDIME
                            ,DBAMV.SERVICO
			    ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And REG_FAT.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
               AND        REG_FAT.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
               AND        ATENDIME.CD_SERVICO = SERVICO.CD_SERVICO(+)
               AND        ITREG_FAT.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
               AND        PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
               AND        REG_FAT.CD_REMESSA = REMESSA.CD_REMESSA
               AND        REMESSA.CD_FATURA = FATURA.CD_FATURA
               AND        REG_FAT.CD_CONVENIO = CONVENIO.CD_CONVENIO
               AND        nvl(PRO_FAT.SN_PACOTE,'N') = 'N'
               AND ITLAN_MED.CD_REG_FAT(+) = ITREG_FAT.CD_REG_FAT
               AND ITLAN_MED.CD_LANCAMENTO(+) = ITREG_FAT.CD_LANCAMENTO
               AND NVL( ITLAN_MED.TP_PAGAMENTO, NVL( ITREG_FAT.TP_PAGAMENTO, 'P' ) ) <> 'C'
               AND        NVL(REG_FAT.SN_DIAGNO,'N' ) =  'N'
               AND        REG_FAT.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
               AND        CONVENIO.TP_CONVENIO <> 'H'
              GROUP BY  SETOR.NM_SETOR
                                   ,SERVICO.DS_SERVICO
               UNION ALL
               SELECT DECODE(SETOR.NM_SETOR, NULL, ' Não Informado', SETOR.NM_SETOR) NM_SETOR
                             ,DECODE(SERVICO.DS_SERVICO, NULL, ' Não Informado', SERVICO.DS_SERVICO)  DS_SERVICO
                             ,SUM(ITREG_AMB.VL_TOTAL_CONTA)          VL_TOTAL
                FROM   DBAMV.REG_AMB
                             ,DBAMV.ITREG_AMB
                             ,DBAMV.PRO_FAT
                             ,DBAMV.GRU_PRO
                             ,DBAMV.SETOR
                             ,DBAMV.REMESSA_FATURA  REMESSA
                             ,DBAMV.FATURA
                             ,DBAMV.CONVENIO
                             ,DBAMV.ATENDIME
                             ,DBAMV.SERVICO
			     ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And REG_AMB.CD_REG_AMB = ITREG_AMB.CD_REG_AMB
                AND        ITREG_AMB.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
                AND        ATENDIME.CD_SERVICO = SERVICO.CD_SERVICO(+)
                AND        ITREG_AMB.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
                AND        PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
                AND        REG_AMB.CD_REMESSA = REMESSA.CD_REMESSA
                AND        REMESSA.CD_FATURA = FATURA.CD_FATURA
                AND        ITREG_AMB.CD_CONVENIO = CONVENIO.CD_CONVENIO
                AND        nvl(PRO_FAT.SN_PACOTE,'N') = 'N'
              AND        NVL( ITREG_AMB.TP_PAGAMENTO, 'X' ) <> 'C'
                AND        NVL(REG_AMB.SN_DIAGNO,'N' ) =  'N'
                AND        REG_AMB.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
                AND        CONVENIO.TP_CONVENIO <> 'H'
                GROUP BY  SETOR.NM_SETOR
                                   ,SERVICO.DS_SERVICO  )
WHERE VL_TOTAL IS NOT NULL
GROUP BY  NM_SETOR
                    ,DS_SERVICO
