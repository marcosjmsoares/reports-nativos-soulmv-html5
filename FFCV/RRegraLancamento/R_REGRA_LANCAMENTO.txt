--R_REGRA_LANCAMENTO
SELECT * FROM (
SELECT  'Regra: ' || R.CD_REGRA_LANCAMENTO cd_regra,
                R.DS_REGRA_LANCAMENTO,
                R.CD_PRO_FAT,
                R.CD_PRO_FAT|| '-'|| P.ds_pro_fat ds_pro_fat ,
                R.CD_PRO_FAT_SUBORDINADO,
                R.TP_ACAO,
                rel.sn_internacao,
                rel.sn_ambulatorial,
                sn_hoca,
                sn_externo,
                sn_urgencia_emergencia
              , REL.CD_CONVENIO,
                REL.CD_MULTI_EMPRESA,
                RPLAN.CD_CON_PLA,
                RSETOR.CD_SETOR,
                REL.CD_REGRA_LANC_RELACIONADOS
              , decode (r.tp_acao, 'E', 'Exclui', 'I', 'Inclui', 'Substitui') acao
              ,R.CD_REGRA_LANCAMENTO
              ,convenio.nm_convenio
FROM    dbamv.regra_lancamento  R,
              dbamv.regra_lancamento_relacionados REL,
              dbamv.regra_lancamento_relac_plano  RPLAN,
              dbamv.regra_lancamento_relac_setor   RSETOR,
              dbamv.pro_fat   P,
              dbamv.convenio
WHERE R.cd_regra_lancamento= REL.cd_regra_lancamento(+)
      AND REL.cd_regra_lanc_relacionados =  RPLAN.cd_regra_lanc_relacionados(+)
      AND REL.cd_regra_lanc_relacionados =  RSETOR.cd_regra_lanc_relacionados(+)
      AND P.cd_pro_fat = R.cd_pro_fat
      AND convenio.cd_convenio = REL.cd_convenio
 ORDER BY R.CD_REGRA_LANCAMENTO, convenio.nm_convenio
)
