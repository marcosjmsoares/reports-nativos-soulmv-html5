--R_LOTE_ATEND_EXTRA
SELECT * FROM (
SELECT
       ATENDIME.CD_ORI_ATE                                  CD_ORI_ATE,
       ATENDIME.CD_ATENDIMENTO                       CD_ATENDIMENTO,
       ATENDIME.CD_PACIENTE                                CD_PACIENTE,
       PACIENTE.NM_PACIENTE                                NM_PACIENTE,
       ATENDIME.NR_GUIA                                        NR_GUIA,
       REG_FAT.DT_INICIO                                         DT_ATENDIMENTO,
       REG_FAT.DT_FINAL                                          DT_ALTA,
       CONVENIO.CD_CONVENIO                               CD_CONVENIO,
       CONVENIO.NM_CONVENIO                              NM_CONVENIO,
       NVL( REG_FAT.VL_TOTAL_CONTA, 0)            VL_TOTAL_CONTA,
       NVL( REG_FAT.VL_TOTAL_EXTRAS, 0)            VL_TOTAL_EXTRA,
       ITAGRUPAMENTO.TP_AGRUPAMENTO         TP_AGRUPAMENTO,
       ITAGRUPAMENTO.CD_AGRUPAMENTO         CD_AGRUPAMENTO,
       AGRUPAMENTO.DS_AGRUPAMENTO            DS_AGRUPAMENTO,
       ORI_ATE.CD_ORI_ATE                                      CD_ORI_ATE1,
       ORI_ATE.DS_ORI_ATE                                      DS_ORI_ATE,
       CARTEIRA.NR_CARTEIRA                                NR_CARTEIRA
FROM   DBAMV.ATENDIME                                      ATENDIME,
       DBAMV.PACIENTE                                             PACIENTE,
       DBAMV.CONVENIO                                            CONVENIO,
       DBAMV.REG_FAT                                               REG_FAT,
       DBAMV.ITAGRUPAMENTO                                ITAGRUPAMENTO,
       DBAMV.AGRUPAMENTO                                   AGRUPAMENTO,
       DBAMV.ORI_ATE                                                ORI_ATE,
       DBAMV.REMESSA_FATURA                              REMESSA_FATURA,
       DBAMV.FATURA                                                 FATURA,
       DBAMV.CARTEIRA                                             CARTEIRA
WHERE  ATENDIME.CD_PACIENTE  =  PACIENTE.CD_PACIENTE
AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND        ATENDIME.CD_CONVENIO  =  CONVENIO.CD_CONVENIO
AND        ATENDIME.CD_ATENDIMENTO  =  REG_FAT.CD_ATENDIMENTO
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND        ATENDIME.CD_ORI_ATE  =   ITAGRUPAMENTO.CD_ORI_ATE
AND       CONVENIO.CD_CONVENIO =  ITAGRUPAMENTO.CD_CONVENIO
AND       ORI_ATE.CD_ORI_ATE  =  ITAGRUPAMENTO.CD_ORI_ATE
AND       AGRUPAMENTO.CD_AGRUPAMENTO  =  ITAGRUPAMENTO.CD_AGRUPAMENTO
AND       REG_FAT.CD_REMESSA  =  REMESSA_FATURA.CD_REMESSA
AND       REMESSA_FATURA.CD_FATURA  =   FATURA.CD_FATURA
AND       REG_FAT.SN_FATURA_IMPRESSA  =   'S'
AND       ITAGRUPAMENTO.TP_AGRUPAMENTO  =   'H'
AND       CONVENIO.CD_CONVENIO  =  CARTEIRA.CD_CONVENIO
AND       PACIENTE.CD_PACIENTE  =  CARTEIRA.CD_PACIENTE
UNION ALL
SELECT
       ATENDIME.CD_ORI_ATE                                          CD_ORI_ATE,
       ATENDIME.CD_ATENDIMENTO                               CD_ATENDIMENTO,
       ATENDIME.CD_PACIENTE                                        CD_PACIENTE,
       PACIENTE.NM_PACIENTE                                         NM_PACIENTE,
       ATENDIME.NR_GUIA                                                 NR_GUIA,
       ATENDIME.DT_ATENDIMENTO                                DT_ATENDIMENTO,
       ATENDIME.DT_ATENDIMENTO                                DT_ALTA,
       CONVENIO.CD_CONVENIO                                        CD_CONVENIO,
       CONVENIO.NM_CONVENIO                                       NM_CONVENIO,
       SUM( NVL( ITREG_AMB.VL_TOTAL_CONTA, 0 ) )    VL_TOTAL_CONTA,
       0                                                                                   VL_TOTAL_EXTRA,
       ITAGRUPAMENTO.TP_AGRUPAMENTO                  TP_AGRUPAMENTO,
       ITAGRUPAMENTO.CD_AGRUPAMENTO                  CD_AGRUPAMENTO,
       AGRUPAMENTO.DS_AGRUPAMENTO                     DS_AGRUPAMENTO,
       ORI_ATE.CD_ORI_ATE                                               CD_ORI_ATE1,
       ORI_ATE.DS_ORI_ATE                                               DS_ORI_ATE,
       CARTEIRA.NR_CARTEIRA                                         NR_CARTEIRA
FROM   DBAMV.ATENDIME                           ATENDIME,
             DBAMV.PACIENTE                            PACIENTE,
             DBAMV.CONVENIO                           CONVENIO,
             DBAMV.REG_AMB                            REG_AMB,
             DBAMV.ITREG_AMB                         ITREG_AMB,
             DBAMV.ITAGRUPAMENTO              ITAGRUPAMENTO,
             DBAMV.AGRUPAMENTO                 AGRUPAMENTO,
             DBAMV.ORI_ATE                              ORI_ATE,
             DBAMV.REMESSA_FATURA            REMESSA_FATURA,
             DBAMV.FATURA                               FATURA,
             DBAMV.CARTEIRA                           CARTEIRA
WHERE   ATENDIME.CD_PACIENTE                     =      PACIENTE.CD_PACIENTE
AND REG_AMB.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND        ATENDIME.CD_ATENDIMENTO            =      ITREG_AMB.CD_ATENDIMENTO
AND        REG_AMB.CD_REG_AMB                      =      ITREG_AMB.CD_REG_AMB
AND        ATENDIME.CD_ORI_ATE                      =      ORI_ATE.CD_ORI_ATE
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND        ITREG_AMB.CD_CONVENIO                  =     ITAGRUPAMENTO.CD_CONVENIO
AND        CONVENIO.CD_CONVENIO                   =      ITAGRUPAMENTO.CD_CONVENIO
AND        ORI_ATE.CD_ORI_ATE                          =     ITAGRUPAMENTO.CD_ORI_ATE
AND        AGRUPAMENTO.CD_AGRUPAMENTO =     ITAGRUPAMENTO.CD_AGRUPAMENTO
AND        REG_AMB.CD_REMESSA                      =     REMESSA_FATURA.CD_REMESSA
AND        REMESSA_FATURA.CD_FATURA         =     FATURA.CD_FATURA
AND        ITREG_AMB.SN_FATURA_IMPRESSA      =    'S'
AND        ITAGRUPAMENTO.TP_AGRUPAMENTO    =   'A'
AND       CONVENIO.CD_CONVENIO  =  CARTEIRA.CD_CONVENIO
AND       PACIENTE.CD_PACIENTE  =  CARTEIRA.CD_PACIENTE
AND       ITREG_AMB.SN_PERTENCE_PACOTE = 'N'
AND       NVL(ITREG_AMB.TP_PAGAMENTO,'X') <> 'C'
GROUP BY     ATENDIME.CD_ATENDIMENTO,
                        ATENDIME.CD_ORI_ATE,
                        ATENDIME.CD_PACIENTE,
                       PACIENTE.NM_PACIENTE,
                       ATENDIME.NR_GUIA,
                       ATENDIME.DT_ATENDIMENTO,
                       ATENDIME.DT_ATENDIMENTO,
                      CONVENIO.CD_CONVENIO,
                      CONVENIO.NM_CONVENIO,
                      ITAGRUPAMENTO.TP_AGRUPAMENTO,
                      ITAGRUPAMENTO.CD_AGRUPAMENTO,
                      AGRUPAMENTO.DS_AGRUPAMENTO,
                      ORI_ATE.CD_ORI_ATE,
                      ORI_ATE.DS_ORI_ATE,
                      CARTEIRA.NR_CARTEIRA
ORDER BY DS_AGRUPAMENTO, TP_AGRUPAMENTO, CD_CONVENIO  :CAMPO_CF_ORDEM
)
ORDER BY DS_AGRUPAMENTO,CD_AGRUPAMENTO,TP_AGRUPAMENTO,
CD_CONVENIO,NM_CONVENIO
