--R_AUDITORIA_PROC_TELA_297
SELECT * FROM (
SELECT  aud.CD_CONTA,
               aud.CD_GRU_PRO,
               gru_pro.ds_gru_pro,
               aud.CD_SETOR,
               Initcap( setor.nm_setor )   nm_setor,
               aud.CD_PRO_FAT,
               pro_fat.ds_pro_fat,
               aud.VL_PERCENTUAL_MULTIPLA,
               Sum(aud.Qt_lancamento) Qt_lancamento,
               SUM(aud.VL_TOT_CONTA) DSP_VL_TOTAL_CONTA,
                aud.vl_unitario,
               SUM(aud.VL_ACRESCIMO) VL_ACRESCIMO,
               SUM(aud.VL_DESCONTO ) VL_DESCONTO
FROM (
 SELECT itreg_fat.CD_REG_FAT CD_CONTA, itreg_fat.CD_PRO_FAT, pro_fat.CD_GRU_PRO,
        CD_SETOR,
        VL_PERCENTUAL_MULTIPLA,
        Qt_lancamento,
        ROUND(NVL( itreg_fat.VL_TOTAL_CONTA, 0) - NVL(VL_ACRESCIMO, 0) +
            NVL(VL_DESCONTO, 0), 2) VL_TOT_CONTA,
        ITREG_FAT.vl_unitario,
        ROUND(NVL(VL_ACRESCIMO,0),2) VL_ACRESCIMO,
        ROUND(NVL(VL_DESCONTO,0),2) VL_DESCONTO
 FROM dbamv.itreg_fat
     ,dbamv.reg_fat
     ,dbamv.pro_fat
 WHERE NVL(SN_PERTENCE_PACOTE,'N') = 'N'
   AND NVL(TP_PAGAMENTO, 'X') <> 'C'
   AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   AND itreg_Fat.cd_pro_fat = pro_fat.cd_pro_fat
   and itreg_fat.cd_reg_fat = :CAMPO_CONTA
   and itreg_fat.cd_reg_fat = reg_fat.cd_reg_fat
   and reg_fat.cd_atendimento = :CAMPO_ATENDIMENTO
 union all
 SELECT CD_REG_AMB CD_CONTA, itreg_amb.CD_PRO_FAT, pro_fat.CD_GRU_PRO,
        CD_SETOR,
        VL_PERCENTUAL_MULTIPLA,
        Qt_lancamento,
        ROUND(NVL(VL_TOTAL_CONTA, 0) - NVL(VL_ACRESCIMO, 0) +
            NVL(VL_DESCONTO, 0), 2) VL_TOT_CONTA,
        ITREG_AMB.vl_unitario,
        ROUND(NVL(VL_ACRESCIMO,0),2) VL_ACRESCIMO,
        ROUND(NVL(VL_DESCONTO,0),2) VL_DESCONTO
 FROM dbamv.itreg_amb
     ,dbamv.pro_fat
 WHERE NVL(SN_PERTENCE_PACOTE,'N') = 'N'
   AND NVL(TP_PAGAMENTO, 'X') <> 'C'
   AND itreg_amb.cd_pro_fat = pro_fat.cd_pro_fat
   and itreg_amb.cd_reg_amb = :CAMPO_CONTA
   and itreg_amb.cd_atendimento = :CAMPO_ATENDIMENTO
 union all
 SELECT CD_REG_AMB CD_CONTA, auditoria_conta.CD_PRO_FAT, pro_fat.CD_GRU_PRO,
        CD_SETOR,
        VL_PERCENTUAL_MULTIPLA,
        Qt_lancamento Qt_lancamento,
        0 VL_TOT_CONTA,
        vl_unitario,
        0 VL_ACRESCIMO,
        0 VL_DESCONTO
 FROM dbamv.auditoria_conta, dbamv.pro_fat
 WHERE auditoria_conta.cd_pro_fat = pro_fat.cd_pro_fat
   AND auditoria_conta.DT_CANCELOU is null
   AND (auditoria_conta.qt_lancamento = 0 or auditoria_conta.vl_percentual_multipla = 0)
   and auditoria_conta.cd_reg_amb = :CAMPO_CONTA
   and auditoria_conta.cd_atendimento = :CAMPO_ATENDIMENTO
   and not exists (select cd_reg_amb from dbamv.itreg_amb
                         where  cd_reg_amb = auditoria_conta.cd_reg_amb
                     and   cd_lancamento = auditoria_conta.cd_lancamento_amb
                     and  cd_setor = auditoria_conta.cd_setor
                     and  auditoria_conta.cd_pro_fat = cd_pro_fat
                     and auditoria_conta.vl_percentual_multiPla = vl_percentual_multiPla
                     and nvl(auditoria_conta.cd_prestador,0) = nvl(cd_prestador,0)
                     and cd_reg_amb = :CAMPO_CONTA
                     and cd_atendimento = :CAMPO_ATENDIMENTO)
 union all
 SELECT CD_REG_FAT CD_CONTA, auditoria_conta.CD_PRO_FAT, pro_fat.CD_GRU_PRO,
        CD_SETOR,
        VL_PERCENTUAL_MULTIPLA,
        Qt_lancamento Qt_lancamento,
        0 VL_TOT_CONTA,
        vl_unitario,
        0 VL_ACRESCIMO,
        0 VL_DESCONTO
 FROM dbamv.auditoria_conta, dbamv.pro_fat
 WHERE auditoria_conta.cd_pro_fat = pro_fat.cd_pro_fat
   AND auditoria_conta.DT_CANCELOU is null
   AND (auditoria_conta.qt_lancamento = 0 or auditoria_conta.vl_percentual_multipla = 0)
   and auditoria_conta.cd_reg_fat = :CAMPO_CONTA
   and not exists (select cd_reg_fat from dbamv.itreg_fat
                    where  cd_reg_fat = auditoria_conta.cd_reg_fat
                     and auditoria_conta.cd_lancamento_fat = cd_lancamento
                     and  cd_setor = auditoria_conta.cd_setor
                     and  auditoria_conta.cd_pro_fat = cd_pro_fat
                     and auditoria_conta.vl_percentual_multiPla = vl_percentual_multiPla
                     and nvl(auditoria_conta.cd_prestador,0) = nvl(cd_prestador,0)
                     and cd_reg_fat = :CAMPO_CONTA)
) Aud,
  dbamv.setor,
  dbamv.gru_pro,
  dbamv.pro_fat
where aud.cd_setor = setor.cd_setor(+)
  and aud.cd_gru_pro = gru_pro.cd_gru_pro(+)
  and aud.cd_pro_fat = pro_fat.cd_pro_fat(+)
GROUP BY  aud.CD_CONTA,
        aud.CD_GRU_PRO,
        gru_pro.ds_gru_pro,
        aud.CD_SETOR,
        Initcap( setor.nm_setor ),
        aud.CD_PRO_FAT,
        pro_fat.ds_pro_fat,
        aud.VL_PERCENTUAL_MULTIPLA,
        aud.vl_unitario
ORDER BY 1, 2, 4, 6
)
ORDER BY CD_CONTA,
CD_GRU_PRO,DS_GRU_PRO,
CD_SETOR,NM_SETOR

SELECT P.CD_PACIENTE ||' - '|| UPPER(P.NM_PACIENTE)   CD_PACIENTE,
               R.CD_REG_FAT,
               R.NR_GUIA,
               NVL( TO_CHAR( A.DT_ATENDIMENTO, 'DD/MM/YYYY'), '' ) DT_INTERN,
               NVL( TO_CHAR( A.DT_ALTA, 'DD/MM/YYYY'), '' ) DT_ALTA,
               A.CD_LEITO ||' - '|| L.DS_LEITO    CD_LEITO,
               R.CD_CONVENIO ||' - '|| C.NM_RAZAO_SOCIAL     CD_CONVENIO,
               R.DT_INICIO,
               R.DT_FINAL,
               A.CD_ATENDIMENTO,
               P.NM_PACIENTE
FROM    DBAMV.PACIENTE P,
        DBAMV.REG_FAT R,
        DBAMV.ATENDIME A,
        DBAMV.LEITO L,
        DBAMV.CONVENIO C,
        DBAMV.AUDITORIA_CONTA AC
WHERE A.CD_ATENDIMENTO = R.CD_ATENDIMENTO
      AND r.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
     AND L.CD_LEITO                 = A.CD_LEITO
      AND C.CD_CONVENIO        = R.CD_CONVENIO
      AND R.CD_ATENDIMENTO = :CAMPO_ATENDIMENTO
      AND R.CD_REG_FAT          = :CAMPO_CONTA
      AND AC.CD_REG_FAT                 = R.CD_REG_FAT
      AND AC.CD_ATENDIMENTO = A.CD_ATENDIMENTO
      AND P.CD_PACIENTE  = A.CD_PACIENTE
      AND AC.DT_CANCELOU IS NULL
      AND  A.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY P.CD_PACIENTE ||' - '|| UPPER(P.NM_PACIENTE),
               R.CD_REG_FAT,
               R.NR_GUIA,
               NVL( TO_CHAR( A.DT_ATENDIMENTO, 'DD/MM/YYYY'), '' ) ,
               NVL( TO_CHAR( A.DT_ALTA, 'DD/MM/YYYY'), '' ),
               A.CD_LEITO ||' - '|| L.DS_LEITO ,
               R.CD_CONVENIO ||' - '|| C.NM_RAZAO_SOCIAL,
               R.DT_INICIO,
               R.DT_FINAL,
               A.CD_ATENDIMENTO,
               P.NM_PACIENTE
UNION ALL
SELECT  P.CD_PACIENTE ||' - '|| UPPER(P.NM_PACIENTE)    CD_PACIENTE,
               R.CD_REG_AMB        CD_REG_FAT,
               ''    NR_GUIA,
               NVL( TO_CHAR( A.DT_ATENDIMENTO, 'DD/MM/YYYY'), '' ) DT_INTERN,
               NVL( TO_CHAR( A.DT_ALTA, 'DD/MM/YYYY'), '' ) DT_ALTA,
               A.CD_LEITO ||' - '|| L.DS_LEITO    CD_LEITO,
               R.CD_CONVENIO ||' - '|| C.NM_RAZAO_SOCIAL     CD_CONVENIO,
               R.DT_LANCAMENTO                      DT_INICIO,
               R.DT_LANCAMENTO_FINAL          DT_FINAL,
               A.CD_ATENDIMENTO,
               P.NM_PACIENTE
FROM    DBAMV.PACIENTE P,
              DBAMV.REG_AMB R,
              DBAMV.ATENDIME A,
              DBAMV.LEITO L,
              DBAMV.CONVENIO C,
              DBAMV.AUDITORIA_CONTA AC
WHERE L.CD_LEITO(+)                 = A.CD_LEITO
      AND r.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
      AND C.CD_CONVENIO        = R.CD_CONVENIO
      AND R.CD_REG_AMB          = :CAMPO_CONTA
      AND A.CD_ATENDIMENTO = :CAMPO_ATENDIMENTO
      AND AC.CD_REG_AMB                   = R.CD_REG_AMB
      AND AC.CD_ATENDIMENTO = A.CD_ATENDIMENTO
      AND P.CD_PACIENTE  = A.CD_PACIENTE
      AND AC.DT_CANCELOU IS NULL
      AND  A.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY  P.CD_PACIENTE ||' - '|| UPPER(P.NM_PACIENTE),
               R.CD_REG_AMB,
               NVL( TO_CHAR( A.DT_ATENDIMENTO, 'DD/MM/YYYY'), '' ),
               NVL( TO_CHAR( A.DT_ALTA, 'DD/MM/YYYY'), '' ),
               A.CD_LEITO ||' - '|| L.DS_LEITO ,
               R.CD_CONVENIO ||' - '|| C.NM_RAZAO_SOCIAL,
               R.DT_LANCAMENTO ,
               R.DT_LANCAMENTO_FINAL ,
               A.CD_ATENDIMENTO,
               P.NM_PACIENTE
