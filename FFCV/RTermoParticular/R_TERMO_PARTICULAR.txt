--R_TERMO_PARTICULAR
SELECT
           ATENDIME.CD_ATENDIMENTO    CD_ATENDIMENTO
          ,REG_FAT.DT_INICIO                      DT_INICIO
          ,REG_FAT.DT_FINAL                       DT_FINAL
          ,dbamv.pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(PACIENTE.CD_PACIENTE, 1)             NM_PACIENTE
          ,PACIENTE.NR_CPF                        NR_CPF
          ,RESPONSA.NM_RESPONSAVEL  NM_RESPONSAVEL
          ,RESPONSA.NR_CPF                      NR_CPF_RESPON
          ,REG_FAT.CD_REG_FAT                NU_CONTA
          ,REG_FAT.VL_TOTAL_CONTA       VL_TOTAL_CONTA
FROM
           DBAMV.ATENDIME
          ,DBAMV.PACIENTE
          ,DBAMV.REG_FAT
          ,DBAMV.RESPONSA
          ,DBAMV.CONVENIO
WHERE     ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND       ATENDIME.CD_ATENDIMENTO = REG_FAT.CD_ATENDIMENTO
AND       ATENDIME.CD_ATENDIMENTO = RESPONSA.CD_ATENDIMENTO (+)
AND       REG_FAT.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND      CONVENIO.TP_CONVENIO = 'P'
AND      REG_FAT.CD_REG_FAT = :CAMPO_CD_REG_FAT
AND      ATENDIME.CD_ATENDIMENTO = :CAMPO_CD_ATENDIMENTO
AND     ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
UNION ALL
SELECT
           ATENDIME.CD_ATENDIMENTO     CD_ATENDIMENTO
          ,ATENDIME.DT_ATENDIMENTO     DT_INICIO
          ,ATENDIME.DT_ALTA                      DT_FINAL
          ,dbamv.pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(paciente.cd_paciente, 1)             NM_PACIENTE
          ,PACIENTE.NR_CPF                        NR_CPF
          ,RESPONSA.NM_RESPONSAVEL  NM_RESPONSAVEL
          ,RESPONSA.NR_CPF                      NR_CPF_RESPON
          ,REG_AMB.CD_REG_AMB              NU_CONTA
          ,SUM(DECODE(NVL(ITREG_AMB.TP_PAGAMENTO,'P'), 'C', 0, Decode(ITREG_AMB.SN_PERTENCE_PACOTE, 'S', 0, ITREG_AMB.VL_TOTAL_CONTA))) VL_TOTAL_CONTA
FROM       DBAMV.ATENDIME
          ,DBAMV.PACIENTE
          ,DBAMV.REG_AMB
          ,DBAMV.ITREG_AMB
          ,DBAMV.RESPONSA
          ,DBAMV.CONVENIO
WHERE     ATENDIME.CD_PACIENTE    = PACIENTE.CD_PACIENTE
AND REG_amb.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND       ATENDIME.CD_ATENDIMENTO = ITREG_AMB.CD_ATENDIMENTO
AND       ATENDIME.CD_ATENDIMENTO = RESPONSA.CD_ATENDIMENTO (+)
AND       REG_AMB.CD_REG_AMB      = ITREG_AMB.CD_REG_AMB
AND       REG_AMB.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND      CONVENIO.TP_CONVENIO = 'P'
AND       REG_AMB.CD_REG_AMB      = :CAMPO_CD_REG_AMB
AND      ATENDIME.CD_ATENDIMENTO = :CAMPO_CD_ATENDIMENTO
AND      ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY
           ATENDIME.CD_ATENDIMENTO
          ,ATENDIME.DT_ATENDIMENTO
          ,ATENDIME.DT_ALTA
          ,pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(PACIENTE.CD_PACIENTE, 1)
          ,PACIENTE.NR_CPF
          ,RESPONSA.NM_RESPONSAVEL
          ,RESPONSA.NR_CPF
          ,REG_AMB.CD_REG_AMB
