--R_TABFAT_PROCED_R_CG__G_TAB_FAT4_CG__Q_TAB_FAT
SELECT  CODIGO, DESCRICAO, DT_VIGENCIA, CD_PRO_FAT, CD_PRO_FAT1, DS_PRO_FAT,
DS_UNIDADE, VL_HONORARIO, VL_OPERACIONAL, VL_TOTAL1, CD_IMPORT, CD_TAB_FAT1,
CALCULAR_VARIACAO
FROM (
SELECT  TAB_FAT.CD_TAB_FAT           CODIGO,
                TAB_FAT.DS_TAB_FAT           DESCRICAO,
                VAL_PRO.DT_VIGENCIA          DT_VIGENCIA,
                VAL_PRO.CD_PRO_FAT          CD_PRO_FAT,
                PRO_FAT.CD_PRO_FAT          CD_PRO_FAT1,
                PRO_FAT.DS_PRO_FAT          DS_PRO_FAT,
                PRO_FAT.DS_UNIDADE          DS_UNIDADE,
                VAL_PRO.VL_HONORARIO     VL_HONORARIO,
                VAL_PRO.VL_OPERACIONAL  VL_OPERACIONAL,
                VAL_PRO.VL_TOTAL                VL_TOTAL1,
                VAL_PRO.CD_IMPORT             CD_IMPORT,
                VAL_PRO.CD_TAB_FAT          CD_TAB_FAT1,
                '1' CALCULAR_VARIACAO
FROM    dbamv.TAB_FAT TAB_FAT,
              dbamv.VAL_PRO VAL_PRO,
              dbamv.PRO_FAT PRO_FAT
WHERE :CAMPO_ULT_SN = 'comparativa'
AND         VAL_PRO.CD_TAB_FAT = TAB_FAT.CD_TAB_FAT
AND	 VAL_PRO.DT_VIGENCIA =   ( Select Max(Dt_Vigencia)
                                      From Dbamv.Val_Pro VP1
                                      Where  VP1.Dt_Vigencia <= current_date
                                      and  VP1.cd_pro_fat   = VAL_PRO.CD_PRO_FAT
                                      and  NVL(VP1.SN_ATIVO, 'S') = decode(:CAMPO_SN_ATIVO, 'T', NVL(VP1.SN_ATIVO, 'S'), :CAMPO_SN_ATIVO)
                                      and  VP1.cd_tab_fat   = VAL_PRO.CD_TAB_FAT)
AND       PRO_FAT.CD_PRO_FAT = VAL_PRO.CD_PRO_FAT
AND       NVL(VAL_PRO.SN_ATIVO, 'S') = decode('', 'T', NVL(VAL_PRO.SN_ATIVO, 'S'), :CAMPO_SN_ATIVO)
UNION ALL
SELECT  TAB_FAT.CD_TAB_FAT           CODIGO,
                TAB_FAT.DS_TAB_FAT           DESCRICAO,
                VAL_PRO.DT_VIGENCIA          DT_VIGENCIA,
                VAL_PRO.CD_PRO_FAT          CD_PRO_FAT,
                PRO_FAT.CD_PRO_FAT          CD_PRO_FAT1,
                PRO_FAT.DS_PRO_FAT          DS_PRO_FAT,
                PRO_FAT.DS_UNIDADE          DS_UNIDADE,
                VAL_PRO.VL_HONORARIO     VL_HONORARIO,
                VAL_PRO.VL_OPERACIONAL  VL_OPERACIONAL,
                VAL_PRO.VL_TOTAL                VL_TOTAL1,
                VAL_PRO.CD_IMPORT             CD_IMPORT,
                VAL_PRO.CD_TAB_FAT          CD_TAB_FAT1,
                '0' CALCULAR_VARIACAO
FROM    dbamv.TAB_FAT TAB_FAT,
              dbamv.VAL_PRO VAL_PRO,
              dbamv.PRO_FAT PRO_FAT
WHERE :CAMPO_ULT_SN = 'comparativa'
AND      VAL_PRO.CD_TAB_FAT = TAB_FAT.CD_TAB_FAT
AND	 VAL_PRO.DT_VIGENCIA =   ( Select Max(VP1.Dt_Vigencia)
                                      From Dbamv.Val_Pro VP1
                                      Where  VP1.Dt_Vigencia <
						 	( Select Max(VP2.Dt_Vigencia)
                                      			From Dbamv.Val_Pro VP2
                                      			Where  VP2.Dt_Vigencia <= current_date
		                                        and  VP2.cd_pro_fat   = VAL_PRO.CD_PRO_FAT
                                      			and  NVL(VP2.SN_ATIVO, 'S') = decode(:CAMPO_SN_ATIVO, 'T', NVL(VP2.SN_ATIVO, 'S'), :CAMPO_SN_ATIVO)
                                      			and  VP2.cd_tab_fat   = VAL_PRO.CD_TAB_FAT )
                                      and  VP1.cd_pro_fat   = VAL_PRO.CD_PRO_FAT
                                      and  NVL(VP1.SN_ATIVO, 'S') = decode(:CAMPO_SN_ATIVO, 'T', NVL(VP1.SN_ATIVO, 'S'), :CAMPO_SN_ATIVO)
                                      and  VP1.cd_tab_fat   = VAL_PRO.CD_TAB_FAT )
AND       PRO_FAT.CD_PRO_FAT = VAL_PRO.CD_PRO_FAT
AND       NVL(VAL_PRO.SN_ATIVO, 'S') = decode('', 'T', NVL(VAL_PRO.SN_ATIVO, 'S'), :CAMPO_SN_ATIVO)
:CAMPO_CF_TAB_FAT
)
