--R_ANALISE_PROT_PACOTE_R_ATENDIMENTO1_Q_1
SELECT * FROM (
SELECT atendime.cd_atendimento      atendimento_q
     , reg_fat.cd_reg_fat           cd_conta_q
     , paciente.nm_paciente         nm_paciente_q
     , pacote.cd_pro_fat_pacote     cd_pro_fat_pacote_q
     , pro_fat1.ds_pro_fat          ds_pacote_q
     , protocolo.cd_pro_fat         cd_pro_fat_q
     , pro_fat2.ds_pro_fat          ds_pro_fat_q
     , protocolo.qt_contratada      qt_protocolo_q
     , conta_pacote.cd_conta_pacote cd_conta_pacote_q
     , atendime.dt_atendimento      dt_atendimento_q
     , itreg_fat.cd_setor           cd_setor_q
     , setor.nm_setor               nm_setor_q
     , itreg_fat.cd_setor||' - '||setor.nm_setor  descricao_setor_Q
     , pro_fat2.cd_gru_pro          cd_gru_pro_q
     , gru_pro.ds_gru_pro           ds_gru_pro_q
     , pro_fat2.cd_gru_pro||' - '||gru_pro.ds_gru_pro  descricao_gru_pro_q
  from dbamv.setor
     , dbamv.atendime
     , dbamv.paciente
     , dbamv.reg_fat
     , dbamv.itreg_fat
     , dbamv.pacote
     , dbamv.protocolo_pacote       protocolo
     , dbamv.conta_pacote
     , dbamv.pro_fat  pro_fat1
     , dbamv.pro_fat  pro_fat2
     , dbamv.gru_pro
 where reg_fat.cd_atendimento    = atendime.cd_atendimento
   and paciente.cd_paciente      = atendime.cd_paciente
   and conta_pacote.cd_reg_fat   = reg_fat.cd_reg_fat
   and itreg_fat.cd_reg_fat      = reg_fat.cd_reg_fat
   and pacote.cd_pacote          = conta_pacote.cd_pacote
   and protocolo.cd_pacote       = pacote.cd_pacote
   and pro_fat2.cd_pro_fat       = protocolo.cd_pro_fat
   and pro_fat1.cd_pro_fat       = pacote.cd_pro_fat_pacote
   and itreg_fat.cd_pro_fat      = pro_fat1.cd_pro_fat
   and atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
   and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   and conta_pacote.sn_principal = 'S'
   and atendime.dt_atendimento   between to_date(:CAMPO_DT_INICIO, 'dd/mm/yyyy')
                                     and (to_date(:CAMPO_DT_FINAL, 'DD/MM/YYYY') + 0.99999)
   and itreg_fat.cd_setor        = setor.cd_setor
   and pro_fat2.cd_gru_pro       = gru_pro.cd_gru_pro
union all
select atendime.cd_atendimento      atendimento_q
     , reg_amb.cd_reg_amb           cd_conta_q
     , paciente.nm_paciente         nm_paciente_q
     , pacote.cd_pro_fat_pacote     cd_pro_fat_pacote_q
     , pro_fat1.ds_pro_fat          ds_pacote_q
     , protocolo.cd_pro_fat         cd_pro_fat_q
     , pro_fat2.ds_pro_fat          ds_pro_fat_q
     , protocolo.qt_contratada      qt_protocolo_q
     , conta_pacote.cd_conta_pacote cd_conta_pacote_q
     , atendime.dt_atendimento      dt_atendimento_q
     , itreg_amb.cd_setor           cd_setor_q
     , setor.nm_setor               nm_setor_q
     , itreg_amb.cd_setor||' - '||setor.nm_setor  descricao_setor_Q
     , pro_fat2.cd_gru_pro          cd_gru_pro_q
     , gru_pro.ds_gru_pro           ds_gru_pro_q
     , pro_fat2.cd_gru_pro||' - '||gru_pro.ds_gru_pro  descricao_gru_pro_q
  from dbamv.setor
     , dbamv.atendime
     , dbamv.paciente
     , dbamv.reg_amb
     , dbamv.itreg_amb
     , dbamv.pacote
     , dbamv.protocolo_pacote       protocolo
     , dbamv.conta_pacote
     , dbamv.pro_fat  pro_fat1
     , dbamv.pro_fat  pro_fat2
     , dbamv.gru_pro
 where itreg_amb.cd_atendimento  = atendime.cd_atendimento
   and paciente.cd_paciente      = atendime.cd_paciente
   and conta_pacote.cd_reg_amb   = reg_amb.cd_reg_amb
   and itreg_amb.cd_reg_amb      = reg_amb.cd_reg_amb
   and pacote.cd_pacote          = conta_pacote.cd_pacote
   and protocolo.cd_pacote       = pacote.cd_pacote
   and pro_fat2.cd_pro_fat       = protocolo.cd_pro_fat
   and pro_fat1.cd_pro_fat       = pacote.cd_pro_fat_pacote
   and itreg_amb.cd_pro_fat      = pro_fat1.cd_pro_fat
   and atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
   and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   and conta_pacote.sn_principal = 'S'
   and atendime.dt_atendimento   between to_date(:CAMPO_DT_INICIO, 'dd/mm/yyyy')
                                     and (to_date(:CAMPO_DT_FINAL, 'DD/MM/YYYY') + 0.99999)
   and itreg_amb.cd_setor        = setor.cd_setor
   and pro_fat2.cd_gru_pro       = gru_pro.cd_gru_pro
    :CAMPO_CF_CD_CONVENIO_AMB
)
ORDER BY ATENDIMENTO_Q,CD_CONTA_Q,NM_PACIENTE_Q,DT_ATENDIMENTO_Q,
CD_PRO_FAT_PACOTE_Q,DS_PACOTE_Q,CD_CONTA_PACOTE_Q,
CD_GRU_PRO_Q,DS_GRU_PRO_Q,DESCRICAO_GRU_PRO_Q,
CD_SETOR_Q,NM_SETOR_Q,DESCRICAO_SETOR_Q
