--R_VAL_PORTE_MEDICO
SELECT TAB_FAT.CD_TAB_FAT                   CD_TAB_FAT1
              ,TAB_FAT.DS_TAB_FAT                   DS_TAB_FAT
              ,POR_MED.CD_PORTE_MEDICO                  CODIGO
              ,POR_MED.DS_PORTE_MEDICO                  DESCRICAO
              ,VAL_POR.DT_VIGENCIA DT_VIGENCIA
              ,VAL_POR.VL_PORTE_MEDICO VL_PORTE_MEDICO
FROM  dbamv.VAL_PORTE_MEDICO       VAL_POR
           ,dbamv.TAB_FAT         TAB_FAT
            ,dbamv.PORTE_MEDICO       POR_MED
WHERE TAB_FAT.CD_TAB_FAT = VAL_POR.CD_TAB_FAT
AND  VAL_POR.CD_PORTE_MEDICO   = POR_MED.CD_PORTE_MEDICO
ORDER BY :CAMPO_ORDEM, VAL_POR.DT_VIGENCIA
