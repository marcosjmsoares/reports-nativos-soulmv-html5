--R_PRO_FAT_HIERARQUIZADO
SELECT * FROM (
SELECT  PRO_HIE.CD_PRO_FAT_HIERARQUIZADO CODIGO
               ,PRO_HIE.DS_PRO_FAT_HIERARQUIZADO DESCRICAO
               ,PRO_HIE.CD_PORTE_MEDICO CD_PORTE_MEDICO
               ,POR_MED.DS_PORTE_MEDICO DS_PORTE_MEDICO
               ,PRO_HIE.CD_PRO_FAT CD_PRO_FAT
               ,PRO_FAT.DS_PRO_FAT DS_PRO_FAT
               ,SUB_GRU.CD_SUB_GRU_PRO_HIERARQUIZADO  CD_SUB_GRU_PRO_HIERARQUIZADO
               ,SUB_GRU.DS_SUB_GRU_PRO_HIERARQUIZADO  DS_SUB_GRU_PRO_HIERARQUIZADO
               ,GRU_PRO.CD_GRU_PRO_HIERARQUIZADO  CD_GRU_PRO_HIERARQUIZADO
               ,GRU_PRO.DS_GRU_PRO_HIERARQUIZADO DS_GRU_PRO_HIERARQUIZADO
FROM  dbamv.SUB_GRU_PRO_HIERARQUIZADO SUB_GRU
           ,dbamv.GRU_PRO_HIERARQUIZADO GRU_PRO
           ,dbamv.PRO_FAT_HIERARQUIZADO PRO_HIE
           ,dbamv.PRO_FAT PRO_FAT
           ,dbamv.PORTE_MEDICO POR_MED
WHERE  GRU_PRO.CD_GRU_PRO_HIERARQUIZADO  =  SUB_GRU.CD_GRU_PRO_HIERARQUIZADO
      AND  PRO_HIE.CD_SUB_GRU_PRO_HIERARQUIZADO  =  SUB_GRU.CD_SUB_GRU_PRO_HIERARQUIZADO
      AND  PRO_FAT.CD_PRO_FAT = PRO_HIE.CD_PRO_FAT
      AND  POR_MED.CD_PORTE_MEDICO = PRO_HIE.CD_PORTE_MEDICO
ORDER BY :CAMPO_ORDEM
)
ORDER BY CD_GRU_PRO_HIERARQUIZADO,DS_GRU_PRO_HIERARQUIZADO,
CD_SUB_GRU_PRO_HIERARQUIZADO,DS_SUB_GRU_PRO_HIERARQUIZADO
