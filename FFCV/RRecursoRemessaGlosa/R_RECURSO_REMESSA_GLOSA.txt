--R_RECURSO_REMESSA_GLOSA
SELECT nota_fiscal,
  dt_emissao            ,
  dt_vencimento      ,
  dt_baixa           ,
vl_faturado,
Sum(vl_pago) AS vl_pago,
Sum(vl_iss) AS vl_iss,
Sum(vl_glosa) AS vl_glosa
FROM (
SELECT nf.cd_nota_fiscal nota_fiscal ,
 nf.dt_emissao dt_emissao            ,
 it.dt_vencimento dt_vencimento      ,
 trunc(r.dt_recebimento) dt_baixa           ,
  nf.vl_total_nota vl_faturado        ,
  r.vl_recebido vl_pago               ,
  nf.vl_iss vl_iss                    ,
  SUM(it_rec.vl_glosa) vl_glosa
   FROM dbamv.nota_fiscal nf    ,
  dbamv.itfat_nota_fiscal it_nf ,
  dbamv.it_recebimento it_rec   ,
  dbamv.con_rec c               ,
  dbamv.itcon_rec it            ,
  dbamv.reccon_rec r
  WHERE nf.cd_nota_fiscal = it_nf.cd_nota_fiscal
AND it_nf.cd_itfat_nf     = it_rec.cd_itfat_nf
AND nf.cd_nota_fiscal     = c.cd_nota_fiscal
AND c.cd_con_rec          = it.cd_con_rec
AND it.cd_itcon_rec       = r.cd_itcon_rec
AND it_rec.cd_reccon_rec  = r.cd_reccon_rec
GROUP BY nf.cd_nota_fiscal ,
  nf.dt_emissao,
  it.dt_vencimento,
  trunc(r.dt_recebimento),
  nf.vl_total_nota         ,
  r.vl_recebido            ,
  nf.vl_iss)
GROUP BY nota_fiscal,
         dt_emissao,
         dt_vencimento,
         dt_baixa,
         vl_faturado
