               AND        REMESSA.SN_FECHADA = 'S'
               and        atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
               UNION ALL
               SELECT        NULL CD_CONTABIL
                            ,'Não Informado'  DS_CONTA

                            ,CONVENIO.CD_CONVENIO   CD_CONVENIO
                            ,CONVENIO.NM_CONVENIO   NM_CONVENIO
                            ,ITREG_FAT.VL_TOTAL_CONTA       VL_TOTAL
               FROM   DBAMV.REG_FAT
                            ,DBAMV.ITREG_FAT
                            ,DBAMV.PRO_FAT
                            ,DBAMV.GRU_PRO
                            ,DBAMV.SETOR
                            ,DBAMV.REMESSA_FATURA     REMESSA
                            ,DBAMV.FATURA
                            ,DBAMV.CONVENIO
                            ,DBAMV.ATENDIME
                            ,DBAMV.GRU_FAT

               AND        ITREG_FAT.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
               AND        PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
               AND        ITREG_FAT.CD_GRU_FAT = GRU_FAT.CD_GRU_FAT(+)

               AND        REG_FAT.CD_CONVENIO = CONVENIO.CD_CONVENIO
               AND         BUSCA_CONTA_CONTABIL(:CAMPO_CD_MULTI_EMPRESA,CONVENIO.CD_CONVENIO,SETOR.CD_SETOR,
                                               GRU_FAT.CD_GRU_FAT,GRU_PRO.CD_GRU_PRO,GRU_FAT.CD_GRU_FAT) IS NULL
               AND        ITREG_FAT.SN_PERTENCE_PACOTE <>'S'
               and       nvl(itreg_fat.tp_pagamento,'X') <> 'C'
               AND        CONVENIO.TP_CONVENIO <> 'H'
               AND        REMESSA.SN_FECHADA = 'S'
