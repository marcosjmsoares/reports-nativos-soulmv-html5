--R_FAT_PREST
SELECT * FROM (
Select  TO_CHAR(fatura.dt_competencia, 'MM/YYYY')          dt_competencia
     ,  fatura.cd_convenio                   cd_convenio
     ,  convenio.nm_convenio            nm_convenio
     ,  itreg_fat.cd_prestador               cd_prestador
     ,  prestador.cd_tip_presta            cd_tip_presta
     ,  tip_presta.nm_tip_presta           nm_tip_presta
     ,  prestador.nm_prestador             nm_prestador
     ,  Sum( NVL(itreg_fat.vl_total_conta, 0) )  vl_total_conta
  From  dbamv.fatura                         fatura
     ,  dbamv.convenio                       convenio
     ,  dbamv.remessa_fatura              remessa_fatura
     ,  dbamv.reg_fat                           reg_fat
     ,  dbamv.itreg_fat                         itreg_fat
     ,  dbamv.prestador                       prestador
     ,  dbamv.tip_presta                      tip_presta
 Where TO_CHAR( dt_competencia, 'MM/YYYY') = TO_CHAR(:CAMPO_COMPET, 'MM/YYYY')
   AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   and  convenio.cd_convenio          like :CAMPO_CONV
   and  fatura.cd_convenio                = convenio.cd_convenio
   and  itreg_fat.cd_prestador            = prestador.cd_prestador
   and  fatura.cd_fatura                     = remessa_fatura.cd_fatura
   and  remessa_fatura.cd_remessa  = reg_fat.cd_remessa
   and  reg_fat.cd_reg_fat                 = itreg_fat.cd_reg_fat
   and  prestador.cd_tip_presta         = tip_presta.cd_tip_presta
Group By fatura.dt_competencia
       ,  fatura.cd_convenio
       ,  convenio.nm_convenio
       ,  itreg_fat.cd_prestador
       ,  prestador.nm_prestador
       ,  prestador.cd_tip_presta
       ,  tip_presta.nm_tip_presta
)
ORDER BY DT_COMPETENCIA,CD_CONVENIO DESC,NM_CONVENIO,
CD_TIP_PRESTA,NM_TIP_PRESTA, CD_PRESTADOR, NM_PRESTADOR
