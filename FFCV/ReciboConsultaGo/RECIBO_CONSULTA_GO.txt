--RECIBO_CONSULTA_GO
select nm_paciente||' ('||paciente.cd_paciente||')' nm_paciente
      ,nm_convenio
      ,'Dr(a) '||nm_prestador||' - CRM - '||ds_codigo_conselho Nm_prestador
      ,atendime.cd_Atendimento
      , cd_log
      , ds_pro_fat
       ,vl_log
 from dbamv.atendime
     ,dbamv.paciente
     ,dbamv.convenio
     ,dbamv.prestador
     ,dbamv.log_recibo
    ,dbamv.pro_fat
     ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
And atendime.cd_prestador = prestador.cd_prestador
  and atendime.cd_paciente  = paciente.cd_paciente
  and atendime.cd_convenio  = convenio.cd_convenio
  and log_recibo.cd_atendimento = atendime.cd_atendimento
  and pro_fat.cd_pro_fat = log_recibo.cd_pro_fat
  and atendime.cd_atendimento = :CAMPO_ATENDIMENTO
