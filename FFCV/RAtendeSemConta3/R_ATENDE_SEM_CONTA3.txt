--R_ATENDE_SEM_CONTA3
SELECT * FROM (
SELECT DISTINCT
               ATENDIME.CD_ATENDIMENTO
              ,ATENDIME.TP_ATENDIMENTO
              ,ATENDIME.CD_PACIENTE
              ,ATENDIME.DT_ATENDIMENTO
              ,ATENDIME.CD_LEITO
              ,ATENDIME.CD_CONVENIO
              ,PACIENTE.NM_PACIENTE
              ,CONVENIO.NM_CONVENIO
              ,LEITO.DS_LEITO
              ,UNID_INT.CD_UNID_INT
               ,nvl( UNID_INT.DS_UNID_INT,ori_ate.ds_ori_ate ) DS_UNID_INT
FROM    DBAMV.ATENDIME
              ,DBAMV.CONVENIO
              ,DBAMV.PACIENTE
              ,DBAMV.LEITO
              ,DBAMV.UNID_INT,
	     dbamv.empresa_convenio
              ,dbamv.ori_ate
WHERE  Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and ATENDIME.CD_ATENDIMENTO NOT IN ( SELECT CD_ATENDIMENTO FROM     DBAMV.ITREG_AMB )
      AND  ATENDIME.CD_ATENDIMENTO NOT IN ( SELECT CD_ATENDIMENTO FROM    DBAMV.REG_FAT where REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA )
      AND  ATENDIME.CD_CONVENIO  =  CONVENIO.CD_CONVENIO
      AND  CONVENIO.TP_CONVENIO  IN  ( 'C','P')
      AND  ATENDIME.CD_PACIENTE  =  PACIENTE.CD_PACIENTE
      AND  ATENDIME.CD_LEITO  =  LEITO.CD_LEITO(+)
      AND  LEITO.CD_UNID_INT   =  UNID_INT.CD_UNID_INT(+)
     and atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
     AND  NVL(ATENDIME.CD_ATENDIMENTO_PAI,0) NOT IN ( SELECT ATENDIME_PAI.CD_ATENDIMENTO
                                                    FROM DBAMV.ATENDIME  ATENDIME_PAI
                                                    WHERE NVL(ATENDIME_PAI.QT_SESSOES,0) > 0 )
and atendime.cd_ori_ate = ori_ate.cd_ori_ate
ORDER BY CD_CONVENIO, DT_ATENDIMENTO
)
ORDER BY NM_CONVENIO,CD_CONVENIO
