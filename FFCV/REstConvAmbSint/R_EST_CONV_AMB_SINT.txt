--R_EST_CONV_AMB_SINT
SELECT * FROM (
Select sum( Decode ( to_char( reg_amb.dt_lancamento, 'MM'), '01', Nvl(itreg_amb.vl_total_conta,0),0 ) ) vl_total_janeiro
      ,sum( Decode ( to_char( reg_amb.dt_lancamento, 'MM'), '02', Nvl(itreg_amb.vl_total_conta,0),0 ) ) vl_total_fevereiro
      ,sum( Decode ( to_char( reg_amb.dt_lancamento, 'MM'), '03', Nvl(itreg_amb.vl_total_conta,0),0 ) ) vl_total_marco
      ,sum( Decode ( to_char( reg_amb.dt_lancamento, 'MM'), '04', Nvl(itreg_amb.vl_total_conta,0),0 ) ) vl_total_abril
      ,sum( Decode ( to_char( reg_amb.dt_lancamento, 'MM'), '05', Nvl(itreg_amb.vl_total_conta,0),0 ) ) vl_total_maio
      ,sum( Decode ( to_char( reg_amb.dt_lancamento, 'MM'), '06', Nvl(itreg_amb.vl_total_conta,0),0 ) ) vl_total_junho
      ,sum( Decode ( to_char( reg_amb.dt_lancamento, 'MM'), '07', Nvl(itreg_amb.vl_total_conta,0),0 ) ) vl_total_julho
      ,sum( Decode ( to_char( reg_amb.dt_lancamento, 'MM'), '08', Nvl(itreg_amb.vl_total_conta,0),0 ) ) vl_total_agosto
      ,sum( Decode ( to_char( reg_amb.dt_lancamento, 'MM'), '09', Nvl(itreg_amb.vl_total_conta,0),0 ) ) vl_total_setembro
      ,sum( Decode ( to_char( reg_amb.dt_lancamento, 'MM'), '10', Nvl(itreg_amb.vl_total_conta,0),0 ) ) vl_total_outubro
      ,sum( Decode ( to_char( reg_amb.dt_lancamento, 'MM'), '11', Nvl(itreg_amb.vl_total_conta,0),0 ) ) vl_total_novembro
      ,sum( Decode ( to_char( reg_amb.dt_lancamento, 'MM'), '12', Nvl(itreg_amb.vl_total_conta,0),0 ) ) vl_total_dezembro
      ,sum( Nvl( itreg_amb.vl_total_conta, 0 ) ) vl_total_conta
      ,ori_ate.cd_ori_ate
      ,ori_ate.ds_ori_ate
      ,convenio.cd_convenio
      ,convenio.nm_convenio
  From dbamv.convenio
      ,dbamv.ori_ate
      ,dbamv.itreg_amb
      ,dbamv.reg_amb
      ,dbamv.atendime
      ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And reg_amb.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
      And reg_amb.dt_lancamento Between To_Date('01/'||:CAMPO_DT_COMPET ,'dd/MM/yyyy')
      AND Last_Day(To_Date('01/'||:CAMPO_DT_COMPET_FINAL,'dd/MM/yyyy'))
      And  itreg_amb.cd_reg_amb = reg_amb.cd_reg_amb
      And atendime.cd_atendimento = itreg_amb.cd_atendimento
      And convenio.cd_convenio = atendime.cd_convenio
      And ori_ate.cd_ori_ate = atendime.cd_ori_ate
Group By
       ori_ate.cd_ori_ate
      ,ori_ate.ds_ori_ate
      ,convenio.cd_convenio
      ,convenio.nm_convenio
)
ORDER BY CD_ORI_ATE,DS_ORI_ATE
