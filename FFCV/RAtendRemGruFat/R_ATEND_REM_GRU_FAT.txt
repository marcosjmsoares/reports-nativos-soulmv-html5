--R_ATEND_REM_GRU_FAT
Select * from
(Select cd_remessa,
           dt_entrega_da_fatura,
           cd_agrupamento,
           dt_competencia,
           cd_convenio,
           nm_convenio,
           nvl(ds_agrupamento,'Não Informado') ds_agrupamento,
           cd_atendimento,
           dt_atendimento,
           nm_paciente,
           vl_total_conta
From
(
Select remessa_fatura.cd_remessa,
       remessa_fatura.dt_entrega_da_fatura,
       remessa_fatura.cd_agrupamento,
       fatura.dt_competencia,
       fatura.cd_convenio,
       convenio.nm_convenio,
       agrupamento.ds_agrupamento,
       reg_fat.cd_atendimento,
       atendime.dt_atendimento,
       paciente.nm_paciente,
       sum(itreg_fat.vl_total_conta) vl_total_conta
  From dbamv.remessa_fatura,
       dbamv.fatura,
       dbamv.convenio,
       dbamv.agrupamento,
       dbamv.reg_fat,
       dbamv.paciente,
       dbamv.atendime,
       dbamv.itreg_fat,
       dbamv.empresa_convenio
 Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
 and remessa_fatura.cd_fatura = fatura.cd_fatura
   AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   and remessa_fatura.cd_agrupamento = agrupamento.cd_agrupamento (+)
   and fatura.cd_convenio = convenio.cd_convenio
   and reg_fat.cd_remessa = remessa_fatura.cd_remessa
   and reg_fat.cd_atendimento = atendime.cd_atendimento
   and atendime.cd_paciente = paciente.cd_paciente
   and reg_fat.cd_reg_fat = itreg_fat.cd_reg_fat
   and       nvl(itreg_fat.sn_pertence_pacote,'N') = 'N'
   and       nvl(reg_fat.sn_diagno,'N') = 'N'
   and       nvl(itreg_fat.tp_pagamento,'X') <> 'C'
   and       convenio.tp_convenio <> 'H'
group by remessa_fatura.cd_remessa,
              remessa_fatura.dt_entrega_da_fatura,
              remessa_fatura.cd_agrupamento,
              fatura.dt_competencia,
              fatura.cd_convenio,
              convenio.nm_convenio,
             agrupamento.ds_agrupamento,
             reg_fat.cd_atendimento,
             atendime.dt_atendimento,
             paciente.nm_paciente
Union All
Select reg_amb.cd_remessa,
       remessa_fatura.dt_entrega_da_fatura,
       remessa_fatura.cd_agrupamento,
       fatura.dt_competencia,
       fatura.cd_convenio,
       convenio.nm_convenio,
       agrupamento.ds_agrupamento,
       itreg_amb.cd_atendimento,
       atendime.dt_atendimento,
       paciente.nm_paciente,
       Sum( itreg_amb.vl_total_conta ) vl_total_conta
  From dbamv.remessa_fatura,
       dbamv.fatura,
       dbamv.convenio,
       dbamv.agrupamento,
       dbamv.reg_amb,
       dbamv.paciente,
       dbamv.atendime,
       dbamv.itreg_amb,
       dbamv.empresa_convenio
 Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
 and remessa_fatura.cd_fatura = fatura.cd_fatura
   AND REG_Amb.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   and remessa_fatura.cd_agrupamento = agrupamento.cd_agrupamento (+)
   and fatura.cd_convenio = convenio.cd_convenio
  and reg_amb.cd_remessa = remessa_fatura.cd_remessa
   and itreg_amb.cd_atendimento = atendime.cd_atendimento
   and atendime.cd_paciente = paciente.cd_paciente
   and reg_amb.cd_reg_amb = itreg_amb.cd_reg_amb
   and       nvl(itreg_amb.sn_pertence_pacote,'N') = 'N'
   and       nvl(reg_amb.sn_diagno,'N') = 'N'
   and       nvl(itreg_amb.tp_pagamento,'X') <> 'C'
   and       convenio.tp_convenio <> 'H'
Group By reg_amb.cd_remessa,
       remessa_fatura.dt_entrega_da_fatura,
       remessa_fatura.cd_agrupamento,
       fatura.dt_competencia,
       fatura.cd_convenio,
       convenio.nm_convenio,
       agrupamento.ds_agrupamento,
       itreg_amb.cd_atendimento,
       atendime.dt_atendimento,
       paciente.nm_paciente
)
:CAMPO_CF_ORDEM
)
ORDER BY CD_REMESSA,DT_ENTREGA_DA_FATURA,CD_AGRUPAMENTO,DT_COMPETENCIA,CD_CONVENIO,NM_CONVENIO,DS_AGRUPAMENTO,NM_PACIENTE
