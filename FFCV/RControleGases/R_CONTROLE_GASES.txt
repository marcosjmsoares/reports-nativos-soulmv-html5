--R_CONTROLE_GASES
SELECT distinct * FROM (
select  mvto_gases.cd_atendimento  cd_atendimento
      ,atendime.dt_atendimento  dt_atendimento
      ,To_Char(atendime.hr_atendimento,'HH24')    hr_atendimento
      ,atendime.dt_alta  dt_alta
      ,To_Char(atendime.hr_alta,'HH24')  hr_alta
      ,dbamv.pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(PACIENTE.CD_PACIENTE, 1)  nm_paciente
      ,mvto_gases.cd_setor  cd_setor
      ,setor.nm_setor  nm_setor
      ,mvto_gases.cd_pro_fat  cd_pro_fat
,decode( dbamv.pkg_mv2000.le_cliente, 270, substr(TIP_PRESC.ds_TIP_PRESC,1,50), pro_fat.ds_pro_fat)  ds_pro_fat  
      ,mvto_gases.dt_liga  dt_liga
      ,mvto_gases.hr_liga  hr_liga
      ,mvto_gases.dt_desliga  dt_desliga
      ,mvto_gases.hr_desliga  hr_desliga
      ,mvto_gases.cd_tip_presc  cd_tip_presc
from   dbamv.mvto_gases  mvto_gases
      ,dbamv.atendime  atendime
      ,dbamv.paciente  paciente
      ,dbamv.setor  setor
      ,dbamv.pro_fat  pro_fat
      ,dbamv.tip_presc 
where  atendime.cd_atendimento  =  :CAMPO_CD_ATENDIMENTO
  and  atendime.cd_atendimento  =  mvto_gases.cd_atendimento
  and  pro_fat.cd_pro_fat  =  mvto_gases.cd_pro_fat
  and  setor.cd_setor  =  mvto_gases.cd_setor
  and  paciente.cd_paciente  =  atendime.cd_paciente
  and  atendime.Cd_Multi_Empresa  = :CAMPO_CD_MULTI_EMPRESA
  and  pro_fat.cd_pro_fat   = tip_presc.cd_pro_fat       
  and  tip_presc.sn_liga_desligar = 'S'                
)
ORDER BY nm_setor,
         ds_pro_fat,
         dt_liga,
         hr_liga
