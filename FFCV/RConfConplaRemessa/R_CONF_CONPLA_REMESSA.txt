--R_CONF_CONPLA_REMESSA
SELECT * FROM (
SELECT CONVENIO.CD_CONVENIO,
       CONVENIO.NM_CONVENIO,
       CON_PLA.TP_TRIBUTO_RETIDO,
       CON_PLA.CD_CON_PLA,
       CON_PLA.DS_CON_PLA,
       REG_FAT.DT_INICIO  DT_ATENDIMENTO,
       REG_FAT.DT_FINAL   DT_ALTA,
       REG_FAT.CD_REG_FAT  CD_CONTA,
       ATENDIME.CD_ATENDIMENTO,
       ATENDIME.CD_ORI_ATE,
       ORI_ATE.DS_ORI_ATE,
       NVL( REG_FAT.VL_TOTAL_CONTA, 0)  VL_TOTAL_CONTA
FROM   DBAMV.ATENDIME,
       DBAMV.CONVENIO,
       DBAMV.REG_FAT,
       DBAMV.ORI_ATE,
       DBAMV.CON_PLA
Where REG_FAT.CD_REMESSA       = :CAMPO_CD_REMESSA
  AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
  AND  ATENDIME.CD_ATENDIMENTO  = REG_FAT.CD_ATENDIMENTO
  AND  REG_FAT.CD_CONVENIO      = CONVENIO.CD_CONVENIO
  AND  CON_PLA.CD_CONVENIO     = CONVENIO.CD_CONVENIO
  AND  CON_PLA.CD_CON_PLA       = REG_FAT.CD_CON_PLA
  AND  ORI_ATE.CD_ORI_ATE       = ATENDIME.CD_ORI_ATE
  AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
UNION ALL
SELECT DISTINCT CONVENIO.CD_CONVENIO,
       CONVENIO.NM_CONVENIO,
       CON_PLA.TP_TRIBUTO_RETIDO,
       CON_PLA.CD_CON_PLA,
       CON_PLA.DS_CON_PLA,
       ATENDIME.DT_ATENDIMENTO,
       ATENDIME.DT_ATENDIMENTO  DT_ALTA,
       ITREG_AMB.CD_REG_AMB  CD_CONTA,
       ATENDIME.CD_ATENDIMENTO,
       ATENDIME.CD_ORI_ATE,
       ORI_ATE.DS_ORI_ATE,
       SUM( NVL( ITREG_AMB.VL_TOTAL_CONTA, 0))  VL_TOTAL_CONTA
FROM   DBAMV.ATENDIME,
       DBAMV.CONVENIO,
       DBAMV.REG_AMB,
       DBAMV.ITREG_AMB,
       DBAMV.ORI_ATE,
       DBAMV.CON_PLA
Where REG_AMB.CD_REMESSA       = :CAMPO_CD_REMESSA
  AND REG_AMB.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
  AND  ITREG_AMB.CD_REG_AMB     = REG_AMB.CD_REG_AMB
  AND  ATENDIME.CD_ATENDIMENTO  = ITREG_AMB.CD_ATENDIMENTO
  AND  REG_AMB.CD_CONVENIO      = CONVENIO.CD_CONVENIO
  AND  CON_PLA.CD_CONVENIO     = CONVENIO.CD_CONVENIO
  AND  CON_PLA.CD_CON_PLA       = ATENDIME.CD_CON_PLA
  AND  ORI_ATE.CD_ORI_ATE       = ATENDIME.CD_ORI_ATE
  AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY CONVENIO.CD_CONVENIO,
       CONVENIO.NM_CONVENIO,
       CON_PLA.TP_TRIBUTO_RETIDO,
       CON_PLA.CD_CON_PLA,
       CON_PLA.DS_CON_PLA,
       ATENDIME.DT_ATENDIMENTO,
       ATENDIME.DT_ATENDIMENTO,
       ITREG_AMB.CD_REG_AMB,
       ATENDIME.CD_ATENDIMENTO,
       ATENDIME.CD_ORI_ATE,
       ORI_ATE.DS_ORI_ATE
ORDER BY 1, 3, 4, 9
)
ORDER BY CD_CONVENIO,NM_CONVENIO,
TP_TRIBUTO_RETIDO,
CD_CON_PLA,DS_CON_PLA, CD_ATENDIMENTO, CD_CONTA
