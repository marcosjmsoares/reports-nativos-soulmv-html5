--R_PORTE_MEDICO_PRO_FAT
SELECT * FROM (
SELECT TAB_FAT.CD_TAB_FAT                   CD_TAB_FAT1
              ,TAB_FAT.DS_TAB_FAT                   DS_TAB_FAT
              ,PRO_FAT.DS_PRO_FAT                   DESCRICAO
              ,PRO_FAT.CD_PRO_FAT                   CODIGO
              ,POR_MED.CD_PORTE_MEDICO                  CD_PORTE_MEDICO
              ,POR_MED.DS_PORTE_MEDICO                  DS_PORTE_MEDICO
              ,POR_PRO.DT_VIGENCIA DT_VIGENCIA
FROM  dbamv.PORTE_MEDICO_PRO_FAT       POR_PRO
           ,dbamv.TAB_FAT         TAB_FAT
           ,dbamv.PRO_FAT        PRO_FAT
           ,dbamv.PORTE_MEDICO       POR_MED
WHERE TAB_FAT.CD_TAB_FAT = POR_PRO.CD_TAB_FAT
AND   PRO_FAT.CD_PRO_FAT    = POR_PRO.CD_PRO_FAT
AND   POR_PRO.CD_PORTE_MEDICO   = POR_MED.CD_PORTE_MEDICO
ORDER BY :CAMPO_ORDEM, POR_PRO.DT_VIGENCIA
)
ORDER BY DS_TAB_FAT,CD_TAB_FAT1
