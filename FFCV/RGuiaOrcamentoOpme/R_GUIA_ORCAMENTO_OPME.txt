--R_GUIA_ORCAMENTO_OPME
SELECT vl_total_autorizado,
       cd_atendimento,
       cd_convenio,
       cd_aviso_cirurgia,
       nm_convenio,
       nm_paciente,
       cd_paciente,
       cd_senha,
       dt_atendimento,
       dt_programada,
       cd_prestador,
       nm_prestador,
       ds_observacao,
       cd_con_pla,
       tp_atendimento,
       dt_vigencia,
       cd_pro_fat,
       observacao_item,
       ds_pro_fat,
       nm_fornecedor,
       nm_laborator,
       qt_autorizado,
       vl_perc_fracao,
       vl_unitario,
       vl_total,
       vl_taxa,
       cd_rms,
       cd_opme_forn,
       qt_solicitada,
       qt_utilizada,
       sn_autorizado
FROM (
select atendime.cd_atendimento,
guia.cd_convenio  cd_convenio,
       aviso.cd_aviso_cirurgia cd_aviso_cirurgia,
       convenio.nm_convenio nm_convenio,
       nvl( dbamv.pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(PACIENTE.CD_PACIENTE, 1), aviso.nm_paciente )  nm_paciente,
       paciente.cd_paciente,
       guia.cd_senha cd_senha,
       nvl( atendime.dt_atendimento, aviso.dt_prev_inter ) dt_atendimento,
       age_cir.dt_inicio_age_cir dt_programada,    
       prestador.cd_prestador cd_prestador,
       prestador.nm_prestador nm_prestador,
       guia.ds_observacao ds_observacao,
       atendime.cd_con_pla cd_con_pla,
       atendime.tp_atendimento tp_atendimento,
       val_opme.dt_vigencia  dt_vigencia,
       it_guia.cd_pro_fat cd_pro_fat,
       it_guia.ds_observacao observacao_item,
decode( pro_fat.cd_pro_fat, null,  it_guia.ds_procedimento, it_guia.cd_pro_fat ||' - '|| pro_fat.ds_pro_fat ) ds_pro_fat,
 	   fornecedor.nm_fantasia nm_fornecedor,
  	   laborator.nm_laborator nm_laborator,
	   it_guia.qt_autorizado qt_autorizado,
  	   val_opme.vl_perc_fracao vl_perc_fracao,
	   nvl(Nvl(val_opme.vl_proposto,val_opme.vl_unitario),0) vl_unitario,
	   nvl(val_opme.vl_total,0) vl_total,
	   nvl(val_opme.vl_taxa_comercializacao,0) vl_taxa,
       nvl(val_opme.vl_total_autorizado,0) vl_total_autorizado,
       val_opme.cd_rms cd_rms,
       val_opme.cd_opme_forn cd_opme_forn
      ,it_guia.qt_autorizado              qt_solicitada
      ,it_guia.qt_utilizada               qt_utilizada
      ,val_opme.sn_autorizado
  from dbamv.guia,
       dbamv.aviso_cirurgia aviso,
       dbamv.convenio,
       dbamv.atendime,
       dbamv.paciente,
       (select prestador_aviso.cd_aviso_cirurgia, prestador_aviso.cd_prestador, prestador_aviso.cd_ati_med
         from dbamv.guia,
                 dbamv.cirurgia_aviso,
                 dbamv.prestador_aviso
 where guia.cd_guia = :CAMPO_CD_GUIA
   and guia.cd_aviso_cirurgia = cirurgia_aviso.cd_aviso_cirurgia
   and cirurgia_aviso.sn_principal = 'S'
   and guia.cd_aviso_cirurgia = prestador_aviso.cd_aviso_cirurgia
   and cirurgia_aviso.cd_cirurgia = prestador_aviso.cd_cirurgia
   and prestador_aviso.sn_principal = 'S') prestador_aviso,
       dbamv.ati_med,
       dbamv.prestador,
       dbamv.it_guia,
       dbamv.val_opme_it_guia val_opme,
       dbamv.pro_fat,
       dbamv.fornecedor,
       dbamv.laborator,
       dbamv.gru_pro
       ,dbamv.age_cir 
 where guia.cd_guia               = :CAMPO_CD_GUIA
   and aviso.cd_aviso_cirurgia    = guia.cd_aviso_cirurgia
   and convenio.cd_convenio(+)    = guia.cd_convenio
   and atendime.cd_atendimento(+) = guia.cd_atendimento
   and paciente.cd_paciente(+)     = aviso.cd_paciente
   and prestador_aviso.cd_aviso_cirurgia = guia.cd_aviso_cirurgia
   and ati_med.cd_ati_med(+)      = prestador_aviso.cd_ati_med
   and ati_med.tp_funcao          = 'C'
   and prestador.cd_prestador(+)  = prestador_aviso.cd_prestador
   and it_guia.cd_guia            = guia.cd_guia
   and val_opme.cd_guia(+)           = it_guia.cd_guia 
   and val_opme.cd_it_guia(+)        = it_guia.cd_it_guia 
   and pro_fat.cd_pro_fat(+)      = it_guia.cd_pro_fat
   and gru_pro.cd_gru_pro(+)      = pro_fat.cd_gru_pro
   and fornecedor.cd_fornecedor(+) = val_opme.cd_fornecedor
   and laborator.cd_laborator(+)  = val_opme.cd_laborator
   and age_cir.cd_aviso_cirurgia(+)  = aviso.cd_aviso_cirurgia 
    and  aviso.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
union
select atendime.cd_atendimento,
       guia.cd_convenio  cd_convenio,
       to_number(null)    cd_aviso_cirurgia,
       convenio.nm_convenio nm_convenio,
       dbamv.pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(PACIENTE.CD_PACIENTE, 1) nm_paciente,
       paciente.cd_paciente,
       guia.cd_senha cd_senha,
       atendime.dt_atendimento  dt_atendimento,
       to_date(null)    dt_programada,
       prestador.cd_prestador cd_prestador,
       prestador.nm_prestador nm_prestador,
       guia.ds_observacao ds_observacao,
       atendime.cd_con_pla cd_con_pla,
       atendime.tp_atendimento tp_atendimento,
       val_opme.dt_vigencia  dt_vigencia,
       it_guia.cd_pro_fat cd_pro_fat,
       it_guia.ds_observacao observacao_item,
       decode( pro_fat.cd_pro_fat, null,  it_guia.ds_procedimento, it_guia.cd_pro_fat ||' - '|| pro_fat.ds_pro_fat ) ds_pro_fat,
 	   fornecedor.nm_fantasia nm_fornecedor,
  	   laborator.nm_laborator nm_laborator,
	   it_guia.qt_autorizado qt_autorizado,
  	   val_opme.vl_perc_fracao vl_perc_fracao,
	   nvl(Nvl(val_opme.vl_proposto,val_opme.vl_unitario),0) vl_unitario,
	   nvl(val_opme.vl_total,0) vl_total,
	   nvl(val_opme.vl_taxa_comercializacao,0) vl_taxa,
       nvl(val_opme.vl_total_autorizado,0) vl_total_autorizado,
       val_opme.cd_rms cd_rms,
       val_opme.cd_opme_forn cd_opme_forn
      ,it_guia.qt_autorizado              qt_solicitada
      ,it_guia.qt_utilizada               qt_utilizada
      ,val_opme.sn_autorizado
  from dbamv.guia,
       dbamv.convenio,
       dbamv.atendime,
       dbamv.paciente,
       dbamv.prestador,
       dbamv.it_guia,
       dbamv.val_opme_it_guia val_opme,
       dbamv.pro_fat,
       dbamv.fornecedor,
       dbamv.laborator,
       dbamv.gru_pro
 where guia.cd_guia               = :CAMPO_CD_GUIA
   and convenio.cd_convenio(+)    = guia.cd_convenio
   and atendime.cd_atendimento(+) = guia.cd_atendimento
   and paciente.cd_paciente(+)     = atendime.cd_paciente
   and prestador.cd_prestador(+)  = atendime.cd_prestador
   and it_guia.cd_guia            = guia.cd_guia
   and val_opme.cd_guia(+)           = it_guia.cd_guia 
   and val_opme.cd_it_guia(+)        = it_guia.cd_it_guia 
   and pro_fat.cd_pro_fat(+)      = it_guia.cd_pro_fat
   and gru_pro.cd_gru_pro(+)      = pro_fat.cd_gru_pro
   and fornecedor.cd_fornecedor(+) = val_opme.cd_fornecedor
   and laborator.cd_laborator(+)  = val_opme.cd_laborator
   and guia.cd_aviso_cirurgia is null
   and guia.cd_atendimento is not null
   :CAMPO_CF_TP_REL
)
ORDER BY CD_CONVENIO,CD_AVISO_CIRURGIA,NM_CONVENIO,NM_PACIENTE,CD_ATENDIMENTO,CD_SENHA,DT_ATENDIMENTO,DT_PROGRAMADA,CD_PRESTADOR,NM_PRESTADOR,DS_OBSERVACAO,CD_CON_PLA,TP_ATENDIMENTO
