--R_TAB_COMPARA
SELECT * FROM (
SELECT  GRU_PRO.CD_GRU_PRO
               ,GRU_PRO.DS_GRU_PRO
               ,VAL_PRO.CD_PRO_FAT
               ,PRO_FAT.DS_PRO_FAT
               ,VAL_PRO.CD_TAB_FAT
               ,VAL_PRO.DT_VIGENCIA
               ,TAB_FAT.CD_TAB_FAT CD_TAB_FAT1
               ,TAB_FAT.DS_TAB_FAT
               ,TAB_FAT.TP_TAB_FAT
               ,VAL_PRO.VL_TOTAL
FROM      DBAMV.GRU_PRO
               ,DBAMV.PRO_FAT
               ,DBAMV.VAL_PRO
               ,DBAMV.TAB_FAT
WHERE  PRO_FAT.CD_PRO_FAT = VAL_PRO.CD_PRO_FAT
AND        GRU_PRO.CD_GRU_PRO = PRO_FAT.CD_GRU_PRO
AND        TAB_FAT.CD_TAB_FAT = VAL_PRO.CD_TAB_FAT
AND        NVL(VAL_PRO.SN_ATIVO, 'S') = decode(:CAMPO_SN_ATIVO, 'T', NVL(VAL_PRO.SN_ATIVO, 'S'), :CAMPO_SN_ATIVO)
AND         VAL_PRO.DT_VIGENCIA = (SELECT MAX(VAL.DT_VIGENCIA)
                                                               FROM   DBAMV.VAL_PRO VAL
                                                               WHERE  VAL.CD_PRO_FAT = VAL_PRO.CD_PRO_FAT
                                                                     AND  NVL(VAL.SN_ATIVO, 'S') = decode(:CAMPO_SN_ATIVO, 'T', NVL(VAL.SN_ATIVO, 'S'), :CAMPO_SN_ATIVO)
                                                               AND    VAL.CD_TAB_FAT = VAL_PRO.CD_TAB_FAT )
:CAMPO_CF_COND_GRU_PRO
)
ORDER BY replace(cd_pro_fat,'X','9'),CD_TAB_FAT
