--R_VAL_PROC
SELECT * FROM (
SELECT VAL_PROC_ORCAMENTO.CD_CONVENIO
      ,CONVENIO.NM_CONVENIO
      ,VAL_PROC_ORCAMENTO.CD_CON_PLA
      ,CON_PLA.DS_CON_PLA
      ,VAL_PROC_ORCAMENTO.DT_REFERENCIA
      ,DECODE( VAL_PROC_ORCAMENTO.TP_ATENDIMENTO, 'I', 'Internação', 'A', 'Ambulatorial', 'E', 'Externo', 'U', 'Urgência' ) TP_ATENDIMENTO
      ,VAL_PROC_ORCAMENTO.CD_TIP_ACOM
      ,TIP_ACOM.DS_TIP_ACOM
      ,VAL_PROC_ORCAMENTO.DS_ESPECIFICACAO
      ,ITVAL_PROC_ORCAMENTO.CD_PRO_FAT
      ,PRO_FAT.DS_PRO_FAT
      ,ITVAL_PROC_ORCAMENTO.VL_PRO_FAT
      ,ITVAL_PROC_ORCAMENTO.VL_PORTE
      ,(ITVAL_PROC_ORCAMENTO.VL_PRO_FAT - ITVAL_PROC_ORCAMENTO.VL_PORTE) VL_PROC
      ,ITVAL_PROC_ORCAMENTO.QTDE
      ,VAL_PROC_ORCAMENTO.DT_VALIDADE_ORCAMENTO
      ,VAL_PROC_ORCAMENTO.CD_PACIENTE
      ,VAL_PROC_ORCAMENTO.NM_PACIENTE
      ,VAL_PROC_ORCAMENTO.NR_TELEFONE
      ,VAL_PROC_ORCAMENTO.EMAIL
  FROM DBAMV.VAL_PROC_ORCAMENTO
      ,DBAMV.CONVENIO
      ,DBAMV.CON_PLA
      ,DBAMV.TIP_ACOM
      ,DBAMV.ITVAL_PROC_ORCAMENTO
      ,DBAMV.PRO_FAT
,Dbamv.Empresa_Con_Pla
Where Empresa_Con_Pla.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And Empresa_Con_Pla.Cd_Convenio = Con_Pla.Cd_Convenio
And Empresa_Con_Pla.Cd_Con_Pla = Con_Pla.Cd_Con_Pla
And CONVENIO.CD_CONVENIO                       = VAL_PROC_ORCAMENTO.CD_CONVENIO
   AND CON_PLA.CD_CONVENIO                        = VAL_PROC_ORCAMENTO.CD_CONVENIO
   AND CON_PLA.CD_CON_PLA                         = VAL_PROC_ORCAMENTO.CD_CON_PLA
   AND TIP_ACOM.CD_TIP_ACOM(+)                   = VAL_PROC_ORCAMENTO.CD_TIP_ACOM
   AND ITVAL_PROC_ORCAMENTO.CD_VAL_PROC_ORCAMENTO = VAL_PROC_ORCAMENTO.CD_VAL_PROC_ORCAMENTO
   AND PRO_FAT.CD_PRO_FAT                         = ITVAL_PROC_ORCAMENTO.CD_PRO_FAT
   AND VAL_PROC_ORCAMENTO.CD_VAL_PROC_ORCAMENTO = :CAMPO_CD_VAL_PROC_ORCAMENTO
)
ORDER BY CD_CONVENIO,NM_CONVENIO,CD_CON_PLA,DS_CON_PLA,DT_REFERENCIA,TP_ATENDIMENTO,CD_TIP_ACOM,DS_TIP_ACOM,DS_ESPECIFICACAO,DT_VALIDADE_ORCAMENTO,CD_PACIENTE,NM_PACIENTE,NR_TELEFONE,EMAIL
