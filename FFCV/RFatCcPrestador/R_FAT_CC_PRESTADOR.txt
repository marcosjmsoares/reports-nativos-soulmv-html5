--R_FAT_CC_PRESTADOR
SELECT * FROM (
SELECT  	NM_CONVENIO
    	,NM_SETOR
	,NM_PRESTADOR
	,SUM(VL_TOTAL_CONTA)  VALOR
FROM
(SELECT  CONVENIO.CD_CONVENIO 					 	CD_CONVENIO
       ,CONVENIO.NM_CONVENIO                  					NM_CONVENIO
       ,DECODE(SETOR.NM_SETOR, NULL, 'Não Informado', SETOR.NM_SETOR) 	NM_SETOR
       ,NVL(ITREG_FAT.VL_TOTAL_CONTA,0)              				VL_TOTAL_CONTA
       ,PRESTADOR.CD_PRESTADOR						CD_PRESTADOR
       ,PRESTADOR.NM_PRESTADOR 						NM_PRESTADOR
FROM    DBAMV.FATURA
       ,DBAMV.REMESSA_FATURA    REMESSA
       ,DBAMV.REG_FAT
       ,DBAMV.ITREG_FAT
       ,DBAMV.CONVENIO
       ,DBAMV.PRESTADOR
       ,DBAMV.ATENDIME
       ,DBAMV.SETOR
       ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And :CAMPO_FATU_CONSU = 'F'
AND   FATURA.CD_FATURA = REMESSA.CD_FATURA
AND   REMESSA.CD_REMESSA = REG_FAT.CD_REMESSA
AND   REG_FAT.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
AND   nvl(ITREG_FAT.SN_PERTENCE_PACOTE,'N') = 'N'
AND   ITREG_FAT.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
AND   ITREG_FAT.CD_SETOR = SETOR.CD_SETOR(+)
AND   NVL(ITREG_FAT.TP_PAGAMENTO,'X') <> 'C'
AND   REG_FAT.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
AND   REG_FAT.CD_CONVENIO    = CONVENIO.CD_CONVENIO
AND   CONVENIO.TP_CONVENIO in ( 'C','P')
AND   REMESSA.SN_FECHADA = 'S'
and        atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
UNION ALL
SELECT  CONVENIO.CD_CONVENIO 					 	CD_CONVENIO
       ,CONVENIO.NM_CONVENIO                  					NM_CONVENIO
       ,DECODE(SETOR.NM_SETOR, NULL, 'Não Informado', SETOR.NM_SETOR) 	NM_SETOR
       , (ITLAN_MED.VL_ATO * ITREG_FAT.QT_LANCAMENTO) +
         ( NVL(ITREG_FAT.VL_FILME_UNITARIO,0) * ITREG_FAT.QT_LANCAMENTO * ( ATI_MED.VL_PERCENTUAL_PAGO  / 100 )  )VL_TOTAL_CONTA
       ,PRESTADOR.CD_PRESTADOR						CD_PRESTADOR
       ,PRESTADOR.NM_PRESTADOR 						NM_PRESTADOR
FROM    DBAMV.FATURA
       ,DBAMV.REMESSA_FATURA    REMESSA
       ,DBAMV.REG_FAT
       ,DBAMV.ITREG_FAT
       ,DBAMV.CONVENIO
       ,DBAMV.PRESTADOR
       ,DBAMV.ATENDIME
       ,DBAMV.SETOR
       ,DBAMV.ITLAN_MED
       ,DBAMV.ATI_MED
       ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And :CAMPO_FATU_CONSU = 'F'
AND   FATURA.CD_FATURA = REMESSA.CD_FATURA
AND   ITLAN_MED.CD_ATI_MED = ATI_MED.CD_ATI_MED
AND   REMESSA.CD_REMESSA = REG_FAT.CD_REMESSA
AND   REG_FAT.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
AND   nvl(ITREG_FAT.SN_PERTENCE_PACOTE,'N') = 'N'
AND   ITREG_FAT.CD_PRESTADOR IS NULL
AND   ITREG_FAT.CD_SETOR = SETOR.CD_SETOR(+)
AND   NVL(ITLAN_MED.TP_PAGAMENTO,'X') <> 'C'
AND   REG_FAT.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
AND   REG_FAT.CD_CONVENIO    = CONVENIO.CD_CONVENIO
AND   CONVENIO.TP_CONVENIO in ( 'C','P')
AND   REMESSA.SN_FECHADA = 'S'
AND   ITREG_FAT.CD_REG_FAT = ITLAN_MED.CD_REG_FAT
AND   ITREG_FAT.CD_LANCAMENTO = ITLAN_MED.CD_LANCAMENTO
AND   ITLAN_MED.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
and        atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
UNION ALL
SELECT  CONVENIO.CD_CONVENIO                  					CD_CONVENIO
       ,CONVENIO.NM_CONVENIO                  					NM_CONVENIO
       ,DECODE(SETOR.NM_SETOR, NULL, 'Não Informado', SETOR.NM_SETOR) 	NM_SETOR
       ,NVL(ITREG_AMB.VL_TOTAL_CONTA,0)              				VL_TOTAL_CONTA
       ,PRESTADOR.CD_PRESTADOR						CD_PRESTADOR
       ,PRESTADOR.NM_PRESTADOR						NM_PRESTADOR
FROM   DBAMV.FATURA
      ,DBAMV.REMESSA_FATURA    REMESSA
      ,DBAMV.REG_AMB
      ,DBAMV.ITREG_AMB
      ,DBAMV.CONVENIO
      ,DBAMV.PRESTADOR
      ,DBAMV.ATENDIME
      ,DBAMV.SETOR
      ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And :CAMPO_FATU_CONSU = 'F'
AND   FATURA.CD_FATURA = REMESSA.CD_FATURA
AND   REMESSA.CD_REMESSA = REG_AMB.CD_REMESSA
AND   REG_AMB.CD_REG_AMB = ITREG_AMB.CD_REG_AMB
AND   nvl(ITREG_AMB.SN_PERTENCE_PACOTE,'N') = 'N'
AND   NVL(ITREG_AMB.TP_PAGAMENTO,'X') <> 'C'
AND   NVL(ITREG_AMB.TP_PAGAMENTO,'X') <> 'C'
AND   ITREG_AMB.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
AND   ITREG_AMB.CD_CONVENIO    = CONVENIO.CD_CONVENIO
AND   ITREG_AMB.CD_PRESTADOR  = PRESTADOR.CD_PRESTADOR
AND   ITREG_AMB.CD_SETOR = SETOR.CD_SETOR(+)
AND   CONVENIO.TP_CONVENIO in ( 'C','P')
AND   REMESSA.SN_FECHADA = 'S'
and        atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
UNION ALL
SELECT  CONVENIO.CD_CONVENIO                  					CD_CONVENIO
       ,CONVENIO.NM_CONVENIO                  					NM_CONVENIO
       ,DECODE(SETOR.NM_SETOR, NULL, 'Não Informado', SETOR.NM_SETOR) 	NM_SETOR
       ,NVL(ITREG_FAT.VL_TOTAL_CONTA,0)              				VL_TOTAL_CONTA
       ,PRESTADOR.CD_PRESTADOR						CD_PRESTADOR
       ,PRESTADOR.NM_PRESTADOR 						NM_PRESTADOR
FROM    DBAMV.FATURA
       ,DBAMV.REMESSA_FATURA    REMESSA
       ,DBAMV.REG_FAT
       ,DBAMV.ITREG_FAT
       ,DBAMV.CONVENIO
       ,DBAMV.PRESTADOR
       ,DBAMV.ATENDIME
       ,DBAMV.PRO_FAT
       ,DBAMV.SETOR
       ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And :CAMPO_FATU_CONSU = 'C'
AND    FATURA.CD_FATURA = REMESSA.CD_FATURA
AND   REMESSA.CD_REMESSA = REG_FAT.CD_REMESSA
AND   REG_FAT.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
AND   ITREG_FAT.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
AND   ITREG_FAT.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
AND   ITREG_FAT.CD_SETOR = SETOR.CD_SETOR(+)
AND   PRO_FAT.SN_PACOTE = 'N'
AND   NVL(ITREG_FAT.TP_PAGAMENTO,'X') <> 'C'
AND   REG_FAT.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
AND   REG_FAT.CD_CONVENIO    = CONVENIO.CD_CONVENIO
AND   CONVENIO.TP_CONVENIO in ( 'C','P')
AND   REMESSA.SN_FECHADA = 'S'
and        atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
UNION ALL
SELECT  CONVENIO.CD_CONVENIO                  					CD_CONVENIO
       ,CONVENIO.NM_CONVENIO                  					NM_CONVENIO
       ,DECODE(SETOR.NM_SETOR, NULL, 'Não Informado', SETOR.NM_SETOR) 	NM_SETOR
       , (ITLAN_MED.VL_ATO * ITREG_FAT.QT_LANCAMENTO) +
         ( NVL(ITREG_FAT.VL_FILME_UNITARIO,0) * ITREG_FAT.QT_LANCAMENTO * ( ATI_MED.VL_PERCENTUAL_PAGO  / 100 )  ) VL_TOTAL_CONTA
       ,PRESTADOR.CD_PRESTADOR						CD_PRESTADOR
       ,PRESTADOR.NM_PRESTADOR 						NM_PRESTADOR
FROM    DBAMV.FATURA
       ,DBAMV.REMESSA_FATURA    REMESSA
       ,DBAMV.REG_FAT
       ,DBAMV.ITREG_FAT
       ,DBAMV.CONVENIO
       ,DBAMV.PRESTADOR
       ,DBAMV.ATENDIME
       ,DBAMV.PRO_FAT
       ,DBAMV.SETOR
       ,DBAMV.ITLAN_MED
       ,DBAMV.ATI_MED
       ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And :CAMPO_FATU_CONSU = 'C'
AND   FATURA.CD_FATURA = REMESSA.CD_FATURA
AND   ITLAN_MED.CD_ATI_MED = ATI_MED.CD_ATI_MED
AND   REMESSA.CD_REMESSA = REG_FAT.CD_REMESSA
AND   REG_FAT.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
AND   ITREG_FAT.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
AND   ITREG_FAT.CD_PRESTADOR IS NULL
AND   ITREG_FAT.CD_SETOR = SETOR.CD_SETOR(+)
AND   PRO_FAT.SN_PACOTE = 'N'
AND   NVL(ITLAN_MED.TP_PAGAMENTO,'X') <> 'C'
AND   REG_FAT.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
AND   REG_FAT.CD_CONVENIO    = CONVENIO.CD_CONVENIO
AND   CONVENIO.TP_CONVENIO in ( 'C','P')
AND   REMESSA.SN_FECHADA = 'S'
AND   ITREG_FAT.CD_REG_FAT = ITLAN_MED.CD_REG_FAT
AND   ITREG_FAT.CD_LANCAMENTO = ITLAN_MED.CD_LANCAMENTO
AND   ITLAN_MED.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
and        atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
UNION ALL
SELECT  CONVENIO.CD_CONVENIO                  					CD_CONVENIO
       ,CONVENIO.NM_CONVENIO                  					NM_CONVENIO
       ,DECODE(SETOR.NM_SETOR, NULL, 'Não Informado', SETOR.NM_SETOR) 	NM_SETOR
       ,NVL(ITREG_AMB.VL_TOTAL_CONTA,0)              				VL_TOTAL_CONTA
       ,PRESTADOR.CD_PRESTADOR						CD_PRESTADOR
       ,PRESTADOR.NM_PRESTADOR						NM_PRESTADOR
FROM   DBAMV.FATURA
      ,DBAMV.REMESSA_FATURA    REMESSA
      ,DBAMV.REG_AMB
      ,DBAMV.ITREG_AMB
      ,DBAMV.PRO_FAT
      ,DBAMV.CONVENIO
      ,DBAMV.PRESTADOR
      ,DBAMV.ATENDIME
      ,DBAMV.SETOR
      ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And :CAMPO_FATU_CONSU = 'C'
AND   FATURA.CD_FATURA = REMESSA.CD_FATURA
AND   REMESSA.CD_REMESSA = REG_AMB.CD_REMESSA
AND   REG_AMB.CD_REG_AMB = ITREG_AMB.CD_REG_AMB
AND   ITREG_AMB.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
AND   PRO_FAT.SN_PACOTE = 'N'
AND   NVL(ITREG_AMB.TP_PAGAMENTO,'X') <> 'C'
AND   ITREG_AMB.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
AND   ITREG_AMB.CD_CONVENIO    = CONVENIO.CD_CONVENIO
AND   ITREG_AMB.CD_PRESTADOR   = PRESTADOR.CD_PRESTADOR
AND   ITREG_AMB.CD_SETOR = SETOR.CD_SETOR(+)
AND   CONVENIO.TP_CONVENIO in ( 'C','P')
AND   REMESSA.SN_FECHADA = 'S'
and        atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
:CAMPO_CF_CONTA_FECHADA_AMB
)
GROUP BY NM_CONVENIO
         ,NM_SETOR
         ,NM_PRESTADOR
)
ORDER BY NM_PRESTADOR,
NM_CONVENIO,
NM_SETOR
