--R_FATURA_ATENDIME_CEDIMAT
SELECT To_char(reg_amb.cd_reg_amb)CONTA,
     reg_amb.dt_lancamento AS DT_LANCAMENTO,
     reg_amb.cd_multi_empresa AS CD_MULTI_EMPRESA,
	 hospital.LOGOTIPO
FROM
     reg_amb,
     itreg_amb,
	 hospital
WHERE
     reg_amb.cd_reg_amb = itreg_amb.cd_reg_amb
	AND cd_atendimento = :CAMPO_CD_ATENDIMENTO
	AND  HOSPITAL.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
	AND ROWNUM = 1
