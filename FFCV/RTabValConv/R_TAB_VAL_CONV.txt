SELECT * FROM (
SELECT PRO_FAT.CD_GRU_PRO             CD_GRU_PRO
              ,PRO_FAT.CD_PRO_FAT              CD_PRO_FAT
              ,PRO_FAT.DS_PRO_FAT               DS_PRO_FAT
              ,ITREGRA.CD_TAB_FAT               CD_TAB_FAT
              ,CON_PLA.CD_CON_PLA              CD_CON_PLA
              ,CON_PLA.DS_CON_PLA               DS_CON_PLA
FROM DBAMV.ITREGRA    ITREGRA
          ,DBAMV.PRO_FAT    PRO_FAT
          ,DBAMV.CON_PLA    CON_PLA
,Dbamv.Empresa_Con_Pla
Where Empresa_Con_Pla.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And Empresa_Con_Pla.Cd_Convenio = Con_Pla.Cd_Convenio
And Empresa_Con_Pla.Cd_Con_Pla = Con_Pla.Cd_Con_Pla
And CON_PLA.CD_CONVENIO = :CAMPO_CD_CONVENIO
      AND ITREGRA.CD_REGRA = empresa_CON_PLA.CD_REGRA  
      AND PRO_FAT.CD_GRU_PRO = ITREGRA.CD_GRU_PRO
      AND PRO_FAT.CD_SUS IS NULL
GROUP BY PRO_FAT.CD_GRU_PRO
                   ,PRO_FAT.CD_PRO_FAT
                   ,PRO_FAT.DS_PRO_FAT
                   ,ITREGRA.CD_TAB_FAT
                   ,CON_PLA.CD_CON_PLA
                   ,CON_PLA.DS_CON_PLA
UNION ALL
SELECT PRO_FAT.CD_GRU_PRO             CD_GRU_PRO
              ,PRO_FAT.CD_PRO_FAT              CD_PRO_FAT
              ,PRO_FAT.DS_PRO_FAT               DS_PRO_FAT
              ,0            CD_TAB_FAT
              ,0    CD_CON_PLA
              ,'TODOS'        DS_CON_PLA
FROM DBAMV.PRO_FAT    PRO_FAT
WHERE 1 = 1
and :CAMPO_SN_MOSTRA_PLANO = 'N'
AND PRO_FAT.CD_SUS IS NULL
:CAMPO_CF_ORDER
)
ORDER BY CD_GRU_PRO,
DS_PRO_FAT,CD_PRO_FAT

--R_TAB_VAL_CONV_R_GRU_PRO3_Q_VAL_PRO2
SELECT * FROM (
SELECT proced.CD_GRU_PRO             CD_GRU_PRO
              , proced.CD_PRO_FAT              CD_PRO_FAT
              , proced.CD_TAB_FAT               CD_TAB_FAT
              , proced.CD_CON_PLA              CD_CON_PLA
              , proced.DS_CON_PLA              DS_CON_PLA
              ,nvl(tab_convenio.dt_vigencia, proced.dt_vigencia)  dt_vigencia
FROM   (SELECT PRO_FAT.CD_GRU_PRO  CD_GRU_PRO
                            ,PRO_FAT.CD_PRO_FAT   CD_PRO_FAT
                            ,ITREGRA.CD_TAB_FAT    CD_TAB_FAT
                            ,CON_PLA.CD_CON_PLA   CD_CON_PLA
                            ,CON_PLA.DS_CON_PLA   DS_CON_PLA
                            ,CON_PLA.CD_CONVENIO CD_CONVENIO
                            ,val_pro.dt_vigencia             dt_vigencia
               FROM DBAMV.ITREGRA             ITREGRA
                          ,DBAMV.PRO_FAT             PRO_FAT
                          ,DBAMV.CON_PLA             CON_PLA
                          ,dbamv.val_pro                    val_pro
,Dbamv.Empresa_Con_Pla
Where Empresa_Con_Pla.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And Empresa_Con_Pla.Cd_Convenio = Con_Pla.Cd_Convenio
And Empresa_Con_Pla.Cd_Con_Pla = Con_Pla.Cd_Con_Pla
And CON_PLA.CD_CONVENIO = :CAMPO_CD_CONVENIO
                  AND ITREGRA.CD_REGRA = empresa_CON_PLA.CD_REGRA
                  AND PRO_FAT.CD_GRU_PRO = ITREGRA.CD_GRU_PRO
                   AND       NVL(VAL_PRO.SN_ATIVO, 'S') = decode(:CAMPO_SN_ATIVO, 'T', NVL(VAL_PRO.SN_ATIVO, 'S'), :CAMPO_SN_ATIVO)
                  and val_pro.cd_pro_fat = pro_fat.cd_pro_fat
                  and val_pro.cd_tab_fat = itregra.cd_tab_fat
                  AND PRO_FAT.CD_SUS IS NULL
             GROUP BY PRO_FAT.CD_GRU_PRO
                                ,PRO_FAT.CD_PRO_FAT
                                ,ITREGRA.CD_TAB_FAT
                                ,CON_PLA.CD_CON_PLA
                                ,CON_PLA.DS_CON_PLA
                                ,CON_PLA.CD_CONVENIO
                                ,val_pro.dt_vigencia) proced
             , dbamv.tab_convenio
where  tab_convenio.cd_pro_fat(+)  = proced.cd_pro_fat
  and tab_convenio.cd_convenio(+) = proced.cd_convenio
  and tab_convenio.cd_con_pla(+)   = proced.cd_con_pla
UNION ALL
SELECT pro_fat.CD_GRU_PRO
              ,pro_fat.CD_PRO_FAT
              ,0            CD_TAB_FAT
              ,0    CD_CON_PLA
              ,'TODOS'        DS_CON_PLA
              ,tab_convenio.dt_vigencia  dt_vigencia
FROM DBAMV.PRO_FAT    pro_fat,
     dbamv.tab_convenio
WHERE 1 = 1
and :CAMPO_SN_MOSTRA_PLANO = 'N'
  and tab_convenio.cd_pro_fat = pro_fat.cd_pro_fat
  and tab_convenio.cd_convenio = :CAMPO_CD_CONVENIO
  and tab_convenio.cd_con_pla is NULL
  AND PRO_FAT.CD_SUS IS NULL
)
ORDER BY  2,3,4,6
