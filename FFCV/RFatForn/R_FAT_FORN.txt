--R_FAT_FORN
SELECT CD_FORNECEDOR
             ,NM_FORNECEDOR
             ,SUM(TOTAL)  TOTAL
FROM
(
SELECT FORN.CD_FORNECEDOR
              ,FORN.NM_FORNECEDOR
              ,ITREG_FAT.VL_TOTAL_CONTA  TOTAL
FROM    DBAMV.ITCOB_PRE                  ITCOB
             ,DBAMV.FORNECEDOR             FORN
             ,DBAMV.REG_FAT                      REG_FAT
             ,DBAMV.REMESSA_FATURA    REMESSA
             ,DBAMV.FATURA                        FATURA
             ,DBAMV.ITREG_FAT                   ITREG_FAT
             ,DBAMV.CONVENIO                    CONVENIO
	     ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And ITCOB.CD_FORNECEDOR = FORN.CD_FORNECEDOR
AND        ITCOB.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
AND        ITCOB.CD_LANCAMENTO = ITREG_FAT.CD_LANCAMENTO
AND        ITREG_FAT.CD_REG_FAT = REG_FAT.CD_REG_FAT
AND        REG_FAT.CD_REMESSA = REMESSA.CD_REMESSA
AND        REMESSA.CD_FATURA = FATURA.CD_FATURA
AND        REG_FAT.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND        CONVENIO.TP_CONVENIO <> 'H'
UNION ALL
SELECT FORN.CD_FORNECEDOR
              ,FORN.NM_FORNECEDOR
              ,ITREG_AMB.VL_TOTAL_CONTA      TOTAL
FROM    DBAMV.ITCOB_PRE_AMBU      ITCOB
             ,DBAMV.FORNECEDOR             FORN
             ,DBAMV.REG_AMB                     REG_AMB
             ,DBAMV.REMESSA_FATURA    REMESSA
             ,DBAMV.FATURA                        FATURA
             ,DBAMV.ITREG_AMB                  ITREG_AMB
             ,DBAMV.CONVENIO                    CONVENIO
	     ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And ITCOB.CD_FORNECEDOR         = FORN.CD_FORNECEDOR
AND        ITCOB.CD_REG_AMB                 = ITREG_AMB.CD_REG_AMB
AND        ITCOB.CD_LANCAMENTO         = ITREG_AMB.CD_LANCAMENTO
AND        ITREG_AMB.CD_REG_AMB      = REG_AMB.CD_REG_AMB
AND        REG_AMB.CD_REMESSA          = REMESSA.CD_REMESSA
AND        REMESSA.CD_FATURA            = FATURA.CD_FATURA
AND        REG_AMB.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND        REG_AMB.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND        CONVENIO.TP_CONVENIO <> 'H'
:CAMPO_CF_COND_CONV_AMBU
)
GROUP BY  CD_FORNECEDOR
                   ,NM_FORNECEDOR
ORDER BY TOTAL DESC
