--R_SITUACAO_RPS
SELECT * FROM (
SELECT SITUACAO,
       NRO_RPS,
       NR_NFE,
       CD_VERIFICACAO_NFE,
       CD_ATENDIMENTO,
       NM_CLIENTE,
       CD_CONVENIO,
       SUBSTR(DT_COMPETENCIA_RPS,4,7) COMPETENCIA,
       DT_COMPETENCIA_RPS,
       DT_EMISSAO_NFE,
       CD_FORMULARIO_NF || ' - ' || NM_FORMULARIO_RPS NM_FORMULARIO_RPS,
       RPS_SUBSTITUIDA,
       STATUS_RPS,
       DS_MOT_SITUACAO,
       DT_CANCELAMENTO,
       CD_NF
      FROM (
            SELECT
                 'SEM ENVIO' SITUACAO,
                  NOTA_FISCAL.NR_ID_NOTA_FISCAL NRO_RPS,
                  NOTA_FISCAL.NR_NOTA_FISCAL_NFE NR_NFE,
                  Decode ( NOTA_FISCAL.CD_VERIFICACAO_NFE, NULL, NULL,'''' || NOTA_FISCAL.CD_VERIFICACAO_NFE || '''') CD_VERIFICACAO_NFE,
                  NOTA_FISCAL.CD_ATENDIMENTO CD_ATENDIMENTO,
                  NOTA_FISCAL.NM_CLIENTE NM_CLIENTE,
                  NOTA_FISCAL.CD_CONVENIO CD_CONVENIO,
                  CONVENIO.NM_CONVENIO NM_CONVENIO,
                  TO_CHAR(NOTA_FISCAL.DT_EMISSAO,'DD/MM/YYYY') DT_COMPETENCIA_RPS,
                  TO_CHAR(NOTA_FISCAL.HR_EMISSAO_NFE,'DD/MM/YYYY') DT_EMISSAO_NFE,
                  NOTA_FISCAL.VL_TOTAL_NOTA VL_TOTAL_RPS,
                  FORMULARIO_NF.DS_FORMULARIO_NF NM_FORMULARIO_RPS,
                  NOTA_FISCAL.CD_NOTA_FISCAL_ORIGEM RPS_SUBSTITUIDA,
                  Decode (NOTA_FISCAL.CD_STATUS,'E', 'Emitida','C', 'Cancelada',NOTA_FISCAL.CD_STATUS) STATUS_RPS,
                  MOTIVO_SITUACAO_CONREC.DS_MOT_SITUACAO,
                  TO_CHAR(NOTA_FISCAL.DT_CANCELAMENTO,'DD/MM/YYYY HH24') DT_CANCELAMENTO,
                  NOTA_FISCAL.CD_NOTA_FISCAL CD_NF,
                  NOTA_FISCAL.CD_FORMULARIO_NF
            FROM
                  DBAMV.NOTA_FISCAL,
                  DBAMV.CONVENIO,
                  DBAMV.FORMULARIO_NF,
                  DBAMV.MOTIVO_SITUACAO_CONREC
            WHERE NOTA_FISCAL.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
              AND NOTA_FISCAL.VL_TOTAL_NOTA > 0
              AND NOTA_FISCAL.NR_NOTA_FISCAL_NFE IS NULL
              AND NOTA_FISCAL.HR_EMISSAO_NFE IS NULL
              AND TRUNC(NOTA_FISCAL.DT_EMISSAO) >= TO_DATE(:CAMPO_DT_INICIAL,'DD/MM/YYYY')
              AND TRUNC(NOTA_FISCAL.DT_EMISSAO) <= TO_DATE(:CAMPO_DT_FINAL,'DD/MM/YYYY')
              AND NOTA_FISCAL.CD_CONVENIO = CONVENIO.CD_CONVENIO(+)
              AND NOTA_FISCAL.CD_FORMULARIO_NF = FORMULARIO_NF.CD_FORMULARIO_NF
              AND NOTA_FISCAL.CD_MOT_SITUACAO = MOTIVO_SITUACAO_CONREC.CD_MOT_SITUACAO(+)
            UNION ALL
            SELECT
                  'SEM RETORNO' SITUACAO,
                  NOTA_FISCAL.NR_ID_NOTA_FISCAL NRO_RPS,
                  NOTA_FISCAL.NR_NOTA_FISCAL_NFE NR_NFE,
                  Decode ( NOTA_FISCAL.CD_VERIFICACAO_NFE, NULL, NULL,'''' || NOTA_FISCAL.CD_VERIFICACAO_NFE || '''') CD_VERIFICACAO_NFE,
                  NOTA_FISCAL.CD_ATENDIMENTO CD_ATENDIMENTO,
                  NOTA_FISCAL.NM_CLIENTE NM_CLIENTE,
                  NOTA_FISCAL.CD_CONVENIO CD_CONVENIO,
                  CONVENIO.NM_CONVENIO NM_CONVENIO,
                  TO_CHAR(NOTA_FISCAL.DT_EMISSAO,'DD/MM/YYYY') DT_COMPETENCIA_RPS,
                  TO_CHAR(NOTA_FISCAL.HR_EMISSAO_NFE,'DD/MM/YYYY') DT_EMISSAO_NFE,
                  NOTA_FISCAL.VL_TOTAL_NOTA VL_TOTAL_RPS,
                  FORMULARIO_NF.DS_FORMULARIO_NF NM_FORMULARIO_RPS,
                  NOTA_FISCAL.CD_NOTA_FISCAL_ORIGEM RPS_SUBSTITUIDA,
                  Decode (NOTA_FISCAL.CD_STATUS,'E', 'Emitida','C', 'Cancelada',NOTA_FISCAL.CD_STATUS) STATUS_RPS,
                  MOTIVO_SITUACAO_CONREC.DS_MOT_SITUACAO,
                  TO_CHAR(NOTA_FISCAL.DT_CANCELAMENTO,'DD/MM/YYYY HH24') DT_CANCELAMENTO,
                  NOTA_FISCAL.CD_NOTA_FISCAL CD_NF,
                  NOTA_FISCAL.CD_FORMULARIO_NF
              FROM (  SELECT MAX(RESP.DS_NUMERO_LOTE) DS_NUMERO_LOTE,
                              MAX(RESP.DS_NUMERO_PROTOCOLO) DS_NUMERO_PROTOCOLO,
                              RPS.DS_NR_RT,
                              RPS.CD_SERIE_RT,
                              NVL(RPS.NM_RAZAO_TOMADOR,RPS.NM_FANTASIA_TOMADOR) NOME_TOMADOR
                        FROM DBAMV.NOTA_FISCAL_ELET_NOT_FIS_DIGI RPS,
                              DBAMV.NOTA_FISCAL_ELETRONICA_LOT_ENV LOTE,
                              DBAMV.NOTA_FISCAL_ELETRONICA_LOT_RSP RESP
                        WHERE LOTE.ID = RPS.ID_PAI
                          AND LOTE.ID = RESP.DS_NUMERO_LOTE
                    GROUP BY RPS.DS_NR_RT,RPS.CD_SERIE_RT,
                              NVL(RPS.NM_RAZAO_TOMADOR,RPS.NM_FANTASIA_TOMADOR)
                    ) PROT ,
                    DBAMV.NOTA_FISCAL_ELETRONICA_LOT_RSP RSSP,
                    DBAMV.NOTA_FISCAL ,
                    DBAMV.FORMULARIO_NF F,
                    DBAMV.CONVENIO,
                    DBAMV.FORMULARIO_NF,
                    DBAMV.MOTIVO_SITUACAO_CONREC
              WHERE RSSP.DS_NUMERO_LOTE = PROT.DS_NUMERO_LOTE
                AND NOTA_FISCAL.NR_ID_NOTA_FISCAL = PROT.DS_NR_RT
                AND NOTA_FISCAL.CD_SERIE = PROT.CD_SERIE_RT
                AND NOTA_FISCAL.NR_NOTA_FISCAL_NFE IS NULL
                AND NOTA_FISCAL.VER_NFE <> 'RE'
                AND NOTA_FISCAL.CD_FORMULARIO_NF = F.CD_FORMULARIO_NF
                AND NVL(F.TP_RPS, 'X') = '1'
                AND RSSP.DS_NUMERO_PROTOCOLO IS NOT NULL
                AND NOTA_FISCAL.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
                AND TRUNC(NOTA_FISCAL.DT_EMISSAO) >= TO_DATE(:CAMPO_DT_INICIAL,'DD/MM/YYYY')
                AND TRUNC(NOTA_FISCAL.DT_EMISSAO) <= TO_DATE(:CAMPO_DT_FINAL,'DD/MM/YYYY')
                AND NOTA_FISCAL.CD_CONVENIO = CONVENIO.CD_CONVENIO(+)
                AND NOTA_FISCAL.CD_FORMULARIO_NF = FORMULARIO_NF.CD_FORMULARIO_NF
                AND NOTA_FISCAL.CD_MOT_SITUACAO = MOTIVO_SITUACAO_CONREC.CD_MOT_SITUACAO(+)
            UNION ALL
              SELECT
                  'COM RETORNO' SITUACAO,
                 NOTA_FISCAL.NR_ID_NOTA_FISCAL NRO_RPS,
                  NOTA_FISCAL.NR_NOTA_FISCAL_NFE NR_NFE,
                  Decode ( NOTA_FISCAL.CD_VERIFICACAO_NFE, NULL, NULL,'''' || NOTA_FISCAL.CD_VERIFICACAO_NFE || '''') CD_VERIFICACAO_NFE,
                  NOTA_FISCAL.CD_ATENDIMENTO CD_ATENDIMENTO,
                  NOTA_FISCAL.NM_CLIENTE NM_CLIENTE,
                  NOTA_FISCAL.CD_CONVENIO CD_CONVENIO,
                  CONVENIO.NM_CONVENIO NM_CONVENIO,
                  TO_CHAR(NOTA_FISCAL.DT_EMISSAO,'DD/MM/YYYY') DT_COMPETENCIA_RPS,
                  TO_CHAR(NOTA_FISCAL.HR_EMISSAO_NFE,'DD/MM/YYYY') DT_EMISSAO_NFE,
                  NOTA_FISCAL.VL_TOTAL_NOTA VL_TOTAL_RPS,
                  FORMULARIO_NF.DS_FORMULARIO_NF NM_FORMULARIO_RPS,
                  NOTA_FISCAL.CD_NOTA_FISCAL_ORIGEM RPS_SUBSTITUIDA,
                  Decode (NOTA_FISCAL.CD_STATUS,'E', 'Emitida','C', 'Cancelada',NOTA_FISCAL.CD_STATUS) STATUS_RPS,
                  MOTIVO_SITUACAO_CONREC.DS_MOT_SITUACAO,
                  TO_CHAR(NOTA_FISCAL.DT_CANCELAMENTO,'DD/MM/YYYY HH24') DT_CANCELAMENTO,
                  NOTA_FISCAL.CD_NOTA_FISCAL CD_NF,
                  NOTA_FISCAL.CD_FORMULARIO_NF
                FROM
                      DBAMV.NOTA_FISCAL,
                      DBAMV.CONVENIO,
                      DBAMV.FORMULARIO_NF,
                      DBAMV.NOTA_FISCAL NF_SUBST,
                      DBAMV.MOTIVO_SITUACAO_CONREC
                WHERE NOTA_FISCAL.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
                  AND NOTA_FISCAL.NR_NOTA_FISCAL_NFE IS NOT NULL
                  AND NOTA_FISCAL.HR_EMISSAO_NFE IS NOT NULL
                  AND TRUNC(NOTA_FISCAL.DT_EMISSAO) >= TO_DATE(:CAMPO_DT_INICIAL,'DD/MM/YYYY')
                  AND TRUNC(NOTA_FISCAL.DT_EMISSAO) <= TO_DATE(:CAMPO_DT_FINAL,'DD/MM/YYYY')
                  AND NOTA_FISCAL.CD_CONVENIO = CONVENIO.CD_CONVENIO(+)
                  AND NOTA_FISCAL.CD_FORMULARIO_NF = FORMULARIO_NF.CD_FORMULARIO_NF
                  AND NOTA_FISCAL.CD_NOTA_FISCAL_ORIGEM = NF_SUBST.CD_NOTA_FISCAL(+)
                  AND NOTA_FISCAL.CD_MOT_SITUACAO = MOTIVO_SITUACAO_CONREC.CD_MOT_SITUACAO(+)
                  :CAMPO_CF_LISTA_RPS
          )
ORDER BY NM_FORMULARIO_RPS, SITUACAO DESC, TO_DATE(DT_COMPETENCIA_RPS,'DD/MM/YYYY'), NRO_RPS)
ORDER BY NM_FORMULARIO_RPS,
SITUACAO
