--R_EST_PREST_DT_LANC
SELECT CD_PRESTADOR,
               NM_PRESTADOR,
              SUM(NVL( VALOR_PAGAR,0))  VALOR_PAGAR
FROM
(
SELECT PRESTADOR.CD_PRESTADOR
              ,PRESTADOR.NM_PRESTADOR
              ,DECODE( :CAMPO_ACRES_DESC,
              'Sim', (ITLAN_MED.VL_ATO * ITREG_FAT.QT_LANCAMENTO)  -
             ( NVL(ITREG_FAT.VL_DESCONTO,0)   * ( ATI_MED.VL_PERCENTUAL_PAGO / 100 ) ) +
             ( NVL(ITREG_FAT.VL_ACRESCIMO,0)  * ( ATI_MED.VL_PERCENTUAL_PAGO / 100 ) ) +
             ( NVL(ITREG_FAT.VL_FILME_UNITARIO,0) * ITREG_FAT.QT_LANCAMENTO *
             ( ATI_MED.VL_PERCENTUAL_PAGO / 100 ) )
              ,'Não', (ITLAN_MED.VL_ATO * ITREG_FAT.QT_LANCAMENTO) +
               ( NVL(ITREG_FAT.VL_FILME_UNITARIO,0) * ITREG_FAT.QT_LANCAMENTO *
               ( ATI_MED.VL_PERCENTUAL_PAGO / 100 ) ) )                  VALOR_PAGAR
FROM   DBAMV.ATENDIME
            ,DBAMV.REG_FAT
            ,DBAMV.CONVENIO
            ,DBAMV.ITREG_FAT
            ,DBAMV.PRO_FAT
            ,DBAMV.GRU_PRO
            ,DBAMV.ITLAN_MED
            ,DBAMV.PRESTADOR
            ,DBAMV.ATI_MED
	    ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And ATENDIME.CD_ATENDIMENTO = REG_FAT.CD_ATENDIMENTO
AND        REG_FAT.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND        REG_FAT.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
AND        ITREG_FAT.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
AND        PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
AND        ITREG_FAT.CD_REG_FAT = ITLAN_MED.CD_REG_FAT
AND        ITREG_FAT.CD_LANCAMENTO = ITLAN_MED.CD_LANCAMENTO
AND        ITLAN_MED.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
AND        ITLAN_MED.CD_ATI_MED = ATI_MED.CD_ATI_MED
AND        ITREG_FAT.SN_PERTENCE_PACOTE = 'N'
AND        NVL(REG_FAT.SN_DIAGNO,'N') = 'N'
AND        ITREG_FAT.DT_LANCAMENTO  BETWEEN TO_DATE(:CAMPO_DT_INICIO, 'DD/MM/YYYY')  AND  (TO_DATE(:CAMPO_DT_FINAL, 'DD/MM/YYYY') + 0.9)
AND          ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND        CONVENIO.TP_CONVENIO <> 'H'
UNION ALL
SELECT PRESTADOR.CD_PRESTADOR
              ,PRESTADOR.NM_PRESTADOR
              ,DECODE( :CAMPO_ACRES_DESC, 'Sim', (ITREG_FAT.VL_TOTAL_CONTA -
               NVL(ITREG_FAT.VL_DESCONTO,0) +  NVL(ITREG_FAT.VL_ACRESCIMO,0)), 'Não',  ITREG_FAT.VL_TOTAL_CONTA)                       VALOR_PAGAR
FROM   DBAMV.ATENDIME
            ,DBAMV.REG_FAT
            ,DBAMV.CONVENIO
            ,DBAMV.ITREG_FAT
            ,DBAMV.PRO_FAT
            ,DBAMV.GRU_PRO
            ,DBAMV.PRESTADOR
	    ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And ATENDIME.CD_ATENDIMENTO = REG_FAT.CD_ATENDIMENTO
AND       ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND        REG_FAT.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND        REG_FAT.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
AND        ITREG_FAT.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
AND        PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
AND        ITREG_FAT.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
AND        NVL(REG_FAT.SN_DIAGNO,'N') = 'N'
AND        ITREG_FAT.SN_PERTENCE_PACOTE = 'N'
AND        ITREG_FAT.DT_LANCAMENTO  BETWEEN TO_DATE(:CAMPO_DT_INICIO, 'DD/MM/YYYY')  AND (TO_DATE( :CAMPO_DT_FINAL, 'DD/MM/YYYY') + 0.9)
AND        CONVENIO.TP_CONVENIO <> 'H'
UNION ALL
SELECT PRESTADOR.CD_PRESTADOR
              ,PRESTADOR.NM_PRESTADOR
              ,DECODE( :CAMPO_ACRES_DESC, 'Sim', (ITREG_AMB.VL_TOTAL_CONTA -
               NVL(ITREG_AMB.VL_DESCONTO,0) +  NVL(ITREG_AMB.VL_ACRESCIMO,0)), 'Não',
               ITREG_AMB.VL_TOTAL_CONTA)                               VALOR_PAGAR
FROM   DBAMV.ATENDIME
            ,DBAMV.ITREG_AMB
            ,DBAMV.REG_AMB
            ,DBAMV.CONVENIO
            ,DBAMV.PRO_FAT
            ,DBAMV.GRU_PRO
            ,DBAMV.PRESTADOR
	    ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And ATENDIME.CD_ATENDIMENTO = ITREG_AMB.CD_ATENDIMENTO
AND        ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND        ITREG_AMB.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND        ITREG_AMB.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
AND        PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
AND        ITREG_AMB.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
AND        ITREG_AMB.CD_REG_AMB = REG_AMB.CD_REG_AMB
AND        NVL(REG_AMB.SN_DIAGNO,'N') = 'N'
AND        ITREG_AMB.SN_PERTENCE_PACOTE = 'N'
AND        ATENDIME.DT_ATENDIMENTO BETWEEN TO_DATE(:CAMPO_DT_INICIO, 'DD/MM/YYYY')   AND  (TO_DATE(:CAMPO_DT_FINAL, 'DD/MM/YYYY') + 0.9)
AND        REG_AMB.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND        CONVENIO.TP_CONVENIO <> 'H'
:CAMPO_CF_COND_CREDEN_AMB
)
GROUP BY CD_PRESTADOR,
                    NM_PRESTADOR
ORDER BY VALOR_PAGAR DESC
