--R_TABFAT_PORT
SELECT * FROM (
SELECT  TAB_FAT.CD_TAB_FAT CD_TAB_FAT
               ,TAB_FAT.DS_TAB_FAT DS_TAB_FAT
               ,VAL_PORTE.DT_VIGENCIA DT_VIGENCIA
               ,TO_CHAR( VAL_PORTE.DT_VIGENCIA, 'YYYY' )  ANO
               ,VAL_PORTE.VL_PORTE VL_PORTE
               ,VAL_PORTE.CD_POR_ANE CD_POR_ANE
              ,POR_ANE.CD_POR_ANE CD_POR_ANE1
              ,POR_ANE.DS_POR_ANE DS_POR_ANE
FROM    dbamv.TAB_FAT TAB_FAT
             ,dbamv.VAL_PORTE VAL_PORTE
             ,dbamv.POR_ANE POR_ANE
WHERE   TAB_FAT.CD_TAB_FAT LIKE :CAMPO_TAB_FAT
AND         VAL_PORTE.CD_TAB_FAT = TAB_FAT.CD_TAB_FAT
AND         POR_ANE.CD_POR_ANE = VAL_PORTE.CD_POR_ANE
ORDER BY TAB_FAT.DS_TAB_FAT
                   ,VAL_PORTE.DT_VIGENCIA
                   ,VAL_PORTE.CD_POR_ANE
)
ORDER BY DS_TAB_FAT,CD_TAB_FAT,
ANO
