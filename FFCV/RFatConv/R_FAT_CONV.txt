--R_FAT_CONV
SELECT * FROM (
SELECT RESUMO.CD_CONVENIO CD_CONVENIO
     , RESUMO.NM_CONVENIO NM_CONVENIO
     , null CD_CON_PLA
     , null DS_CON_PLA
     , SUM(VL_INTERNACAO)
     , SUM(VL_AMBULATORIAL)
     , SUM(VL_EXTERNO)
     , SUM(VL_URGENCIA)
     , SUM(QT_INTERNACAO)
     , SUM(QT_AMBULATORIAL)
     , SUM(QT_EXTERNO)
     , SUM(QT_URGENCIA)
     , SUM(VL_TOTAL_CONTA) VALOR
     , COUNT(VL_TOTAL_CONTA) QTD
     , SUM(VL_REPASSE) VL_REPASSE
  FROM
       (SELECT ATENDIME.CD_ATENDIMENTO
             , CONVENIO.CD_CONVENIO
             , CONVENIO.NM_CONVENIO
             , REG_FAT.CD_REG_FAT
             , SUM( DECODE(ATENDIME.TP_ATENDIMENTO, 'I', NVL(NVL(ITLAN_MED.VL_LIQUIDO, ITREG_FAT.VL_TOTAL_CONTA),0),  'H', NVL(NVL(ITLAN_MED.VL_LIQUIDO, ITREG_FAT.VL_TOTAL_CONTA),0), NULL)) VL_INTERNACAO
             , SUM( DECODE(ATENDIME.TP_ATENDIMENTO, 'A', NVL(ITREG_FAT.VL_TOTAL_CONTA,0), NULL)) VL_AMBULATORIAL
             , SUM( DECODE(ATENDIME.TP_ATENDIMENTO, 'E', NVL(ITREG_FAT.VL_TOTAL_CONTA,0), NULL)) VL_EXTERNO
             , SUM( DECODE(ATENDIME.TP_ATENDIMENTO, 'U', NVL(ITREG_FAT.VL_TOTAL_CONTA,0), NULL)) VL_URGENCIA
             , DECODE(ATENDIME.TP_ATENDIMENTO, 'I', 1,'H',1,0) QT_INTERNACAO
             , DECODE(ATENDIME.TP_ATENDIMENTO, 'A', 1,0) QT_AMBULATORIAL
             , DECODE(ATENDIME.TP_ATENDIMENTO, 'E', 1,0) QT_EXTERNO
             , DECODE(ATENDIME.TP_ATENDIMENTO, 'U', 1,0) QT_URGENCIA
             , SUM( NVL(NVL(ITLAN_MED.VL_LIQUIDO, ITREG_FAT.VL_TOTAL_CONTA),0))              VL_TOTAL_CONTA
             , REG_FAT.CD_CON_PLA
             , SUM(DECODE(GRU_FAT.TP_GRU_FAT,'SP',DECODE(NVL(ITLAN_MED.TP_PAGAMENTO, NVL(ITREG_FAT.TP_PAGAMENTO,'X')),'P',NVL(ITLAN_MED.VL_LIQUIDO, ITREG_FAT.VL_TOTAL_CONTA),0),0)) VL_REPASSE
          FROM DBAMV.FATURA
             , DBAMV.REMESSA_FATURA REMESSA
             , DBAMV.REG_FAT
             , DBAMV.ITREG_FAT
             , DBAMV.CONVENIO
             , DBAMV.ATENDIME
             , DBAMV.SERVICO
             , DBAMV.GRU_FAT
             , DBAMV.ITLAN_MED
             , dbamv.Empresa_Convenio
         WHERE FATURA.CD_FATURA = REMESSA.CD_FATURA
           AND REG_FAT.CD_REMESSA = REMESSA.CD_REMESSA
           AND REG_FAT.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
           And Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
           And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
           AND REMESSA.SN_FECHADA = 'S'
           AND REG_FAT.CD_CONVENIO    = CONVENIO.CD_CONVENIO
           AND REG_FAT.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
           AND ITREG_FAT.CD_GRU_FAT = GRU_FAT.CD_GRU_FAT
           AND ITREG_FAT.CD_REG_FAT = ITLAN_MED.CD_REG_FAT(+)
           AND ITREG_FAT.CD_LANCAMENTO = ITLAN_MED.CD_LANCAMENTO(+)
           AND NVL(ITLAN_MED.TP_PAGAMENTO, nvl(ITREG_FAT.TP_PAGAMENTO,'X')) <> 'C'
           AND ATENDIME.CD_SERVICO = SERVICO.CD_SERVICO(+)
           AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
           AND CONVENIO.TP_CONVENIO IN ('C','P')
         GROUP BY ATENDIME.CD_ATENDIMENTO
             , CONVENIO.CD_CONVENIO
             , CONVENIO.NM_CONVENIO
             , REG_FAT.CD_REG_FAT
             , DECODE(ATENDIME.TP_ATENDIMENTO, 'I', 1,'H',1,0)
             , DECODE(ATENDIME.TP_ATENDIMENTO, 'A', 1,0)
             , DECODE(ATENDIME.TP_ATENDIMENTO, 'E', 1,0)
             , DECODE(ATENDIME.TP_ATENDIMENTO, 'U', 1,0)
             , REG_FAT.CD_CON_PLA
         UNION ALL
        SELECT ATENDIME.CD_ATENDIMENTO
             , CONVENIO.CD_CONVENIO
             , CONVENIO.NM_CONVENIO
             , TO_NUMBER(NULL) CD_REG_AMB
             , SUM( DECODE(ATENDIME.TP_ATENDIMENTO, 'I', NVL(ITREG_AMB.VL_TOTAL_CONTA,0), 'H', NVL(ITREG_AMB.VL_TOTAL_CONTA,0), NULL)) VL_INTERNACAO
             , SUM( DECODE(ATENDIME.TP_ATENDIMENTO, 'A', NVL(ITREG_AMB.VL_TOTAL_CONTA,0), NULL)) VL_AMBULATORIAL
             , SUM( DECODE(ATENDIME.TP_ATENDIMENTO, 'E', NVL(ITREG_AMB.VL_TOTAL_CONTA,0), NULL)) VL_EXTERNO
             , SUM( DECODE(ATENDIME.TP_ATENDIMENTO, 'U', NVL(ITREG_AMB.VL_TOTAL_CONTA,0), NULL)) VL_URGENCIA
             , DECODE(ATENDIME.TP_ATENDIMENTO, 'I', 1,'H',1,0) QT_INTERNACAO
             , DECODE(ATENDIME.TP_ATENDIMENTO, 'A', 1,0) QT_AMBULATORIAL
             , DECODE(ATENDIME.TP_ATENDIMENTO, 'E', 1,0) QT_EXTERNO
             , DECODE(ATENDIME.TP_ATENDIMENTO, 'U', 1,0) QT_URGENCIA
             , SUM( NVL(ITREG_AMB.VL_TOTAL_CONTA,0) )             VL_TOTAL_CONTA
             , ITREG_AMB.CD_CON_PLA
             , SUM(DECODE(GRU_FAT.TP_GRU_FAT,'SP',DECODE(NVL(ITREG_AMB.TP_PAGAMENTO,'P'),'P',ITREG_AMB.VL_TOTAL_CONTA,0),0)) VL_REPASSE
          FROM DBAMV.FATURA
             , DBAMV.REMESSA_FATURA REMESSA
             , DBAMV.REG_AMB
             , DBAMV.ITREG_AMB
             , DBAMV.CONVENIO
             , DBAMV.ATENDIME
             , DBAMV.SERVICO
             , DBAMV.GRU_FAT
             , Dbamv.Empresa_Convenio
         WHERE FATURA.CD_FATURA = REMESSA.CD_FATURA
           AND REG_AMB.CD_REMESSA = REMESSA.CD_REMESSA
           AND REG_AMB.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
           AND REG_AMB.CD_MULTI_EMPRESA = ATENDIME.CD_MULTI_EMPRESA
           and Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
           And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
           AND REMESSA.SN_FECHADA = 'S'
           and atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
           AND REG_AMB.CD_REG_AMB = ITREG_AMB.CD_REG_AMB
           AND ITREG_AMB.CD_GRU_FAT = GRU_FAT.CD_GRU_FAT
           AND ITREG_AMB.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
           AND ITREG_AMB.CD_CONVENIO    = CONVENIO.CD_CONVENIO
           AND REG_AMB.CD_CONVENIO = CONVENIO.CD_CONVENIO
           AND NVL(ITREG_AMB.TP_PAGAMENTO,'X') <> 'C'
           AND CONVENIO.TP_CONVENIO IN ('C','P')
           AND ATENDIME.CD_SERVICO = SERVICO.CD_SERVICO(+)
         GROUP BY ATENDIME.CD_ATENDIMENTO
             , CONVENIO.CD_CONVENIO
             , CONVENIO.NM_CONVENIO
             , DECODE(ATENDIME.TP_ATENDIMENTO, 'I', 1,'H',1,0)
             , DECODE(ATENDIME.TP_ATENDIMENTO, 'A', 1,0)
             , DECODE(ATENDIME.TP_ATENDIMENTO, 'E', 1,0)
             , DECODE(ATENDIME.TP_ATENDIMENTO, 'U', 1,0)
             , ITREG_AMB.CD_CON_PLA
      ) RESUMO , DBAMV.CON_PLA
WHERE RESUMO.CD_CON_PLA = CON_PLA.CD_CON_PLA
  AND RESUMO.CD_CONVENIO = CON_PLA.CD_CONVENIO
  GROUP BY  RESUMO.CD_CONVENIO
          , RESUMO.NM_CONVENIO
ORDER BY RESUMO.CD_CONVENIO, CON_PLA.CD_CON_PLA
)
ORDER BY NM_CONVENIO asc

             , DBAMV.REMESSA_FATURA    REMESSA
             , DBAMV.REG_AMB
             , DBAMV.ITREG_AMB
             , DBAMV.CONVENIO
             , DBAMV.ATENDIME
             , DBAMV.SERVICO
             , DBAMV.GRU_FAT
             , DBAMV.PRO_FAT
         WHERE FATURA.CD_FATURA = REMESSA.CD_FATURA
           AND REG_AMB.CD_REMESSA = REMESSA.CD_REMESSA
           AND REG_AMB.CD_REG_AMB = ITREG_AMB.CD_REG_AMB
           AND REG_AMB.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
           AND REG_AMB.CD_MULTI_EMPRESA = ATENDIME.CD_MULTI_EMPRESA
           AND ITREG_AMB.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
           AND ITREG_AMB.CD_GRU_FAT = GRU_FAT.CD_GRU_FAT
           AND ITREG_AMB.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
           AND ITREG_AMB.CD_CONVENIO    = CONVENIO.CD_CONVENIO
           AND REG_AMB.CD_CONVENIO = CONVENIO.CD_CONVENIO
           AND NVL(ITREG_AMB.TP_PAGAMENTO,'X') <> 'C'
           AND ATENDIME.CD_SERVICO = SERVICO.CD_SERVICO(+)
           AND CONVENIO.TP_CONVENIO NOT IN ('H','A')
           AND REMESSA.SN_FECHADA = 'S'
         GROUP BY ATENDIME.CD_ATENDIMENTO
             , CONVENIO.CD_CONVENIO
             , CONVENIO.NM_CONVENIO
             , DECODE(ATENDIME.TP_ATENDIMENTO, 'I', 1,'H',1,0)
             , DECODE(ATENDIME.TP_ATENDIMENTO, 'A', 1,0)
             , DECODE(ATENDIME.TP_ATENDIMENTO, 'E', 1,0)
             , DECODE(ATENDIME.TP_ATENDIMENTO, 'U', 1,0)
             , ITREG_AMB.CD_CON_PLA
      ) RESUMO , DBAMV.CON_PLA, DBAMV.EMPRESA_CONVENIO EC
WHERE RESUMO.CD_CON_PLA = CON_PLA.CD_CON_PLA
  AND RESUMO.CD_CONVENIO = CON_PLA.CD_CONVENIO
  AND EC.CD_CONVENIO = RESUMO.CD_CONVENIO
  AND EC.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
  GROUP BY  RESUMO.CD_CONVENIO
          , RESUMO.NM_CONVENIO
		  , CASE WHEN '' = 'S' THEN CON_PLA.CD_CON_PLA END
          , CASE WHEN '' = 'S' THEN CON_PLA.DS_CON_PLA END
ORDER BY RESUMO.CD_CONVENIO, CON_PLA.CD_CON_PLA
)
