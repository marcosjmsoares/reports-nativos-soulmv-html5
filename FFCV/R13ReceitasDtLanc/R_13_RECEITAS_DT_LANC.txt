--R_13_RECEITAS_DT_LANC
SELECT * FROM (
SELECT  NM_CONVENIO
       , to_date(MES, 'MM') MES
       ,DS_CON_PLA
       ,SUM(VL_TOTAL_CONTA)  VALOR
FROM
(SELECT  CONVENIO.CD_CONVENIO          CD_CONVENIO
       ,CONVENIO.NM_CONVENIO                  NM_CONVENIO
       ,TO_CHAR(ITREG_FAT.DT_LANCAMENTO, 'MM')  MES
       ,NVL(ITREG_FAT.VL_TOTAL_CONTA,0)              VL_TOTAL_CONTA
       ,CON_PLA.CD_CON_PLA
       ,CON_PLA.DS_CON_PLA
FROM    DBAMV.REG_FAT
       ,DBAMV.ITREG_FAT
       ,DBAMV.CONVENIO
       ,DBAMV.CON_PLA
       ,DBAMV.ATENDIME
       ,DBAMV.SERVICO
WHERE :CAMPO_FATU_CONSU = 'F'
AND   REG_FAT.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
AND   ITREG_FAT.SN_PERTENCE_PACOTE = 'N'
AND   NVL(ITREG_FAT.TP_PAGAMENTO,'X') <> 'C'
AND   NVL(REG_FAT.SN_DIAGNO, 'N') = 'N'
AND   REG_FAT.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
AND   REG_FAT.CD_CONVENIO    = CONVENIO.CD_CONVENIO
AND   REG_FAT.CD_CONVENIO    = CON_PLA.CD_CONVENIO
AND   REG_FAT.CD_CON_PLA     = CON_PLA.CD_CON_PLA
AND   ITREG_FAT.DT_LANCAMENTO  BETWEEN  to_date(:CAMPO_DT_INICIO, 'dd/mm/yyyy')   AND  (to_date(:CAMPO_DT_FINAL, 'dd/mm/yyyy') + 0.9)
AND   CONVENIO.TP_CONVENIO <> 'H'
and   atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
AND   ATENDIME.CD_SERVICO = SERVICO.CD_SERVICO(+)
UNION ALL
SELECT  CONVENIO.CD_CONVENIO                  CD_CONVENIO
       ,CONVENIO.NM_CONVENIO                  NM_CONVENIO
       ,TO_CHAR(ATENDIME.DT_ATENDIMENTO, 'MM')  MES
       ,NVL(ITREG_AMB.VL_TOTAL_CONTA,0)              VL_TOTAL_CONTA
       ,CON_PLA.CD_CON_PLA
       ,CON_PLA.DS_CON_PLA
FROM   DBAMV.REG_AMB
      ,DBAMV.ITREG_AMB
      ,DBAMV.CONVENIO
      ,DBAMV.CON_PLA
      ,DBAMV.ATENDIME
      ,DBAMV.SERVICO
WHERE  :CAMPO_FATU_CONSU = 'F'
AND   REG_AMB.CD_REG_AMB = ITREG_AMB.CD_REG_AMB
AND   ITREG_AMB.SN_PERTENCE_PACOTE = 'N'
AND   NVL(ITREG_AMB.TP_PAGAMENTO,'X') <> 'C'
AND   NVL(REG_AMB.SN_DIAGNO, 'N') = 'N'
AND   ITREG_AMB.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
AND   ITREG_AMB.CD_CONVENIO    = CONVENIO.CD_CONVENIO
AND   ITREG_AMB.CD_CONVENIO    = CON_PLA.CD_CONVENIO
AND   ITREG_AMB.CD_CON_PLA     = CON_PLA.CD_CON_PLA
AND        ATENDIME.DT_ATENDIMENTO BETWEEN  to_date(:CAMPO_DT_INICIO, 'dd/mm/yyyy')   AND  (to_date(:CAMPO_DT_FINAL, 'dd/mm/yyyy') + 0.9)
AND   REG_AMB.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND   CONVENIO.TP_CONVENIO <> 'H'
and   atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
AND   ATENDIME.CD_SERVICO = SERVICO.CD_SERVICO(+)
UNION ALL
SELECT  CONVENIO.CD_CONVENIO                  CD_CONVENIO
       ,CONVENIO.NM_CONVENIO                  NM_CONVENIO
       ,TO_CHAR(ITREG_FAT.DT_LANCAMENTO, 'MM')  MES
       ,NVL(ITREG_FAT.VL_TOTAL_CONTA,0)              VL_TOTAL_CONTA
       ,CON_PLA.CD_CON_PLA
       ,CON_PLA.DS_CON_PLA
FROM  DBAMV.REG_FAT
       ,DBAMV.ITREG_FAT
       ,DBAMV.CONVENIO
       ,DBAMV.CON_PLA
       ,DBAMV.ATENDIME
       ,DBAMV.PRO_FAT
       ,DBAMV.SERVICO
WHERE :CAMPO_FATU_CONSU = 'C'
AND   REG_FAT.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
AND   ITREG_FAT.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
AND   PRO_FAT.SN_PACOTE = 'N'
AND   NVL(REG_FAT.SN_DIAGNO, 'N') = 'N'
AND   REG_FAT.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
AND   REG_FAT.CD_CONVENIO    = CONVENIO.CD_CONVENIO
AND   REG_FAT.CD_CONVENIO    = CON_PLA.CD_CONVENIO
AND   REG_FAT.CD_CON_PLA     = CON_PLA.CD_CON_PLA
 AND        ITREG_FAT.DT_LANCAMENTO  BETWEEN  to_date(:CAMPO_DT_INICIO, 'dd/mm/yyyy')   AND  (to_date(:CAMPO_DT_FINAL, 'dd/mm/yyyy') + 0.9)
AND   CONVENIO.TP_CONVENIO <> 'H'
and   atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
AND   ATENDIME.CD_SERVICO = SERVICO.CD_SERVICO(+)
UNION ALL
SELECT  CONVENIO.CD_CONVENIO                  CD_CONVENIO
       ,CONVENIO.NM_CONVENIO                  NM_CONVENIO
       ,TO_CHAR(ATENDIME.DT_ATENDIMENTO, 'MM')  MES
       ,NVL(ITREG_AMB.VL_TOTAL_CONTA,0)              VL_TOTAL_CONTA
       ,CON_PLA.CD_CON_PLA
       ,CON_PLA.DS_CON_PLA
FROM   DBAMV.REG_AMB
      ,DBAMV.ITREG_AMB
      ,DBAMV.PRO_FAT
      ,DBAMV.CONVENIO
      ,DBAMV.CON_PLA
      ,DBAMV.ATENDIME
      ,DBAMV.SERVICO
WHERE  :CAMPO_FATU_CONSU = 'C'
AND   REG_AMB.CD_REG_AMB = ITREG_AMB.CD_REG_AMB
AND   ITREG_AMB.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
AND   PRO_FAT.SN_PACOTE = 'N'
AND   NVL(REG_AMB.SN_DIAGNO, 'N') = 'N'
AND   ITREG_AMB.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
AND   ITREG_AMB.CD_CONVENIO    = CONVENIO.CD_CONVENIO
AND   ITREG_AMB.CD_CONVENIO    = CON_PLA.CD_CONVENIO
AND   ITREG_AMB.CD_CON_PLA     = CON_PLA.CD_CON_PLA
AND        ATENDIME.DT_ATENDIMENTO BETWEEN  to_date(:CAMPO_DT_INICIO, 'dd/mm/yyyy')   AND  (to_date(:CAMPO_DT_FINAL, 'dd/mm/yyyy') + 0.9)
AND   REG_AMB.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND   CONVENIO.TP_CONVENIO <> 'H'
and   atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
AND   ATENDIME.CD_SERVICO = SERVICO.CD_SERVICO(+)
:CAMPO_CF_CONTA_FECHADA_AMB
)
GROUP BY NM_CONVENIO
         ,MES
         ,DS_CON_PLA
ORDER BY VALOR DESC
)
ORDER BY NM_CONVENIO,
DS_CON_PLA,
MES
