--R_EST_CONV_AMB
SELECT * FROM (
Select sum( itreg_amb.vl_total_conta ) vl_total_conta
      ,ori_ate.cd_ori_ate
      ,ori_ate.ds_ori_ate
      ,convenio.cd_convenio
      ,convenio.nm_convenio
      ,count( distinct itreg_amb.cd_atendimento ) qtd_atendime
  From dbamv.convenio
      ,dbamv.ori_ate
      ,dbamv.itreg_amb
      ,dbamv.reg_amb
      ,dbamv.atendime
      ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
And reg_amb.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
      And to_char( reg_amb.dt_lancamento, 'mm/yyyy') = :CAMPO_DT_COMPET
      And  itreg_amb.cd_reg_amb = reg_amb.cd_reg_amb
      And atendime.cd_atendimento = itreg_amb.cd_atendimento
      And convenio.cd_convenio = atendime.cd_convenio
      And ori_ate.cd_ori_ate = atendime.cd_ori_ate
Group By
       ori_ate.cd_ori_ate
      ,ori_ate.ds_ori_ate
      ,convenio.cd_convenio
      ,convenio.nm_convenio
)
ORDER BY CD_ORI_ATE,DS_ORI_ATE
