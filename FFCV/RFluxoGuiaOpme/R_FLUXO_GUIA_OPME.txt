--R_FLUXO_GUIA_OPME
SELECT  guia.cd_atendimento, guia.cd_res_lei, guia.cd_aviso_cirurgia, guia.cd_movimento_agenda_central,
 	guia.nr_guia, guia.cd_senha, guia.tp_situacao, Decode(guia.tp_situacao,'A','Autorizada','P',		'Pendente','S','Solicitada','N','Negada','G','Em Negociação','C','Cancelada','R','Pré-Autorizada' 	)status_situacao,guia.cd_convenio || ' - ' || c.nm_convenio as ds_convenio, guia.dt_geracao,
      	p.nm_paciente, guia.tp_status_opme, Decode(guia.tp_status_opme,'AR','A Revisar','EC',		'Em Cotação','DR','Devolvido p/ Revisão','PR','Pré OC.','RN','Renegociar','CA','Cancelada',		'RE','Revisada','CO','Cotada','AU','Autorizada') status_opme
 FROM   dbamv.guia , dbamv.convenio c, dbamv.paciente p
WHERE   guia.cd_convenio = c.cd_convenio
  AND   guia.cd_paciente = p.cd_paciente
  AND   guia.tp_guia = 'O'
 ORDER BY guia.dt_geracao
