--R_AGENDA_REMESSA
Select agenda_remessa.cd_convenio,
           convenio.nm_convenio,
           agenda_remessa.cd_agrupamento,
           agrupamento.ds_agrupamento,
           agenda_remessa.dt_abertura,
           agenda_remessa.dt_fechamento
  From dbamv.agenda_remessa,
           dbamv.convenio,
           dbamv.agrupamento,
           dbamv.Empresa_Convenio
 Where convenio.cd_convenio                  = agenda_remessa.cd_convenio
     and agrupamento.cd_agrupamento (+)  = agenda_remessa.cd_agrupamento
     and agenda_remessa.dt_fechamento >=  :CAMPO_FECH_INI
     and agenda_remessa.dt_fechamento <=  :CAMPO_FECH_FIM
     And Agenda_Remessa.Cd_Convenio = Empresa_Convenio.Cd_Convenio
     And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
union
Select to_number(null) cd_convenio,
       null nm_convenio,
       to_number(null) cd_agrupamento,
       null ds_agrupamento,
       to_date(null) dt_abertura,
       to_date(null) dt_fechamento
  From dual
Order by NM_CONVENIO,CD_CONVENIO, 2, 4, 6
