--R_LOG_MOTIV_PEND
SELECT g.nr_guia, g.cd_atendimento, g.cd_aviso_cirurgia, g.cd_res_lei,
       Decode(g.tp_guia, 'I', 'Internação', 'C', 'Consulta', 'R', 'Prorrogação',
                         'P', 'Procedimento', 'O', 'OPME', 'Q', 'Quimioterapia',
                         'D', 'Radioterapia', 'S', 'SADT', 'M', 'Medicamento', 'I', 'Materiais') tp_guia,
       g.tp_situacao, l.cd_log_mot_pend, l.dt_movimento, l.cd_usu_movimento,
       Decode(l.tp_movimento, 'I', 'Inclusão', 'U', 'Alteração', 'D', 'Exclusão' ) tp_movimento,
       l.cd_motiv_pend_ant, l.cd_motiv_pend_atu,
       m1.cd_motivo_pendencia_guia cod_aterior, m1.ds_motivo_pendencia_guia ds_aterior,
       m2.cd_motivo_pendencia_guia cod_atual, m2.ds_motivo_pendencia_guia ds_atual
FROM dbamv.guia g,
     dbamv.log_mot_pend l,
     dbamv.motivo_pendencia_guia m1,
     dbamv.motivo_pendencia_guia m2
WHERE g.cd_guia = :CAMPO_CD_GUIA
   AND l.cd_guia = g.cd_guia
   AND l.cd_motiv_pend_ant = m1.cd_motivo_pendencia_guia(+)
   AND l.cd_motiv_pend_atu = m2.cd_motivo_pendencia_guia(+)
ORDER BY g.cd_guia, l.cd_log_mot_pend
