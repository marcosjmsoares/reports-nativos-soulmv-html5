--R_NOTAS_FORMULARIO
SELECT N.NR_CGC_CPF CNPJ_CPF,
	   CASE WHEN Length(REPLACE(REPLACE(N.NR_CGC_CPF,'-',''),'.','')) <= 11 AND Length(REPLACE(REPLACE(N.NR_CGC_CPF,'-',''),'.','')) > 0  THEN '2'
			WHEN Length(REPLACE(REPLACE(N.NR_CGC_CPF,'-',''),'.','')) > 11 THEN '1'
			ELSE '3' END   TIPO_RNC,
	   N.NM_CLIENTE,
	   N.CD_FORMULARIO_NF,
	   F.DS_FORMULARIO_NF,
	   N.NR_ID_NOTA_FISCAL NR_NOTA_FISCAL_NFE,
	   SUBSTR(N.DS_OBSERVACOES,INSTR(N.DS_OBSERVACOES,'/',1)+1, INSTR(N.DS_OBSERVACOES,'/',1,2) - (INSTR(N.DS_OBSERVACOES,'/',1)+1))  NF_MODIFICADA,
	   DETALHE.VL_TRIBUTO VLR_ITBIS_FATURADO,
	   TRUNC(N.DT_EMISSAO) DT_EMISSAO,
	   N.VL_TOTAL_NOTA VL_TOTAL_NOTA,
	   N.CD_NOTA_FISCAL
FROM DBAMV.NOTA_FISCAL N, DBAMV.FORMULARIO_NF F,( SELECT CD_NOTA_FISCAL, VL_TRIBUTO FROM DBAMV.NOTA_FISCAL_TRIBUTO :CAMPO_CF_DETALHAMENTO ) DETALHE
WHERE N.CD_FORMULARIO_NF = F.CD_FORMULARIO_NF
AND N.CD_STATUS NOT IN ('C')
AND N.CD_NOTA_FISCAL = DETALHE.CD_NOTA_FISCAL(+)
ORDER BY N.CD_FORMULARIO_NF, N.NR_ID_NOTA_FISCAL
