--R_DEM_TISS_CTA_R_DEM_CTA_Q_DEM_CTA
SELECT * FROM (
select cta.id id_cta
       , cta.id_pai id_pai_cta
       , cta.nr_registro_ans
       , cta.ds_operadora
       , cta.cd_cgc_operadora
       , cta.nr_demonstrativo
       , cta.dt_emissao
       , cta.cd_operadora_prestador
       , cta.cd_cgc_prestador
       , cta.cd_cpf_prestador
       , cta.nm_prestador
       , cta.cd_cnes
       , cta.nr_fatura
       , cta.vl_processado_fatura
       , cta.vl_liberado_fatura
       , cta.vl_glosa_fatura
       , cta.vl_processado
       , cta.vl_liberado
       , cta.vl_glosa
       , lte.id id_lte
       , lte.id_pai id_pai_lte
       , lte.nr_lote
       , lte.dt_envio_lote
       , lte.nr_protocolo
       , lte.vl_protocolo
       , lte.vl_glosa vl_glosa_lte
       , guia.id id_guia
       , guia.id_pai id_pai_guia
       , guia.nr_guia
       , guia.nr_guia_operadora
       , guia.cd_guia
       , guia.nm_paciente
       , guia.nr_beneficiario
       , guia.nr_carteira
       , guia.dt_realizacao
       , guia.vl_processado_guia
       , guia.vl_liberado_guia
       , guia.vl_glosa_guia
from dbamv.tiss_retorno_demon_conta cta
     , dbamv.tiss_retorno_demon_conta_lote lte
     , dbamv.tiss_retorno_demon_conta_guia guia
     , dbamv.tiss_mensagem tm
where  tm.id_mensagem_envio = :CAMPO_ID_TISS_MSG
   and cta.id_pai = tm.id
   and lte.id_pai = cta.id
   and guia.id_pai = lte.id
)
ORDER BY ID_CTA,ID_PAI_CTA,NR_REGISTRO_ANS,DS_OPERADORA,CD_CGC_OPERADORA,NR_DEMONSTRATIVO,DT_EMISSAO,CD_OPERADORA_PRESTADOR,CD_CGC_PRESTADOR,CD_CPF_PRESTADOR,NM_PRESTADOR,CD_CNES,NR_FATURA,VL_PROCESSADO_FATURA,VL_LIBERADO_FATURA,VL_GLOSA_FATURA,VL_PROCESSADO,VL_LIBERADO,VL_GLOSA,
ID_LTE,ID_PAI_LTE,NR_LOTE,DT_ENVIO_LOTE,NR_PROTOCOLO,VL_PROTOCOLO,VL_GLOSA_LTE
