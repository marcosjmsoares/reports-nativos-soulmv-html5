--R_GUIA_PRORROGACAO_SUL_AMERICA
SELECT ROWNUM LINHA
     , GUIA.CD_ATENDIMENTO
     , GUIA.DS_JUSTIFICATIVA
     , GUIA.CD_CONVENIO
     , NVL(GUIA.NR_DIAS_SOLICITADOS,0) NR_DIARIAS
     , TO_CHAR(SYSDATE,'DD/MM/YYYY') DT_EMISSAO
     , (SELECT DS_RAZAO_SOCIAL FROM DBAMV.MULTI_EMPRESAS WHERE CD_MULTI_EMPRESA = Nvl(DBAMV.PKG_MV2000.LE_EMPRESA,1)) DS_RAZAO_SOCIAL
     , (SELECT CD_CGC FROM DBAMV.MULTI_EMPRESAS WHERE CD_MULTI_EMPRESA = Nvl(DBAMV.PKG_MV2000.LE_EMPRESA,1)) CD_CNPJ
     , PACIENTE.NM_PACIENTE
     , CARTEIRA.NR_MATRICULA_TITULAR
     , TO_CHAR(CARTEIRA.DT_VALIDADE_CARTEIRA,'DD/MM/YYYY') DT_VALIDADE_CARTEIRA
     , GUIA.CD_SENHA
     , TO_CHAR(GUIA.DT_VALIDADE,'DD/MM/YYYY') DT_VALIDADE
     , CONVENIO.NM_CONVENIO
     , CONVENIO.NR_REGISTRO_OPERADORA_ANS
     , DBAMV.FN_IDADE(PACIENTE.DT_NASCIMENTO,'a A m M d D',GUIA.DT_SOLICITACAO) NR_IDADE
     , (SELECT CD_UF FROM DBAMV.MULTI_EMPRESAS WHERE CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA) CD_UF
     , (SELECT CIDADE.NM_CIDADE
          FROM DBAMV.MULTI_EMPRESAS ME
             , DBAMV.CIDADE
         WHERE ME.CD_CIDADE = CIDADE.CD_CIDADE
           AND CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA) DS_CIDADE
    , ATENDIME.CD_LEITO
    , TO_CHAR(ATENDIME.DT_ATENDIMENTO,'DD/MM/YYYY') DT_INTERNACAO
    , TO_CHAR(ATENDIME.DT_ALTA,'DD/MM/YYYY') DT_ALTA
    , DBAMV.PKG_TISS_UTIL.FNC_TRADUZ_PROC(GUIA.CD_CONVENIO,'D',ATENDIME.CD_PRO_INT) DS_DIAG
    , DBAMV.PKG_TISS_UTIL.FNC_TRADUZ_PROC(GUIA.CD_CONVENIO,'C',IT_GUIA.CD_PRO_FAT) CD_PROCEDIMENTO_ITEM
    , DBAMV.PKG_TISS_UTIL.FNC_TRADUZ_PROC(GUIA.CD_CONVENIO,'D',IT_GUIA.CD_PRO_FAT) DS_PROCEDIMENTO_ITEM
    , IT_GUIA.QT_UTILIZADA
    , NVL(IT_GUIA.DS_OBSERVACAO,GUIA.DS_JUSTIFICATIVA) DS_JUSTIFICATIVA
    , To_Char(IT_GUIA.DT_GERACAO,'DD/MM/YYYY') DT_GERACAO
  FROM DBAMV.GUIA
     , DBAMV.PACIENTE
     , DBAMV.CONVENIO
     , DBAMV.ATENDIME
     , DBAMV.IT_GUIA
     , ( SELECT NR_MATRICULA_TITULAR, DT_VALIDADE_CARTEIRA, CD_CONVENIO, CD_CON_PLA
           FROM (SELECT NVL(CARTEIRA.NR_MATRICULA_TITULAR,CARTEIRA.NR_CARTEIRA) NR_MATRICULA_TITULAR, MAX(CARTEIRA.DT_VALIDADE) DT_VALIDADE_CARTEIRA,
                        CARTEIRA.CD_CONVENIO, CARTEIRA.CD_CON_PLA
                   FROM DBAMV.CARTEIRA, DBAMV.GUIA
                  WHERE GUIA.CD_GUIA = :CAMPO_CD_GUIA_FILHA
                    AND CARTEIRA.CD_PACIENTE = GUIA.CD_PACIENTE
                    AND CARTEIRA.CD_CONVENIO = GUIA.CD_CONVENIO
                  GROUP BY NVL(CARTEIRA.NR_MATRICULA_TITULAR,CARTEIRA.NR_CARTEIRA), CARTEIRA.CD_CONVENIO, CARTEIRA.CD_CON_PLA )
          WHERE ROWNUM=1 ) CARTEIRA
 WHERE PACIENTE.CD_PACIENTE = GUIA.CD_PACIENTE
   AND CARTEIRA.CD_CONVENIO(+)   = GUIA.CD_CONVENIO
   AND CONVENIO.CD_CONVENIO   = GUIA.CD_CONVENIO
   AND ATENDIME.CD_ATENDIMENTO = GUIA.CD_ATENDIMENTO
   AND IT_GUIA.CD_GUIA(+) = GUIA.CD_GUIA
   AND GUIA.CD_GUIA = :CAMPO_CD_GUIA_FILHA
 ORDER BY ROWNUM,IT_GUIA.CD_IT_GUIA
