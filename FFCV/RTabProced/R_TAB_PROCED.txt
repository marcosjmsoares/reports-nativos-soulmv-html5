--R_TAB_PROCED
SELECT * FROM (
SELECT TAB_FAT.CD_TAB_FAT                   CD_TAB_FAT1
              ,TAB_FAT.DS_TAB_FAT                   DS_TAB_FAT
              ,TAB_FAT.TP_TAB_FAT                   TP_TAB_FAT
              ,VAL_PRO.CD_PRO_FAT                  CODIGO
              ,VAL_PRO.CD_TAB_FAT                   CD_TAB_FAT
              ,TRUNC(VAL_PRO.DT_VIGENCIA)   DT_VIGENCIA
              ,VAL_PRO.VL_TOTAL                        VL_TOTAL
              ,PRO_FAT.DS_PRO_FAT                   DESCRICAO
              ,PRO_FAT.CD_PRO_FAT                   CD_PRO_FAT2
              ,PRO_FAT.DS_UNIDADE                   DS_UNIDADE1
              ,PRO_FAT.NR_AUXILIAR                   NR_AUXILIAR1
              ,PRO_FAT.TP_SEXO                          TP_SEXO1
              ,0		                           VL_FILME1
              ,POR_ANE.CD_POR_ANE                  CD_POR_ANE1
              ,POR_ANE.DS_POR_ANE                  DS_POR_ANE
              ,GRU_PRO.CD_GRU_PRO                CD_GRU_PRO1
              ,GRU_PRO.DS_GRU_PRO                DS_GRU_PRO
              ,val_pro.vl_honorario                            vl_honorario
              ,val_pro.vl_operacional                         vl_operacional
FROM  dbamv.VAL_PRO       VAL_PRO
           ,dbamv.TAB_FAT        TAB_FAT
           ,dbamv.PRO_FAT        PRO_FAT
           ,dbamv.POR_ANE       POR_ANE
           ,dbamv.GRU_PRO      GRU_PRO
WHERE TAB_FAT.CD_TAB_FAT = VAL_PRO.CD_TAB_FAT
AND   TAB_FAT.CD_TAB_FAT=  :CAMPO_TAB_FAT
AND   PRO_FAT.CD_PRO_FAT    = VAL_PRO.CD_PRO_FAT
AND   PRO_FAT.SN_ATIVO = 'S'
AND       NVL(VAL_PRO.SN_ATIVO, 'S') = decode(:CAMPO_SN_ATIVO, 'T', NVL(VAL_PRO.SN_ATIVO, 'S'), :CAMPO_SN_ATIVO)
AND   GRU_PRO.CD_GRU_PRO = PRO_FAT.CD_GRU_PRO
AND   PRO_FAT.CD_POR_ANE   = POR_ANE.CD_POR_ANE (+)
AND   TRUNC(VAL_PRO.DT_VIGENCIA)  = (SELECT MAX(TRUNC(VAL_PRO_3.DT_VIGENCIA))
                                     FROM DBAMV.VAL_PRO    VAL_PRO_3
                                     WHERE VAL_PRO_3.CD_TAB_FAT = VAL_PRO.CD_TAB_FAT
                                           AND NVL(VAL_PRO_3.SN_ATIVO, 'S') = decode(:CAMPO_SN_ATIVO, 'T', NVL(VAL_PRO_3.SN_ATIVO, 'S'), :CAMPO_SN_ATIVO)
                                           AND   VAL_PRO_3.CD_PRO_FAT = VAL_PRO.CD_PRO_FAT)
:CAMPO_CF_COND_GRU_PRO
)
ORDER BY CD_TAB_FAT1, CD_GRU_PRO1, :CAMPO_ORDEM
