--R_DIARIAS_CONS_X_FAT_ANA
SELECT * FROM (
SELECT CD_ATENDIMENTO
      ,CD_PACIENTE
      ,DT_ATENDIMENTO
      ,DT_ALTA
      ,CD_TIP_ACOM
      ,CD_PRO_FAT
      ,SUM(NR_CONSUMIDAS)  NR_CONSUMIDAS
      ,DBAMV.RETORNA_DIARIAS_FATURADAS(CD_ATENDIMENTO
                                      ,:CAMPO_DT_INICIO
                                      ,:CAMPO_DT_FIM
                                      ,CD_PRO_FAT
                                      ,'Faturamento' )  NR_FATURADAS_MANUAL
      ,DBAMV.RETORNA_DIARIAS_FATURADAS(CD_ATENDIMENTO
                                      ,:CAMPO_DT_INICIO
                                      ,:CAMPO_DT_FIM
                                      ,CD_PRO_FAT
                                      ,'Diaria' )       NR_FATURADAS_AUTOMAT
FROM
(
SELECT ATENDIME.CD_ATENDIMENTO
      ,ATENDIME.CD_PACIENTE
      ,ATENDIME.DT_ATENDIMENTO
      ,ATENDIME.DT_ALTA
      ,MOV_INT.CD_TIP_ACOM
      ,( TRUNC(MOV_INT.DT_LIB_MOV) - :CAMPO_DT_INICIO ) NR_CONSUMIDAS
      ,ACOMOD_PRO_FAT.CD_PRO_FAT  CD_PRO_FAT
FROM   DBAMV.ATENDIME
      ,DBAMV.MOV_INT
      ,DBAMV.ACOMOD_PRO_FAT
       ,dbamv.convenio                       
	,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
And ( TRUNC(MOV_INT.DT_MOV_INT) < :CAMPO_DT_INICIO AND TRUNC(MOV_INT.DT_LIB_MOV) BETWEEN :CAMPO_DT_INICIO AND :CAMPO_DT_FIM )
AND    ACOMOD_PRO_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND    (MOV_INT.TP_MOV = 'I' OR MOV_INT.TP_MOV = 'O')
AND    ATENDIME.TP_ATENDIMENTO = 'I'
AND    ( TRUNC(ATENDIME.DT_ALTA) > :CAMPO_DT_INICIO OR ATENDIME.DT_ALTA IS NULL)
AND    ATENDIME.CD_ATENDIMENTO = MOV_INT.CD_ATENDIMENTO
AND    ATENDIME.CD_CONVENIO = ACOMOD_PRO_FAT.CD_CONVENIO
AND    ATENDIME.CD_CON_PLA  = ACOMOD_PRO_FAT.CD_CON_PLA
AND    MOV_INT.CD_TIP_ACOM = ACOMOD_PRO_FAT.CD_TIP_ACOM
and   CONVENIO.CD_CONVENIO = DBAMV.ATENDIME.CD_CONVENIO    
and    CONVENIO.TP_CONVENIO IN ('C','P')                                                  
AND    MOV_INT.CD_MOV_INT = ( SELECT MAX(CD_MOV_INT)
                              FROM   DBAMV.MOV_INT MOV_AUX
                              WHERE  MOV_AUX.CD_ATENDIMENTO = MOV_INT.CD_ATENDIMENTO
                              AND    TRUNC(MOV_AUX.DT_MOV_INT) = TRUNC(MOV_INT.DT_MOV_INT)
                              AND   (MOV_INT.TP_MOV = 'I' OR MOV_INT.TP_MOV = 'O') )
UNION ALL
SELECT ATENDIME.CD_ATENDIMENTO
      ,ATENDIME.CD_PACIENTE
      ,ATENDIME.DT_ATENDIMENTO
      ,ATENDIME.DT_ALTA
      ,MOV_INT.CD_TIP_ACOM
      ,DECODE(MOV_INT.CD_MOV_INT,(SELECT MAX(M.CD_MOV_INT)
                                  FROM   DBAMV.MOV_INT M
                                  WHERE  M.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
                                  AND    TRUNC(M.DT_MOV_INT) <= :CAMPO_DT_FIM),
                                  ( :CAMPO_DT_FIM - :CAMPO_DT_INICIO ) + 1,
                                  ( :CAMPO_DT_FIM - :CAMPO_DT_INICIO ) )  NR_CONSUMIDAS
      ,ACOMOD_PRO_FAT.CD_PRO_FAT  CD_PRO_FAT
FROM   DBAMV.ATENDIME
      ,DBAMV.MOV_INT
      ,DBAMV.ACOMOD_PRO_FAT
     ,dbamv.convenio,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
And  ( ( TRUNC(MOV_INT.DT_MOV_INT) < :CAMPO_DT_INICIO AND MOV_INT.DT_LIB_MOV IS NULL )
OR ( TRUNC(MOV_INT.DT_MOV_INT) < :CAMPO_DT_INICIO AND MOV_INT.DT_LIB_MOV > :CAMPO_DT_FIM )  )
AND    (MOV_INT.TP_MOV = 'I' OR MOV_INT.TP_MOV = 'O')
AND    ATENDIME.TP_ATENDIMENTO = 'I'
AND ACOMOD_PRO_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND    ( TRUNC(ATENDIME.DT_ALTA) > :CAMPO_DT_INICIO OR ATENDIME.DT_ALTA IS NULL)
AND    ATENDIME.CD_ATENDIMENTO = MOV_INT.CD_ATENDIMENTO
AND    ATENDIME.CD_CONVENIO = ACOMOD_PRO_FAT.CD_CONVENIO
AND    ATENDIME.CD_CON_PLA  = ACOMOD_PRO_FAT.CD_CON_PLA
AND    MOV_INT.CD_TIP_ACOM = ACOMOD_PRO_FAT.CD_TIP_ACOM
and   CONVENIO.CD_CONVENIO = DBAMV.ATENDIME.CD_CONVENIO    
and    CONVENIO.TP_CONVENIO IN ('C','P')                                                  
AND    MOV_INT.CD_MOV_INT = ( SELECT MAX(CD_MOV_INT)
                              FROM   DBAMV.MOV_INT MOV_AUX
                              WHERE  MOV_AUX.CD_ATENDIMENTO = MOV_INT.CD_ATENDIMENTO
                              AND    TRUNC(MOV_AUX.DT_MOV_INT) = TRUNC(MOV_INT.DT_MOV_INT)
                              AND   (MOV_INT.TP_MOV = 'I' OR MOV_INT.TP_MOV = 'O') )
UNION ALL
SELECT ATENDIME.CD_ATENDIMENTO
      ,ATENDIME.CD_PACIENTE
      ,ATENDIME.DT_ATENDIMENTO
      ,ATENDIME.DT_ALTA
      ,MOV_INT.CD_TIP_ACOM
      ,DECODE(TRUNC(ATENDIME.DT_ALTA),TRUNC(MOV_INT.DT_MOV_INT),1,(NVL(TRUNC(MOV_INT.DT_LIB_MOV), :CAMPO_DT_FIM) - TRUNC(MOV_INT.DT_MOV_INT))) NR_CONSUMIDAS
      ,ACOMOD_PRO_FAT.CD_PRO_FAT  CD_PRO_FAT
FROM   DBAMV.ATENDIME
      ,DBAMV.MOV_INT
      ,DBAMV.ACOMOD_PRO_FAT
     ,dbamv.convenio
     ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
And TRUNC(MOV_INT.DT_MOV_INT) BETWEEN :CAMPO_DT_INICIO AND 
AND    TRUNC(MOV_INT.DT_LIB_MOV) BETWEEN :CAMPO_DT_INICIO AND 
AND ACOMOD_PRO_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND    (MOV_INT.TP_MOV = 'I' OR MOV_INT.TP_MOV = 'O')
AND    ATENDIME.TP_ATENDIMENTO = 'I'
AND    ( TRUNC(ATENDIME.DT_ALTA) > :CAMPO_DT_INICIO OR ATENDIME.DT_ALTA IS NULL)
AND    ATENDIME.CD_ATENDIMENTO = MOV_INT.CD_ATENDIMENTO
AND    ATENDIME.CD_CONVENIO = ACOMOD_PRO_FAT.CD_CONVENIO
AND    ATENDIME.CD_CON_PLA  = ACOMOD_PRO_FAT.CD_CON_PLA
AND    MOV_INT.CD_TIP_ACOM = ACOMOD_PRO_FAT.CD_TIP_ACOM
and   CONVENIO.CD_CONVENIO = DBAMV.ATENDIME.CD_CONVENIO    
and    CONVENIO.TP_CONVENIO IN ('C','P')                                                  
AND    MOV_INT.CD_MOV_INT = ( SELECT MAX(CD_MOV_INT)
                              FROM   DBAMV.MOV_INT MOV_AUX
                              WHERE  MOV_AUX.CD_ATENDIMENTO = MOV_INT.CD_ATENDIMENTO
                              AND    TRUNC(MOV_AUX.DT_MOV_INT) = TRUNC(MOV_INT.DT_MOV_INT)
                              AND   (MOV_INT.TP_MOV = 'I' OR MOV_INT.TP_MOV = 'O') )
UNION ALL
SELECT ATENDIME.CD_ATENDIMENTO
      ,ATENDIME.CD_PACIENTE
      ,ATENDIME.DT_ATENDIMENTO
      ,ATENDIME.DT_ALTA
      ,MOV_INT.CD_TIP_ACOM
      ,DECODE(MOV_INT.CD_MOV_INT,(SELECT MAX(M.CD_MOV_INT)
                                  FROM   DBAMV.MOV_INT M
                                  WHERE  M.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
                                  AND    TRUNC(M.DT_MOV_INT) <= :CAMPO_DT_FIM),
                                ( :CAMPO_DT_FIM - TRUNC(MOV_INT.DT_MOV_INT) ) + 1,
                                ( :CAMPO_DT_FIM - TRUNC(MOV_INT.DT_MOV_INT) ) ) NR_CONSUMIDAS
      ,ACOMOD_PRO_FAT.CD_PRO_FAT  CD_PRO_FAT
FROM   DBAMV.ATENDIME
      ,DBAMV.MOV_INT
      ,DBAMV.ACOMOD_PRO_FAT
      ,dbamv.convenio
      ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
And TRUNC(MOV_INT.DT_MOV_INT) BETWEEN :CAMPO_DT_INICIO AND 
AND  ( TRUNC(MOV_INT.DT_LIB_MOV) > TRUNC(:CAMPO_DT_FIM) OR  MOV_INT.DT_LIB_MOV IS NULL )
AND ACOMOD_PRO_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND    (MOV_INT.TP_MOV = 'I' OR MOV_INT.TP_MOV = 'O')
AND    ATENDIME.TP_ATENDIMENTO = 'I'
AND    ( TRUNC(ATENDIME.DT_ALTA) > :CAMPO_DT_INICIO OR ATENDIME.DT_ALTA IS NULL)
AND    ATENDIME.CD_ATENDIMENTO = MOV_INT.CD_ATENDIMENTO
AND    ATENDIME.CD_CONVENIO = ACOMOD_PRO_FAT.CD_CONVENIO
AND    ATENDIME.CD_CON_PLA  = ACOMOD_PRO_FAT.CD_CON_PLA
AND    MOV_INT.CD_TIP_ACOM = ACOMOD_PRO_FAT.CD_TIP_ACOM
and   CONVENIO.CD_CONVENIO = DBAMV.ATENDIME.CD_CONVENIO    
and    CONVENIO.TP_CONVENIO IN ('C','P')                                                  
AND    MOV_INT.CD_MOV_INT = ( SELECT MAX(CD_MOV_INT)
                              FROM   DBAMV.MOV_INT MOV_AUX
                              WHERE  MOV_AUX.CD_ATENDIMENTO = MOV_INT.CD_ATENDIMENTO
                              AND    TRUNC(MOV_AUX.DT_MOV_INT) = TRUNC(MOV_INT.DT_MOV_INT)
                              AND   (MOV_INT.TP_MOV = 'I' OR MOV_INT.TP_MOV = 'O') )
)
GROUP BY CD_ATENDIMENTO
        ,CD_PACIENTE
        ,DT_ATENDIMENTO
        ,DT_ALTA
        ,CD_TIP_ACOM
        ,CD_PRO_FAT
      ,DBAMV.RETORNA_DIARIAS_FATURADAS(CD_ATENDIMENTO
                                      ,:CAMPO_DT_INICIO
                                      ,:CAMPO_DT_FIM
                                      ,CD_PRO_FAT
                                      ,'Faturamento' )
      ,DBAMV.RETORNA_DIARIAS_FATURADAS(CD_ATENDIMENTO
                                      ,:CAMPO_DT_INICIO
                                      ,:CAMPO_DT_FIM
                                      ,CD_PRO_FAT
                                      ,'Diaria' )
)
ORDER BY CD_ATENDIMENTO,CD_PACIENTE,DT_ATENDIMENTO,DT_ALTA,
CD_TIP_ACOM
