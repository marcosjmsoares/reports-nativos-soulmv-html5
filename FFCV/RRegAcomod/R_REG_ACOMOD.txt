--R_REG_ACOMOD
SELECT REGRA.CD_REGRA CODIGO,
               REGRA.DS_REGRA DESCRICAO,
               REGRA.SN_PAGA_HORARIO_ESPECIAL SN_PAGA_HORARIO_ESPECIAL,
               REGRA.CD_HORARIO CD_HORARIO
FROM    dbamv.REGRA REGRA
WHERE dbamv.FNC_FFCV_RESTRINGE_REGRA_VALOR( null,'MODEL',null,'EMPRESA_CON_PLA','CD_REGRA', REGRA.CD_REGRA, null, null) = 'N'
ORDER BY
