--R_EXTRATO_CAPTA
SELECT  CD_CONVENIO,
       :CAMPO_NM_CONVENIO NM_CONVENIO,
       (:CAMPO_DT_INICIAL+ rownum-1) DT_REFERENCIA
  FROM DBASGU.MENU
  WHERE ROWNUM <= (:CAMPO_DT_FINAL - :CAMPO_DT_INICIAL +1 )
ORDER BY NM_CONVENIO, DT_REFERENCIA
