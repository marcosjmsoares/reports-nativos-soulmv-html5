--R_CRED_PREST
SELECT PRES_CON.CD_PRESTADOR
      ,PRESTADOR.NM_PRESTADOR
      ,PRES_CON.CD_CONVENIO
      ,CONVENIO.NM_CONVENIO
      ,PRES_CON.SN_PAGA_PELO_CONVENIO
      ,PRES_CON.CD_PRESTADOR_CONVENIADO
  FROM dbamv.PRES_CON         PRES_CON
    ,  dbamv.PRESTADOR      PRESTADOR
    ,  dbamv.CONVENIO          CONVENIO
      ,Dbamv.Empresa_Convenio
 Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
   And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   And PRESTADOR.CD_PRESTADOR = PRES_CON.CD_PRESTADOR
   And CONVENIO.CD_CONVENIO = PRES_CON.CD_CONVENIO
   and (pres_con.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA or pres_con.cd_multi_empresa is null)
   And PRES_CON.sn_paga_pelo_convenio  = 'S'
 ORDER BY PRESTADOR.NM_PRESTADOR,
          PRES_CON.CD_PRESTADOR,
          CONVENIO.NM_CONVENIO,
          CONVENIO.CD_CONVENIO,
          PRES_CON.CD_PRESTADOR_CONVENIADO,
          PRES_CON.SN_PAGA_PELO_CONVENIO
