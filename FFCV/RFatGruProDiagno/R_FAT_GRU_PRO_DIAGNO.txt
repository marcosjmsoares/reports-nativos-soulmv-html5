--R_FAT_GRU_PRO_DIAGNO
SELECT * FROM (
SELECT GRU_PRO.CD_GRU_PRO
              ,GRU_PRO.DS_GRU_PRO
              ,PRO_FAT.CD_PRO_FAT
              ,PRO_FAT.DS_PRO_FAT
              ,ITREG_FAT.QT_LANCAMENTO
              ,ITREG_FAT.VL_TOTAL_CONTA         VALOR_PAGAR
FROM     DBAMV.REG_FAT
              ,DBAMV.CONVENIO
              ,DBAMV.ITREG_FAT
              ,DBAMV.PRO_FAT
              ,DBAMV.GRU_PRO
              ,DBAMV.REMESSA_FATURA                 REMESSA
              ,DBAMV.FATURA
	      ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And REG_FAT.CD_REG_FAT = ITREG_FAT.CD_REG_FAT
AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND        REG_FAT.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND        ITREG_FAT.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
AND        PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
AND        REG_FAT.CD_REMESSA = REMESSA.CD_REMESSA
AND        REMESSA.CD_FATURA = FATURA.CD_FATURA
AND        ITREG_FAT.SN_PERTENCE_PACOTE = 'N'
AND        NVL(REG_FAT.SN_DIAGNO,'N') = 'S'
UNION all
SELECT GRU_PRO.CD_GRU_PRO
              ,GRU_PRO.DS_GRU_PRO
              ,PRO_FAT.CD_PRO_FAT
              ,PRO_FAT.DS_PRO_FAT
              ,ITREG_AMB.QT_LANCAMENTO
              ,ITREG_AMB.VL_TOTAL_CONTA      VALOR_PAGAR
FROM    DBAMV.ITREG_AMB
             ,DBAMV.CONVENIO
             ,DBAMV.REG_AMB
             ,DBAMV.PRO_FAT
             ,DBAMV.GRU_PRO
             ,DBAMV.REMESSA_FATURA              REMESSA
             ,DBAMV.FATURA
	     ,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And ITREG_AMB.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND REG_amb.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND        ITREG_AMB.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
AND        PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
AND        ITREG_AMB.CD_REG_AMB = REG_AMB.CD_REG_AMB
AND        REG_AMB.CD_REMESSA = REMESSA.CD_REMESSA
AND        REMESSA.CD_FATURA = FATURA.CD_FATURA
AND        ITREG_AMB.SN_PERTENCE_PACOTE = 'N'
AND        NVL(REG_AMB.SN_DIAGNO,'N') = 'S'
:CAMPO_CF_COND_GRU_PRO
)
ORDER BY CD_GRU_PRO,DS_GRU_PRO,
CD_PRO_FAT,DS_PRO_FAT
