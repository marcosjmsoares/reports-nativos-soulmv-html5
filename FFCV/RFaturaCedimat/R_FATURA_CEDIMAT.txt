--R_FATURA_AMBULATORIAL_ITENS_CEDIMAT
SELECT CD_GRU_PRO, DS_GRU_PRO, CD_PRO_FAT, DS_PRO_FAT, SUM(QT_LANCAMENTO) QT_LANCAMENTOS,
VL_UNITARIO, SUM(VL_BRUTO) VALOR_TOTAL,
SUM(VL_COBERTURA) VL_COBERTURAS,
(SUM(VL_BRUTO) - SUM(VL_COBERTURA)) VL_DIFERENCIA,
CD_LANCAMENTO
FROM (
SELECT GRU_PRO.CD_GRU_PRO CD_GRU_PRO,
       GRU_PRO.DS_GRU_PRO DS_GRU_PRO,
       PRO_FAT.CD_PRO_FAT CD_PRO_FAT,
       PRO_FAT.DS_PRO_FAT DS_PRO_FAT,
       ITREG_AMB.QT_LANCAMENTO QT_LANCAMENTO,
       ITREG_AMB.VL_PERCENTUAL_MULTIPLA VL_PERCENTUAL_MULTIPLA,
       NVL(ITREG_AMB.VL_UNITARIO,0) VL_UNITARIO,
       (DECODE (NVL(ITREG_AMB.VL_DESCONTO_CONTA,0), 0,
       NVL(ITREG_AMB.VL_TOTAL_CONTA,0),
       (NVL(ITREG_AMB.VL_TOTAL_CONTA,0) + NVL(ITREG_AMB.VL_DESCONTO_CONTA,0)))) VL_BRUTO,
       DECODE(CONVENIO.TP_CONVENIO, 'P', 0, NVL(ITREG_AMB.VL_TOTAL_CONTA,0)) VL_COBERTURA,
       ITREG_AMB.CD_LANCAMENTO CD_LANCAMENTO
FROM  ITREG_AMB,
       PRO_FAT,
       GRU_PRO,
       REG_AMB,
       CONVENIO
WHERE ITREG_AMB.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
AND PRO_FAT.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
AND ITREG_AMB.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
AND ITREG_AMB.CD_REG_AMB = REG_AMB.CD_REG_AMB
AND CONVENIO.CD_CONVENIO = REG_AMB.CD_CONVENIO
AND ITREG_AMB.CD_ATENDIMENTO = :CAMPO_CD_ATENDIMENTO
AND REG_AMB.CD_REG_AMB = :CAMPO_CD_REG_AMB
AND REG_AMB.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND NVL(ITREG_AMB.TP_PAGAMENTO,'P') <> 'C'
AND NVL(ITREG_AMB.SN_PERTENCE_PACOTE,'N') = 'N'
)
GROUP BY CD_GRU_PRO, DS_GRU_PRO, CD_PRO_FAT, DS_PRO_FAT, VL_UNITARIO, CD_LANCAMENTO
ORDER BY CD_GRU_PRO, CD_PRO_FAT

--R_FATURA_AMBULATORIAL_ITENS_PRESTADORES_CEDIMAT
SELECT PRESTADOR.NM_PRESTADOR NM_PRESTADOR,
       PRESTADOR.CD_PRESTADOR CD_PRESTADOR,
       ATI_MED.DS_ATI_MED DS_ATIVIDADE,
       ATI_MED.VL_PERCENTUAL_PAGO VL_PERCENTUAL_PAGO,
       SUM(DECODE (NVL(ITREG_AMB.VL_DESCONTO_CONTA,0), 0,
       NVL(ITREG_AMB.VL_TOTAL_CONTA,0),
       (NVL(ITREG_AMB.VL_TOTAL_CONTA,0) + NVL(ITREG_AMB.VL_DESCONTO_CONTA,0)))) VL_BRUTO,
       SUM(DECODE(CONVENIO.TP_CONVENIO, 'P', 0, NVL(ITREG_AMB.VL_TOTAL_CONTA,0))) VL_COBERTURA,
       SUM((DECODE (NVL(ITREG_AMB.VL_DESCONTO_CONTA,0), 0,
       NVL(ITREG_AMB.VL_TOTAL_CONTA,0),
       (NVL(ITREG_AMB.VL_TOTAL_CONTA,0) + NVL(ITREG_AMB.VL_DESCONTO_CONTA,0))))
       -
       (DECODE(CONVENIO.TP_CONVENIO, 'P', 0, NVL(ITREG_AMB.VL_TOTAL_CONTA,0))) ) VL_NETO
FROM  ITREG_AMB,
      PRO_FAT,
      REG_AMB,
      CONVENIO,
      PRESTADOR,
      ATI_MED
WHERE ITREG_AMB.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
AND ITREG_AMB.CD_PRO_FAT = PRO_FAT.CD_PRO_FAT
AND ITREG_AMB.CD_REG_AMB = REG_AMB.CD_REG_AMB
AND CONVENIO.CD_CONVENIO = REG_AMB.CD_CONVENIO
AND ITREG_AMB.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
AND ITREG_AMB.CD_ATI_MED = ATI_MED.CD_ATI_MED
AND ITREG_AMB.CD_ATENDIMENTO = :CAMPO_CD_ATENDIMENTO
AND REG_AMB.CD_REG_AMB = :CAMPO_CD_REG_AMB
AND REG_AMB.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND ITREG_AMB.CD_PRO_FAT = :CAMPO_CD_PRO_FAT
AND PRO_FAT.CD_GRU_PRO = :CAMPO_CD_GRU_PRO
AND NVL( ITREG_AMB.TP_PAGAMENTO, 'P' )  <> 'C'
AND NVL(ITREG_AMB.SN_PERTENCE_PACOTE,'N') = 'N'
AND ITREG_AMB.CD_PRESTADOR IS NOT NULL
GROUP BY PRESTADOR.CD_PRESTADOR, PRESTADOR.NM_PRESTADOR, ATI_MED.DS_ATI_MED, ATI_MED.VL_PERCENTUAL_PAGO
ORDER BY PRESTADOR.NM_PRESTADOR
