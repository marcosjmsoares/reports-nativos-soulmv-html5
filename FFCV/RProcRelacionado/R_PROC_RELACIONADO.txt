--R_PROC_RELACIONADO
SELECT * FROM (
select itreg_fat.cd_pro_fat,
       pro_fat.ds_pro_fat,
       itreg_fat_rel.cd_pro_fat cd_pro_fat_rel,
       pro_fat_rel.ds_pro_fat   ds_pro_fat_rel,
       itreg_fat_rel.dt_lancamento,
       itreg_fat_rel.qt_lancamento,
       itreg_fat_rel.vl_total_conta,
       itreg_fat_rel.cd_lancamento,
       reg_fat.cd_atendimento,
       atendime.cd_paciente,
       'HOSPITALAR' tp_conta
  from dbamv.itreg_fat,
       dbamv.pro_fat,
       dbamv.reg_fat,
       dbamv.atendime,
       dbamv.itreg_fat itreg_fat_rel,
       dbamv.pro_fat pro_fat_rel
 where itreg_fat.cd_reg_fat    = :CAMPO_CONTA
   AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   and itreg_fat.cd_lancamento = :CAMPO_CD_LANCAMENTO
   and reg_fat.cd_reg_fat      = itreg_fat.cd_reg_fat
   and atendime.cd_atendimento = reg_fat.cd_atendimento
   and pro_fat.cd_pro_fat      = itreg_fat.cd_pro_fat
   and itreg_fat.cd_reg_fat    = itreg_fat_rel.cd_reg_fat
   and itreg_fat.cd_lancamento = itreg_fat_rel.cd_lancamento_rel
   and pro_fat_rel.cd_pro_fat  = itreg_fat_rel.cd_pro_fat
   and atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
union all
select distinct itreg_amb.cd_pro_fat,
       pro_fat.ds_pro_fat,
       itreg_amb_rel.cd_pro_fat cd_pro_fat_rel,
       pro_fat_rel.ds_pro_fat   ds_pro_fat_rel,
       nvl(itreg_amb.dt_sessao,atendime.dt_atendimento) dt_lancamento,
       itreg_amb_rel.qt_lancamento,
       itreg_amb_rel.vl_total_conta,
       itreg_amb_rel.cd_lancamento,
       atendime.cd_atendimento,
       atendime.cd_paciente,
       'AMBULATORIAL'   tp_conta
  from dbamv.itreg_amb,
       dbamv.pro_fat,
       dbamv.itreg_amb itreg_amb_rel,
       dbamv.pro_fat pro_fat_rel,
       dbamv.atendime
 where itreg_amb.cd_reg_amb    = :CAMPO_CONTA
   and itreg_amb.cd_lancamento = :CAMPO_CD_LANCAMENTO
   and atendime.cd_atendimento = itreg_amb.cd_atendimento
   and pro_fat.cd_pro_fat      = itreg_amb.cd_pro_fat
   and itreg_amb.cd_reg_amb    = itreg_amb_rel.cd_reg_amb
   and itreg_amb.cd_lancamento = itreg_amb_rel.cd_lancamento_rel
   and pro_fat_rel.cd_pro_fat  = itreg_amb_rel.cd_pro_fat
   and atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
 order by 8
)
ORDER BY CD_ATENDIMENTO,CD_PACIENTE,TP_CONTA,CD_PRO_FAT,DS_PRO_FAT
