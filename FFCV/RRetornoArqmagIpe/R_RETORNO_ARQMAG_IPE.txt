--R_RETORNO_ARQMAG_IPE
SELECT R.CD_RETORNO_ARQMAG_IPE, R.CD_RETORNO_IT_ARQMAG_IPE, R.NR_NOTA, R.TP_ARQMAG_IPE, R.CD_REF_RETORNO_ARQMAG_IPE,
                        R.CD_PROCEDIMENTO, R.DS_LOG, I.NM_ARQUIVO
          FROM DBAMV.RETORNO_IT_ARQMAG_IPE R, dbamv.log_retorno_arqmag_ipe I, dbamv.retorno_arqmag_ipe T
		 WHERE R.CD_RETORNO_ARQMAG_IPE = :CAMPO_PID_RETORNO 
       AND T.CD_RETORNO_ARQMAG_IPE = R.CD_RETORNO_ARQMAG_IPE
       AND T.CD_RETORNO_LOG_IPE = I.CD_RETORNO_LOG_IPE
         ORDER BY 1,2,3,4,5
