--R_FCESP_DS_PROC
SELECT * FROM (
select l.nr_nota,
       LPad(
        SubStr(
        Rtrim(
        Ltrim(
        Replace(
        Replace(
        Replace(
        Replace(c.nr_carteira, ',', ''),'.',''),'-','')
                ,' ',''))),1,11),11,'0') identificacao,
       LPad(
        SubStr(
        Rtrim(
        Ltrim(
        Replace(
        Replace(
        Replace(
        Replace(c.nr_carteira, ',', ''),'.',''),'-','')
                ,' ',''))),12, 3), 3,'0') beneficiario,
       c.nm_titular,
       p.nm_paciente,
       sum(l.vl_sh),
       l.seq_nota,
       l.tp_nota,
       l.cd_atendimento
  from dbamv.log_meio_mag l,
         (select distinct nr_carteira, nm_titular,cd_con_pla,cd_paciente,cd_convenio
            from dbamv.carteira) c, 
         dbamv.paciente p,
         dbamv.atendime a
 where l.cd_remessa        = :CAMPO_CD_REMESSA
   and  a.cd_atendimento = l.cd_atendimento
   and  p.cd_paciente       = a.cd_paciente
   and  c.cd_convenio      = a.cd_convenio
   and  c.cd_paciente       = a.cd_paciente
   and  c.cd_con_pla        = a.cd_con_pla
   and a.cd_multi_empresa =  :CAMPO_CD_MULTI_EMPRESA 
group by l.nr_nota,
         LPad(
          SubStr(
          Rtrim(
          Ltrim(
          Replace(
          Replace(
          Replace(
          Replace(c.nr_carteira, ',', ''),'.',''),'-','')
                  ,' ',''))),1,11),11,'0'),
         LPad(
          SubStr(
          Rtrim(
          Ltrim(
          Replace(
          Replace(
          Replace(
          Replace(c.nr_carteira, ',', ''),'.',''),'-','')
                  ,' ',''))),12, 3), 3,'0'),
         c.nm_titular,
         p.nm_paciente,
         l.seq_nota,
         l.tp_nota,
         l.cd_atendimento
order by l.seq_nota, l.tp_nota, l.nr_nota
)
