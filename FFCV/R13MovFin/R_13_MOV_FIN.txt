               DBAMV.RECCON_REC,
               DBAMV.REMESSA_FATURA REMESSA,
               DBAMV.FATURA,
               DBAMV.CONVENIO 
            WHERE
               CON_REC.TP_CON_REC IN 
               (
                  'P',
                  'C' 
               )
               AND CON_REC.CD_CON_REC = ITCON_REC.CD_CON_REC 
               AND ITCON_REC.CD_ITCON_REC = RECCON_REC.CD_ITCON_REC 
               AND TO_CHAR(RECCON_REC.DT_RECEBIMENTO, 'yyyy') = :CAMPO_DT_COMPET 
               AND CON_REC.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA 
               AND CON_REC.CD_REMESSA = REMESSA.CD_REMESSA 
               AND REMESSA.CD_FATURA = FATURA.CD_FATURA 
               AND FATURA.CD_CONVENIO = CONVENIO.CD_CONVENIO 
               AND ITCON_REC.TP_QUITACAO IN 
               (
                  'Q',
                  'P',
                  'G' 
               )
            UNION ALL
            SELECT
               0 CD_CONVENIO,
               'AVULSA' NM_CONVENIO,
               TO_CHAR (RECCON_REC.DT_RECEBIMENTO, 'MM') MES,
               RECCON_REC.VL_Recebido VL_TOTAL_CONTA,
               'Recebido' DS_CON_PLA,
               3 COD 
            FROM
               DBAMV.CON_REC,
               DBAMV.NOTA_FISCAL,
               DBAMV.ITCON_REC,
               DBAMV.RECCON_REC 
            WHERE
               NOTA_FISCAL.CD_NOTA_FISCAL = CON_REC.CD_NOTA_FISCAL 
               AND CON_REC.CD_CON_REC = ITCON_REC.CD_CON_REC 
               AND ITCON_REC.CD_ITCON_REC = RECCON_REC.CD_ITCON_REC 
               AND TO_CHAR (RECCON_REC.DT_RECEBIMENTO, 'YYYY') = :CAMPO_DT_COMPET 
               AND CON_REC.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA 
               AND ITCON_REC.TP_QUITACAO IN 
               (
                  'Q',
                  'P',
                  'G' 
               )
               AND NOTA_FISCAL.TP_NOTA_FISCAL = 'A' 
               :CAMPO_CF_COND_CONV 
         )
      GROUP BY
         NM_CONVENIO,
         MES,
         DS_CON_PLA,
         COD 
      ORDER BY
         VALOR DESC 
   )
ORDER BY
   NM_CONVENIO,
   COD,
   DS_CON_PLA,
   MES

FROM
  (
    SELECT
      MES MES_RESUMO,
      SUM(VL_TOTAL_CONTA) VALOR_RESUMO,
      TIPO TIPO_RESUMO,
      COD_RESUMO 
    FROM
      (
        SELECT
          CONVENIO . CD_CONVENIO CD_CONVENIO,
          CONVENIO.NM_CONVENIO NM_CONVENIO,
          TO_CHAR(ITREG_FAT.DT_LANCAMENTO, 'MM') MES,
          NVL(NVL(ITLAN_MED.VL_LIQUIDO, ITREG_FAT.VL_TOTAL_CONTA), 0) VL_TOTAL_CONTA,
          'LANÇADO' TIPO,
          1 COD_RESUMO 
        FROM
          DBAMV.REG_FAT,
          DBAMV.ITREG_FAT,
          DBAMV.CONVENIO,
          DBAMV.EMPRESA_CONVENIO,
          DBAMV . ATENDIME,
          DBAMV.ITLAN_MED 
        WHERE
          :CAMPO_FATU_CONSU = 'F' 
          AND REG_FAT.CD_REG_FAT = ITREG_FAT.CD_REG_FAT 
          AND EMPRESA_CONVENIO.CD_CONVENIO = CONVENIO.CD_CONVENIO 
          AND EMPRESA_CONVENIO.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA 
          AND ITREG_FAT.CD_REG_FAT = ITLAN_MED.CD_REG_FAT( + ) 
          AND ITREG_FAT.CD_LANCAMENTO = ITLAN_MED.CD_LANCAMENTO( + )
