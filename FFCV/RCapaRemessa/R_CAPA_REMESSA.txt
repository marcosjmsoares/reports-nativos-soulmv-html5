--R_CAPA_REMESSA
Select Sum( NVL( log_meio_mag.qtd_referencia, 0 ) ) tot_itens,
       Sum( NVL( log_meio_mag.vl_sh, 0 ) + NVL( log_meio_mag.vl_sp, 0 ) + NVL( log_meio_mag.vl_sd, 0 ) ) vl_total,
       count( distinct log_meio_mag.nr_nota ) tot_contas
  from dbamv.log_meio_mag
 where log_meio_mag.cd_remessa = :CAMPO_CD_REMESSA
   and log_meio_mag.tp_nota = :CAMPO_TP_NOTA
