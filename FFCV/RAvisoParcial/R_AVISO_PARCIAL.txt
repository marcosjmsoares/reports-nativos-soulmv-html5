--R_AVISO_PARCIAL
select setor.nm_setor,
          atendime.cd_atendimento,
          reg_fat.cd_reg_fat,
          dbamv.pkg_ffcv_paciente.FNC_RETORNA_NOME_PACIENTE(paciente.cd_paciente, 1) nm_paciente,
          reg_fat.dt_inicio,
          reg_fat.dt_final,
          convenio.nm_convenio,
          atendime.dt_atendimento
  from dbamv.atendime,
          dbamv.convenio,
          dbamv.paciente,
          dbamv.reg_fat,
          dbamv.setor,
          dbamv.ori_ate,
	  dbamv.empresa_convenio
where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And  atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
 and atendime.cd_atendimento = reg_fat.cd_atendimento
     AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
     and paciente.cd_paciente    = atendime.cd_paciente
     and ori_ate.cd_ori_ate      = atendime.cd_ori_ate
     and setor.cd_setor          = ori_ate.cd_setor
     and convenio.cd_convenio    = reg_fat.cd_convenio
     and nvl(reg_fat.sn_fechada,'N') = 'N'
     and convenio.tp_convenio    in ('C','P')
     and ( trunc(sysdate) - trunc(reg_fat.dt_inicio) ) >= :CAMPO_DIAS_PARCIAL
 order by nm_setor, nm_paciente
