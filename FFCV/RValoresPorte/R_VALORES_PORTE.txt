--R_VALORES_PORTE
SELECT * FROM (
SELECT  VAL_PORTE.CD_POR_ANE CODIGO
              ,POR_ANE.DS_POR_ANE DESCRICAO
              ,VAL_PORTE.CD_TAB_FAT CD_TAB_FAT
              ,TAB_FAT.DS_TAB_FAT DS_TAB_FAT
              ,VAL_PORTE.DT_VIGENCIA DT_VIGENCIA
              ,VAL_PORTE.VL_PORTE VL_PORTE
FROM    dbamv.VAL_PORTE VAL_PORTE
       ,dbamv.POR_ANE POR_ANE
       ,dbamv.TAB_FAT TAB_FAT
WHERE POR_ANE.CD_POR_ANE = VAL_PORTE.CD_POR_ANE
AND   TAB_FAT.CD_TAB_FAT = VAL_PORTE.CD_TAB_FAT
ORDER BY  TAB_FAT.DS_TAB_FAT
)
ORDER BY CODIGO,DESCRICAO,CD_TAB_FAT
