--R_ORCAMENTO_CONTA
SELECT  GRPROC.CD_GRU_PRO
              ,GRU_PRO.DS_GRU_PRO
              ,SUM( NVL(GRPROC.VL_TOTAL_CONTA_HOSP,0) + NVL(GRPROC.VL_TOTAL_CONTA_AMBU,0) )   VALOR
FROM     DBAMV.V_CONTA_GRPROC  GRPROC
              ,DBAMV.GRU_PRO
WHERE  GRPROC.CD_GRU_PRO = GRU_PRO.CD_GRU_PRO
GROUP BY GRPROC.CD_GRU_PRO
                   ,GRU_PRO.DS_GRU_PRO
ORDER BY GRPROC.CD_GRU_PRO
