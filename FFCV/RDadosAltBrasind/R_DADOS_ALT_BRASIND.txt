--R_DADOS_ALT_BRASIND
SELECT  PRO_FAT.CD_PRO_FAT
               ,PRO_FAT.DS_PRO_FAT
               ,PRO_FAT.DS_UNIDADE
               ,B_PRECOS.CD_APRESENTACAO
               ,B_PRECOS.CD_MEDICAMENTO
               ,B_PRECOS.CD_LABORATORIO
               ,B_MEDICAME.DS_MEDICAMENTO
               ,B_MEDICAME.TP_MEDICAMENTO
               ,IMP_BRA.CD_LABORATORIO CD_LABORATORIO2
FROM    DBAMV.IMP_BRA
             ,DBAMV.PRO_FAT
             ,DBAMV.B_PRECOS
             ,DBAMV.B_MEDICAME
             ,DBAMV.GRU_PRO
WHERE  IMP_BRA.CD_PRO_FAT                = PRO_FAT.CD_PRO_FAT
    AND    IMP_BRA.CD_MEDICAMENTO      = B_MEDICAME.CD_MEDICAMENTO
    AND    B_PRECOS.CD_MEDICAMENTO   = B_MEDICAME.CD_MEDICAMENTO
    AND    B_PRECOS.CD_APRESENTACAO = IMP_BRA.CD_APRESENTACAO
    AND    B_PRECOS.CD_IMPORT                = (  SELECT MAX(B.CD_IMPORT) FROM DBAMV.B_PRECOS B
                                                                               WHERE B.CD_MEDICAMENTO   = B_MEDICAME.CD_MEDICAMENTO
                                                                                     AND B.CD_APRESENTACAO = IMP_BRA.CD_APRESENTACAO)
	AND pro_fat.cd_gru_pro = gru_pro.cd_gru_pro
    AND gru_pro.tp_gru_pro IN ('MD','MT')
MINUS
SELECT DISTINCT PRO_FAT.CD_PRO_FAT
               ,PRO_FAT.DS_PRO_FAT
               ,PRO_FAT.DS_UNIDADE
               ,IMP_BRA.CD_APRESENTACAO
               ,IMP_BRA.CD_MEDICAMENTO
               ,IMP_BRA.CD_LABORATORIO
               ,B_MEDICAME.DS_MEDICAMENTO
               ,B_MEDICAME.TP_MEDICAMENTO
               ,IMP_BRA.CD_LABORATORIO CD_LABORATORIO2
FROM    DBAMV.IMP_BRA
             ,DBAMV.PRO_FAT
             ,DBAMV.B_MEDICAME
             ,DBAMV.GRU_PRO
WHERE IMP_BRA.CD_PRO_FAT            =  PRO_FAT.CD_PRO_FAT
    AND   IMP_BRA.CD_MEDICAMENTO =   B_MEDICAME.CD_MEDICAMENTO
    AND pro_fat.cd_gru_pro = gru_pro.cd_gru_pro
    AND gru_pro.tp_gru_pro IN ('MD','MT')
