--R_RESUMO_CONTA_GO
SELECT * FROM (
SELECT ATENDIME.CD_ATENDIMENTO
      ,CARTEIRA.NM_TITULAR
      ,CARTEIRA.NR_CARTEIRA MATRICULA
      ,PACIENTE.NM_PACIENTE
      ,DT_ATENDIMENTO DATA
      ,ATENDIME.DT_ALTA
      ,ATENDIME.CD_CID
      ,CID.DS_CID
      ,GRU_FAT.DS_GRU_FAT
      ,DECODE(GRU_FAT.TP_GRU_FAT,'SH',1,'MD',1,'MT',1,'MM',1,2) tipo
      ,SUM(ITREG_FAT.VL_TOTAL_CONTA) as VL_TOTAL_CONTA
      ,SUM(ITREG_FAT.VL_ACRESCIMO) as VL_ACRESCIMO
      ,LEITO.DS_LEITO
      ,PRESTADOR.DS_CODIGO_CONSELHO
      ,PRESTADOR.NM_PRESTADOR
  FROM DBAMV.ATENDIME,
       DBAMV.PACIENTE,
       DBAMV.CARTEIRA,
       DBAMV.CID,
       DBAMV.REG_FAT,
       DBAMV.ITREG_FAT,
       DBAMV.LEITO,
       DBAMV.PRESTADOR,
       DBAMV.GRU_FAT
 WHERE ATENDIME.CD_PACIENTE    =  PACIENTE.CD_PACIENTE
   AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   AND     ATENDIME.CD_PACIENTE    =  CARTEIRA.CD_PACIENTE(+)
   AND     ATENDIME.CD_CONVENIO    = CARTEIRA.CD_CONVENIO(+)
   AND     ATENDIME.CD_CON_PLA     =  CARTEIRA.CD_CON_PLA(+)
   AND    CID.CD_CID(+)                         =  ATENDIME.CD_CID
   AND    REG_FAT.CD_ATENDIMENTO  = ATENDIME.CD_ATENDIMENTO
   AND    REG_FAT.CD_REG_FAT           = ITREG_FAT.CD_REG_FAT
   AND    GRU_FAT.CD_GRU_FAT(+)      = ITREG_FAT.CD_GRU_FAT
   AND    LEITO.CD_LEITO(+)                   = ATENDIME.CD_LEITO
   AND    PRESTADOR.CD_PRESTADOR(+) = ATENDIME.CD_PRESTADOR
   AND    ATENDIME.CD_ATENDIMENTO = :CAMPO_ATENDIME
   AND    REG_FAT.CD_REG_FAT              = TO_NUMBER ( :CAMPO_CD_REG_FAT )
   AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY
ATENDIME.CD_ATENDIMENTO
      ,CARTEIRA.NM_TITULAR
      ,CARTEIRA.NR_CARTEIRA
      ,PACIENTE.NM_PACIENTE
      ,DT_ATENDIMENTO
      ,ATENDIME.DT_ALTA
      ,ATENDIME.CD_CID
      ,CID.DS_CID
      ,DECODE(GRU_FAT.TP_GRU_FAT,'SH',1,'MD',1,'MT',1,'MM',1,2)
      ,GRU_FAT.DS_GRU_FAT
      ,LEITO.DS_LEITO
      ,PRESTADOR.DS_CODIGO_CONSELHO
      ,PRESTADOR.NM_PRESTADOR
)
ORDER BY NM_TITULAR,MATRICULA,NM_PACIENTE,DATA,DT_ALTA,CD_CID,DS_CID,CD_ATENDIMENTO,NM_PRESTADOR,DS_LEITO,DS_CODIGO_CONSELHO, TIPO, DS_GRU_FAT

--R_RESUMO_CONTA_GO_R_G_TP_GRU_FAT3_Q_2
SELECT * FROM (
SELECT 'Acr/dec. - '||RPAD(GRU_FAT.DS_GRU_FAT,34,'.') de_acres
      ,SUM(VL_ACRESCIMO) vl_acres
,ATENDIME.CD_ATENDIMENTO
  FROM DBAMV.ATENDIME,
       DBAMV.PACIENTE,
       DBAMV.CARTEIRA,
       DBAMV.REG_FAT,
       DBAMV.ITREG_FAT,
       DBAMV.GRU_FAT
 WHERE ATENDIME.CD_PACIENTE          = PACIENTE.CD_PACIENTE
   AND REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
   AND     PACIENTE.CD_PACIENTE           = CARTEIRA.CD_PACIENTE
   AND     ATENDIME.CD_CONVENIO         = CARTEIRA.CD_CONVENIO
   AND     REG_FAT.CD_ATENDIMENTO    = ATENDIME.CD_ATENDIMENTO
   AND     REG_FAT.CD_REG_FAT              = ITREG_FAT.CD_REG_FAT
   AND     GRU_FAT.CD_GRU_FAT             = ITREG_FAT.CD_GRU_FAT
   AND     ATENDIME.CD_ATENDIMENTO = :CAMPO_ATENDIME
   AND     REG_FAT.CD_REG_FAT              = TO_NUMBER ( :CAMPO_CD_REG_FAT )
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY
ATENDIME.CD_ATENDIMENTO
      ,CARTEIRA.NM_TITULAR
      ,CARTEIRA.NR_CARTEIRA
      ,PACIENTE.NM_PACIENTE
      ,DT_ATENDIMENTO
      ,ATENDIME.DT_ALTA
      ,DECODE(GRU_FAT.TP_GRU_FAT,'SH',1,'MD',1,'MT',1,'MM',1,2)
      ,GRU_FAT.DS_GRU_FAT
)
WHERE CD_ATENDIMENTO = :CAMPO_CD_ATENDIMENTO
ORDER BY CD_ATENDIMENTO
