--R_DISSAN
SELECT * FROM (
SELECT  DIS_SAN.CD_DIS_SAN CODIGO
       ,DIS_SAN.NM_DIS_SAN DESCRICAO
       ,CIDADE.CD_CIDADE CD_CIDADE
       ,CIDADE.NM_CIDADE NM_CIDADE
FROM    dbamv.DIS_SAN DIS_SAN
       ,dbamv.CIDADE CIDADE
WHERE CIDADE.CD_CIDADE = DIS_SAN.CD_CIDADE
AND CIDADE.CD_CIDADE LIKE :CAMPO_CIDADE
ORDER BY CIDADE.NM_CIDADE, DIS_SAN.NM_DIS_SAN
)
