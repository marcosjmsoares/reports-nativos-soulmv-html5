--R_TERMO_RESPONSABILIDADE
SELECT NM_USUARIO,
              nvl(CPF,'Não Informado') CPF,
(SELECT cidade.nm_cidade
        FROM multi_empresas, cidade
        WHERE cidade.cd_cidade = multi_empresas.cd_cidade
        AND multi_empresas.cd_multi_empresa = pkg_mv2000.le_empresa) nm_cidade,
              SYSDATE DT_TERMO
              FROM dbasgu.usuarios
              WHERE usuarios.cd_usuario = :CAMPO_CD_USUARIO
