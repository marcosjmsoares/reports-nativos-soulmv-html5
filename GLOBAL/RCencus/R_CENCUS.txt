--R_CENCUS
SELECT  DISTINCT
               GRUPO_DE_CUSTO.CD_GRUPO_DE_CUSTO
              ,GRUPO_DE_CUSTO.DS_GRUPO_DE_CUSTO
FROM DBAMV.GRUPO_DE_CUSTO
          ,DBAMV.SETOR
WHERE GRUPO_DE_CUSTO.CD_GRUPO_DE_CUSTO  =  SETOR.CD_GRUPO_DE_CUSTO
AND       SETOR.CD_MULTI_EMPRESA LIKE :CAMPO_CD_MULTI_EMPRESA
