--R_ORIATE
SELECT  ORI_ATE.CD_ORI_ATE CODIGO
       ,ORI_ATE.DS_ORI_ATE DESCRICAO
       ,ORI_ATE.TP_ORIGEM TIPO
       ,decode(Nvl(ORI_ATE.SN_ATIVO,'S'),'S', 'Sim', 'Não') SN_ATIVO
FROM    dbamv.ORI_ATE ORI_ATE
where ORI_ATE.cd_multi_empresa = nvl(dbamv.pkg_mv2000.le_empresa, ORI_ATE.cd_multi_empresa)
ORDER BY
