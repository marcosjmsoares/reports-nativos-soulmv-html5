--R_CONVENIO
SELECT  CONVENIO.CD_CONVENIO CODIGO
               ,CONVENIO.NM_CONVENIO DESCRICAO
               ,CONVENIO.SN_COBRA_INSUMOS
               ,CONVENIO.TP_CONVENIO TIPO
FROM      DBAMV.CONVENIO
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
ORDER BY
