--R_PAC_D_COMPULSORIA
SELECT * FROM (
SELECT atendime.cd_atendimento                                 cd_atendimento
     , atendime.cd_paciente                                    cd_paciente
     , paciente.nm_paciente                                    nm_paciente
     , cid.cd_cid                                              cd_cid
     , cid.ds_cid                                              ds_cid
     , to_char(atendime.dt_atendimento, 'dd/mm/yyyy')  dt_atendimento
     , to_char(atendime.hr_atendimento, 'hh24')  hr_atendimento
     , To_Char(atendime.dt_alta,  'dd/mm/yyyy')  dt_alta
     , To_Char(atendime.hr_alta,  'hh24')  hr_alta
     , prestador.nm_prestador                                  nm_prestador
     , ori_ate.ds_ori_ate                                           ds_ori_ate
  FROM dbamv.atendime
     , dbamv.cid
     , dbamv.paciente
     , dbamv.prestador
     , dbamv.ori_ate
 WHERE atendime.cd_cid               = cid.cd_cid
   AND atendime.cd_paciente          = paciente.cd_paciente
   AND atendime.cd_ori_ate             = ori_ate.cd_ori_ate
   AND atendime.cd_prestador         = prestador.cd_prestador
   AND cid.sn_notificavel                  = 'S'
   AND atendime.dt_atendimento between :CAMPO_DT_INICIAL and     
   AND atendime.cd_multi_empresa = nvl(dbamv.pkg_mv2000.le_empresa, atendime.cd_multi_empresa)
  :CAMPO_CF_COND_CD_DOENCA
)
ORDER BY CD_CID,DS_CID
