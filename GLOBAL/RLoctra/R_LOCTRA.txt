--R_LOCTRA
SELECT  LOC_TRANS.CD_LOC_TRANS CODIGO
       ,LOC_TRANS.DS_LOC_TRANS DESCRICAO
       ,LOC_TRANS.DS_ENDERECO
       ,LOC_TRANS.NR_ENDERECO
       ,LOC_TRANS.DS_COMPLEMENTO
       ,LOC_TRANS.NM_BAIRRO
       ,CIDADE.CD_UF
       ,CIDADE.NM_CIDADE
       ,LOC_TRANS.NR_CEP
       ,LOC_TRANS.NR_FONE
       ,LOC_TRANS.NR_LOC_TRANS_SUS
FROM    dbamv.LOC_TRANS LOC_TRANS
              ,DBAMV.CIDADE
WHERE LOC_TRANS.CD_CIDADE = CIDADE.CD_CIDADE(+)
ORDER BY
