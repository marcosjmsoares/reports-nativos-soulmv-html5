--R_PAPEL_USUARIOS
SELECT PAPEL.CD_PAPEL ||'-'||PAPEL.DS_PAPEL AS DS_PAPEL, Decode(PAPEL_USUARIOS.TP_PAPEL,'A','AUXILIAR','P','PRINCIPAL')TP_PAPEL, PAPEL_USUARIOS.CD_USUARIO 
FROM DBASGU.PAPEL_USUARIOS ,DBASGU.PAPEL 
WHERE DBASGU.PAPEL_USUARIOS.CD_PAPEL = DBASGU.PAPEL.CD_PAPEL
AND PAPEL_USUARIOS.TP_PAPEL IS NOT NULL 
AND PAPEL_USUARIOS.TP_PAPEL IN   ('A','P')
AND  PAPEL.CD_PAPEL=:CAMPO_CD_PAPEL
ORDER BY PAPEL_USUARIOS.TP_PAPEL DESC ,PAPEL_USUARIOS.CD_USUARIO
