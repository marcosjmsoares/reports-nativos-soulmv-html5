--R_VIEW_MVREPORT
Select Dbvi.View_name Table_name
      ,Dbtc.Comments
  From all_views DBVI
      ,all_tab_comments DBTC
 Where Dbvi.View_name like 'VDIC%'
   And Dbvi.Owner in ('DBAMV','DBASGU')
   And Dbtc.Table_name (+) = Dbvi.View_name
 Order
    By Dbvi.View_name
