--R_USUARIO_SISTEMA
SELECT * FROM (
SELECT sistema.cd_sistema   cd_sistema
              ,sistema.nm_sistema  nm_sistema
              ,usuarios.cd_usuario  cd_usuario
              ,usuarios.nm_usuario nm_usuario
              ,usuarios.cd_papel    cd_papel
              ,papel.ds_papel         ds_papel
              ,modulos.cd_modulo  cd_modulo
              ,modulos.nm_modulo nm_modulo
FROM dbasgu.sistema
          ,dbasgu.mod_sis
          ,dbasgu.modulos
          ,dbasgu.papel_mod
          ,dbasgu.usuarios
          ,dbasgu.papel_usuarios
          ,dbasgu.papel    papel
WHERE sistema.cd_sistema like :CAMPO_CD_SISTEMA
      AND papel.cd_papel like :CAMPO_CD_PAPEL
      AND sistema.cd_sistema = mod_sis.cd_sistema
      AND mod_sis.cd_modulo = modulos.cd_modulo
      AND usuarios.cd_usuario like :CAMPO_CD_USUARIO
      AND usuarios.cd_usuario = papel_usuarios.cd_usuario
      AND papel_usuarios.cd_papel = papel.cd_papel
      AND papel_usuarios.cd_papel = papel_mod.cd_papel
      AND papel_mod.cd_modulo = modulos.cd_modulo
ORDER BY nm_modulo
)
ORDER BY CD_SISTEMA,NM_SISTEMA,
NM_USUARIO,CD_USUARIO,
CD_PAPEL,DS_PAPEL, NM_MODULO, CD_MODULO
