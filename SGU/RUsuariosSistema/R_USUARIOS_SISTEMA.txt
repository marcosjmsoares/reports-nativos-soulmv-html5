--R_USUARIOS_SISTEMA
SELECT * FROM (
select  usuarios.cd_usuario
       ,usuarios.nm_usuario
       ,decode(usuarios.sn_ativo,'S','Ativo','N','Inativo','') sn_ativo
       ,Modulos.cd_sistema_dono Cd_sistema
       ,sistema.nm_sistema
       ,modulos.cd_modulo
       ,modulos.nm_modulo
  from  dbasgu.usuarios
       ,dbasgu.aut_mod
       ,dbasgu.modulos
       ,dbasgu.sistema
  Where aut_mod.cd_usuario          = usuarios.cd_usuario
    and modulos.cd_modulo           = aut_mod.cd_modulo
    and sistema.cd_sistema          = modulos.cd_sistema_dono
Order by usuarios.cd_usuario
        ,sistema.cd_sistema
        ,modulos.nm_modulo
)
