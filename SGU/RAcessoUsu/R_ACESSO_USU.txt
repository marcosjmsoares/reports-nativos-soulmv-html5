--R_ACESSO_USU
SELECT * FROM (
select sistema.nm_sistema, modulos.nm_modulo, usuarios.cd_usuario, usuarios.nm_usuario, modulos.tp_modulo, 'Usuário' Tp_acesso
from dbasgu.sistema,
        dbasgu.modulos,
        dbasgu.usuarios,
        dbasgu.aut_mod,
        dbasgu.mod_sis
where  usuarios.cd_usuario   = aut_mod.cd_usuario
and   aut_mod.cd_modulo     = modulos.cd_modulo
and   modulos.cd_modulo     = mod_sis.cd_modulo(+)
and   mod_sis.cd_sistema    = sistema.cd_sistema(+)
UNION ALL
select sistema.nm_sistema, modulos.nm_modulo, usuarios.cd_usuario, usuarios.nm_usuario, modulos.tp_modulo tp_modulo1, 'Papel' Tp_acesso2
from dbasgu.sistema,
        dbasgu.mod_sis,
        dbasgu.modulos,
        dbasgu.papel_mod,
        dbasgu.papel_usuarios,
        dbasgu.usuarios
where sistema.cd_sistema = mod_sis.cd_sistema
   and mod_sis.cd_modulo = modulos.cd_modulo
   and modulos.cd_modulo = papel_mod.cd_modulo
   and papel_mod.cd_papel = papel_usuarios.cd_papel
   and papel_usuarios.cd_usuario = usuarios.cd_usuario
order by 1, 2, 5
)
ORDER BY CD_USUARIO,NM_USUARIO,
NM_SISTEMA,nm_modulo
