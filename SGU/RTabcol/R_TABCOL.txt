--R_TABCOL
Select Cd_Tabela
          ,Coluna
          ,Tp_Dado
          ,Decode(Sn_Pk,'S','Sim','   ') SN_PK
          ,Decode(Sn_Fk,'S','Sim','   ') SN_FK
          ,Decode(Sn_Insert,'S','Sim','Não') SN_INSERT
          ,Decode(Sn_Update,'S','Sim','Não') SN_UPDATE
          ,Decode(Sn_Delete,'S','Sim','Não') SN_DELETE
 From Dbamv.Colunas
WHERE 1=1
Order By Cd_Tabela Asc
              ,Sn_Pk Desc
              ,Coluna Asc
