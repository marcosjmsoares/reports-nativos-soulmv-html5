--R_PRIVUSU
SELECT * FROM (
Select F.Cd_Usuario, F.Nm_Usuario, U.Created, P.Granted_Role
From Dbasgu.Usuarios F, Sys.All_Users U, Sys.Dba_Role_Privs P
Where F.Cd_Usuario like :CAMPO_CD_USUARIO
And  U.Username = F.Cd_Usuario
And  P.Granted_Role Like :CAMPO_GRANTEE
And P.Grantee = F.Cd_Usuario
Order By F.Nm_Usuario
)
ORDER BY CD_USUARIO,
NM_USUARIO,
CREATED
