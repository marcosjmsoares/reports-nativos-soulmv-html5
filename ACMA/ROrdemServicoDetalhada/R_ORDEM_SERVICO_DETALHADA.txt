--R_ORDEM_SERVICO_DETALHADA
SELECT * FROM (
SELECT Ofi.cd_oficina  cd_oficina1
             , Ofi.ds_oficina ds_oficina1
             , St.nm_setor nm_setor1
             , So.cd_os cd_os1
             , So.dt_pedido dt_pedido1
             , So.ds_servico ds_servico1
             , Tip.ds_tipo_os ds_tipo_os1
             , Bm.Ds_bem ds_bem1
             , Bm.Ds_plaqueta ds_plaqueta1
             , So.dt_execucao dt_execucao1
             , So.dt_prev_exec dt_prev_exec1
             , So.Tp_situacao
             ,decode(tp_situacao,'A','ABERTO','C','CONCLUIDO','N','NÃO ATENDIDO','M','AGUARDANDO MATERIAL','E','CONSERTO EXTERNO','S','SOLICITACAO','L','AGUARDANDO LIB.SETOR','F','A AGENDAR', 'D', 'CANCELADO','') Situacao
             , So.nm_solicitante
             , decode('S','S',(So.dt_execucao - So.dt_pedido) * 24,null) Tempo
FROM Dbamv.Solicitacao_Os So
          , Dbamv.Bens Bm
          , Dbamv.Setor St
         , Dbamv.Oficina Ofi
         , Dbamv.Tipo_Os Tip
WHERE Bm.Cd_Bem (+) = So.cd_bem
      and St.cd_setor   = So.cd_setor
      and ofi.cd_oficina   = So.cd_oficina
      and Tip.cd_tipo_os   = So.cd_tipo_os
     and decode(:CAMPO_OPCAO_PERIODO, 'S', trunc(so.dt_pedido), 'C' , trunc(so.dt_execucao))   BETWEEN :CAMPO_DATA_INI AND (:CAMPO_DATA_FIM)
and ofi.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
     AND so.cd_multi_empresa = dbamv.pkg_mv2000.le_empresa
:CAMPO_CF_SITUACAO
)
ORDER BY CD_OFICINA1,DS_OFICINA1,NM_SETOR1
