--R_ABERTURAOS_FCF
SELECT * FROM (
SELECT
	L_SETOR.NM_SETOR 			NM_SETOR,
  	L_SETOR.CD_SETOR 			CD_SETOR,
  	L_MANU_ESPEC.DS_ESPEC 		DS_ESPEC,
  	L_MANU_ESPEC.CD_ESPEC 		CD_ESPEC,
  	OS.NM_USUARIO 				NM_USUARIO,
  	OS.NM_SOLICITANTE 			NM_SOLICITANTE,
  	OS.DT_PEDIDO 				DT_PEDIDO,
  	OS.DS_SERVICO 				DS_SERVICO,
  	OS.DS_OBSERVACAO 			DS_OBSERVACAO,
  	OS.CD_OS                    CD_OS,
  	OS.TP_SITUACAO                      		TP_SITUACAO,
  	MANU_SERV.NM_SERVICO 	                	NM_SERVICO,
  	MANU_SERV.CD_SERVICO 	                	CD_SERVICO,
  	FUNCIONARIO.NM_FUNC                     NM_FUNC,
  	FUNCIONARIO.CD_FUNC 		            CD_FUNC,
  	ITSOLICITACAO_OS.VL_TEMPO_GASTO 	VL_TEMPO_GASTO,
  	ITSOLICITACAO_OS.HR_INICIO          HR_INICIO,
  	ITSOLICITACAO_OS.HR_FINAL 		HR_FINAL,
  	ITSOLICITACAO_OS.CD_ITSOLICITACAO_OS 	CD_ITSOLICITACAO_OS
FROM
	DBAMV.SETOR               	L_SETOR,
	DBAMV.MANU_ESPEC          L_MANU_ESPEC,
	DBAMV.SOLICITACAO_OS    OS,
	DBAMV.MANU_SERV,
	DBAMV.FUNCIONARIO,
	DBAMV.ITSOLICITACAO_OS
WHERE 		OS.CD_ESPEC                 		= L_MANU_ESPEC.CD_ESPEC
  	AND 	OS.CD_SETOR                 		= L_SETOR.CD_SETOR
	AND 	ITSOLICITACAO_OS.CD_FUNC 	= FUNCIONARIO.CD_FUNC
  	AND 	ITSOLICITACAO_OS.CD_SERVICO 	= MANU_SERV.CD_SERVICO
  	AND	OS.CD_OS		                = ITSOLICITACAO_OS.CD_OS
  	AND 	OS.DT_PEDIDO                  		BETWEEN NVL(:CAMPO_DT_INICIO, TO_DATE('01/01/1900', 'DD/MM/YYYY')) AND NVL(:CAMPO_DT_FIM,TO_DATE('01/01/3000', 'DD/MM/YYYY'))
  :CAMPO_CF_ESPECIALIDADE
)
ORDER BY CD_SETOR,NM_SETOR,CD_ESPEC,DS_ESPEC,NM_USUARIO,NM_SOLICITANTE,DT_PEDIDO,CD_OS,DS_SERVICO,TP_SITUACAO
