--R_PLANO_BEM_PROJECT
SELECT :CAMPO_CF_DS_BEM || cd_bem || ' ' ||
       :CAMPO_CF_DS_MARCA || ds_marca || ' ' ||
       :CAMPO_CF_DS_MODELO || ds_modelo || ' ' ||
       :CAMPO_CF_DS_SERIE || nr_serie || ' ' ||
       :CAMPO_CF_DS_ETIQUETA || ds_plaqueta || ' ' ||
       :CAMPO_CF_DS_PLANO || ds_plano || ' ' ||
       :CAMPO_CF_DS_OFICINA || ds_oficina ||
       ';' ||
       To_Char(ULTIMA_MANUTENCAO, 'dd/mm/yyyy') || Nvl2(ULTIMA_MANUTENCAO, ';', '') ||
       qt_horas_execucao || Nvl2(qt_horas_execucao, ' hrs;', '') AS PLANO_BEM_PROJECT
  FROM (
SELECT  bem.cd_bem
      , bem.ds_marca
      , bem.ds_modelo
      , bem.nr_serie
      , bem.ds_plaqueta
      , pla.ds_plano
      , ofi.ds_oficina
      , prg.dt_ultima_manutencao Ultima_Manutencao
      , plo.qt_horas_execucao
FROM dbamv.programacao_plano prg
   , dbamv.plano_manutencao pla
   , dbamv.plano_manutencao_oficina plo
   , dbamv.oficina ofi
   , dbamv.bens bem
   , dbamv.localidade loc
   , dbamv.setor setor
WHERE prg.cd_plano = pla.cd_plano
  AND pla.cd_plano = plo.cd_plano
  AND plo.cd_oficina = ofi.cd_oficina
  AND prg.cd_bem = bem.cd_bem
  AND prg.cd_localidade = loc.cd_localidade
  AND bem.cd_setor = Setor.cd_setor
  AND (         ( :CAMPO_OPCAO_PERIODO = 'S' )
             OR  ( :CAMPO_OPCAO_PERIODO = 'U'  AND Prg.dt_Ultima_Manutencao BETWEEN to_date( to_char(:CAMPO_DATA_INI,'dd/mm/yyyy')|| ' 00','dd/mm/yyyy hh24')
                                                                                                                                     AND to_date( to_char(:CAMPO_DATA_FIM,'dd/mm/yyyy')|| ' 23','dd/mm/yyyy hh24')   )
             OR  ( :CAMPO_OPCAO_PERIODO = 'P'  AND Prg.dt_Ultima_Geracao BETWEEN to_date( to_char(:CAMPO_DATA_INI,'dd/mm/yyyy')|| ' 00','dd/mm/yyyy hh24')
                                                                                                                                     AND to_date( to_char(:CAMPO_DATA_FIM,'dd/mm/yyyy')|| ' 23','dd/mm/yyyy hh24')   )
           )
  AND (       ( :CAMPO_TIPO_PLANO <> 'P' )
            OR (Prg.Qt_Dias = :CAMPO_PERIODICIDADE )
           )
ORDER BY bem.cd_bem
)
ORDER BY CD_BEM,DS_PLAQUETA
