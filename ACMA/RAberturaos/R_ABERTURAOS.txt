--R_ABERTURAOS
SELECT L_SETOR.NM_SETOR L_SETOR_NM_SETOR,
               L_SETOR.CD_SETOR L_SETOR_CD_SETOR,
               L_MANU_ESPEC.DS_ESPEC L_MANU_ESPEC_DS_ESPEC,
               L_MANU_ESPEC.CD_ESPEC L_MANU_ESPEC_CD_ESPEC,
               USUARIOS.NM_USUARIO NM_USUARIO,
               OS.NM_SOLICITANTE NM_SOLICITANTE,
               OS.DT_PEDIDO DT_PEDIDO,
               OS.DS_SERVICO DS_SERVICO,
               OS.DS_OBSERVACAO DS_OBSERVACAO,
               OS.CD_SETOR CD_SETOR,
               OS.CD_OS CD_OS1,
               OS.CD_ESPEC CD_ESPEC,
               OS.TP_SITUACAO TP_SITUACAO,
               DECODE(OS.SN_PACIENTE,'S','Sim','Não') SN_PACIENTE,
               LOC.DS_LOCALIDADE,
               substr(tipo_os.ds_tipo_os, 1 , 20) ds_tipo_os,
               OFICINA.CD_OFICINA   ,
               OFICINA.DS_OFICINA,
               BENS.CD_BEM,
               BENS.DS_BEM
FROM     DBAMV.SETOR                       L_SETOR,
               DBAMV.MANU_ESPEC           L_MANU_ESPEC,
               DBAMV.SOLICITACAO_OS     OS,
               DBAMV.LOCALIDADE             LOC,
               DBAMV.TIPO_OS                    TIPO_OS,
               DBAMV.OFICINA                      OFICINA,
               DBAMV.BENS                          BENS,
               DBASGU.USUARIOS
WHERE OS.CD_ESPEC            = L_MANU_ESPEC.CD_ESPEC (+)
      AND OS.CD_SETOR           = L_SETOR.CD_SETOR
      AND OS.CD_LOCALIDADE = LOC.CD_LOCALIDADE(+)
      AND OS.cd_tipo_os             = tipo_os.cd_tipo_os
      AND OS.CD_OFICINA         = OFICINA.CD_OFICINA(+)
      AND OS.CD_BEM               = BENS.CD_BEM(+)
      AND OS.NM_SOLICITANTE = USUARIOS.CD_USUARIO (+)
      AND OFICINA.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
ORDER BY OS.CD_OS, os.nm_solicitante
