--R_GASTO_MANUTENCAO_SINT
Select *
From (Select Distinct SOS.CD_OS, SOS.TP_SITUACAO, SOS.DS_SERVICO, SOS.TP_LOCAL, SOS.DT_EXECUCAO, SOS.CD_SETOR, S.NM_SETOR,
T.DS_TIPO_OS, T.CD_TIPO_OS, SOS.CD_OFICINA, (Select To_Char(Nvl(Sum(ISOS.VL_TEMPO_GASTO), 0) +
Trunc(Nvl(Sum(ISOS.VL_TEMPO_GASTO_MIN), 0) / 60)) || ':' || To_Char(Mod(Nvl(Sum(ISOS.VL_TEMPO_GASTO_MIN), 0), 60), 'FM00') From
DBAMV.ITSOLICITACAO_OS ISOS Where ISOS.CD_OS = SOS.CD_OS) TEMPO
From DBAMV.SOLICITACAO_OS SOS, DBAMV.TIPO_OS T, DBAMV.SETOR S 
Where SOS.CD_OS Is Not Null And SOS.CD_TIPO_OS = T.CD_TIPO_OS And Trunc(SOS.DT_PEDIDO) Between :CAMPO_DTINI And  And
SOS.CD_SETOR = S.CD_SETOR     
:CAMPO_CF_WHERE_TP_SITUACAO)
Order By CD_SETOR, NM_SETOR, DS_TIPO_OS, CD_TIPO_OS
