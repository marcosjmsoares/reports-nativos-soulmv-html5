--R_GASTO_MANUTENCAO_IOB
SELECT * FROM (
SELECT  IOB.CD_CAPITULO
       ,IOB.DS_CAPITULO
       ,SUB_IOB.CD_SUB_CAP
       ,SUB_IOB.DS_SUB_CAP
       ,SOS.CD_OS
       ,SOS.DS_SERVICO
      ,SOS.TP_LOCAL
FROM    DBAMV.IOB
       ,DBAMV.BENS B
       ,DBAMV.SOLICITACAO_OS SOS
       ,DBAMV.SUB_IOB
WHERE   B.CD_CAPITULO = IOB.CD_CAPITULO
  AND   B.CD_CAPITULO = SUB_IOB.CD_CAPITULO  
  AND   B.CD_BEM = SOS.CD_BEM
  AND   B.CD_SUB_CAP = SUB_IOB.CD_SUB_CAP
  AND   SOS.CD_OS IS NOT NULL
  AND   SOS.TP_SITUACAO = 'C'
:CAMPO_CF_CD_SUB_CAP
)
ORDER BY CD_CAPITULO,
CD_SUB_CAP, DS_SERVICO
