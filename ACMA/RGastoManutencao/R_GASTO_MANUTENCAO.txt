SELECT ST.CD_SETOR
      ,ST.NM_SETOR
      ,L.CD_LOCALIDADE
      ,L.DS_LOCALIDADE
      ,SOS_F.CD_OS
      ,SOS_F.DS_SERVICO
      ,SOS_F.TP_SITUACAO TP_SITUACAO
      ,SOS_F.DT_PEDIDO
      ,decode(SOS_F.TP_LOCAL, 'I', 'Interno',
                            'E', 'Externo') tp_local
  From Dbamv.Solicitacao_Os Sos
     , Dbamv.Solicitacao_Os Sos_F
     , DBAMV.SETOR ST
     , DBAMV.LOCALIDADE L
     , dbamv.itsolicitacao_ext ite
 Where Sos.Cd_Os           = Sos_F.Cd_Ordem_Servico_Principal
   AND SOS_F.CD_SETOR      = ST.CD_SETOR
   AND SOS_F.CD_LOCALIDADE = L.CD_LOCALIDADE(+)
   AND trunc(SOS_F.DT_PEDIDO) BETWEEN :CAMPO_DTINI AND 
   and :CAMPO_SN_OS_RELACIONADA = 'S'
   And Sos_F.Cd_Os = Ite.Cd_Os(+)
and sos_f.cd_os = 6522
Union All
SELECT ST.CD_SETOR
      ,ST.NM_SETOR
      ,L.CD_LOCALIDADE
      ,L.DS_LOCALIDADE
      ,SOS.CD_OS
      ,SOS.DS_SERVICO
      ,SOS.TP_SITUACAO TP_SITUACAO
      ,SOS.DT_PEDIDO
      ,decode(SOS.TP_LOCAL, 'I', 'Interno',
                            'E', 'Externo') tp_local
  From Dbamv.Solicitacao_Os Sos
     , DBAMV.SETOR ST
     , DBAMV.LOCALIDADE L
 Where SOS.CD_SETOR         = ST.CD_SETOR
   AND SOS.CD_LOCALIDADE    = L.CD_LOCALIDADE(+)
   AND trunc(SOS.DT_PEDIDO) BETWEEN :CAMPO_DTINI
