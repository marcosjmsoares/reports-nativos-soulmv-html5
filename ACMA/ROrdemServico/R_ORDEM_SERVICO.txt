--R_ORDEM_SERVICO
select sol.cd_os,
       sol.ds_servico,
       sol.dt_pedido,
       sol.dt_execucao,
       sol.cd_espec,
       manu_espec.ds_espec,
       decode(sol.tp_situacao,'A', 'ABERTO',
                              'C','CONCLUÍDO',
                              'N','NÃO ATENDIDO',
                              'M','AGUARDANDO MATERIAL',
                              'E','CONSERTO EXTERNO',
                              'S','SOLICITAÇÃO',
							  'D','CANCELADA'
								) situacao,
       sol.cd_tipo_os,
       tipo_os.ds_tipo_os,
       sol.nm_solicitante,
       sol.cd_setor,
       setor.nm_setor,
       localidade.ds_localidade,
       sol.ds_observacao,
       itsol.cd_servico,
       manu_serv.nm_servico,
       itsol.cd_func,
       func.nm_func,
       itsol.hr_inicio,
       itsol.hr_final,
       itsol.vl_tempo_gasto,
       itsol.vl_tempo_gasto_min,
       itsol.ds_servico ds_serv_real,
       bens.cd_bem,
       bens.ds_bem,
       bens.ds_marca,
       bens.ds_modelo,
       bens.nr_serie,
       bens.ds_plaqueta,
     Sol.Cd_Ordem_servico_principal Cd_Ord_ser_pri
from dbamv.solicitacao_os sol,
     dbamv.itsolicitacao_os itsol,
     dbamv.manu_espec,
     dbamv.tipo_os,
     dbamv.setor,
     dbamv.manu_serv,
     dbamv.funcionario func,
     dbamv.localidade,
     dbamv.bens
where TRUNC(dt_pedido) between :CAMPO_DT_INI and  and
      itsol.cd_os(+) = sol.cd_os and
      sol.cd_espec = manu_espec.cd_espec(+) and
      sol.cd_tipo_os = tipo_os.cd_tipo_os and
      sol.cd_setor = setor.cd_setor and
      itsol.cd_servico = manu_serv.cd_servico(+) and
      itsol.cd_func = func.cd_func(+) and
      sol.cd_localidade = localidade.cd_localidade(+) and
      sol.cd_bem  = bens.cd_bem(+)
     AND sol.cd_multi_empresa = dbamv.pkg_mv2000.le_empresa
order by sol.cd_os
