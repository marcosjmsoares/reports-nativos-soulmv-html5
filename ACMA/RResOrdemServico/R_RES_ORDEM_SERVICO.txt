--R_RES_ORDEM_SERVICO
SELECT DISTINCT SETOR.CD_SETOR Cd_Setor
        ,SETOR.NM_SETOR NM_Setor
       ,SOLICITACAO_OS.CD_OS Cd_os
       , Solicitacao_Os.Ds_Servico Ds_Servico
       ,SUB_CLAS.CD_SUB_CLA Cd_Sub_Cla
       ,SUB_CLAS.DS_SUB_CLA Ds_Sub_Cla
       ,BENS.CD_BEM Cd_Bem
       ,BENS.DS_MODELO Ds_Modelo
       ,BENS.DS_MARCA Ds_Marca
       ,BENS.DS_BEM Ds_Bem
       ,BENS.DS_PLAQUETA Ds_Plaqueta
       ,SOLICITACAO_OS.DT_PEDIDO Dt_Pedido
       ,SOLICITACAO_OS.CD_ORDEM_SERVICO_PRINCIPAL Cd_Ord_ser_pri
  FROM DBAMV.TIPO_OS
       ,DBAMV.SOLICITACAO_OS
       ,DBAMV.ITSOLICITACAO_OS
       ,DBAMV.OFICINA
       ,DBAMV.BENS
       ,DBAMV.ESPECIE
       ,DBAMV.CLASSE
       ,DBAMV.SUB_CLAS
       ,DBAMV.SETOR
WHERE SOLICITACAO_OS.CD_OS      = ITSOLICITACAO_OS.CD_OS (+)
  AND SOLICITACAO_OS.CD_TIPO_OS = TIPO_OS.CD_TIPO_OS
  AND SOLICITACAO_OS.CD_OFICINA = OFICINA.CD_OFICINA(+)
  AND SOLICITACAO_OS.CD_BEM     = BENS.CD_BEM(+)
  AND BENS.CD_SUB_CLA           = SUB_CLAS.CD_SUB_CLA(+)
  AND BENS.CD_CLASSE            = SUB_CLAS.Cd_CLASSE(+)
  AND BENS.CD_ESPECIE           = SUB_CLAS.CD_ESPECIE(+)
  AND SUB_CLAS.CD_CLASSE        = CLASSE.CD_CLASSE(+)
  AND SUB_CLAS.CD_ESPECIE       = CLASSE.CD_ESPECIE(+)
  AND CLASSE.CD_ESPECIE         = ESPECIE.CD_ESPECIE(+)
  AND SETOR.CD_SETOR            = SOLICITACAO_OS.CD_SETOR
  AND TRUNC(SOLICITACAO_OS.DT_PEDIDO) BETWEEN :CAMPO_DT_INI AND 
  AND SOLICITACAO_OS.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
 and oficina.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
ORDER BY DT_PEDIDO, CD_SETOR
