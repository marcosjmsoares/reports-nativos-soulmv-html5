--R_CONSUMO_ENERGIA_ELETRICA
Select To_Char(A.DT_REFERENCIA, 'YYYYMM') || LPad(To_Char(RowNum), 2, '0') As MES_ANO_REFERENCIA, A.VL_TOTAL, Decode(:CAMPO_S_VALOR_ESCOLHIDO, '1', (Select B.QT_ITEM_CONSUMO From
DBAMV.ITEM_CONSUMO B Where B.CD_CONSUMO = A.CD_CONSUMO And B.CD_TIPO_CONSUMO = :CAMPO_N_CD_TIPO_CONSUMO), (Select B.VL_ITEM_CONSUMO From DBAMV.ITEM_CONSUMO B
Where B.CD_CONSUMO = A.CD_CONSUMO And B.CD_TIPO_CONSUMO = :CAMPO_N_CD_TIPO_CONSUMO)) As VALOR_OPCIONAL, (Select C.DS_TIPO_CONSUMO From DBAMV.TIPO_CONSUMO C
Where C.CD_TIPO_CONSUMO = :CAMPO_N_CD_TIPO_CONSUMO) As DESCRICAO_TIPO_CONSUMO, D.DS_UNIDADE_CONSUMIDORA, ' ' As ROTULO_PADRAO From DBAMV.UNIDADE_CONSUMIDORA D, DBAMV.CONSUMO A
Where A.CD_UNIDADE_CONSUMIDORA = :CAMPO_N_CD_UNIDADE_CONSUMIDORA And To_Char(A.DT_REFERENCIA, 'YYYYMM') Between To_Char(Add_Months(:CAMPO_DT_REFERENCIA, -11), 'YYYYMM') And To_Char(:CAMPO_DT_REFERENCIA, 'YYYYMM') And A.CD_UNIDADE_CONSUMIDORA = D.CD_UNIDADE_CONSUMIDORA Order By 1
