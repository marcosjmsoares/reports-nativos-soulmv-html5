--R_BEM_CONTRATO_MANUT
SELECT * FROM (
select  contrato_manutencao.cd_contrato_manutencao
       ,contrato_manutencao.ds_contrato_manutencao
       ,contrato_manutencao.cd_fornecedor
       ,fornecedor.nm_fantasia
       ,contrato_manutencao.dt_inicio
       ,contrato_manutencao.dt_termino
       ,contrato_manutencao.cd_usuario_responsavel
       ,contrato_manutencao.tp_situacao
       ,contrato_manutencao.vl_contrato_manutencao
       ,contrato_manutencao.nr_dia_vencimento
       ,contrato_manutencao.tp_periodo_pagamento
       ,contrato_manutencao.ds_clausula
       ,itcontrato_manutencao.cd_bem
       ,bens.ds_bem
       ,itcontrato_manutencao.dt_inicio dt_inicio_bem
  from  dbamv.contrato_manutencao
       ,dbamv.itcontrato_manutencao
       ,dbamv.fornecedor
       ,dbamv.bens
 where contrato_manutencao.cd_contrato_manutencao = itcontrato_manutencao.cd_contrato_manutencao(+)
   and contrato_manutencao.cd_fornecedor = fornecedor.cd_fornecedor
   and itcontrato_manutencao.cd_bem = bens.cd_bem (+)
   and contrato_manutencao.cd_multi_empresa =  :CAMPO_CD_MULTI_EMPRESA
:CAMPO_CF_BEMM
)
ORDER BY CD_CONTRATO_MANUTENCAO,DS_CONTRATO_MANUTENCAO,CD_FORNECEDOR,NM_FANTASIA,DT_INICIO,DT_TERMINO,CD_USUARIO_RESPONSAVEL,TP_SITUACAO,VL_CONTRATO_MANUTENCAO,NR_DIA_VENCIMENTO,TP_PERIODO_PAGAMENTO
