--R_TEMPO_SERVICO_EXE
SELECT * FROM (
SELECT T.cd_multi_empresa, T.cd_ano, T.cd_mes, T.cd_func, F.nm_func
          ,Sum(T.nr_hora_01) Dia01, Sum(T.nr_hora_02) Dia02, Sum(T.nr_hora_03) Dia03, Sum(T.nr_hora_04) Dia04, Sum(T.nr_hora_05) Dia05
          ,Sum(T.nr_hora_06) Dia06, Sum(T.nr_hora_07) Dia07, Sum(T.nr_hora_08) Dia08, Sum(T.nr_hora_09) Dia09, Sum(T.nr_hora_10) Dia10
          ,Sum(T.nr_hora_11) Dia11, Sum(T.nr_hora_12) Dia12, Sum(T.nr_hora_13) Dia13, Sum(T.nr_hora_14) Dia14, Sum(T.nr_hora_15) Dia15
          ,Sum(T.nr_hora_16) Dia16, Sum(T.nr_hora_17) Dia17, Sum(T.nr_hora_18) Dia18, Sum(T.nr_hora_19) Dia19, Sum(T.nr_hora_20) Dia20
          ,Sum(T.nr_hora_21) Dia21, Sum(T.nr_hora_22) Dia22, Sum(T.nr_hora_23) Dia23, Sum(T.nr_hora_24) Dia24, Sum(T.nr_hora_25) Dia25
          ,Sum(T.nr_hora_26) Dia26, Sum(T.nr_hora_27) Dia27, Sum(T.nr_hora_28) Dia28, Sum(T.nr_hora_29) Dia29, Sum(T.nr_hora_30) Dia30, Sum(T.nr_hora_31) Dia31
          ,Sum( T.nr_hora_01 + T.nr_hora_02 + T.nr_hora_03 + T.nr_hora_04 + T.nr_hora_05 + T.nr_hora_06 + T.nr_hora_07 + T.nr_hora_08 + T.nr_hora_09 + T.nr_hora_10
              + T.nr_hora_11 + T.nr_hora_12 + T.nr_hora_13 + T.nr_hora_14 + T.nr_hora_15 + T.nr_hora_16 + T.nr_hora_17 + T.nr_hora_18 + T.nr_hora_19 + T.nr_hora_20
              + T.nr_hora_21 + T.nr_hora_22 + T.nr_hora_23 + T.nr_hora_24 + T.nr_hora_25 + T.nr_hora_26 + T.nr_hora_27 + T.nr_hora_28 + T.nr_hora_29 + T.nr_hora_30 + T.nr_hora_31
              ) Total_Func
FROM dbamv.Tempo_Execucao_Servicos T, dbamv.funcionario F
WHERE T.cd_func = F.cd_func
      AND T.cd_ano = To_Number(TO_CHAR(:CAMPO_DATA_INI,'YYYY'))
      AND T.cd_mes = To_Number(TO_CHAR(:CAMPO_DATA_INI,'MM'))
      AND T.cd_usuario_logado = USER
      AND T.cd_multi_empresa = dbamv.pkg_mv2000.le_empresa
GROUP BY T.cd_multi_empresa, T.cd_ano, T.cd_mes, T.cd_func, F.nm_func
ORDER BY F.nm_func
)
ORDER BY NM_FUNC,CD_MULTI_EMPRESA,CD_ANO,CD_MES,CD_FUNC
