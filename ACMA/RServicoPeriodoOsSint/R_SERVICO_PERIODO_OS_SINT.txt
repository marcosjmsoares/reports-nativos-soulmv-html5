--R_SERVICO_PERIODO_OS_SINT
select solicitacao_os.cd_tipo_os,
      tipo_os.ds_tipo_os,
       count(*) qtd,
     sum(vl_tempo_gasto) horas,
     sum(vl_tempo_gasto_min) minutos,
     sum(vl_tempo_gasto) + (sum(vl_tempo_gasto_min)/60) tempo_total
from dbamv.itsolicitacao_os,
     dbamv.solicitacao_os,
     dbamv.tipo_os,
     dbamv.manu_serv
where trunc(hr_final) between :CAMPO_DTINI and   and
      tipo_os.cd_tipo_os  = solicitacao_os.cd_tipo_os and
      solicitacao_os.cd_os = itsolicitacao_os.cd_os and
      manu_serv.cd_servico = itsolicitacao_os.cd_servico
     AND solicitacao_os.cd_multi_empresa = dbamv.pkg_mv2000.le_empresa
group by solicitacao_os.cd_tipo_os,
      tipo_os.ds_tipo_os
