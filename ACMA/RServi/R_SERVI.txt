--R_SERVI
select cd_servico,
       nm_servico,
       peso,
       qtd_hora_min,
       qtd_hora_max,
      Case When Manu_Serv.Tp_Categoria = '1' Then 'Urgente'
               When Manu_Serv.Tp_Categoria = '2' Then 'Não Urgente'
               When Manu_Serv.Tp_Categoria = '3' Then 'Específico de Terceiros'
               When Manu_Serv.Tp_Categoria = '4' Then 'Calibragem'
               When Manu_Serv.Tp_Categoria = '5' Then 'Check List'
       End As Tp_Categoria
from dbamv.manu_serv
