--R_ORDEM_SERVICO_DETALHAD2
SELECT * FROM (
select Ofi.cd_oficina  cd_oficina
     , Ofi.ds_oficina
     , St.nm_setor
     , Lc.ds_localidade
     , So.cd_os
     , So.dt_pedido
     , So.ds_servico
     , Tip.ds_tipo_os
     , Bm.Ds_bem
     , Bm.Ds_plaqueta
     , So.dt_execucao
     , So.dt_prev_exec
     , So.Tp_situacao
     ,decode(tp_situacao,'A','ABERTO','C','CONCLUIDO','N','NÃO ATENDIDO','M','AGUARDANDO MATERIAL','E','CONSERTO EXTERNO','S','SOLICITACAO','L','AGUARDANDO LIB.SETOR','F','A AGENDAR') Situacao
     , So.nm_solicitante
     , decode(:CAMPO_IMPRIME_TEMPO,'S',(So.dt_execucao - So.dt_pedido),null) Tempo
     , Func.Nm_Func
    from Dbamv.Solicitacao_Os So
     , Dbamv.Bens Bm
     , Dbamv.Setor St
     , Dbamv.Localidade Lc
     , Dbamv.Oficina Ofi
     , Dbamv.Tipo_Os Tip
     , (select distinct Os.Cd_Os
                , Fu.Nm_Func
          from Dbamv.Itsolicitacao_Os Os
                , Dbamv.Funcionario Fu
        Where Fu.cd_func = Os.Cd_Func) Func
 where Bm.Cd_Bem (+) = So.cd_bem
   and Func.Cd_Os (+) = So.Cd_Os
   and St.cd_setor   = So.cd_setor
   and Lc.cd_setor   = St.cd_setor
   and So.Cd_localidade = Lc.cd_localidade
   and ofi.cd_oficina   = So.cd_oficina
   and Tip.cd_tipo_os   = So.cd_tipo_os
   and decode(:CAMPO_OPCAO_PERIODO, 'S', so.dt_pedido , 'C' , so.dt_execucao)   BETWEEN :CAMPO_DATA_INI AND (:CAMPO_DATA_FIM + 1)
and ofi.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
:CAMPO_CF_SITUACAO
)
ORDER BY CD_OFICINA,DS_OFICINA,
NM_SETOR,
TEMPO,NM_SOLICITANTE,CD_OS,DT_PEDIDO,DS_SERVICO,DS_TIPO_OS,DS_BEM,DS_PLAQUETA,DT_EXECUCAO,DT_PREV_EXEC,SITUACAO,DS_LOCALIDADE
