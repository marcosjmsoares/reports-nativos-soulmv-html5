--R_SOL_EXTERNA
Select  os.cd_os
       ,os.dt_pedido
       ,os.nm_solicitante
       ,usuarios.nm_usuario
       ,os.cd_tipo_os
       ,tipo.tp_preliminar
       ,tipo.ds_tipo_os
       ,os.cd_setor
       ,setor.nm_setor
       ,nvl(os.cd_localidade, 0) cd_localidade
       ,nvl(loc.ds_localidade, 'NÃO ESPECIFICADA') ds_localidade
       ,nvl(os.cd_espec,0) cd_espec
       ,nvl(espec.ds_espec,'NÃO ESPECIFICADA') ds_espec
       ,os.ds_observacao
       ,os.ds_servico
       ,DECODE(OS.SN_PACIENTE,'S','Sim','Não') SN_PACIENTE
       ,bens.cd_bem
       ,bens.ds_bem
       ,os.tp_situacao
 From dbamv.solicitacao_os os
     ,dbamv.tipo_os tipo
     ,dbamv.setor
     ,dbamv.localidade loc
     ,dbamv.manu_espec espec
     ,dbamv.bens
     ,dbasgu.usuarios
where os.cd_os = :CAMPO_OS
  and os.nm_solicitante = usuarios.cd_usuario (+)
  and os.cd_bem = bens.cd_bem (+)
  and tipo.cd_tipo_os   = os.cd_tipo_os
  and setor.cd_setor    = os.cd_setor
  and loc.cd_localidade (+) = os.cd_localidade
  and espec.cd_espec (+)    = os.cd_espec
