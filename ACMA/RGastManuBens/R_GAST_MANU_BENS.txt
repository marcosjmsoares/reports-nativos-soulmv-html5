--R_GAST_MANU_BENS
SELECT * FROM (
SELECT B.CD_BEM
      ,B.DS_PLAQUETA
      ,B.DS_MARCA
      ,B.DS_MODELO
      ,B.DS_BEM
      ,B.CD_SETOR
      ,ST.NM_SETOR
      ,B.CD_LOCALIDADE
      ,L.DS_LOCALIDADE
      ,SOS.CD_OS
      ,SOS.DS_SERVICO
      ,SOS.TP_LOCAL
FROM DBAMV.SOLICITACAO_OS SOS
    ,DBAMV.BENS B
    ,DBAMV.SETOR ST
    ,DBAMV.LOCALIDADE L
WHERE SOS.CD_BEM                 = B.CD_BEM
     AND B.CD_SETOR              = ST.CD_SETOR
     AND B.CD_LOCALIDADE         = L.CD_LOCALIDADE
     AND SOS.CD_OS IS NOT NULL
    AND SOS.TP_SITUACAO = 'C'
     AND sos.cd_multi_empresa = dbamv.pkg_mv2000.le_empresa
:CAMPO_CF_CD_BEM
)
ORDER BY CD_BEM,DS_BEM,CD_SETOR,NM_SETOR,CD_LOCALIDADE,DS_LOCALIDADE,DS_PLAQUETA,DS_MARCA,DS_MODELO
