--R_SERVICO_PERIODO_ESPE
SELECT * FROM (
select NVL(solicitacao_os.cd_espec,0) CD_ESPEC,
       NVL(manu_espec.ds_espec,'NÃO INFORMADA') DS_ESPEC,
       itsolicitacao_os.cd_servico,
       manu_serv.nm_servico,
       count(*) qtd,
       sum(vl_tempo_gasto) horas,
       sum(vl_tempo_gasto_min) minutos,
       sum((vl_tempo_gasto*60) + vl_tempo_gasto_min) total_horas_minutos
from dbamv.itsolicitacao_os,
     dbamv.solicitacao_os,
     dbamv.manu_espec,
     dbamv.manu_serv
where trunc(hr_final) between :CAMPO_DTINI and  and
      manu_espec.cd_espec(+)  = solicitacao_os.cd_espec and
      solicitacao_os.cd_os = itsolicitacao_os.cd_os and
      manu_serv.cd_servico = itsolicitacao_os.cd_servico
     AND solicitacao_os.cd_multi_empresa = dbamv.pkg_mv2000.le_empresa
group by solicitacao_os.cd_espec,
       manu_espec.ds_espec,
       itsolicitacao_os.cd_servico,
       manu_serv.nm_servico
)
ORDER BY DS_ESPEC,CD_ESPEC, NM_SERVICO
