--R_SERVICO_PERIODO_FUNC
SELECT * FROM (
select itsolicitacao_os.cd_func,
       funcionario.nm_func,
       itsolicitacao_os.cd_servico,
       manu_serv.nm_servico,
       Nvl(sum(vl_tempo_gasto),0) horas,
       Nvl(sum(vl_tempo_gasto_min),0) minutos,
       Nvl(sum(Decode(vl_tempo_gasto,0,(vl_tempo_gasto_min * 1),((vl_tempo_gasto * 60) + vl_tempo_gasto_min))),0) tempo_total,
       count(*) qtd
from dbamv.itsolicitacao_os,
     dbamv.solicitacao_os,
     dbamv.funcionario,
     dbamv.manu_serv
where trunc(hr_final) between :CAMPO_DTINI and  and
      funcionario.cd_func = itsolicitacao_os.cd_func and
      solicitacao_os.cd_os = itsolicitacao_os.cd_os and
      manu_serv.cd_servico = itsolicitacao_os.cd_servico
group by itsolicitacao_os.cd_func,
       funcionario.nm_func,
       itsolicitacao_os.cd_servico,
       manu_serv.nm_servico
order by funcionario.nm_func,
       manu_serv.nm_servico
)
ORDER BY NM_FUNC
