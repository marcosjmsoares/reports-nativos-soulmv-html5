--R_TERMO_ABERTURAOS
SELECT * FROM (
SELECT S.CD_OS ORDEM_SERVICO
      ,DT_EXECUCAO DATA_PROGRAMADA
      ,S.CD_BEM||' - '||BENS.DS_BEM EQUIPAMENTO
      ,NM_SOLICITANTE SOLICITATE
      ,Setor.cd_setor||' - '||Setor.nm_setor  CENTRO_CUSTO
      ,Oficina.cd_oficina || ' - ' || Oficina.ds_oficina  SETOR_EXECUTANTE
      ,Tipo_OS.cd_tipo_os||' - '||Tipo_OS.ds_tipo_os TIPO_MANUTENCAO
      ,Localidade.cd_localidade||' - '||Localidade.ds_localidade LOCALIZACAO
      ,Usuarios.cd_usuario||' - '||Usuarios.nm_usuario RESPONSAVEL
      ,DT_PREV_EXEC PRAZO_ENTREGA
      ,DS_RAMAL RAMAL
,Decode(TP_PRIORIDADE,'M','Muito Alta','A','Alta','E','Média','B','Baixa','U','Muito Baixa') PRIORIDADE
      ,S.DS_SERVICO DESCRICAO_SERVICO
      ,S.DS_OBSERVACAO OBSERVACAO
      ,SERV.NM_SERVICO
      ,I.CD_SERVICO
      ,F.NM_FUNC EXECUTANTE_SERVICO
      ,HR_INICIO INICIO_SERVICO
      ,HR_FINAL FIM_SERVICO
      ,I.DS_SERVICO SERVICO_EXECUTADO
  FROM DBAMV.SOLICITACAO_OS S
      ,DBAMV.ITSOLICITACAO_OS I
      ,DBAMV.SETOR
      ,DBAMV.FUNCIONARIO F
      ,DBAMV.BENS
      ,DBAMV.MANU_SERV SERV
      ,Dbamv.Oficina
      ,dbamv.Tipo_OS
      ,Dbamv.Localidade
      ,DbaSgu.Usuarios
 WHERE S.CD_OS            = I.CD_OS(+)
   AND F.CD_FUNC(+)       = I.CD_FUNC
   AND S.CD_SETOR         = SETOR.CD_SETOR
   AND S.CD_BEM           = BENS.CD_BEM(+)
   AND S.Cd_Oficina       = Oficina.cd_oficina(+)
   AND S.Cd_Tipo_OS       = Tipo_OS.Cd_Tipo_OS(+)
   AND S.Cd_Setor         = Localidade.Cd_Setor
   AND S.Cd_Localidade    = Localidade.Cd_Localidade(+)
   AND S.Cd_Responsavel   = Usuarios.Cd_Usuario(+)
   AND SERV.CD_SERVICO(+) = I.CD_SERVICO
AND S.dt_pedido BETWEEN  to_date( to_char(:CAMPO_DTINI,'dd/mm/yyyy')|| ' 00','dd/mm/yyyy hh24')
                                        AND to_date( to_char(:CAMPO_DTFIM,'dd/mm/yyyy')|| ' 23','dd/mm/yyyy hh24')
ORDER BY S.CD_OS
)
ORDER BY ORDEM_SERVICO,DATA_PROGRAMADA,EQUIPAMENTO,SOLICITATE,SETOR_EXECUTANTE,TIPO_MANUTENCAO,CENTRO_CUSTO,LOCALIZACAO,RESPONSAVEL,PRAZO_ENTREGA,RAMAL,PRIORIDADE,DESCRICAO_SERVICO
