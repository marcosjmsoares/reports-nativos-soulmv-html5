--R_FUNC_ESPEC
SELECT * FROM (
SELECT L_MANU_FUNC.NM_FUNC L_MANU_FUNC_NM_FUNC,
               MANU_ESPEC.DS_ESPEC L_MANU_ESPEC_DS_ESPEC,
               FUNC_ESPEC2.CD_FUNC CD_FUNC,
               FUNC_ESPEC2.CD_ESPEC CD_ESPEC,
               L_MANU_FUNC.VL_HORA VL_HORA
FROM     DBAMV.FUNCIONARIO L_MANU_FUNC,
               DBAMV.MANU_ESPEC MANU_ESPEC,
               DBAMV.FUNC_ESPEC FUNC_ESPEC2
WHERE  FUNC_ESPEC2.CD_ESPEC = MANU_ESPEC.CD_ESPEC
      AND  FUNC_ESPEC2.CD_FUNC = L_MANU_FUNC.CD_FUNC
:CAMPO_CF_ESPEC
)
ORDER BY L_MANU_FUNC_NM_FUNC,VL_HORA,CD_ESPEC
