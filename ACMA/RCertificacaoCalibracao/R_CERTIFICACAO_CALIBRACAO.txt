--R_CERTIFICACAO_CALIBRACAO
SELECT * FROM (
SELECT SOLICITACAO_OS.CD_OS
       ,TIPO_OS.CD_TIPO_OS
       ,TIPO_OS.DS_TIPO_OS
       ,MANU_ESPEC.DS_ESPEC
       ,SOLICITACAO_OS.NM_SOLICITANTE
       ,SOLICITACAO_OS.DT_PEDIDO
       ,BENS.CD_BEM
       ,BENS.DS_BEM
       ,BENS.DS_MARCA
       ,BENS.DS_MODELO
       ,BENS.NR_SERIE
       ,BENS.DS_ESPECIFICACOES
       ,CHECK_LIST.DS_PROC_QUALIDADE
       ,MANU_SERV.CD_SERVICO
       ,MANU_SERV.NM_SERVICO
       ,MANU_SERV.DS_UNIDADE_MEDIDA
       ,AVG(ITSOLICITACAO_OS.VL_REFERENCIA) VL_REFERENCIA
       ,AVG(ITSOLICITACAO_OS.VL_REAL) VL_REAL
       ,BENS_ITEM.CD_BEM CD_BEM_ITEM
       ,BENS_ITEM.DS_BEM DS_BEM_ITEM
       ,OFICINA.CD_OFICINA cd_oficina
       ,OFICINA.DS_OFICINA
  FROM DBAMV.SOLICITACAO_OS
       ,DBAMV.TIPO_OS
       ,DBAMV.MANU_ESPEC
       ,DBAMV.BENS
       ,DBAMV.CHECK_LIST
       ,DBAMV.MANU_SERV
       ,DBAMV.ITSOLICITACAO_OS
       ,DBAMV.BENS   BENS_ITEM
       ,DBAMV.OFICINA
 WHERE SOLICITACAO_OS.CD_TIPO_OS = TIPO_OS.CD_TIPO_OS
   AND SOLICITACAO_OS.CD_ESPEC         = MANU_ESPEC.CD_ESPEC(+)
   AND SOLICITACAO_OS.CD_BEM            = BENS.CD_BEM(+)
   AND BENS.CD_CHECK_LIST                   = CHECK_LIST.CD_CHECK_LIST(+)
   AND SOLICITACAO_OS.CD_OS               = ITSOLICITACAO_OS.CD_OS
   AND ITSOLICITACAO_OS.CD_SERVICO = MANU_SERV.CD_SERVICO(+)
   AND ITSOLICITACAO_OS.CD_BEM         = BENS_ITEM.CD_BEM (+)
   AND SOLICITACAO_OS.CD_OFICINA      = OFICINA.CD_OFICINA(+)
   AND MANU_SERV.TP_CATEGORIA        = '4' 
   AND SOLICITACAO_OS.CD_OS     = :CAMPO_CD_OS
   AND oficina.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
GROUP BY
SOLICITACAO_OS.CD_OS
       ,TIPO_OS.DS_TIPO_OS
       ,MANU_ESPEC.DS_ESPEC
       ,SOLICITACAO_OS.NM_SOLICITANTE
       ,SOLICITACAO_OS.DT_PEDIDO
       ,BENS.CD_BEM
       ,BENS.DS_BEM
       ,BENS.DS_MARCA
       ,BENS.DS_MODELO
       ,BENS.NR_SERIE
       ,BENS.DS_ESPECIFICACOES
       ,CHECK_LIST.DS_PROC_QUALIDADE
       ,MANU_SERV.CD_SERVICO
       ,MANU_SERV.NM_SERVICO
       ,MANU_SERV.DS_UNIDADE_MEDIDA
       ,BENS_ITEM.CD_BEM
       ,BENS_ITEM.DS_BEM
       ,OFICINA.CD_OFICINA
       ,OFICINA.DS_OFICINA
       ,TIPO_OS.CD_TIPO_OS
)
ORDER BY CD_OS,CD_OFICINA,CD_SERVICO, CD_TIPO_OS
