--R_AUDITORIA_OS
SELECT solicitacao_os.cd_os
     , solicitacao_os.dt_pedido
     , solicitacao_os.nm_solicitante Solicitante
     , solicitacao_os.ds_observacao
     , solicitacao_os.ds_servico
     , solicitacao_os.ds_ramal
     , solicitacao_os.dt_entrega Prazo_Entrega
     , usuarios.nm_usuario Responsavel
     , solicitacao_os.dt_ultima_atualizacao Ultima_Atualizacao
     , tipo_os.ds_tipo_os
     , Decode(Nvl(solicitacao_os.tp_situacao,'A'),'A' , 'Aberto'
                                    ,'C' , 'Concluído'
                                    ,'N' , 'Não Atendido'
                                    ,'M' , 'Aguardando material'
                                    ,'E' , 'Concerto Externo'
                                    ,'S' , 'Solicitação'
                                    ,'F' , 'A Agendar'
                                    ,'L' , 'Aguardando liberação do setor') Situacao
     , Decode(Nvl(avaliacao_servico.tp_avaliacao,'N'),'N', 'Não Avaliado'
                                      ,'E', 'Excelente'
                                      ,'B', 'Bom'
                                      ,'R', 'Razoável'
                                      ,'U', 'Ruim') Avaliacao
    , solicitacao_os.cd_usuario_cadastro_ss
    , solicitacao_os.dt_usuario_cadastro_ss
    , (Select nm_usuario from dbasgu.usuarios where cd_usuario = solicitacao_os.cd_usuario_cadastro_ss) nm_usuario_cadastro_ss
    , solicitacao_os.cd_usuario_cadastro_os
    , solicitacao_os.dt_usuario_cadastro_os
    , (Select nm_usuario from dbasgu.usuarios where cd_usuario = solicitacao_os.cd_usuario_cadastro_os) nm_usuario_cadastro_os
    , solicitacao_os.cd_usuario_ultima_modificacao
    , solicitacao_os.dt_usuario_ultima_modificacao
    , (Select nm_usuario from dbasgu.usuarios where cd_usuario = solicitacao_os.cd_usuario_ultima_modificacao) nm_ultima_modificacao
    , solicitacao_os.cd_usuario_recebe_ss
    , solicitacao_os.dt_usuario_recebe_ss
    , (Select nm_usuario from dbasgu.usuarios where cd_usuario = solicitacao_os.cd_usuario_recebe_ss) nm_usuario_recebe_ss
    , solicitacao_os.cd_usuario_fecha_os
    , solicitacao_os.dt_usuario_fecha_os
    , (Select nm_usuario from dbasgu.usuarios where cd_usuario = solicitacao_os.cd_usuario_fecha_os) nm_usuario_fecha_os
    , solicitacao_os.cd_usuario_gera_os
    , solicitacao_os.dt_usuario_gera_os
    , (Select nm_usuario from dbasgu.usuarios where cd_usuario = solicitacao_os.cd_usuario_gera_os) nm_usuario_gera_os
FROM dbamv.solicitacao_os
   , dbamv.avaliacao_servico
   , dbasgu.usuarios
   , dbamv.tipo_os
WHERE solicitacao_os.cd_os = avaliacao_servico.cd_solicitacao_ordem_srv(+)
  AND solicitacao_os.cd_responsavel = usuarios.cd_usuario(+)
  AND solicitacao_os.cd_tipo_os = tipo_os.cd_tipo_os
  AND ( ( :CAMPO_FUNCIONARIO = '%' )   OR
             ( :CAMPO_FUNCIONARIO <> '%'   :CAMPO_CF_EXECUTANTE )
          )
  AND (         ( :CAMPO_OPCAO_PERIODO = 'CS'  AND solicitacao_os.dt_usuario_cadastro_ss BETWEEN  to_date( to_char(:CAMPO_DATA_INI,'dd/mm/yyyy')|| ' 00','dd/mm/yyyy hh24')
                                                                                                                                AND to_date( to_char(:CAMPO_DATA_FIM,'dd/mm/yyyy')|| ' 23','dd/mm/yyyy hh24')  )
             OR  ( :CAMPO_OPCAO_PERIODO = 'RS'  AND solicitacao_os.dt_usuario_recebe_ss BETWEEN to_date( to_char(:CAMPO_DATA_INI,'dd/mm/yyyy')|| ' 00','dd/mm/yyyy hh24')
                                                                                                                                     AND to_date( to_char(:CAMPO_DATA_FIM,'dd/mm/yyyy')|| ' 23','dd/mm/yyyy hh24')   )
             OR  ( :CAMPO_OPCAO_PERIODO = 'CO'  AND solicitacao_os.dt_usuario_cadastro_os BETWEEN to_date( to_char(:CAMPO_DATA_INI,'dd/mm/yyyy')|| ' 00','dd/mm/yyyy hh24')
                                                                                                                                     AND to_date( to_char(:CAMPO_DATA_FIM,'dd/mm/yyyy')|| ' 23','dd/mm/yyyy hh24')   )
             OR  ( :CAMPO_OPCAO_PERIODO = 'UO'  AND solicitacao_os.dt_usuario_ultima_modificacao BETWEEN to_date( to_char(:CAMPO_DATA_INI,'dd/mm/yyyy')|| ' 00','dd/mm/yyyy hh24')
                                                                                                                                     AND to_date( to_char(:CAMPO_DATA_FIM,'dd/mm/yyyy')|| ' 23','dd/mm/yyyy hh24')   )
             OR  ( :CAMPO_OPCAO_PERIODO = 'FO'  AND solicitacao_os.dt_usuario_fecha_os BETWEEN to_date( to_char(:CAMPO_DATA_INI,'dd/mm/yyyy')|| ' 00','dd/mm/yyyy hh24')
                                                                                                                                     AND to_date( to_char(:CAMPO_DATA_FIM,'dd/mm/yyyy')|| ' 23','dd/mm/yyyy hh24')   )
             OR  ( :CAMPO_OPCAO_PERIODO = 'GO'  AND solicitacao_os.dt_usuario_gera_os BETWEEN to_date( to_char(:CAMPO_DATA_INI,'dd/mm/yyyy')|| ' 00','dd/mm/yyyy hh24')
                                                                                                                                     AND to_date( to_char(:CAMPO_DATA_FIM,'dd/mm/yyyy')|| ' 23','dd/mm/yyyy hh24')   )
             OR  ( :CAMPO_OPCAO_PERIODO = 'SF'  )
           )
ORDER BY solicitacao_os.cd_os, solicitacao_os.dt_pedido
