--R_CONSUMO_AGUA_GAS
Select To_Char(A.DT_REFERENCIA, 'YYYYMM') || LPad(To_Char(RowNum), 2, '0') As MES_ANO_REFERENCIA, A.VL_TOTAL, A.QT_CONSUMO,
D.DS_UNIDADE_CONSUMIDORA, ' ' As ROTULO_PADRAO
From DBAMV.UNIDADE_CONSUMIDORA D, DBAMV.CONSUMO A
Where A.CD_UNIDADE_CONSUMIDORA = :CAMPO_N_CD_UNIDADE_CONSUMIDORA And
To_Char(A.DT_REFERENCIA, 'YYYYMM') Between To_Char(Add_Months(:CAMPO_DT_REFERENCIA, -11), 'YYYYMM') And
To_Char(:CAMPO_DT_REFERENCIA, 'YYYYMM') And A.CD_UNIDADE_CONSUMIDORA = D.CD_UNIDADE_CONSUMIDORA
Order By 1
