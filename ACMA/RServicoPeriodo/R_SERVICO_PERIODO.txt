--R_SERVICO_PERIODO
SELECT * FROM (
select solicitacao_os.cd_setor,
       setor.nm_setor,
       itsolicitacao_os.cd_servico,
       manu_serv.nm_servico,
       count(*) qtd,
     sum(vl_tempo_gasto) horas,
     sum(vl_tempo_gasto_min) minutos,
       round(sum((vl_tempo_gasto*60) + vl_tempo_gasto_min),2) total_horas_minutos
from dbamv.itsolicitacao_os,
     dbamv.solicitacao_os,
     dbamv.setor,
     dbamv.manu_serv
where trunc(hr_final ) between :CAMPO_DTINI and  and
      setor.cd_setor = solicitacao_os.cd_setor and
      solicitacao_os.cd_os = itsolicitacao_os.cd_os and
      manu_serv.cd_servico = itsolicitacao_os.cd_servico and
      setor.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
group by solicitacao_os.cd_setor,
       setor.nm_setor,
       itsolicitacao_os.cd_servico,
       manu_serv.nm_servico
)
ORDER BY CD_SETOR,NM_SETOR
