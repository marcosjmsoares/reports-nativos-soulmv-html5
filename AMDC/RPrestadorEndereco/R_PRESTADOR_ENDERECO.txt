--R_PRESTADOR_ENDERECO
SELECT * FROM (
SELECT PRESTADOR.CD_PRESTADOR
              ,PRESTADOR.NM_PRESTADOR
              ,CIDADE.NM_CIDADE NM_CIDADE
              ,COUNT(*)  QT_PRESTADOR
              ,PRESTADOR.DS_BAIRRO_COMERCIAL BAIRRO_COM
              ,ESPECIALID.DS_ESPECIALID ESPEC_PRINC
FROM DBAMV.TIP_PRESTA
          ,DBAMV.PRESTADOR
          ,DBAMV.ESP_MED
          ,DBAMV.ESPECIALID
          ,DBAMV.CIDADE
WHERE TIP_PRESTA.CD_TIP_PRESTA = PRESTADOR.CD_TIP_PRESTA
AND PRESTADOR.CD_PRESTADOR = ESP_MED.CD_PRESTADOR
AND ESP_MED.CD_ESPECIALID = ESPECIALID.CD_ESPECIALID
AND ESP_MED.SN_ESPECIAL_PRINCIPAL = 'S'
AND PRESTADOR.CD_CIDADE_COMERCIAL = CIDADE.CD_CIDADE
GROUP BY  CIDADE.NM_CIDADE
                    ,PRESTADOR.CD_PRESTADOR
                    ,PRESTADOR.NM_PRESTADOR
                    ,PRESTADOR.NR_PISPASEP
                    ,PRESTADOR.INSCR_AUTONOMO
                    ,PRESTADOR.nr_agencia||'/'||PRESTADOR.nr_conta
                    ,PRESTADOR.DS_BAIRRO_COMERCIAL
                    ,ESPECIALID.DS_ESPECIALID
:CAMPO_CF_ORDEM
)
