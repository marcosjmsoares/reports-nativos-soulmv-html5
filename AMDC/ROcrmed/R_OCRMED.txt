--R_OCRMED
SELECT * FROM (
SELECT  OCO_MED1.DT_OCO_MED DT_OCO_MED1
       ,OCO_MED1.CD_PRESTADOR CD_PRESTADOR3
       ,PRESTADOR.CD_PRESTADOR CODIGO
       ,PRESTADOR.NM_PRESTADOR DESCRICAO
       ,OCO_MED1.CD_OCO_MED CD_OCO_MED
       ,OCO_MED1.DS_OCO_MED DS_OCO_MED
FROM    dbamv.OCO_MED OCO_MED1
       ,dbamv.PRESTADOR PRESTADOR
WHERE PRESTADOR.CD_PRESTADOR = OCO_MED1.CD_PRESTADOR
AND PRESTADOR.CD_PRESTADOR LIKE :CAMPO_PRESTADOR
AND OCO_MED1.DT_OCO_MED BETWEEN :CAMPO_DT_INIC ORDER BY OCO_MED1.DT_OCO_MED
        ,OCO_MED1.CD_PRESTADOR
        ,PRESTADOR.CD_PRESTADOR
        ,PRESTADOR.NM_PRESTADOR
)
ORDER BY DT_OCO_MED1,
CD_PRESTADOR3,CODIGO,DESCRICAO
