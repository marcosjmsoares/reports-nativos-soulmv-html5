--R_PRESTADOR_R_13_Q_DADOS_REPASSE
SELECT * FROM (
SELECT  EMPRESA_PRESTADOR.CD_PRESTADOR
       ,EMPRESA_PRESTADOR.CD_REG_REPASSE
       ,EMPRESA_PRESTADOR.CD_REG_REPASSE_SIA
       ,EMPRESA_PRESTADOR.CD_REG_REPASSE_SIH
       ,EMPRESA_PRESTADOR.CD_INDICE
       ,EMPRESA_PRESTADOR.CD_FORNECEDOR
       ,:CAMPO_CF_TP_VINCULO TP_VINCULO
       ,REG_REPASSE.DS_REG_REPASSE
       ,REG_REPASSE_SIA.DS_REG_REPASSE_SIA
       ,REG_REPASSE_SIH.DS_REG_REPASSE_SIH
       ,INDICE.NM_INDICE
       ,FORNECEDOR.NM_FORNECEDOR
       ,EMPRESA_PRESTADOR.VL_PERC_REPASSE_CONTA
FROM    DBAMV.EMPRESA_PRESTADOR
       ,DBAMV.PRESTADOR
       ,DBAMV.REG_REPASSE
       ,DBAMV.REG_REPASSE_SIA
       ,DBAMV.REG_REPASSE_SIH
       ,DBAMV.FORNECEDOR
       ,DBAMV.INDICE
       ,DBAMV.PRESTADOR_TIPO_VINCULO
WHERE  EMPRESA_PRESTADOR.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
  AND  EMPRESA_PRESTADOR.CD_REG_REPASSE = REG_REPASSE.CD_REG_REPASSE(+)
  AND  EMPRESA_PRESTADOR.CD_REG_REPASSE_SIA = REG_REPASSE_SIA.CD_REG_REPASSE_SIA(+)
  AND  EMPRESA_PRESTADOR.CD_REG_REPASSE_SIH = REG_REPASSE_SIH.CD_REG_REPASSE_SIH(+)
  AND  EMPRESA_PRESTADOR.CD_FORNECEDOR = FORNECEDOR.CD_FORNECEDOR(+)
  AND  EMPRESA_PRESTADOR.CD_INDICE = INDICE.CD_INDICE(+)
  AND PRESTADOR_TIPO_VINCULO.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
  :CAMPO_CF_CD_MULTI_EMPRESA
)
WHERE CD_PRESTADOR = :CAMPO_CD_PRESTADOR

--R_PRESTADOR_R_TIP_PRESTA1_Q_TIP_PRESTA
SELECT * FROM (
SELECT  DISTINCT
               TIP_PRESTA.CD_TIP_PRESTA
              ,TIP_PRESTA.NM_TIP_PRESTA
              ,PRESTADOR.CD_PRESTADOR  PRESTA
              ,( SELECT COUNT(distinct PRESTADOR.CD_PRESTADOR)
FROM DBAMV.TIP_PRESTA   tip
          ,DBAMV.PRESTADOR
          ,DBAMV.ESP_MED
          ,DBAMV.ESPECIALID
	  ,DBAMV.PRESTADOR_TIPO_VINCULO
WHERE TIP_PRESTA.CD_TIP_PRESTA = PRESTADOR.CD_TIP_PRESTA
      AND PRESTADOR.CD_PRESTADOR = ESP_MED.CD_PRESTADOR(+)
      AND ESP_MED.CD_ESPECIALID = ESPECIALID.CD_ESPECIALID(+)
      AND TIP.CD_TIP_PRESTA     =  TIP_PRESTA.CD_TIP_PRESTA
      AND PRESTADOR_TIPO_VINCULO.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
      :CAMPO_CF_CD_MULTI_EMPRESA
 ) TOTAL
 FROM  DBAMV.TIP_PRESTA
           ,DBAMV.PRESTADOR
           ,DBAMV.PRESTADOR_TIP_COMUN
           ,DBAMV.TIP_COMUN
           ,DBAMV.ESP_MED
	   ,DBAMV.ESPECIALID
	   ,DBAMV.PRESTADOR_TIPO_VINCULO
WHERE TIP_PRESTA.CD_TIP_PRESTA = PRESTADOR.CD_TIP_PRESTA
AND       PRESTADOR.CD_PRESTADOR = PRESTADOR_TIP_COMUN.CD_PRESTADOR(+)
AND       PRESTADOR_TIP_COMUN.CD_TIP_COMUN =  TIP_COMUN.CD_TIP_COMUN(+)
AND 	  PRESTADOR.CD_PRESTADOR = ESP_MED.CD_PRESTADOR(+)
AND  	  ESP_MED.CD_ESPECIALID = ESPECIALID.CD_ESPECIALID(+)
AND 	  PRESTADOR_TIPO_VINCULO.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
:CAMPO_CF_CD_MULTI_EMPRESA
) ORDER BY  CD_TIP_PRESTA ,NM_TIP_PRESTA

--R_PRESTADOR_R_TIP_PRESTA_Q_TIP_PRESTA
SELECT * FROM (
SELECT  DISTINCT
               TIP_PRESTA.CD_TIP_PRESTA
              ,TIP_PRESTA.NM_TIP_PRESTA
              ,PRESTADOR.CD_PRESTADOR  PRESTA
              ,(SELECT COUNT(distinct PRESTADOR.CD_PRESTADOR)
		FROM DBAMV.TIP_PRESTA   tip
		  ,DBAMV.PRESTADOR
	          ,DBAMV.ESP_MED
	          ,DBAMV.ESPECIALID
		  ,DBAMV.PRESTADOR_TIPO_VINCULO
		WHERE TIP_PRESTA.CD_TIP_PRESTA = PRESTADOR.CD_TIP_PRESTA
		AND PRESTADOR.CD_PRESTADOR = ESP_MED.CD_PRESTADOR(+)
	      	AND ESP_MED.CD_ESPECIALID = ESPECIALID.CD_ESPECIALID(+)
      		AND TIP.CD_TIP_PRESTA     =  TIP_PRESTA.CD_TIP_PRESTA
		AND PRESTADOR_TIPO_VINCULO.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
      		:CAMPO_CF_ATUANTE
	) TOTAL
 FROM  DBAMV.TIP_PRESTA
           ,DBAMV.PRESTADOR
           ,DBAMV.PRESTADOR_TIP_COMUN
           ,DBAMV.TIP_COMUN
           ,DBAMV.ESP_MED
           ,DBAMV.ESPECIALID
	   ,DBAMV.PRESTADOR_TIPO_VINCULO
WHERE TIP_PRESTA.CD_TIP_PRESTA = PRESTADOR.CD_TIP_PRESTA
AND       PRESTADOR.CD_PRESTADOR = PRESTADOR_TIP_COMUN.CD_PRESTADOR(+)
AND       PRESTADOR_TIP_COMUN.CD_TIP_COMUN =  TIP_COMUN.CD_TIP_COMUN(+)
AND PRESTADOR.CD_PRESTADOR = ESP_MED.CD_PRESTADOR(+)
AND ESP_MED.CD_ESPECIALID = ESPECIALID.CD_ESPECIALID(+)
AND PRESTADOR_TIPO_VINCULO.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
:CAMPO_CF_CD_MULTI_EMPRESA
) ORDER BY  CD_TIP_PRESTA ,NM_TIP_PRESTA
