--R_LICMED
SELECT * FROM (
    SELECT  LIC_MED.DT_INICIAL DT_INIC
          ,LIC_MED.DT_FINAL           DT_FIM
          ,LIC_MED.CD_PRESTADOR CD_PRESTADOR1
          ,PRESTADOR1.CD_PRESTADOR CD_PRESTADOR
          ,PRESTADOR1.NM_PRESTADOR NM_PRESTADOR1
          ,LIC_MED.CD_LIC_MED CD_LIC_MED
          ,LIC_MED.DS_LIC_MED DS_LIC_MED
    FROM    dbamv.LIC_MED LIC_MED
          ,dbamv.PRESTADOR PRESTADOR1
    WHERE PRESTADOR1.CD_PRESTADOR = LIC_MED.CD_PRESTADOR
      AND PRESTADOR1.CD_PRESTADOR LIKE :CAMPO_PRESTADOR
      AND LIC_MED.DT_INICIAL BETWEEN :CAMPO_DT_INIC AND 
      AND LIC_MED.CD_MULTI_EMPRESA = NVL(DBAMV.PKG_MV2000.LE_EMPRESA, LIC_MED.CD_MULTI_EMPRESA)
    ORDER BY LIC_MED.DT_INICIAL
            ,LIC_MED.CD_PRESTADOR
            ,PRESTADOR1.CD_PRESTADOR
          ,PRESTADOR1.NM_PRESTADOR
)
ORDER BY DT_INIC,DT_FIM,CD_PRESTADOR1,CD_PRESTADOR,NM_PRESTADOR1
