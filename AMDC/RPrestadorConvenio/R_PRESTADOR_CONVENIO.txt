--R_PRESTADOR_CONVENIO
SELECT DISTINCT  PRESTADOR.CD_PRESTADOR
               ,PRESTADOR.NM_PRESTADOR
               ,CONV.CD_CONVENIO
               ,CONV.NM_CONVENIO
FROM    DBAMV.PRESTADOR
             ,DBAMV.CONVENIO CONV
             ,DBAMV.TIP_PRESTA
             ,DBAMV.PRES_CON
            ,Dbamv.Empresa_Convenio
WHERE PRES_CON.CD_CONVENIO        = CONV.CD_CONVENIO
      AND  PRES_CON.CD_PRESTADOR   = PRESTADOR.CD_PRESTADOR
      AND  TIP_PRESTA.CD_TIP_PRESTA = PRESTADOR.CD_TIP_PRESTA
      And Empresa_Convenio.Cd_Convenio         = Conv.Cd_Convenio
      And Empresa_Convenio.Cd_Multi_Empresa =  :CAMPO_CD_MULTI_EMPRESA
ORDER BY PRESTADOR.NM_PRESTADOR
