--R_PRESTADOR_ANIVER
SELECT PRESTADOR.SN_ATUANTE, PRESTADOR.NM_PRESTADOR, PRESTADOR.DS_ENDERECO, PRESTADOR.DT_NASCIMENTO,
PRESTADOR.DS_CODIGO_CONSELHO, PRESTADOR.CD_PRESTADOR
FROM DBAMV.PRESTADOR
WHERE DBAMV.PRESTADOR.SN_ATUANTE = 'S'
AND     to_char( PRESTADOR.DT_NASCIMENTO, 'MM/DD' )  >= to_char( :CAMPO_DT_ANIV_INI, 'MM/DD' )
AND     to_char( PRESTADOR.DT_NASCIMENTO, 'MM/DD' )  <= to_char( :CAMPO_DT_ANIV_FIM, 'MM/DD' )
ORDER BY TO_CHAR(PRESTADOR.DT_NASCIMENTO,'MM/DD')
