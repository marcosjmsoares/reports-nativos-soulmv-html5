--R_PRESTADOR_ESPECIALIDADE
SELECT  COUNT(PRESTADOR.CD_PRESTADOR) QUANT
               ,PRESTADOR.CD_PRESTADOR             CODIGO
               ,PRESTADOR.NM_MNEMONICO            MNEMONICO
               ,PRESTADOR.NM_PRESTADOR            DESCRICAO
               ,DECODE(PRESTADOR.TP_SITUACAO, 'A', 'ATIVO', 'INATIVO')   TP_SITUACAO
               ,DECODE(PRESTADOR.TP_CORPO_CLINICO, 'S', 'SIM', 'NÃO')   TP_CORPO_CLINICO
               , PRESTADOR.NR_CPF_CGC
               ,PRESTADOR.DS_CODIGO_CONSELHO
               ,ESPECIALID.CD_ESPECIALID
               ,ESPECIALID.DS_ESPECIALID
FROM   dbamv.PRESTADOR
             ,dbamv.ESP_MED
             ,dbamv.ESPECIALID
WHERE  PRESTADOR.CD_PRESTADOR = ESP_MED.CD_PRESTADOR(+)
AND        ESP_MED.CD_ESPECIALID = ESPECIALID.CD_ESPECIALID(+)
AND        PRESTADOR.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
GROUP BY PRESTADOR.NM_PRESTADOR
                  ,PRESTADOR.CD_PRESTADOR
                  ,PRESTADOR.NM_MNEMONICO
                  ,TP_SITUACAO
                  ,TP_CORPO_CLINICO
                  ,PRESTADOR.NR_CPF_CGC
                  ,PRESTADOR.DS_CODIGO_CONSELHO
                  ,ESPECIALID.CD_ESPECIALID
                 ,ESPECIALID.DS_ESPECIALID
