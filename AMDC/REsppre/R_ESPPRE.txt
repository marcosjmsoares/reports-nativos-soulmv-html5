--R_ESPPRE
SELECT * FROM (
SELECT  COUNT(PRESTADOR.CD_PRESTADOR)   QUANT
               ,COUNT(ESPECIALID.CD_ESPECIALID)      QUANT_ESP
               ,ESP_MED.CD_ESPECIALID                        CD_ESPECIALID2
               ,DECODE(ESP_MED.SN_ESPECIAL_PRINCIPAL, 'S', '', 'N', '')       SN_ESPECIAL_PRINCIPAL
               ,ESPECIALID.CD_ESPECIALID                     ESPECIALIDADE
               ,ESPECIALID.DS_ESPECIALID                     DESCRICAO
               ,PRESTADOR.CD_PRESTADOR                 CD_PRESTADOR
               ,PRESTADOR.NM_PRESTADOR                 NM_MNEMONICO
               ,PRESTADOR.DS_CODIGO_CONSELHO    DS_CODIGO_CONSELHO
FROM    dbamv.ESP_MED ESP_MED
              ,dbamv.ESPECIALID ESPECIALID
              ,dbamv.PRESTADOR PRESTADOR
WHERE ESPECIALID.CD_ESPECIALID LIKE :CAMPO_ESPECIALIDADE
AND       ESPECIALID.CD_ESPECIALID = ESP_MED.CD_ESPECIALID
AND       PRESTADOR.CD_PRESTADOR = ESP_MED.CD_PRESTADOR
AND       SN_ESPECIAL_PRINCIPAL LIKE :CAMPO_TP_ESPECIALIDADE 
GROUP BY ESP_MED.CD_ESPECIALID,
                    ESP_MED.SN_ESPECIAL_PRINCIPAL,
                    ESPECIALID.CD_ESPECIALID,
                    ESPECIALID.DS_ESPECIALID,
                    PRESTADOR.CD_PRESTADOR,
                    PRESTADOR.NM_PRESTADOR,
                    PRESTADOR.DS_CODIGO_CONSELHO
ORDER BY :CAMPO_ORDEM
)
