--R_PRESTADOR_ATUACAO
SELECT DISTINCT  MAX (ATENDIME.DT_ATENDIMENTO) DT_ULT_ATEND
              ,PRESTADOR.CD_PRESTADOR
              ,PRESTADOR.NM_PRESTADOR  NM_PRESTADOR
              ,ATENDIME.CD_CONVENIO
              ,PACIENTE.NM_PACIENTE
              ,DECODE(ATENDIME.TP_ATENDIMENTO, 'I', 'INTERNACAO'
                                             , 'A', 'AMBULATORIAIS'
                                             , 'U','URGÊNCIA/EMERGENCIA'
                                             , 'E','EXTERNO'
                                             , 'H','HOME CARE') TP_ATEND
              ,ATENDIME.CD_ATENDIMENTO
              ,CONVENIO.NM_CONVENIO
              ,TRUNC(SYSDATE - ULT_ATEND.DT_ULT_ATEND2) DIFERENCA_DIAS
FROM    DBAMV.PRESTADOR
       ,DBAMV.ATENDIME
       ,DBAMV.PACIENTE
       ,DBAMV.CONVENIO
       ,( SELECT DISTINCT  MAX (ATENDIME.DT_ATENDIMENTO) DT_ULT_ATEND2
           FROM DBAMV.ATENDIME
          WHERE ATENDIME.DT_ATENDIMENTO  BETWEEN  :CAMPO_DTINIC  AND  
            AND :CAMPO_TP_ATEND IN ('A','I','%')) ULT_ATEND
WHERE  PRESTADOR.CD_PRESTADOR = ATENDIME.CD_PRESTADOR
AND ATENDIME.DT_ATENDIMENTO  BETWEEN  :CAMPO_DTINIC  AND  
AND :CAMPO_TP_ATEND IN ('A','I','%')
AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND CONVENIO.CD_CONVENIO = ATENDIME.CD_CONVENIO
AND ULT_ATEND.DT_ULT_ATEND2 =ATENDIME.DT_ATENDIMENTO
GROUP BY PRESTADOR.CD_PRESTADOR
                  ,PRESTADOR.NM_PRESTADOR
                  ,ATENDIME.CD_CONVENIO
                  ,PACIENTE.NM_PACIENTE
                  ,DECODE(ATENDIME.TP_ATENDIMENTO, 'I', 'INTERNACAO'
                                                 , 'A', 'AMBULATORIAIS'
                                                 , 'U','URGÊNCIA/EMERGENCIA'
                                                 , 'E','EXTERNO'
                                                 , 'H','HOME CARE')
                  ,ATENDIME.CD_ATENDIMENTO
                  ,CONVENIO.NM_CONVENIO
                  ,TRUNC(SYSDATE - ULT_ATEND.DT_ULT_ATEND2)
UNION
SELECT DISTINCT  MAX (AVISO_CIRURGIA.DT_REALIZACAO) DT_ULT_ATEND
              ,PRESTADOR.CD_PRESTADOR
              ,PRESTADOR.NM_PRESTADOR  NM_PRESTADOR
              ,ATENDIME.CD_CONVENIO
              ,PACIENTE.NM_PACIENTE
              ,DECODE(ATENDIME.TP_ATENDIMENTO, 'I', 'INTERNACAO'
                                             , 'A', 'AMBULATORIAIS'
                                             , 'U','URGÊNCIA/EMERGENCIA'
                                             , 'E','EXTERNO'
                                             , 'H','HOME CARE')  TP_ATEND
              ,ATENDIME.CD_ATENDIMENTO
              ,CONVENIO.NM_CONVENIO
              ,TRUNC(SYSDATE - ULT_ATEND.DT_ULT_ATEND2) DIFERENCA_DIAS
FROM    DBAMV.PRESTADOR
             ,DBAMV.ATENDIME
             ,DBAMV.AVISO_CIRURGIA
             ,DBAMV.CIRURGIA_AVISO
             ,DBAMV.PRESTADOR_AVISO
             ,DBAMV.ATI_MED
             ,DBAMV.PACIENTE
             ,DBAMV.CONVENIO
             ,( SELECT DISTINCT  MAX (ATENDIME.DT_ATENDIMENTO) DT_ULT_ATEND2
           	 FROM DBAMV.ATENDIME
          	WHERE ATENDIME.DT_ATENDIMENTO  BETWEEN  :CAMPO_DTINIC  AND  
                  AND :CAMPO_TP_ATEND IN ('A','I','%')) ULT_ATEND
WHERE PRESTADOR.CD_PRESTADOR = ATENDIME.CD_PRESTADOR
AND  AVISO_CIRURGIA.DT_REALIZACAO BETWEEN :CAMPO_DTINIC  AND  
AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = CIRURGIA_AVISO.CD_AVISO_CIRURGIA
AND AVISO_CIRURGIA.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
AND CIRURGIA_AVISO.CD_CIRURGIA = PRESTADOR_AVISO.CD_CIRURGIA
AND PRESTADOR_AVISO.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
AND PRESTADOR_AVISO.CD_ATI_MED = ATI_MED.CD_ATI_MED
AND :CAMPO_TP_ATEND IN ('CIR','PER','ANE','AUX','%')
AND ATENDIME.CD_MULTI_EMPRESA =  DBAMV.PKG_MV2000.LE_EMPRESA 
AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND CONVENIO.CD_CONVENIO = ATENDIME.CD_CONVENIO
AND ULT_ATEND.DT_ULT_ATEND2 =ATENDIME.DT_ATENDIMENTO
GROUP BY PRESTADOR.CD_PRESTADOR
                  ,PRESTADOR.NM_PRESTADOR
                  ,ATENDIME.CD_CONVENIO
                  ,PACIENTE.NM_PACIENTE
                  ,DECODE(ATENDIME.TP_ATENDIMENTO, 'I', 'INTERNACAO'
                                                 , 'A', 'AMBULATORIAIS'
                                                 , 'U','URGÊNCIA/EMERGENCIA'
                                                 , 'E','EXTERNO'
                                                 , 'H','HOME CARE')
                  ,ATENDIME.CD_ATENDIMENTO
                  ,CONVENIO.NM_CONVENIO
                  ,TRUNC(SYSDATE - ULT_ATEND.DT_ULT_ATEND2)
