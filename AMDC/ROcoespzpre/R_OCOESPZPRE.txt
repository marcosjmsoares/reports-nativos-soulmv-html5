SELECT  HIS_MED.CD_ESPECIALIZ CODIGO
       ,ESPECIALIZ.CD_ESPECIALIZ CD_ESPECIALIZ1
       ,ESPECIALIZ.DS_ESPECIALIZ DESCRICAO
       ,HIS_MED.DT_HIS_MED DT_HIS_MED
       ,PRESTADOR4.CD_PRESTADOR CD_PRESTADOR4
       ,PRESTADOR4.NM_PRESTADOR NM_PRESTADOR5
       ,CONSELHO.CD_CONSELHO CD_CONSELHO
       ,CONSELHO.DS_CONSELHO DS_CONSELHO
       ,HIS_MED.DS_HIS_MED DS_HIS_MED
FROM    dbamv.HIS_MED HIS_MED
       ,dbamv.ESPECIALIZACOES ESPECIALIZ
       ,dbamv.PRESTADOR PRESTADOR4
       ,dbamv.CONSELHO CONSELHO
WHERE ESPECIALIZ.CD_ESPECIALIZ LIKE :CAMPO_ESPECI
  AND HIS_MED.DT_HIS_MED BETWEEN :CAMPO_DT_INIC AND 
  AND HIS_MED.CD_MULTI_EMPRESA = NVL(DBAMV.PKG_MV2000.LE_EMPRESA, HIS_MED.CD_MULTI_EMPRESA)
  AND ESPECIALIZ.CD_ESPECIALIZ = HIS_MED.CD_ESPECIALIZ
  AND PRESTADOR4.CD_PRESTADOR = HIS_MED.CD_PRESTADOR
  AND CONSELHO.CD_CONSELHO = PRESTADOR4.CD_CONSELHO
ORDER BY :CAMPO_CF_ORDEM
        ,HIS_MED.DT_HIS_MED
        ,PRESTADOR4.CD_PRESTADOR
