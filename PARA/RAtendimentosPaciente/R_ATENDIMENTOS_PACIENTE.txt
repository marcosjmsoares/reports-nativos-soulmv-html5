--R_ATENDIMENTOS_PACIENTE
SELECT PACIENTE.CD_PACIENTE     CD_PACIENTE
      ,PACIENTE.NM_PACIENTE     NM_PACIENTE
      ,(SELECT PA.NM_SOCIAL_PACIENTE
      FROM DBAMV.PACIENTE PA
      WHERE PA.CD_PACIENTE = PACIENTE.CD_PACIENTE
        AND NVL(PA.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE
      ,PRESTADOR.NM_PRESTADOR   NM_PRESTADOR
      ,ATENDIME.CD_ATENDIMENTO  CD_ATENDIMENTO
      ,ATENDIME.DT_ATENDIMENTO  DT_ATENDIMENTO
      ,ATENDIME.HR_ATENDIMENTO  HR_ATENDIMENTO
      ,ATENDIME.TP_ATENDIMENTO  TP_ATENDIMENTO
      ,CONVENIO.NM_CONVENIO     NM_CONVENIO
      ,CONVENIO.CD_CONVENIO     CD_CONVENIO
      ,CON_PLA.CD_CON_PLA       CD_CON_PLA
      ,CON_PLA.DS_CON_PLA       DS_CON_PLA
      ,DES_ATE.DS_DES_ATE       DS_DES_ATE
      ,ESPECIALID.DS_ESPECIALID DS_ESPECIALID
FROM DBAMV.ATENDIME
    ,DBAMV.PACIENTE
    ,DBAMV.CON_PLA
    ,DBAMV.PRESTADOR
    ,DBAMV.CONVENIO
    ,DBAMV.DES_ATE
    ,DBAMV.ESPECIALID
WHERE ATENDIME.DT_ATENDIMENTO BETWEEN TO_DATE(TO_CHAR(:CAMPO_DATINI, 'DD/MM/YYYY') || ' 00', 'DD/MM/YYYY HH24') AND
			    		TO_DATE(TO_CHAR(:CAMPO_DATFIM, 'DD/MM/YYYY') || ' 23', 'DD/MM/YYYY HH24')
  AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
  AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
  AND ATENDIME.CD_CONVENIO = CON_PLA.CD_CONVENIO(+)
  AND ATENDIME.CD_CON_PLA  = CON_PLA.CD_CON_PLA(+)
  AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
  AND ATENDIME.CD_DES_ATE  = DES_ATE.CD_DES_ATE(+)
  AND ATENDIME.CD_ESPECIALID = ESPECIALID.CD_ESPECIALID(+)
  AND ATENDIME.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
ORDER BY PACIENTE.NM_PACIENTE
        ,ATENDIME.DT_ATENDIMENTO DESC
