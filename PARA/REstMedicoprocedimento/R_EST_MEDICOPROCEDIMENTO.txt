--R_EST_MEDICOPROCEDIMENTO
SELECT PRESTADOR.CD_PRESTADOR
              ,PRESTADOR.NM_PRESTADOR
              ,COUNT(*) QT_PRESTA
FROM   DBAMV.ATENDIME
             ,DBAMV.PRESTADOR
WHERE ATENDIME.DT_ATENDIMENTO between to_date(to_char(:CAMPO_DATINI,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
			    					  to_date(to_char(:CAMPO_DATFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND        ATENDIME.CD_PRO_INT IS NOT NULL
AND        ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
 AND (
       TP_ATENDIMENTO IN ('I','H') AND CD_ATENDIMENTO_PAI IS NULL
       OR TP_ATENDIMENTO NOT IN ('I','H')
)
GROUP BY PRESTADOR.CD_PRESTADOR
                   ,PRESTADOR.NM_PRESTADOR
ORDER BY QT_PRESTA DESC
