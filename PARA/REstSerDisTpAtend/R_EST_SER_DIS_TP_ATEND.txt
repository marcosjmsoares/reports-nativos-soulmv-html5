--R_EST_SER_DIS_TP_ATEND
SELECT * FROM (
SELECT ATENDIME.CD_SER_DIS                                          CD_SER_DIS
             ,NVL(SER_DIS.DS_SER_DIS,'NAO INFORMADO')     DS_SER_DIS
             ,TRUNC(ATENDIME.DT_ATENDIMENTO)                  DT_ATENDIMENTO
             ,ATENDIME.CD_TIP_MAR                                           CD_TIP_MAR
             ,NVL(TIP_MAR.DS_TIP_MAR,'NAO INFORMADO')    DS_TIP_MAR
             ,COUNT(*)                                                                     QT_ATENDE
FROM   DBAMV.ATENDIME
            ,DBAMV.SER_DIS
            ,DBAMV.TIP_MAR
WHERE atendime.dt_atendimento between to_date(to_char(:CAMPO_DT_INICIAL,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND
									  to_date(to_char(:CAMPO_DT_FINAL,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND        ATENDIME.CD_SER_DIS = SER_DIS.CD_SER_DIS
AND        ATENDIME.CD_TIP_MAR = TIP_MAR.CD_TIP_MAR
AND        ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY ATENDIME.CD_SER_DIS
                   ,NVL(SER_DIS.DS_SER_DIS,'NAO INFORMADO')
                  ,TRUNC(ATENDIME.DT_ATENDIMENTO)
                  ,ATENDIME.CD_TIP_MAR
                  ,NVL(TIP_MAR.DS_TIP_MAR,'NAO INFORMADO')
)
ORDER BY DS_SER_DIS,CD_SER_DIS,
DT_ATENDIMENTO
