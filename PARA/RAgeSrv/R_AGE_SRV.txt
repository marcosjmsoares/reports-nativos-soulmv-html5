--R_AGE_SRV
SELECT  MARCACAO.CD_SER_DIS
               ,SER_DIS.DS_SER_DIS
               ,PRESTADOR.CD_PRESTADOR
               ,PRESTADOR.NM_MNEMONICO
               ,MARCACAO.DT_MARCACAO
               ,MARCACAO.HR_INICIAL
               ,MARCACAO.HR_FINAL
               ,TIP_MAR.DS_TIP_MAR
               ,ITMARCACAO_TIP_MAR.NR_CONSULTAS
               ,nvl(ITMARCACAO_TIP_MAR.NR_MARCADOS,0) NR_MARCADOS
               ,MARCACAO.NR_TEMPO_MEDIO
               ,MARCACAO.SN_SIA
FROM      Dbamv.MARCACAO
               ,Dbamv.MARCACAO_SER_DIS
               ,Dbamv.SER_DIS
               ,Dbamv.PRESTADOR
               ,Dbamv.TIP_MAR
               ,Dbamv.ITMARCACAO_TIP_MAR
WHERE  PRESTADOR.CD_PRESTADOR  =  MARCACAO.CD_PRESTADOR
AND        MARCACAO.CD_MARCACAO  =  MARCACAO_SER_DIS.CD_MARCACAO
AND        SER_DIS.CD_SER_DIS  =  MARCACAO_SER_DIS.CD_SER_DIS
AND        TIP_MAR.CD_TIP_MAR  =  ITMARCACAO_TIP_MAR.CD_TIP_MAR
AND        MARCACAO.CD_MARCACAO  =  ITMARCACAO_TIP_MAR.CD_MARCACAO
AND        SER_DIS.CD_SER_DIS  LIKE  :CAMPO_SERVICO
AND 	   MARCACAO.DT_MARCACAO between to_date(to_char(:CAMPO_DT_INIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
										to_date(to_char(:CAMPO_DT_FIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
ORDER BY  MARCACAO.CD_SER_DIS
                   ,SER_DIS.DS_SER_DIS
                   ,PRESTADOR.CD_PRESTADOR
                   ,MARCACAO.DT_MARCACAO
                   ,to_date(to_char(MARCACAO.HR_INICIAL,'hh24'),'hh24')
