--R_TRANSF_DADOS_PAC
SELECT DISTINCT DECODE(LR.DS_MOTIVO_TROCPAC, 'D', 'PACIENTE DUPLICADO', 'E', 'TROCA DE EMPRESA') DS_MOTIVO_TROCPAC,  
			LR.DS_MOTIVO_REVERSAO,  
			LR.CD_LOG_TROCPAC_REVERSAO,  
			to_char(LR.DH_LOG_TROCPAC, 'DD/MM/YYYY HH24') DH_LOG_TROCPAC,
			to_char(LR.DH_LOG_REVERSAO, 'DD/MM/YYYY HH24') DH_LOG_REVERSAO, 
			LR.NM_USUARIO_LOG_TROCPAC,  
			P.CD_PACIENTE,  
			P.NM_PACIENTE,  
			(SELECT PA.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE PA WHERE PA.CD_PACIENTE = P.CD_PACIENTE AND NVL(PA.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE,  
			to_char(P.DT_NASCIMENTO, 'DD/MM/YYYY') DT_NASCIMENTO, 
			P.NM_MAE,
			to_char(P.DT_CADASTRO, 'DD/MM/YYYY') DT_CADASTRO,
			LR.DS_JUSTIFICATIVA_UNIFICACAO 
FROM DBAMV.LOG_TROCPAC_REVERSAO LR, 
		DBAMV.PACIENTE P, 
		DBAMV.LOG_UNIFICA_PACIENTE_INTEGRA LU 
WHERE LU.CD_PACIENTE_NEW = P.CD_PACIENTE 
  AND LU.CD_LOG_TROCPAC_REVERSAO = LR.CD_LOG_TROCPAC_REVERSAO 
UNION
SELECT DISTINCT DECODE(LR.DS_MOTIVO_TROCPAC, 'D', 'PACIENTE DUPLICADO', 'E', 'TROCA DE EMPRESA') DS_MOTIVO_TROCPAC,  
			LR.DS_MOTIVO_REVERSAO,  
			LR.CD_LOG_TROCPAC_REVERSAO,  
			to_char(LR.DH_LOG_TROCPAC, 'DD/MM/YYYY HH24') DH_LOG_TROCPAC,
			to_char(LR.DH_LOG_REVERSAO, 'DD/MM/YYYY HH24') DH_LOG_REVERSAO,  
			LR.NM_USUARIO_LOG_TROCPAC,  
			P.CD_PACIENTE,  
			P.NM_PACIENTE,
			(SELECT PA.NM_SOCIAL_PACIENTE FROM DBAMV.TROCPAC_PACIENTE PA WHERE PA.CD_PACIENTE = P.CD_PACIENTE AND NVL(PA.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE,  
			to_char(P.DT_NASCIMENTO, 'DD/MM/YYYY') DT_NASCIMENTO, 
			P.NM_MAE,
			to_char(P.DT_CADASTRO, 'DD/MM/YYYY') DT_CADASTRO,
			LR.DS_JUSTIFICATIVA_UNIFICACAO 
FROM DBAMV.LOG_TROCPAC_REVERSAO LR, 
		DBAMV.TROCPAC_PACIENTE P, 
		DBAMV.LOG_UNIFICA_PACIENTE_INTEGRA LU 
WHERE LU.CD_PACIENTE_NEW = P.CD_PACIENTE 
  AND LU.CD_LOG_TROCPAC_REVERSAO = LR.CD_LOG_TROCPAC_REVERSAO 
  ORDER BY CD_PACIENTE
