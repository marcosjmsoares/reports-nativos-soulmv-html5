--R_COMPROVANTE_AGENDA_EXA
SELECT * FROM (
SELECT horario.data_da_marcacao  data_da_marcacao
              ,plantao.dt_marcacao dt_marcacao
              ,To_Char(horario.hora_agenda, 'hh24') hora_agenda
              ,horario.cd_paciente cd_paciente
              ,horario.cd_paciente||' - '||horario.nm_paciente paciente
              ,horario.cd_convenio cd_convenio
              ,convenio.cd_convenio||' - '||convenio.nm_convenio convenio
              ,horario.cd_con_pla cd_con_pla
              ,horario.cd_con_pla||' - '||con_pla.ds_con_pla con_pla
              ,plantao.cd_unidade_regional cd_unidade_regional
              ,plantao.cd_unidade_regional||' - '||unidade_regional.ds_unidade_regional  unidade_regional
              ,plantao.cd_prestador cd_prestador
              ,plantao.cd_prestador||' - '||prestador.nm_prestador prestador
              ,plantao.cd_recurso
              ,plantao.cd_recurso||' - '||srecurso.nm_recurso recurso
              ,exa_rx.cd_exa_rx cd_exa_rx
              ,exa_rx.cd_exa_rx||' - '||exa_rx.ds_exa_rx exa_rx
              ,exa_rx.ds_orientacao
  FROM dbamv.v_central_marcacao_plantao plantao
            ,dbamv.v_central_marcacao_horario horario
            ,dbamv.convenio
            ,dbamv.con_pla
            ,dbamv.exa_rx_it_escala_dia
            ,dbamv.exa_rx
            ,dbamv.srecurso
            ,dbamv.prestador
            ,dbamv.unidade_regional
Where plantao.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And plantao.codigo_agenda = horario.codigo_agenda
       AND plantao.tipo_agenda   = horario.tipo_agenda
       AND plantao.cd_ser_dis = horario.cd_ser_dis(+)
       AND plantao.cd_tip_mar = horario.cd_tip_mar(+)
       AND plantao.cd_recurso = srecurso.cd_recurso(+)
       AND plantao.cd_prestador = prestador.cd_prestador(+)
       AND plantao.cd_unidade_regional = unidade_regional.cd_unidade_regional(+)
       AND horario.cd_convenio   = convenio.cd_convenio
       AND horario.cd_convenio   = con_pla.cd_convenio
       AND horario.cd_con_pla    = con_pla.cd_con_pla
       AND exa_rx_it_escala_dia.cd_escala_dia = horario.codigo_agenda
       AND exa_rx_it_escala_dia.cd_exa_rx  = exa_rx.cd_exa_rx
       AND plantao.tipo_agenda   = 'I'
       AND plantao.codigo_agenda = :CAMPO_CODIGO_AGENDA
       AND to_char(horario.hora_agenda, 'hh24') = :CAMPO_HR_AGENDA
       AND to_char(exa_rx_it_escala_dia.hr_agenda, 'hh24') = :CAMPO_HR_AGENDA
)
ORDER BY CD_CONVENIO,CD_CON_PLA,UNIDADE_REGIONAL,PRESTADOR,CD_RECURSO,RECURSO,CD_PACIENTE,PACIENTE,CONVENIO,CON_PLA,DATA_DA_MARCACAO,DT_MARCACAO,HORA_AGENDA,CD_UNIDADE_REGIONAL,CD_PRESTADOR
