--R_PROD_ORI_SER
SELECT ORI_ATE.CD_ORI_ATE                                               CD_ORI_ATE
             ,ORI_ATE.DS_ORI_ATE                                                DS_ORI_ATE
             ,MARCACAO_SER_TIPO.CD_SER_DIS                        CD_SER_DIS
             ,SER_DIS.DS_SER_DIS                                                 DS_SER_DIS
             ,MARCACAO.DT_MARCACAO                                        DT_MARCACAO
             ,TIP_MAR.CD_TIP_MAR                                               CD_TIP_MAR
             ,TIP_MAR.DS_TIP_MAR                                              DS_TIP_MAR
             ,MARCACAO.CD_MARCACAO                                      CD_MARCACAO
             ,nvl(SUM(MARCACAO_SER_TIPO.NR_CONSULTAS),0)      CONSULTA
             ,nvl(SUM(MARCACAO_SER_TIPO.NR_MARCADOS),0)       MARCACAO
FROM   DBAMV.MARCACAO
             ,DBAMV.ORI_ATE
             ,DBAMV.TIP_MAR
             ,DBAMV.MARCACAO_SER_TIPO
             ,DBAMV.SER_DIS
WHERE MARCACAO.DT_MARCACAO between to_date(to_char(:CAMPO_DTINIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
							 	   to_date(to_char(:CAMPO_DTFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND        NVL(MARCACAO.SN_ATIVO,'S')  =  'S'
AND        MARCACAO.CD_ORI_ATE  =  ORI_ATE.CD_ORI_ATE
AND        MARCACAO_SER_TIPO.CD_TIP_MAR  =  TIP_MAR.CD_TIP_MAR
AND        MARCACAO.CD_MARCACAO  =  MARCACAO_SER_TIPO.CD_MARCACAO
AND        MARCACAO_SER_TIPO.CD_SER_DIS  =  SER_DIS.CD_SER_DIS
GROUP BY  ORI_ATE.CD_ORI_ATE
                    ,ORI_ATE.DS_ORI_ATE
                    ,MARCACAO_SER_TIPO.CD_SER_DIS
                    ,SER_DIS.DS_SER_DIS
                    ,MARCACAO.DT_MARCACAO
                    ,MARCACAO.CD_TIP_MAR
                    ,TIP_MAR.DS_TIP_MAR
                    ,TIP_MAR.CD_TIP_MAR
                    ,MARCACAO.CD_MARCACAO
ORDER BY ORI_ATE.DS_ORI_ATE, SER_DIS.DS_SER_DIS,
                    MARCACAO.DT_MARCACAO,
                    TIP_MAR.CD_TIP_MAR
