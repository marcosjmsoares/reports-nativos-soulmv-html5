--R_EST_IDADE_SEXO
SELECT  FAIXA_ETARIA                  FAIXA_ETARIA
               ,SUM(TOT_MASCULINO)  TOT_MASCULINO
               ,SUM(TOT_FEMININO)      TOT_FEMININO
               ,SUM(TOT_AMBOS)           TOT_INDEFINIDO
FROM
(
SELECT DBAMV.RETORNA_FAIXA_ETARIA_RELATORIO( PACIENTE.DT_NASCIMENTO, 'D' ) FAIXA_ETARIA
              ,NVL(DBAMV.RETORNA_FAIXA_ETARIA_RELATORIO( PACIENTE.DT_NASCIMENTO, 'C' ), 1000) CD_ETARIA
              ,COUNT(*)   TOT_MASCULINO
              ,SUM(0)       TOT_FEMININO
              ,SUM(0)       TOT_AMBOS
FROM  DBAMV.ATENDIME
           ,DBAMV.PACIENTE
WHERE ATENDIME.DT_ATENDIMENTO between to_date(to_char(:CAMPO_DATINI,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND
									  to_date(to_char(:CAMPO_DATFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND        ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND        PACIENTE.TP_SEXO = 'M'
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY DBAMV.RETORNA_FAIXA_ETARIA_RELATORIO( PACIENTE.DT_NASCIMENTO, 'D' )
       ,DBAMV.RETORNA_FAIXA_ETARIA_RELATORIO( PACIENTE.DT_NASCIMENTO, 'C' )
UNION ALL
SELECT DBAMV.RETORNA_FAIXA_ETARIA_RELATORIO( PACIENTE.DT_NASCIMENTO, 'D' ) FAIXA_ETARIA
       ,NVL(DBAMV.RETORNA_FAIXA_ETARIA_RELATORIO( PACIENTE.DT_NASCIMENTO, 'C' ),1000) CD_ETARIA
              ,SUM(0)       TOT_MASCULINO
              ,COUNT(*)   TOT_FEMININO
              ,SUM(0)       TOT_AMBOS
FROM  DBAMV.ATENDIME
           ,DBAMV.PACIENTE
WHERE ATENDIME.DT_ATENDIMENTO between to_date(to_char(:CAMPO_DATINI,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND
									  to_date(to_char(:CAMPO_DATFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND       ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND       PACIENTE.TP_SEXO = 'F'
GROUP BY DBAMV.RETORNA_FAIXA_ETARIA_RELATORIO( PACIENTE.DT_NASCIMENTO, 'D' )
       ,DBAMV.RETORNA_FAIXA_ETARIA_RELATORIO( PACIENTE.DT_NASCIMENTO, 'C' )
UNION ALL
SELECT DBAMV.RETORNA_FAIXA_ETARIA_RELATORIO( PACIENTE.DT_NASCIMENTO, 'D' ) FAIXA_ETARIA
       ,NVL(DBAMV.RETORNA_FAIXA_ETARIA_RELATORIO( PACIENTE.DT_NASCIMENTO, 'C' ),1000) CD_ETARIA
              ,SUM(0)       TOT_MASCULINO
              ,SUM(0)       TOT_FEMININO
              ,COUNT(*)   TOT_AMBOS
FROM  DBAMV.ATENDIME
           ,DBAMV.PACIENTE
WHERE ATENDIME.DT_ATENDIMENTO between to_date(to_char(:CAMPO_DATINI,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND
									  to_date(to_char(:CAMPO_DATFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND       ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND      PACIENTE.TP_SEXO = 'I'
GROUP BY DBAMV.RETORNA_FAIXA_ETARIA_RELATORIO( PACIENTE.DT_NASCIMENTO, 'D' )
       ,DBAMV.RETORNA_FAIXA_ETARIA_RELATORIO( PACIENTE.DT_NASCIMENTO, 'C' )
)
GROUP BY FAIXA_ETARIA, CD_ETARIA
ORDER BY TO_NUMBER(CD_ETARIA )
