SELECT   PRESTADOR.CD_PRESTADOR           CD_PRESTADOR,
         PRESTADOR.NM_PRESTADOR           NM_PRESTADOR,
         MARCACAO.DT_MARCACAO                DT_MARCACAO,
         MARCACAO.HR_INICIAL                        HR_INICIAL,
         MARCACAO.HR_FINAL                          HR_FINAL,
         TIP_MAR.DS_TIP_MAR                         DS_TIP_MAR,
         ITMARCACAO_TIP_MAR.NR_CONSULTAS            NR_CONSULTAS,
         ITMARCACAO_TIP_MAR.NR_MARCADOS             NR_MARCADOS,
         ITMARCACAO_TIP_MAR.NR_ENCAIXES                NR_ENCAIXES,
         MARCACAO.CD_MARCACAO                                  CD_MARCACAO,
         DECODE( MARCACAO.TP_PRODUCAO, 'P', 'PRODUCAO', 'CONTRATO' )  TP_PRODUCAO,
         ITMARCACAO_TIP_MAR.                                        CD_TIP_MAR
FROM     DBAMV.PRESTADOR,
         DBAMV.MARCACAO,
         DBAMV.ITMARCACAO_TIP_MAR,
         DBAMV.TIP_MAR
WHERE    MARCACAO.CD_PRESTADOR =  PRESTADOR.CD_PRESTADOR
AND      MARCACAO.CD_MARCACAO  =  ITMARCACAO_TIP_MAR.CD_MARCACAO
AND      TIP_MAR.CD_TIP_MAR    =  ITMARCACAO_TIP_MAR.CD_TIP_MAR
AND      MARCACAO.DT_MARCACAO  >= TO_DATE( TO_CHAR( :CAMPO_DT_INICIAL, 'DD/MM/YYYY' ) || ' 00', 'DD/MM/YYYY HH24' )
AND      MARCACAO.DT_MARCACAO  <= TO_DATE( TO_CHAR( :CAMPO_DT_FINAL  , 'DD/MM/YYYY' ) || ' 23', 'DD/MM/YYYY HH24' )
ORDER BY PRESTADOR.NM_PRESTADOR, TP_PRODUCAO,
         MARCACAO.DT_MARCACAO,
         MARCACAO.HR_INICIAL,
         TIP_MAR.CD_TIP_MAR
