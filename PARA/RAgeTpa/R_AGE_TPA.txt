--R_AGE_TPA
SELECT * FROM (
SELECT  ITMARCACAO_TIP_MAR.CD_TIP_MAR
              ,TIP_MAR.DS_TIP_MAR
              ,PRESTADOR.CD_PRESTADOR       CD_PRESTADOR1
              ,PRESTADOR.NM_MNEMONICO       NM_PRESTADOR
              ,MARCACAO.DT_MARCACAO
              ,MARCACAO.HR_INICIAL
              ,MARCACAO.HR_FINAL
              ,SER_DIS.DS_SER_DIS
              ,ITMARCACAO_TIP_MAR.NR_CONSULTAS
              ,ITMARCACAO_TIP_MAR.NR_MARCADOS
              ,MARCACAO.NR_TEMPO_MEDIO
              ,MARCACAO.SN_SIA
FROM    Dbamv.MARCACAO
             ,Dbamv.TIP_MAR
             ,Dbamv.PRESTADOR
             ,Dbamv.MARCACAO_SER_DIS
             ,Dbamv.SER_DIS
             ,Dbamv.ITMARCACAO_TIP_MAR
WHERE  PRESTADOR.CD_PRESTADOR  =  MARCACAO.CD_PRESTADOR
AND        MARCACAO.CD_MARCACAO  =  MARCACAO_SER_DIS.CD_MARCACAO
AND        MARCACAO_SER_DIS.CD_SER_DIS  =  SER_DIS.CD_SER_DIS
AND        TIP_MAR.CD_TIP_MAR  =  ITMARCACAO_TIP_MAR.CD_TIP_MAR
AND        MARCACAO.CD_MARCACAO  =  ITMARCACAO_TIP_MAR.CD_MARCACAO
AND        TIP_MAR.CD_TIP_MAR  LIKE  :CAMPO_PLANTAO
AND        MARCACAO.DT_MARCACAO BETWEEN to_date(to_char(:CAMPO_DT_INIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
										to_date(to_char(:CAMPO_DT_FIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
ORDER BY MARCACAO.CD_TIP_MAR
                  ,TIP_MAR.DS_TIP_MAR
                  ,PRESTADOR.CD_PRESTADOR
                  ,MARCACAO.DT_MARCACAO
                  ,to_date(to_char(MARCACAO.HR_INICIAL,'hh24'),'hh24')
)
ORDER BY CD_TIP_MAR,DS_TIP_MAR,
CD_PRESTADOR1,NM_PRESTADOR,DT_MARCACAO,HR_INICIAL,HR_FINAL,NR_CONSULTAS,NR_MARCADOS,NR_TEMPO_MEDIO,SN_SIA
