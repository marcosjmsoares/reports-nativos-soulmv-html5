Select  marcacao.cd_marcacao  cd_marcacao
     ,  marcacao.cd_prestador       cd_prestador
     ,  prestador.nm_prestador       nm_prestador
     ,  marcacao.cd_ser_dis          cd_ser_dis
     ,  ser_dis.ds_ser_dis               ds_ser_dis
     ,  marcacao.hr_inicial              hr_inicial
     ,  marcacao.hr_final                hr_final
     ,  marcacao.dt_marcacao       dt_marcacao
     ,  it_marcacao.cd_atendimento    cd_atendimento
     ,  it_marcacao.cd_paciente          cd_paciente
     ,  it_marcacao.sn_encaixe            sn_encaixe
     ,  paciente.nm_paciente              nm_paciente
     ,  (SELECT PA.NM_SOCIAL_PACIENTE
          FROM DBAMV.PACIENTE PA
          WHERE PA.CD_PACIENTE = paciente.CD_PACIENTE
            AND NVL(PA.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE
     ,  it_marcacao.cd_tip_mar            cd_tip_mar
     ,  tip_mar.ds_tip_mar                    ds_tip_mar
     ,  same.cd_cad_same                  cd_cad_same
     ,  same.nr_matricula_same           nr_matricula_same
     ,  same.ds_linha                           ds_linha
     ,  same.ds_coluna                        ds_coluna
     ,  cad_same.ds_cad_same           ds_cad_same
     ,  armario_same.ds_armario_same    ds_armario_same
     ,  sala_same.ds_sala_same              ds_sala_same
     ,  decode(sn_sia,'S','SIA/SUS','N','Particular')   sn_Sia
     ,  ds_ori_ate   ds_ori_ate
     ,  ori_ate.cd_ori_ate   cd_ori_ate
  From  dbamv.marcacao
     ,  dbamv.prestador
     ,  dbamv.ser_dis
     ,  dbamv.it_marcacao
     ,  dbamv.paciente
     ,  dbamv.tip_mar
     ,  dbamv.same
     ,  dbamv.cad_same
     ,  dbamv.armario_same
     ,  dbamv.sala_same
     ,  dbamv.ori_ate
 Where marcacao.dt_marcacao between to_date(to_char(:CAMPO_DT_INIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND
 									to_date(to_char(:CAMPO_DT_FIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
   and  marcacao.cd_prestador     =  prestador.cd_prestador
   and  marcacao.cd_marcacao      =  it_marcacao.cd_marcacao
   and  it_marcacao.cd_paciente   =  paciente.cd_paciente
   and  it_marcacao.cd_tip_mar    =  tip_mar.cd_tip_mar(+)
   and  paciente.cd_paciente      =  same.cd_paciente(+)
   and  same.cd_cad_same          =  cad_same.cd_cad_same(+)
   and  it_marcacao.cd_ser_dis       =  ser_dis.cd_ser_dis
   and  it_marcacao.cd_paciente   is not null
   and  same.cd_armario_same      = armario_same.cd_armario_same(+)
   and  armario_same.cd_sala_same = sala_same.cd_sala_same(+)
   and  ori_ate.cd_ori_ate        = marcacao.cd_ori_ate
ORDER BY CD_MARCACAO, PACIENTE.NM_PACIENTE
