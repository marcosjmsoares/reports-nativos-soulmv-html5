--R_SERE
SELECT  ATENDIME.CD_SER_DIS
              ,SER_DIS.DS_SER_DIS
              ,COUNT(*)   QT_ATENDE
FROM    DBAMV.ATENDIME
              ,DBAMV.SER_DIS
WHERE   SER_DIS.CD_SER_DIS(+)  =  ATENDIME.CD_SER_DIS
AND ATENDIME.DT_ATENDIMENTO between to_date(to_char(:CAMPO_DT_INIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
									to_date(to_char(:CAMPO_DT_FIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
and  SER_DIS.cd_ser_dis is not null
GROUP BY  ATENDIME.CD_SER_DIS
                    ,SER_DIS.DS_SER_DIS
ORDER BY QT_ATENDE DESC
