--R_COMPROVANTE_AGENDA_AMB
SELECT plantao.dt_marcacao dt_marcacao
             ,horario.data_da_marcacao data_da_marcacao
             ,To_Char(horario.hora_agenda, 'hh24') hora_agenda
             ,horario.cd_paciente cd_paciente
             ,horario.cd_paciente||' - '||horario.nm_paciente paciente
             ,horario.cd_convenio||' - '||convenio.nm_convenio convenio
             ,horario.cd_con_pla||' - '||con_pla.ds_con_pla con_pla
             ,plantao.cd_unidade_regional cd_unidade_regional
             ,unidade_regional.ds_unidade_regional ds_unidade_regional
             ,plantao.cd_prestador cd_prestador
             ,prestador.nm_prestador nm_prestador
             ,horario.cd_ser_dis cd_ser_dis
             ,ser_dis.ds_ser_dis ds_ser_dis
             ,horario.cd_tip_mar cd_tip_mar
             ,tip_mar.ds_tip_mar ds_tip_mar
   FROM dbamv.v_central_marcacao_plantao plantao
             ,dbamv.v_central_marcacao_horario horario
             ,dbamv.convenio
             ,dbamv.con_pla
             ,dbamv.tip_mar
             ,dbamv.ser_dis
             ,dbamv.prestador
             ,dbamv.unidade_regional
Where    plantao.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
      And   plantao.codigo_agenda = horario.codigo_agenda
      AND  plantao.tipo_agenda   = horario.tipo_agenda
      AND  plantao.cd_prestador = prestador.cd_prestador(+)
      AND  plantao.cd_ser_dis = horario.cd_ser_dis(+)
      AND  plantao.cd_tip_mar = horario.cd_tip_mar(+)
      AND  horario.cd_ser_dis   = ser_dis.cd_ser_dis(+)
      AND  horario.cd_tip_mar   = tip_mar.cd_tip_mar(+)
      AND  plantao.cd_unidade_regional = unidade_regional.cd_unidade_regional(+)
      AND  horario.cd_convenio   = convenio.cd_convenio
      AND  horario.cd_convenio   = con_pla.cd_convenio
      AND  nvl(horario.cd_con_pla, con_pla.cd_con_pla)    = con_pla.cd_con_pla
      AND  plantao.tipo_agenda   = 'A'
      AND  plantao.codigo_agenda = :CAMPO_CODIGO_AGENDA
      AND  to_char(horario.hora_agenda, 'hh24') = :CAMPO_HR_AGENDA
