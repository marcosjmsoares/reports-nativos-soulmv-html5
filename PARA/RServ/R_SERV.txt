SELECT  COUNT(*)       CONT_SER_DIS
               ,ATENDIME.CD_SER_DIS                     CD_SER_DIS1
               ,decode(SER_DIS.DS_SER_DIS, null, 'Não Informado', SER_DIS.DS_SER_DIS) DS_SER_DIS
FROM    Dbamv.ATENDIME ATENDIME
              ,Dbamv.SER_DIS SER_DIS
WHERE  SER_DIS.CD_SER_DIS(+) = ATENDIME.CD_SER_DIS
AND      ATENDIME.TP_ATENDIMENTO = 'A'
AND      ATENDIME.DT_ATENDIMENTO between to_date(to_char(:CAMPO_DT_INIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
										 to_date(to_char(:CAMPO_DT_FIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND       ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY ATENDIME.CD_SER_DIS, SER_DIS.DS_SER_DIS
ORDER BY CONT_SER_DIS desc
