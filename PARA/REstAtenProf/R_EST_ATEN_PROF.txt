--R_EST_ATEN_PROF
SELECT COUNT(*) TotalProf,
       decode(prof.NM_PROFISSAO, null, 'Não Informada', prof.NM_PROFISSAO) NM_PROFISSAO
  FROM DBAMV.ATENDIME  aten,
             DBAMV.PACIENTE  paci,
             DBAMV.PROFISSAO prof
 WHERE aten.dt_atendimento  between
to_date(to_char(:CAMPO_DATINI,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND to_date(to_char(:CAMPO_DATFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
       AND aten.CD_PACIENTE     = paci.CD_PACIENTE
       AND paci.CD_PROFISSAO   = prof.CD_PROFISSAO(+)
      and aten.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY prof.NM_PROFISSAO
ORDER BY TotalProf  DESC
