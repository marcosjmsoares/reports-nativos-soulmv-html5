--R_AGE_PRE
SELECT * FROM (
SELECT  MARCACAO.CD_PRESTADOR                           CD_PRESTADOR
               ,PRESTADOR.CD_PRESTADOR                        CD_PRESTADOR1
               ,PRESTADOR.NM_MNEMONICO                        NM_PRESTADOR
               ,MARCACAO.CD_MARCACAO                             CD_MARCACAO
               ,MARCACAO.DT_MARCACAO                             DT_MARCACAO
               ,MARCACAO.HR_INICIAL                                    HR_INICIAL
               ,MARCACAO.HR_FINAL                                      HR_FINAL
               ,TIP_MAR.DS_TIP_MAR                                     DS_TIP_MAR
               ,Nvl(ITMARCACAO_TIP_MAR.NR_CONSULTAS,0)              NR_CONSULTAS
               ,Nvl(ITMARCACAO_TIP_MAR.NR_MARCADOS,0)               NR_MARCADOS
               ,Nvl(MARCACAO.NR_TEMPO_MEDIO,0)                      NR_TEMPO_MEDIO
               ,MARCACAO.SN_SIA SN_SIA
               ,DECODE( TO_CHAR(MARCACAO.DT_MARCACAO, 'D') , '1', 'DOMINGO',
                                                                                                             '2', 'SEGUNDA',
                                                                                                             '3', 'TERCA',
                                                                                                             '4', 'QUARTA',
                                                                                                             '5', 'QUINTA',
                                                                                                             '6', 'SEXTA',
                                                                                                             'SABADO'    )  DIA_SEMANA
FROM     Dbamv.MARCACAO
              ,Dbamv.PRESTADOR
              ,Dbamv.TIP_MAR
              ,Dbamv.ITMARCACAO_TIP_MAR
WHERE  PRESTADOR.CD_PRESTADOR  =  MARCACAO.CD_PRESTADOR
AND        TIP_MAR.CD_TIP_MAR  =   ITMARCACAO_TIP_MAR.CD_TIP_MAR
AND        MARCACAO.CD_MARCACAO  =   ITMARCACAO_TIP_MAR.CD_MARCACAO
AND        MARCACAO.CD_PRESTADOR LIKE :CAMPO_CD_MEDICO
AND        MARCACAO.DT_MARCACAO between to_date(to_char(:CAMPO_DT_INIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
										to_date(to_char(:CAMPO_DT_FIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
ORDER BY  PRESTADOR.CD_PRESTADOR
                    ,MARCACAO.CD_PRESTADOR
                    ,PRESTADOR.NM_MNEMONICO
                    ,MARCACAO.DT_MARCACAO
                    ,to_date(to_char(MARCACAO.HR_INICIAL,'hh24'),'hh24')
)
ORDER BY CD_PRESTADOR,CD_PRESTADOR1,NM_PRESTADOR,
DT_MARCACAO,DIA_SEMANA,HR_INICIAL,HR_FINAL,NR_TEMPO_MEDIO,SN_SIA,CD_MARCACAO
