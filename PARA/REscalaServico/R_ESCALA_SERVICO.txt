--R_ESCALA_SERVICO
SELECT * FROM (
SELECT SER_DIS.CD_SER_DIS
              ,SER_DIS.DS_SER_DIS
              ,PRESTADOR.CD_PRESTADOR
              ,PRESTADOR.NM_PRESTADOR
              ,P_ESCALA.TP_DIA_SEMANA
              ,DECODE(TO_CHAR(P_ESCALA.TP_DIA_SEMANA), '1', 'DOMINGO',
                                                                                                    '2', 'SEGUNDA-FEIRA',
                                                                                                    '3', 'TERÇA-FEIRA',
                                                                                                    '4', 'QUARTA-FEIRA',
                                                                                                    '5', 'QUINTA-FEIRA',
                                                                                                    '6', 'SEXTA-FEIRA',
                                                                                                    '7', 'SÁBADO')           DIA
             ,P_ESCALA.HR_INIC
             ,P_ESCALA.HR_FIM
             ,P_ESCALA.NR_ATENDIMENTOS
             ,TIP_MAR.CD_TIP_MAR
             ,TIP_MAR.DS_TIP_MAR
             ,ITP_ESCALA.QT_AGENDAR
             ,P_ESCALA.SN_QTD_TIPO
             ,To_Date(trunc((to_number(to_char(P_ESCALA.HR_FIM,'sssss')) - to_number(to_char(P_ESCALA.HR_INIC,'sssss'))) / P_ESCALA.NR_ATENDIMENTOS),'sssss') tempo_medio
FROM   DBAMV.PRESTADOR
             ,DBAMV.P_ESCALA
             ,DBAMV.ITP_ESCALA
             ,DBAMV.TIP_MAR
             ,DBAMV.P_ESCALA_SER_DIS
             ,DBAMV.SER_DIS
WHERE  P_ESCALA.SN_ATIVO = 'S'
AND        PRESTADOR.CD_PRESTADOR = P_ESCALA.CD_PRESTADOR
AND        P_ESCALA.CD_P_ESCALA = ITP_ESCALA.CD_P_ESCALA
AND        ITP_ESCALA.CD_TIP_MAR = TIP_MAR.CD_TIP_MAR
AND        P_ESCALA.CD_P_ESCALA = P_ESCALA_SER_DIS.CD_P_ESCALA
AND        P_ESCALA_SER_DIS.CD_SER_DIS = SER_DIS.CD_SER_DIS
:CAMPO_CF_MULTI_EMPRESA
)
ORDER BY CD_SER_DIS,DS_SER_DIS,
CD_PRESTADOR,NM_PRESTADOR,
TP_DIA_SEMANA,DIA,HR_INIC,HR_FIM,NR_ATENDIMENTOS
