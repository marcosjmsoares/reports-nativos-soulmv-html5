--R_ATENDIMENTOSMEDICO
SELECT * FROM (
SELECT PRES.CD_PRESTADOR,
       PRES.NM_PRESTADOR,
       ATEN.CD_ATENDIMENTO,
       ATEN.CD_TIPO_INTERNACAO,
       TIPO_INTERNACAO.DS_TIPO_INTERNACAO,
       PACI.NM_PACIENTE,
       (SELECT PA.NM_SOCIAL_PACIENTE
          FROM DBAMV.PACIENTE PA
          WHERE PA.CD_PACIENTE = PACI.CD_PACIENTE
            AND NVL(PA.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE,
       ATEN.DT_ATENDIMENTO,
       to_char(ATEN.HR_ATENDIMENTO,'HH24') hr_atendimento,
       COPL.DS_CON_PLA,
       CONV.NM_CONVENIO,
       DEAT.DS_DES_ATE,
       ATEN.NR_CARTEIRA,
       GUIA.NR_GUIA,
       DECODE(ATEN.SN_RETORNO,'S','Retorno','N','Primeira Consulta','Não Informado') SN_RETORNO,
       DECODE(CONV.TP_CONVENIO, 'A', TAB_SIASUS.CD_SSM, PRO_FAT.DS_PRO_FAT)   DS_PRO_FAT
  FROM DBAMV.ATENDIME        ATEN,
       DBAMV.PACIENTE        PACI,
       DBAMV.CON_PLA         COPL,
       DBAMV.PRESTADOR   PRES,
       DBAMV.CONVENIO      CONV,
       DBAMV.DES_ATE         DEAT,
       DBAMV.GUIA,
       dbamv.TIPO_INTERNACAO,
       DBAMV.PRO_FAT,
       DBAMV.TAB_SIASUS
 WHERE ATEN.DT_ATENDIMENTO  BETWEEN  to_date(to_char(:CAMPO_DATINI,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND to_date(to_char(:CAMPO_DATFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
   AND ATEN.CD_PACIENTE  =  PACI.CD_PACIENTE
   AND ATEN.CD_CONVENIO  =  CONV.CD_CONVENIO
   AND ATEN.CD_CONVENIO  =  COPL.CD_CONVENIO(+)
   AND ATEN.CD_CON_PLA   =  COPL.CD_CON_PLA(+)
   AND ATEN.CD_PRESTADOR =  PRES.CD_PRESTADOR(+)
   AND ATEN.CD_DES_ATE   =  DEAT.CD_DES_ATE(+)   
   AND ATEN.CD_ATENDIMENTO = GUIA.CD_ATENDIMENTO(+)
   AND ATEN.CD_PRO_INT     =  PRO_FAT.CD_PRO_FAT(+)
   AND ATEN.CD_PRO_INT     =  TAB_SIASUS.CD_SSM(+)
   AND ATEN.CD_TIPO_INTERNACAO = TIPO_INTERNACAO.CD_TIPO_INTERNACAO(+)
   AND ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
   :CAMPO_CF_MULTI_EMPRESA
)
ORDER BY CD_PRESTADOR,NM_PRESTADOR,  TRUNC(DT_ATENDIMENTO),
                    HR_ATENDIMENTO,
                    NM_PACIENTE
