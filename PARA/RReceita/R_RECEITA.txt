--R_RECEITA
SELECT TEXTOS_ATD.CD_TEXTOS_ATD      CD_TEXTOS_ATD,
       'Para: ' || PACIENTE.NM_PACIENTE                         NM_PACIENTE,
       ATENDIME.CD_PRESTADOR                    CD_PRESTADOR,
       TEXTOS_ATD.CD_ATENDIMENTO           CD_ATENDIMENTO,
       SYSDATE                                                     DATA_RECEITA,
       TEXTOS_ATD.DS_TEXTOS_ATD              DS_TEXTOS_ATD,
       PRESTADOR.NM_PRESTADOR                NM_PRESTADOR
FROM   DBAMV.TEXTOS_ATD,
       DBAMV.ATENDIME,
       DBAMV.PRESTADOR,
       DUAL,
       DBAMV.PACIENTE
WHERE  TEXTOS_ATD.TP_TEXTOS_ATD = 'R'
AND    ATENDIME.CD_PACIENTE      =  PACIENTE.CD_PACIENTE
AND    ATENDIME.CD_PRESTADOR     = PRESTADOR.CD_PRESTADOR 
AND   TEXTOS_ATD.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
