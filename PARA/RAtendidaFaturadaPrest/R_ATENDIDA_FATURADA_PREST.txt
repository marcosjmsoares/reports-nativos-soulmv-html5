--R_ATENDIDA_FATURADA_PREST
SELECT PRESTADOR.CD_PRESTADOR
              ,PRESTADOR.NM_PRESTADOR
              ,COUNT(*)  NR_AGENDADAS
FROM    DBAMV.MARCACAO
             ,DBAMV.PRESTADOR
             ,DBAMV.IT_MARCACAO
WHERE  MARCACAO.DT_MARCACAO between to_date(to_char(:CAMPO_DTINIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
									 to_date(to_char(:CAMPO_DTFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND        NVL(MARCACAO.SN_ATIVO,'S')  =  'S'
AND        MARCACAO.CD_PRESTADOR  =  PRESTADOR.CD_PRESTADOR
AND        MARCACAO.CD_MARCACAO  =  IT_MARCACAO.CD_MARCACAO
AND        (IT_MARCACAO.CD_PACIENTE IS NOT NULL  OR  IT_MARCACAO.NM_PACIENTE IS NOT NULL)
GROUP BY PRESTADOR.CD_PRESTADOR
                   ,PRESTADOR.NM_PRESTADOR
ORDER BY PRESTADOR.NM_PRESTADOR
