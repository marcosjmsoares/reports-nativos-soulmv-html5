--R_PROCEDIMENTOS_ATENDIMENTOS
SELECT * FROM (
SELECT  nvl(tip_mar.DS_TIP_MAR,'Não informado')    DS_TIP_ATE
      , E.CD_PROCEDIMENTO          CD_SSM
      , P.DS_PROCEDIMENTO          DS_TITULO
      , SUM(E.QT_LANCADA) QTDE
   FROM DBAMV.EVE_SIASUS E
      , DBAMV.tip_mar
      , DBAMV.PROCEDIMENTO_SUS P
      , DBAMV.ATENDIME
      , DBAMV.ESPECIALID
 WHERE E.CD_PROCEDIMENTO = P.CD_PROCEDIMENTO
   AND atendime.cd_tip_mar = tip_mar.cd_tip_mar(+)
   AND ATENDIME.CD_ATENDIMENTO = E.CD_aTENDIMENTO
   AND E.DT_EVE_SIASUS >= TO_DATE( TO_CHAR( :CAMPO_DT_INICIAL, 'DD/MM/YYYY' ) || ' 00', 'DD/MM/YYYY HH24' )
   AND E.DT_EVE_SIASUS <= TO_DATE( TO_CHAR( :CAMPO_DT_FINAL  , 'DD/MM/YYYY' ) || ' 23', 'DD/MM/YYYY HH24' )
   AND ATENDIME.cd_especialid = especialid.cd_especialid(+)
GROUP BY  tip_mar.DS_TIP_MAR ,E.CD_PROCEDIMENTO
        , P.DS_PROCEDIMENTO
ORDER BY tip_mar.DS_TIP_MAR
        , E.CD_PROCEDIMENTO
)
ORDER BY DS_TIP_ATE
