--R_SESSOES_REALIZADAS
SELECT * FROM (
 SELECT ATENDIME.DT_ATENDIMENTO DT_ATENDIMENTO
       ,ATENDIME.CD_ATENDIMENTO CD_ATENDIMENTO
       ,ATENDIME.CD_PACIENTE CD_PACIENTE
       ,PACIENTE.NM_PACIENTE  NM_PACIENTE
       ,(SELECT PA.NM_SOCIAL_PACIENTE
          FROM DBAMV.PACIENTE PA
          WHERE PA.CD_PACIENTE = PACIENTE.CD_PACIENTE
            AND NVL(PA.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE
       ,ATENDIME_SESSAO.CD_PROCEDIMENTO_SIA CD_SSM
       ,PROCEDIMENTO_SUS.DS_PROCEDIMENTO DS_TITULO
       ,ATENDIME.QT_SESSOES QT_SESSOES
   FROM DBAMV.ATENDIME
       ,DBAMV.ATENDIME_SESSAO
       ,DBAMV.PRESTADOR
       ,DBAMV.SER_DIS
       ,DBAMV.ORI_ATE
       ,DBAMV.PACIENTE
       ,DBAMV.PROCEDIMENTO_SUS
  WHERE ATENDIME.CD_ATENDIMENTO = ATENDIME_SESSAO.CD_ATENDIMENTO_PAI
    AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
    AND ATENDIME.CD_SER_DIS = SER_DIS.CD_SER_DIS
    AND ATENDIME.CD_ORI_ATE = ORI_ATE.CD_ORI_ATE
    AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
    AND ATENDIME.CD_PROCEDIMENTO = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
    AND ATENDIME_SESSAO.CD_PROCEDIMENTO_SIA = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
    AND ATENDIME.SN_ATENDIMENTO_APAC = 'S'
    AND  ATENDIME.DT_ATENDIMENTO BETWEEN TO_DATE(TO_CHAR(:CAMPO_DT_INICIAL, 'DD/MM/YYYY') || ' 00', 'DD/MM/YYYY HH24') AND 
    									 TO_DATE(TO_CHAR(:CAMPO_DT_FINAL, 'DD/MM/YYYY') || ' 23', 'DD/MM/YYYY HH24')
    AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
UNION ALL
 SELECT ATENDIME.DT_ATENDIMENTO DT_ATENDIMENTO
       ,ATENDIME.CD_ATENDIMENTO CD_ATENDIMENTO
       ,ATENDIME.CD_PACIENTE CD_PACIENTE
       ,PACIENTE.NM_PACIENTE NM_PACIENTE
       ,(SELECT PA.NM_SOCIAL_PACIENTE
          FROM DBAMV.PACIENTE PA
          WHERE PA.CD_PACIENTE = PACIENTE.CD_PACIENTE
            AND NVL(PA.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE
       ,ATENDIME_SESSAO.CD_PROCEDIMENTO_SIA CD_SSM
       ,PROCEDIMENTO_SUS.DS_PROCEDIMENTO DS_TITULO
       ,ATENDIME_SESSAO.QT_LANCADA
   FROM DBAMV.ATENDIME
       ,DBAMV.ATENDIME_SESSAO
       ,DBAMV.PRESTADOR
       ,DBAMV.SER_DIS
       ,DBAMV.ORI_ATE
       ,DBAMV.PACIENTE
       ,DBAMV.PROCEDIMENTO_SUS
  WHERE ATENDIME.CD_ATENDIMENTO = ATENDIME_SESSAO.CD_ATENDIMENTO_PAI
    AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
    AND ATENDIME.CD_SER_DIS = SER_DIS.CD_SER_DIS
    AND ATENDIME.CD_ORI_ATE = ORI_ATE.CD_ORI_ATE
    AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
    AND ATENDIME.CD_PROCEDIMENTO = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
    AND ATENDIME_SESSAO.CD_PROCEDIMENTO_SIA = PROCEDIMENTO_SUS.CD_PROCEDIMENTO
    AND ATENDIME.SN_ATENDIMENTO_APAC = 'N'
    AND  ATENDIME.DT_ATENDIMENTO BETWEEN TO_DATE(TO_CHAR(:CAMPO_DT_INICIAL, 'DD/MM/YYYY') || ' 00', 'DD/MM/YYYY HH24') AND 
    									  TO_DATE(TO_CHAR(:CAMPO_DT_FINAL, 'DD/MM/YYYY') || ' 23', 'DD/MM/YYYY HH24')
    AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
:CAMPO_CF_COND_ORI_ATE
)
ORDER BY DT_ATENDIMENTO, CD_ATENDIMENTO, CD_PACIENTE, NM_PACIENTE
