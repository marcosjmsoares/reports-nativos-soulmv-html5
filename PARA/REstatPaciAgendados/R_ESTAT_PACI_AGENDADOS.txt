--R_ESTAT_PACI_AGENDADOS
SELECT TRUNC(MARC.DT_MARCACAO)   DT_AGENDADO
              ,COUNT(*)                                       TOTAL_DIA
FROM    DBAMV.MARCACAO          MARC
              ,DBAMV.IT_MARCACAO    ITMA
WHERE MARC.DT_MARCACAO between to_date(to_char(:CAMPO_DATINI,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND
							   to_date(to_char(:CAMPO_DATFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND       MARC.CD_MARCACAO = ITMA.CD_MARCACAO
AND       ITMA.NM_PACIENTE IS NOT NULL
GROUP BY TRUNC(MARC.DT_MARCACAO)
ORDER BY TRUNC(MARC.DT_MARCACAO)
