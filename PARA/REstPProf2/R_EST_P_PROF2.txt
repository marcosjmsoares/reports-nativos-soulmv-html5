--R_EST_P_PROF2
SELECT * FROM (
SELECT prest.CD_PRESTADOR                       CD_PRESTADOR,
               prest.NM_PRESTADOR                      NM_PRESTADOR,
               marc.DT_MARCACAO                         DT_MARCACAO,
               marc.HR_INICIAL                                HR_INICIAL,
               marc.HR_FINAL                                  HR_FINAL,
               tipo.DS_TIP_MAR                               DS_TIP_MAR,
               mase.NR_CONSULTAS                        NR_CONSULTAS,
               marc.CD_MARCACAO                        CD_MARCACAO,
               tipo.CD_TIP_MAR                               CD_TIP_MAR,
               mase.CD_SER_DIS                            CD_SER_DIS,
               serv.DS_SER_DIS                              DS_SER_DIS
    FROM      DBAMV.MARCACAO                  marc,
                    DBAMV.ITMARCACAO_TIP_MAR        mati,
                    DBAMV.MARCACAO_SER_tipo          mase,
                    DBAMV.TIP_MAR                   tipo,
                    DBAMV.PRESTADOR                 prest,
                    DBAMV.SER_DIS                   serv
        WHERE  marc.DT_MARCACAO between to_date(to_char(:CAMPO_DTINIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
       	 	   							to_date(to_char(:CAMPO_DTFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
        AND        nvl(marc.SN_ATIVO,'S')  =  'S'
        AND        marc.CD_MARCACAO  =  mati.CD_MARCACAO
        AND        mati.CD_TIP_MAR   =  tipo.CD_TIP_MAR
        AND        marc.CD_PRESTADOR =  prest.CD_PRESTADOR
        AND        mati.CD_MARCACAO  =  mase.CD_MARCACAO
        AND        mati.CD_TIP_MAR  =  mase.CD_TIP_MAR
        AND        mase.CD_SER_DIS  =   serv.CD_SER_DIS
           ORDER BY    prest.NM_PRESTADOR,
                                  serv.DS_SER_DIS,
                                  marc.DT_MARCACAO,
                                  marc.HR_INICIAL,
                                  tipo.DS_TIP_MAR
)
ORDER BY NM_PRESTADOR, CD_PRESTADOR,
DS_SER_DIS, CD_SER_DIS,
DT_MARCACAO, DS_TIP_MAR
