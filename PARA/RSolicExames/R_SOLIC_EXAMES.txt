--R_SOLIC_EXAMES
SELECT * FROM (
SELECT SOLIC_ATD.CD_SOLIC_ATD,
       SYSDATE DT_SOLIC_ATD,
       ATENDIME.CD_CONVENIO,
       CONVENIO.NM_CONVENIO,
       CARTEIRA.NR_CARTEIRA,
       CARTEIRA.DT_VALIDADE,
       CARTEIRA.NM_TITULAR,
       PACIENTE.NM_PACIENTE,
       ATENDIME.CD_CID,
       CID.DS_CID,
       NVL(DECODE(CONVENIO.TP_CONVENIO, 'A', IT_SOLIC_ATD.CD_PROCEDIMENTO, IT_SOLIC_ATD.CD_PRO_FAT) , IT_SOLIC_ATD.CD_PRO_FAT)  CD_PRO_FAT,
       NVL(DECODE(CONVENIO.TP_CONVENIO, 'A', PROCEDIMENTO_SUS.DS_PROCEDIMENTO, PRO_FAT.DS_PRO_FAT) , PRO_FAT.DS_PRO_FAT)  DS_PRO_FAT,     
       IT_SOLIC_ATD.QT_SOLICITADA,
       SOLIC_ATD.DS_OBSERVACAO,
       ATENDIME.CD_PRESTADOR,
       PRESTADOR.NM_PRESTADOR,
       PRESTADOR.DS_CODIGO_CONSELHO,
      SOLIC_ATD.QT_SESSOES,
      SOLIC_ATD.TP_SOLIC_ATD,
atendime.cd_atendimento 
FROM   DBAMV.SOLIC_ATD,
       DUAL,
       DBAMV.ATENDIME,
       DBAMV.CONVENIO,
       DBAMV.CARTEIRA,
       DBAMV.PACIENTE,
       DBAMV.CID,
       DBAMV.IT_SOLIC_ATD,
       DBAMV.PRO_FAT,
       DBAMV.PROCEDIMENTO_SUS,
       DBAMV.PRESTADOR
WHERE  SOLIC_ATD.CD_ATENDIMENTO  =  ATENDIME.CD_ATENDIMENTO
AND    ATENDIME.CD_CONVENIO      =  CONVENIO.CD_CONVENIO
AND    ATENDIME.CD_PACIENTE      =  CARTEIRA.CD_PACIENTE(+)
AND    ATENDIME.CD_CONVENIO      =  CARTEIRA.CD_CONVENIO(+)
and     atendime.cd_con_pla = carteira.cd_con_pla(+)
and     atendime.nr_Carteira  = carteira.nr_carteira(+)
AND    ATENDIME.CD_PACIENTE      =  PACIENTE.CD_PACIENTE
AND    ATENDIME.CD_CID           =  CID.CD_CID(+)
AND    SOLIC_ATD.CD_SOLIC_ATD    =  IT_SOLIC_ATD.CD_SOLIC_ATD
AND    IT_SOLIC_ATD.CD_PRO_FAT   =  PRO_FAT.CD_PRO_FAT(+)
AND    IT_SOLIC_ATD.CD_PROCEDIMENTO   =  PROCEDIMENTO_SUS.CD_PROCEDIMENTO(+)
AND    ATENDIME.CD_PRESTADOR     =  PRESTADOR.CD_PRESTADOR
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND SOLIC_ATD.CD_SOLIC_ATD = :CAMPO_CD_SOLIC_ATD
)
ORDER BY CD_SOLIC_ATD,DT_SOLIC_ATD,CD_CONVENIO,NM_CONVENIO,NR_CARTEIRA,DT_VALIDADE,NM_TITULAR,NM_PACIENTE,CD_CID,DS_CID,QT_SESSOES,DS_OBSERVACAO,CD_PRESTADOR,NM_PRESTADOR,DS_CODIGO_CONSELHO,TP_SOLIC_ATD,CD_ATENDIMENTO
