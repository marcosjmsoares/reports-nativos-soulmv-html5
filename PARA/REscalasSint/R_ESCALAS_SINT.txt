--R_ESCALAS_SINT
SELECT  distinct
        PRESTADOR.NM_MNEMONICO
       ,PRESTADOR.DS_CODIGO_CONSELHO
       ,PRESTADOR.CD_PRESTADOR
       ,to_char(P_ESCALA.HR_INIC, 'hh24')  hr_inic
       ,p_escala.cd_ori_ate
       ,p_escala_ser_dis.cd_ser_dis
       ,SER_DIS.DS_SER_DIS
       ,ORI_ATE.DS_ORI_ATE
FROM    Dbamv.P_ESCALA
       ,Dbamv.PRESTADOR
       ,Dbamv.ORI_ATE
       ,Dbamv.P_ESCALA_SER_DIS
       ,Dbamv.SER_DIS
WHERE P_ESCALA.CD_PRESTADOR  =  PRESTADOR.CD_PRESTADOR
AND       P_ESCALA.CD_ORI_ATE  =  ORI_ATE.CD_ORI_ATE
AND       P_ESCALA.CD_P_ESCALA  =  P_ESCALA_SER_DIS.CD_P_ESCALA
AND       P_ESCALA_SER_DIS.CD_SER_DIS  =  SER_DIS.CD_SER_DIS
ORDER BY PRESTADOR.NM_MNEMONICO
