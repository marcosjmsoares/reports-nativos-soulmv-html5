--R_EST_P_ORI
SELECT * FROM (
SELECT ori_ate.ds_ori_ate
              ,marcacao.cd_ori_ate
              ,marcacao.dt_marcacao
              ,TO_CHAR(marcacao.hr_inicial, 'hh24') hr_inicial
              ,TO_CHAR(marcacao.hr_final,   'hh24') hr_final
              ,marcacao.cd_tip_mar
              ,tip_mar.ds_tip_mar
              ,nvl(SUM(marcacao_ser_tipo.nr_consultas),0) consultas
              ,nvl(SUM(marcacao_ser_tipo.nr_marcados),0)  marcacoes
   FROM dbamv.marcacao
             ,dbamv.ori_ate
             ,dbamv.marcacao_ser_tipo
             ,dbamv.tip_mar
WHERE marcacao.cd_ori_ate  = ori_ate.cd_ori_ate
      AND marcacao.cd_marcacao = marcacao_ser_tipo.cd_marcacao
      AND tip_mar.cd_tip_mar   = marcacao_ser_tipo.cd_tip_mar
      AND marcacao.dt_marcacao between to_date(to_char(:CAMPO_DTINIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
              						   to_date(to_char(:CAMPO_DTFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')		
GROUP BY ori_ate.ds_ori_ate
                   ,marcacao.cd_ori_ate
                   ,marcacao.dt_marcacao
                   ,TO_CHAR(marcacao.hr_inicial, 'hh24')
                   ,TO_CHAR(marcacao.hr_final  , 'hh24')
                   ,marcacao.cd_tip_mar
                   ,tip_mar.ds_tip_mar
)
ORDER BY DS_ORI_ATE,DT_MARCACAO,ds_tip_mar
