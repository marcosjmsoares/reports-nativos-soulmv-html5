--R_EST_P_GERAL
SELECT * FROM (
SELECT MARCACAO.DT_MARCACAO  DT_MARCACAO,
              TIP_MAR.CD_TIP_MAR        CD_TIP_MAR,
              TIP_MAR.DS_TIP_MAR        DS_TIP_MAR,
              SUM(MARCACAO_SER_TIPO.NR_CONSULTAS) Consultas,
              nvl(SUM(MARCACAO_SER_TIPO.NR_MARCADOS),0) Marcacoes
   FROM DBAMV.MARCACAO,
              DBAMV.MARCACAO_SER_TIPO,
              DBAMV.TIP_MAR
WHERE MARCACAO.DT_MARCACAO between to_date(to_char(:CAMPO_DTINIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
							       to_date(to_char(:CAMPO_DTFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
      AND  NVL(MARCACAO.SN_ATIVO,'S')  =  'S'
      AND  MARCACAO_SER_TIPO.CD_MARCACAO =  MARCACAO.CD_MARCACAO
      AND  MARCACAO_SER_TIPO.CD_TIP_MAR  =  TIP_MAR.CD_TIP_MAR
 GROUP BY MARCACAO.DT_MARCACAO,
                    TIP_MAR.CD_TIP_MAR,
                    TIP_MAR.DS_TIP_MAR
)
ORDER BY DT_MARCACAO
