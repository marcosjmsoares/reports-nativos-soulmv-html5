--R_ATEN_EMPRESA
SELECT * FROM (
  SELECT ATEN.CD_ATENDIMENTO ATEND,
         CART.NM_EMPRESA     EMPRESA,
         PACI.NM_PACIENTE    PACIENTE,
         (SELECT PA.NM_SOCIAL_PACIENTE
          FROM DBAMV.PACIENTE PA
          WHERE PA.CD_PACIENTE = PACI.CD_PACIENTE
            AND NVL(PA.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE,
         LEIT.DS_LEITO       LEITO,
         ATEN.DT_ATENDIMENTO DT_INTERNA,
         ATEN.DT_ALTA        DT_ALTA,
         CONVENIO.NM_CONVENIO CONVENIO,
         PRES.NM_PRESTADOR    MEDICO
  FROM DBAMV.ATENDIME  ATEN,
       DBAMV.PACIENTE  PACI,
       DBAMV.CONVENIO  CONVENIO,
       DBAMV.LEITO     LEIT,
       DBAMV.CARTEIRA  CART,
       DBAMV.PRESTADOR PRES
  WHERE ATEN.DT_ATENDIMENTO BETWEEN TO_DATE(TO_CHAR(:CAMPO_DTINIC, 'DD/MM/YYYY') || ' 00', 'DD/MM/YYYY HH24') AND 
      					  TO_DATE(TO_CHAR(:CAMPO_DTFIM, 'DD/MM/YYYY') || ' 23', 'DD/MM/YYYY HH24')
    AND ATEN.CD_PACIENTE      = PACI.CD_PACIENTE
    AND ATEN.CD_CONVENIO      = CONVENIO.CD_CONVENIO
    AND ATEN.CD_PRESTADOR     = PRES.CD_PRESTADOR(+)
    AND ATEN.CD_LEITO         = LEIT.CD_LEITO(+)
    AND ATEN.NR_CARTEIRA      = CART.NR_CARTEIRA(+)
    AND ATEN.CD_PACIENTE      = CART.CD_PACIENTE(+)
    AND ATEN.CD_CONVENIO      = CART.CD_CONVENIO(+)
    AND ATEN.CD_CON_PLA       = CART.CD_CON_PLA(+)
    AND ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
    :CAMPO_CF_MULTI_EMPRESA
)
:CAMPO_CF_ORDEM, ATEND
