--R_PAC_AGD_SESSOES
SELECT * FROM (
SELECT atendime_sessao.dt_agendamento
             ,atendime_sessao.cd_atendimento
             ,paciente.cd_paciente||' - '||paciente.nm_paciente paciente
             ,paciente.nm_paciente
             ,(SELECT PA.NM_SOCIAL_PACIENTE
          		FROM DBAMV.PACIENTE PA
          		WHERE PA.CD_PACIENTE = paciente.CD_PACIENTE
            	  AND NVL(PA.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE
             ,atendime_sessao.dia_qt
             ,atendime_sessao.esquema
             ,atendime_sessao.observacao
FROM   dbamv.atendime
            ,dbamv.atendime_sessao
            ,dbamv.ser_dis
            ,dbamv.prestador
            ,dbamv.ori_ate
            ,dbamv.paciente
WHERE       atendime_sessao.cd_atendimento = atendime.cd_atendimento
AND         atendime.cd_ser_dis        = ser_dis.cd_ser_dis
AND         atendime.cd_prestador      = prestador.cd_prestador
AND         atendime.cd_ori_ate        = ori_ate.cd_ori_ate
AND         atendime.cd_paciente       = paciente.cd_paciente
AND 		atendime_sessao.dt_agendamento between to_date(to_char(:CAMPO_DT_INICIAL,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
												   to_date(to_char(:CAMPO_DT_FINAL,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND        atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
order by atendime_sessao.dt_agendamento, paciente.nm_paciente
)
