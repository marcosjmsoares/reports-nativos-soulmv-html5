--R_DESC_CIRURGICA
SELECT ATENDIME.CD_ATENDIMENTO
              ,ATENDIME.DT_ATENDIMENTO
             ,TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24')   HR_ATENDIMENTO
             ,TO_CHAR(ATENDIME.DT_ALTA,'DD/MM/YYYY') || ' - ' || TO_CHAR(ATENDIME.HR_ALTA,'HH24')   DT_ALTA
             ,ATENDIME.CD_CONVENIO
             ,ATENDIME.CD_CID
             ,CID.DS_CID
             ,ATENDIME.CD_PRO_INT
             ,DBAMV.PROCEDIAS(ATENDIME.CD_CONVENIO, ATENDIME.CD_PRO_INT, 'D')    DS_PRO_INT
             ,ATENDIME.CD_PRESTADOR
             ,PRESTADOR.NM_PRESTADOR
             ,ATENDIME.CD_PACIENTE
             ,PACIENTE.NM_PACIENTE
             ,PACIENTE.TP_SEXO
             ,DBAMV.FNC_ESTADO_CIVIL(PACIENTE.TP_ESTADO_CIVIL) TP_ESTADO_CIVIL1
             ,PACK_INTERNAMENTO.RETORNA_IDADE( PACIENTE.DT_NASCIMENTO, SYSDATE)   IDADE
             ,PACIENTE.TP_COR
             ,PACIENTE.DS_ENDERECO
             ,PACIENTE.NR_ENDERECO
             ,PACIENTE.DS_COMPLEMENTO
             ,PACIENTE.NM_BAIRRO
             ,PACIENTE.NR_FONE
             ,PACIENTE.NR_CEP
             ,(CIDADE.NM_CIDADE || '-' || CIDADE.CD_UF)  NM_CIDADE
             ,RESPONSA.NM_RESPONSAVEL
             ,RESPONSA.DS_ENDERECO                           DS_ENDERECO_RESP
             ,RESPONSA.NM_BAIRRO                                 NM_BAIRRO_RESP
             ,RESPONSA.NR_FONE                                     NR_FORNE_RESP
             ,TIP_PAREN.DS_TIP_PAREN
             ,(CID.NM_CIDADE || '-' || CID.CD_UF)  NM_CIDADE_RESP
             ,CASOS_ATD.DS_HDA
             ,CASOS_ATD.DS_EX_FIS
             ,CASOS_ATD.DS_DIAGNOSTICO
             ,CASOS_ATD.DS_TRATAMENTO
FROM    DBAMV.ATENDIME
             ,DBAMV.PACIENTE
             ,DBAMV.RESPONSA
             ,DBAMV.CIDADE
             ,DBAMV.CIDADE  CID
             ,DBAMV.TIP_PAREN
             ,DBAMV.CASOS_ATD
             ,DBAMV.CID
             ,DBAMV.PRESTADOR
WHERE ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND       ATENDIME.CD_ATENDIMENTO = RESPONSA.CD_ATENDIMENTO(+)
AND      PACIENTE.CD_CIDADE = CIDADE.CD_CIDADE(+)
AND      RESPONSA.CD_CIDADE = CID.CD_CIDADE(+)
AND      RESPONSA.CD_TIP_PAREN = TIP_PAREN.CD_TIP_PAREN(+)
AND      ATENDIME.CD_CID = CID.CD_CID(+)
AND      ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
AND      ATENDIME.CD_CASOS_ATD = CASOS_ATD.CD_CASOS_ATD(+)
AND      ATENDIME.CD_ATENDIMENTO = :CAMPO_CD_ATENDE
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
