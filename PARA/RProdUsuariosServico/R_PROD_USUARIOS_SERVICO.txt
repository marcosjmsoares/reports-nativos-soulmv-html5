--R_PROD_USUARIOS_SERVICO_R_SERVICO4_AGENDAMENTOS2
SELECT * FROM (
SELECT  it_marcacao.cd_ser_dis  cd_ser_dis2
            ,   ser_dis.ds_ser_dis  ds_ser_dis2
            ,   it_marcacao.nm_usuario  nm_usuario2
            ,   trunc(it_marcacao.dt_agendado)  dt_agendado2
            ,   it_marcacao.cd_tip_mar  cd_tip_mar2
            ,   tip_mar.ds_tip_mar  ds_tip_mar2
            ,   nvl(ds_papel,'USUÁRIO SEM PAPEL CADASTRADO' )ds_papel2
FROM     dbamv.it_marcacao
            ,  dbamv.marcacao
            ,  dbamv.tip_mar
            ,  dbasgu.usuarios
            ,  dbasgu.papel
            ,  dbamv.ser_dis
            , dbasgu.papel_usuarios
WHERE  ( nm_paciente is not null OR  cd_paciente is not null )
      AND                                  = 'N'
      AND it_marcacao.dt_agendado between to_date(to_char(:CAMPO_DTINIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
      									  to_date(to_char(:CAMPO_DTFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
      AND marcacao.cd_marcacao                 = it_marcacao.cd_marcacao
      AND it_marcacao.nm_usuario                 is not null
      AND it_marcacao.nm_usuario                 = usuarios.cd_usuario
      AND it_marcacao.cd_tip_mar                  = tip_mar.cd_tip_mar
      AND it_marcacao.cd_ser_dis                  = ser_dis.cd_ser_dis
      AND nvl(marcacao.sn_Ativo, 'S') = 'S'
      and papel.cd_papel = papel_usuarios.cd_papel
      and usuarios.cd_usuario = papel_usuarios.cd_usuario (+)
GROUP BY it_marcacao.cd_ser_dis
                 ,   ser_dis.ds_ser_dis
                 ,   it_marcacao.nm_usuario
                 ,   trunc(it_marcacao.dt_agendado)
                 ,   it_marcacao.cd_tip_mar
                 ,   tip_mar.ds_tip_mar
                 ,   ds_papel
ORDER BY   it_marcacao.nm_usuario
                   ,  it_marcacao.cd_ser_dis
                   ,   trunc(it_marcacao.dt_agendado)
                   ,   tip_mar.ds_tip_mar
)
ORDER BY CD_SER_DIS2,DS_SER_DIS2,
DT_AGENDADO2
