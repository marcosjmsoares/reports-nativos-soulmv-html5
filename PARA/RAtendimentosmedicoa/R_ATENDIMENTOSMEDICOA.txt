--R_ATENDIMENTOSMEDICOA
SELECT * FROM (
SELECT PRES.CD_PRESTADOR,
       PRES.NM_PRESTADOR,
       ATEN.CD_ATENDIMENTO,
       ATEN.CD_TIPO_INTERNACAO,
       TIPO_INTERNACAO.DS_TIPO_INTERNACAO,
       PACI.NM_PACIENTE,
       ATEN.DT_ATENDIMENTO,
       ATEN.HR_ATENDIMENTO,
       COPL.DS_CON_PLA,
       CONV.NM_CONVENIO,
       DEAT.DS_DES_ATE,
       CART.NR_CARTEIRA,
       ATEN.NR_GUIA,
       DECODE(ATEN.SN_RETORNO,'S','Primeira Consulta','Retorno') SN_RETORNO,
       PRO_FAT.DS_PRO_FAT
  FROM DBAMV.ATENDIME        ATEN,
       DBAMV.PACIENTE        PACI,
       DBAMV.CON_PLA         COPL,
       DBAMV.PRESTADOR   PRES,
       DBAMV.CONVENIO      CONV,
       DBAMV.DES_ATE         DEAT,
       DBAMV.CARTEIRA       CART,
       dbamv.TIPO_INTERNACAO,
       DBAMV.PRO_FAT
 WHERE ATEN.DT_ATENDIMENTO between to_date(to_char(:CAMPO_DATINI,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
 								   to_date(to_char(:CAMPO_DATFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
   AND ATEN.CD_PACIENTE  =  PACI.CD_PACIENTE
   AND ATEN.CD_CONVENIO  =  CONV.CD_CONVENIO
   AND ATEN.CD_CONVENIO  =  COPL.CD_CONVENIO(+)
   AND ATEN.CD_CON_PLA   =  COPL.CD_CON_PLA(+)
   AND ATEN.CD_PRESTADOR =  PRES.CD_PRESTADOR(+)
   AND ATEN.CD_DES_ATE   =  DEAT.CD_DES_ATE(+)
   AND ATEN.CD_PACIENTE  =  CART.CD_PACIENTE(+)
   AND ATEN.CD_CONVENIO  =  CART.CD_CONVENIO(+)
   AND ATEN.CD_CON_PLA    =  CART.CD_CON_PLA(+)
   AND ATEN.NR_CARTEIRA  =  CART.NR_CARTEIRA(+)
   AND ATEN.CD_PRO_INT     =  PRO_FAT.CD_PRO_FAT(+)
   AND ATEN.CD_TIPO_INTERNACAO = TIPO_INTERNACAO.CD_TIPO_INTERNACAO(+)
   AND ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
ORDER BY PRES.NM_PRESTADOR,
                    TRUNC(ATEN.DT_ATENDIMENTO),
                    TO_CHAR(ATEN.HR_ATENDIMENTO,'hh24'),
                    PACI.NM_PACIENTE
)
ORDER BY CD_PRESTADOR,NM_PRESTADOR,
         CD_ATENDIMENTO, NM_PACIENTE, DT_ATENDIMENTO,
         HR_ATENDIMENTO, DS_CON_PLA,
         NM_CONVENIO, DS_TIPO_INTERNACAO,
         DS_DES_ATE,  NR_CARTEIRA,  NR_GUIA,
         SN_RETORNO,  DS_PRO_FAT
