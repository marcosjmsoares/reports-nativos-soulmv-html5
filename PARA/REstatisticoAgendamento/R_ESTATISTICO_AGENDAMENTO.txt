--R_ESTATISTICO_AGENDAMENTO
SELECT decode(atendime.cd_atendimento, null, decode(:CAMPO_CD_COLUNA, null, 0, :CAMPO_CD_COLUNA), :CAMPO_CD_COLUNA2)    CODIGO
              ,decode(atendime.cd_atendimento, null, decode(:CAMPO_DS_COLUNA, null, 'NÃO INFORMADO', :CAMPO_DS_COLUNA), :CAMPO_DS_COLUNA2)      DESCRICAO
              ,COUNT(*)                 QT_AGENDADAS
FROM DBAMV.MARCACAO
    ,DBAMV.IT_MARCACAO
    ,DBAMV.CONVENIO
    ,DBAMV.SER_DIS
    ,DBAMV.PRESTADOR
    ,DBAMV.PACIENTE
    ,DBAMV.ATENDIME
    ,DBAMV.CONVENIO  CONVENIO2
WHERE     MARCACAO.DT_MARCACAO between to_date(to_char(:CAMPO_DT_INI,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
							           to_date(to_char(:CAMPO_DT_FIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND       MARCACAO.CD_MARCACAO = IT_MARCACAO.CD_MARCACAO
AND       nvl(marcacao.sn_ativo, 'S') = 'S'
AND       (IT_MARCACAO.CD_PACIENTE is not null
OR        IT_MARCACAO.NM_PACIENTE is not null )
AND       IT_MARCACAO.CD_PACIENTE = PACIENTE.CD_PACIENTE(+)
AND       IT_MARCACAO.CD_CONVENIO = CONVENIO.CD_CONVENIO(+)
AND       IT_MARCACAO.CD_SER_DIS  = SER_DIS.CD_SER_DIS(+)
AND       IT_MARCACAO.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO(+)
AND       ATENDIME.CD_CONVENIO   = CONVENIO2.CD_CONVENIO(+)
AND       MARCACAO.CD_PRESTADOR   = PRESTADOR.CD_PRESTADOR(+)
AND      ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY  decode(atendime.cd_atendimento, null, decode(:CAMPO_CD_COLUNA, null, 0, :CAMPO_CD_COLUNA), :CAMPO_CD_COLUNA2)
                   ,decode(atendime.cd_atendimento, null, decode(:CAMPO_DS_COLUNA, null, 'NÃO INFORMADO', :CAMPO_DS_COLUNA), :CAMPO_DS_COLUNA2)
