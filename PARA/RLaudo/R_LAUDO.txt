--R_LAUDO
SELECT TEXTOS_ATD.DS_TEXTOS_ATD,
               ATENDIME.CD_CID,
               ATENDIME.CD_PRESTADOR,
              ATENDIME.CD_PACIENTE,
              CID.DS_CID,
              PRESTADOR.NM_PRESTADOR,
              PRESTADOR.DS_CODIGO_CONSELHO,
              SYSDATE  DATA_LAUDO,
              PACIENTE.NM_PACIENTE,
              PACIENTE.DT_NASCIMENTO
FROM    DBAMV.TEXTOS_ATD,
              DBAMV.ATENDIME,
              DBAMV.CID,
              DBAMV.PRESTADOR,
              DBAMV.PACIENTE,
              DBAMV.CASOS_ATD
WHERE  TEXTOS_ATD.TP_TEXTOS_ATD    = 'L'
AND    TEXTOS_ATD.CD_CASOS_ATD     = CASOS_ATD.CD_CASOS_ATD
AND        ATENDIME.CD_CID                          = CID.CD_CID(+)
AND     CASOS_ATD.CD_PRESTADOR     =  PRESTADOR.CD_PRESTADOR
AND        ATENDIME.CD_PACIENTE               =  PACIENTE.CD_PACIENTE
AND        ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND    ATENDIME.CD_ATENDIMENTO  = CASOS_ATD.CD_ATENDIMENTO
