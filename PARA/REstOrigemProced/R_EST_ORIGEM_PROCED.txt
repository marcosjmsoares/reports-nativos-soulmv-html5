--R_EST_ORIGEM_PROCED
SELECT ORI_ATE.CD_ORI_ATE
              ,ORI_ATE.DS_ORI_ATE
              ,COUNT(*)  QT_ORIGEM
FROM   DBAMV.ATENDIME
            ,DBAMV.ORI_ATE
WHERE  ATENDIME.DT_ATENDIMENTO between to_date(to_char(:CAMPO_DATINI,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
									   to_date(to_char(:CAMPO_DATFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND        ATENDIME.CD_ORI_ATE = ORI_ATE.CD_ORI_ATE
AND  ATENDIME.CD_MULTI_EMPRESA  = ORI_ATE.CD_MULTI_EMPRESA
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY ORI_ATE.CD_ORI_ATE
                   ,ORI_ATE.DS_ORI_ATE
ORDER BY QT_ORIGEM   DESC
