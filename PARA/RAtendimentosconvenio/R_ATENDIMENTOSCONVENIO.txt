--R_ATENDIMENTOSCONVENIO
SELECT * FROM dbamv.v_r_atendimentos_convenio
        WHERE DT_ATENDIMENTO between to_date(to_CHAR(:CAMPO_DATINI,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') 
		AND to_date(to_CHAR(:CAMPO_DATFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
  :CAMPO_CF_ORDEM ,CD_CONVENIO
