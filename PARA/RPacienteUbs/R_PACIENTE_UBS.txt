--R_PACIENTE_UBS
SELECT * FROM (
SELECT UNIDADE_REGIONAL.CD_UNIDADE_REGIONAL
      ,UNIDADE_REGIONAL.DS_UNIDADE_REGIONAL
      ,TO_CHAR(MARCACAO.DT_MARCACAO,'DD/MM/YYYY')||' '||TO_CHAR(IT_MARCACAO.HR_AGENDADO,'HH24')
      ,PACIENTE.CD_PACIENTE
      ,PACIENTE.NM_PACIENTE
      ,SER_DIS.CD_SER_DIS codigo
      ,SER_DIS.DS_SER_DIS descricao
  FROM DBAMV.UNIDADE_REGIONAL
      ,DBAMV.SER_DIS
      ,DBAMV.IT_MARCACAO
      ,DBAMV.MARCACAO_SER_TIPO
      ,DBAMV.MARCACAO
      ,DBAMV.PACIENTE
 WHERE SER_DIS.CD_SER_DIS = MARCACAO_SER_TIPO.CD_SER_DIS
   AND UNIDADE_REGIONAL.CD_UNIDADE_REGIONAL = IT_MARCACAO.CD_UNIDADE_REGIONAL_UBS
   AND MARCACAO.CD_MARCACAO =  IT_MARCACAO.CD_MARCACAO
   AND IT_MARCACAO.SN_LIBERA_VAGA_UBS = 'S'
   AND PACIENTE.CD_PACIENTE = IT_MARCACAO.CD_PACIENTE
   AND MARCACAO_SER_TIPO.CD_MARCACAO = IT_MARCACAO.CD_MARCACAO
   AND CD_DIVISAO_UNIDADE_REGIONAL IS NOT NULL
 GROUP BY UNIDADE_REGIONAL.CD_UNIDADE_REGIONAL
         ,UNIDADE_REGIONAL.DS_UNIDADE_REGIONAL
         ,TO_CHAR(MARCACAO.DT_MARCACAO,'DD/MM/YYYY')||' '||TO_CHAR(IT_MARCACAO.HR_AGENDADO,'HH24')
         ,PACIENTE.CD_PACIENTE
         ,PACIENTE.NM_PACIENTE
         ,SER_DIS.CD_SER_DIS
         ,SER_DIS.DS_SER_DIS
UNION ALL
SELECT UNIDADE_REGIONAL.CD_UNIDADE_REGIONAL
      ,UNIDADE_REGIONAL.DS_UNIDADE_REGIONAL
      ,TO_CHAR(ESCALA_DIA.DT_ESCALA_DIA,'DD/MM/YYYY')||' '||TO_CHAR(IT_ESCALA_DIA.HR_AGENDA,'HH24')
      ,PACIENTE.CD_PACIENTE
      ,PACIENTE.NM_PACIENTE
      ,RECURSO.CD_RECURSO  codigo
      ,RECURSO.NM_RECURSO  descricao
  FROM DBAMV.UNIDADE_REGIONAL
      ,DBAMV.RECURSO
      ,DBAMV.IT_ESCALA_DIA
      ,DBAMV.ESCALA_DIA
      ,DBAMV.PACIENTE
 WHERE RECURSO.CD_RECURSO = ESCALA_DIA.CD_RECURSO
   AND UNIDADE_REGIONAL.CD_UNIDADE_REGIONAL = IT_ESCALA_DIA.CD_UNIDADE_REGIONAL_UBS
   AND IT_ESCALA_DIA.SN_LIBERA_VAGA_UBS = 'S'
   AND IT_ESCALA_DIA.CD_ESCALA_DIA = ESCALA_DIA.CD_ESCALA_DIA
   AND PACIENTE.CD_PACIENTE = IT_ESCALA_DIA.CD_PACIENTE
   AND CD_DIVISAO_UNIDADE_REGIONAL IS NOT NULL
 GROUP BY UNIDADE_REGIONAL.CD_UNIDADE_REGIONAL
         ,UNIDADE_REGIONAL.DS_UNIDADE_REGIONAL
         ,TO_CHAR(ESCALA_DIA.DT_ESCALA_DIA,'DD/MM/YYYY')||' '||TO_CHAR(IT_ESCALA_DIA.HR_AGENDA,'HH24')
         ,PACIENTE.CD_PACIENTE
         ,PACIENTE.NM_PACIENTE
         ,RECURSO.CD_RECURSO
         ,RECURSO.NM_RECURSO
)
ORDER BY CD_UNIDADE_REGIONAL,DS_UNIDADE_REGIONAL
