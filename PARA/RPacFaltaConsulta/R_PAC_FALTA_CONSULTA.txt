--R_PAC_FALTA_CONSULTA
SELECT  MARCACAO.CD_MARCACAO CD_MARCACAO,
  NVL(
    (CASE WHEN (IT_MARCACAO.CD_PACIENTE IS NOT NULL AND
    IT_MARCACAO.CD_CONVENIO IS NOT NULL AND
    IT_MARCACAO.CD_CON_PLA  IS NOT NULL) THEN
    (SELECT NVL(CARTEIRA.NM_EMPRESA, 'EMPRESA NÃO ENCONTRADA')
      FROM DBAMV.CARTEIRA
      WHERE CARTEIRA.CD_PACIENTE = IT_MARCACAO.CD_PACIENTE
        AND CARTEIRA.CD_CONVENIO = IT_MARCACAO.CD_CONVENIO
        AND CARTEIRA.CD_CON_PLA  = IT_MARCACAO.CD_CON_PLA
        AND CARTEIRA.DT_VALIDADE = (SELECT MAX(DT_VALIDADE)
      FROM DBAMV.CARTEIRA CART
      WHERE CART.CD_PACIENTE = CARTEIRA.CD_PACIENTE
        AND CART.CD_CONVENIO = CARTEIRA.CD_CONVENIO
        AND CART.CD_CON_PLA =  CARTEIRA.CD_CON_PLA )
        AND ROWNUM = 1)
    END)
    ,'EMPRESA NÃO ENCONTRADA' ) EMPRESA
  ,MARCACAO.DT_MARCACAO      DT_MARCACAO
  ,IT_MARCACAO.CD_PACIENTE   CD_PACIENTE
  ,IT_MARCACAO.NM_PACIENTE   NM_PACIENTE
  ,(SELECT PA.NM_SOCIAL_PACIENTE
      FROM DBAMV.PACIENTE PA
      WHERE PA.CD_PACIENTE = IT_MARCACAO.CD_PACIENTE
        AND NVL(PA.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE
  ,IT_MARCACAO.NR_TELEFONE   NR_TELEFONE
  ,TO_CHAR(IT_MARCACAO.HR_AGENDADO,'HH24') HR_AGENDADO
  ,SER_DIS.CD_SER_DIS CD_SER_DIS
  ,SER_DIS.DS_SER_DIS DS_SER_DIS
  ,1 CONT
FROM DBAMV.MARCACAO      MARCACAO
    ,DBAMV.UNIDADE_REGIONAL  UNIDADE_REGIONAL
    ,DBAMV.IT_MARCACAO       IT_MARCACAO
    ,DBAMV.SER_DIS
WHERE UNIDADE_REGIONAL.CD_UNIDADE_REGIONAL(+) = MARCACAO.CD_UNIDADE_REGIONAL
  AND MARCACAO.CD_MARCACAO = IT_MARCACAO.CD_MARCACAO
  AND IT_MARCACAO.CD_SER_DIS = SER_DIS.CD_SER_DIS(+)
  AND IT_MARCACAO.NM_PACIENTE IS NOT NULL
  AND IT_MARCACAO.CD_ATENDIMENTO IS NULL
  AND	MARCACAO.DT_MARCACAO BETWEEN TO_DATE(TO_CHAR(:CAMPO_DT_INICIAL,'DD/MM/YYYY')||' 00','DD/MM/YYYY HH24') 
      AND TO_DATE(TO_CHAR(:CAMPO_DT_FINAL,'DD/MM/YYYY')||' 23','DD/MM/YYYY HH24')
ORDER BY
NVL(
  (CASE WHEN (IT_MARCACAO.CD_PACIENTE IS NOT NULL AND
    IT_MARCACAO.CD_CONVENIO IS NOT NULL AND
    IT_MARCACAO.CD_CON_PLA  IS NOT NULL) THEN
    (SELECT NVL(CARTEIRA.NM_EMPRESA, 'EMPRESA NÃO ENCONTRADA')
  FROM DBAMV.CARTEIRA
  WHERE CARTEIRA.CD_PACIENTE = IT_MARCACAO.CD_PACIENTE
    AND CARTEIRA.CD_CONVENIO = IT_MARCACAO.CD_CONVENIO
    AND CARTEIRA.CD_CON_PLA  = IT_MARCACAO.CD_CON_PLA
    AND CARTEIRA.DT_VALIDADE = (SELECT MAX(DT_VALIDADE)
  FROM DBAMV.CARTEIRA CART
  WHERE CART.CD_PACIENTE = CARTEIRA.CD_PACIENTE
    AND CART.CD_CONVENIO = CARTEIRA.CD_CONVENIO
    AND CART.CD_CON_PLA =  CARTEIRA.CD_CON_PLA )
    AND ROWNUM = 1)
  END)
  ,'EMPRESA NÃO ENCONTRADA' )
  ,DS_SER_DIS
  ,DT_MARCACAO
  ,TO_DATE(TO_CHAR(IT_MARCACAO.HR_AGENDADO,'HH24'),'HH24')
  ,NM_PACIENTE
  ,NR_TELEFONE
