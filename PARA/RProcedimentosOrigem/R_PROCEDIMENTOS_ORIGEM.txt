--R_PROCEDIMENTOS_ORIGEM
SELECT * FROM (
SELECT    ORI.DS_ORI_ATE        DS_ORIGEM,
          CID.NM_CIDADE         NM_CIDADE,
          CID.CD_UF             CD_UF,
          PSUS.CD_PROCEDIMENTO CD_SSM,
          PSUS.DS_PROCEDIMENTO DS_TITULO,
          COUNT(*)              QTD
FROM      DBAMV.ATENDIME    ATE,
          DBAMV.ORI_ATE     ORI,
          DBAMV.PACIENTE    PAC,
          DBAMV.CIDADE      CID,
          DBAMV.PROCEDIMENTO_SUS PSUS
WHERE     ATE.DT_ATENDIMENTO   >= TO_DATE( TO_CHAR( :CAMPO_DT_INICIAL, 'DD/MM/YYYY' ) || ' 00', 'DD/MM/YYYY HH24' )
AND       ATE.DT_ATENDIMENTO   <= TO_DATE( TO_CHAR( :CAMPO_DT_FINAL  , 'DD/MM/YYYY' ) || ' 23', 'DD/MM/YYYY HH24' )
AND       ATE.CD_ORI_ATE       =  ORI.CD_ORI_ATE
AND       ATE.CD_PACIENTE      =  PAC.CD_PACIENTE
AND       PAC.CD_CIDADE        =  CID.CD_CIDADE
AND       ATE.CD_PROCEDIMENTO = PSUS.CD_PROCEDIMENTO
GROUP BY  ATE.CD_ORI_ATE,
          ORI.DS_ORI_ATE,
          CID.NM_CIDADE,
          CID.CD_UF,
          PSUS.CD_PROCEDIMENTO,
          PSUS.DS_PROCEDIMENTO
)
ORDER BY NM_CIDADE,
DS_ORIGEM,CD_UF
