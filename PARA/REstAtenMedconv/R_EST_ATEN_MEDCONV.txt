--R_EST_ATEN_MEDCONV
SELECT * FROM (
SELECT ATENDIME.CD_PRESTADOR
              ,ATENDIME.CD_CONVENIO
              ,PRESTADOR.NM_PRESTADOR
              ,CONVENIO.NM_CONVENIO
              ,COUNT(*)
FROM    DBAMV.ATENDIME
             ,DBAMV.PRESTADOR
             ,DBAMV.CONVENIO
WHERE ATENDIME.DT_ATENDIMENTO  BETWEEN to_date(to_char(:CAMPO_DATINI,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND to_date(to_char(:CAMPO_DATFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND       CONVENIO.CD_CONVENIO  =  ATENDIME.CD_CONVENIO
AND       PRESTADOR.CD_PRESTADOR(+) = ATENDIME.CD_PRESTADOR
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY PRESTADOR.NM_PRESTADOR
                   ,ATENDIME.CD_PRESTADOR
                   ,CONVENIO.NM_CONVENIO
                   ,ATENDIME.CD_CONVENIO
)
ORDER BY NM_PRESTADOR,CD_PRESTADOR
