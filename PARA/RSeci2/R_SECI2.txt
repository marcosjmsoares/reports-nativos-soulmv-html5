SELECT COUNT(ATENDIME.CD_SER_DIS) QTDATE
              ,decode(CIDADE.NM_CIDADE, null, 'NÃO INFORMADA', ltrim (CIDADE.NM_CIDADE)  ) NM_CIDADE
       ,decode(CIDADE.CD_UF, null , 'NI', cidade.cd_uf)  CD_UF
       ,SER_DIS.DS_SER_DIS DS_SER_DIS
       ,:CAMPO_FC_PERCENTUAL PERCT
FROM    Dbamv.ATENDIME ATENDIME
       ,Dbamv.PACIENTE PACIENTE
       ,Dbamv.CIDADE CIDADE
       ,Dbamv.SER_DIS SER_DIS
WHERE PACIENTE.CD_PACIENTE = ATENDIME.CD_PACIENTE
AND   CIDADE.CD_CIDADE(+) = PACIENTE.CD_CIDADE
AND   SER_DIS.CD_SER_DIS = ATENDIME.CD_SER_DIS
AND   ATENDIME.TP_ATENDIMENTO = 'A'
AND   ATENDIME.DT_ATENDIMENTO BETWEEN to_date(to_char(:CAMPO_DT_INIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
		     						  to_date(to_char(:CAMPO_DT_FIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')      
GROUP BY CIDADE.CD_UF
                   ,CIDADE.NM_CIDADE
                   ,SER_DIS.DS_SER_DIS
ORDER BY PERCT DESC
