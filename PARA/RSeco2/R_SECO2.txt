--R_SECO2
SELECT * FROM (
SELECT  COUNT(ATENDIME.CD_CONVENIO) COUNT_ATENDIME
       ,ATENDIME.CD_SER_DIS CD_SER_DIS
       ,SER_DIS.DS_SER_DIS DS_SER_DIS
       ,SER_DIS.CD_SER_DIS CD_SER_DIS1
       ,ATENDIME.CD_CONVENIO CD_CONVENIO1
       ,CONVENIO.NM_CONVENIO NM_CONVENIO
FROM    Dbamv.ATENDIME ATENDIME
       ,Dbamv.SER_DIS SER_DIS
       ,Dbamv.CONVENIO CONVENIO
WHERE CONVENIO.CD_CONVENIO = ATENDIME.CD_CONVENIO
AND   SER_DIS.CD_SER_DIS = ATENDIME.CD_SER_DIS
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND   ATENDIME.TP_ATENDIMENTO = 'A'
AND  ATENDIME.DT_ATENDIMENTO BETWEEN to_date(to_char(:CAMPO_DT_INIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
									 to_date(to_char(:CAMPO_DT_FIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')      
GROUP BY ATENDIME.CD_SER_DIS,
         SER_DIS.DS_SER_DIS,
         SER_DIS.CD_SER_DIS,
         ATENDIME.CD_CONVENIO,
         CONVENIO.NM_CONVENIO
ORDER BY SER_DIS.DS_SER_DIS
        ,CONVENIO.NM_CONVENIO
)
ORDER BY DS_SER_DIS,CD_SER_DIS,CD_SER_DIS1
