--R_SOS_CIRU
SELECT * FROM (
SELECT  paciente.cd_paciente,
        paciente.nm_paciente,
        sos_it_agenda_ciru.dt_agenda,
        sos_it_agenda_ciru.cd_pro_fat,
        pro_fat.ds_pro_fat,
        sos_cirurgia.ds_cirurgia,
        sos_cirurgia.ds_risco_cirurgico,
        sos_cirurgia.ds_intercorrencia
FROM    dbamv.sos_cirurgia        sos_cirurgia,
        dbamv.atendime            atendime,
        dbamv.paciente            paciente,
        dbamv.sos_it_agenda_ciru  sos_it_agenda_ciru,
        dbamv.pro_fat             pro_fat
WHERE   sos_cirurgia.cd_atendimento   =  :CAMPO_CD_ATENDIMENTO
   and  sos_cirurgia.cd_atendimento   =  atendime.cd_atendimento
   and  atendime.cd_paciente          =  paciente.cd_paciente
   and  atendime.cd_atendimento       =  sos_it_agenda_ciru.cd_atendimento
   and  sos_it_agenda_ciru.cd_pro_fat =  pro_fat.cd_pro_fat(+)
:CAMPO_CF_MULTI_EMPRESA
)
ORDER BY DT_AGENDA
