--R_AGE_OPER
SELECT * FROM (
SELECT it_marcacao.nm_usuario
            ,   trunc(it_marcacao.dt_agendado)  dt_agendado
            ,   it_marcacao.cd_tip_mar
            ,   tip_mar.ds_tip_mar
            ,   ds_papel
FROM     dbamv.it_marcacao
            ,  dbamv.marcacao
            ,  dbamv.tip_mar
            ,  dbasgu.usuarios
            ,  dbasgu.papel
            ,  dbasgu.papel_usuarios
WHERE (it_marcacao.nm_paciente is not null
        OR it_marcacao.cd_paciente is not null )
      AND marcacao.cd_marcacao                 = it_marcacao.cd_marcacao
      AND it_marcacao.nm_usuario                 is not null
      AND it_marcacao.nm_usuario                 = papel_usuarios.cd_usuario
      AND papel_usuarios.cd_usuario  LIKE                 :CAMPO_USUARIO
      AND it_marcacao.cd_tip_mar                  = tip_mar.cd_tip_mar
      AND nvl(marcacao.sn_ativo, 'S') = 'S'
      AND papel_usuarios.cd_papel                          =  papel.cd_papel
      AND papel_usuarios.cd_papel                          LIKE  :CAMPO_CD_PAPEL
      AND papel_usuarios.cd_usuario               =   usuarios.cd_usuario
      AND it_marcacao.dt_agendado between to_date(to_char(:CAMPO_DTINIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
      									  to_date(to_char(:CAMPO_DTFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
GROUP BY  it_marcacao.nm_usuario
                 ,   trunc(it_marcacao.dt_agendado)
                 ,   it_marcacao.cd_tip_mar
                 ,   tip_mar.ds_tip_mar
                 ,   ds_papel
)
ORDER BY DS_PAPEL,NM_USUARIO,
DT_AGENDADO
