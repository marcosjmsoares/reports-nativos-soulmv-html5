--R_EST_P_ORI_S
SELECT ori_ate.ds_ori_ate
              ,ori_ate.cd_ori_ate
              ,tip_mar.ds_tip_mar
              ,marcacao_ser_tipo.cd_tip_mar
              ,SUM(marcacao_ser_tipo.nr_consultas) totcons
              ,SUM(marcacao_ser_tipo.nr_marcados) totmarc
   FROM dbamv.marcacao
             ,dbamv.marcacao_ser_tipo
             ,dbamv.tip_mar
             ,dbamv.ori_ate
WHERE marcacao.cd_ori_ate  = ori_ate.cd_ori_ate
      AND marcacao.cd_marcacao = marcacao_ser_tipo.cd_marcacao
      AND tip_mar.cd_tip_mar   = marcacao_ser_tipo.cd_tip_mar
      AND dt_marcacao between to_date(to_char(:CAMPO_DTINIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
      						  to_date(to_char(:CAMPO_DTFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
 GROUP BY ori_ate.ds_ori_ate
                   ,ori_ate.cd_ori_ate
                   ,tip_mar.ds_tip_mar
                   ,marcacao_ser_tipo.cd_tip_mar
 ORDER BY ori_ate.ds_ori_ate
              ,ori_ate.cd_ori_ate
              ,tip_mar.ds_tip_mar
              ,marcacao_ser_tipo.cd_tip_mar
              ,totcons
              ,totmarc
