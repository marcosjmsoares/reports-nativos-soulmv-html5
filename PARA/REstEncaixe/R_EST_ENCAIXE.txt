--R_EST_ENCAIXE
SELECT * FROM (
SELECT  PRESTADOR.CD_PRESTADOR
               ,PRESTADOR.NM_PRESTADOR
               ,MARCACAO.DT_MARCACAO
               ,COUNT(*)  QT_ENCAIXE
FROM      DBAMV.MARCACAO
               ,DBAMV.IT_MARCACAO  IT_MARCA
               ,DBAMV.PRESTADOR
WHERE MARCACAO.DT_MARCACAO between to_date(to_char(:CAMPO_DT_INI,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
								   to_date(to_char(:CAMPO_DT_FIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND       MARCACAO.SN_ATIVO = 'S'
AND       MARCACAO.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
AND       MARCACAO.CD_MARCACAO = IT_MARCA.CD_MARCACAO
AND       NVL(IT_MARCA.SN_ENCAIXE,'N') = 'S'
GROUP BY PRESTADOR.CD_PRESTADOR
                   ,PRESTADOR.NM_PRESTADOR
                   ,MARCACAO.DT_MARCACAO
)
ORDER BY CD_PRESTADOR,NM_PRESTADOR, DT_MARCACAO, QT_ENCAIXE
