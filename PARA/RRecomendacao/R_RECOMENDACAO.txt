--R_RECOMENDACAO
SELECT * FROM (
SELECT  SOLIC_ATD.CD_SOLIC_ATD,
               CARTEIRA.NR_CARTEIRA,
               ATENDIME.CD_CONVENIO,
               CONVENIO.NM_CONVENIO,
               ATENDIME.CD_PACIENTE,
               PACIENTE.NM_PACIENTE,
               CARTEIRA.NM_TITULAR,
               CARTEIRA.DT_VALIDADE,
               ATENDIME.CD_PRESTADOR,
               CIRURGIAO.NM_PRESTADOR    CIRURGIAO,
               CIRURGIAO.DS_CODIGO_CONSELHO,
               IT_SOLIC_ATD.CD_PRO_FAT,
               DECODE(CONVENIO.TP_CONVENIO, 'A', TAB_SIASUS.DS_TITULO, PRO_FAT.DS_PRO_FAT)    DS_PRO_FAT,
               IT_SOLIC_ATD.CD_CID,
               CID.DS_CID,
               SOLIC_ATD.DS_OBSERVACAO,
               SOLIC_ATD.QT_DIAS_INTERN,
               SYSDATE
FROM    DBAMV.SOLIC_ATD,
              DBAMV.IT_SOLIC_ATD,
              DBAMV.ATENDIME,
              DBAMV.CARTEIRA,
              DBAMV.CONVENIO,
              DBAMV.PACIENTE,
              DBAMV.PRESTADOR  CIRURGIAO,
              DBAMV.CID,
              DBAMV.PRO_FAT,
              DBAMV.TAB_SIASUS,
              DUAL
WHERE  SOLIC_ATD.CD_SOLIC_ATD         = :CAMPO_CD_SOLIC_ATD
AND        SOLIC_ATD.CD_SOLIC_ATD         = IT_SOLIC_ATD.CD_SOLIC_ATD
AND        SOLIC_ATD.CD_ATENDIMENTO   = ATENDIME.CD_ATENDIMENTO
AND        ATENDIME.CD_PACIENTE            = CARTEIRA.CD_PACIENTE(+)
AND        ATENDIME.CD_CONVENIO           = CARTEIRA.CD_CONVENIO(+)
AND        ATENDIME.CD_CONVENIO           = CONVENIO.CD_CONVENIO(+)
AND        ATENDIME.CD_PACIENTE            = PACIENTE.CD_PACIENTE
AND        ATENDIME.CD_PRESTADOR        = CIRURGIAO.CD_PRESTADOR
AND        IT_SOLIC_ATD.CD_CID                 = CID.CD_CID(+)
AND        IT_SOLIC_ATD.CD_PRO_FAT       = PRO_FAT.CD_PRO_FAT(+)
AND        IT_SOLIC_ATD.CD_PRO_FAT       = TAB_SIASUS.CD_SSM(+)
:CAMPO_CF_MULTI_EMPRESA
)
ORDER BY CD_SOLIC_ATD,NR_CARTEIRA,CD_CONVENIO,NM_CONVENIO,CD_PACIENTE,NM_PACIENTE,NM_TITULAR,DT_VALIDADE,CD_PRESTADOR,CIRURGIAO,DS_OBSERVACAO,QT_DIAS_INTERN,DS_CODIGO_CONSELHO,SYSDATE
