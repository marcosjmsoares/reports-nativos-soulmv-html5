--R_AVALIACAO
SELECT  PACIENTE.NM_PACIENTE,
               PRESTADOR.NM_PRESTADOR,
               ATENDIME.DT_ATENDIMENTO
FROM    DBAMV.SOLIC_ATD,
              DBAMV.ATENDIME,
              DBAMV.PACIENTE,
              DBAMV.PRESTADOR
WHERE  SOLIC_ATD.CD_SOLIC_ATD         = :CAMPO_CD_SOLIC_ATD
AND        SOLIC_ATD.CD_ATENDIMENTO   = ATENDIME.CD_ATENDIMENTO
AND        ATENDIME.CD_PACIENTE            = PACIENTE.CD_PACIENTE
AND       ATENDIME.CD_PRESTADOR  = PRESTADOR.CD_PRESTADOR
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
