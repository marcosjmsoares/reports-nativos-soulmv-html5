--R_LISTA_PACIENTES_AGENDA
SELECT * FROM (
SELECT PLANTAO.CODIGO_AGENDA
               ,PRESTADOR.NM_PRESTADOR
               ,PLANTAO.CD_RECURSO
               ,UNIDADE_REGIONAL.DS_UNIDADE_REGIONAL
               ,TO_CHAR(PLANTAO.DT_MARCACAO,'DD/MM/YYYY') DT_MARCACAO
               ,TO_CHAR(HORARIO.HORA_AGENDA,'hh24') HORA_AGENDA
               ,DECODE(PLANTAO.TIPO_AGENDA,'L','LAB')TIPO_AGENDA
                ,HORARIO.CD_PACIENTE
               ,HORARIO.NM_PACIENTE
               ,HORARIO.NR_TELEFONE
               ,NVL(HORARIO.SN_ENCAIXE,'N') SN_ENCAIXE
               ,SER_DIS.DS_SER_DIS
               ,TIP_MAR.DS_TIP_MAR
               ,CONVENIO.NM_CONVENIO
               ,CON_PLA.DS_CON_PLA
  FROM DBAMV.V_CENTRAL_MARCACAO_PLANTAO PLANTAO
              ,DBAMV.V_CENTRAL_MARCACAO_HORARIO HORARIO
              ,DBAMV.CONVENIO
              ,DBAMV.CON_PLA
              ,DBAMV.PRESTADOR
              ,DBAMV.TIP_MAR
              ,DBAMV.SER_DIS
              ,DBAMV.UNIDADE_REGIONAL
 WHERE PLANTAO.CODIGO_AGENDA                   = HORARIO.CODIGO_AGENDA
      AND  PLANTAO.TIPO_AGENDA                         = HORARIO.TIPO_AGENDA
      AND  PLANTAO.CD_PRESTADOR                     = PRESTADOR.CD_PRESTADOR(+)
      AND  PLANTAO.CD_SER_DIS                            = HORARIO.CD_SER_DIS(+)
      AND  PLANTAO.CD_TIP_MAR                            = HORARIO.CD_TIP_MAR(+)
      AND  HORARIO.CD_TIP_MAR                            = TIP_MAR.CD_TIP_MAR(+)
      AND  HORARIO.CD_SER_DIS                            = SER_DIS.CD_SER_DIS(+)
      AND  PLANTAO.CD_UNIDADE_REGIONAL       = UNIDADE_REGIONAL.CD_UNIDADE_REGIONAL(+)
      AND  HORARIO.CD_CONVENIO                          = CONVENIO.CD_CONVENIO(+)
      AND  HORARIO.CD_CONVENIO                          = CON_PLA.CD_CONVENIO(+)
      AND  HORARIO.CD_CON_PLA                            = CON_PLA.CD_CON_PLA(+)
      AND  TO_DATE(TO_CHAR(PLANTAO.DT_MARCACAO, 'DD/MM/YYYY')||' '||TO_CHAR(HORARIO.HORA_AGENDA, 'HH24'), 'DD/MM/YYYY HH24') BETWEEN  TO_DATE(:CAMPO_PDT_INICIAL,'DD/MM/YYYY HH24') AND TO_DATE(:CAMPO_PDT_FINAL,'DD/MM/YYYY HH24')
      AND  HORARIO.NM_PACIENTE IS NOT NULL
      AND  PLANTAO.TIPO_AGENDA                          = 'L'
     AND   PLANTAO.SN_ATIVO                                  = 'S'
     AND  (:CAMPO_PTP_AGENDAMENTO                                = 'T'
       OR                                   = 'L')
UNION
SELECT PLANTAO.CODIGO_AGENDA
               ,PRESTADOR.NM_PRESTADOR
               ,PLANTAO.CD_RECURSO
               ,UNIDADE_REGIONAL.DS_UNIDADE_REGIONAL
               ,TO_CHAR(PLANTAO.DT_MARCACAO,'DD/MM/YYYY') DT_MARCACAO
               ,TO_CHAR(HORARIO.HORA_AGENDA,'hh24') HORA_AGENDA
               ,DECODE(PLANTAO.TIPO_AGENDA,'I','IMAG')   TIPO_AGENDA
                ,HORARIO.CD_PACIENTE
               ,HORARIO.NM_PACIENTE
               ,HORARIO.NR_TELEFONE
               ,NVL(HORARIO.SN_ENCAIXE,'N') SN_ENCAIXE
               ,SER_DIS.DS_SER_DIS
               ,TIP_MAR.DS_TIP_MAR
               ,CONVENIO.NM_CONVENIO
               ,CON_PLA.DS_CON_PLA
  FROM DBAMV.V_CENTRAL_MARCACAO_PLANTAO PLANTAO
              ,DBAMV.V_CENTRAL_MARCACAO_HORARIO HORARIO
              ,DBAMV.CONVENIO
              ,DBAMV.CON_PLA
              ,DBAMV.PRESTADOR
              ,DBAMV.TIP_MAR
              ,DBAMV.SER_DIS
              ,DBAMV.UNIDADE_REGIONAL
 WHERE PLANTAO.CODIGO_AGENDA                   = HORARIO.CODIGO_AGENDA
      AND  PLANTAO.TIPO_AGENDA                         = HORARIO.TIPO_AGENDA
      AND  PLANTAO.CD_PRESTADOR                     = PRESTADOR.CD_PRESTADOR(+)
      AND  PLANTAO.CD_SER_DIS                            = HORARIO.CD_SER_DIS(+)
      AND  PLANTAO.CD_TIP_MAR                            = HORARIO.CD_TIP_MAR(+)
      AND  HORARIO.CD_TIP_MAR                            = TIP_MAR.CD_TIP_MAR(+)
      AND  HORARIO.CD_SER_DIS                            = SER_DIS.CD_SER_DIS(+)
      AND  PLANTAO.CD_UNIDADE_REGIONAL       = UNIDADE_REGIONAL.CD_UNIDADE_REGIONAL(+)
      AND  HORARIO.CD_CONVENIO                          = CONVENIO.CD_CONVENIO(+)
      AND  HORARIO.CD_CONVENIO                          = CON_PLA.CD_CONVENIO(+)
      AND  HORARIO.CD_CON_PLA                            = CON_PLA.CD_CON_PLA(+)
      AND  TO_DATE(TO_CHAR(PLANTAO.DT_MARCACAO, 'DD/MM/YYYY')||' '||TO_CHAR(HORARIO.HORA_AGENDA, 'HH24'), 'DD/MM/YYYY HH24') BETWEEN  TO_DATE(:CAMPO_PDT_INICIAL,'DD/MM/YYYY HH24') AND TO_DATE(:CAMPO_PDT_FINAL,'DD/MM/YYYY HH24')
     AND  HORARIO.NM_PACIENTE IS NOT NULL
     AND  PLANTAO.TIPO_AGENDA                         = 'I'
    AND  PLANTAO.SN_ATIVO                                  = 'S'
     AND  (:CAMPO_PTP_AGENDAMENTO                                = 'T'
       OR                                   = 'I')
UNION
SELECT PLANTAO.CODIGO_AGENDA
               ,PRESTADOR.NM_PRESTADOR
               ,PLANTAO.CD_RECURSO
               ,UNIDADE_REGIONAL.DS_UNIDADE_REGIONAL
               ,TO_CHAR(PLANTAO.DT_MARCACAO,'DD/MM/YYYY') DT_MARCACAO
               ,TO_CHAR(HORARIO.HORA_AGENDA,'hh24') HORA_AGENDA
               ,DECODE(PLANTAO.TIPO_AGENDA,'A','AMB')   TIPO_AGENDA
                ,HORARIO.CD_PACIENTE
               ,HORARIO.NM_PACIENTE
               ,HORARIO.NR_TELEFONE
               ,NVL(HORARIO.SN_ENCAIXE,'N') SN_ENCAIXE
               ,SER_DIS.DS_SER_DIS
               ,TIP_MAR.DS_TIP_MAR
               ,CONVENIO.NM_CONVENIO
               ,CON_PLA.DS_CON_PLA
  FROM DBAMV.V_CENTRAL_MARCACAO_PLANTAO PLANTAO
              ,DBAMV.V_CENTRAL_MARCACAO_HORARIO HORARIO
              ,DBAMV.CONVENIO
              ,DBAMV.CON_PLA
              ,DBAMV.PRESTADOR
              ,DBAMV.TIP_MAR
              ,DBAMV.SER_DIS
              ,DBAMV.UNIDADE_REGIONAL
 WHERE PLANTAO.CODIGO_AGENDA                   = HORARIO.CODIGO_AGENDA
      AND  PLANTAO.TIPO_AGENDA                         = HORARIO.TIPO_AGENDA
      AND  PLANTAO.CD_PRESTADOR                     = PRESTADOR.CD_PRESTADOR(+)
      AND  PLANTAO.CD_SER_DIS                            = HORARIO.CD_SER_DIS(+)
      AND  PLANTAO.CD_TIP_MAR                            = HORARIO.CD_TIP_MAR(+)
      AND  HORARIO.CD_TIP_MAR                            = TIP_MAR.CD_TIP_MAR(+)
      AND  HORARIO.CD_SER_DIS                            = SER_DIS.CD_SER_DIS(+)
      AND  PLANTAO.CD_UNIDADE_REGIONAL       = UNIDADE_REGIONAL.CD_UNIDADE_REGIONAL(+)
      AND  HORARIO.CD_CONVENIO                          = CONVENIO.CD_CONVENIO(+)
      AND  HORARIO.CD_CONVENIO                          = CON_PLA.CD_CONVENIO(+)
      AND  HORARIO.CD_CON_PLA                            = CON_PLA.CD_CON_PLA(+)
      AND  TO_DATE(TO_CHAR(PLANTAO.DT_MARCACAO, 'DD/MM/YYYY')||' '||TO_CHAR(HORARIO.HORA_AGENDA, 'HH24'), 'DD/MM/YYYY HH24') BETWEEN  TO_DATE(:CAMPO_PDT_INICIAL,'DD/MM/YYYY HH24') AND TO_DATE(:CAMPO_PDT_FINAL,'DD/MM/YYYY HH24')
      AND  HORARIO.NM_PACIENTE IS NOT NULL
      AND  PLANTAO.TIPO_AGENDA                         = 'A'
      AND  PLANTAO.SN_ATIVO                                  = 'S'
      AND  (:CAMPO_PTP_AGENDAMENTO                                = 'T'
       OR                                   = 'A')
ORDER BY HORA_AGENDA ASC
)
ORDER BY DT_MARCACAO,
CODIGO_AGENDA,DS_UNIDADE_REGIONAL,NM_PRESTADOR
