--R_EST_P_SER_S_RF_SERVICO_Q_MARCACAO1
SELECT * FROM (
SELECT  distinct ser_dis.cd_ser_dis cd_ser_dis,
                marcacao.cd_marcacao,
                ser_dis.ds_ser_dis ds_ser_dis,
                ori_ate.cd_ori_ate cd_ori_ate,
                tip_mar.cd_tip_mar cd_tip_mar,
                decode(tip_mar.ds_tip_mar,NULL,'NÃO INFORMADA', tip_mar.ds_tip_mar) ds_tip_mar,
                sum(temp.consultas) consultas,
                sum(temp.marcadas) marcadas,
                sum(temp.encaixes)  encaixes,
                sum(temp.atendidas)  atendidas
FROM DBAMV.ORI_ATE,
     DBAMV.TIP_MAR,
     DBAMV.SER_DIS,
     DBAMV.MARCACAO,
(
SELECT  temporaria.cd_marcacao cd_marcacao,
                temporaria.cd_ser_dis   cd_ser_dis,
                temporaria.cd_ori_ate   cd_ori_ate,
                temporaria.cd_tip_mar  cd_tip_mar,
                SUM(total_consulta) consultas,
                SUM(total_marcado) marcadas,
                SUM(total_encaixe) encaixes,
                SUM(total_atendida) atendidas,
                SUM(total_falta_paciente) faltas_paciente,
                SUM(total_falta_medica) faltas_medico
FROM dbamv.marcacao,
(       select marcacao.cd_marcacao cd_marcacao,
               marcacao_ser_tipo.cd_ser_dis cd_ser_dis,
               marcacao.cd_ori_ate cd_ori_ate,
               marcacao_ser_tipo.cd_tip_mar,
               marcacao_ser_tipo.nr_consultas total_consulta,
               nvl(marcacao_ser_tipo.nr_marcados,0) total_marcado,
               nvl(marcacao_ser_tipo.nr_encaixes,0)  total_encaixe,
               0 total_atendida,
               0 total_falta_paciente,
               0 total_falta_medica
        from dbamv.marcacao_ser_tipo,
             dbamv.marcacao
        where marcacao.dt_marcacao between to_date(to_char(:CAMPO_DTINIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
        			     				   to_date(to_char(:CAMPO_DTFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
        and   marcacao_ser_tipo.cd_marcacao = marcacao.cd_marcacao
        and   nvl(marcacao.sn_ativo,'S')  =  'S'
UNION ALL
       select it_marcacao.cd_marcacao   cd_marcacao,
               it_marcacao.cd_ser_dis          cd_ser_dis,
              marcacao.cd_ori_ate            cd_ori_ate,
              it_marcacao.cd_tip_mar          cd_tip_mar,
              0 total_consulta,
              0 total_marcado,
              0 total_encaixe,
              count(*) total_atendida,
              0 total_falta_paciente,
              0 total_falta_medica
       from dbamv.atendime,
              dbamv.it_marcacao,
              dbamv.marcacao
       where dt_marcacao between to_date(to_char(:CAMPO_DTINIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
       					   		 to_date(to_char(:CAMPO_DTFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
       and atendime.tp_atendimento = 'A'
       and it_marcacao.cd_atendimento = atendime.cd_atendimento
       and it_marcacao.cd_marcacao = marcacao.cd_marcacao
       and atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
       group by it_marcacao.cd_marcacao,
                     it_marcacao.cd_ser_dis,
                    marcacao.cd_ori_ate,
                    it_marcacao.cd_tip_mar
) temporaria
where temporaria.cd_marcacao = marcacao.cd_marcacao
GROUP BY   temporaria.cd_marcacao,
                      temporaria.cd_ser_dis,
                      temporaria.cd_ori_ate,
                      temporaria.cd_tip_mar
)TEMP
WHERE TEMP.CD_SER_DIS = SER_DIS.CD_SER_DIS(+)
AND   TEMP.CD_TIP_MAR = TIP_MAR.CD_TIP_MAR(+)
AND   TEMP.CD_ORI_ATE = ORI_ATE.CD_ORI_ATE(+)
AND   TEMP.CD_MARCACAO = MARCACAO.CD_MARCACAO
GROUP BY ser_dis.cd_ser_dis,
                marcacao.cd_marcacao,
                ser_dis.ds_ser_dis,
                ori_ate.cd_ori_ate,
                ori_ate.ds_ori_ate,
                tip_mar.cd_tip_mar,
                decode(tip_mar.ds_tip_mar,NULL,'NÃO INFORMADA', tip_mar.ds_tip_mar)
ORDER BY  ser_dis.ds_ser_dis,
                     decode(tip_mar.ds_tip_mar,NULL,'NÃO INFORMADA', tip_mar.ds_tip_mar)
)
ORDER BY DS_SER_DIS,CD_SER_DIS,
DS_TIP_MAR,CD_TIP_MAR
