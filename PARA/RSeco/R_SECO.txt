--R_SECO
SELECT  ATENDIME.CD_SER_DIS
               ,decode(SER_DIS.DS_SER_DIS, null, 'Não Informado', SER_DIS.DS_SER_DIS) DS_SER_DIS
              ,COUNT(*)   QT_SERVICO
FROM    DBAMV.ATENDIME
              ,DBAMV.SER_DIS
WHERE ATENDIME.DT_ATENDIMENTO between to_date(to_char(:CAMPO_DT_INIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
									  to_date(to_char(:CAMPO_DT_FIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND        SER_DIS.CD_SER_DIS(+) = ATENDIME.CD_SER_DIS
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND      atendime.tp_atendimento = 'A'
GROUP BY ATENDIME.CD_SER_DIS
                  ,SER_DIS.DS_SER_DIS
ORDER BY QT_SERVICO  DESC
