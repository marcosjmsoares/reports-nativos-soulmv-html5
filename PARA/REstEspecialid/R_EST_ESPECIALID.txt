--R_EST_ESPECIALID
SELECT * FROM (
SELECT atendime.CD_ORI_ATE
               ,ori_ate.DS_ORI_ATE
              ,decode(ESPECIALID.DS_ESPECIALID, null, 'Não Informada', ESPECIALID.DS_ESPECIALID) DS_ESPECIALID
             ,COUNT(*)       CONT_ESPECIALID
               ,ATENDIME.CD_ESPECIALID      CD_ESPECIALID
FROM    Dbamv.ATENDIME ATENDIME
              ,Dbamv.ESPECIALID ESPECIALID
              ,DBAMV.ORI_ATE
WHERE  atendime.cd_ori_ate  = ori_ate.cd_ori_ate
and          ESPECIALID.CD_ESPECIALID(+) = ATENDIME.CD_ESPECIALID
AND   ATENDIME.DT_ATENDIMENTO between to_date(to_char(:CAMPO_DT_INIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
									  to_date(to_char(:CAMPO_DT_FIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND       ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY ATENDIME.CD_ORI_ATE, ORI_ATE.DS_ORI_ATE, ATENDIME.CD_ESPECIALID, ESPECIALID.DS_ESPECIALID
ORDER BY CD_ORI_ATE,DS_ORI_ATE,CONT_ESPECIALID DESC
)
