--R_PAC_AGD_DIA
SELECT * FROM (
SELECT
         TO_CHAR( IT_MARCACAO.DT_AGENDADO, 'dd/MM/yyyy')   DT_AGENDADO
       , MARCACAO.CD_MARCACAO CD_MARCACAO
       ,PRESTADOR.CD_PRESTADOR CD_PRESTADOR1
       ,PRESTADOR.NM_PRESTADOR NM_PRESTADOR
       ,MARCACAO.DT_MARCACAO DT_MARCACAO
       ,MARCACAO.HR_INICIAL HR_INICIAL
       ,MARCACAO.HR_FINAL HR_FINAL
       ,TIP_MAR.DS_TIP_MAR DS_TIP_MAR
        ,MARCACAO.CD_SER_DIS
       ,SER_DIS.DS_SER_DIS DS_SER_DIS
       ,MARCACAO.NR_TEMPO_MEDIO NR_TEMPO_MEDIO
       ,MARCACAO.SN_SIA SN_SIA
       ,IT_MARCACAO.CD_PACIENTE
       ,IT_MARCACAO.NM_PACIENTE
       ,IT_MARCACAO.NR_TELEFONE
FROM    Dbamv.MARCACAO MARCACAO
       ,Dbamv.PRESTADOR PRESTADOR
       ,Dbamv.SER_DIS SER_DIS
       ,Dbamv.IT_MARCACAO
       ,Dbamv.TIP_MAR
       ,DBAMV.MARCACAO_SER_DIS
WHERE PRESTADOR.CD_PRESTADOR    = MARCACAO.CD_PRESTADOR
AND    MARCACAO_SER_DIS.CD_MARCACAO =  MARCACAO.CD_MARCACAO
AND   SER_DIS.CD_SER_DIS        =  MARCACAO_SER_DIS.CD_SER_DIS
AND   MARCACAO.CD_MARCACAO      = IT_MARCACAO.CD_MARCACAO
AND   IT_MARCACAO.CD_TIP_MAR    = TIP_MAR.CD_TIP_MAR
AND   MARCACAO.CD_PRESTADOR  LIKE  :CAMPO_CD_MEDICO
AND   MARCACAO.DT_MARCACAO between to_date(to_char(:CAMPO_DT_INIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
								   to_date(to_char(:CAMPO_DT_FIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
ORDER BY MARCACAO.CD_MARCACAO
        ,PRESTADOR.CD_PRESTADOR
        ,MARCACAO.DT_MARCACAO
        ,to_date(to_char(MARCACAO.HR_INICIAL,'hh24'),'hh24')
)
ORDER BY DT_AGENDADO,
CD_MARCACAO,CD_PRESTADOR1,NM_PRESTADOR,DT_MARCACAO,HR_INICIAL,HR_FINAL,CD_SER_DIS,DS_SER_DIS,NR_TEMPO_MEDIO,SN_SIA
