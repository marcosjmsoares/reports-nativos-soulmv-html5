--R_GRAFICO_ATEND_HORA_R_GRID_Q_GRID
SELECT SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '00',1,0))  HR_HORA0
              ,SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '01',1,0))  HR_HORA1
              ,SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '02',1,0))  HR_HORA2
              ,SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '03',1,0))  HR_HORA3
              ,SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '04',1,0))  HR_HORA4
              ,SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '05',1,0))  HR_HORA5
              ,SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '06',1,0))  HR_HORA6
              ,SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '07',1,0))  HR_HORA7
              ,SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '08',1,0))  HR_HORA8
              ,SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '09',1,0))  HR_HORA9
              ,SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '10',1,0))  HR_HORA10
              ,SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '11',1,0))  HR_HORA11
              ,SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '12',1,0))  HR_HORA12
              ,SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '13',1,0))  HR_HORA13
              ,SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '14',1,0))  HR_HORA14
              ,SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '15',1,0))  HR_HORA15
              ,SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '16',1,0))  HR_HORA16
              ,SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '17',1,0))  HR_HORA17
              ,SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '18',1,0))  HR_HORA18
              ,SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '19',1,0))  HR_HORA19
              ,SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '20',1,0))  HR_HORA20
              ,SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '21',1,0))  HR_HORA21
              ,SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '22',1,0))  HR_HORA22
              ,SUM(DECODE(TO_CHAR(ATENDIME.HR_ATENDIMENTO,'HH24'), '23',1,0))  HR_HORA23
              ,SUM(1)  TOT
FROM     DBAMV.ATENDIME, dbamv.ori_ate
WHERE atendime.cd_ori_ate = ori_ate.cd_ori_ate
and ATENDIME.DT_ATENDIMENTO between to_date(to_char(:CAMPO_DT_INIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND
                    to_date(to_char(:CAMPO_DT_FIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND ori_ate.cd_multi_empresa  = :CAMPO_CD_MULTI_EMPRESA
