--R_RESUMO_ATENDIMENTOS
SELECT      CD_QUEBRA    CD_QUEBRA
                   ,DS_QUEBRA    DS_QUEBRA
                   ,SUM(RETORNO)  RETORNOS
                   ,SUM(TURNO_1)   TURNO1
                   ,SUM(TURNO_2)   TURNO2
                   ,SUM(TURNO_3)   TURNO3
                   ,SUM(TURNO_4)   TURNO4
FROM(
               SELECT   DECODE('', 'E', ATENDIME.CD_ESPECIALID, ATENDIME.CD_CONVENIO) CD_QUEBRA
                               ,DECODE('', 'E', ESPECIALID.DS_ESPECIALID, CONVENIO.NM_CONVENIO) DS_QUEBRA
                               ,COUNT(*)  RETORNO
                               ,0         TURNO_1
                               ,0         TURNO_2
                               ,0         TURNO_3
                               ,0         TURNO_4
               FROM     dbamv.atendime
                             ,dbamv.convenio
                             ,dbamv.especialid
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And atendime.cd_convenio = convenio.cd_convenio
and atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
               AND          atendime.cd_especialid = especialid.cd_especialid
               AND          atendime.tp_atendimento = 'A'
               AND          NVL(atendime.sn_retorno, 'N') = 'S'
               AND          atendime.dt_atendimento between to_date(to_char(:CAMPO_DT_INICIAL,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
               												to_date(to_char(:CAMPO_DT_FINAL,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
               GROUP BY  DECODE('', 'E', ATENDIME.CD_ESPECIALID, ATENDIME.CD_CONVENIO)
                                   ,DECODE('', 'E', ESPECIALID.DS_ESPECIALID, CONVENIO.NM_CONVENIO)
               UNION
               SELECT   DECODE('', 'E', ATENDIME.CD_ESPECIALID, ATENDIME.CD_CONVENIO) CD_QUEBRA
                               ,DECODE('', 'E', ESPECIALID.DS_ESPECIALID, CONVENIO.NM_CONVENIO) DS_QUEBRA
                               ,0                 RETORNO
                               ,COUNT(*)   TURNO_1
                               ,0                 TURNO_2
                               ,0                 TURNO_3
                               ,0                 TURNO_4
               FROM     dbamv.atendime
                             ,dbamv.convenio
                             ,dbamv.especialid
,Dbamv.Empresa_Convenio

               GROUP BY  DECODE('', 'E', ATENDIME.CD_ESPECIALID, ATENDIME.CD_CONVENIO)

               UNION
               SELECT   DECODE('', 'E', ATENDIME.CD_ESPECIALID, ATENDIME.CD_CONVENIO) CD_QUEBRA
                               ,DECODE('', 'E', ESPECIALID.DS_ESPECIALID, CONVENIO.NM_CONVENIO) DS_QUEBRA
                               ,0                RETORNO
                               ,0                TURNO_1
                               ,COUNT(*)  TURNO_2
                               ,0                TURNO_3
                               ,0                TURNO_4
               FROM     dbamv.atendime
                             ,dbamv.convenio
                             ,dbamv.especialid
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And atendime.cd_convenio = convenio.cd_convenio
and atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
               AND          atendime.cd_especialid = especialid.cd_especialid
               AND          atendime.tp_atendimento = 'A'
               AND          NVL(atendime.sn_retorno, 'N') = 'N'
