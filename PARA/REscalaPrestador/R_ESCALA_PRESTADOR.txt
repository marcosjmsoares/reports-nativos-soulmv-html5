--R_ESCALA_PRESTADOR
SELECT * FROM (
SELECT To_Number(Substr(PRESTADOR.CD_PRESTADOR,1,12)) CD_PRESTADOR 
              ,PRESTADOR.NM_PRESTADOR
              ,P_ESCALA.TP_DIA_SEMANA
              ,DECODE(TO_CHAR(P_ESCALA.TP_DIA_SEMANA), '1', 'DOMINGO',
                                                                                                    '2', 'SEGUNDA-FEIRA',
                                                                                                    '3', 'TERÇA-FEIRA',
                                                                                                    '4', 'QUARTA-FEIRA',
                                                                                                    '5', 'QUINTA-FEIRA',
                                                                                                    '6', 'SEXTA-FEIRA',
                                                                                                    '7', 'SÁBADO')           DIA
             ,P_ESCALA.HR_INIC
             ,P_ESCALA.HR_FIM
             ,TIP_MAR.CD_TIP_MAR
             ,TIP_MAR.DS_TIP_MAR
             ,ITP_ESCALA.QT_AGENDAR
FROM   DBAMV.PRESTADOR
             ,DBAMV.P_ESCALA
             ,DBAMV.ITP_ESCALA
             ,DBAMV.TIP_MAR
WHERE  P_ESCALA.SN_ATIVO = 'S'
AND        PRESTADOR.CD_PRESTADOR = P_ESCALA.CD_PRESTADOR
AND        P_ESCALA.CD_P_ESCALA = ITP_ESCALA.CD_P_ESCALA
AND        ITP_ESCALA.CD_TIP_MAR = TIP_MAR.CD_TIP_MAR
ORDER BY PRESTADOR.NM_PRESTADOR,  P_ESCALA.TP_DIA_SEMANA
)
ORDER BY CD_PRESTADOR,NM_PRESTADOR,
TP_DIA_SEMANA,DIA,HR_INIC,HR_FIM
