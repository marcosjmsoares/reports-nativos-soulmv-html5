--R_AGE_PLANTAO
SELECT * FROM (
SELECT IT_MARCA.CD_MARCACAO          CD_MARCACAO
      ,MARCACAO.DT_MARCACAO          DT_MARCACAO
      ,MARCACAO.DS_CONSULTORIO       DS_CONSULTORIO
      ,PRESTADOR.CD_PRESTADOR        CD_PRESTADOR
      ,PRESTADOR.NM_PRESTADOR        NM_PRESTADOR
      ,IT_MARCA.CD_SER_DIS           CD_SER_DIS
      ,SER_DIS.DS_SER_DIS            DS_SER_DIS
      ,MARCACAO.CD_ORI_ATE           CD_ORI_ATE
      ,ORI_ATE.DS_ORI_ATE            DS_ORI_ATE
      ,MARCACAO.HR_INICIAL           HR_INICIAL
      ,MARCACAO.HR_FINAL             HR_FINAL
      ,NVL(MARCACAO.NR_CONSULTAS, 0) NR_CONSULTAS
      ,NVL(MARCACAO.NR_MARCADOS, 0)  NR_MARCADOS
      ,MARCACAO.NR_TEMPO_MEDIO       NR_TEMPO_MEDIO
      ,IT_MARCA.HR_AGENDADO          HR_AGENDADO
      ,IT_MARCA.SN_ENCAIXE           SN_ENCAIXE
      ,IT_MARCA.CD_PACIENTE          CD_PACIENTE
      ,IT_MARCA.DS_OBSERVACAO        DS_OBSERVACAO
      ,NVL(PACIENTE.NM_PACIENTE, IT_MARCA.NM_PACIENTE ) NM_PACIENTE
      ,(SELECT PA.NM_SOCIAL_PACIENTE
        FROM DBAMV.PACIENTE PA
        WHERE PA.CD_PACIENTE = PACIENTE.CD_PACIENTE
          AND NVL(PA.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE
      ,IT_MARCA.NR_TELEFONE         NR_FONE
      ,IT_MARCA.CD_IT_MARCACAO      CD_IT_MARCACAO
      ,CONVENIO.NM_CONVENIO         NM_CONVENIO
      ,TIP_MAR.DS_TIP_MAR           DS_TIP_MAR
      ,IT_MARCA.CD_ATENDIMENTO      CD_ATENDIMENTO
      ,IT_MARCA.CD_PROCEDIMENTO_SIA CD_SSM_SIA
FROM DBAMV.IT_MARCACAO IT_MARCA
    ,DBAMV.MARCACAO
    ,DBAMV.PRESTADOR
    ,DBAMV.SER_DIS
    ,DBAMV.ORI_ATE
    ,DBAMV.PACIENTE
    ,DBAMV.CONVENIO
    ,DBAMV.TIP_MAR
WHERE MARCACAO.DT_MARCACAO BETWEEN TO_DATE(TO_CHAR(:CAMPO_DT_INICIO, 'DD/MM/YYYY') || ' 00', 'DD/MM/YYYY HH24') AND 
			    TO_DATE(TO_CHAR(:CAMPO_DT_FINAL, 'DD/MM/YYYY') || ' 23', 'DD/MM/YYYY HH24')
  AND (MARCACAO.SN_ATIVO = 'S'
      OR MARCACAO.SN_ATIVO IS NULL)
  AND PACIENTE.CD_PACIENTE(+) = IT_MARCA.CD_PACIENTE
  AND SER_DIS.CD_SER_DIS(+) = IT_MARCA.CD_SER_DIS
  AND CONVENIO.CD_CONVENIO(+) = IT_MARCA.CD_CONVENIO
  AND TIP_MAR.CD_TIP_MAR(+) = IT_MARCA.CD_TIP_MAR
  AND MARCACAO.CD_MARCACAO = IT_MARCA.CD_MARCACAO
  AND PRESTADOR.CD_PRESTADOR = MARCACAO.CD_PRESTADOR
  AND ORI_ATE.CD_ORI_ATE = MARCACAO.CD_ORI_ATE
  AND NVL(PACIENTE.NM_PACIENTE, IT_MARCA.NM_PACIENTE) IS NOT NULL
  :CAMPO_CF_ORDEM
)
ORDER BY CD_MARCACAO,
         DT_MARCACAO,
         DS_CONSULTORIO,
         CD_PRESTADOR,
         NM_PRESTADOR,
         CD_ORI_ATE,
         DS_ORI_ATE,
         HR_INICIAL,
         HR_FINAL,
         NR_CONSULTAS,
         NR_MARCADOS,
         NR_TEMPO_MEDIO
