SELECT COUNT(*)        QTDATD
              ,decode(CIDADE.CD_CIDADE, null, 0, CIDADE.CD_CIDADE)                 CD_CIDADE
              ,decode(CIDADE.NM_CIDADE, null, 'NÃO INFORMADA', ltrim (CIDADE.NM_CIDADE)  || ' - ' ||  CIDADE.CD_UF) NM_CIDADE
FROM    Dbamv.ATENDIME ATENDIME
              ,Dbamv.PACIENTE PACIENTE
              ,Dbamv.CIDADE CIDADE
WHERE ATENDIME.CD_PACIENTE =  PACIENTE.CD_PACIENTE
AND       PACIENTE.CD_CIDADE      =  CIDADE.CD_CIDADE(+)
AND       ATENDIME.DT_ATENDIMENTO between to_date(to_char(:CAMPO_DT_INIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND 
										  to_date(to_char(:CAMPO_DT_FIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY CIDADE.CD_UF
                  ,decode(CIDADE.CD_CIDADE, null, 0, CIDADE.CD_CIDADE)
                  ,decode(CIDADE.NM_CIDADE, null, 'NÃO INFORMADA', ltrim (CIDADE.NM_CIDADE)  || ' - ' ||  CIDADE.CD_UF)
ORDER BY QTDATD  DESC
