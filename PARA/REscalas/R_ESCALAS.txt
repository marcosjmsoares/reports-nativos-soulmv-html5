--R_ESCALAS
SELECT * FROM (
SELECT  PRESTADOR.CD_PRESTADOR
               ,PRESTADOR.NM_PRESTADOR
               ,P_ESCALA.CD_P_ESCALA              CD_ESCALA
               ,P_ESCALA.TP_DIA_SEMANA
               ,P_ESCALA.SN_ATIVO
               ,P_ESCALA.SN_SIA
               ,P_ESCALA.SN_ATENDE_FERIADO
               ,P_ESCALA.HR_INIC
               ,P_ESCALA.HR_FIM
               ,SER_DIS.DS_SER_DIS
               ,ORI_ATE.DS_ORI_ATE
               ,P_ESCALA.NR_ATENDIMENTOS
FROM     Dbamv.P_ESCALA
              ,Dbamv.PRESTADOR
              ,Dbamv.ORI_ATE
              ,Dbamv.P_ESCALA_SER_DIS
              ,Dbamv.SER_DIS
WHERE P_ESCALA.CD_PRESTADOR  =  PRESTADOR.CD_PRESTADOR
AND       P_ESCALA.CD_ORI_ATE  =  ORI_ATE.CD_ORI_ATE
AND       P_ESCALA.CD_P_ESCALA  =  P_ESCALA_SER_DIS.CD_P_ESCALA
AND       P_ESCALA_SER_DIS.CD_SER_DIS  =  SER_DIS.CD_SER_DIS
ORDER BY PRESTADOR.CD_PRESTADOR,  P_ESCALA.TP_DIA_SEMANA
)
