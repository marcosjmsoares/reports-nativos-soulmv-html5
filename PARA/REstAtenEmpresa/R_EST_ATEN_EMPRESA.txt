--R_EST_ATEN_EMPRESA
SELECT cart.NM_EMPRESA,
       count(*) count
  FROM DBAMV.ATENDIME aten,
       DBAMV.CARTEIRA cart,
       DBAMV.PACIENTE paci
 WHERE aten.DT_ATENDIMENTO between
to_date(to_char(:CAMPO_DTINIC,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND to_date(to_char(:CAMPO_DTFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
       AND aten.CD_PACIENTE    = paci.CD_PACIENTE
       AND ATEN.CD_PACIENTE    = cart.CD_PACIENTE(+)
       AND ATEN.CD_CONVENIO    = CART.CD_CONVENIO(+)
       AND ATEN.CD_CON_PLA     = CART.CD_CON_PLA(+)
      and aten.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY cart.NM_EMPRESA
ORDER BY COUNT(*) DESC
