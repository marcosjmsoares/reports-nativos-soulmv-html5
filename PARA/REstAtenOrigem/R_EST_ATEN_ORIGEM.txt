--R_EST_ATEN_ORIGEM
SELECT * FROM (
SELECT Decode( aten.TP_ATENDIMENTO, 'I', 'INTERNACAO',
                                    'A', 'AMBULATORIAL',
                                    'E', 'EXTERNO',
                                    'U', 'URGENCIA/EMERGENCIA',
                                    'H', 'HOME CARE',
                                    'O', 'ATENDIMENTO SOS',
                                    'S', 'AIH',
                                    'NAO IDENTIFICADO' ) TP_ATENDIMENTO,
               orig.DS_ORI_ATE,
               orig.CD_ORI_ATE,
               COUNT(*)         QTDATD
FROM     DBAMV.ATENDIME  aten,
               DBAMV.ORI_ATE   orig,
               DBAMV.CONVENIO  conv
WHERE aten.CD_ORI_ATE  =  orig.CD_ORI_ATE
AND       aten.DT_ATENDIMENTO between
to_date(to_char(:CAMPO_DATINI,'dd/mm/yyyy')||' 00','dd/mm/yyyy HH24') AND to_date(to_char(:CAMPO_DATFIM,'dd/mm/yyyy')||' 23','dd/mm/yyyy HH24')
AND      aten.CD_CONVENIO = conv.CD_CONVENIO
AND       ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND       ATEN.CD_ATENDIMENTO_PAI IS NULL
GROUP BY aten.TP_ATENDIMENTO,
                    orig.DS_ORI_ATE,
                    orig.CD_ORI_ATE
)
ORDER BY TP_ATENDIMENTO, QTDATD  DESC
