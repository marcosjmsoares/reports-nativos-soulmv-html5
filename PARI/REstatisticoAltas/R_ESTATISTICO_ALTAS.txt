--R_ESTATISTICO_ALTAS
SELECT 1 FROM DUAL


/*----------------------------------------------------------------------------*/
--R_ESTATISTICO_ALTAS_RF_MOTIVOS1_Q_QUEBRA_TP_INTERNACAO
SELECT * FROM (
SELECT  DECODE(TOTAL_GERAL.CODIGO, '-1', NULL, TOTAL_GERAL.CODIGO) CODIGO
               ,TOTAL_GERAL.DESCRICAO
               ,QT_ALTAS TOTAL
               ,TIPO_INTERNACAO.CD_TIPO_INTERNACAO
               ,TIPO_INTERNACAO.DS_TIPO_INTERNACAO
               ,COUNT(*) SUBTOTAL
FROM DBAMV.ATENDIME
           ,DBAMV.MOT_ALT
           ,DBAMV.CONVENIO
           ,DBAMV.ESPECIALID
           ,DBAMV.PRESTADOR
           ,DBAMV.CID
           ,DBAMV.LEITO
           ,DBAMV.UNID_INT
           ,DBAMV.TIPO_INTERNACAO
           ,(SELECT   NVL(:CAMPO_CD_COLUNA,-1)    CODIGO
                             ,:CAMPO_DS_COLUNA     DESCRICAO
                             ,COUNT(*) QT_ALTAS
            FROM DBAMV.ATENDIME
                       ,DBAMV.MOT_ALT
                        ,DBAMV.CONVENIO
                        ,DBAMV.ESPECIALID
                        ,DBAMV.PRESTADOR
                        ,DBAMV.CID
                        ,DBAMV.LEITO
                        ,DBAMV.UNID_INT
                        ,DBAMV.TIPO_INTERNACAO
             WHERE ATENDIME.TP_ATENDIMENTO IN ('I','B','H')
                  AND ATENDIME.CD_ATENDIMENTO_PAI IS NULL
                   AND ATENDIME.DT_ALTA IS NOT NULL
                   AND ATENDIME.DT_ALTA BETWEEN To_Date(Concat(To_Char(:CAMPO_DT_INI,'dd/mm/yyyy'),' 00'),'dd/mm/yyyy hh24') 
				   AND To_Date(Concat(To_Char(:CAMPO_DT_FIM,'dd/mm/yyyy'),' 23'),'dd/mm/yyyy hh24')
                   AND ATENDIME.CD_MOT_ALT = MOT_ALT.CD_MOT_ALT
                   AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
                   AND ATENDIME.CD_TIPO_INTERNACAO = TIPO_INTERNACAO.CD_TIPO_INTERNACAO(+)
                   AND ATENDIME.CD_ESPECIALID = ESPECIALID.CD_ESPECIALID(+)
                   AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
                   AND ATENDIME.CD_CID = CID.CD_CID(+)
                   AND ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
                   AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)
                   AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
             GROUP BY 
                                ,:CAMPO_DS_COLUNA
              ) TOTAL_GERAL
WHERE ATENDIME.TP_ATENDIMENTO IN ('I','B','H')
AND ATENDIME.CD_ATENDIMENTO_PAI IS NULL
AND ATENDIME.DT_ALTA IS NOT NULL
AND ATENDIME.DT_ALTA BETWEEN To_Date(Concat(To_Char(:CAMPO_DT_INI,'dd/mm/yyyy'),' 00'),'dd/mm/yyyy hh24') 
AND To_Date(Concat(To_Char(:CAMPO_DT_FIM,'dd/mm/yyyy'),' 23'),'dd/mm/yyyy hh24')
AND ATENDIME.CD_MOT_ALT = MOT_ALT.CD_MOT_ALT
AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND ATENDIME.CD_TIPO_INTERNACAO = TIPO_INTERNACAO.CD_TIPO_INTERNACAO(+)
AND ATENDIME.CD_ESPECIALID = ESPECIALID.CD_ESPECIALID(+)
AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
AND ATENDIME.CD_CID = CID.CD_CID(+)
AND ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY TIPO_INTERNACAO.CD_TIPO_INTERNACAO
                    ,TIPO_INTERNACAO.DS_TIPO_INTERNACAO
                    ,TOTAL_GERAL.CODIGO
                    ,TOTAL_GERAL.DESCRICAO
                    ,QT_ALTAS
ORDER BY TOTAL_GERAL.QT_ALTAS DESC ,SUBTOTAL DESC
)
ORDER BY TOTAL DESC,CODIGO,DESCRICAO


/*----------------------------------------------------------------------------*/
--R_ESTATISTICO_ALTAS_RF_MOTIVOS_Q_ESTATISTICA
SELECT     CODIGO
              ,:CAMPO_DS_COLUNA     DESCRICAO
              ,COUNT(*)                 QT_ALTAS
FROM DBAMV.ATENDIME
          ,DBAMV.MOT_ALT
          ,DBAMV.CONVENIO
          ,DBAMV.ESPECIALID
          ,DBAMV.SETOR
          ,DBAMV.PRESTADOR
          ,DBAMV.CID
          ,DBAMV.LEITO
          ,DBAMV.UNID_INT
	  ,DBAMV.TIPO_INTERNACAO
WHERE ATENDIME.TP_ATENDIMENTO IN ('I','B','H')
AND       ATENDIME.DT_ALTA IS NOT NULL
AND       ATENDIME.DT_ALTA BETWEEN To_Date(Concat(To_Char(:CAMPO_DT_INI,'dd/mm/yyyy'),' 00'),'dd/mm/yyyy hh24')
AND       To_Date(Concat(To_Char(:CAMPO_DT_FIM,'dd/mm/yyyy'),' 23'),'dd/mm/yyyy hh24')
AND       ATENDIME.CD_MOT_ALT = MOT_ALT.CD_MOT_ALT
AND       ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND 	  ATENDIME.CD_TIPO_INTERNACAO = TIPO_INTERNACAO.CD_TIPO_INTERNACAO(+)
AND       ATENDIME.CD_ESPECIALID = ESPECIALID.CD_ESPECIALID(+)
AND       ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
AND       ATENDIME.CD_CID = CID.CD_CID(+)
AND       ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
AND       LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)
AND 	  UNID_INT.CD_SETOR = SETOR.CD_SETOR
AND       SETOR.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND ATENDIME.CD_ATENDIMENTO_PAI IS NULL
GROUP BY  :CAMPO_CD_COLUNA
                    ,:CAMPO_DS_COLUNA
ORDER BY  QT_ALTAS  DESC
