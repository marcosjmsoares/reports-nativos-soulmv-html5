--R_CON_PERMAN
SELECT ATENDIME.CD_PRESTADOR MEDICO ,
       PRESTADOR.NM_MNEMONICO NM_MNEMONICO ,
       ATENDIME.CD_LEITO CD_LEITO ,
       LEITO.DS_RESUMO DS_LEITO ,
       LEITO.DS_ENFERMARIA DS_ENFERMARIA ,
       UNID_INT.CD_UNID_INT CD_UNID_INT ,
       ATENDIME.CD_PACIENTE PRONTUARIO ,
       PACIENTE.NM_PACIENTE NOME ,
   (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
       ATENDIME.CD_ATENDIMENTO CD_ATENDIMENTO ,
       ATENDIME.DT_ATENDIMENTO DT_ATENDIMENTO ,
       NVL(ATENDIME.DT_PREVISTA_ALTA, ATENDIME.DT_ATENDIMENTO) DATA_ALTA ,
       DECODE(CONVENIO.TP_CONVENIO, 'C', ATENDIME.CD_PRO_INT, 'P', ATENDIME.CD_PRO_INT, ATENDIME.CD_PROCEDIMENTO) PRO_INT ,
       TRUNC(SYSDATE) - TRUNC(NVL(ATENDIME.DT_PREVISTA_ALTA, DT_ATENDIMENTO)) VCTO ,
       DECODE(CONVENIO.TP_CONVENIO, 'C', PRO_FAT.NR_DIAS_INTERNACAO, 'P', PRO_FAT.NR_DIAS_INTERNACAO, PROCEDIMENTO_SUS.NR_DIAS_INTERNACAO) DIAS ,
       (TRUNC(SYSDATE) - TRUNC(ATENDIME.DT_ATENDIMENTO))PERMAN
FROM dbamv.ATENDIME ,
     dbamv.CONVENIO ,
     dbamv.PRESTADOR ,
     dbamv.LEITO ,
     dbamv.UNID_INT ,
     dbamv.PACIENTE ,
     dbamv.PRO_FAT ,
     dbamv.PROCEDIMENTO_SUS ,
     DUAL
WHERE ATENDIME.TP_ATENDIMENTO = 'I'
  AND TRUNC(SYSDATE) - TRUNC(NVL(ATENDIME.DT_PREVISTA_ALTA, DT_ATENDIMENTO)) >= :CAMPO_VCTO
  AND ATENDIME.DT_ALTA IS NULL
  AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
  AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
  AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
  AND ATENDIME.CD_LEITO = LEITO.CD_LEITO
  AND ATENDIME.CD_PRO_INT = PRO_FAT.CD_PRO_FAT (+)
  AND ATENDIME.CD_PROCEDIMENTO = PROCEDIMENTO_SUS.CD_PROCEDIMENTO (+)
  AND UNID_INT.CD_UNID_INT LIKE :CAMPO_UNIDADE
  AND ATENDIME.CD_PRESTADOR LIKE :CAMPO_PRESTADOR
  AND UNID_INT.CD_UNID_INT = LEITO.CD_UNID_INT
  AND UNID_INT.SN_ATIVO = 'S'
  AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA    
ORDER BY MEDICO,
         :CAMPO_ORDEM
