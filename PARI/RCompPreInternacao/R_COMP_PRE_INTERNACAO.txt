--R_COMP_PRE_INTERNACAO
SELECT * FROM(
SELECT  DISTINCT RES_LEI.CD_RES_LEI
       ,TO_CHAR(RES_LEI.DT_RESERVA,'DD/MM/YYYY') DT_RESERVA
       ,RES_LEI.NM_USUARIO
       ,RES_LEI.CD_PACIENTE
       ,RES_LEI.NM_PACIENTE
       ,RES_LEI.CD_ORI_ATE
       ,TO_CHAR(RES_LEI.DT_PREV_INTERNACAO,'DD/MM/YYYY HH24') DT_PREV_INTERNACAO
       ,Decode(Convenio.Tp_Convenio, 'H', Procedimento_sus.Cd_procedimento,RES_LEI.CD_PRO_INT) CD_PRO_INT
       ,RES_LEI.DS_OBSERVACAO
       ,CONVENIO.NM_CONVENIO
       ,CON_PLA.DS_CON_PLA
       ,Decode(Convenio.Tp_Convenio, 'H', Procedimento_sus.ds_procedimento,pro_fat.DS_PRO_FAT) DS_PRO_FAT
       ,TIP_ACOM.DS_TIP_ACOM
       ,PRESTADOR.NM_PRESTADOR
       ,ESPECIALID.DS_ESPECIALID
       ,SERVICO.DS_SERVICO
       ,SAME.NR_MATRICULA_SAME
       ,RES_LEI.NR_CARTEIRA
       ,SUB_PLANO.DS_SUB_PLANO
  FROM  DBAMV.RES_LEI
       ,DBAMV.CONVENIO
       ,DBAMV.CON_PLA
       ,DBAMV.SUB_PLANO
       ,DBAMV.PACIENTE
       ,DBASGU.USUARIOS
       ,DBAMV.PRO_FAT
       ,dbamv.procedimento_sus
       ,DBAMV.PRESTADOR
       ,DBAMV.ESPECIALID
       ,DBAMV.SERVICO
       ,DBAMV.TIP_ACOM
       ,DBAMV.SAME 
WHERE RES_LEI.CD_PACIENTE          = PACIENTE.CD_PACIENTE(+)
  AND RES_LEI.CD_CONVENIO          = CONVENIO.CD_CONVENIO(+)
  AND RES_LEI.CD_CON_PLA           = CON_PLA.CD_CON_PLA(+)
  AND RES_LEI.CD_CONVENIO          = CON_PLA.CD_CONVENIO(+)
  AND RES_LEI.CD_SUB_PLANO         = SUB_PLANO.CD_SUB_PLANO(+)
  AND RES_LEI.CD_CON_PLA           =  SUB_PLANO.CD_CON_PLA(+)
  AND RES_LEI.NM_USUARIO           = USUARIOS.CD_USUARIO
  AND RES_LEI.CD_PACIENTE          = SAME.CD_PACIENTE(+)
  AND RES_LEI.CD_PRO_INT           = PRO_FAT.CD_PRO_FAT(+)
  AND RES_LEI.CD_PRESTADOR         = PRESTADOR.CD_PRESTADOR(+)
  AND RES_LEI.CD_ESPECIALID        = ESPECIALID.CD_ESPECIALID(+)
  AND RES_LEI.CD_SERVICO           = SERVICO.CD_SERVICO(+)
  AND RES_LEI.CD_TIP_ACOM          = TIP_ACOM.CD_TIP_ACOM(+)
  And Res_Lei.Cd_Procedimento      = procedimento_sus.cd_procedimento(+)
  AND RES_LEI.CD_MULTI_EMPRESA     = :CAMPO_CD_MULTI_EMPRESA
  AND RES_LEI.CD_RES_LEI                     = :CAMPO_PCD_RES_LEI
  AND RES_LEI.DT_RESERVA IS NOT NULL
  AND RES_LEI.DT_PREV_INTERNACAO IS NOT NULL
)ORDER BY DS_SUB_PLANO, CD_PRO_INT, NR_MATRICULA_SAME, NR_CARTEIRA, DS_ESPECIALID, DT_PREV_INTERNACAO,
NM_USUARIO, CD_RES_LEI, DT_RESERVA, CD_PACIENTE, NM_PACIENTE, DS_OBSERVACAO, NM_CONVENIO, DS_CON_PLA,
DS_PRO_FAT, DS_TIP_ACOM, NM_PRESTADOR, DS_SERVICO
