--R_ALTASESPECIALIDADE
SELECT * FROM (
SELECT  'ALTAS DE CONVENIOS OU PARTICULAR'      TIPO_ALTA
              ,DECODE(ESPECIALID.CD_ESPECIALID, NULL, 0, ESPECIALID.CD_ESPECIALID)                       CD_ESPECIALID
              ,DECODE(ESPECIALID.DS_ESPECIALID, NULL, 'SEM ESPECIALIDADE', ESPECIALID.DS_ESPECIALID)     DS_ESPECIALID
              ,ATENDIME.DT_ALTA
              ,ATENDIME.CD_ATENDIMENTO
              ,ATENDIME.CD_PACIENTE
              ,ATENDIME.DT_ATENDIMENTO
              ,PACIENTE.NM_PACIENTE
              ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  CD_PACIENTE) NM_SOCIAL_PACIENTE
              ,PRESTADOR.NM_PRESTADOR
              ,ATENDIME.CD_PRO_INT
              ,PRO_FAT.DS_PRO_FAT     DS_PRO_INT
              ,MOT_ALT.DS_MOT_ALT
              ,LEITO.DS_RESUMO
              ,UNID_INT.DS_UNID_INT
FROM   DBAMV.ATENDIME
             ,DBAMV.ESPECIALID
             ,DBAMV.PRO_FAT
             ,DBAMV.PACIENTE
             ,DBAMV.CONVENIO
             ,DBAMV.PRESTADOR
             ,DBAMV.MOT_ALT
             ,DBAMV.LEITO
             ,DBAMV.UNID_INT
WHERE  TRUNC(ATENDIME.DT_ALTA) BETWEEN :CAMPO_DT_INI AND 
AND        ATENDIME.DT_ALTA IS NOT NULL
AND        ATENDIME.TP_ATENDIMENTO in ('I', 'H')
AND        ATENDIME.CD_CONVENIO  =  CONVENIO.CD_CONVENIO
AND        CONVENIO.TP_CONVENIO IN ('C', 'P')
AND        ATENDIME.CD_PRO_INT  =  PRO_FAT.CD_PRO_FAT(+)
AND        NVL(ATENDIME.CD_ESPECIALID_SAIDA,ATENDIME.CD_ESPECIALID) = ESPECIALID.CD_ESPECIALID (+)
AND        ATENDIME.CD_PACIENTE  =  PACIENTE.CD_PACIENTE
AND        ATENDIME.CD_PRESTADOR  =  PRESTADOR.CD_PRESTADOR(+)
AND        ATENDIME.CD_MOT_ALT = MOT_ALT.CD_MOT_ALT(+)
AND        ATENDIME.CD_LEITO  =  LEITO.CD_LEITO(+)
AND        LEITO.CD_UNID_INT  =  UNID_INT.CD_UNID_INT(+)
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND ATENDIME.CD_ATENDIMENTO_PAI IS NULL
UNION ALL
SELECT  'ALTAS DO SUS'      TIPO_ALTA
              ,DECODE(ESPECIALID.CD_ESPECIALID, NULL, 0, ESPECIALID.CD_ESPECIALID)                       CD_ESPECIALID
              ,DECODE(ESPECIALID.DS_ESPECIALID, NULL, 'SEM ESPECIALIDADE', ESPECIALID.DS_ESPECIALID)     DS_ESPECIALID
              ,ATENDIME.DT_ALTA
              ,ATENDIME.CD_ATENDIMENTO
              ,ATENDIME.CD_PACIENTE
              ,ATENDIME.DT_ATENDIMENTO
              ,PACIENTE.NM_PACIENTE
              ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  CD_PACIENTE) NM_SOCIAL_PACIENTE
              ,PRESTADOR.NM_PRESTADOR
              ,PROCEDIMENTO_SUS.CD_PROCEDIMENTO
              ,PROCEDIMENTO_SUS.DS_PROCEDIMENTO
              ,MOT_ALT.DS_MOT_ALT
              ,LEITO.DS_RESUMO
              ,UNID_INT.DS_UNID_INT
FROM   DBAMV.ATENDIME
             ,DBAMV.PROCEDIMENTO_SUS
             ,DBAMV.ESPECIALID
             ,DBAMV.PACIENTE
             ,DBAMV.CONVENIO
             ,DBAMV.PRESTADOR
             ,DBAMV.MOT_ALT
             ,DBAMV.LEITO
             ,DBAMV.UNID_INT
WHERE  TRUNC(ATENDIME.DT_ALTA) BETWEEN :CAMPO_DT_INI AND 
AND        ATENDIME.DT_ALTA IS NOT NULL
AND        ATENDIME.TP_ATENDIMENTO IN ('I','H')
AND        ATENDIME.CD_CONVENIO  =  CONVENIO.CD_CONVENIO
AND        CONVENIO.TP_CONVENIO IN ('H','A')
AND        ATENDIME.CD_PROCEDIMENTO = PROCEDIMENTO_SUS.CD_PROCEDIMENTO (+)
AND        NVL(ATENDIME.CD_ESPECIALID_SAIDA,ATENDIME.CD_ESPECIALID) = ESPECIALID.CD_ESPECIALID (+)
AND        ATENDIME.CD_PACIENTE  =  PACIENTE.CD_PACIENTE
AND        ATENDIME.CD_PRESTADOR  =  PRESTADOR.CD_PRESTADOR(+)
AND        ATENDIME.CD_MOT_ALT = MOT_ALT.CD_MOT_ALT(+)
AND        ATENDIME.CD_LEITO  =  LEITO.CD_LEITO(+)
AND        LEITO.CD_UNID_INT  =  UNID_INT.CD_UNID_INT(+)
AND     ATENDIME.CD_ATENDIMENTO_PAI IS NULL
:CAMPO_CF_MULTI_EMPRESA
)
ORDER BY TIPO_ALTA,
CD_ESPECIALID,DS_ESPECIALID,
DT_ALTA
