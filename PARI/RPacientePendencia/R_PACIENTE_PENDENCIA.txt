--R_PACIENTE_PENDENCIA
SELECT paci.CD_PACIENTE,
			       paci.NM_PACIENTE,
			  (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACI.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
			       paci.NR_FONE,
			       tipo.DS_TIPO_PENDENCIA,
			       aten.CD_ATENDIMENTO,
			       aten.DT_ATENDIMENTO,
			       pend.NM_RESPONSAVEL
			FROM DBAMV.ATENDIME aten,
			     DBAMV.PENDENCIA_ATENDIMENTO pend,
			     DBAMV.TIPO_PENDENCIA tipo,
			     DBAMV.PACIENTE paci
			WHERE pend.DT_BAIXA_PENDENCIA IS NULL
			  AND pend.CD_ATENDIMENTO = aten.CD_ATENDIMENTO
			  AND pend.CD_TIPO_PENDENCIA = tipo.CD_TIPO_PENDENCIA
			  AND aten.CD_PACIENTE = paci.CD_PACIENTE
			  AND aten.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
			ORDER BY paci.CD_PACIENTE,
			         tipo.DS_TIPO_PENDENCIA
