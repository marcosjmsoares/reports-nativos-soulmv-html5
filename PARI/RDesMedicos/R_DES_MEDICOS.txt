--R_DES_MEDICOS
SELECT A.CD_PRESTADOR, A.MEDICOS, B.CONSULTAS, C.INTERNACOES,D.CIRURGIAS,
        E.ESTUDOS_HEMODINAMICOS, F.ECOGRAFIAS, G.ERGOMETRIAS, H.HOLTER_MAPA,
        I.ECG, J.ANALISES_CLINICAS
FROM
   (SELECT DISTINCT A.CD_PRESTADOR , A.NM_PRESTADOR  MEDICOS
    FROM   DBAMV.ATENDIME B, DBAMV.PRESTADOR A
    WHERE  B.CD_PRESTADOR = A.CD_PRESTADOR) A ,
   (SELECT A.CD_PRESTADOR , A.NM_PRESTADOR ,COUNT(C.CD_MARCACAO) CONSULTAS
    FROM   DBAMV.ATENDIME B , DBAMV.PRESTADOR A , DBAMV.IT_MARCACAO C
    WHERE  (B.CD_PRESTADOR = A.CD_PRESTADOR) AND (C.CD_ATENDIMENTO = B.CD_ATENDIMENTO)
			AND B.DT_ATENDIMENTO BETWEEN :CAMPO_DT_INI GROUP BY A.CD_PRESTADOR , A.NM_PRESTADOR) B ,
   (SELECT A.CD_PRESTADOR , A.NM_PRESTADOR ,COUNT(B.CD_ATENDIMENTO) INTERNACOES
    FROM   DBAMV.ATENDIME B , DBAMV.PRESTADOR A
    WHERE (B.CD_PRESTADOR = A.CD_PRESTADOR) AND B.TP_ATENDIMENTO = 'I'
			AND B.DT_ATENDIMENTO BETWEEN :CAMPO_DT_INI GROUP BY A.CD_PRESTADOR , A.NM_PRESTADOR) C ,
   (SELECT A.CD_PRESTADOR , A.NM_PRESTADOR ,COUNT(C. CD_AVISO_CIRURGIA) CIRURGIAS
    FROM   DBAMV.ATENDIME B , DBAMV.PRESTADOR A , DBAMV.AVISO_CIRURGIA C
    WHERE (B.CD_PRESTADOR = A.CD_PRESTADOR) AND (B.CD_ATENDIMENTO = C.CD_ATENDIMENTO)
	       AND (C.CD_CEN_CIR = 1) AND B.DT_ATENDIMENTO BETWEEN :CAMPO_DT_INI GROUP BY A.CD_PRESTADOR , A.NM_PRESTADOR) D ,
   (SELECT A.CD_PRESTADOR , A.NM_PRESTADOR ,COUNT(C. CD_AVISO_CIRURGIA) ESTUDOS_HEMODINAMICOS
    FROM   DBAMV.ATENDIME B , DBAMV.PRESTADOR A , DBAMV.AVISO_CIRURGIA C
    WHERE  (B.CD_PRESTADOR = A.CD_PRESTADOR) AND (B.CD_ATENDIMENTO = C.CD_ATENDIMENTO)
	       AND (C.CD_CEN_CIR = 2) AND B.DT_ATENDIMENTO BETWEEN :CAMPO_DT_INI GROUP BY A.CD_PRESTADOR , A.NM_PRESTADOR) E ,
   (SELECT A.CD_PRESTADOR , A.NM_PRESTADOR,COUNT(C.CD_REG_AMB) ECOGRAFIAS
    FROM   DBAMV.ATENDIME B , DBAMV.PRESTADOR A , DBAMV.ITREG_AMB C
    WHERE  (B.CD_PRESTADOR = A.CD_PRESTADOR) AND (B.CD_ATENDIMENTO = C.CD_ATENDIMENTO)
	       AND (C.CD_PRO_FAT BETWEEN '20010095' AND '20010230' OR
		        C.CD_PRO_FAT BETWEEN '33010013' AND '33010242' OR
		        C.CD_PRO_FAT BETWEEN '20010389' AND '20010486')
			AND B.DT_ATENDIMENTO BETWEEN :CAMPO_DT_INI GROUP BY A.CD_PRESTADOR , A.NM_PRESTADOR) F ,
   (SELECT A.CD_PRESTADOR , A.NM_PRESTADOR,COUNT(C.CD_REG_AMB) ERGOMETRIAS
    FROM  DBAMV.ATENDIME B, DBAMV.PRESTADOR A, DBAMV.ITREG_AMB C
    WHERE  (B.CD_PRESTADOR = A.CD_PRESTADOR) AND (B.CD_ATENDIMENTO = C.CD_ATENDIMENTO)
        	AND (C.CD_PRO_FAT = '20010028') AND B.DT_ATENDIMENTO BETWEEN :CAMPO_DT_INI GROUP BY A.CD_PRESTADOR , A.NM_PRESTADOR) G ,
   (SELECT A.CD_PRESTADOR , A.NM_PRESTADOR,COUNT(C.CD_REG_AMB) HOLTER_MAPA
    FROM   DBAMV.ATENDIME B, DBAMV.PRESTADOR A, DBAMV.ITREG_AMB C
    WHERE (B.CD_PRESTADOR = A.CD_PRESTADOR)	AND (B.CD_ATENDIMENTO = C.CD_ATENDIMENTO)
    	   AND (C.CD_PRO_FAT IN ('20010052','20010079','2020058','20010249'))
			AND B.DT_ATENDIMENTO BETWEEN :CAMPO_DT_INI GROUP BY A.CD_PRESTADOR , A.NM_PRESTADOR) H ,
   (SELECT A.CD_PRESTADOR , A.NM_PRESTADOR,COUNT(C.CD_REG_AMB) ECG
    FROM   DBAMV.ATENDIME B, DBAMV.PRESTADOR A, DBAMV.ITREG_AMB C
    WHERE (B.CD_PRESTADOR = A.CD_PRESTADOR)	AND (B.CD_ATENDIMENTO = C.CD_ATENDIMENTO)
            AND (C.CD_PRO_FAT = '20010010') AND B.DT_ATENDIMENTO BETWEEN :CAMPO_DT_INI GROUP BY A.CD_PRESTADOR , A.NM_PRESTADOR) I ,
   (SELECT A.CD_PRESTADOR , A.NM_PRESTADOR,COUNT(C.CD_REG_AMB) ANALISES_CLINICAS
    FROM   DBAMV.ATENDIME B, DBAMV.PRESTADOR A, DBAMV.ITREG_AMB C
    WHERE (B.CD_PRESTADOR = A.CD_PRESTADOR)	AND (B.CD_ATENDIMENTO = C.CD_ATENDIMENTO)
	       AND (C.CD_PRO_FAT LIKE '28%') AND B.DT_ATENDIMENTO BETWEEN :CAMPO_DT_INI GROUP BY A.CD_PRESTADOR , A.NM_PRESTADOR) J
WHERE
    A.CD_PRESTADOR = B.CD_PRESTADOR(+) AND
    A.CD_PRESTADOR = C.CD_PRESTADOR(+) AND
    A.CD_PRESTADOR = D.CD_PRESTADOR(+) AND
    A.CD_PRESTADOR = E.CD_PRESTADOR(+) AND
    A.CD_PRESTADOR = F.CD_PRESTADOR(+) AND
    A.CD_PRESTADOR = G.CD_PRESTADOR(+) AND
    A.CD_PRESTADOR = H.CD_PRESTADOR(+) AND
    A.CD_PRESTADOR = I.CD_PRESTADOR(+) AND
    A.CD_PRESTADOR = J.CD_PRESTADOR(+) AND
	(B.CONSULTAS			IS NOT NULL OR
	C.INTERNACOES			IS NOT NULL OR
	D.CIRURGIAS				IS NOT NULL OR
	E.ESTUDOS_HEMODINAMICOS	IS NOT NULL OR
	F.ECOGRAFIAS			IS NOT NULL OR
	G.ERGOMETRIAS			IS NOT NULL OR
	H.HOLTER_MAPA			IS NOT NULL OR
	I.ECG					IS NOT NULL OR
	J.ANALISES_CLINICAS		IS NOT NULL)
