--R_PRE_INTERNACAO_PERIODO
SELECT * FROM (
SELECT  DISTINCT PACIENTE.CD_PACIENTE
       ,NVL(PACIENTE.NM_PACIENTE,RES_LEI.NM_PACIENTE)  NM_PACIENTE
       ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  CD_PACIENTE) NM_SOCIAL_PACIENTE
       ,PACIENTE.NR_FONE
       ,SAME.NR_MATRICULA_SAME
       ,SUBSTR(SAME.NR_MATRICULA_SAME, -2 , 1) DIGITO
       ,CONVENIO.NM_CONVENIO
       ,CON_PLA.DS_CON_PLA
       ,PRESTADOR.NM_PRESTADOR
       ,LEITO.DS_RESUMO
       ,TIP_ACOM.DS_TIP_ACOM
       ,DECODE(RES_LEI.SN_ACOMPANHANTE,NULL,'','S','Sim','N','Não') SN_ACOMP
       ,DECODE(RES_LEI.SN_ACOMPANHANTE,NULL,'','S','Sim','N','Nao') SN_ACOMP1
       ,TRUNC(RES_LEI.DT_PREV_INTERNACAO) DT_INTERNACAO
       ,RES_LEI.CD_PRO_INT
       ,PRO_FAT.DS_PRO_FAT
       ,CID.CD_CID
       ,CID.DS_CID
       ,RES_LEI.CD_AVISO_CIRURGIA 
       ,AVISO_CIRURGIA.DT_AVISO_CIRURGIA 
       ,RES_LEI.SN_ATIVO
  FROM  DBAMV.ATENDIME
       ,DBAMV.PACIENTE
       ,DBAMV.CONVENIO
       ,DBAMV.CON_PLA
       ,DBAMV.SAME
       ,DBAMV.PRESTADOR
       ,DBAMV.TIP_ACOM
       ,DBAMV.LEITO
       ,DBAMV.CID
       ,DBAMV.PRO_FAT
       ,DBAMV.RES_LEI
       ,DBAMV.AVISO_CIRURGIA 
 WHERE RES_LEI.CD_PACIENTE                 = PACIENTE.CD_PACIENTE(+)
   AND     RES_LEI.CD_ATENDIMENTO        = ATENDIME.CD_ATENDIMENTO(+)
   AND     RES_LEI.CD_CONVENIO                = CONVENIO.CD_CONVENIO(+)
   AND     RES_LEI.CD_CON_PLA                  = CON_PLA.CD_CON_PLA(+)
   AND     RES_LEI.CD_CONVENIO                = CON_PLA.CD_CONVENIO(+)
   AND     PACIENTE.CD_PACIENTE              = SAME.CD_PACIENTE(+)
   AND     RES_LEI.CD_PRESTADOR            = PRESTADOR.CD_PRESTADOR(+)
   AND     RES_LEI.CD_TIP_ACOM                = TIP_ACOM.CD_TIP_ACOM(+)
   AND     RES_LEI.CD_LEITO                        = LEITO.CD_LEITO(+)
   AND     RES_LEI.CD_CID                            = CID.CD_CID(+)
   AND     RES_LEI.CD_PRO_INT                  = PRO_FAT.CD_PRO_FAT(+)
   AND     RES_LEI.CD_AVISO_CIRURGIA      = AVISO_CIRURGIA.CD_AVISO_CIRURGIA(+) 
   AND     RES_LEI.DT_PREV_INTERNACAO IS NOT NULL
   AND    TRUNC(RES_LEI.DT_PREV_INTERNACAO) BETWEEN  trunc(:CAMPO_DT_INI) AND trunc(:CAMPO_DT_FIM)
    AND    ATENDIME.CD_MULTI_EMPRESA (+) = :CAMPO_CD_MULTI_EMPRESA  
   AND    RES_LEI.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
  ORDER BY    DT_INTERNACAO
                        ,DIGITO
                        ,NM_PACIENTE
)
ORDER BY DT_INTERNACAO
