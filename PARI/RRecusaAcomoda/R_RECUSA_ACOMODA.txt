--R_RECUSA_ACOMODA
SELECT * FROM (
SELECT RECUSA_ACOMODA.CD_RECUSA_ACOM              CD_RECUSA_ACOM
              ,RECUSA_ACOMODA.DT_SOLICITACAO                  DT_SOLICITACAO
              ,RECUSA_ACOMODA.HR_SOLICITACAO                  HR_SOLICITACAO
              ,RECUSA_ACOMODA.NM_PACIENTE                       NM_PACIENTE
              ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND RECUSA_ACOMODA.NM_PACIENTE =  P.NM_PACIENTE) NM_SOCIAL_PACIENTE
              ,RECUSA_ACOMODA.NM_USUARIO                        NM_USUARIO
              ,RECUSA_ACOMODA.NM_PRESTADOR_SOLIC      NM_PRESTADOR_SOLIC
              ,CONVENIO.NM_CONVENIO                                      NM_CONVENIO
              ,ORI_ATE.DS_ORI_ATE                                             DS_ORI_ATE
              ,TIP_ACOM.DS_TIP_ACOM                                        DS_TIP_ACOM
              ,PRESTADOR.NM_PRESTADOR                               NM_PRESTADOR
              ,LOC_TRANS.DS_LOC_TRANS                                  DS_LOC_TRANS
FROM   DBAMV.RECUSA_ACOMODA
            ,DBAMV.CONVENIO
            ,DBAMV.ORI_ATE
            ,DBAMV.TIP_ACOM
            ,DBAMV.PRESTADOR
            ,DBAMV.LOC_TRANS
WHERE TRUNC(RECUSA_ACOMODA.DT_SOLICITACAO) BETWEEN :CAMPO_DATINI AND 
AND       RECUSA_ACOMODA.CD_CONVENIO = CONVENIO.CD_CONVENIO(+)
AND       RECUSA_ACOMODA.CD_ORI_ATE = ORI_ATE.CD_ORI_ATE
AND       RECUSA_ACOMODA.CD_TIP_ACOM = TIP_ACOM.CD_TIP_ACOM
AND       RECUSA_ACOMODA.CD_PRESTADOR_DEST = PRESTADOR.CD_PRESTADOR(+)
AND       RECUSA_ACOMODA.CD_LOC_TRANS = LOC_TRANS.CD_LOC_TRANS(+)
:CAMPO_CF_MULTI_EMPRESA
)
ORDER BY DS_TIP_ACOM
