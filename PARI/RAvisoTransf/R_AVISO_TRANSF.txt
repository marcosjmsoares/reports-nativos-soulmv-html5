--R_AVISO_TRANSF
SELECT ATENDIME.CD_ATENDIMENTO     CD_ATENDIMENTO
             ,LEITO_ANT.DS_RESUMO                DS_RESUMO_ANT
             ,LEITO_ANT.CD_LEITO                     CD_LEITO_ANT
             ,UNID_ANT.CD_UNID_INT                CD_UNID_ANT
             ,UNID_ANT.DS_UNID_INT                DS_UNID_ANT
             ,TIP_ACOM_ANT.CD_TIP_ACOM      CD_TIP_ACOM_ANT
             ,TIP_ACOM_ANT.DS_TIP_ACOM      DS_TIP_ACOM_ANT
             ,PACIENTE.CD_PACIENTE                CD_PACIENTE
             ,PACIENTE.NM_PACIENTE               NM_PACIENTE
             ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  CD_PACIENTE) NM_SOCIAL_PACIENTE
             ,CONVENIO.CD_CONVENIO              CD_CONVENIO
             ,CONVENIO.NM_CONVENIO             NM_CONVENIO
             ,PRESTADOR.CD_PRESTADOR      CD_PRESTADOR
             ,PRESTADOR.NM_PRESTADOR      NM_PRESTADOR
             ,ATENDIME.DT_ATENDIMENTO      DT_ATENDIMENTO
             ,ATENDIME.HR_ATENDIMENTO      HR_ATENDIMENTO
             ,MOV_INT.NM_USUARIO                  NM_USUARIO
             ,MOV_INT.DT_MOV_INT                   DT_MOV_INT
             ,MOV_INT.HR_MOV_INT                   HR_MOV_INT
             ,LEITO.DS_RESUMO                         DS_LEITO
             ,UNID_INT.CD_UNID_INT                 CD_UNID_INT
             ,UNID_INT.DS_UNID_INT                 DS_UNID_INT
             ,TIP_ACOM.CD_TIP_ACOM               CD_TIP_ACOM
             ,TIP_ACOM.DS_TIP_ACOM               DS_TIP_ACOM
FROM DBAMV.MOV_INT
          ,DBAMV.ATENDIME
          ,DBAMV.LEITO           LEITO_ANT
          ,DBAMV.UNID_INT    UNID_ANT
          ,DBAMV.TIP_ACOM   TIP_ACOM_ANT
          ,DBAMV.LEITO
          ,DBAMV.UNID_INT
          ,DBAMV.TIP_ACOM
          ,DBAMV.PRESTADOR
          ,DBAMV.CONVENIO
          ,DBAMV.PACIENTE
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And MOV_INT.TP_MOV = 'O'
AND   MOV_INT.CD_MOV_INT = :CAMPO_PCD_MOV_INT
AND   MOV_INT.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
AND   MOV_INT.CD_LEITO = LEITO.CD_LEITO(+)
AND   LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)
AND   LEITO.CD_TIP_ACOM = TIP_ACOM.CD_TIP_ACOM(+)
AND   MOV_INT.CD_LEITO_ANTERIOR = LEITO_ANT.CD_LEITO(+)
AND   LEITO_ANT.CD_UNID_INT = UNID_ANT.CD_UNID_INT(+)
AND   LEITO_ANT.CD_TIP_ACOM = TIP_ACOM_ANT.CD_TIP_ACOM(+)
AND   MOV_INT.CD_CONVENIO = CONVENIO.CD_CONVENIO(+)
AND   MOV_INT.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
AND   ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
