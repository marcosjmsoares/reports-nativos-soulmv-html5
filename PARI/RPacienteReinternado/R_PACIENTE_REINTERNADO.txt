--R_PACIENTE_REINTERNADO
SELECT DISTINCT ATENDIME.CD_ATENDIMENTO ,
                PACIENTE.CD_PACIENTE ,
                PACIENTE.NM_PACIENTE ,
        (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
                PACIENTE.DT_NASCIMENTO ,
                ATENDIME.DT_ATENDIMENTO ,
                CIDADE.NM_CIDADE
FROM DBAMV.ATENDIME ,
     DBAMV.MOT_ALT ,
     DBAMV.LEITO ,
     DBAMV.PACIENTE ,
     DBAMV.CIDADE ,
     DBAMV.ATENDIME ATE
WHERE ATENDIME.TP_ATENDIMENTO = 'I'
  AND TRUNC(ATENDIME.DT_ATENDIMENTO) BETWEEN :CAMPO_DT_INI AND 
  AND ATENDIME.CD_MOT_ALT = MOT_ALT.CD_MOT_ALT(+)
  AND (ATENDIME.CD_MOT_ALT IS NULL
       OR MOT_ALT.TP_MOT_ALTA <> 'D')
  AND ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
  AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
  AND PACIENTE.CD_CIDADE = CIDADE.CD_CIDADE(+)
  AND TRUNC(ATE.DT_ATENDIMENTO) BETWEEN :CAMPO_DT_INI AND 
  AND ATE.TP_ATENDIMENTO = 'I'
  AND ATENDIME.CD_PACIENTE = ATE.CD_PACIENTE
  AND ATENDIME.CD_ATENDIMENTO <> ATE.CD_ATENDIMENTO 
  AND ATE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA 
  AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
