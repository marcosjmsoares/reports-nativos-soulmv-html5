--R_INT_PAC
SELECT PACIENTE.CD_PACIENTE CD_PACIENTE ,
			       PACIENTE.NM_PACIENTE NM_PACIENTE ,
			  (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
			       LEITO.DS_RESUMO DS_RESUMO ,
			       LEITO.DS_ENFERMARIA DS_ENFERMARIA ,
			       ATENDIME.DT_ATENDIMENTO DT_ATENDIMENTO ,
			       ATENDIME.HR_ATENDIMENTO HR_ATENDIMENTO ,
			       PRESTADOR.NM_PRESTADOR NM_PRESTADOR ,
			       ATENDIME.CD_ATENDIMENTO CD_ATENDIMENTO ,
			       CONVENIO.CD_CONVENIO CD_CONVENIO ,
			       CONVENIO.NM_CONVENIO NM_CONVENIO
			FROM DBAMV.ATENDIME ,
			     DBAMV.PACIENTE ,
			     DBAMV.LEITO LEITO ,
			     DBAMV.PRESTADOR ,
			     DBAMV.CONVENIO ,
			     DBAMV.CON_PLA CON_PLA , 
			     DBAMV.empresa_con_pla ,  
			     DBAMV.UNID_INT UNID_INT 
			WHERE EMPRESA_CON_PLA.CD_CONVENIO = CON_PLA.CD_CONVENIO 
			  AND EMPRESA_CON_PLA.CD_CON_PLA = CON_PLA.CD_CON_PLA 
			  AND EMPRESA_CON_PLA.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA 
			  AND ATENDIME.CD_CONVENIO = CON_PLA.CD_CONVENIO(+)
			  AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO(+)
			  AND ATENDIME.CD_CON_PLA = CON_PLA.CD_CON_PLA(+) 
			  AND ATENDIME.TP_ATENDIMENTO IN ('H',
			                                  'I')
			  AND ATENDIME.CD_ATENDIMENTO_PAI IS NULL
			  AND LEITO.CD_LEITO(+) = ATENDIME.CD_LEITO
			  AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)
			  AND PACIENTE.CD_PACIENTE = ATENDIME.CD_PACIENTE
			  AND PRESTADOR.CD_PRESTADOR = ATENDIME.CD_PRESTADOR(+)
			  AND DBAMV.FNC_MV_USUARIO_UNIDADE_SETOR(NULL, NULL, atendime.cd_ori_ate, atendime.cd_leito) = 'S' 
			  AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA      
			ORDER BY PACIENTE.NM_PACIENTE
