--R_EST_INTER_UNID_CONV_ATUAL
SELECT UNID_INT.CD_UNID_INT
             ,UNID_INT.DS_UNID_INT
             ,COUNT(*)  QUANT
             ,(SELECT COUNT(*)  QUANT
FROM   DBAMV.ATENDIME
            ,DBAMV.LEITO
            ,DBAMV.UNID_INT
            ,DBAMV.CONVENIO
WHERE TRUNC(ATENDIME.DT_ATENDIMENTO)  BETWEEN :CAMPO_DT_INICIO AND 
AND        ATENDIME.CD_LEITO  =  LEITO.CD_LEITO
AND        LEITO.CD_UNID_INT  =  UNID_INT.CD_UNID_INT
AND        ATENDIME.CD_ATENDIMENTO_PAI IS NULL
AND        ATENDIME.TP_ATENDIMENTO IN ('I', 'H')
AND        ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
:CAMPO_CF_MULTI_EMPRESA
) SOMA_FINAL
FROM   DBAMV.ATENDIME
            ,DBAMV.LEITO
            ,DBAMV.UNID_INT
            ,DBAMV.CONVENIO
WHERE TRUNC(ATENDIME.DT_ATENDIMENTO)  BETWEEN :CAMPO_DT_INICIO AND 
AND        ATENDIME.CD_LEITO  =  LEITO.CD_LEITO
AND        LEITO.CD_UNID_INT  =  UNID_INT.CD_UNID_INT
AND        ATENDIME.CD_ATENDIMENTO_PAI IS NULL
AND        ATENDIME.TP_ATENDIMENTO IN ('I', 'H')
AND        ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY  UNID_INT.CD_UNID_INT
                    ,UNID_INT.DS_UNID_INT
ORDER BY  QUANT DESC
