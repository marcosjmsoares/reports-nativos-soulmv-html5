--R_CENSO_CONV
SELECT * FROM (
SELECT  'U' TIPO
               ,LEITO.CD_UNID_INT CD_UNID_INT
               ,LEITO.DS_RESUMO
               ,LEITO.TP_OCUPACAO
               ,LEITO.DS_ENFERMARIA
               ,UNID_INT.DS_UNID_INT
               ,PACIENTE.NM_PACIENTE
               ,PACIENTE.DT_NASCIMENTO
               ,CONVENIO.NM_CONVENIO
               ,PRESTADOR.NM_PRESTADOR
               ,ATENDIME.DT_ATENDIMENTO
               ,ATENDIME.CD_ATENDIMENTO
               ,LEITO.CD_LEITO
               ,TO_NUMBER(NULL) CODIGO_RESUMO
               ,NULL DESCRICAO_RESUMO
               ,0 INTERNOS
               ,0 INTERNOS_DIA
               ,0 OBITOS_DIA
               ,0 ALTAS_DIA
               ,0 ATIVOS_ACOM
               ,0 EXTRAS_ACOM
               ,0 INTERNOS_ACOM
               ,LEITO.SN_EXTRA
               ,TO_NUMBER(NULL) CD_CONV_UNID
               ,NULL DS_CONV_UNID
               ,0 QT_CONV_UNID
FROM      DBAMV.V_INTERNADOS
                ,DBAMV.LEITO
                ,DBAMV.UNID_INT
                ,DBAMV.ATENDIME
                ,DBAMV.PACIENTE
                ,DBAMV.CONVENIO
                ,DBAMV.PRESTADOR
WHERE   V_INTERNADOS.CD_LEITO (+) =  LEITO.CD_LEITO
      AND   LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT
      AND :CAMPO_DATA >= leito.dt_ativacao
      AND :CAMPO_DATA <= nvl(leito.dt_desativacao,sysdate)
      AND   V_INTERNADOS.CD_ATENDIMENTO =  ATENDIME.CD_ATENDIMENTO (+)
      AND   ATENDIME.CD_PACIENTE  =  PACIENTE.CD_PACIENTE (+)
      AND   ATENDIME.CD_CONVENIO   =   CONVENIO.CD_CONVENIO (+)
      AND   PRESTADOR.CD_PRESTADOR (+) =  ATENDIME.CD_PRESTADOR
      AND ( LEITO.TP_SITUACAO = DECODE( :CAMPO_LEITO, 'S', 'I', 'N', '1' )    OR  LEITO.TP_SITUACAO <> 'I' )
UNION ALL
(SELECT  'C' TIPO
       ,TO_NUMBER(NULL) CD_UNID_INT,NULL,NULL,NULL,NULL,NULL,TO_DATE(NULL),NULL,NULL,TO_DATE(NULL)
       ,TO_NUMBER(NULL),TO_NUMBER(NULL)
       ,CONVENIO.CD_CONVENIO
       ,CONVENIO.NM_CONVENIO
       ,SUM(INTERNOS)     INTERNOS
       ,SUM(INTERNOS_DIA) INTERNOS_DIA
       ,SUM(OBITOS_DIA)   OBITOS_DIA
       ,SUM(ALTAS_DIA)    ALTAS_DIA
       ,0 ATIVOS_ACOM
       ,0 EXTRAS_ACOM
       ,0 INTERNOS_ACOM
       ,NULL
       ,TO_NUMBER(NULL) CD_CONV_UNID
       ,NULL DS_CONV_UNID
      ,0 QT_CONV_UNID
FROM
      (SELECT
       CONVENIO.CD_CONVENIO
      ,CONVENIO.NM_CONVENIO
      ,COUNT(ATENDIME.CD_CONVENIO) INTERNOS
      ,0 INTERNOS_DIA
      ,0 OBITOS_DIA
      ,0 ALTAS_DIA
       FROM        DBAMV.V_INTERNADOS    V_INTERNADOS
                  ,DBAMV.LEITO           LEITO
                  ,DBAMV.UNID_INT        UNID_INT
                  ,DBAMV.ATENDIME        ATENDIME
                  ,DBAMV.CONVENIO        CONVENIO
       WHERE       V_INTERNADOS.CD_LEITO  =  LEITO.CD_LEITO
       AND        (LEITO.TP_SITUACAO  =  DECODE(:CAMPO_LEITO, 'S', 'I', 'N', '1' ) OR LEITO.TP_SITUACAO <> 'I')
       AND        LEITO.CD_UNID_INT  =  UNID_INT.CD_UNID_INT
       AND        :CAMPO_DATA >= leito.dt_ativacao
       AND        :CAMPO_DATA <= nvl(leito.dt_desativacao,sysdate)
       AND        ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
       AND        V_INTERNADOS.CD_ATENDIMENTO  =  ATENDIME.CD_ATENDIMENTO
       GROUP BY CONVENIO.CD_CONVENIO
               ,CONVENIO.NM_CONVENIO
       UNION ALL
       SELECT
       CONVENIO.CD_CONVENIO
      ,CONVENIO.NM_CONVENIO
      ,0 INTERNOS
      ,SUM(DECODE(ATENDIME.DT_ATENDIMENTO,:CAMPO_DATA,1,0))                        INTERNOS_DIA
      ,SUM(DECODE(ATENDIME.DT_ALTA || MOT_ALT.TP_MOT_ALTA,:CAMPO_DATA || 'O',1,0)) OBITOS_DIA
      ,SUM(DECODE(ATENDIME.DT_ALTA || MOT_ALT.TP_MOT_ALTA,:CAMPO_DATA || 'D',1,
          :CAMPO_DATA || 'A',1,:CAMPO_DATA || 'T',1,0))                          ALTAS_DIA
       FROM        DBAMV.LEITO           LEITO
                  ,DBAMV.UNID_INT        UNID_INT
                  ,DBAMV.ATENDIME        ATENDIME
                  ,DBAMV.CONVENIO        CONVENIO
                  ,DBAMV.MOT_ALT         MOT_ALT
       WHERE       ATENDIME.CD_LEITO  =  LEITO.CD_LEITO
       AND        (LEITO.TP_SITUACAO  =  DECODE(:CAMPO_LEITO, 'S', 'I', 'N', '1' ) OR LEITO.TP_SITUACAO <> 'I')
       AND        LEITO.CD_UNID_INT  =  UNID_INT.CD_UNID_INT
       AND        :CAMPO_DATA >= leito.dt_ativacao
       AND        :CAMPO_DATA <= nvl(leito.dt_desativacao,sysdate)
       AND        ATENDIME.TP_ATENDIMENTO = 'I'
       AND        ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
       AND        ATENDIME.CD_MOT_ALT = MOT_ALT.CD_MOT_ALT(+)
       GROUP BY CONVENIO.CD_CONVENIO
               ,CONVENIO.NM_CONVENIO) CONVENIO
WHERE NVL(CONVENIO.INTERNOS,0) > 0   OR
      NVL(CONVENIO.INTERNOS_DIA,0) > 0  OR
      NVL(CONVENIO.OBITOS_DIA,0) > 0  OR
      NVL(CONVENIO.ALTAS_DIA,0) > 0
GROUP BY CONVENIO.CD_CONVENIO
        ,CONVENIO.NM_CONVENIO
UNION ALL
SELECT  'A' TIPO
       ,TO_NUMBER(NULL) CD_UNID_INT,NULL,NULL,NULL,NULL,NULL,TO_DATE(NULL),NULL,NULL,TO_DATE(NULL)
       ,TO_NUMBER(NULL),TO_NUMBER(NULL)
       ,TIP_ACOM.CD_TIP_ACOM
       ,TIP_ACOM.DS_TIP_ACOM
       ,0 INTERNOS
       ,0 INTERNOS_DIA
       ,0 OBITOS_DIA
       ,0 ALTAS_DIA
       ,COUNT(*)                         ATIVOS_ACOM
       ,SUM(DECODE(SN_EXTRA,'S',1,0))    EXTRAS_ACOM
       ,SUM(DECODE(TP_OCUPACAO,'O',1,0)) INTERNOS_ACOM
       ,NULL
       ,TO_NUMBER(NULL) CD_CONV_UNID
      ,NULL DS_CONV_UNID
       ,0 QT_CONV_UNID
FROM    DBAMV.LEITO
       ,DBAMV.UNID_INT
       ,DBAMV.TIP_ACOM
WHERE  :CAMPO_DATA >= leito.dt_ativacao
  AND  :CAMPO_DATA <= nvl(leito.dt_desativacao,sysdate)
  AND  (LEITO.TP_SITUACAO = DECODE(:CAMPO_LEITO, 'S', 'I', 'N', '1' )  or  LEITO.TP_SITUACAO <> 'I')
  AND   LEITO.CD_UNID_INT =  UNID_INT.CD_UNID_INT
  AND   LEITO.CD_TIP_ACOM =  TIP_ACOM.CD_TIP_ACOM
GROUP BY  TIP_ACOM.DS_TIP_ACOM
         ,TIP_ACOM.CD_TIP_ACOM)
UNION ALL
SELECT  'R' TIPO
           ,LEITO.CD_UNID_INT CD_UNID_INT
          ,NULL,NULL,NULL,NULL,NULL,TO_DATE(NULL),NULL,NULL,TO_DATE(NULL)
          ,TO_NUMBER(NULL),TO_NUMBER(NULL)
               ,TO_NUMBER(NULL) CODIGO_RESUMO
               ,NULL DESCRICAO_RESUMO
               ,0 INTERNOS
               ,0 INTERNOS_DIA
               ,0 OBITOS_DIA
               ,0 ALTAS_DIA
               ,0 ATIVOS_ACOM
               ,0 EXTRAS_ACOM
               ,0 INTERNOS_ACOM
               ,NULL
               ,ATENDIME.CD_CONVENIO CD_CONV_UNID
               ,CONVENIO.NM_CONVENIO DS_CONV_UNID
               ,COUNT(*) QT_CONV_UNID
FROM      DBAMV.V_INTERNADOS
                ,DBAMV.LEITO
                ,DBAMV.UNID_INT
                ,DBAMV.ATENDIME
                ,DBAMV.PACIENTE
                ,DBAMV.CONVENIO
                ,DBAMV.PRESTADOR
WHERE   V_INTERNADOS.CD_LEITO (+) =  LEITO.CD_LEITO
      AND   LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT
      AND :CAMPO_DATA >= leito.dt_ativacao
      AND :CAMPO_DATA <= nvl(leito.dt_desativacao,sysdate)
      AND   V_INTERNADOS.CD_ATENDIMENTO =  ATENDIME.CD_ATENDIMENTO (+)
      AND   ATENDIME.CD_PACIENTE  =  PACIENTE.CD_PACIENTE (+)
      AND   ATENDIME.CD_CONVENIO   =   CONVENIO.CD_CONVENIO (+)
      AND   PRESTADOR.CD_PRESTADOR (+) =  ATENDIME.CD_PRESTADOR
      AND ( LEITO.TP_SITUACAO = DECODE( :CAMPO_LEITO, 'S', 'I', 'N', '1' )    OR  LEITO.TP_SITUACAO <> 'I' )
      AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY LEITO.CD_UNID_INT
                  ,CONVENIO.NM_CONVENIO
                  ,ATENDIME.CD_CONVENIO
order by 6
:CAMPO_CF_ORDEM
)
ORDER BY DT_NASCIMENTO,CD_UNID_INT,CODIGO_RESUMO,DESCRICAO_RESUMO,DS_UNID_INT,INTERNOS,INTERNOS_DIA,OBITOS_DIA,ALTAS_DIA,ATIVOS_ACOM,EXTRAS_ACOM,INTERNOS_ACOM,TIPO ASC
