--R_PACIENTE_POR_CID_UNID
SELECT * FROM
  (SELECT UNID_INT.CD_UNID_INT ,
          UNID_INT.DS_UNID_INT ,
          ATENDIME.CD_ATENDIMENTO ,
          ATENDIME.TP_ATENDIMENTO ,
          ATENDIME.DT_ATENDIMENTO ,
          ATENDIME.CD_PACIENTE ,
          PACIENTE.NM_PACIENTE ,
     (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
          ATENDIME.CD_CID ,
          CID.DS_CID
   FROM DBAMV.ATENDIME ,
        DBAMV.LEITO ,
        DBAMV.UNID_INT ,
        DBAMV.PACIENTE ,
        DBAMV.CID
   WHERE TRUNC(ATENDIME.DT_ATENDIMENTO) BETWEEN :CAMPO_DT_INICIO AND 
     AND ATENDIME.CD_LEITO = LEITO.CD_LEITO
     AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT
     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
     AND ATENDIME.CD_CID = CID.CD_CID
     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA    
   ORDER BY NM_PACIENTE,
            DT_ATENDIMENTO)
ORDER BY CD_UNID_INT,
         DS_UNID_INT,
         CD_CID,
         DS_CID
