--R_PACIENTE_POR_CID_R_ATENDIMENTO_Q_PAC_CID
SELECT ATENDIME.CD_ATENDIMENTO CD_ATENDIMENTO ,
			       ATENDIME.TP_ATENDIMENTO TP_ATENDIMENTO ,
			       ATENDIME.DT_ATENDIMENTO DT_ATENDIMENTO ,
			       ATENDIME.DT_ALTA DT_ALTA ,
			       ATENDIME.CD_PACIENTE CD_PACIENTE ,
			       PACIENTE.NM_PACIENTE NM_PACIENTE ,
			  (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
			       ATENDIME.CD_CID CD_CID ,
			       CID.DS_CID DS_CID ,
			       PACIENTE.TP_SEXO TP_SEXO ,
			       DBAMV.PACK_INTERNAMENTO.RETORNA_IDADE(PACIENTE.DT_NASCIMENTO, sysdate) IDADE_PAC ,
			       CONVENIO.NM_CONVENIO NM_CONVENIO ,
			       CON_PLA.DS_CON_PLA DS_CON_PLA ,
			       CARTEIRA.NM_EMPRESA NM_EMPRESA
			FROM DBAMV.ATENDIME ,
			     DBAMV.PACIENTE ,
			     DBAMV.CID ,
			     DBAMV.CONVENIO ,
			     DBAMV.CON_PLA ,
			     DBAMV.CARTEIRA
			WHERE TRUNC(ATENDIME.DT_ATENDIMENTO) BETWEEN :CAMPO_DT_INICIO AND 
			  AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
			  AND ATENDIME.CD_CID = CID.CD_CID
			  AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
			  AND ATENDIME.CD_CONVENIO = CON_PLA.CD_CONVENIO(+)
			  AND ATENDIME.CD_CON_PLA = CON_PLA.CD_CON_PLA(+)
			  AND ATENDIME.CD_PACIENTE = CARTEIRA.CD_PACIENTE(+)
			  AND ATENDIME.CD_CONVENIO = CARTEIRA.CD_CONVENIO(+)
			  AND ATENDIME.CD_CON_PLA = CARTEIRA.CD_CON_PLA(+)
			  AND atendime.nr_carteira = carteira.nr_carteira(+)
			  AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
			  AND ATENDIME.TP_ATENDIMENTO LIKE :CAMPO_TP_ATENDIMENTO
