--R_ALTASCONVENIO_ENDERECO
SELECT * FROM (
SELECT conv.CD_CONVENIO,
               conv.NM_CONVENIO,
               aten.CD_ATENDIMENTO,
               paci.NM_PACIENTE,
               aten.DT_ATENDIMENTO,
               aten.DT_ALTA,
               aten.HR_ALTA,
               copl.DS_CON_PLA,
               pres.NM_PRESTADOR,
               DECODE(CONV.TP_CONVENIO, 'C',  ATEN.CD_PRO_INT,
                                                                      'P', ATEN.CD_PRO_INT,
                                                                      'A', PRO_FAT.CD_SUS,
                                                                      'H', PRO_FAT.CD_SUS)     CD_PRO_INT,
               PRO_FAT.DS_PRO_FAT      DS_PRO_INT,
               leito.DS_RESUMO,
               MOT_ALT.DS_MOT_ALT,
               paci.DS_ENDERECO  ||  '  -  ' || paci.NM_BAIRRO   DS_ENDERECO,
               paci.NR_FONE,
               ori.DS_ORI_ATE
   FROM DBAMV.ATENDIME aten,
              DBAMV.PRO_FAT,
              DBAMV.PACIENTE paci,
              DBAMV.CON_PLA copl,
              DBAMV.CONVENIO conv,
              DBAMV.PRESTADOR pres,
              DBAMV.LEITO  leito,
              DBAMV.MOT_ALT,
              DBAMV.ORI_ATE  ori
 WHERE aten.TP_ATENDIMENTO IN ( 'I', 'H' )
       AND aten.DT_ALTA is not null
       AND aten.DT_ALTA between :CAMPO_DATINI and 
       AND aten.CD_PACIENTE         = paci.CD_PACIENTE
       AND aten.CD_PRO_INT           = pro_fat.CD_PRO_FAT(+)
       AND aten.CD_CONVENIO        = copl.CD_CONVENIO(+)
       AND aten.CD_CON_PLA          = copl.CD_CON_PLA(+)
       AND aten.CD_LEITO                = leito.CD_LEITO(+)
       AND aten.CD_CONVENIO        = conv.CD_CONVENIO
       AND aten.CD_PRESTADOR    = pres.CD_PRESTADOR
       AND aten.CD_MOT_ALT          = mot_alt.CD_MOT_ALT
      AND  aten.CD_ORI_ATE(+)       = ori.CD_ORI_ATE
      AND ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
ORDER BY conv.CD_CONVENIO,
		paci.NM_PACIENTE
)
ORDER BY NM_CONVENIO, NM_PACIENTE,CD_CONVENIO
