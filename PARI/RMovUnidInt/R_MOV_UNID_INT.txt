--R_MOV_UNID_INT_R_15_Q_TRANSFER_PA
SELECT * FROM (
			    SELECT unid_int.cd_unid_int cd_unid_int_tpa ,
			           unid_int.ds_unid_int ds_unid_int_tpa ,
			           trunc(mov_int.dt_mov_int) dt_mov_int_tpa ,
			           atendime.cd_atendimento cd_atendimento_tpa ,
			           paciente.cd_paciente cd_paciente_tpa ,
			           paciente.nm_paciente nm_paciente_tpa ,
			       (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
			           SubStr(To_Char(mov_int.hr_mov_int, 'hh24'), 0, 5) hr_mov_int_tpa ,
			           'TRANSFER. PARA' tp_mov_tpa ,
			                            leito_ori.ds_resumo ds_resumo_ori_tpa ,
			                            leito_des.ds_resumo ds_resumo_des_tpa ,
			                            cid.cd_cid cd_cid_tpa ,
			                            cid.cd_cid ||' - '|| cid.ds_cid ds_cid_tpa ,
			                            convenio.nm_convenio nm_convenio_tpa ,
			                            ' ' ds_mot_alt_tpa ,
			                                tip_acom.ds_tip_acom ds_tip_acom_tpa
			    FROM dbamv.MOV_INT ,
			        dbamv.ATENDIME ,
			        dbamv.PACIENTE ,
			        dbamv.CONVENIO ,
			        dbamv.leito leito_ori ,
			        dbamv.leito leito_des ,
			        dbamv.UNID_INT ,
			        dbamv.cid ,
			        dbamv.tip_acom,
			        Dbamv.Empresa_Convenio
			    WHERE Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
			     AND Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
			     AND MOV_INT.TP_MOV = 'O'
			     AND TRUNC(MOV_INT.DT_MOV_INT) BETWEEN :CAMPO_DT_INICIAL AND 
			     AND ATENDIME.CD_TIP_ACOM_COBERTURA = TIP_ACOM.CD_TIP_ACOM(+)
			     AND TRUNC(NVL(MOV_INT.DT_LIB_MOV, SYSDATE)) >= TRUNC(MOV_INT.DT_MOV_INT)
			     AND ATENDIME.TP_ATENDIMENTO = 'I'
			     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
			     AND MOV_INT.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
			     AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
			     AND UNID_INT.SN_ATIVO = 'S'
			     AND ATENDIME.CD_CID = CID.CD_CID(+)
			     AND MOV_INT.CD_LEITO_ANTERIOR = LEITO_ORI.CD_LEITO
			     AND LEITO_ORI.CD_UNID_INT = UNID_INT.CD_UNID_INT
			     AND MOV_INT.CD_LEITO = LEITO_DES.CD_LEITO
			     AND LEITO_ORI.CD_UNID_INT != LEITO_DES.CD_UNID_INT  
			    ORDER BY cd_unid_int_tpa,
			            dt_mov_int_tpa,
			            nm_paciente_tpa)
			ORDER BY CD_UNID_INT_TPA,
			         DS_UNID_INT_TPA,
			         DT_MOV_INT_TPA
