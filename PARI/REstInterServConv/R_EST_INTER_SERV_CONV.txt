Select Servico.Cd_Servico
       ,Servico.Ds_Servico
       ,Convenio.Cd_Convenio
       ,Convenio.Nm_Convenio
       ,Count(*) qtd_Servico
  From Dbamv.Atendime
       ,Dbamv.Servico
       ,Dbamv.Convenio
       ,Dbamv.Leito
       ,Dbamv.Unid_Int
 Where atendime.cd_servico = servico.cd_servico(+)
   And Atendime.Cd_Convenio = Convenio.Cd_Convenio
   And Atendime.Cd_Leito = Leito.Cd_Leito(+)
   And Leito.Cd_Unid_Int = Unid_Int.Cd_Unid_Int(+)
   And Atendime.Tp_Atendimento IN ('I','H')
   And Atendime.Cd_Atendimento_Pai Is Null
  AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
   And Atendime.Dt_Atendimento Between :CAMPO_DT_INICIO
                                  And :CAMPO_DT_FINAL + 0.999988425925926
Group By Servico.Cd_Servico
      ,Servico.Ds_Servico
      ,Convenio.Cd_Convenio
      ,Convenio.Nm_Convenio
Order By Cd_Servico
      ,Ds_Servico
      ,Count(*) Desc
      ,Cd_Convenio desc
