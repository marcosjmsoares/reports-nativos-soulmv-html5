--R_INT_ESPEC_ORI
SELECT * FROM (
SELECT TIPO_ALTA
      ,CD_ESPECIALID
      ,DS_ESPECIALID
      ,CD_TIPO_INTERNACAO
      ,DS_TIPO_INTERNACAO
      ,DT_ATENDIMENTO
      ,DT_ALTA
      ,CD_ATENDIMENTO
      ,CD_PACIENTE
      ,NM_PACIENTE
      ,NM_PRESTADOR
 FROM
(
SELECT 'INTERNAÇÕES POR CONVÊNIOS OU PARTICULAR'   TIPO_ALTA
      ,DECODE(TO_CHAR(ESPECIALID.CD_ESPECIALID), NULL, '0', TO_CHAR(ESPECIALID.CD_ESPECIALID))   CD_ESPECIALID
      ,DECODE(ESPECIALID.DS_ESPECIALID, NULL, 'SEM ESPECIALIDADE', ESPECIALID.DS_ESPECIALID)       DS_ESPECIALID
      ,ATENDIME.CD_TIPO_INTERNACAO
      ,TIPO_INTERNACAO.DS_TIPO_INTERNACAO
      ,ATENDIME.DT_ATENDIMENTO
      ,ATENDIME.DT_ALTA
      ,ATENDIME.CD_ATENDIMENTO
      ,ATENDIME.CD_PACIENTE
      ,PACIENTE.NM_PACIENTE
      ,PRESTADOR.NM_PRESTADOR
 FROM  DBAMV.ATENDIME
      ,DBAMV.PACIENTE
      ,DBAMV.PRESTADOR
      ,DBAMV.ESPECIALID
      ,DBAMV.CONVENIO
      ,DBAMV.TIPO_INTERNACAO
 WHERE ATENDIME.CD_PACIENTE   = PACIENTE.CD_PACIENTE
   AND ATENDIME.CD_PRESTADOR  = PRESTADOR.CD_PRESTADOR(+)
   AND ATENDIME.TP_ATENDIMENTO IN ( 'I', 'H')
   AND ATENDIME.CD_ESPECIALID = ESPECIALID.CD_ESPECIALID(+)
   AND ATENDIME.CD_CONVENIO   =  CONVENIO.CD_CONVENIO
   AND TIPO_INTERNACAO.CD_TIPO_INTERNACAO = ATENDIME.CD_TIPO_INTERNACAO
  AND       ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
UNION ALL
SELECT 'INTERNAÇÕES SUS'                                         TIPO_ALTA
      ,DECODE(ESPEC_SUS.CD_ESPEC_SUS, NULL, '0', ESPEC_SUS.CD_ESPEC_SUS)                                       CD_ESPECIALID
      ,DECODE(ESPEC_SUS.DS_ESPEC_SUS, NULL, 'SEM ESPECIALIDADE', ESPEC_SUS.DS_ESPEC_SUS)     DS_ESPECIALID
      ,ATENDIME.CD_TIPO_INTERNACAO
      ,TIPO_INTERNACAO.DS_TIPO_INTERNACAO
      ,ATENDIME.DT_ATENDIMENTO
      ,ATENDIME.DT_ALTA
      ,ATENDIME.CD_ATENDIMENTO
      ,ATENDIME.CD_PACIENTE
      ,PACIENTE.NM_PACIENTE
      ,PRESTADOR.NM_PRESTADOR
 FROM  DBAMV.ATENDIME
      ,DBAMV.PACIENTE
      ,DBAMV.PRESTADOR
      ,DBAMV.CONVENIO
      ,DBAMV.PRO_FAT
      ,DBAMV.ESPEC_SUS
      ,DBAMV.TIPO_INTERNACAO
 WHERE ATENDIME.CD_PACIENTE  = PACIENTE.CD_PACIENTE
   AND ATENDIME.TP_ATENDIMENTO IN ( 'I', 'H')
   AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
   AND ATENDIME.CD_CONVENIO  =  CONVENIO.CD_CONVENIO
   AND ATENDIME.CD_PRO_INT   = PRO_FAT.CD_PRO_FAT(+)
   AND PRO_FAT.CD_ESPEC_SUS  =  ESPEC_SUS.CD_ESPEC_SUS(+)
   AND TIPO_INTERNACAO.CD_TIPO_INTERNACAO = ATENDIME.CD_TIPO_INTERNACAO
   AND       ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
   :CAMPO_CF_MULTI_EMPRESA
    )
   :CAMPO_CF_ORDEM
)
ORDER BY TIPO_ALTA,
CD_ESPECIALID,DS_ESPECIALID,
CD_TIPO_INTERNACAO,DS_TIPO_INTERNACAO, CD_ATENDIMENTO
