--R_LIBERACAO_LEITO
SELECT leito.cd_Unid_int    Cd_Unid_Int
      ,ds_unid_int          Ds_Unid_Int
      , Avg((to_date(to_char(Nvl(dt_lib_mov, SYSDATE),'dd/mm/yyyy')||' '||to_char(Nvl(hr_lib_mov, SYSDATE),'HH24'),'dd/mm/yyyy hh24')
           - to_date(to_char(Nvl(dt_mov_int, SYSDATE),'dd/mm/yyyy')||' '||to_char(Nvl(hr_mov_int, SYSDATE),'HH24'),'dd/mm/yyyy hh24')
            )*24) DIF_HH_Mi
      ,Count(DISTINCT(atendime.cd_atendimento)) Qt_Altas
  FROM dbamv.atendime
     , dbamv.leito
     , dbamv.unid_int
     , dbamv.mov_int
 WHERE Trunc(dt_alta) BETWEEN :CAMPO_DT_INI and 
   AND atendime.cd_leito = leito.cd_leito(+)
   AND leito.cd_unid_int = unid_int.cd_unid_int(+)
   AND atendime.cd_atendimento = mov_int.cd_atendimento (+)
   AND tp_atendimento = 'I'
GROUP BY Leito.cd_Unid_int
       , ds_unid_int
ORDER BY cd_Unid_int
       , ds_unid_int
