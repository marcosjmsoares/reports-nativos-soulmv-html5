--R_EST_OCUP_SINT
SELECT * FROM (
SELECT  CD_UNID_INT
       ,DS_UNID_INT
       ,DS_RESUMO
       ,DS_TIP_ACOM
       ,cd_leito
       ,sum(QT_PAC ) QT_PACIENTES
       ,SUM(QT_DIA ) QT_DIARIAS
from
(
  SELECT  UNID_INT.CD_UNID_INT          CD_UNID_INT
         ,UNID_INT.DS_UNID_INT                  DS_UNID_INT
         ,LEITO.DS_RESUMO                         DS_RESUMO
         ,TIP_ACOM.DS_TIP_ACOM               DS_TIP_ACOM
         ,COUNT(*)                                           QT_PAC
         ,SUM(DECODE(SUBSTR(:CAMPO_DATA_FIM - TRUNC(MOV_INT.DT_LIB_MOV) ,1,1)
          ,'-',:CAMPO_DATA_FIM
          ,nvl(TRUNC(MOV_INT.DT_LIB_MOV), :CAMPO_DATA_FIM))
          -
        DECODE(SUBSTR(TRUNC(MOV_INT.DT_MOV_INT) - :CAMPO_DATA_INI,1,1)
          ,'-',:CAMPO_DATA_INI
          ,MOV_INT.DT_MOV_INT)
    ) QT_DIA
         ,LEITO.cd_leito                cd_leito
  FROM    DBAMV.MOV_INT
         ,DBAMV.TIP_ACOM
         ,DBAMV.LEITO
         ,DBAMV.UNID_INT
         ,DBAMV.CONVENIO
         ,DBAMV.SETOR
 WHERE   TRUNC(mov_int.dt_mov_int)  <  :CAMPO_DATA_INI
 AND     TRUNC(mov_int.dt_lib_mov) BETWEEN :CAMPO_DATA_INI  AND  
 AND      mov_int.CD_TIP_ACOM    = TIP_ACOM.CD_TIP_ACOM
 AND      mov_int.CD_LEITO           = LEITO.CD_LEITO
 AND      LEITO.CD_UNID_INT       = UNID_INT.CD_UNID_INT
 AND      mov_int.CD_CONVENIO   = CONVENIO.CD_CONVENIO
 AND      UNID_INT.CD_SETOR     = SETOR.CD_SETOR
 AND     MOV_INT.TP_MOV IN ('O','I')
 group by UNID_INT.CD_UNID_INT
         ,UNID_INT.DS_UNID_INT
         ,LEITO.DS_RESUMO
         ,TIP_ACOM.DS_TIP_ACOM
        ,LEITO.cd_leito
union all
  SELECT  UNID_INT.CD_UNID_INT          CD_UNID_INT
         ,UNID_INT.DS_UNID_INT                 DS_UNID_INT
         ,LEITO.DS_RESUMO                         DS_RESUMO
         ,TIP_ACOM.DS_TIP_ACOM               DS_TIP_ACOM
         ,COUNT(*)                                          QT_PAC
         ,SUM(DECODE(SUBSTR(:CAMPO_DATA_FIM - TRUNC(MOV_INT.DT_LIB_MOV) ,1,1)
          ,'-',:CAMPO_DATA_FIM
          ,nvl(TRUNC(MOV_INT.DT_LIB_MOV), :CAMPO_DATA_FIM))
          -
        DECODE(SUBSTR(TRUNC(MOV_INT.DT_MOV_INT) - :CAMPO_DATA_INI,1,1)
          ,'-',:CAMPO_DATA_INI
          ,MOV_INT.DT_MOV_INT)
    ) QT_DIA
         ,LEITO.cd_leito                                   cd_leito
  FROM    DBAMV.MOV_INT
         ,DBAMV.TIP_ACOM
         ,DBAMV.LEITO
         ,DBAMV.UNID_INT
         ,DBAMV.CONVENIO
         ,DBAMV.SETOR
 WHERE    TRUNC(mov_int.dt_mov_int)  <  :CAMPO_DATA_INI      AND mov_int.dt_lib_mov IS NULL
 AND      mov_int.CD_TIP_ACOM    = TIP_ACOM.CD_TIP_ACOM
 AND      mov_int.CD_LEITO            = LEITO.CD_LEITO
 AND      LEITO.CD_UNID_INT      = UNID_INT.CD_UNID_INT
 AND      mov_int.CD_CONVENIO   = CONVENIO.CD_CONVENIO
 AND      UNID_INT.CD_SETOR     = SETOR.CD_SETOR
 AND     MOV_INT.TP_MOV IN ('O','I')
 group by UNID_INT.CD_UNID_INT
         ,UNID_INT.DS_UNID_INT
         ,LEITO.DS_RESUMO
         ,TIP_ACOM.DS_TIP_ACOM
        ,LEITO.cd_leito
union all
 SELECT  UNID_INT.CD_UNID_INT          CD_UNID_INT
        ,UNID_INT.DS_UNID_INT                  DS_UNID_INT
        ,LEITO.DS_RESUMO                         DS_RESUMO
        ,TIP_ACOM.DS_TIP_ACOM               DS_TIP_ACOM
        ,COUNT(*)                                           QT_PAC
         ,SUM(DECODE(SUBSTR(:CAMPO_DATA_FIM - TRUNC(MOV_INT.DT_LIB_MOV) ,1,1)
          ,'-',:CAMPO_DATA_FIM
          ,nvl(TRUNC(MOV_INT.DT_LIB_MOV), :CAMPO_DATA_FIM))
          -
        DECODE(SUBSTR(TRUNC(MOV_INT.DT_MOV_INT) - :CAMPO_DATA_INI,1,1)
          ,'-',:CAMPO_DATA_INI
          ,MOV_INT.DT_MOV_INT)
    ) QT_DIA
        ,LEITO.cd_leito                                   cd_leito
 FROM    DBAMV.MOV_INT
        ,DBAMV.TIP_ACOM
        ,DBAMV.LEITO
        ,DBAMV.UNID_INT
        ,DBAMV.CONVENIO
        ,DBAMV.SETOR
 WHERE   TRUNC(mov_int.DT_mov_int)  BETWEEN :CAMPO_DATA_INI  AND 
   and  (TRUNC(mov_int.dt_lib_mov) BETWEEN :CAMPO_DATA_INI  AND    or mov_int.dt_lib_mov IS NULL )
AND      mov_int.CD_TIP_ACOM    = TIP_ACOM.CD_TIP_ACOM
AND      mov_int.CD_LEITO            = LEITO.CD_LEITO
AND      LEITO.CD_UNID_INT       = UNID_INT.CD_UNID_INT
AND      mov_int.CD_CONVENIO    = CONVENIO.CD_CONVENIO
AND      UNID_INT.CD_SETOR     = SETOR.CD_SETOR
AND     MOV_INT.TP_MOV IN ('O','I')
group by UNID_INT.CD_UNID_INT
        ,UNID_INT.DS_UNID_INT
        ,LEITO.DS_RESUMO
        ,TIP_ACOM.DS_TIP_ACOM
        ,LEITO.cd_leito
union all
SELECT  UNID_INT.CD_UNID_INT          CD_UNID_INT
       ,UNID_INT.DS_UNID_INT                 DS_UNID_INT
       ,LEITO.DS_RESUMO                         DS_RESUMO
       ,TIP_ACOM.DS_TIP_ACOM                DS_TIP_ACOM
       ,COUNT(*)                                           QT_PAC
         ,SUM(DECODE(SUBSTR(:CAMPO_DATA_FIM - TRUNC(MOV_INT.DT_LIB_MOV) ,1,1)
          ,'-',:CAMPO_DATA_FIM
          ,nvl(TRUNC(MOV_INT.DT_LIB_MOV), :CAMPO_DATA_FIM))
          -
        DECODE(SUBSTR(TRUNC(MOV_INT.DT_MOV_INT) - :CAMPO_DATA_INI,1,1)
          ,'-',:CAMPO_DATA_INI
          ,MOV_INT.DT_MOV_INT)
    ) QT_DIA
       ,LEITO.cd_leito                                   cd_leito
FROM    DBAMV.MOV_INT
       ,DBAMV.TIP_ACOM
       ,DBAMV.LEITO
       ,DBAMV.UNID_INT
       ,DBAMV.CONVENIO
       ,DBAMV.SETOR
WHERE  TRUNC(mov_int.DT_mov_int)  BETWEEN :CAMPO_DATA_INI  AND 
and      mov_int.dt_lib_mov > :CAMPO_DATA_FIM
AND    mov_int.CD_TIP_ACOM    = TIP_ACOM.CD_TIP_ACOM
AND    mov_int.CD_LEITO       = LEITO.CD_LEITO
AND    LEITO.CD_UNID_INT      = UNID_INT.CD_UNID_INT
AND    mov_int.CD_CONVENIO    = CONVENIO.CD_CONVENIO
AND    UNID_INT.CD_SETOR = SETOR.CD_SETOR
AND   MOV_INT.TP_MOV IN ('O','I')
group by UNID_INT.CD_UNID_INT
        ,UNID_INT.DS_UNID_INT
        ,LEITO.DS_RESUMO
        ,TIP_ACOM.DS_TIP_ACOM
        ,LEITO.cd_leito
  union all
  SELECT  UNID_INT.CD_UNID_INT         CD_UNID_INT
         ,UNID_INT.DS_UNID_INT         DS_UNID_INT
         ,LEITO.DS_RESUMO              DS_RESUMO
         ,TIP_ACOM.DS_TIP_ACOM         DS_TIP_ACOM
         ,COUNT(*)                     QT_PAC
         ,SUM(DECODE(SUBSTR(:CAMPO_DATA_FIM - TRUNC(MOV_INT.DT_LIB_MOV) ,1,1)
          ,'-',:CAMPO_DATA_FIM
          ,nvl(TRUNC(MOV_INT.DT_LIB_MOV), :CAMPO_DATA_FIM))
          -
        DECODE(SUBSTR(TRUNC(MOV_INT.DT_MOV_INT) - :CAMPO_DATA_INI,1,1)
          ,'-',:CAMPO_DATA_INI
          ,MOV_INT.DT_MOV_INT)
    ) QT_DIA
         ,LEITO.cd_leito               cd_leito
  FROM    DBAMV.MOV_INT
         ,DBAMV.TIP_ACOM
         ,DBAMV.LEITO
         ,DBAMV.UNID_INT
         ,DBAMV.CONVENIO
         ,DBAMV.SETOR
 WHERE  TRUNC(mov_int.dt_mov_int)  <  :CAMPO_DATA_INI
   and trunc(mov_int.dt_lib_mov) > :CAMPO_DATA_FIM
 AND      mov_int.CD_TIP_ACOM   = TIP_ACOM.CD_TIP_ACOM
 AND      mov_int.CD_LEITO      = LEITO.CD_LEITO
 AND      LEITO.CD_UNID_INT     = UNID_INT.CD_UNID_INT
 AND      mov_int.CD_CONVENIO   = CONVENIO.CD_CONVENIO
 AND      UNID_INT.CD_SETOR     = SETOR.CD_SETOR
 AND      MOV_INT.TP_MOV IN ('O','I')
 group by UNID_INT.CD_UNID_INT
         ,UNID_INT.DS_UNID_INT
         ,LEITO.DS_RESUMO
         ,TIP_ACOM.DS_TIP_ACOM
        ,LEITO.cd_leito
)
group by CD_UNID_INT
        ,DS_UNID_INT
        ,DS_RESUMO
        ,DS_TIP_ACOM
        ,cd_leito
)
ORDER BY DS_UNID_INT,CD_UNID_INT
