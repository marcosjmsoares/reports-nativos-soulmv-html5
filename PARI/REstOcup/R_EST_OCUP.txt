--R_EST_OCUP
SELECT * FROM (
select leito.cd_unid_int cd_unid_int
         ,unid_int.ds_unid_int ds_unid_int
         ,'Diárias' tipo
         ,'1' ordem
         ,sum ( least ( :CAMPO_DATAFIM, atendime.dt_alta ) -
          greatest ( :CAMPO_DATAINI, atendime.dt_atendimento ) ) dias
from   dbamv.atendime atendime
        ,dbamv.leito leito
        ,dbamv.unid_int unid_int
where  atendime.dt_atendimento between :CAMPO_DATAINI and  
and    atendime.tp_atendimento = 'I'
and    atendime.cd_leito = leito.cd_leito
and    leito.cd_unid_int = unid_int.cd_unid_int
and    atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
group by leito.cd_unid_int
        ,unid_int.ds_unid_int
union
select leito.cd_unid_int cd_unid_int
      ,unid_int.ds_unid_int ds_unid_int
      ,'Diárias' tipo
      ,'1' ordem
      ,sum ( least ( :CAMPO_DATAFIM, atendime.dt_alta ) -
       greatest ( :CAMPO_DATAINI, atendime.dt_atendimento ) ) dias
from   dbamv.atendime atendime
      ,dbamv.leito leito
      ,dbamv.unid_int unid_int
where  atendime.dt_alta between :CAMPO_DATAINI and
and    atendime.tp_atendimento = 'I'
and    atendime.cd_leito = leito.cd_leito
and    leito.cd_unid_int = unid_int.cd_unid_int
And atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
group by leito.cd_unid_int
        ,unid_int.ds_unid_int
union
select leito.cd_unid_int cd_unid_int
      ,unid_int.ds_unid_int ds_unid_int
      ,'Diárias' tipo
      ,'1' ordem
      ,sum ( :CAMPO_DATAFIM - greatest ( :CAMPO_DATAINI, atendime.dt_atendimento ) ) dias
from   dbamv.atendime atendime
      ,dbamv.leito leito
      ,dbamv.unid_int unid_int
where  atendime.dt_alta IS NULL
and    atendime.tp_atendimento = 'I'
and    atendime.cd_leito = leito.cd_leito
and    leito.cd_unid_int = unid_int.cd_unid_int
And atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
group by leito.cd_unid_int
        ,unid_int.ds_unid_int
union
select leito.cd_unid_int cd_unid_int
      ,unid_int.ds_unid_int ds_unid_int
      ,'Pacientes' tipo
      ,'2' ordem
      ,count ( atendime.cd_paciente ) dias
from   dbamv.atendime atendime
      ,dbamv.leito leito
      ,dbamv.unid_int unid_int
where  atendime.dt_atendimento between :CAMPO_DATAINI and
and    atendime.tp_atendimento = 'I'
and    atendime.cd_leito = leito.cd_leito
and    leito.cd_unid_int = unid_int.cd_unid_int
And atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
group by leito.cd_unid_int
        ,unid_int.ds_unid_int
union
select leito.cd_unid_int cd_unid_int
      ,unid_int.ds_unid_int ds_unid_int
      ,'Pacientes' tipo
      ,'2' ordem
      ,count ( atendime.cd_paciente ) dias
from   dbamv.atendime atendime
      ,dbamv.leito leito
      ,dbamv.unid_int
where  atendime.dt_alta between :CAMPO_DATAINI and
and    atendime.tp_atendimento = 'I'
and    atendime.cd_leito = leito.cd_leito
and    leito.cd_unid_int = unid_int.cd_unid_int
And atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
group by leito.cd_unid_int
        ,unid_int.ds_unid_int
union
select leito.cd_unid_int cd_unid_int
      ,unid_int.ds_unid_int ds_unid_int
      ,'Pacientes' tipo
      ,'2' ordem
      ,count ( atendime.cd_paciente ) dias
from   dbamv.atendime atendime
      ,dbamv.leito leito
      ,dbamv.unid_int unid_int
where  atendime.dt_alta IS NULL
and    atendime.tp_atendimento = 'I'
and    atendime.cd_leito = leito.cd_leito
and    leito.cd_unid_int = unid_int.cd_unid_int
And atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
group by leito.cd_unid_int
        ,unid_int.ds_unid_int
union
SELECT  leito.cd_unid_int cd_unid_int
       ,unid_int.ds_unid_int
       ,'Internados' tipo
       ,'3' ordem
       ,count ( atendime.cd_atendimento ) dias
FROM    dbamv.atendime atendime
       ,dbamv.leito leito
       ,dbamv.unid_int unid_int
where  atendime.dt_atendimento between :CAMPO_DATAINI and
and    atendime.tp_atendimento = 'I'
and    atendime.cd_leito = leito.cd_leito
and    leito.cd_unid_int = unid_int.cd_unid_int
And atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
group by leito.cd_unid_int
        ,unid_int.ds_unid_int
union
SELECT  leito.cd_unid_int cd_unid_int
       ,unid_int.ds_unid_int ds_unid_int
       ,'Óbitos' tipo
       ,'4' ordem
       ,count ( atendime.cd_atendimento ) dias
FROM    dbamv.atendime atendime
       ,dbamv.leito leito
       ,dbamv.mot_alt mot_alt
       ,dbamv.unid_int unid_int
where  atendime.dt_alta between :CAMPO_DATAINI and
and    atendime.tp_atendimento = 'I'
and    atendime.cd_leito = leito.cd_leito
and    leito.cd_unid_int = unid_int.cd_unid_int
and    atendime.cd_mot_alt = mot_alt.cd_mot_alt
And atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and    mot_alt.tp_mot_alta = 'O'
group by leito.cd_unid_int
        ,unid_int.ds_unid_int
union
SELECT  leito.cd_unid_int cd_unid_int
       ,unid_int.ds_unid_int ds_unid_int
       ,'Altas' tipo
       ,'5' ordem
       ,count ( atendime.cd_atendimento ) dias
FROM    dbamv.atendime atendime
       ,dbamv.leito leito
       ,dbamv.unid_int unid_int
       ,dbamv.mot_alt mot_alt
where  atendime.dt_alta between :CAMPO_DATAINI and
and    atendime.tp_atendimento = 'I'
and    atendime.cd_leito = leito.cd_leito
and    leito.cd_unid_int = unid_int.cd_unid_int
and    atendime.cd_mot_alt = mot_alt.cd_mot_alt
And atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
and    mot_alt.tp_mot_alta <> 'O'
group by leito.cd_unid_int
        ,unid_int.ds_unid_int
)
ORDER BY CD_UNID_INT,DS_UNID_INT,
ORDEM,  TIPO
