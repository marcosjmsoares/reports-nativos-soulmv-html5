--R_CENSO_SINT
SELECT  LEITO.CD_LEITO                    CD_LEITO,
                LEITO.CD_UNID_INT             CD_UNID_INT,
                ATENDIME.CD_ATENDIMENTO    CD_ATENDIMENTO,
                LEITO.DS_LEITO                                      DS_LEITO,
                LEITO.TP_OCUPACAO                            TP_OCUPACAO,
                LEITO.TP_SITUACAO                              TP_SITUACAO,
                LEITO.DS_ENFERMARIA                         DS_ENFERMARIA,
                UNID_INT.DS_UNID_INT                         DS_UNID_INT,
                ATENDIME.CD_PACIENTE                       CD_PACIENTE,
                PACIENTE.NM_PACIENTE                       NM_PACIENTE,
                CONVENIO.CD_CONVENIO                      CD_CONVENIO,
                CONVENIO.NM_CONVENIO                     NM_CONVENIO,
                PRESTADOR.CD_PRESTADOR               CD_PRESTADOR,
                PRESTADOR.NM_MNEMONICO              NM_MNEMONICO
FROM      DBAMV.V_INTERNADOS                          V_INTERNADOS,
                DBAMV.LEITO                                           LEITO,
                DBAMV.UNID_INT                                    UNID_INT,
                DBAMV.ATENDIME                                   ATENDIME,
                DBAMV.PACIENTE                                    PACIENTE,
                DBAMV.CONVENIO                                   CONVENIO,
                DBAMV.PRESTADOR                               PRESTADOR
WHERE  V_INTERNADOS.CD_LEITO  (+) =  LEITO.CD_LEITO
AND    LEITO.CD_UNID_INT =  UNID_INT.CD_UNID_INT
AND    V_INTERNADOS.CD_ATENDIMENTO =  ATENDIME.CD_ATENDIMENTO (+)
AND    PACIENTE.CD_PACIENTE (+) =  ATENDIME.CD_PACIENTE
AND    CONVENIO.CD_CONVENIO (+) =  ATENDIME.CD_CONVENIO
AND    PRESTADOR.CD_PRESTADOR (+) =  ATENDIME.CD_PRESTADOR
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
AND    UNID_INT.CD_UNID_INT  LIKE  :CAMPO_UNID_INT
AND   ( LEITO.TP_SITUACAO = DECODE( :CAMPO_LEITO, 'S', 'I', 'N', '1' )
OR     LEITO.TP_SITUACAO <> 'I' )
ORDER BY UNID_INT.DS_UNID_INT, LEITO.DS_ENFERMARIA, LEITO.DS_LEITO
