--R_OBITOS_FAIXA_ETARIA
SELECT * FROM (
SELECT FAIXA_ETARIA.CD_FAIXA_ETARIA_RELATORIO,
               FAIXA_ETARIA.DS_FAIXA_ETARIA_RELATORIO   idade,
               ATENDIME.CD_CONVENIO,
               CONVENIO.NM_CONVENIO,
               UNID_INT.DS_UNID_INT,
               ESPEC_SUS.DS_ESPEC_SUS,
               COUNT(*) QT_OBITOS
FROM     DBAMV.ATENDIME,
               DBAMV.PACIENTE,
               DBAMV.CONVENIO,
               DBAMV.PRO_FAT,
               DBAMV.LEITO,
               DBAMV.UNID_INT,
               DBAMV.FAIXA_ETARIA_RELATORIO   FAIXA_ETARIA,
               DBAMV.ESPEC_SUS,
               DBAMV.MOT_ALT
WHERE  TRUNC(ATENDIME.DT_ALTA) BETWEEN :CAMPO_DT_INIC AND 
AND        ATENDIME.TP_ATENDIMENTO IN ('I', 'H')
AND        ATENDIME.CD_PACIENTE   = PACIENTE.CD_PACIENTE
AND        ATENDIME.CD_CONVENIO  = CONVENIO.CD_CONVENIO
AND        ATENDIME.CD_PRO_INT     = PRO_FAT.CD_PRO_FAT(+)
AND        PRO_FAT.CD_ESPEC_SUS   = ESPEC_SUS.CD_ESPEC_SUS(+)
AND        ATENDIME.CD_LEITO          =  LEITO.CD_LEITO
AND        LEITO.CD_UNID_INT            =  UNID_INT.CD_UNID_INT
AND        DBAMV.RETORNA_FAIXA_ETARIA_RELATORIO(PACIENTE.DT_NASCIMENTO, 'C') = FAIXA_ETARIA.CD_FAIXA_ETARIA_RELATORIO
AND        ATENDIME.CD_MOT_ALT  =   MOT_ALT.CD_MOT_ALT
AND        MOT_ALT.TP_MOT_ALTA = 'O'
AND         ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY  FAIXA_ETARIA.CD_FAIXA_ETARIA_RELATORIO,
                     FAIXA_ETARIA.DS_FAIXA_ETARIA_RELATORIO,
                     ATENDIME.CD_CONVENIO,
                     CONVENIO.NM_CONVENIO,
                     UNID_INT.DS_UNID_INT,
                     ESPEC_SUS.DS_ESPEC_SUS
ORDER BY CD_FAIXA_ETARIA_RELATORIO
)
ORDER BY DS_UNID_INT,
DS_ESPEC_SUS,
CD_CONVENIO,NM_CONVENIO,
CD_FAIXA_ETARIA_RELATORIO,IDADE
