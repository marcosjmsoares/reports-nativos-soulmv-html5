--R_EXCLUSAO_ATENDIMENTO
SELECT * FROM (
SELECT DISTINCT trunc(log_excl_atendime.dt_log_excl_atendime)  dt_exclusao,
              log_excl_atendime.nm_usuario                             nm_usuario_exclusao
FROM     dbamv.log_excl_atendime
WHERE   trunc(log_excl_atendime.dt_log_excl_atendime) BETWEEN :CAMPO_DTINIC AND 
:CAMPO_COND_USUARIO
)
ORDER BY DT_EXCLUSAO,
NM_USUARIO_EXCLUSAO
