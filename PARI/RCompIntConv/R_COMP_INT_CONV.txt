--R_COMP_INT_CONV
SELECT A.CD_CONVENIO, A.NM_CONVENIO, SUM(A.QTD_INTER_ANO1) QTD_INT1, SUM(A.QTD_INTER_ANO2) QTD_INT2
FROM
(SELECT CONV.CD_CONVENIO CD_CONVENIO
      , CONV.NM_CONVENIO NM_CONVENIO
      , COUNT(*)         QTD_INTER_ANO1
      , 0                QTD_INTER_ANO2
  FROM DBAMV.ATENDIME   ATEN,
       DBAMV.CONVENIO   CONV
  WHERE ATEN.DT_ATENDIMENTO  BETWEEN  :CAMPO_DT_INICIAL1  AND   
        AND ATEN.TP_ATENDIMENTO IN ( 'I', 'H' )
        AND ATEN.CD_ATENDIMENTO_PAI IS NULL
        AND ATEN.CD_CONVENIO  =  CONV.CD_CONVENIO
        AND ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
  GROUP BY CONV.CD_CONVENIO,
           CONV.NM_CONVENIO
UNION ALL
  SELECT CONV.CD_CONVENIO CD_CONVENIO
       , CONV.NM_CONVENIO NM_CONVENIO
       , 0                QTD_INTER_ANO1
       , COUNT(*)         QTD_INTER_ANO2
  FROM DBAMV.ATENDIME   ATEN,
       DBAMV.CONVENIO   CONV
  WHERE ATEN.DT_ATENDIMENTO  BETWEEN  :CAMPO_DT_INICIAL2  AND   
        AND ATEN.TP_ATENDIMENTO IN ( 'I', 'H' )
        AND ATEN.CD_ATENDIMENTO_PAI IS NULL
        AND ATEN.CD_CONVENIO  =  CONV.CD_CONVENIO
       AND ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
  GROUP BY CONV.CD_CONVENIO,
           CONV.NM_CONVENIO) A
GROUP BY A.CD_CONVENIO,
         A.NM_CONVENIO
ORDER BY A.NM_CONVENIO
