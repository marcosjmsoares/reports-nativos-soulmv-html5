--R_OPE_INT_CID
SELECT * FROM
			  (SELECT CIDADE.NM_CIDADE ,
			          PACIENTE.CD_PACIENTE ,
			          PACIENTE.NM_PACIENTE ,
			      (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
			          ATENDIME.DT_ATENDIMENTO ,
			          TO_CHAR(ATENDIME.HR_ATENDIMENTO, 'HH24') HR_ATENDIMENTO ,
			          ATENDIME.DT_ALTA ,
			          TO_CHAR(ATENDIME.HR_ALTA, 'HH24') HR_ALTA ,
			          PRESTADOR.NM_PRESTADOR
			   FROM DBAMV.ATENDIME ATENDIME ,
			        DBAMV.PACIENTE PACIENTE ,
			        DBAMV.PRESTADOR PRESTADOR ,
			        DBAMV.CIDADE ,
			        DBAMV.LEITO ,
			        DBAMV.UNID_INT ,
			        DBAMV.CONVENIO
			   WHERE ATENDIME.TP_ATENDIMENTO IN ('I',
			                                     'H')
			     AND ATENDIME.DT_ATENDIMENTO BETWEEN :CAMPO_DT_INIC AND 
			     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE(+)
			     AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
			     AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
			     AND PACIENTE.CD_CIDADE = CIDADE.CD_CIDADE(+)
			     AND ATENDIME.CD_LEITO = LEITO.CD_LEITO
			     AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT
			     AND ATENDIME.CD_ATENDIMENTO_PAI IS NULL
			     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA      
			   ORDER BY ATENDIME.DT_ATENDIMENTO ,
			            HR_ATENDIMENTO ASC)
			ORDER BY NM_CIDADE,
			         DT_ATENDIMENTO,
			         HR_ATENDIMENTO ASC
