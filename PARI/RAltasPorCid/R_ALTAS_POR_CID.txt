--R_ALTAS_POR_CID
SELECT * FROM (
			SELECT ATENDIME.CD_ATENDIMENTO					CD_ATENDIMENTO
			              ,ATENDIME.TP_ATENDIMENTO    		TP_ATENDIMENTO
			              ,ATENDIME.DT_ATENDIMENTO    		DT_ATENDIMENTO
			              ,ATENDIME.DT_ALTA           		DT_ALTA
			              ,ATENDIME.CD_PACIENTE				CD_PACIENTE
			              ,PACIENTE.NM_PACIENTE				NM_PACIENTE
			              ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  CD_PACIENTE) NM_SOCIAL_PACIENTE
			              ,ATENDIME.CD_CID            		CD_CID
			              ,NVL(CID.DS_CID, 'CID NAO INFORMADO')    DS_CID
			             ,PACIENTE.TP_SEXO                         TP_SEXO
			             ,DBAMV.PACK_INTERNAMENTO.RETORNA_IDADE(PACIENTE.DT_NASCIMENTO, sysdate)      IDADE_PAC
			             ,CONVENIO.NM_CONVENIO				NM_CONVENIO
			             ,ESPECIALID.DS_ESPECIALID			DS_ESPECIALID
			             ,DECODE(MOT_ALT.TP_MOT_ALTA, 'A', 'ALTA', 'D', 'ALTA ADMINISTRATIVA', 'O', 'OBITO', 'T', 'TRANSFERENCIA', 'INDEFINIDO')   TP_ALTA
			FROM     DBAMV.ATENDIME
			               ,DBAMV.PACIENTE
			               ,DBAMV.CID
			               ,DBAMV.CONVENIO
			              , DBAMV.MOT_ALT
			              , DBAMV.ESPECIALID
			WHERE   TRUNC(ATENDIME.DT_ALTA)  BETWEEN :CAMPO_DT_INICIO AND 
			AND     DT_ALTA IS NOT NULL
			AND     ATENDIME.CD_PACIENTE   =  PACIENTE.CD_PACIENTE
			AND     ATENDIME.CD_CID               =  CID.CD_CID(+)
			AND     ATENDIME.CD_CONVENIO  =  CONVENIO.CD_CONVENIO
			AND     ATENDIME.CD_ESPECIALID = ESPECIALID.CD_ESPECIALID(+)
			AND     ATENDIME.CD_MOT_ALT  =  MOT_ALT.CD_MOT_ALT(+)
			AND     ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
			:CAMPO_CF_ORDEM_PD
			)
			ORDER BY CD_CID,DS_CID
