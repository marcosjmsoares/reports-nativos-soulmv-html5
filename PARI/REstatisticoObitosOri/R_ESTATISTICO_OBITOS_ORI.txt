--R_ESTATISTICO_OBITOS_ORI
SELECT     CODIGO
      ,NVL(:CAMPO_DS_COLUNA,'NÃO ESPECIFICADO')   DESCRICAO
      ,COUNT(*)        QT_ALTAS
FROM DBAMV.ATENDIME
          ,DBAMV.MOT_ALT
          ,DBAMV.CONVENIO
          ,DBAMV.ESPECIALID
          ,DBAMV.PRESTADOR
          ,DBAMV.CID
          ,DBAMV.LEITO
          ,DBAMV.UNID_INT
WHERE ATENDIME.TP_ATENDIMENTO = 'I'
  AND ATENDIME.DT_ALTA        IS NOT NULL
  AND TRUNC(ATENDIME.DT_ALTA) BETWEEN :CAMPO_DT_INI AND 
  AND ATENDIME.CD_MOT_ALT    = MOT_ALT.CD_MOT_ALT(+)
  AND MOT_ALT.TP_MOT_ALTA    = 'O'
  AND ATENDIME.CD_CONVENIO   = CONVENIO.CD_CONVENIO
  AND ATENDIME.CD_ESPECIALID = ESPECIALID.CD_ESPECIALID(+)
  AND ATENDIME.CD_PRESTADOR  = PRESTADOR.CD_PRESTADOR(+)
  AND ATENDIME.CD_CID        = CID.CD_CID(+)
  AND ATENDIME.CD_LEITO      = LEITO.CD_LEITO(+)
  AND LEITO.CD_UNID_INT      = UNID_INT.CD_UNID_INT(+)
GROUP BY  :CAMPO_CD_COLUNA
         ,:CAMPO_DS_COLUNA
ORDER BY  QT_ALTAS  DESC
