--R_EST_ATEN_MED_CONV_G_PRESTADOR1_Q_6
SELECT * FROM
		  (SELECT PRESTADOR.CD_PRESTADOR CD_PRESTADOR,
		          PRESTADOR.NM_PRESTADOR NM_PRESTADOR,
		          CONVENIO.CD_CONVENIO CD_CONVENIO,
		          CONVENIO.NM_CONVENIO NM_CONVENIO,
		          PACIENTE.CD_PACIENTE CD_PACIENTE,
		          PACIENTE.NM_PACIENTE NM_PACIENTE,
		     (SELECT P.NM_SOCIAL_PACIENTE
		      FROM DBAMV.PACIENTE P
		      WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S'
		        AND PACIENTE.CD_PACIENTE = CD_PACIENTE) NM_SOCIAL_PACIENTE, 
		   ATENDIME.DT_ATENDIMENTO DT_ATENDIMENTO,
		   TO_CHAR(ATENDIME.HR_ATENDIMENTO, 'HH24') HR_ATENDIMENTO,
		   1 QTDE_ATENDIMENTO
		   FROM DBAMV.ATENDIME,
		        DBAMV.PACIENTE,
		        DBAMV.PRESTADOR,
		        DBAMV.CONVENIO,
		        DBAMV.ORI_ATE 
		   WHERE TRUNC(ATENDIME.DT_ATENDIMENTO) BETWEEN :CAMPO_DT_INICIO AND 
		     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
		     AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
		     AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
		     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
		     AND ATENDIME.CD_ORI_ATE = ORI_ATE.CD_ORI_ATE
		     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA      
		   GROUP BY PRESTADOR.CD_PRESTADOR,
		            PRESTADOR.NM_PRESTADOR,
		            CONVENIO.CD_CONVENIO,
		            CONVENIO.NM_CONVENIO,
		            PACIENTE.CD_PACIENTE,
		            PACIENTE.NM_PACIENTE,
		            ATENDIME.DT_ATENDIMENTO,
		            ATENDIME.HR_ATENDIMENTO
		   ORDER BY NM_PRESTADOR)
		ORDER BY NM_PRESTADOR,
		         NM_CONVENIO,
		         HR_ATENDIMENTO

--R_EST_ATEN_MED_CONV_R_9_Q_1
SELECT * FROM
		  (SELECT turno ,
		          CD_TURNO ,
		          CD_PRESTADOR ,
		          NM_PRESTADOR ,
		          CD_CONVENIO ,
		          NM_CONVENIO ,
		          CD_PACIENTE ,
		          NM_PACIENTE ,
		          DT_ATENDIMENTO ,
		          HR_ATENDIMENTO ,
		          QTDE_ATENDIMENTO
		   FROM
		     (SELECT 'Manha' turno ,
		                     '1' CD_TURNO ,
		                         PRESTADOR.CD_PRESTADOR CD_PRESTADOR ,
		                         PRESTADOR.NM_PRESTADOR NM_PRESTADOR ,
		                         CONVENIO.CD_CONVENIO CD_CONVENIO ,
		                         CONVENIO.NM_CONVENIO NM_CONVENIO ,
		                         PACIENTE.CD_PACIENTE CD_PACIENTE ,
		                         PACIENTE.NM_PACIENTE NM_PACIENTE ,
		                         ATENDIME.DT_ATENDIMENTO DT_ATENDIMENTO ,
		                         TO_CHAR(ATENDIME.HR_ATENDIMENTO, 'HH24') HR_ATENDIMENTO ,
		                         1 QTDE_ATENDIMENTO
		      FROM DBAMV.ATENDIME ,
		           DBAMV.PACIENTE ,
		           DBAMV.PRESTADOR ,
		           DBAMV.CONVENIO ,
		           DBAMV.ORI_ATE
		      WHERE TRUNC(ATENDIME.DT_ATENDIMENTO) BETWEEN :CAMPO_DT_INICIO AND 
		        AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
		        AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
		        AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
		        AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
		        AND ATENDIME.CD_ORI_ATE = ORI_ATE.CD_ORI_ATE
		        AND TO_CHAR(ATENDIME.HR_ATENDIMENTO, 'HH24') BETWEEN :CAMPO_HR_MANHA_INI AND 
		        AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA      
		      GROUP BY PRESTADOR.CD_PRESTADOR ,
		               PRESTADOR.NM_PRESTADOR ,
		               CONVENIO.CD_CONVENIO ,
		               CONVENIO.NM_CONVENIO ,
		               PACIENTE.CD_PACIENTE ,
		               PACIENTE.NM_PACIENTE ,
		               ATENDIME.DT_ATENDIMENTO ,
		               ATENDIME.HR_ATENDIMENTO
		      UNION SELECT 'Tarde' turno ,
		                           '2' CD_TURNO ,
		                               PRESTADOR.CD_PRESTADOR CD_PRESTADOR ,
		                               PRESTADOR.NM_PRESTADOR NM_PRESTADOR ,
		                               CONVENIO.CD_CONVENIO CD_CONVENIO ,
		                               CONVENIO.NM_CONVENIO NM_CONVENIO ,
		                               PACIENTE.CD_PACIENTE CD_PACIENTE ,
		                               PACIENTE.NM_PACIENTE NM_PACIENTE ,
		                               ATENDIME.DT_ATENDIMENTO DT_ATENDIMENTO ,
		                               TO_CHAR(ATENDIME.HR_ATENDIMENTO, 'HH24') HR_ATENDIMENTO ,
		                               1 QTDE_ATENDIMENTO
		      FROM DBAMV.ATENDIME ,
		           DBAMV.PACIENTE ,
		           DBAMV.PRESTADOR ,
		           DBAMV.CONVENIO ,
		           DBAMV.ORI_ATE
		      WHERE TRUNC(ATENDIME.DT_ATENDIMENTO) BETWEEN :CAMPO_DT_INICIO AND 
		        AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
		        AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
		        AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
		        AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
		        AND ATENDIME.CD_ORI_ATE = ORI_ATE.CD_ORI_ATE
		        AND TO_CHAR(ATENDIME.HR_ATENDIMENTO, 'HH24') BETWEEN :CAMPO_HR_TARDE_INI AND 
		        AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA      
		      GROUP BY PRESTADOR.CD_PRESTADOR ,
		               PRESTADOR.NM_PRESTADOR ,
		               CONVENIO.CD_CONVENIO ,
		               CONVENIO.NM_CONVENIO ,
		               PACIENTE.CD_PACIENTE ,
		               PACIENTE.NM_PACIENTE ,
		               ATENDIME.DT_ATENDIMENTO ,
		               ATENDIME.HR_ATENDIMENTO
		      UNION SELECT 'Noite' turno ,
		                           '3' CD_TURNO ,
		                               PRESTADOR.CD_PRESTADOR CD_PRESTADOR ,
		                               PRESTADOR.NM_PRESTADOR NM_PRESTADOR ,
		                               CONVENIO.CD_CONVENIO CD_CONVENIO ,
		                               CONVENIO.NM_CONVENIO NM_CONVENIO ,
		                               PACIENTE.CD_PACIENTE CD_PACIENTE ,
		                               PACIENTE.NM_PACIENTE NM_PACIENTE ,
		                               ATENDIME.DT_ATENDIMENTO DT_ATENDIMENTO ,
		                               TO_CHAR(ATENDIME.HR_ATENDIMENTO, 'HH24') HR_ATENDIMENTO ,
		                               1 QTDE_ATENDIMENTO
		      FROM DBAMV.ATENDIME ,
		           DBAMV.PACIENTE ,
		           DBAMV.PRESTADOR ,
		           DBAMV.CONVENIO ,
		           DBAMV.ORI_ATE
		      WHERE TRUNC(ATENDIME.DT_ATENDIMENTO) BETWEEN :CAMPO_DT_INICIO AND 
		        AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
		        AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
		        AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
		        AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
		        AND ATENDIME.CD_ORI_ATE = ORI_ATE.CD_ORI_ATE
		        AND TO_CHAR(ATENDIME.HR_ATENDIMENTO, 'HH24') BETWEEN :CAMPO_HR_NOITE_INI AND 
		        AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA      
		      GROUP BY PRESTADOR.CD_PRESTADOR ,
		               PRESTADOR.NM_PRESTADOR ,
		               CONVENIO.CD_CONVENIO ,
		               CONVENIO.NM_CONVENIO ,
		               PACIENTE.CD_PACIENTE ,
		               PACIENTE.NM_PACIENTE ,
		               ATENDIME.DT_ATENDIMENTO ,
		               ATENDIME.HR_ATENDIMENTO)Turno
		   GROUP BY turno ,
		            CD_TURNO ,
		            CD_PRESTADOR ,
		            NM_PRESTADOR ,
		            CD_CONVENIO ,
		            NM_CONVENIO ,
		            CD_PACIENTE ,
		            NM_PACIENTE ,
		            DT_ATENDIMENTO ,
		            HR_ATENDIMENTO ,
		            QTDE_ATENDIMENTO
		   ORDER BY CD_TURNO,
		            DT_ATENDIMENTO DESC, HR_ATENDIMENTO DESC)
		ORDER BY CD_TURNO,
		         TURNO
