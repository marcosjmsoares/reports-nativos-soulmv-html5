--R_AVISO_ALTA_MEDICA
SELECT ATENDIME.CD_ATENDIMENTO                        CD_ATENDIMENTO
             ,UNID_INT.CD_UNID_INT                              CD_UNID_INT
             ,UNID_INT.DS_UNID_INT                              DS_UNID_INT
             ,PACIENTE.CD_PACIENTE                              CD_PACIENTE
             ,PACIENTE.NM_PACIENTE                              NM_PACIENTE
             ,CONVENIO.CD_CONVENIO                              CD_CONVENIO
             ,CONVENIO.NM_CONVENIO                              NM_CONVENIO
             ,PRESTADOR.CD_PRESTADOR                            CD_PRESTADOR
             ,PRESTADOR.NM_PRESTADOR                            NM_PRESTADOR
             ,ATENDIME.DT_ATENDIMENTO                           DT_ATENDIMENTO
             ,to_char(ATENDIME.HR_ATENDIMENTO,'HH24')        HR_ATENDIMENTO
             ,ATENDIME.NM_USUARIO                               NM_USUARIO
             ,ATENDIME.DT_ALTA_MEDICA                           DT_ALTA
             ,to_char(ATENDIME.HR_ALTA_MEDICA,'HH24')        HR_ALTA
             ,ATENDIME.NM_USUARIO_ALTA_MEDICA       NM_USUARIO
             ,MOT_ALT.TP_MOT_ALTA                                  TP_MOT_ALTA
             ,MOT_ALT.CD_MOT_ALT                                    CD_MOT_ALT
             ,MOT_ALT.DS_MOT_ALT                                    DS_MOT_ALT
             ,LEITO.DS_RESUMO                                           DS_LEITO
             ,LEITO.CD_LEITO                                                CD_LEITO
FROM DBAMV.ATENDIME
          ,DBAMV.LEITO
          ,DBAMV.UNID_INT
          ,DBAMV.PACIENTE
          ,DBAMV.CONVENIO
          ,DBAMV.PRESTADOR
          ,DBAMV.MOT_ALT
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
And ATENDIME.CD_ATENDIMENTO = :CAMPO_PCD_ATENDIMENTO
AND   ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
AND   LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)
AND   ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND   ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND   ATENDIME.CD_MOT_ALT = MOT_ALT.CD_MOT_ALT(+)
AND   ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
