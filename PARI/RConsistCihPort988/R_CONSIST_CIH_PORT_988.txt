Select Atendime.Cd_Pro_Int Cd_Pro_Int2,
           Atendime.Cd_Pro_Int_Procedimento_Entrad Cd_Pro_Int,
           To_Char(Atendime.Dt_Atendimento,'DD/MM/YYYY') Dt_Atende,
           To_Char(Atendime.Dt_Alta,'DD/MM/YYYY') Dt_Alta,
           Atendime.Cd_Cid,
           Atendime.Dt_Atendimento,
           Atendime.Cd_Mot_Alt,
           Atendime.Cd_Atendimento,
           Atendime.Cd_Ssm_Cih,
           Atendime.Nr_Declaracao_Obito,
           Paciente.Cd_Paciente,
           Paciente.Nm_Paciente,
           Paciente.Ds_Endereco,
           Paciente.Ds_Complemento,
           Paciente.Nr_Endereco,
           Paciente.Nr_Cep,
           Paciente.Dt_Nascimento,
           Paciente.Tp_Sexo,
           Paciente.Nr_Cns,
           Paciente.Cd_Cidade,
           Cidade.Cd_Ibge,
           Cidade.Cd_Uf,
           Cidade.Nm_Cidade,
           Convenio.Tp_Convenio,
           Convenio.Sn_Filantropia,
           Convenio.Tp_Fonte_Remuneracao,
           Convenio.Nr_Registro_Operadora_Ans,
           Convenio.Nr_Cgc,
           Carteira.Nr_Carteira
      From Dbamv.Atendime,
           Dbamv.Paciente,
           Dbamv.Convenio,
           Dbamv.Unid_Int,
           Dbamv.Leito,
           Dbamv.Carteira,
           Dbamv.Cidade
     Where Paciente.Cd_Paciente        = Atendime.Cd_Paciente
       And Atendime.Tp_Atendimento     = 'I'
       And Atendime.Cd_Atendimento_Pai Is Null
       And Atendime.Cd_Paciente        = Carteira.Cd_Paciente(+)
       And Atendime.Cd_Convenio        = Carteira.Cd_Convenio(+)
       And Atendime.Cd_Con_Pla         = Carteira.Cd_Con_Pla(+)
       And Atendime.Dt_Alta            Between :CAMPO_DT_INI And 
       And Convenio.Cd_Convenio        = Atendime.Cd_Convenio
       And Convenio.Tp_Convenio In ('C','P')
       And Leito.Cd_Leito              = Atendime.Cd_Leito
       And Leito.Cd_Unid_Int           = Unid_Int.Cd_Unid_Int
       And Paciente.Cd_Cidade = Cidade.Cd_Cidade(+)
       and atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
