--R_CAUCAO
SELECT * FROM (
SELECT cauc.DT_RETIRADO,
       cauc.TP_PAGAMENTO,
       aten.CD_ATENDIMENTO,
       paci.CD_PACIENTE,
       paci.NM_PACIENTE,
       cauc.VL_CAUCAO,
       cauc.NR_CHEQUE,
       cauc.CD_FIN_CAR,
       finc.NM_FIN_CAR,
       banc.NM_BANCO,
       cauc.NM_USUARIO_RETIRADO
  FROM DBAMV.ATENDIME aten,
       DBAMV.CAUCAO cauc,
       DBAMV.PACIENTE paci,
       DBAMV.FIN_CAR finc,
       DBAMV.BANCO banc
 WHERE cauc.DT_RETIRADO between :CAMPO_DATINI AND 
   AND cauc.CD_ATENDIMENTO = aten.CD_ATENDIMENTO
   AND aten.CD_PACIENTE    = paci.CD_PACIENTE
   AND cauc.CD_FIN_CAR     = finc.CD_FIN_CAR(+)
   AND cauc.CD_BANCO       = banc.CD_BANCO(+)
   and aten.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
ORDER BY cauc.DT_RETIRADO,
         cauc.TP_PAGAMENTO,
         paci.NM_PACIENTE
)
ORDER BY DT_RETIRADO,
TP_PAGAMENTO
