--R_PAC_INT_IDADE
SELECT PACIENTE.CD_PACIENTE ,
       PACIENTE.NM_PACIENTE ,
   (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
       PACIENTE.DT_NASCIMENTO ,
       ATENDIME.DT_ATENDIMENTO ,
       CIDADE.NM_CIDADE ,
       DECODE(CONVENIO.TP_CONVENIO, 
                'C', PRO_FAT.CD_PRO_FAT , 
                'P', PRO_FAT.CD_PRO_FAT , 
                'A', PRO_FAT.CD_SUS , 
                'H', PRO_FAT.CD_SUS) CD_PRO_FAT ,
       PRO_FAT.DS_PRO_FAT
FROM DBAMV.ATENDIME ,
     DBAMV.PACIENTE ,
     DBAMV.CONVENIO ,
     DBAMV.LEITO ,
     DBAMV.CIDADE ,
     DBAMV.PRO_FAT
WHERE ATENDIME.TP_ATENDIMENTO = 'I'
  AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
  AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
  AND ATENDIME.CD_LEITO = LEITO.CD_LEITO
  AND TRUNC((TRUNC(SYSDATE) - TRUNC(PACIENTE.DT_NASCIMENTO)) / 365.25) BETWEEN :CAMPO_IDADE_INI AND 
  AND PACIENTE.CD_CIDADE = CIDADE.CD_CIDADE(+)
  AND ATENDIME.CD_PRO_INT = PRO_FAT.CD_PRO_FAT(+)
  AND ATENDIME.DT_ALTA IS NULL
  AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA    
ORDER BY PACIENTE.NM_PACIENTE
