--R_CAUCAO_RECEB
SELECT trunc(cauc.DT_CAUCAO) Dt_Caucao,
       cauc.TP_PAGAMENTO    Tp_Pagamento,
       aten.CD_ATENDIMENTO   Cd_Atendimento,
       paci.CD_PACIENTE     Cd_Paciente,
       paci.NM_PACIENTE    Nm_Paciente,
       (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACI.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE,
       cauc.VL_CAUCAO   Vl_Caucao,
       cauc.DT_RETIRADO  Dt_Retirado,
       cauc.NR_CHEQUE   Nr_Cheque,
       cauc.CD_FIN_CAR   Cd_Fin_Car,
       finc.NM_FIN_CAR    Nm_Fin_Car,
       banc.NM_BANCO     Nm_Banco,
       cauc.NM_USUARIO   Nm_Usuario,
       decode(cauc.TP_CAUCAO,'R','Retida','D','Deposito' )  TP_CAUCAO
  FROM DBAMV.ATENDIME aten,
       DBAMV.CAUCAO cauc,
       DBAMV.PACIENTE paci,
       DBAMV.FIN_CAR finc,
       DBAMV.BANCO banc
 WHERE trunc(cauc.DT_CAUCAO) between trunc(:CAMPO_DATINI) AND trunc(:CAMPO_DATFIM)
  AND cauc.CD_ATENDIMENTO = aten.CD_ATENDIMENTO
  AND aten.CD_PACIENTE    = paci.CD_PACIENTE
  AND cauc.CD_FIN_CAR     = finc.CD_FIN_CAR(+)
  AND cauc.CD_BANCO       = banc.CD_BANCO(+)
  AND  ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY  trunc(cauc.DT_CAUCAO),
          cauc.TP_PAGAMENTO    ,
      aten.CD_ATENDIMENTO,
      paci.CD_PACIENTE  ,
      paci.NM_PACIENTE  ,
      cauc.VL_CAUCAO   ,
      cauc.DT_RETIRADO ,
      cauc.NR_CHEQUE   ,
      cauc.CD_FIN_CAR  ,
      finc.NM_FIN_CAR  ,
      banc.NM_BANCO    ,
      cauc.NM_USUARIO   ,
      cauc.cd_caucao,
      decode(cauc.TP_CAUCAO,'R','Retida','D','Deposito' )
ORDER BY trunc(cauc.DT_CAUCAO),
         aten.CD_ATENDIMENTO
