--R_MOV_PREST_PROV
SELECT CD_PRESTADOR, NM_PRESTADOR FROM (SELECT PRESTADOR.CD_PRESTADOR,
       PRESTADOR.NM_PRESTADOR
  FROM DBAMV.ATENDIME,
       DBAMV.PRESTADOR
 WHERE ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
   AND PRESTADOR.SN_ATUANTE = 'P'
   AND DBAMV.FNC_MV_RECUPERA_DATA_HORA(DT_ATENDIMENTO, HR_ATENDIMENTO) BETWEEN :CAMPO_DT_INICIAL UNION
SELECT PRESTADOR.CD_PRESTADOR,
       PRESTADOR.NM_PRESTADOR
  FROM DBAMV.AVISO_CIRURGIA,
       DBAMV.PRESTADOR_AVISO,
       DBAMV.AGE_CIR,
       DBAMV.PRESTADOR
 WHERE AVISO_CIRURGIA.CD_AVISO_CIRURGIA = AGE_CIR.CD_AVISO_CIRURGIA (+)
   AND AVISO_CIRURGIA.CD_AVISO_CIRURGIA = PRESTADOR_AVISO.CD_AVISO_CIRURGIA
   AND PRESTADOR.CD_PRESTADOR           = PRESTADOR_AVISO.CD_PRESTADOR
   AND PRESTADOR.SN_ATUANTE             = 'P'
   AND AGE_CIR.DT_INICIO_AGE_CIR BETWEEN :CAMPO_DT_INICIAL UNION
SELECT PRESTADOR.CD_PRESTADOR,
       PRESTADOR.NM_PRESTADOR
  FROM DBAMV.RES_LEI,
       DBAMV.PRESTADOR
 WHERE RES_LEI.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
   AND PRESTADOR.SN_ATUANTE = 'P'
   AND RES_LEI.DT_PREV_INTERNACAO BETWEEN :CAMPO_DT_INICIAL AND 
:CAMPO_CF_CD_PRESTADOR)
