--R_MOV_GERAL_R_DT_ALTA_A_Q_ALTAS
SELECT ATENDIME.CD_ATENDIMENTO CD_ATENDIMENTO_A ,
       ATENDIME.DT_ATENDIMENTO DT_ATENDIMENTO_A ,
       MOT_ALT.TP_MOT_ALTA TP_MOT_ALTA_A ,
       ATENDIME.DT_ALTA DT_ALTA_A ,
       PACIENTE.NM_PACIENTE NM_PACIENTE_A ,
       (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE_A ,
       PACIENTE.CD_PACIENTE CD_PACIENTE_A ,
       CIDADE.NM_CIDADE NM_CIDADE_A ,
       ATENDIME.CD_LOC_TRANS CD_LOC_TRANS_A ,
       LOC_TRANS.DS_LOC_TRANS DS_LOC_TRANS_A ,
       CONVENIO.NM_CONVENIO NM_CONVENIO_A ,
       PRESTADOR.NM_MNEMONICO NM_PRESTADOR_A ,
       (LEITO.DS_LEITO || ' - ' || LEITO.DS_ENFERMARIA) DS_LEITO_A ,
       UNID_INT.DS_UNID_INT DS_UNID_INT_A
FROM dbamv.ATENDIME ,
     dbamv.PACIENTE ,
     dbamv.LOC_TRANS ,
     dbamv.CONVENIO ,
     dbamv.PRESTADOR ,
     dbamv.LEITO ,
     dbamv.UNID_INT ,
     dbamv.CIDADE ,
     dbamv.MOT_ALT
WHERE ATENDIME.TP_ATENDIMENTO IN ('I', 'H')
  AND TRUNC(ATENDIME.DT_ALTA) BETWEEN :CAMPO_DT_INICIAL AND 
  AND ATENDIME.CD_MOT_ALT = MOT_ALT.CD_MOT_ALT(+)
  AND MOT_ALT.TP_MOT_ALTA IN ('T', 'A', 'D')
  AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
  AND PACIENTE.CD_CIDADE = CIDADE.CD_CIDADE(+)
  AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
  AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
  AND ATENDIME.CD_LOC_TRANS = LOC_TRANS.CD_LOC_TRANS(+)
  AND ATENDIME.CD_LEITO = LEITO.CD_LEITO
  AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT
  AND UNID_INT.SN_ATIVO = 'S'
  AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA 
ORDER BY DS_UNID_INT

--R_MOV_GERAL_R_DT_ATENDIMENTO_Q_INTERNACOES
SELECT ATENDIME.CD_ATENDIMENTO CD_ATENDIMENTO_T ,
       ATENDIME.DT_ATENDIMENTO DT_ATENDIMENTO_T ,
       MOT_ALT.TP_MOT_ALTA TP_MOT_ALTA_T ,
       ATENDIME.DT_ALTA DT_ALTA_T ,
       PACIENTE.NM_PACIENTE NM_PACIENTE_T ,
       PACIENTE.CD_PACIENTE CD_PACIENTE_T ,
       CIDADE.NM_CIDADE NM_CIDADE_T ,
       ATENDIME.CD_LOC_TRANS CD_LOC_TRANS_T ,
       LOC_TRANS.DS_LOC_TRANS DS_LOC_TRANS_T ,
       CONVENIO.NM_CONVENIO NM_CONVENIO_T ,
       PRESTADOR.NM_MNEMONICO NM_PRESTADOR_T ,
       (LEITO.DS_LEITO || ' - ' || LEITO.DS_ENFERMARIA) DS_LEITO_T ,
       UNID_INT.DS_UNID_INT DS_UNID_INT_T ,
       ATENDIME.SENHA_SUS SENHA_SUS,
       (SELECT P.NM_SOCIAL_PACIENTE  FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE = CD_PACIENTE) NM_SOCIAL_PACIENTE_T
FROM dbamv.MOV_INT ,
     dbamv.ATENDIME ,
     dbamv.PACIENTE ,
     dbamv.LOC_TRANS ,
     dbamv.CONVENIO ,
     dbamv.PRESTADOR ,
     dbamv.LEITO ,
     dbamv.UNID_INT ,
     dbamv.CIDADE ,
     dbamv.MOT_ALT
WHERE MOV_INT.TP_MOV = 'I'
  AND TRUNC(MOV_INT.DT_MOV_INT) BETWEEN :CAMPO_DT_INICIAL AND 
  AND TRUNC(NVL(MOV_INT.DT_LIB_MOV, SYSDATE)) >= TRUNC(MOV_INT.DT_MOV_INT)
  AND ATENDIME.TP_ATENDIMENTO = 'I'
  AND MOV_INT.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
  AND ATENDIME.CD_LOC_TRANS = LOC_TRANS.CD_LOC_TRANS(+)
  AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
  AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
  AND ATENDIME.CD_MOT_ALT = MOT_ALT.CD_MOT_ALT(+)
  AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
  AND PACIENTE.CD_CIDADE = CIDADE.CD_CIDADE(+)
  AND MOV_INT.CD_LEITO = LEITO.CD_LEITO
  AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT
  AND UNID_INT.SN_ATIVO = 'S'
  AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA 
ORDER BY DS_UNID_INT
