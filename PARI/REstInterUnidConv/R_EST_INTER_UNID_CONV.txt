--R_EST_INTER_UNID_CONV_R_UNIDADE_Q_UNIDADE
SELECT  UNID_INT.CD_UNID_INT
       ,UNID_INT.DS_UNID_INT
       ,COUNT(*)  QUANT
       ,(SELECT Sum(QUANT) FROM (
            SELECT  UNID_INT.CD_UNID_INT
                   ,UNID_INT.DS_UNID_INT
                   ,COUNT(*)  QUANT
            FROM    DBAMV.ATENDIME
                   ,DBAMV.LEITO
                   ,DBAMV.UNID_INT
                   ,DBAMV.MOV_INT
                   ,DBAMV.CONVENIO
            WHERE  ATENDIME.TP_ATENDIMENTO  IN  ('I', 'H')
              AND  ATENDIME.CD_ATENDIMENTO_PAI IS NULL
              AND  TRUNC(MOV_INT.DT_MOV_INT)  BETWEEN  :CAMPO_DT_INICIO  AND  
               AND  MOV_INT.CD_LEITO   =  LEITO.CD_LEITO
              AND  LEITO.CD_UNID_INT        = UNID_INT.CD_UNID_INT
              AND  ATENDIME.CD_ATENDIMENTO  = MOV_INT.CD_ATENDIMENTO
              AND  ATENDIME.CD_CONVENIO     = CONVENIO.CD_CONVENIO
              AND  MOV_INT.TP_MOV           = 'I'
            GROUP BY  UNID_INT.CD_UNID_INT
                                ,UNID_INT.DS_UNID_INT
                               )
        ) TOT_UNID
FROM    DBAMV.ATENDIME
       ,DBAMV.LEITO
       ,DBAMV.UNID_INT
       ,DBAMV.MOV_INT
       ,DBAMV.CONVENIO
WHERE  ATENDIME.TP_ATENDIMENTO  IN  ('I', 'H')
  AND  ATENDIME.CD_ATENDIMENTO_PAI IS NULL
  AND  TRUNC(MOV_INT.DT_MOV_INT)  BETWEEN  :CAMPO_DT_INICIO  AND  
   AND  MOV_INT.CD_LEITO   =  LEITO.CD_LEITO
  AND  LEITO.CD_UNID_INT        = UNID_INT.CD_UNID_INT
  AND  ATENDIME.CD_ATENDIMENTO  = MOV_INT.CD_ATENDIMENTO
  AND  ATENDIME.CD_CONVENIO     = CONVENIO.CD_CONVENIO
  AND  MOV_INT.TP_MOV           = 'I'
GROUP BY  UNID_INT.CD_UNID_INT
                    ,UNID_INT.DS_UNID_INT
ORDER BY  QUANT DESC
