           NULL DS_SERVICO
FROM
  (SELECT CONVENIO.CD_CONVENIO ,
          CONVENIO.NM_CONVENIO ,
          NVL(COUNT(ATENDIME.CD_CONVENIO), 0) INTERNOS ,
          0 INTERNOS_DIA ,
          0 OBITOS_DIA ,
          0 ALTAS_DIA
   FROM DBAMV.LEITO LEITO ,
        DBAMV.UNID_INT UNID_INT ,
        DBAMV.ATENDIME ATENDIME ,
        DBAMV.CONVENIO CONVENIO
   WHERE atendime.CD_LEITO = LEITO.CD_LEITO
    AND LEITO.TP_OCUPACAO IN ('O', 'I')
    AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT
    AND :CAMPO_DATA >= leito.dt_ativacao
    AND :CAMPO_DATA <= nvl(leito.dt_desativacao, sysdate)
    AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
    AND LEITO.TP_SITUACAO = 'A' 
    AND atendime.tp_atendimento IN ('I', 'U')
    AND trunc(atendime.dt_alta) IS NULL 
    AND :CAMPO_RESUMO = 'S' 
   GROUP BY CONVENIO.CD_CONVENIO ,
            CONVENIO.NM_CONVENIO
   UNION ALL 
   SELECT CONVENIO.CD_CONVENIO ,
          CONVENIO.NM_CONVENIO ,
          0 INTERNOS ,
          NVL(SUM(DECODE(ATENDIME.DT_ATENDIMENTO, :CAMPO_DATA,1, 0)), 0) INTERNOS_DIA ,
          NVL(SUM(DECODE(ATENDIME.DT_ALTA || MOT_ALT.TP_MOT_ALTA, :CAMPO_DATA || 'O', 1, 0)), 0) OBITOS_DIA ,
          NVL(SUM(DECODE(ATENDIME.DT_ALTA || MOT_ALT.TP_MOT_ALTA, :CAMPO_DATA || 'D', 1, :CAMPO_DATA || 'A', 1, :CAMPO_DATA || 'T', 1, 0)), 0) ALTAS_DIA
   FROM DBAMV.LEITO LEITO ,
        DBAMV.UNID_INT UNID_INT ,
        DBAMV.ATENDIME ATENDIME ,
        DBAMV.CONVENIO CONVENIO ,
        DBAMV.MOT_ALT MOT_ALT
   WHERE ATENDIME.CD_LEITO = LEITO.CD_LEITO 
     AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT
     AND :CAMPO_DATA >= leito.dt_ativacao
     AND :CAMPO_DATA <= nvl(leito.dt_desativacao, sysdate)
     AND ATENDIME.TP_ATENDIMENTO = 'I'
     AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
     AND ATENDIME.CD_MOT_ALT = MOT_ALT.CD_MOT_ALT(+)
     AND :CAMPO_RESUMO = 'S'
     AND LEITO.TP_SITUACAO = 'A' 
   GROUP BY CONVENIO.CD_CONVENIO ,
            CONVENIO.NM_CONVENIO) CONVENIO
WHERE NVL(CONVENIO.INTERNOS, 0) > 0
  OR NVL(CONVENIO.INTERNOS_DIA, 0) > 0
  OR NVL(CONVENIO.OBITOS_DIA, 0) > 0
  OR NVL(CONVENIO.ALTAS_DIA, 0) > 0
GROUP BY CONVENIO.CD_CONVENIO ,
         CONVENIO.NM_CONVENIO
UNION ALL

SELECT 'C'                                          TIPO
       ,TO_NUMBER(NULL)                             CD_UNID_INT
       ,NULL                                        DS_RESUMO
       ,NULL                                        TP_OCUPACAO
       ,NULL                                        DS_ENFERMARIA
       ,NULL                                        DS_UNID_INT
       ,NULL                                        NM_PACIENTE
       ,TO_DATE(NULL)                               DT_NASCIMENTO
       ,NULL                                        NM_CONVENIO
       ,NULL                                        NM_PRESTADOR
       ,TO_DATE(NULL)                               DT_ATENDIMENTO
       ,TO_NUMBER(NULL)                             CD_ATENDIMENTO
       ,TO_NUMBER(NULL)                             CD_LEITO
       ,CONVENIO.CD_CONVENIO                        CODIGO_RESUMO
       ,CONVENIO.NM_CONVENIO                        DESCRICAO_RESUMO
       ,NVL(SUM(INTERNOS),0)     INTERNOS
       ,NVL(SUM(INTERNOS_DIA),0) INTERNOS_DIA
       ,NVL(SUM(OBITOS_DIA),0)   OBITOS_DIA
       ,NVL(SUM(ALTAS_DIA),0)    ALTAS_DIA
       ,0 ATIVOS_ACOM
       ,0 EXTRAS_ACOM
       ,0 INTERNOS_ACOM
       ,NULL                                        SN_EXTRA
       ,to_number(null)                             CD_CONV_UNID
       ,null                                        DS_CONV_UNID
       ,0 QT_CONV_UNID
       ,TO_NUMBER(NULL) CD_PACIENTE
       ,NULL                       DS_SUB_PLANO
FROM
      (SELECT
       CONVENIO.CD_CONVENIO
      ,CONVENIO.NM_CONVENIO
      ,NVL(COUNT(ATENDIME.CD_CONVENIO),0) INTERNOS
      ,0 INTERNOS_DIA
      ,0 OBITOS_DIA
      ,0 ALTAS_DIA
       FROM        DBAMV.V_INTERNADOS    V_INTERNADOS
                  ,DBAMV.LEITO           LEITO
                  ,DBAMV.UNID_INT        UNID_INT
                  ,DBAMV.ATENDIME        ATENDIME
                  ,DBAMV.CONVENIO        CONVENIO
       WHERE       V_INTERNADOS.CD_LEITO  =  LEITO.CD_LEITO
       AND        LEITO.CD_UNID_INT  =  UNID_INT.CD_UNID_INT
       AND        :CAMPO_DATA >= leito.dt_ativacao
       AND        :CAMPO_DATA <= nvl(leito.dt_desativacao,sysdate)
       AND        ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
       AND        V_INTERNADOS.CD_ATENDIMENTO  =  ATENDIME.CD_ATENDIMENTO
       AND         LEITO.TP_SITUACAO = 'A'
       AND        LEITO.TP_OCUPACAO IN ('O','I')
      AND  :CAMPO_RESUMO = 'S'
       GROUP BY CONVENIO.CD_CONVENIO
               ,CONVENIO.NM_CONVENIO
       UNION ALL
       SELECT
       CONVENIO.CD_CONVENIO
      ,CONVENIO.NM_CONVENIO
      ,0 INTERNOS
      ,NVL(SUM(DECODE(ATENDIME.DT_ATENDIMENTO,:CAMPO_DATA,1,0)),0)                        INTERNOS_DIA
      ,NVL(SUM(DECODE(ATENDIME.DT_ALTA || MOT_ALT.TP_MOT_ALTA,:CAMPO_DATA || 'O',1,0)),0) OBITOS_DIA
      ,NVL(SUM(DECODE(ATENDIME.DT_ALTA || MOT_ALT.TP_MOT_ALTA,:CAMPO_DATA || 'D',1,
          :CAMPO_DATA || 'A',1,:CAMPO_DATA || 'T',1,0)),0)                          ALTAS_DIA
       FROM        DBAMV.LEITO           LEITO
                  ,DBAMV.UNID_INT        UNID_INT
                  ,DBAMV.ATENDIME        ATENDIME
                  ,DBAMV.CONVENIO        CONVENIO
                  ,DBAMV.MOT_ALT         MOT_ALT
       WHERE       ATENDIME.CD_LEITO  =  LEITO.CD_LEITO
       AND        LEITO.CD_UNID_INT  =  UNID_INT.CD_UNID_INT
       AND        :CAMPO_DATA >= leito.dt_ativacao
       AND        :CAMPO_DATA <= nvl(leito.dt_desativacao,sysdate)
       AND        ATENDIME.TP_ATENDIMENTO = 'I'
       AND        ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
       AND        ATENDIME.CD_MOT_ALT = MOT_ALT.CD_MOT_ALT(+)
      AND  :CAMPO_RESUMO = 'S'
      AND         LEITO.TP_SITUACAO = 'A'
       AND        LEITO.TP_OCUPACAO IN ('O','I')
       GROUP BY CONVENIO.CD_CONVENIO
               ,CONVENIO.NM_CONVENIO) CONVENIO
WHERE NVL(CONVENIO.INTERNOS,0) > 0   OR
      NVL(CONVENIO.INTERNOS_DIA,0) > 0  OR
      NVL(CONVENIO.OBITOS_DIA,0) > 0  OR
      NVL(CONVENIO.ALTAS_DIA,0) > 0
GROUP BY CONVENIO.CD_CONVENIO
        ,CONVENIO.NM_CONVENIO
 ORDER BY codigo_resumo
