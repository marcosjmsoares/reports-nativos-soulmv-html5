--R_EST_SERVICO_R_CG__G_ATENDIME2_Q_SERVICO
SELECT  SERVICO.CD_SERVICO
               ,SERVICO.DS_SERVICO
               ,COUNT( ATENDIME.CD_ATENDIMENTO)                QTD_ATENDE
               ,TRUNC( AVG(NVL( TRUNC(ATENDIME.DT_ALTA), TRUNC(SYSDATE))  -  TRUNC(DT_ATENDIMENTO)))    MEDIA_PERMAN
	       , (SELECT SUM(QTD_ATENDE) FROM (
		   SELECT SERVICO.CD_SERVICO
                   ,Nvl(SERVICO.DS_SERVICO,'NÃO INFORMADO') DS_SERVICO 
                   ,COUNT( ATENDIME.CD_ATENDIMENTO)                QTD_ATENDE
                   ,TRUNC( AVG(NVL( TRUNC(ATENDIME.DT_ALTA), TRUNC(SYSDATE))  -  TRUNC(DT_ATENDIMENTO)))    MEDIA_PERMAN
		FROM     DBAMV.ATENDIME
                  ,DBAMV.SERVICO
                  ,DBAMV.TIPO_INTERNACAO INTERN
                  ,DBAMV.LEITO
                  ,DBAMV.UNID_INT
		WHERE SERVICO.CD_SERVICO(+)  =  ATENDIME.CD_SERVICO
		AND       ATENDIME.CD_TIPO_INTERNACAO = INTERN.CD_TIPO_INTERNACAO(+)
		AND       ATENDIME.TP_ATENDIMENTO IN ( 'I', 'H' )
		AND       ATENDIME.CD_ATENDIMENTO_PAI IS NULL
		AND       TRUNC(ATENDIME.DT_ATENDIMENTO)  BETWEEN :CAMPO_DATAINI  AND  
		AND       ATENDIME.CD_LEITO                = LEITO.CD_LEITO(+)
		AND       LEITO.CD_UNID_INT                  = UNID_INT.CD_UNID_INT(+)
		AND       ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
		GROUP BY SERVICO.CD_SERVICO
                   ,SERVICO.DS_SERVICO)) TOTAL_GERAL
	FROM     DBAMV.ATENDIME
         ,DBAMV.SERVICO
         ,DBAMV.TIPO_INTERNACAO INTERN
         ,DBAMV.LEITO
         ,DBAMV.UNID_INT
WHERE SERVICO.CD_SERVICO(+)  =  ATENDIME.CD_SERVICO
AND       ATENDIME.CD_TIPO_INTERNACAO = INTERN.CD_TIPO_INTERNACAO(+)
AND       ATENDIME.TP_ATENDIMENTO IN ( 'I', 'H' )
AND       ATENDIME.CD_ATENDIMENTO_PAI IS NULL
AND       TRUNC(ATENDIME.DT_ATENDIMENTO)  BETWEEN :CAMPO_DATAINI  AND  
AND       ATENDIME.CD_LEITO                = LEITO.CD_LEITO(+)
AND       LEITO.CD_UNID_INT                  = UNID_INT.CD_UNID_INT(+)
AND       ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY SERVICO.CD_SERVICO
                   ,SERVICO.DS_SERVICO
ORDER BY  QTD_ATENDE  DESC

--R_EST_SERVICO_R_CG__G_ATENDIME_CG__Q_ATENDIME
SELECT  SERVICO.CD_SERVICO
               ,SERVICO.DS_SERVICO
               ,COUNT( ATENDIME.CD_ATENDIMENTO)                QT_ATENDE
               ,TRUNC( AVG(NVL( TRUNC(ATENDIME.DT_ALTA), TRUNC(SYSDATE))  -  TRUNC(DT_ATENDIMENTO)))    MEDIA_PERMAN
         ,(SELECT SUM(QT_ATENDE) FROM (
                    SELECT  SERVICO.CD_SERVICO
                           ,Nvl(SERVICO.DS_SERVICO,'NÃO INFORMADO') DS_SERVICO 
                           ,COUNT( ATENDIME.CD_ATENDIMENTO)                QT_ATENDE
                           ,TRUNC( AVG(NVL( TRUNC(ATENDIME.DT_ALTA), TRUNC(SYSDATE))  -  TRUNC(DT_ATENDIMENTO)))    MEDIA_PERMAN
                    FROM     DBAMV.ATENDIME
                            ,DBAMV.SERVICO
                            ,DBAMV.LEITO
                            ,DBAMV.UNID_INT
                    WHERE SERVICO.CD_SERVICO(+)  =  ATENDIME.CD_SERVICO
                      AND       ATENDIME.TP_ATENDIMENTO IN ( 'I', 'H' )
                      AND       ATENDIME.CD_ATENDIMENTO_PAI IS NULL
                      AND       TRUNC(ATENDIME.DT_ATENDIMENTO)  BETWEEN :CAMPO_DATAINI  AND   
AND       ATENDIME.CD_LEITO                = LEITO.CD_LEITO(+)
AND       LEITO.CD_UNID_INT   = UNID_INT.CD_UNID_INT(+)
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY  SERVICO.CD_SERVICO
                   ,SERVICO.DS_SERVICO
)) TOTAL_GERAL
FROM     DBAMV.ATENDIME
               ,DBAMV.SERVICO
               ,DBAMV.TIPO_INTERNACAO INTERN
               ,DBAMV.LEITO
               ,DBAMV.UNID_INT
WHERE SERVICO.CD_SERVICO(+)  =  ATENDIME.CD_SERVICO
AND       ATENDIME.CD_TIPO_INTERNACAO = INTERN.CD_TIPO_INTERNACAO(+)
AND       ATENDIME.TP_ATENDIMENTO IN ( 'I', 'H' )
AND       ATENDIME.CD_ATENDIMENTO_PAI IS NULL
AND       TRUNC(ATENDIME.DT_ATENDIMENTO)  BETWEEN :CAMPO_DATAINI  AND   
AND       ATENDIME.CD_LEITO                = LEITO.CD_LEITO(+)
AND       LEITO.CD_UNID_INT                  = UNID_INT.CD_UNID_INT(+)
AND       ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY SERVICO.CD_SERVICO
                   ,SERVICO.DS_SERVICO
ORDER BY  QT_ATENDE  DESC
