--R_HOMONIMOS_POR_COMPET
SELECT PACIENTE.NM_PACIENTE
              ,COUNT(PACIENTE.NM_PACIENTE)   QT_ATENDE
FROM     DBAMV.ATENDIME
              ,DBAMV.PACIENTE
WHERE TO_CHAR(ATENDIME.DT_ATENDIMENTO,'MM/YYYY')  =  :CAMPO_COMPETENCIA
AND        ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY PACIENTE.NM_PACIENTE
HAVING COUNT(PACIENTE.NM_PACIENTE) > 1
ORDER BY PACIENTE.NM_PACIENTE
