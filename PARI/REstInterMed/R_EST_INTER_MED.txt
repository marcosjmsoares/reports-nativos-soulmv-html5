--R_EST_INTER_MED
SELECT PRES.CD_PRESTADOR,
               PRES.NM_PRESTADOR,
               COUNT(*)         INTERNACOES
FROM     DBAMV.ATENDIME      ATEN
              ,DBAMV.PRESTADOR   PRES
              ,DBAMV.LEITO
              ,DBAMV.UNID_INT
WHERE  TRUNC(ATEN.DT_ATENDIMENTO)  BETWEEN  :CAMPO_DATINI  AND  
AND        InStr('HI',ATEN.TP_ATENDIMENTO) > 0
AND        ATEN.CD_ATENDIMENTO_PAI IS NULL
AND        ATEN.CD_PRESTADOR  =  PRES.CD_PRESTADOR
AND       ATEN.CD_LEITO                = leito.CD_LEITO(+)
AND       LEITO.CD_UNID_INT   = UNID_INT.CD_UNID_INT(+)
AND    ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY  PRES.NM_PRESTADOR,
                     PRES.CD_PRESTADOR
ORDER BY INTERNACOES DESC
