--R_DECLARA_COMP
SELECT PACIENTE.CD_PACIENTE                    CD_PACIENTE,
               PACIENTE.NM_PACIENTE                   NM_PACIENTE,
               (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE = P.CD_PACIENTE) NM_SOCIAL_PACIENTE,
               PACIENTE.DT_NASCIMENTO              DT_NASCIMENTO,
               PACIENTE.TP_SEXO                            TP_SEXO,
               PACIENTE.TP_COR                              TP_COR,
               DBAMV.FNC_ESTADO_CIVIL(PACIENTE.TP_ESTADO_CIVIL) TP_ESTADO_CIVIL1,
               PACIENTE.NR_DOCUMENTO              NR_DOCUMENTO,
               PACIENTE.NM_MAE                             NM_MAE,
               PACIENTE.DS_ENDERECO                 DS_ENDERECO,
               PACIENTE.NR_ENDERECO                 NR_ENDERECO,
               PACIENTE.NM_BAIRRO                       NM_BAIRRO,
               PACIENTE.NR_CEP                              NR_CEP,
               PACIENTE.NR_FONE                           NR_FONE,
               PACIENTE.DT_CADASTRO                  DT_CADASTRO,
               PACIENTE.NR_IDENTIDADE               NR_IDENTIDADE,
               PACIENTE.DS_OM_IDENTIDADE        DS_OM_IDENTIDADE,
               PACIENTE.NR_CPF                               NR_CPF,
               ATENDIME.CD_ATENDIMENTO          CD_ATENDIMENTO,
               ATENDIME.CD_PACIENTE                   CD_PACIENTE,
               ATENDIME.DT_ATENDIMENTO          DT_ATENDIMENTO,
               ATENDIME.DT_ALTA                           DT_ALTA,
               ATENDIME.HR_ALTA                           HR_ALTA,
               ATENDIME.HR_ATENDIMENTO          HR_ATENDIMENTO,
               ATENDIME.TP_ATENDIMENTO          TP_ATENDIMENTO,
               ATENDIME.CD_CONVENIO                  CD_CONVENIO,
               ORI_ATE.DS_ORI_ATE                         DS_ORI_ATE,
               PRESTADOR.NM_PRESTADOR          NM_PRESTADOR,
               PRESTADOR.NR_CPF_CGC                 NR_CPF_CGC,
               PRESTADOR.DS_CODIGO_CONSELHO   DS_CODIGO_CONSELHO,
               LEITO.DS_ENFERMARIA                      DS_ENFERMARIA,
               LEITO.DS_LEITO                                   DS_LEITO,
               UNID_INT.CD_UNID_INT                      CD_UNID_INT,
               UNID_INT.DS_UNID_INT                      DS_UNID_INT,
               UNID_INT.DS_LOCALIZACAO               DS_LOCALIZACAO,
               SETOR.NM_SETOR                        NM_SETOR,
               CONVENIO.NM_CONVENIO                  NM_CONVENIO,
               RESPONSA.NM_RESPONSAVEL          NM_RESPONSAVEL,
               RESPONSA.NR_FONE                            NR_FONE_RESPONSAVEL,
               RESPONSA.DS_DOCUMENTO               DS_DOCUMENTO_RESPONSAVEL,
               RESPONSA.NR_CPF                               NR_CPF_RESP,
               RESPONSA.DS_ENDERECO                  DS_ENDERECO_RESP,
               RESPONSA.NM_BAIRRO                        NM_BAIRRO_RESP,
               CID_RESP.NM_CIDADE                          NM_CIDADE_RESP,
               RESPONSA.DS_CONTATO_FAMILIA     DS_CONTATO_FAMILIA,
               SUBSTR( CIDADE.NM_CIDADE, 1, 100)   NM_CIDADE,
               CIDADE.CD_UF                                        CD_UF,
               DS_INFO_ATENDIMENTO                      DS_INFO_ATENDIMENTO
FROM  DBAMV.ATENDIME,
            DBAMV.PACIENTE,
            DBAMV.CIDADE,
            DBAMV.ORI_ATE,
            DBAMV.PRESTADOR,
            DBAMV.LEITO,
            DBAMV.UNID_INT,
            DBAMV.SETOR,
            DBAMV.CONVENIO,
            DBAMV.RESPONSA,
            DBAMV.CIDADE            CID_RESP,
            DBAMV.ATENDIME_INFO
WHERE  ATENDIME.CD_ATENDIMENTO  =  :CAMPO_PCD_ATENDIMENTO
AND        ATENDIME.CD_ATENDIMENTO  =  ATENDIME_INFO.CD_ATENDIMENTO (+)
AND        ATENDIME.CD_PACIENTE  =  PACIENTE.CD_PACIENTE
AND        PACIENTE.CD_CIDADE  =  CIDADE.CD_CIDADE(+)
AND        ATENDIME.CD_ORI_ATE  =  ORI_ATE.CD_ORI_ATE
AND        ATENDIME.CD_PRESTADOR  =  PRESTADOR.CD_PRESTADOR(+)
AND        ATENDIME.CD_LEITO  =  LEITO.CD_LEITO(+)
AND        LEITO.CD_UNID_INT  =  UNID_INT.CD_UNID_INT(+)
AND        UNID_INT.CD_SETOR = SETOR.CD_SETOR(+)
AND        ATENDIME.CD_CONVENIO  =  CONVENIO.CD_CONVENIO
AND        ATENDIME.CD_ATENDIMENTO  =  RESPONSA.CD_ATENDIMENTO (+)
AND        RESPONSA.CD_CIDADE  =  CID_RESP.CD_CIDADE(+)
AND        ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
