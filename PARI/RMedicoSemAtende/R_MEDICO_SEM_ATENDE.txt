--R_MEDICO_SEM_ATENDE
SELECT  PRESTADOR.CD_PRESTADOR
               ,PRESTADOR.NM_PRESTADOR
              ,(TRUNC(SYSDATE) - MAX(TRUNC(ATENDIME.DT_ATENDIMENTO)))   QT_DIAS
               ,MAX(TRUNC(ATENDIME.DT_ATENDIMENTO))     DT_ATENDIMENTO
FROM    DBAMV.ATENDIME
              ,DBAMV.PRESTADOR
WHERE      PRESTADOR.TP_SITUACAO  =  'A'
AND            ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
and atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY PRESTADOR.CD_PRESTADOR
                   ,PRESTADOR.NM_PRESTADOR
HAVING  (TRUNC(SYSDATE) - MAX(TRUNC(ATENDIME.DT_ATENDIMENTO)))   >=  :CAMPO_QT_DIAS
