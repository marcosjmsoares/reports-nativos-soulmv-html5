--R_EST_ALTAS_MOT
SELECT moal.CD_MOT_ALT,
       moal.DS_MOT_ALT,
       COUNT(*)   QTDALTAS
FROM   DBAMV.ATENDIME aten,
       DBAMV.MOT_ALT moal
WHERE aten.DT_ALTA between :CAMPO_DATINI and 
AND    aten.CD_MOT_ALT= moal.CD_MOT_ALT
AND    ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY moal.DS_MOT_ALT,
         moal.CD_MOT_ALT
ORDER BY QTDALTAS DESC
