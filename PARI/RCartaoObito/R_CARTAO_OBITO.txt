--R_CARTAO_OBITO_RF_ETIQ1_Q_ATENDIMENTO
SELECT * FROM (
SELECT '1'  A1
             ,ATENDIME.CD_PACIENTE
             ,NVL(ATENDIME.DT_ALTA_MEDICA,ATENDIME.DT_ALTA) DT_ALTA_MEDICA
             ,NVL(ATENDIME.HR_ALTA_MEDICA,ATENDIME.HR_ALTA) HR_ALTA_MEDICA
             ,PACIENTE.NM_PACIENTE
             ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE
             ,(PACIENTE.NR_IDENTIDADE || ' ' || PACIENTE.DS_OM_IDENTIDADE)      IDENTIDADE
             ,PACK_INTERNAMENTO.RETORNA_IDADE(PACIENTE.DT_NASCIMENTO, SYSDATE)    NR_IDADE
             ,UNID_INT.CD_UNID_INT
             ,UNID_INT.DS_UNID_INT
             ,DECODE(:CAMPO_IML,'S','SIM','NAO')  PARA_IML
             ,CIDADE.NM_CIDADE
             ,UNID_INT.CD_SETOR CD_SETOR
FROM   DBAMV.ATENDIME
             ,DBAMV.PACIENTE
             ,DBAMV.MOT_ALT
             ,DBAMV.LEITO
             ,DBAMV.UNID_INT
             ,DBAMV.CIDADE
WHERE  ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND        PACIENTE.CD_CIDADE = CIDADE.CD_CIDADE(+)
AND        ATENDIME.CD_MOT_ALT = MOT_ALT.CD_MOT_ALT
AND        ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
AND        LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)
AND        MOT_ALT.TP_MOT_ALTA = 'O'
AND        ATENDIME.DT_ALTA IS NOT NULL
AND        ATENDIME.CD_PACIENTE = :CAMPO_CD_PACIENTE
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
UNION
SELECT '1'  A1
             ,ATENDIME.CD_PACIENTE
             ,NVL(ATENDIME.DT_ALTA_MEDICA,ATENDIME.DT_ALTA) DT_ALTA_MEDICA
             ,NVL(ATENDIME.HR_ALTA_MEDICA,ATENDIME.HR_ALTA) HR_ALTA_MEDICA
             ,PACIENTE.NM_PACIENTE
             ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE
             ,(PACIENTE.NR_IDENTIDADE || ' ' || PACIENTE.DS_OM_IDENTIDADE)      IDENTIDADE
             ,PACK_INTERNAMENTO.RETORNA_IDADE(PACIENTE.DT_NASCIMENTO, SYSDATE)    NR_IDADE
             ,NULL CD_UNID_INT
             ,NULL DS_UNID_INT
             ,DECODE(:CAMPO_IML,'S','SIM','NAO')  PARA_IML
             ,CIDADE.NM_CIDADE
             ,NULL CD_SETOR
FROM   DBAMV.ATENDIME
             ,DBAMV.PACIENTE
             ,DBAMV.CIDADE
             ,DBAMV.TIP_RES
WHERE  ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND        PACIENTE.CD_CIDADE = CIDADE.CD_CIDADE(+)
AND       ATENDIME.CD_TIP_RES = TIP_RES.CD_TIP_RES
AND       TIP_RES.SN_OBITO = 'S'
AND        ATENDIME.DT_ALTA IS NOT NULL
AND        ATENDIME.CD_PACIENTE = :CAMPO_CD_PACIENTE
AND  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
:CAMPO_CF_MULTI_EMPRESA
)
WHERE A1 = :CAMPO_A
