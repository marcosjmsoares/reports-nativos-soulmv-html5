--R_PAC_ACOMOD
SELECT * FROM
			  (SELECT convenio.cd_convenio ,
			          convenio.nm_convenio ,
			          empresa_con_pla.cd_con_pla ,
			          atendime.cd_tip_acom_cobertura,
			          atendime.cd_atendimento ,
			          paciente.cd_paciente ,
			          paciente.nm_paciente ,
			      (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
			          leito.ds_leito ,
			          leito.ds_enfermaria ,
			          tip_acom.ds_tip_acom ,
			          atendime.dt_atendimento ,
			          mov_int.dt_mov_int ,
			          mov_int.dt_lib_mov ,
			          1 qtd
			   FROM dbamv.mov_int,
			        dbamv.atendime,
			        dbamv.paciente,
			        dbamv.tip_acom,
			        dbamv.leito,
			        dbamv.convenio,
			        dbamv.empresa_con_pla
			   WHERE atendime.cd_atendimento = mov_int.cd_atendimento
			     AND mov_int.cd_tip_acom = tip_acom.cd_tip_acom
			     AND mov_int.cd_leito = leito.cd_leito
			     AND atendime.cd_paciente = paciente.cd_paciente
			     AND atendime.cd_tip_acom_cobertura <> mov_int.cd_tip_acom
			     AND atendime.cd_convenio = convenio.cd_convenio
			     AND empresa_con_pla.cd_convenio = atendime.cd_convenio
			     AND atendime.cd_con_pla = empresa_con_pla.cd_con_pla
			     AND empresa_con_pla.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
			     AND atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA     
			     AND atendime.dt_atendimento BETWEEN :CAMPO_DT_INIC ORDER BY mov_int.dt_mov_int)
			ORDER BY CD_CONVENIO,
			         NM_CONVENIO,
			         CD_CON_PLA,
			         CD_TIP_ACOM_COBERTURA,
			         dt_mov_int
