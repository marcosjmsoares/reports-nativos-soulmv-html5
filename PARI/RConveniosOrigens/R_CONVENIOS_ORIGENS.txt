--R_CONVENIOS_ORIGENS
SELECT * FROM(
SELECT  ATE.CD_CONVENIO  CD_CONVEINO,
        CON.NM_CONVENIO  NM_CONVENIO,
        ATE.CD_ORI_ATE   CD_ORI_ATE,
        ORI.DS_ORI_ATE   DS_ORI_ATE,
        COUNT(*)         QTD
FROM    DBAMV.ATENDIME   ATE,
        DBAMV.CONVENIO   CON,
        DBAMV.ORI_ATE    ORI
WHERE   ATE.DT_ATENDIMENTO   >= TO_DATE( TO_CHAR( :CAMPO_DT_INICIAL, 'DD/MM/YYYY' ) || ' 00', 'DD/MM/YYYY HH24' )
AND     ATE.DT_ATENDIMENTO   <= TO_DATE( TO_CHAR( :CAMPO_DT_FINAL  , 'DD/MM/YYYY' ) || ' 23', 'DD/MM/YYYY HH24' )
AND     ATE.CD_CONVENIO      =  CON.CD_CONVENIO
AND     ATE.CD_ORI_ATE       =  ORI.CD_ORI_ATE
AND     ATE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY ATE.CD_CONVENIO,
         CON.NM_CONVENIO,
         ATE.CD_ORI_ATE,
         ORI.DS_ORI_ATE
)
ORDER BY NM_CONVENIO
