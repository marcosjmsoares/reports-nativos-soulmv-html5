SELECT * FROM (
SELECT  PRESTADOR.NM_PRESTADOR
               ,PRESTADOR.CD_PRESTADOR
               ,LEITO.DS_ENFERMARIA
               ,LEITO.DS_RESUMO
               ,PACIENTE.CD_PACIENTE
               ,PACIENTE.NM_PACIENTE
               ,ATENDIME.CD_ATENDIMENTO
               ,ATENDIME.DT_ATENDIMENTO
               ,ATENDIME.HR_ATENDIMENTO
               ,CID.CD_CID CD_CID
               ,CID.DS_CID
               ,LEITO.DS_LEITO
FROM    dbamv.PRESTADOR
              ,dbamv.ATENDIME
              ,dbamv.LEITO
              ,dbamv.PACIENTE
              ,dbamv.CID
WHERE   ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
      AND   LEITO.CD_LEITO = ATENDIME.CD_LEITO
      AND   PACIENTE.CD_PACIENTE = ATENDIME.CD_PACIENTE
      AND   ATENDIME.CD_CID = CID.CD_CID(+)
      AND   ATENDIME.TP_ATENDIMENTO IN ( 'I', 'H')
     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
ORDER BY PRESTADOR.NM_MNEMONICO
                   ,PRESTADOR.NM_PRESTADOR
                   ,LEITO.DS_ENFERMARIA
                   ,LEITO.DS_LEITO
)
ORDER BY NM_PRESTADOR,CD_PRESTADOR,DS_ENFERMARIA,DS_LEITO
