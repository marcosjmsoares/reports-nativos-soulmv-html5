--R_ALTASMEDICO
SELECT * FROM
			  (SELECT pres.CD_PRESTADOR,
			          pres.NM_PRESTADOR,
			          aten.CD_ATENDIMENTO,
			          paci.NM_PACIENTE,
			          (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACI.CD_PACIENTE =  CD_PACIENTE) NM_SOCIAL_PACIENTE,
			          aten.DT_ATENDIMENTO,
			          aten.DT_ALTA,
			          aten.HR_ALTA,
			          conv.NM_CONVENIO,
			          copl.DS_CON_PLA
			   FROM DBAMV.ATENDIME aten,
			        DBAMV.PACIENTE paci,
			        DBAMV.CON_PLA copl,
			        DBAMV.CONVENIO conv,
			        DBAMV.PRESTADOR pres,
			        DBAMV.LEITO,
			        DBAMV.UNID_INT
			   WHERE aten.TP_ATENDIMENTO IN ('I', 'H')
			     AND aten.DT_ALTA IS NOT NULL
			     AND trunc(aten.DT_ALTA) BETWEEN trunc(to_date(:CAMPO_DATINI)) AND trunc(to_date(:CAMPO_DATFIM))
			     AND aten.CD_PACIENTE = paci.CD_PACIENTE
			     AND aten.CD_CONVENIO = copl.CD_CONVENIO(+)
			     AND aten.CD_CON_PLA = copl.CD_CON_PLA(+)
			     AND aten.CD_CONVENIO = conv.CD_CONVENIO
			     AND aten.CD_PRESTADOR = pres.CD_PRESTADOR
			     AND aten.CD_LEITO = leito.CD_LEITO(+)
			     AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)
			     AND ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
			     AND ATEN.CD_ATENDIMENTO_PAI IS NULL   
			   ORDER BY pres.NM_PRESTADOR, paci.NM_PACIENTE)
			ORDER BY NM_PRESTADOR, CD_PRESTADOR
