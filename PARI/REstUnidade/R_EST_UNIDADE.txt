SELECT UNID_INT.CD_UNID_INT          CD_UNID_INT
              ,UNID_INT.DS_UNID_INT          DS_UNID_INT
              ,COUNT(CD_ATENDIMENTO)    NR_ATENDIMENTO
FROM     DBAMV.ATENDIME                   ATENDIME
              ,DBAMV.LEITO                            LEITO
              ,DBAMV.UNID_INT                     UNID_INT
WHERE  ATENDIME.TP_ATENDIMENTO  =  'I'
AND        ATENDIME.CD_LEITO                  =  LEITO.CD_LEITO
AND        UNID_INT.CD_UNID_INT             =  LEITO.CD_UNID_INT
AND      ATENDIME.DT_ATENDIMENTO BETWEEN :CAMPO_DATAINI AND 
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY UNID_INT.DS_UNID_INT
                   ,UNID_INT.CD_UNID_INT
