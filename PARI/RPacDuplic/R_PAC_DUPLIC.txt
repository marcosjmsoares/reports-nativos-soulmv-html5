--R_PAC_DUPLIC
SELECT CD_PACIENTE Codigo ,
			       DS_CHECAPAC ,
			       NM_PACIENTE ,
			       NM_SOCIAL_PACIENTE ,
			       DT_NASCIMENTO ,
			       NM_MAE ,
			       NM_USUARIO
			FROM
			  (SELECT PACIENTE.CD_PACIENTE CD_PACIENTE ,
			          PACIENTE.DS_CHECAPAC DS_CHECAPAC ,
			          PACIENTE.NM_PACIENTE NM_PACIENTE ,
			      (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
			          PACIENTE.DT_NASCIMENTO DT_NASCIMENTO ,
			          PACIENTE.NM_MAE NM_MAE ,
			          PACIENTE.NM_USUARIO NM_USUARIO
			   FROM DBAMV.PACIENTE PACIENTE
			   WHERE EXISTS
			       (SELECT 'X'
			        FROM dbamv.paciente paci
			        WHERE PACIENTE.CD_PACIENTE <> PACI.CD_PACIENTE
			          AND Paciente.Nm_Paciente = Paci.Nm_Paciente
			          AND TRUNC(Paciente.Dt_Nascimento) = Trunc(Paci.Dt_Nascimento)
			          AND Paciente.NM_MAE = Paci.NM_MAE  :CAMPO_CF_NOME_SUB )  
			   ORDER BY nm_paciente,
			            DS_CHECAPAC)
			WHERE rownum <= :CAMPO_QUANTIDADE
