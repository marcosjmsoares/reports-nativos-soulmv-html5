--R_RESUMO_ENTRADA_SAIDA
SELECT * FROM (
Select cd_especialid
          ,ds_especialid
          ,data
          ,sum(qt_interna)  qt_interna
          ,sum(qt_alta)       qt_alta
          ,sum(qt_obito)     qt_obito
          ,sum(qt_trans)     qt_trans
from (
select especialid.cd_especialid
          ,decode(especialid.ds_especialid, null, 'NÃO INFORMADA', especialid.ds_especialid)  ds_especialid
          ,mov_int.dt_mov_int   data
          ,count(*)  qt_interna
          ,0         qt_alta
          ,0         qt_obito
          ,0         qt_trans
from   dbamv.atendime
         ,dbamv.especialid
         ,dbamv.unid_int
         ,dbamv.leito
         ,dbamv.mov_int
where atendime.cd_especialid = especialid.cd_especialid(+)
and     unid_int.sn_ativo = 'S'
and     leito.cd_unid_int = unid_int.cd_unid_int
and     leito.cd_leito = mov_int.cd_leito
and     mov_int.tp_mov = 'I'
and     atendime.cd_atendimento = mov_int.cd_atendimento
and     atendime.cd_atendimento_pai is null
and     trunc(mov_int.dt_mov_int) between :CAMPO_DT_INI and 
And atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
group by especialid.cd_especialid
              ,decode(especialid.ds_especialid, null, 'NÃO INFORMADA', especialid.ds_especialid)
              ,mov_int.dt_mov_int
Union All
select especialid.cd_especialid
          ,decode(especialid.ds_especialid, null, 'NÃO INFORMADA', especialid.ds_especialid)  ds_especialid
         ,atendime.dt_alta  data
          ,0        qt_interna
          ,count(*) qt_alta
          ,0        qt_obito
          ,0        qt_trans
from   dbamv.atendime
         ,dbamv.especialid
         ,dbamv.unid_int
         ,dbamv.leito
         ,dbamv.mot_alt
where atendime.tp_atendimento = 'I'
and     atendime.cd_atendimento_pai is null
and     atendime.cd_especialid = especialid.cd_especialid(+)
and     leito.cd_leito = atendime.cd_leito
and     unid_int.sn_ativo = 'S'
and     unid_int.cd_unid_int = leito.cd_unid_int
and     mot_alt.tp_mot_alta <> 'O'
and     mot_alt.cd_mot_alt = atendime.cd_mot_alt
and     trunc(atendime.dt_alta) between :CAMPO_DT_INI and 
And atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
group by especialid.cd_especialid
              ,decode(especialid.ds_especialid, null, 'NÃO INFORMADA', especialid.ds_especialid)
              ,atendime.dt_alta
Union All
select especialid.cd_especialid
          ,decode(especialid.ds_especialid, null, 'NÃO INFORMADA', especialid.ds_especialid)  ds_especialid
         ,atendime.dt_alta  data
         ,0         qt_interna
         ,0         qt_alta
         ,count(*)  qt_obito
         ,0         qt_trans
from   dbamv.atendime
         ,dbamv.especialid
         ,dbamv.unid_int
         ,dbamv.leito
         ,dbamv.mot_alt
where  atendime.tp_atendimento = 'I'
and     atendime.cd_atendimento_pai is null
and     atendime.cd_especialid = especialid.cd_especialid(+)
and     unid_int.sn_ativo = 'S'
and     leito.cd_leito = atendime.cd_leito
and     leito.cd_unid_int = unid_int.cd_unid_int
and     mot_alt.tp_mot_alta = 'O'
and     mot_alt.cd_mot_alt = atendime.cd_mot_alt
and     trunc(atendime.dt_alta) between :CAMPO_DT_INI and 
And atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
group by especialid.cd_especialid
              ,decode(especialid.ds_especialid, null, 'NÃO INFORMADA', especialid.ds_especialid)
              ,atendime.dt_alta
Union All
select especialid.cd_especialid
          ,decode(especialid.ds_especialid, null, 'NÃO INFORMADA', especialid.ds_especialid)  ds_especialid
         ,mov_int.dt_mov_int  data
         ,0            qt_interna
         ,0            qt_alta
         ,0            qt_obito
         ,count(*)  qt_trans
from   dbamv.atendime
         ,dbamv.especialid
         ,dbamv.leito
         ,dbamv.unid_int
         ,dbamv.mov_int
where atendime.cd_especialid = especialid.cd_especialid(+)
and     unid_int.sn_ativo = 'S'
and     atendime.cd_atendimento =  mov_int.cd_atendimento
and     mov_int.cd_leito = leito.cd_leito
and     leito.cd_unid_int = unid_int.cd_unid_int
and     trunc(mov_int.dt_mov_int) between :CAMPO_DT_INI and 
and     mov_int.tp_mov = 'O'
and     atendime.cd_atendimento = mov_int.cd_atendimento
And atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
group by especialid.cd_especialid
              ,decode(especialid.ds_especialid, null, 'NÃO INFORMADA', especialid.ds_especialid)
              ,mov_int.dt_mov_int     )
group by cd_especialid
              ,ds_especialid
              ,data
)
ORDER BY CD_ESPECIALID,DS_ESPECIALID, DATA
