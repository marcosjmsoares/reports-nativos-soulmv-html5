SELECT aten.CD_ATENDIMENTO,
       paci.CD_PACIENTE,
       paci.NM_PACIENTE,
       nvl(paci.NM_SOCIAL_PACIENTE, 'Nao Atribuido') NM_SOCIAL_PACIENTE,
       paci.NR_IDENTIDADE NR_DOCUMENTO,
       paci.DS_OM_IDENTIDADE,
       paci.NR_CPF,
       paci.DS_ENDERECO,
       paci.NR_ENDERECO,
       paci.NM_BAIRRO,
       paci.NR_CEP,
       paci.NR_FONE,
       cida.NM_CIDADE,
       cida.CD_UF,
       conv.NM_CONVENIO,
       (SELECT cidade.nm_cidade
                FROM dbamv.cidade, dbamv.hospital
                WHERE cidade.cd_cidade = hospital.cd_cidade
                AND hospital.cd_multi_empresa = 1)  NM_CIDADE_HOSPITAL 
FROM   DBAMV.ATENDIME aten,
       DBAMV.PACIENTE paci,
       DBAMV.CIDADE cida,
       DBAMV.CONVENIO conv
,Dbamv.Empresa_Convenio
Where Empresa_Convenio.Cd_Convenio = Conv.Cd_Convenio
And Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
And aten.CD_ATENDIMENTO = :CAMPO_ATEND
  AND  aten.CD_PACIENTE    = paci.CD_PACIENTE
  AND  paci.CD_CIDADE      = cida.CD_CIDADE
  AND  aten.CD_CONVENIO    = conv.CD_CONVENIO
