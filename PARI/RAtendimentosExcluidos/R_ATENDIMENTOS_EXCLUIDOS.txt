--R_ATENDIMENTOS_EXCLUIDOS
SELECT EXCLU.CD_ATENDIMENTO CD_ATENDIMENTO
              ,EXCLU.TP_ATENDIMENTO TP_ATENDIMENTO
              ,to_date(to_char(EXCLU.DT_LOG_EXCL_ATENDIME, 'dd/mm/yyyy')  || to_char(EXCLU.HR_LOG_EXCL_ATENDIME,'HH24') , 'dd/mm/yyyyhh24') DT_LOG_EXCL_ATENDIME
              ,EXCLU.DS_MOTIVO  DS_MOTIVO
              ,EXCLU.NM_USUARIO    NM_USUARIO
              ,PACIENTE.NM_PACIENTE  NM_PACIENTE
              ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE = P.CD_PACIENTE) NM_SOCIAL_PACIENTE
              ,EXCLU.DT_ATENDIMENTO  DT_ATENDIMENTO
              ,EXCLU.CD_MULTI_EMPRESA
FROM     DBAMV.LOG_EXCL_ATENDIME  EXCLU
              ,DBAMV.PACIENTE
WHERE  EXCLU.CD_PACIENTE = PACIENTE.CD_PACIENTE
AND        TRUNC(EXCLU.DT_LOG_EXCL_ATENDIME)  BETWEEN :CAMPO_DT_INICIO  AND 
AND exclu.cd_multi_empresa = dbamv.pkg_mv2000.le_empresa
AND exclu.cd_atendimento NOT IN (SELECT cd_atendimento FROM dbamv.atendime)
ORDER BY  EXCLU.CD_ATENDIMENTO
