--R_TERMO_ATENDIMENTO
SELECT PACIENTE.CD_PACIENTE
       ,PACIENTE.NM_PACIENTE
       ,PACIENTE.DT_NASCIMENTO
       ,PACIENTE.TP_SEXO
       ,PACIENTE.TP_COR
       ,DBAMV.FNC_ESTADO_CIVIL(PACIENTE.TP_ESTADO_CIVIL) TP_ESTADO_CIVIL1
       ,PACIENTE.NR_DOCUMENTO
       ,PACIENTE.NR_IDENTIDADE
       ,PACIENTE.CD_NATURALIDADE
       ,PACIENTE.DS_OM_IDENTIDADE
       ,PACIENTE.NM_MAE
       ,Nvl(PACIENTE.NM_PAI,' ') NM_PAI
       ,PACIENTE.DS_ENDERECO
       ,PACIENTE.NR_ENDERECO
       ,PACIENTE.DS_COMPLEMENTO
       ,PACIENTE.NR_CPF
       ,PACIENTE.NM_BAIRRO
       ,PACIENTE.NR_CEP
       ,PACIENTE.NR_FONE
       ,PACIENTE.DT_CADASTRO
       ,PACIENTE.NM_CONJUGE
       ,ATENDIME.CD_ATENDIMENTO
       ,ATENDIME.CD_PACIENTE
       ,ATENDIME.DT_ATENDIMENTO
       ,ATENDIME.HR_ATENDIMENTO
       ,ATENDIME.DT_ALTA
       ,ATENDIME.CD_SERVICO
       ,ATENDIME.CD_SER_DIS
       ,ATENDIME.TP_ATENDIMENTO
       ,ATENDIME.NM_USUARIO
       ,ATENDIME.CD_CONVENIO
       ,ORI_ATE.DS_ORI_ATE
       ,DECODE(
        	CONVENIO.TP_CONVENIO, 'C',
          ATENDIME.CD_PRO_INT,  'P',
          ATENDIME.CD_PRO_INT,  'A',
          PRO_FAT.CD_SUS,       'H',
        	PRO_FAT.CD_SUS
       ) CD_PRO_INT
       ,PRO_FAT.DS_PRO_FAT DS_PRO_INT
       ,PRESTADOR.NM_PRESTADOR
       ,PRESTADOR.NR_CPF_CGC
       ,Nvl(PRESTADOR.DS_CODIGO_CONSELHO,' ') DS_CODIGO_CONSELHO
       ,LEITO.DS_ENFERMARIA
       ,LEITO.DS_LEITO
       ,UNID_INT.CD_UNID_INT
       ,nvl(UNID_INT.DS_UNID_INT,' ') DS_UNID_INT
       ,UNID_INT.DS_LOCALIZACAO
       ,CONVENIO.NM_CONVENIO
       ,CID.DS_CID
       ,SERVICO.DS_SERVICO
       ,SERVICO.TP_ANAMNESE TP_ANAM_SERVICO
       ,SER_DIS.DS_SER_DIS
       ,SER_DIS.TP_ANAMNESE TP_ANAM_SER_DIS
       ,TIP_ACOM.DS_TIP_ACOM
       ,RESPONSA.NM_RESPONSAVEL
       ,Nvl(RESPONSA.NR_FONE,' ') NR_FONE_RESPONSAVEL
       ,RESPONSA.DS_DOCUMENTO DS_DOCUMENTO_RESPONSAVEL
       ,RESPONSA.DS_ENDERECO DS_ENDERECO_RESPONSAVEL
       ,Nvl(RESPONSA.NR_CPF,' ') NR_CPF_RESP
       ,RESPONSA.NR_CEP NR_CEP_RESP
       ,RESPONSA.NM_BAIRRO NM_BAIRRO_RESP
       ,RESPONSA.NR_ENDERECO NR_ENDERECO_RESP
       ,RESPONSA.DS_COMPLEMENTO DS_COMPLEMENTO_RESP
       ,RESPONSA.TP_ESTADO_CIVIL TP_ESTADO_CIVIL_RESP
       ,RESPONSA.DS_CONTATO_FAMILIA
       ,CID_RESP.NM_CIDADE NM_CIDADE_RESP
       ,PROF_RESP.NM_PROFISSAO NM_PROFISSAO_RESP
       ,Nvl(TIP_PAREN.DS_TIP_PAREN,' ') DS_TIP_PAREN_RESP
       ,CON_PLA.CD_CON_PLA
       ,CON_PLA.DS_CON_PLA
       ,CIDADE.NM_CIDADE
       ,CIDADE.CD_UF
       ,CON_PLA.DS_CON_PLA
       ,PROFISSAO.NM_PROFISSAO
       ,atendime_info.ds_info_atendimento
       ,SUB_PLANO.DS_SUB_PLANO
  FROM DBAMV.PACIENTE,
       DBAMV.CIDADE,
       DBAMV.ATENDIME,
       DBAMV.ORI_ATE,
       DBAMV.PRO_FAT,
       DBAMV.PRESTADOR,
       DBAMV.LEITO,
       DBAMV.UNID_INT ,
       DBAMV.CONVENIO,
       DBAMV.CID,
       DBAMV.SERVICO ,
       DBAMV.SER_DIS,
       DBAMV.TIP_ACOM ,
       DBAMV.RESPONSA,
       DBAMV.CIDADE CID_RESP,
       DBAMV.CON_PLA,
       DBAMV.PROFISSAO,
       DBAMV.PROFISSAO PROF_RESP,
       DBAMV.TIP_PAREN,
       dbamv.atendime_info,
       DBAMV.SUB_PLANO
WHERE	ATENDIME.CD_ATENDIMENTO = TO_CHAR( :CAMPO_PCD_ATENDIMENTO	)
	AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
	AND PACIENTE.CD_CIDADE = CIDADE.CD_CIDADE(+)
	AND PACIENTE.CD_PROFISSAO = PROFISSAO.CD_PROFISSAO(+)
	AND ATENDIME.CD_ORI_ATE = ORI_ATE.CD_ORI_ATE(+)
	AND ATENDIME.CD_PRO_INT = PRO_FAT.CD_PRO_FAT(+)
	AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
	AND ATENDIME.CD_LEITO = LEITO.CD_LEITO (+)
	AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT (+)
	AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
  AND	ATENDIME.CD_CID = CID.CD_CID (+)
	AND ATENDIME.CD_ATENDIMENTO = atendime_info.cd_atendimento(+)
	AND ATENDIME.CD_SERVICO = SERVICO.CD_SERVICO (+)
	AND ATENDIME.CD_SER_DIS = SER_DIS.CD_SER_DIS (+)
	AND ATENDIME.CD_TIP_ACOM = TIP_ACOM.CD_TIP_ACOM (+)
	AND ATENDIME.CD_ATENDIMENTO = RESPONSA.CD_ATENDIMENTO (+)
	AND RESPONSA.CD_CIDADE = CID_RESP.CD_CIDADE(+)
	AND ATENDIME.CD_CONVENIO = CON_PLA.CD_CONVENIO(+)
	AND ATENDIME.CD_CON_PLA = CON_PLA.CD_CON_PLA(+)
	AND RESPONSA.CD_PROFISSAO = PROF_RESP.CD_PROFISSAO(+)
	AND RESPONSA.CD_TIP_PAREN = TIP_PAREN.CD_TIP_PAREN(+)
	AND atendime.cd_convenio = sub_plano.cd_convenio(+)
	AND atendime.cd_con_pla = sub_plano.cd_con_pla(+)
	AND atendime.cd_sub_plano = sub_plano.cd_sub_plano(+)
               AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
