--R_EST_USUARIO
SELECT  COUNT(ATENDIME.CD_ATENDIMENTO)       COUNT_USUARIO
               ,ATENDIME.NM_USUARIO||' - '||u.nm_usuario                               NM_USUARIO
FROM    Dbamv.ATENDIME ATENDIME,
        DBASGU.USUARIOS U
WHERE  TRUNC(ATENDIME.DT_ATENDIMENTO)  BETWEEN  :CAMPO_DT_INIC  AND 
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND ATENDIME.NM_USUARIO = U.CD_USUARIO
GROUP BY ATENDIME.NM_USUARIO, u.nm_usuario
ORDER BY ATENDIME.NM_USUARIO
