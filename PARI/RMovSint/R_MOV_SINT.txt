--R_MOV_SINT
SELECT * FROM (
select   trunc(mov_int.dt_mov_int)   DT_ATENDIMENTO_T
           ,'1'                                                        ordem
           ,'Transf. Internas'                                   TIPO
           ,count(*)                                            QTD_TRANSFERENCIA
           ,(select sum(qtd_transferencia) from(
           select   trunc(mov_int.dt_mov_int)   DT_ATENDIMENTO_T
           ,'1'                                                        ordem
           ,'Transf. Internas'                                   TIPO
           ,count(*)                                            QTD_TRANSFERENCIA
from   dbamv.atendime
         ,dbamv.especialid
         ,dbamv.leito
         ,dbamv.unid_int
         ,dbamv.mov_int
where atendime.cd_especialid = especialid.cd_especialid(+)
and     atendime.cd_atendimento =  mov_int.cd_atendimento
and     mov_int.cd_leito = leito.cd_leito
and     leito.cd_unid_int = unid_int.cd_unid_int
and     trunc(mov_int.dt_mov_int) BETWEEN  :CAMPO_DT_INICIAL  AND  
and     mov_int.tp_mov = 'O'
and     atendime.cd_atendimento = mov_int.cd_atendimento
and atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
group by trunc(mov_int.dt_mov_int)
)) total_transferencia
from   dbamv.atendime
         ,dbamv.especialid
         ,dbamv.leito
         ,dbamv.unid_int
         ,dbamv.mov_int
where atendime.cd_especialid = especialid.cd_especialid(+)
and     atendime.cd_atendimento =  mov_int.cd_atendimento
and     mov_int.cd_leito = leito.cd_leito
and     leito.cd_unid_int = unid_int.cd_unid_int
and     trunc(mov_int.dt_mov_int) BETWEEN  :CAMPO_DT_INICIAL  AND  
and     mov_int.tp_mov = 'O'
and     atendime.cd_atendimento = mov_int.cd_atendimento
and atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
group by trunc(mov_int.dt_mov_int)
UNION
SELECT  TRUNC(MOV_INT.DT_MOV_INT)   DT_ATENDIMENTO_I
               ,'2' ordem
               ,'Internações'                                             TIPO
               ,count(*)                                                     QTD_INTERNACOES
               ,(select sum(qtd_internacoes) from (
               SELECT  TRUNC(MOV_INT.DT_MOV_INT)   DT_ATENDIMENTO_I
               ,'2' ordem
               ,'Internações'                                             TIPO
               ,count(*)                                                     QTD_INTERNACOES
FROM   DBAMV.ATENDIME
            ,DBAMV.MOV_INT
           ,DBAMV.UNID_INT
           ,DBAMV.LEITO
WHERE  MOV_INT.TP_MOV =  'I'
AND       MOV_INT.CD_LEITO  =  LEITO.CD_LEITO
AND       LEITO.CD_UNID_INT  =  UNID_INT.CD_UNID_INT
AND        TRUNC(NVL(MOV_INT.DT_LIB_MOV,SYSDATE))  >=  TRUNC(MOV_INT.DT_MOV_INT)
AND        ATENDIME.TP_ATENDIMENTO =  'I'
AND        MOV_INT.CD_ATENDIMENTO  =  ATENDIME.CD_ATENDIMENTO
AND        TRUNC(MOV_INT.DT_MOV_INT)  BETWEEN  :CAMPO_DT_INICIAL  AND  
and atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
GROUP BY  TRUNC(MOV_INT.DT_MOV_INT)
)) total_internacoes
FROM   DBAMV.ATENDIME
            ,DBAMV.MOV_INT
           ,DBAMV.UNID_INT
           ,DBAMV.LEITO
WHERE  MOV_INT.TP_MOV =  'I'
AND       MOV_INT.CD_LEITO  =  LEITO.CD_LEITO
AND       LEITO.CD_UNID_INT  =  UNID_INT.CD_UNID_INT
AND        TRUNC(NVL(MOV_INT.DT_LIB_MOV,SYSDATE))  >=  TRUNC(MOV_INT.DT_MOV_INT)
AND        ATENDIME.TP_ATENDIMENTO =  'I'
AND        MOV_INT.CD_ATENDIMENTO  =  ATENDIME.CD_ATENDIMENTO
AND        TRUNC(MOV_INT.DT_MOV_INT)  BETWEEN  :CAMPO_DT_INICIAL  AND  
and atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
GROUP BY  TRUNC(MOV_INT.DT_MOV_INT)
UNION
SELECT  TRUNC(ATENDIME.DT_ALTA)   DT_ALTA_O
               ,'3'                                                  ordem
               ,'Altas'                                            TIPO
               ,COUNT(*)                                     QTD_OBITO
               ,(select sum(qtd_obito) from (
               SELECT  TRUNC(ATENDIME.DT_ALTA)   DT_ALTA_O
               ,'3'                                                  ordem
               ,'Altas'                                            TIPO
               ,COUNT(*)                                     QTD_OBITO
FROM  dbamv.ATENDIME
           ,dbamv.MOT_ALT
           ,dbamv.LEITO
           ,dbamv.UNID_INT
WHERE ATENDIME.TP_ATENDIMENTO =  'I'
AND       TRUNC(ATENDIME.DT_ALTA)  BETWEEN  :CAMPO_DT_INICIAL  AND  
AND       ATENDIME.CD_MOT_ALT = MOT_ALT.CD_MOT_ALT
AND       MOT_ALT.TP_MOT_ALTA <> 'O'
AND       ATENDIME.CD_LEITO  =  LEITO.CD_LEITO
AND       LEITO.CD_UNID_INT  =  UNID_INT.CD_UNID_INT
GROUP BY  TRUNC(ATENDIME.DT_ALTA)
)) total_obito
FROM  dbamv.ATENDIME
           ,dbamv.MOT_ALT
           ,dbamv.LEITO
           ,dbamv.UNID_INT
WHERE ATENDIME.TP_ATENDIMENTO =  'I'
AND       TRUNC(ATENDIME.DT_ALTA)  BETWEEN  :CAMPO_DT_INICIAL  AND  
AND       ATENDIME.CD_MOT_ALT = MOT_ALT.CD_MOT_ALT
AND       MOT_ALT.TP_MOT_ALTA <> 'O'
AND       ATENDIME.CD_LEITO  =  LEITO.CD_LEITO
AND       LEITO.CD_UNID_INT  =  UNID_INT.CD_UNID_INT
GROUP BY  TRUNC(ATENDIME.DT_ALTA)
UNION
SELECT  TRUNC(ATENDIME.DT_ALTA)        DT_ALTA_O
               ,'4'                                                       ordem
               ,'Óbitos'                                               TIPO
               ,COUNT(*)                                          QTD_OBITO
               ,(select sum(qtd_obito) from (
SELECT  TRUNC(ATENDIME.DT_ALTA)        DT_ALTA_O
               ,'4'                                                       ordem
               ,'Óbitos'                                               TIPO
               ,COUNT(*)                                          QTD_OBITO
FROM  dbamv.ATENDIME
           ,dbamv.MOT_ALT
           ,dbamv.LEITO
           ,dbamv.UNID_INT
WHERE  ATENDIME.TP_ATENDIMENTO =  'I'
AND        TRUNC(ATENDIME.DT_ALTA)  BETWEEN  :CAMPO_DT_INICIAL  AND  
AND         MOT_ALT.TP_MOT_ALTA = 'O'
AND         ATENDIME.CD_MOT_ALT = MOT_ALT.CD_MOT_ALT
AND       ATENDIME.CD_LEITO  =  LEITO.CD_LEITO
AND       LEITO.CD_UNID_INT  =  UNID_INT.CD_UNID_INT
and atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
GROUP BY  TRUNC(ATENDIME.DT_ALTA))) total_obito2
FROM  dbamv.ATENDIME
           ,dbamv.MOT_ALT
           ,dbamv.LEITO
           ,dbamv.UNID_INT
WHERE  ATENDIME.TP_ATENDIMENTO =  'I'
AND        TRUNC(ATENDIME.DT_ALTA)  BETWEEN  :CAMPO_DT_INICIAL  AND  
AND         MOT_ALT.TP_MOT_ALTA = 'O'
AND         ATENDIME.CD_MOT_ALT = MOT_ALT.CD_MOT_ALT
AND       ATENDIME.CD_LEITO  =  LEITO.CD_LEITO
AND       LEITO.CD_UNID_INT  =  UNID_INT.CD_UNID_INT
and atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
GROUP BY  TRUNC(ATENDIME.DT_ALTA)
)
ORDER BY DT_ATENDIMENTO_T,
ORDEM,TIPO
