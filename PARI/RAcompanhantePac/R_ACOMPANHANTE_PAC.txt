--R_ACOMPANHANTE_PAC
SELECT * FROM (
SELECT UNID_INT.CD_UNID_INT
      ,UNID_INT.DS_UNID_INT
      ,LEITO.CD_LEITO
      ,LEITO.DS_LEITO
      ,PACIENTE.NM_PACIENTE
      ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE
      ,LEITO.DS_ENFERMARIA
      ,LEITO.NR_RAMAL
      ,TIP_ACOM.DS_TIP_ACOM
FROM DBAMV.ATENDIME
    ,DBAMV.UNID_INT
    ,DBAMV.TIP_ACOM
    ,DBAMV.LEITO
    ,DBAMV.PACIENTE
WHERE ATENDIME.CD_LEITO = LEITO.CD_LEITO
  AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT
  AND LEITO.CD_TIP_ACOM = TIP_ACOM.CD_TIP_ACOM
  AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
  AND ATENDIME.SN_ACOMPANHANTE = 'S'
  AND ATENDIME.DT_ALTA IS NULL
  AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
ORDER BY LEITO.DS_LEITO
)
ORDER BY CD_UNID_INT,DS_UNID_INT
