--R_OPE_ALTA_CIDADE
SELECT * FROM
			  (SELECT CIDADE.NM_CIDADE NM_CIDADE ,
			          PACIENTE.CD_PACIENTE CD_PACIENTE ,
			          PACIENTE.NM_PACIENTE NM_PACIENTE ,
			      (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
			          TO_CHAR(ATENDIME.DT_ATENDIMENTO, 'DD/MM/YYYY') || ' ' || TO_CHAR(ATENDIME.HR_ATENDIMENTO, 'HH24') DT_ATENDIMENTO ,
			          TO_CHAR(ATENDIME.DT_ALTA, 'DD/MM/YYYY') || ' ' || TO_CHAR(ATENDIME.HR_ALTA, 'HH24') DT_ALTA ,
			          PRESTADOR.NM_PRESTADOR NM_PRESTADOR ,
			          CONVENIO.NM_CONVENIO NM_CONVENIO
			   FROM DBAMV.ATENDIME ,
			        DBAMV.PACIENTE ,
			        DBAMV.PRESTADOR ,
			        DBAMV.CIDADE ,
			        DBAMV.CONVENIO
			   WHERE ATENDIME.TP_ATENDIMENTO IN ('I',
			                                     'H')
			     AND TRUNC(ATENDIME.DT_ALTA) BETWEEN :CAMPO_DT_INIC AND 
			     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
			     AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
			     AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
			     AND PACIENTE.CD_CIDADE = CIDADE.CD_CIDADE(+)
			     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA 
			     AND ATENDIME.CD_ATENDIMENTO_PAI IS NULL  
			   ORDER BY CIDADE.NM_CIDADE)
			ORDER BY NM_CIDADE
