--R_ATEND_CONV_PLANO
SELECT * FROM
  (SELECT CONVENIO.NM_CONVENIO ,
          CON_PLA.CD_CON_PLA ,
          CON_PLA.DS_CON_PLA ,
          PACIENTE.CD_PACIENTE ,
          PACIENTE.NM_PACIENTE ,
          (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
          LEITO.DS_ENFERMARIA ,
          LEITO.DS_RESUMO ,
          PRESTADOR.NM_PRESTADOR ,
          GUIA.NR_GUIA ,
          GUIA.DT_VALIDADE ,
          GUIA.CD_SENHA ,
          ATENDIME.DT_ATENDIMENTO ,
          ATENDIME.HR_ATENDIMENTO ,
          ATENDIME.CD_ATENDIMENTO ,
          ATENDIME.CD_CONVENIO
   FROM DBAMV.CONVENIO ,
        DBAMV.ATENDIME ,
        DBAMV.PACIENTE ,
        DBAMV.LEITO ,
        DBAMV.PRESTADOR ,
        DBAMV.GUIA ,
        DBAMV.CON_PLA
   WHERE ATENDIME.TP_ATENDIMENTO IN ('I', 'H')
     AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
     AND ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
     AND ATENDIME.CD_ATENDIMENTO = GUIA.CD_ATENDIMENTO(+)
     AND ATENDIME.CD_CONVENIO = GUIA.CD_CONVENIO(+)
     AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
     AND ATENDIME.CD_CONVENIO = CON_PLA.CD_CONVENIO
     AND ATENDIME.CD_CON_PLA = CON_PLA.CD_CON_PLA
     AND GUIA.TP_GUIA(+) = 'I'
     AND ATENDIME.CD_ATENDIMENTO_PAI IS NULL
     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA 
   ORDER BY CONVENIO.NM_CONVENIO ,
            CON_PLA.DS_CON_PLA ,
            PACIENTE.NM_PACIENTE)
ORDER BY NM_CONVENIO,
         CD_CONVENIO,
         CD_CON_PLA,
         DS_CON_PLA,
         NM_PACIENTE
