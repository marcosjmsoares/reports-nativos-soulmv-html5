--R_ETIQUE_PACI
SELECT PACIENTE.CD_PACIENTE                                                        CD_PACIENTE,
               PACIENTE.NM_PACIENTE                                                       NM_PACIENTE,
               PACIENTE.TP_SEXO                                                                TP_SEXO,
               PACIENTE.DS_ENDERECO                                                      DS_ENDERECO,
               PACIENTE.NM_MAE                                                                  NM_MAE,
               PACIENTE.NM_PAI                                                                    NM_PAI,
               PACIENTE.NR_FONE                                                                NR_FONE,
               TO_CHAR(PACIENTE.DT_NASCIMENTO, 'DD/MM/YYYY')   DT_NASCIMENTO,
               TO_CHAR(PACIENTE.DT_CADASTRO, 'DD/MM/YYYY')       DT_CADASTRO
FROM     DBAMV.PACIENTE
WHERE PACIENTE.CD_PACIENTE =  :CAMPO_CD_PACIENTE
