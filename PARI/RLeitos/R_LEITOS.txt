--R_LEITOS
SELECT  SETOR.CD_SETOR                CD_SETOR
              ,SETOR.NM_SETOR                NM_SETOR
              ,UNID_INT.CD_UNID_INT       CD_UNID_INT
              ,UNID_INT.DS_UNID_INT       DS_UNID_INT
              ,LEITO.DS_ENFERMARIA       DS_ENFERMARIA
              ,LEITO.CD_LEITO                    Codigo
              ,LEITO.DS_LEITO                    Descricao
              ,LEITO.TP_SEXO                    TP_SEXO
              ,LEITO.TP_OCUPACAO          TP_OCUPACAO
FROM    DBAMV.LEITO                         LEITO
             ,DBAMV.UNID_INT                   UNID_INT
             ,DBAMV.SETOR                       SETOR
WHERE  SETOR.CD_SETOR    =  UNID_INT.CD_SETOR
AND        LEITO.CD_UNID_INT     =  UNID_INT.CD_UNID_INT
AND        UNID_INT.CD_UNID_INT LIKE :CAMPO_UNID_INT
AND       UNID_INT.SN_ATIVO = 'S'
AND       LEITO.DT_DESATIVACAO IS NULL
order by  CD_UNID_INT
