--R_EST_PERM_CID
SELECT CID.CD_CID
              ,CID.DS_CID
              ,count(atendime.cd_atendimento)    SUM_CID 
              ,ROUND(AVG(NVL(TRUNC(ATENDIME.DT_ALTA), TRUNC(SYSDATE)) - TRUNC(ATENDIME.DT_ATENDIMENTO)))     MEDIA_PERMAN
FROM   DBAMV.ATENDIME
             ,DBAMV.CID
WHERE  TRUNC(ATENDIME.DT_ATENDIMENTO) BETWEEN :CAMPO_DATA_INI AND 
AND        ATENDIME.TP_ATENDIMENTO = 'I'
AND        ATENDIME.CD_ATENDIMENTO_PAI IS NULL
AND        ATENDIME.CD_CID  =  CID.CD_CID
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY CID.CD_CID
                   ,CID.DS_CID
ORDER BY MEDIA_PERMAN DESC
