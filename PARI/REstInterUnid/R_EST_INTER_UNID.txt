--R_EST_INTER_UNID_RF_ATEND_Q_ATENDIMENTOS
SELECT CD_UNID_INT
              ,NVL(DS_UNID_INT,'HOME CARE') DS_UNID_INT
              ,SUM(INTERNACOES)   Internacoes
FROM (  SELECT UNID_INT.CD_UNID_INT
                            ,UNID_INT.DS_UNID_INT
                            ,0            Internacoes
               FROM DBAMV.UNID_INT
               Where 1 = 1
               UNION ALL
                SELECT UNID_INT.CD_UNID_INT
                             ,UNID_INT.DS_UNID_INT
                             ,COUNT(*)     Internacoes
               FROM    DBAMV.MOV_INT
                             ,DBAMV.LEITO
                             ,DBAMV.UNID_INT
                             ,Dbamv.ATENDIME
                             ,Dbamv.Convenio
               WHERE  ATENDIME.TP_ATENDIMENTO  IN ('I','H')
               AND        MOV_INT.TP_MOV = 'I'
               AND        TRUNC(MOV_INT.DT_MOV_INT)  BETWEEN :CAMPO_DATINI  AND  
               AND        MOV_INT.CD_LEITO  =  LEITO.CD_LEITO(+)
               AND        MOV_INT.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO(+)
			   AND        LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)
               AND        ATENDIME.CD_ATENDIMENTO_PAI IS NULL
               AND        ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
              AND        ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
               GROUP  BY UNID_INT.CD_UNID_INT,
                                    UNID_INT.DS_UNID_INT )
GROUP BY  CD_UNID_INT
                    ,DS_UNID_INT
ORDER  BY DS_UNID_INT
