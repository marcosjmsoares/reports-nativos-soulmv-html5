--R_INTERNACOES_ESQUEMA
SELECT * FROM
  (SELECT DISTINCT 'VISITA MÉDICA' DS_TIP_ESQ , 
                                   UNID_INT.CD_UNID_INT ,
                                   Nvl(nullif(trunc(sysdate) - trunc(ATENDIME.DT_ATENDIMENTO), 0), 1) QT_DIAS ,
                                   TO_DATE(TO_CHAR(ATENDIME.DT_ATENDIMENTO, 'dd/MM/yyyy') || ' ' || to_char(ATENDIME.hr_atendimento, 'HH24'), 'dd/MM/yyyy HH24') DT_HR_ATENDIMENTO ,
                                   UNID_INT.DS_UNID_INT ,
                                   ATENDIME.CD_ATENDIMENTO ,
                                   ATENDIME.CD_PACIENTE ,
          (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
                                   PACIENTE.NM_PACIENTE ,
                                   TO_DATE(TO_CHAR(ATENDIME.DT_ATENDIMENTO, 'dd/MM/yyyy'), 'dd/MM/yyyy') DT_ATENDIMENTO ,
                                   CONVENIO.NM_CONVENIO ,
                                   PRESTADOR.NM_PRESTADOR ,
                                   PACIENTE.NR_IDENTIDADE ,
                                   PACK_INTERNAMENTO.RETORNA_IDADE(PACIENTE.DT_NASCIMENTO, SYSDATE) NR_IDADE ,
                                   TIPO_INTERNACAO.DS_TIPO_INTERNACAO ,
                                   leito.ds_resumo 
FROM DBAMV.ATENDIME ,
     DBAMV.PACIENTE ,
     DBAMV.PRESTADOR ,
     DBAMV.CONVENIO ,
     DBAMV.TIPO_INTERNACAO ,
     DBAMV.LEITO ,
     DBAMV.UNID_INT
   WHERE ATENDIME.TP_ATENDIMENTO IN ('I', 'H')
     AND ATENDIME.DT_ALTA IS NULL
     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
     AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
     AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
     AND TIPO_INTERNACAO.CD_TIPO_INTERNACAO = ATENDIME.CD_TIPO_INTERNACAO
     AND ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
     AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)
     AND ATENDIME.CD_ATENDIMENTO_PAI IS NULL
     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
   UNION ALL 
   SELECT DISTINCT TIP_ESQ.DS_TIP_ESQ ,
                 UNID_INT.CD_UNID_INT ,
                 Nvl(nullif(trunc(sysdate) - trunc(ATENDIME.DT_ATENDIMENTO), 0), 1) QT_DIAS ,
                 TO_DATE(TO_CHAR(ATENDIME.DT_ATENDIMENTO, 'dd/MM/yyyy') || ' ' || to_char(ATENDIME.hr_atendimento, 'HH24'), 'dd/MM/yyyy HH24') DT_HR_ATENDIMENTO ,
                 UNID_INT.DS_UNID_INT ,
                 ATENDIME.CD_ATENDIMENTO ,
                 ATENDIME.CD_PACIENTE ,
                 PACIENTE.NM_PACIENTE ,
           (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
                 TO_DATE(To_Char(ATENDIME.DT_ATENDIMENTO, 'dd/MM/yyyy') || TO_CHAR(ATENDIME.HR_ATENDIMENTO, 'HH24'), 'dd/MM/yyyy hh24') DT_ATENDIMENTO ,
                 CONVENIO.NM_CONVENIO ,
                 PRESTADOR.NM_PRESTADOR ,
                 PACIENTE.NR_IDENTIDADE ,
                 PACK_INTERNAMENTO.RETORNA_IDADE(PACIENTE.DT_NASCIMENTO, SYSDATE) NR_IDADE ,
                 TIPO_INTERNACAO.DS_TIPO_INTERNACAO ,
                 leito.ds_resumo 
FROM DBAMV.ATENDIME ,
     DBAMV.PACIENTE ,
     DBAMV.PRESTADOR ,
     DBAMV.CONVENIO ,
     DBAMV.TIPO_INTERNACAO ,
     DBAMV.LEITO ,
     DBAMV.UNID_INT ,
     DBAMV.PRE_MED ,
     DBAMV.ITPRE_MED ,
     DBAMV.TIP_PRESC ,
     DBAMV.TIP_ESQ
   WHERE ATENDIME.TP_ATENDIMENTO IN ('I', 'H')
     AND TRUNC(PRE_MED.DT_PRE_MED) BETWEEN TO_DATE(:CAMPO_DATINI, 'dd/MM/yyyy') AND TO_DATE(:CAMPO_DATFIM, 'dd/MM/yyyy')
     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
     AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
     AND TIPO_INTERNACAO.CD_TIPO_INTERNACAO = ATENDIME.CD_TIPO_INTERNACAO
     AND ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
     AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)
     AND ATENDIME.CD_ATENDIMENTO_PAI IS NULL
     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
     AND PRE_MED.CD_PRE_MED = ITPRE_MED.CD_PRE_MED
     AND ATENDIME.CD_ATENDIMENTO = PRE_MED.CD_ATENDIMENTO
     AND TIP_PRESC.CD_TIP_PRESC = ITPRE_MED.CD_TIP_PRESC
     AND TIP_PRESC.CD_TIP_ESQ = ITPRE_MED.CD_TIP_ESQ
     AND ITPRE_MED.CD_TIP_ESQ = TIP_ESQ.CD_TIP_ESQ
     AND PRESTADOR.CD_PRESTADOR = NVL(ITPRE_MED.CD_PRESTADOR, 0) 
   ORDER BY 1,
            2,
            13)
ORDER BY DS_TIP_ESQ,
         CD_UNID_INT,
         DS_UNID_INT,
         NM_PACIENTE
