--R_LISTA_ATEND_PREV_ALTA
SELECT DISTINCT Atendime.Cd_Atendimento ATENDIMENTO,
			                Atendime.Cd_Paciente PACIENTE,
			                to_char(Atendime.Dt_Atendimento, 'dd/mm/yyyy') DT_ATENDIMENTO,
			                to_char(Atendime.Dt_Prevista_Alta, 'dd/mm/yyyy') DT_PREVISTA_ALTA,
			                Atendime.Dt_Prevista_Alta,
							(CASE NVL(SN_UTILIZA_NOME_SOCIAL, 'N')
								WHEN 'S' THEN Paciente.NM_SOCIAL_PACIENTE
								ELSE Paciente.nm_paciente 
							END) nm_paciente, 
					  		Leito.Ds_Resumo LEITO,
			                Convenio.Cd_Convenio CD_CONVENIO,
			                Convenio.Nm_Convenio CONVENIO,
			                Convenio.Tp_Convenio TP_CONVENIO,
			                Prestador.Nm_Prestador PRESTADOR,
			                TRUNC(SYSDATE - DT_ATENDIMENTO) NR_DIAS
			FROM Dbamv.Atendime,
			     Dbamv.Paciente,
			     Dbamv.Leito,
			     Dbamv.Convenio,
			     Dbamv.Prestador,
			     Dbamv.Unid_Int
			WHERE Atendime.Cd_Paciente = Paciente.Cd_Paciente
			  AND Atendime.Cd_Leito = Leito.Cd_Leito
			  AND Atendime.Cd_Convenio = Convenio.Cd_Convenio
			  AND Atendime.Cd_Prestador = Prestador.Cd_Prestador
			  AND Atendime.Dt_Prevista_Alta IS NOT NULL
			  AND Atendime.Dt_Prevista_Alta BETWEEN :CAMPO_DT_INICIAL AND 
			  AND atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA 
			  AND leito.cd_unid_int = unid_int.cd_unid_int(+)  
			ORDER BY Atendime.Dt_Prevista_Alta
