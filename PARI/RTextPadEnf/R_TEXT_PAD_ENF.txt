SELECT PACIENTE.NM_PACIENTE,
               PACK_INTERNAMENTO.RETORNA_IDADE(PACIENTE.DT_NASCIMENTO, SYSDATE) A,
               DECODE(CONVENIO.TP_CONVENIO, 'C',  ATENDIME.CD_PRO_INT,
              'P',  ATENDIME.CD_PRO_INT,
              'A',  PRO_FAT.CD_SUS,
              'H', PRO_FAT.CD_SUS)     CD_PRO_INT,
               PRO_FAT.DS_PRO_FAT    DS_PRO_INT,
               ATENDIME.CD_PRESTADOR,
               PRESTADOR.NM_PRESTADOR,
               RESPONSA.NM_RESPONSAVEL,
               ATENDIME.CD_LEITO,
               LEITO.DS_LEITO,
               LEITO.CD_UNID_INT,
               UNID_INT.DS_UNID_INT,
               ATENDIME.CD_CONVENIO,
               CONVENIO.NM_CONVENIO
FROM   DBAMV.PACIENTE,
             DBAMV.ATENDIME,
             DBAMV.PRO_FAT,
             DBAMV.PRESTADOR,
             DBAMV.RESPONSA,
             DBAMV.LEITO,
             DBAMV.UNID_INT,
             DBAMV.CONVENIO
WHERE  ATENDIME.CD_PACIENTE        = PACIENTE.CD_PACIENTE
AND       ATENDIME.CD_PRO_INT      = PRO_FAT.CD_PRO_FAT
AND       ATENDIME.CD_PRESTADOR    = PRESTADOR.CD_PRESTADOR
AND       ATENDIME.CD_ATENDIMENTO  = RESPONSA.CD_ATENDIMENTO
AND       ATENDIME.CD_LEITO        = LEITO.CD_LEITO
AND       LEITO.CD_UNID_INT        = UNID_INT.CD_UNID_INT
AND       ATENDIME.CD_CONVENIO     = CONVENIO.CD_CONVENIO
AND      ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
