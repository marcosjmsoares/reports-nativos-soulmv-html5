--R_CAUCAO_ULTRAPASSADA
SELECT *
FROM (SELECT   ATENDIME.CD_ATENDIMENTO
              ,ATENDIME.CD_PACIENTE
              ,PACIENTE.NM_PACIENTE
              ,(SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  CD_PACIENTE) NM_SOCIAL_PACIENTE
              ,ATENDIME.CD_CONVENIO
              ,CONVENIO.NM_CONVENIO
              ,ATENDIME.CD_LEITO
              ,LEITO.DS_RESUMO
              ,LEITO.CD_UNID_INT
              ,UNID_INT.DS_UNID_INT
              ,SUM(CAUCAO.VL_CAUCAO)    VL_CAUCAO
              ,((nvl((SELECT Sum(NVL(REG_FAT.VL_TOTAL_CONTA,0))
		                  FROM dbamv.atendime ATen
                          ,DBAMV.REG_FAT
		                      ,DBAMV.CONVENIO
		                  WHERE REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
		                    AND REG_FAT.CD_ATENDIMENTO = ATENDIME.cd_atendimento
		                    AND REG_FAT.CD_CONVENIO = CONVENIO.CD_CONVENIO
		                    AND CONVENIO.TP_CONVENIO = 'P'
                        AND aten.cd_Atendimento = ATENDIME.CD_ATENDIMENTO
		                    and Aten.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA) , 0)) +
		                  (nvl((SELECT Sum(NVL(REG_AMB.VL_TOTAL_CONTA,0))
		                        FROM  dbamv.atendime ATen, DBAMV.ITREG_AMB
		                             ,DBAMV.REG_AMB
		                             ,DBAMV.CONVENIO
		                        WHERE REG_amb.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
		                          AND ITREG_AMB.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
		                          AND ITREG_AMB.CD_REG_AMB = REG_AMB.CD_REG_AMB
		                          AND ITREG_AMB.CD_CONVENIO = CONVENIO.CD_CONVENIO
		                          AND CONVENIO.TP_CONVENIO = 'P'
                                          AND aten.cd_Atendimento = atendime.cd_atendimento
		                          and Aten.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA),0))) vl_conta
   ,CAUCAO.TP_CAUCAO
FROM DBAMV.ATENDIME
    ,DBAMV.PACIENTE
    ,DBAMV.CAUCAO
    ,DBAMV.LEITO
    ,DBAMV.UNID_INT
    ,DBAMV.CONVENIO
WHERE ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
  AND ATENDIME.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
  AND (CAUCAO.TP_RETIRADO IS NULL or CAUCAO.TP_RETIRADO='D')
  AND ATENDIME.CD_ATENDIMENTO = CAUCAO.CD_ATENDIMENTO
  AND ATENDIME.CD_LEITO = LEITO.CD_LEITO
  AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT
  AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
GROUP BY ATENDIME.CD_ATENDIMENTO
                   ,ATENDIME.CD_PACIENTE
                   ,PACIENTE.NM_PACIENTE
                   ,ATENDIME.CD_CONVENIO
                   ,CONVENIO.NM_CONVENIO
                   ,ATENDIME.CD_LEITO
                   ,LEITO.DS_RESUMO
                   ,LEITO.CD_UNID_INT
                   ,UNID_INT.DS_UNID_INT
                   ,CAUCAO.TP_CAUCAO)
WHERE (((nvl(Vl_Conta,0) * 100) / Vl_Caucao) >= :CAMPO_PERCENT) OR  IS NULL
ORDER BY DS_UNID_INT,CD_UNID_INT,
CD_ATENDIMENTO,CD_PACIENTE,NM_PACIENTE,CD_LEITO,DS_RESUMO,CD_CONVENIO,NM_CONVENIO

--R_CAUCAO_ULTRAPASSADA_R_3_Q_VALOR_CONTA
SELECT * FROM (
SELECT CD_ATENDE
            ,NVL(SUM(VALOR_CONTA),0)  VALOR_CONTA
FROM (
SELECT ATENDIME.CD_ATENDIMENTO        CD_ATENDE
              ,NVL(REG_FAT.VL_TOTAL_CONTA,0)           VALOR_CONTA
FROM     DBAMV.ATENDIME
               ,DBAMV.REG_FAT
               ,DBAMV.CONVENIO
WHERE  REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND        ATENDIME.CD_ATENDIMENTO = REG_FAT.CD_ATENDIMENTO
AND        REG_FAT.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND        CONVENIO.TP_CONVENIO = 'P'
UNION ALL
SELECT ATENDIME.CD_ATENDIMENTO    CD_ATENDE
              ,NVL(REG_AMB.VL_TOTAL_CONTA,0)      VALOR_CONTA
FROM   DBAMV.ATENDIME
             ,DBAMV.ITREG_AMB
             ,DBAMV.REG_AMB
             ,DBAMV.CONVENIO
WHERE REG_amb.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND        ATENDIME.CD_ATENDIMENTO = ITREG_AMB.CD_ATENDIMENTO
AND        ITREG_AMB.CD_REG_AMB = REG_AMB.CD_REG_AMB
AND        ITREG_AMB.CD_CONVENIO = CONVENIO.CD_CONVENIO
AND        CONVENIO.TP_CONVENIO = 'P'
UNION ALL
SELECT ATENDIME.CD_ATENDIMENTO    CD_ATENDE
              ,0      VALOR_CONTA
FROM   DBAMV.ATENDIME
             ,DBAMV.ITREG_AMB
             ,DBAMV.REG_AMB
             ,DBAMV.CONVENIO
WHERE REG_amb.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND        ATENDIME.CD_ATENDIMENTO = ITREG_AMB.CD_ATENDIMENTO
AND        ITREG_AMB.CD_REG_AMB = REG_AMB.CD_REG_AMB
AND        ITREG_AMB.CD_CONVENIO = CONVENIO.CD_CONVENIO
UNION ALL
SELECT ATENDIME.CD_ATENDIMENTO        CD_ATENDE
              ,0          VALOR_CONTA
FROM     DBAMV.ATENDIME
               ,DBAMV.REG_FAT
               ,DBAMV.CONVENIO
WHERE  REG_FAT.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
AND        ATENDIME.CD_ATENDIMENTO = REG_FAT.CD_ATENDIMENTO
AND        REG_FAT.CD_CONVENIO = CONVENIO.CD_CONVENIO
:CAMPO_CF_MULTI_EMPRESA
)
GROUP BY CD_ATENDE
)
WHERE CD_ATENDE = :CAMPO_CD_ATENDIMENTO
