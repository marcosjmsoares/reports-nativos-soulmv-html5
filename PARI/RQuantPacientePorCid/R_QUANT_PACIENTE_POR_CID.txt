--R_QUANT_PACIENTE_POR_CID
SELECT    ATENDIME.CD_CID
                ,CID.DS_CID
                ,COUNT(*)      QT_CID
                ,SUM(NVL(TRUNC(ATENDIME.DT_ALTA), TRUNC(SYSDATE)) - TRUNC(ATENDIME.DT_ATENDIMENTO))                   SUM_CID
                ,ROUND(AVG(NVL(TRUNC(ATENDIME.DT_ALTA), TRUNC(SYSDATE)) - TRUNC(ATENDIME.DT_ATENDIMENTO)))     MEDIA_PERMAN
FROM       DBAMV.ATENDIME
                 ,DBAMV.CID
WHERE   TRUNC(ATENDIME.DT_ATENDIMENTO)  BETWEEN :CAMPO_DT_INICIO AND 
AND         ATENDIME.CD_CID = CID.CD_CID
AND         atendime.tp_atendimento LIKE :CAMPO_TP_ATENDIMENTO
AND        ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY  ATENDIME.CD_CID
                   ,CID.DS_CID
ORDER BY QT_CID    DESC
