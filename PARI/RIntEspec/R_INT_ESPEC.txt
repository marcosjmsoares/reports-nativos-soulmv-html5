--R_INT_ESPEC
SELECT TIPO_ALTA ,
			       CD_ESPECIALID ,
			       DS_ESPECIALID ,
			       CD_TIPO_INTERNACAO ,
			       DS_TIPO_INTERNACAO ,
			       DT_ATENDIMENTO ,
			       DT_ALTA ,
			       CD_ATENDIMENTO ,
			       CD_PACIENTE ,
			       NM_PACIENTE ,
			       NM_SOCIAL_PACIENTE,
			       NM_PRESTADOR ,
			       CD_CID ,
			       DS_CID
			FROM
			  (SELECT 'INTERNAÇÕES POR CONVÊNIOS OU PARTICULAR' TIPO_ALTA ,
			                                                    DECODE(TO_CHAR(ESPECIALID.CD_ESPECIALID), NULL, '0', TO_CHAR(ESPECIALID.CD_ESPECIALID)) CD_ESPECIALID ,
			                                                    DECODE(ESPECIALID.DS_ESPECIALID, NULL, 'SEM ESPECIALIDADE', ESPECIALID.DS_ESPECIALID) DS_ESPECIALID ,
			                                                    ATENDIME.CD_TIPO_INTERNACAO ,
			                                                    TIPO_INTERNACAO.DS_TIPO_INTERNACAO ,
			                                                    ATENDIME.DT_ATENDIMENTO ,
			                                                    ATENDIME.DT_ALTA ,
			                                                    ATENDIME.CD_ATENDIMENTO ,
			                                                    ATENDIME.CD_PACIENTE ,
			                                                    PACIENTE.NM_PACIENTE ,
			        (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
			                                                    PRESTADOR.NM_PRESTADOR ,
			                                                    ATENDIME.CD_CID ,
			                                                    CID.DS_CID
			   FROM DBAMV.ATENDIME ,
			        DBAMV.PACIENTE ,
			        DBAMV.PRESTADOR ,
			        DBAMV.ESPECIALID ,
			        DBAMV.CONVENIO ,
			        DBAMV.TIPO_INTERNACAO ,
			        DBAMV.CID
			   WHERE ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
			     AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
			     AND ATENDIME.TP_ATENDIMENTO IN ('I',
			                                     'H')
			     AND ATENDIME.CD_ESPECIALID = ESPECIALID.CD_ESPECIALID(+)
			     AND ATENDIME.CD_TIPO_INTERNACAO = TIPO_INTERNACAO.CD_TIPO_INTERNACAO(+)
			     AND CONVENIO.TP_CONVENIO IN ('C',
			                                  'P')
			     AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
			     AND ATENDIME.CD_CID = CID.CD_CID(+)
			     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA        
			   UNION ALL SELECT 'INTERNAÇÕES SUS' TIPO_ALTA ,
			                                      DECODE(ATENDIME.CD_ESPECIALID, NULL, '0', ATENDIME.CD_ESPECIALID) CD_ESPECIALID ,
			                                      DECODE(ESPECIALID.DS_ESPECIALID, NULL, 'SEM ESPECIALIDADE', ESPECIALID.DS_ESPECIALID) DS_ESPECIALID ,
			                                      ATENDIME.CD_TIPO_INTERNACAO ,
			                                      TIPO_INTERNACAO.DS_TIPO_INTERNACAO ,
			                                      ATENDIME.DT_ATENDIMENTO ,
			                                      ATENDIME.DT_ALTA ,
			                                      ATENDIME.CD_ATENDIMENTO ,
			                                      ATENDIME.CD_PACIENTE ,
			                                      PACIENTE.NM_PACIENTE ,
			               (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
			                                      PRESTADOR.NM_PRESTADOR ,
			                                      ATENDIME.CD_CID ,
			                                      CID.DS_CID
			   FROM DBAMV.ATENDIME ,
			        DBAMV.PACIENTE ,
			        DBAMV.PRESTADOR ,
			        DBAMV.CONVENIO ,
			        DBAMV.PRO_FAT ,
			        DBAMV.ESPEC_SUS ,
			        DBAMV.TIPO_INTERNACAO ,
			        DBAMV.CID ,
			        DBAMV.ESPECIALID
			   WHERE ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
			     AND ATENDIME.TP_ATENDIMENTO IN ('I',
			                                     'H')
			     AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
			     AND CONVENIO.TP_CONVENIO = 'H'
			     AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
			     AND ATENDIME.CD_TIPO_INTERNACAO = TIPO_INTERNACAO.CD_TIPO_INTERNACAO(+)
			     AND ATENDIME.CD_PRO_INT = PRO_FAT.CD_PRO_FAT(+)
			     AND PRO_FAT.CD_ESPEC_SUS = ESPEC_SUS.CD_ESPEC_SUS(+)
			     AND ATENDIME.CD_CID = CID.CD_CID(+)
			     AND ATENDIME.CD_ESPECIALID = ESPECIALID.CD_ESPECIALID(+)
			     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA         :CAMPO_CF_MULTI_EMPRESA )
