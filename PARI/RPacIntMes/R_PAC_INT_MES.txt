--R_PAC_INT_MES_R_UNID_INT_Q_REPORT
SELECT UNID_INT.CD_UNID_INT ,
       UNID_INT.DS_UNID_INT ,
       CONVENIO.TP_CONVENIO ,
       DECODE(CONVENIO.TP_CONVENIO, 
                'C', ATENDIME.CD_ESPECIALID, 
                'P', ATENDIME.CD_ESPECIALID, 
                'A', PRO_FAT.CD_ESPEC_SUS, 
                'H', PRO_FAT.CD_ESPEC_SUS) CD_ESPECIALIDADE ,
       DECODE(CONVENIO.TP_CONVENIO, 
                'C', ESPECIALID.DS_ESPECIALID, 
                'P', ESPECIALID.DS_ESPECIALID, 
                'A', ESPEC_SUS.DS_ESPEC_SUS, 
                'H', ESPEC_SUS.DS_ESPEC_SUS) DS_ESPECIALIDADE ,
       CONVENIO.CD_CONVENIO ,
       CONVENIO.NM_CONVENIO ,
       ATENDIME.CD_ATENDIMENTO ,
       ATENDIME.CD_PACIENTE ,
       ATENDIME.DT_ATENDIMENTO ,
       ATENDIME.DT_ALTA ,
       DECODE(CONVENIO.TP_CONVENIO, 
                'C', ATENDIME.CD_PRO_INT, 
                'P', ATENDIME.CD_PRO_INT, 
                'A', PRO_FAT.CD_SUS, 
                'H', PRO_FAT.CD_SUS) CD_PRO_INT ,
       PACIENTE.NM_PACIENTE ,
  (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
       PRESTADOR.NM_PRESTADOR
FROM DBAMV.ESPEC_SUS ,
     DBAMV.ATENDIME ,
     DBAMV.PACIENTE ,
     DBAMV.CONVENIO ,
     DBAMV.PRO_FAT ,
     DBAMV.PRESTADOR ,
     DBAMV.LEITO ,
     DBAMV.UNID_INT ,
     DBAMV.ESPECIALID,
     Dbamv.Empresa_Convenio
WHERE Empresa_Convenio.Cd_Convenio = Convenio.Cd_Convenio
	AND Empresa_Convenio.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
	AND TRUNC(ATENDIME.DT_ATENDIMENTO) BETWEEN :CAMPO_DT_INI AND 
	AND TO_NUMBER(TO_CHAR(ATENDIME.DT_ATENDIMENTO, 'DD')) >= TO_NUMBER(TO_CHAR(:CAMPO_NR_DIA,'DD')) 
	AND (ATENDIME.DT_ALTA > :CAMPO_DT_FIM
		OR ATENDIME.DT_ALTA IS NULL)
	AND ATENDIME.TP_ATENDIMENTO = 'I'
	AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
	AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
	AND ATENDIME.CD_PRO_INT = PRO_FAT.CD_PRO_FAT(+)
	AND PRO_FAT.CD_ESPEC_SUS = ESPEC_SUS.CD_ESPEC_SUS(+)
	AND ATENDIME.CD_ESPECIALID = ESPECIALID.CD_ESPECIALID(+)
	AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR(+)
	AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
	AND ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
	AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)     
ORDER BY UNID_INT.CD_UNID_INT,
         ATENDIME.DT_ATENDIMENTO
