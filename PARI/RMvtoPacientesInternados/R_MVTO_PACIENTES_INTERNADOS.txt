--R_MVTO_PACIENTES_INTERNADOS
SELECT * FROM
			  (SELECT TRUNC(ATE.DT_ATENDIMENTO) DT_ATENDIMENTO,
			          ATE.CD_CONVENIO,
			          CON.NM_CONVENIO,
			          ATE.CD_PACIENTE,
			          PAC.NM_PACIENTE,
			      (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE,
			          UNI.DS_UNID_INT,
			          LEI.CD_LEITO,
			          PRE.NM_PRESTADOR,
			          DECODE(CON.TP_CONVENIO, 'C', ATE.CD_PRO_INT, 'P', ATE.CD_PRO_INT, 'A', PROC_SUS.CD_PROCEDIMENTO, 'H', PROC_SUS.CD_PROCEDIMENTO) CD_PRO_INT,
			          ATE.CD_ATENDIMENTO,
			          SAM.NR_MATRICULA_SAME,
			          SER.DS_SERVICO,
			          ltrim(rtrim(CID.NM_CIDADE) || ' ' || CID.CD_UF) NM_CIDADE
			   FROM DBAMV.ATENDIME ATE,
			        DBAMV.PACIENTE PAC,
			        DBAMV.PRO_FAT,
			        DBAMV.PROCEDIMENTO_SUS PROC_SUS,
			        DBAMV.LEITO LEI,
			        DBAMV.UNID_INT UNI,
			        DBAMV.PRESTADOR PRE,
			        DBAMV.SAME SAM,
			        DBAMV.SERVICO SER,
			        DBAMV.CIDADE CID,
			        DBAMV.CONVENIO CON,
			        DBAMV.LEITO,
			        DBAMV.UNID_INT
			   WHERE ATE.TP_ATENDIMENTO = 'I'
			     AND ATE.DT_ATENDIMENTO >= TO_DATE(TO_CHAR(:CAMPO_DT_INICIAL, 'DD/MM/YYYY') || ' 00', 'DD/MM/YYYY HH24')
			     AND ATE.DT_ATENDIMENTO <= TO_DATE(TO_CHAR(:CAMPO_DT_FINAL , 'DD/MM/YYYY') || ' 23', 'DD/MM/YYYY HH24')
			     AND ATE.CD_LEITO = LEITO.CD_LEITO(+)
			     AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)    
			     AND ATE.CD_PACIENTE = PAC.CD_PACIENTE
			     AND ATE.CD_PRO_INT = PRO_FAT.CD_PRO_FAT (+)
			     AND ATE.CD_PROCEDIMENTO = PROC_SUS.CD_PROCEDIMENTO (+)
			     AND ATE.CD_LEITO = LEI.CD_LEITO(+)
			     AND LEI.CD_UNID_INT = UNI.CD_UNID_INT
			     AND ATE.CD_PRESTADOR = PRE.CD_PRESTADOR(+)
			     AND PAC.CD_PACIENTE = SAM.CD_PACIENTE(+)
			     AND ATE.CD_SERVICO = SER.CD_SERVICO(+)
			     AND PAC.CD_CIDADE = CID.CD_CIDADE(+)
			     AND ATE.CD_CONVENIO = CON.CD_CONVENIO
			     AND ATE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA 
			     AND ATE.CD_ATENDIMENTO_PAI IS NULL
			   ORDER BY PAC.NM_PACIENTE,
			            UNI.DS_UNID_INT)
			ORDER BY DT_ATENDIMENTO,
			         NM_CONVENIO,
			         CD_CONVENIO
