--R_MOV_LEITO
SELECT leito.cd_leito,
			       leito.ds_resumo,
			       fnc_mv_recupera_data_hora(mov_int.dt_mov_int, mov_int.hr_mov_int) dh_mov_int,
			       DECODE (mov_int.tp_mov,
			               'I', 'Internação',
			               'O', 'Transferência',
			               'L', 'Limpeza',
			               'R', 'Reserva',
			               'M', 'Manutenção',
			               'E', 'Reforma',
			               'N', 'Interdição',
			               'F', 'Infecção',
			               'A', 'Acompanhante',
			       'Movimentação') tp_mov,
			       mov_int.cd_atendimento,
			       paciente.cd_paciente,
			       paciente.nm_paciente,
			   (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
			              mov_int.nm_usuario
			FROM dbamv.mov_int,
			     dbamv.leito,
			     dbamv.paciente,
			     dbamv.atendime
			WHERE mov_int.cd_leito = leito.cd_leito
			  AND mov_int.cd_atendimento = atendime.cd_atendimento(+)
			  AND paciente.cd_paciente(+) = atendime.cd_paciente
			  AND mov_int.cd_leito = :CAMPO_LEITO 
			ORDER BY fnc_mv_recupera_data_hora(mov_int.dt_mov_int, mov_int.hr_mov_int) DESC
