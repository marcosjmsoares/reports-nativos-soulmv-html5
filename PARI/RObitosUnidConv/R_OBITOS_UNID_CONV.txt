SELECT ATENDIME.CD_ATENDIMENTO CD_ATENDIMENTO ,
			       PACIENTE.CD_PACIENTE CD_PACIENTE ,
			       PACIENTE.NM_PACIENTE NM_PACIENTE ,
			   (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
			       ATENDIME.DT_ATENDIMENTO DT_ATENDIMENTO ,
			       ATENDIME.HR_ATENDIMENTO HR_ATENDIMENTO ,
			       ATENDIME.DT_ALTA_MEDICA DT_ALTA ,
			       ATENDIME.HR_ALTA_MEDICA HR_ALTA ,
			       PACIENTE.DT_NASCIMENTO DT_NASCIMENTO ,
			       PACIENTE.TP_SEXO TP_SEXO ,
			       UNID_INT.CD_UNID_INT CD_UNID_INT ,
			       UNID_INT.DS_UNID_INT DS_UNID_INT ,
			       CONVENIO.CD_CONVENIO CD_CONVENIO ,
			       CONVENIO.NM_CONVENIO NM_CONVENIO
			FROM DBAMV.ATENDIME ,
			     DBAMV.PACIENTE ,
			     DBAMV.LEITO ,
			     DBAMV.MOT_ALT ,
			     DBAMV.UNID_INT ,
			     DBAMV.CONVENIO
			WHERE TRUNC(ATENDIME.DT_ALTA) BETWEEN :CAMPO_DT_INI AND 
			  AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
			  AND ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
			  AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)
			  AND ATENDIME.CD_MOT_ALT = MOT_ALT.CD_MOT_ALT
			  AND MOT_ALT.TP_MOT_ALTA = 'O'
			  AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
			  AND ATENDIME.DT_ALTA IS NOT NULL
			  AND ATENDIME.DT_ATENDIMENTO IS NOT NULL
			  AND ATENDIME.TP_ATENDIMENTO = 'I'
			  AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA    
			ORDER BY cd_unid_int
