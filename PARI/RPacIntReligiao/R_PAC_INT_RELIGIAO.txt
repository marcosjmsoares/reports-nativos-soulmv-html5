--R_PAC_INT_RELIGIAO
SELECT * FROM
  (SELECT RELIGIAO.DS_RELIGIAO ,
          LEITO.DS_ENFERMARIA ,
          LEITO.DS_RESUMO ,
          PACIENTE.CD_PACIENTE ,
          DECODE(nvl(PACIENTE.SN_UTILIZA_NOME_SOCIAL, 'N'),'S', Nvl(PACIENTE.NM_SOCIAL_PACIENTE, PACIENTE.NM_PACIENTE), PACIENTE.NM_PACIENTE) NM_PACIENTE ,     
          PACIENTE.DT_NASCIMENTO ,
          ATENDIME.CD_ATENDIMENTO ,
          ATENDIME.DT_ATENDIMENTO ,
          ATENDIME.DT_ALTA ,
          ATENDIME.HR_ATENDIMENTO ,
          UNID_INT.CD_UNID_INT ,
          UNID_INT.DS_UNID_INT ,
          PRESTADOR.NM_PRESTADOR ,
          PRESTADOR.CD_PRESTADOR ,
          DBAMV.PACK_INTERNAMENTO.RETORNA_IDADE(PACIENTE.DT_NASCIMENTO, sysdate) IDADE_T
   FROM dbamv.religiao ,
        dbamv.ATENDIME ,
        dbamv.LEITO ,
        dbamv.PACIENTE ,
        dbamv.PRESTADOR ,
        DBAMV.UNID_INT
   WHERE ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
     AND ATENDIME.TP_ATENDIMENTO = 'I'
     AND PACIENTE.CD_RELIGIAO = RELIGIAO.CD_RELIGIAO(+)
     AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
     AND ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
     AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)
     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA       
   ORDER BY UNID_INT.DS_UNID_INT ,
            RELIGIAO.DS_RELIGIAO ,
            LEITO.DS_ENFERMARIA ,
            LEITO.DS_LEITO)
ORDER BY CD_UNID_INT,
         DS_UNID_INT
