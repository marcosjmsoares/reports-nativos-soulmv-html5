SELECT   PACIENTE.CD_PACIENTE,
    PACIENTE.NM_PACIENTE,
    PACIENTE.DT_NASCIMENTO,
    PACIENTE.TP_SEXO,
    PACIENTE.TP_COR,
    DBAMV.FNC_ESTADO_CIVIL(PACIENTE.TP_ESTADO_CIVIL) TP_ESTADO_CIVIL1,
    PACIENTE.NR_DOCUMENTO,
    PACIENTE.NM_MAE,
    PACIENTE.DS_ENDERECO,
    PACIENTE.NR_ENDERECO,
    PACIENTE.NM_BAIRRO,
    PACIENTE.NR_CEP,
    PACIENTE.NR_FONE,
    PACIENTE.DT_CADASTRO,
    PACIENTE.NR_IDENTIDADE,
    PACIENTE.DS_OM_IDENTIDADE,
    PACIENTE.NR_CPF,
    ATENDIME.CD_ATENDIMENTO,
    ATENDIME.CD_PACIENTE,
    ATENDIME.DT_ATENDIMENTO,
    ATENDIME.HR_ATENDIMENTO,
    ATENDIME.CD_SERVICO,
    ATENDIME.CD_SER_DIS,
    ATENDIME.TP_ATENDIMENTO,
    ORI_ATE.DS_ORI_ATE,
    DECODE(CONVENIO.TP_CONVENIO, 'C',  ATENDIME.CD_PRO_INT,
                                 'P',  ATENDIME.CD_PRO_INT,
                                 'A',  PRO_FAT.CD_SUS,
                                 'H', PRO_FAT.CD_SUS)     CD_PRO_INT,
    PRO_FAT.DS_PRO_FAT       DS_PRO_INT,
    PRESTADOR.NM_PRESTADOR,
    PRESTADOR.DS_CODIGO_CONSELHO,
    LEITO.DS_ENFERMARIA,
    LEITO.DS_LEITO,
    CONVENIO.NM_CONVENIO,
    CID.DS_CID,
    SERVICO.DS_SERVICO,
    SERVICO.TP_ANAMNESE TP_ANAM_SERVICO,
    SER_DIS.DS_SER_DIS,
    SER_DIS.TP_ANAMNESE TP_ANAM_SER_DIS,
    TIP_ACOM.DS_TIP_ACOM,
    RESPONSA.NM_RESPONSAVEL,
    RESPONSA.DS_DOCUMENTO  DS_DOCUMENTO_RESPONSAVEL,
    RESPONSA.DS_CONTATO_FAMILIA,
    RESPONSA.DS_COMPLEMENTO,
    RESPONSA.NR_ENDERECO,
    RESPONSA.DS_ENDERECO ENDERECO_RESPOSAVEL,
    RESPONSA.TP_ESTADO_CIVIL,
    RESPONSA.NR_CPF CPF_RESPONSAVEL,
    RESPONSA.NR_CEP  NR_CEP_RESPONSAVEL,
    RESPONSA.NM_BAIRRO NM_BAIRRO_RESPONSAVEL,
    CARTEIRA.NR_CARTEIRA,
    CARTEIRA.NM_TITULAR,
    CIDADE.NM_CIDADE,
    CIDADE.CD_UF,
    CON_PLA.DS_CON_PLA,
    CONVENIO.CD_CONVENIO,
    PROFISSAO.NM_PROFISSAO,
    CIDADE_RESP.NM_CIDADE NM_CIDADE_RESP,
    CIDADE_RESP.CD_UF CD_UF_CIDADE_RESPONSAVEL,
    GUIA.CD_SENHA,
    CARTEIRA.NM_EMPRESA,
    CARTEIRA.DT_ULT_PGTO,
    PACIENTE.DS_COMPLEMENTO  DS_COMPLEMENTO_END,
    TIP_PAREN.DS_TIP_PAREN DS_TIP_PAREN,
    RESPONSA.NR_FONE NR_FONE_RESP,
    RESPONSA.NR_CPF NR_CPF_RESP,
    PACIENTE.NM_PAI NM_PAI_RESP,
    PRESTADOR.DS_CODIGO_CONSELHO,
    UNID_INT.DS_UNID_INT
FROM      DBAMV.PACIENTE,
                 DBAMV.CIDADE ,
                 DBAMV.ATENDIME,
                 DBAMV.ORI_ATE,
                 DBAMV.PRO_FAT,
                 DBAMV.PRESTADOR,
                 DBAMV.LEITO,
                 DBAMV.CONVENIO,
                 DBAMV.CON_PLA,
                 DBAMV.CID,
                 DBAMV.SERVICO,
                 DBAMV.SER_DIS,
                 DBAMV.TIP_ACOM,
                 DBAMV.RESPONSA,
                 DBAMV.CARTEIRA,
                 DBAMV.PROFISSAO,
                 DBAMV.CIDADE CIDADE_RESP,
                 DBAMV.GUIA,
                 DBAMV.UNID_INT,
                 DBAMV.TIP_PAREN
WHERE  ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
And    ATENDIME.CD_ATENDIMENTO   = TO_CHAR(:CAMPO_PCD_ATENDIMENTO)
AND    ATENDIME.CD_PACIENTE      = PACIENTE.CD_PACIENTE
AND    PACIENTE.CD_CIDADE        = CIDADE.CD_CIDADE
AND    ATENDIME.CD_ORI_ATE       = ORI_ATE.CD_ORI_ATE
AND    RESPONSA.CD_TIP_PAREN     = TIP_PAREN.CD_TIP_PAREN(+)
AND    ATENDIME.CD_PRO_INT       = PRO_FAT.CD_PRO_FAT(+)
AND    ATENDIME.CD_PRESTADOR     = PRESTADOR.CD_PRESTADOR
AND    ATENDIME.CD_LEITO         = LEITO.CD_LEITO (+)
AND    ATENDIME.CD_CONVENIO      = CONVENIO.CD_CONVENIO
AND    ATENDIME.CD_CONVENIO      = CON_PLA.CD_CONVENIO
AND    ATENDIME.CD_CON_PLA       = CON_PLA.CD_CON_PLA
AND    ATENDIME.CD_CID           = CID.CD_CID (+)
AND    ATENDIME.CD_SERVICO       = SERVICO.CD_SERVICO(+)
AND    ATENDIME.CD_SER_DIS       = SER_DIS.CD_SER_DIS (+)
AND    ATENDIME.CD_TIP_ACOM      = TIP_ACOM.CD_TIP_ACOM(+)
AND    ATENDIME.CD_ATENDIMENTO   = RESPONSA.CD_ATENDIMENTO(+)
AND    UNID_INT.CD_UNID_INT      = LEITO.CD_UNID_INT
AND    RESPONSA.CD_PROFISSAO     = PROFISSAO.CD_PROFISSAO(+)
AND    RESPONSA.CD_CIDADE        = CIDADE_RESP.CD_CIDADE(+)
AND    ATENDIME.CD_ATENDIMENTO   = GUIA.CD_ATENDIMENTO(+)
AND    ATENDIME.CD_PACIENTE      = CARTEIRA.CD_PACIENTE (+)
AND    ATENDIME.CD_CONVENIO      = CARTEIRA.CD_CONVENIO(+)
AND    ATENDIME.CD_CON_PLA       = CARTEIRA.CD_CON_PLA(+)
AND    ATENDIME.NR_CARTEIRA      = CARTEIRA.NR_CARTEIRA(+)
