--R_SIT_ATEND
SELECT ATENDIME.CD_ATENDIMENTO,
               ATENDIME.DT_ATENDIMENTO,
              ATENDIME.DT_ALTA,
              PRO_FAT.DS_PRO_FAT     PROCEDIMENTO,
              GUIA.NR_GUIA
FROM DBAMV.ATENDIME,
           DBAMV.CONVENIO,
           DBAMV.PRO_FAT,
           DBAMV.GUIA
where atendime.cd_multi_empresa = :CAMPO_CD_MULTI_EMPRESA
and  TRUNC(ATENDIME.DT_ATENDIMENTO)  BETWEEN  :CAMPO_DATINI  AND  
AND       ATENDIME.CD_CONVENIO  =  CONVENIO.CD_CONVENIO
AND       ATENDIME.CD_PRO_INT  =  PRO_FAT.CD_PRO_FAT
AND       CONVENIO.TP_CONVENIO = 'H'
AND       ATENDIME.CD_ATENDIMENTO  =   GUIA.CD_ATENDIMENTO(+)
AND       ATENDIME.CD_CONVENIO  =  GUIA.CD_CONVENIO(+)
AND      GUIA.TP_GUIA(+)  =  DECODE(ATENDIME.TP_ATENDIMENTO, 'I', 'I', 'E', 'P', 'C' )
ORDER BY ATENDIME.CD_ATENDIMENTO
