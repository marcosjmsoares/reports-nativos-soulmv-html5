--R_INTERNACOES_UNIDADE
SELECT * FROM
			  (SELECT UNID_INT.CD_UNID_INT ,
			          UNID_INT.DS_UNID_INT ,
			          ATENDIME.CD_ATENDIMENTO ,
			          ATENDIME.CD_PACIENTE ,
			          PACIENTE.NM_PACIENTE ,
			      (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
			          ATENDIME.DT_ATENDIMENTO ,
			          ATENDIME.DT_ALTA ,
			          CONVENIO.NM_CONVENIO ,
			          PRESTADOR.NM_PRESTADOR ,
			          PACIENTE.NR_IDENTIDADE ,
			          PACK_INTERNAMENTO.RETORNA_IDADE(PACIENTE.DT_NASCIMENTO, SYSDATE) NR_IDADE ,
			          TIPO_INTERNACAO.DS_TIPO_INTERNACAO ,
			          leito.ds_resumo ,
			          trunc(NVL(ATENDIME.DT_ALTA, TRUNC(SYSDATE)) - TRUNC(ATENDIME.DT_ATENDIMENTO)) QTD_DIAS_INT
			   FROM DBAMV.ATENDIME ,
			        DBAMV.PACIENTE ,
			        DBAMV.PRESTADOR ,
			        DBAMV.CONVENIO ,
			        DBAMV.TIPO_INTERNACAO ,
			        DBAMV.MOV_INT ,
			        DBAMV.LEITO ,
			        DBAMV.UNID_INT
			   WHERE ATENDIME.TP_ATENDIMENTO IN ('I',
			                                     'H',
			                                     'B')
			     AND TRUNC(MOV_INT.DT_MOV_INT) BETWEEN TRUNC(:CAMPO_DATINI) AND To_Date(Concat(To_Char(:CAMPO_DATFIM,'dd/mm/yyyy'), ' 23'), 'dd/mm/yyyy hh24')
			     AND ATENDIME.CD_ATENDIMENTO_PAI IS NULL
			     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
			     AND ATENDIME.CD_PRESTADOR = PRESTADOR.CD_PRESTADOR
			     AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
			     AND TIPO_INTERNACAO.CD_TIPO_INTERNACAO(+) = ATENDIME.CD_TIPO_INTERNACAO
			     AND ATENDIME.CD_ATENDIMENTO = MOV_INT.CD_ATENDIMENTO
			     AND MOV_INT.TP_MOV = 'I'
			     AND MOV_INT.CD_LEITO = LEITO.CD_LEITO
			     AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)
			     AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA     :CAMPO_CF_ATEND_FILHO )
