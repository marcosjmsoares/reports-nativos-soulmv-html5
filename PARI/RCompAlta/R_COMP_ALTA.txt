--R_COMP_ALTA
SELECT ATENDIME.CD_ATENDIMENTO                     CD_ATENDIMENTO
     , ATENDIME.CD_PACIENTE                        CD_PACIENTE
     , PACIENTE.NM_PACIENTE                        NM_PACIENTE
     , (SELECT PA.NM_SOCIAL_PACIENTE
        FROM DBAMV.PACIENTE PA
        WHERE PA.CD_PACIENTE = PACIENTE.CD_PACIENTE
          AND NVL(PA.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE
     , ATENDIME.DT_ATENDIMENTO                     DT_ATENDIMENTO
     , TO_CHAR(ATENDIME.HR_ATENDIMENTO, 'HH24') HR_ATENDIMENTO
     , SERVICO.CD_SERVICO                          CD_SERVICO
     , SERVICO.DS_SERVICO                          DS_SERVICO
     , LEITO.CD_LEITO                              CD_LEITO
     , LEITO.DS_LEITO                              DS_LEITO
     , LEITO.CD_UNID_INT                           CD_UNID_INT
     , DECODE(:CAMPO_TP_ALTA, 'Medica', ATENDIME.DT_ALTA_MEDICA, ATENDIME.DT_ALTA) DT_ALTA
     , DECODE(:CAMPO_TP_ALTA, 'Medica', ATENDIME.HR_ALTA_MEDICA, ATENDIME.HR_ALTA) HR_ALTA
     , MOT_ALT.CD_MOT_ALT                       CD_MOT_ALT
     , MOT_ALT.DS_MOT_ALT                       DS_MOT_ALT
     , DECODE( CONVENIO.TP_CONVENIO
             , 'A', PROCEDIMENTO_SUS.Cd_Procedimento
             , 'H', PROCEDIMENTO_SUS.Cd_Procedimento
             , 'C', ATENDIME.CD_PRO_INT_PROCEDIMENTO_ENTRAD
             , 'P', ATENDIME.CD_PRO_INT_PROCEDIMENTO_ENTRAD
             , ATENDIME.CD_PRO_INT_PROCEDIMENTO_ENTRAD
             ) CD_PRO_INT_PROCEDIMENTO_ENTRAD
     , ATENDIME.DS_OBS_ALTA                  DS_OBS_ALTA
     , ATENDIME.CD_CONVENIO                  CD_CONVENIO
     , CONVENIO.NM_CONVENIO                 NM_CONVENIO
     , ATENDIME.CD_CON_PLA                    CD_CON_PLA
     , CON_PLA.DS_CON_PLA                      DS_CON_PLA
     , ATENDIME.NM_USUARIO_ALTA         NM_USUARIO_ALTA
     , ATENDIME.NM_USUARIO_ALTA_MEDICA                    NM_USUARIO 
     , DECODE( CONVENIO.TP_CONVENIO
             , 'C', decode( PRO_FAT.DS_PRO_FAT
                          , NULL, :CAMPO_PROC_NAO_INF, PRO_FAT.DS_PRO_FAT)
             , 'P', decode( PRO_FAT.DS_PRO_FAT
                          , NULL, :CAMPO_PROC_NAO_INF, PRO_FAT.DS_PRO_FAT)
             , 'A', decode( PROCEDIMENTO_SUS.Ds_Procedimento
                          , NULL, :CAMPO_PROC_NAO_INF, PROCEDIMENTO_SUS.Ds_Procedimento)
             , 'H', DECODE( PROCEDIMENTO_SUS.Ds_Procedimento
                          , NULL, :CAMPO_PROC_NAO_INF, PROCEDIMENTO_SUS.Ds_Procedimento)
             , PRO_FAT.DS_PRO_FAT
             ) PROCEDIMENTO
     , ATENDIME.CD_CID 
     , CID.DS_CID
     , PRESTADOR.NM_PRESTADOR
     , PRESTADOR.CD_PRESTADOR 
     , PRESTADOR.DS_CODIGO_CONSELHO
     , CONSELHO.DS_CONSELHO     DESC_CONSELHO
  FROM DBAMV.ATENDIME
     , DBAMV.PACIENTE
     , DBAMV.PRO_FAT
     , DBAMV.SERVICO
     , DBAMV.LEITO
     , DBAMV.MOT_ALT
     , DBAMV.CONVENIO
     , DBAMV.CON_PLA
     , DBAMV.PRESTADOR
     , DBAMV.CID
     , DBAMV.CONSELHO
     , Dbamv.PROCEDIMENTO_SUS
 WHERE ATENDIME.TP_ATENDIMENTO IN ( 'I', 'H', 'B')
   AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
   AND ATENDIME.CD_PRO_INT_PROCEDIMENTO_ENTRAD  =  PRO_FAT.CD_PRO_FAT(+)
   AND ATENDIME.CD_CONVENIO   = CONVENIO.CD_CONVENIO
   AND ATENDIME.CD_SERVICO = SERVICO.CD_SERVICO(+)
   AND ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
   AND ATENDIME.CD_MOT_ALT = MOT_ALT.CD_MOT_ALT(+)
   AND ATENDIME.CD_CONVENIO = CON_PLA.CD_CONVENIO(+)
   AND ATENDIME.CD_CON_PLA   = CON_PLA.CD_CON_PLA(+)
   AND ATENDIME.CD_CID = CID.CD_CID(+)
   AND PRESTADOR.CD_PRESTADOR = ATENDIME.CD_PRESTADOR 
   AND PRESTADOR.CD_CONSELHO  =  CONSELHO.CD_CONSELHO
   AND Atendime.CD_PROcedimento_Alta = PROCEDIMENTO_SUS.Cd_Procedimento(+)
