--R_TP_INTERN_X_CONV
SELECT * FROM(
SELECT  ATE.CD_CONVENIO          CD_CONVENIO,
        CON.NM_CONVENIO          NM_CONVENIO,
        NVL(TPI.DS_TIPO_INTERNACAO,'NAO DEFINIDO')   DS_TIPO_INTERNACAO,
        COUNT(*)         QTD,
        nvl( car.nm_empresa, 'NAO DEFINIDA') NM_EMPRESA
FROM    DBAMV.ATENDIME        ATE,
        DBAMV.CONVENIO        CON,
        DBAMV.ORI_ATE         ORI,
        DBAMV.TIPO_INTERNACAO TPI,
       dbamv.carteira         CAR
WHERE  TRUNC(ATE.DT_ATENDIMENTO)  BETWEEN  :CAMPO_DT_INICIAL  AND  
AND        ATE.CD_CONVENIO         =  CON.CD_CONVENIO
AND        ATE.CD_ORI_ATE          =  ORI.CD_ORI_ATE
AND        ATE.CD_TIPO_INTERNACAO  =  TPI.CD_TIPO_INTERNACAO(+)
AND        ATE.CD_PACIENTE         =  CAR.CD_PACIENTE(+)
AND        ATE.CD_CONVENIO         =  CAR.CD_CONVENIO(+)
AND        ATE.CD_CON_PLA          =  CAR.CD_CON_PLA(+)
AND        ATE.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY  CON.NM_CONVENIO, ATE.CD_CONVENIO, car.nm_empresa ,DS_TIPO_INTERNACAO
)
ORDER BY NM_CONVENIO,DS_TIPO_INTERNACAO
