--R_MAPA_TRANSF
SELECT * FROM
  (SELECT PACIENTE.CD_PACIENTE,
          DECODE(SN_UTILIZA_NOME_SOCIAL,'S', Nvl(NM_SOCIAL_PACIENTE, NM_PACIENTE), 'N', NM_PACIENTE) NM_PACIENTE,
          PACIENTE.NR_RG_NASC,
          ATENDIME.CD_ATENDIMENTO,
          MOV_INT.CD_LEITO_ANTERIOR,
          LEITO_ANT.CD_UNID_INT CD_UNID_INT_ANT,
          UNID_INT_ANT.DS_UNID_INT DS_UNID_INT_ANT, 
          LEITO_ANT.DS_RESUMO DS_LEITO_ANT,
          MOV_INT.CD_LEITO CD_LEITO_ATUAL,
          LEITO_ATU.CD_UNID_INT CD_UNID_INT_ATU,
          UNID_INT_ATU.DS_UNID_INT DS_UNID_INT_ATU,  
          LEITO_ATU.DS_RESUMO DS_LEITO_ATU,
          TRUNC(MOV_INT.DT_MOV_INT) DT_MOV_INT,
          TO_CHAR(MOV_INT.HR_MOV_INT, 'HH24') HR_MOV_INT,
          MOTIVO_TRANSF_LEITO.CD_MOTIVO_TRANSF_LEITO,
          NVL(MOTIVO_TRANSF_LEITO.DS_MOTIVO_TRANSF_LEITO, 'NÃO EXISTE MOTIVO PARA ESSE(S) ATENDIMENTO(S)') DS_MOTIVO_TRANSF_LEITO,
          CONVENIO.NM_CONVENIO,
          SERVICO.DS_SERVICO,
          MOV_INT.CD_MOV_INT  ,
          TIP_ACOM.DS_TIP_ACOM 
   FROM DBAMV.MOV_INT,
        DBAMV.LEITO LEITO_ANT,
        DBAMV.LEITO LEITO_ATU,
        DBAMV.MOTIVO_TRANSF_LEITO,
        DBAMV.ATENDIME,
        DBAMV.PACIENTE,
        DBAMV.CONVENIO,
        DBAMV.SERVICO,
        DBAMV.UNID_INT UNID_INT_ANT,  
        DBAMV.UNID_INT UNID_INT_ATU, 
        DBAMV.TIP_ACOM 
   WHERE TRUNC(MOV_INT.DT_MOV_INT) BETWEEN :CAMPO_DT_INICIAL AND 
     AND MOV_INT.TP_MOV = 'O'
     AND MOV_INT.CD_LEITO_ANTERIOR = LEITO_ANT.CD_LEITO(+)
     AND MOV_INT.CD_LEITO = LEITO_ATU.CD_LEITO
     AND MOV_INT.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO
     AND MOV_INT.CD_MOTIVO_TRANSF_LEITO = MOTIVO_TRANSF_LEITO.CD_MOTIVO_TRANSF_LEITO(+) 
     AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
     AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
     AND ATENDIME.CD_SERVICO = SERVICO.CD_SERVICO(+)
     AND leito_atu.cd_unid_int = unid_int_atu.cd_unid_int 
     AND leito_ant.cd_unid_int = unid_int_ant.cd_unid_int 
     AND atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA 
     AND MOV_INT.CD_TIP_ACOM = TIP_ACOM.CD_TIP_ACOM 
   ORDER BY PACIENTE.CD_PACIENTE,
            ATENDIME.CD_ATENDIMENTO,
            MOV_INT.DT_MOV_INT,
            MOV_INT.HR_MOV_INT)
ORDER BY CD_MOTIVO_TRANSF_LEITO,
         DS_MOTIVO_TRANSF_LEITO
