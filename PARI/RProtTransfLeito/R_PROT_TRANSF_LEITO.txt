--R_PROT_TRANSF_LEITO
SELECT ATENDIME.CD_ATENDIMENTO
      ,PACIENTE.CD_PACIENTE
      ,DECODE(PACIENTE.SN_UTILIZA_NOME_SOCIAL,'S',PACIENTE.NM_SOCIAL_PACIENTE,PACIENTE.NM_PACIENTE) AS NM_PACIENTE
      ,CONVENIO.CD_CONVENIO
      ,CONVENIO.NM_CONVENIO
      ,CON_PLA.CD_CON_PLA
      ,CON_PLA.DS_CON_PLA
      ,SUB_PLANO.CD_SUB_PLANO
      ,SUB_PLANO.DS_SUB_PLANO
      ,LEITO_ANT.CD_LEITO CD_LEITO_ANT
      ,LEITO_ANT.DS_LEITO DS_LEITO_ANT
      ,TIP_ACOM_ANT.CD_TIP_ACOM CD_TIP_ACOM_ANT
      ,TIP_ACOM_ANT.DS_TIP_ACOM DS_TIP_ACOM_ANT
      ,SOLIC_TRANSFERENCIA_LEITO.CD_TIP_ACOM
      ,TIP_ACOM.DS_TIP_ACOM
      ,SOLIC_TRANSFERENCIA_LEITO.CD_LEITO_RESERVA
      ,LEITO.DS_LEITO
      ,SOLIC_TRANSFERENCIA_LEITO.NM_SOLICITANTE
      ,DECODE(SOLIC_TRANSFERENCIA_LEITO.TP_DOCUMENTO,'I','IDENTIDADE'
                                                    ,'C','CPF'
                                                    ,'T','CARTEIRA DE TRABALHO'
                                                    ,'P','PASSAPORTE'
                                                    ,'O','OUTROS') TP_DOCUMENTO
      ,SOLIC_TRANSFERENCIA_LEITO.DS_DOCUMENTO
      ,TIP_PAREN.CD_TIP_PAREN
      ,TIP_PAREN.DS_TIP_PAREN
      ,MOTIVO_TRANSF_LEITO.CD_MOTIVO_TRANSF_LEITO
      ,MOTIVO_TRANSF_LEITO.DS_MOTIVO_TRANSF_LEITO
      ,SOLIC_TRANSFERENCIA_LEITO.DT_SOLIC_TRANSF
      ,SOLIC_TRANSFERENCIA_LEITO.HR_SOLIC_TRANSF
      ,SOLIC_TRANSFERENCIA_LEITO.CD_USUARIO_SOLICITACAO
      ,SOLIC_TRANSFERENCIA_LEITO.DH_SOLICITACAO
FROM DBAMV.ATENDIME
    ,DBAMV.SOLIC_TRANSFERENCIA_LEITO
    ,DBAMV.PACIENTE
    ,DBAMV.CONVENIO
    ,DBAMV.CON_PLA
    ,DBAMV.SUB_PLANO
    ,DBAMV.MOTIVO_TRANSF_LEITO
    ,DBAMV.TIP_PAREN
    ,DBAMV.TIP_ACOM
    ,DBAMV.TIP_ACOM TIP_ACOM_ANT
    ,DBAMV.LEITO
    ,DBAMV.LEITO LEITO_ANT
WHERE ATENDIME.CD_PACIENTE   = PACIENTE.CD_PACIENTE
 AND CONVENIO.CD_CONVENIO    = ATENDIME.CD_CONVENIO
 AND CON_PLA.CD_CON_PLA      = ATENDIME.CD_CON_PLA
 AND CONVENIO.CD_CONVENIO    = CON_PLA.CD_CONVENIO
 AND ATENDIME.CD_CON_PLA    = CON_PLA.CD_CON_PLA(+)
 AND ATENDIME.CD_CONVENIO  = SUB_PLANO.CD_CONVENIO(+)
 AND ATENDIME.CD_SUB_PLANO = SUB_PLANO.CD_SUB_PLANO(+)
 AND ATENDIME.CD_TIP_ACOM  = TIP_ACOM_ANT.CD_TIP_ACOM(+)
 AND SOLIC_TRANSFERENCIA_LEITO.CD_TIP_ACOM = TIP_ACOM.CD_TIP_ACOM
 AND LEITO_ANT.CD_LEITO = ATENDIME.CD_LEITO
 AND LEITO.CD_LEITO(+) = SOLIC_TRANSFERENCIA_LEITO.CD_LEITO_RESERVA
 AND MOTIVO_TRANSF_LEITO.CD_MOTIVO_TRANSF_LEITO = SOLIC_TRANSFERENCIA_LEITO.CD_MOTIVO_TRANSF_LEITO
 AND ATENDIME.CD_ATENDIMENTO = SOLIC_TRANSFERENCIA_LEITO.CD_ATENDIMENTO
 AND SOLIC_TRANSFERENCIA_LEITO.CD_TIP_PAREN = TIP_PAREN.CD_TIP_PAREN(+)
 AND SOLIC_TRANSFERENCIA_LEITO.CD_SOLIC_TRANSFERENCIA_LEITO = :CAMPO_CD_SOLIC_TRANSFERENCIA_LEITO
 AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
