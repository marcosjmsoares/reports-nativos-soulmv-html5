--R_ALTASCONVENIO
SELECT * FROM (
SELECT conv.CD_CONVENIO,
               conv.NM_CONVENIO,
               aten.CD_ATENDIMENTO,
               paciente.NM_PACIENTE,
               (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  CD_PACIENTE) NM_SOCIAL_PACIENTE,
               aten.DT_ATENDIMENTO,
               aten.DT_ALTA,
               aten.HR_ALTA,
               copl.DS_CON_PLA,
               pres.NM_PRESTADOR,
                  DECODE(CONV.Tp_Convenio, 'C', ATEN.Cd_Pro_INT,
                                               'P', ATEN.Cd_Pro_INT,
                                               'A', Procedimento_Sus.Cd_Procedimento,
                                               'H', Procedimento_Sus.Cd_Procedimento) CD_PRO_INT,
                         DECODE(CONV.Tp_Convenio, 'C', PRO_FAT.DS_PRO_FAT ,
                                               'P', PRO_FAT.DS_PRO_FAT ,
                                               'A', Procedimento_Sus.DS_Procedimento,
                                               'H', Procedimento_Sus.DS_Procedimento) DS_PRO_INT,
               leito.DS_RESUMO,
               MOT_ALT.DS_MOT_ALT,
               ORI.DS_ORI_ATE ,
               MOT_ALT.CD_MOT_ALT,
               DBAMV.PACK_INTERNAMENTO.RETORNA_IDADE(PACIENTE.DT_NASCIMENTO, sysdate)      IDADE_PAC,
               CIDADE.NM_CIDADE,
               ESPECIALID.DS_ESPECIALID,
               paciente.nr_fone
   FROM DBAMV.ATENDIME aten,
              DBAMV.PRO_FAT,
              DBAMV.PACIENTE,
              DBAMV.CON_PLA copl,
              DBAMV.CONVENIO conv,
              DBAMV.PRESTADOR pres,
              DBAMV.LEITO  leito,
              DBAMV.MOT_ALT,
              DBAMV.ORI_ATE ori,
              DBAMV.CIDADE,
              DBAMV.ESPECIALID,
              DBAMV.UNID_INT,
             DBAMV.PROCEDIMENTO_SUS
 WHERE aten.TP_ATENDIMENTO IN ( 'I', 'H' ,'B')
       AND aten.DT_ALTA is not null
       AND trunc(aten.DT_ALTA) between trunc(:CAMPO_DATINI) and trunc(:CAMPO_DATFIM)
       AND aten.CD_PACIENTE         = paciente.CD_PACIENTE
       AND aten.CD_PRO_INT           = pro_fat.CD_PRO_FAT(+)
       AND aten.CD_CONVENIO        = copl.CD_CONVENIO(+)
       AND aten.CD_CON_PLA          = copl.CD_CON_PLA(+)
       AND aten.CD_LEITO                = leito.CD_LEITO(+)
       AND aten.CD_CONVENIO        = conv.CD_CONVENIO
       AND aten.CD_PRESTADOR    = pres.CD_PRESTADOR(+)
       AND aten.CD_MOT_ALT          = mot_alt.CD_MOT_ALT(+)
       AND aten.CD_ORI_ATE(+)       = ori.CD_ORI_ATE
       AND PACIENTE.CD_CIDADE = CIDADE.CD_CIDADE (+)
       AND NVL(ATEN.CD_ESPECIALID_SAIDA,ATEN.CD_ESPECIALID) = ESPECIALID.CD_ESPECIALID (+)
       AND LEITO.CD_UNID_INT   = UNID_INT.CD_UNID_INT(+)
      AND  ATEN.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
      AND PROCEDIMENTO_SUS.CD_PROCEDIMENTO (+) =  ATEN.CD_PROCEDIMENTO
      :CAMPO_CF_UNID_INT
)
ORDER BY NM_CONVENIO, CD_CONVENIO, NM_PACIENTE
