--R_HISTORICO_PACI
SELECT ATENDIME.CD_ATENDIMENTO CD_ATENDIMENTO,
       ATENDIME.CD_CASOS_ATD CD_CASOS_ATD,
       ATENDIME.DT_ATENDIMENTO DT_ATENDIMENTO,
       ATENDIME.HR_ATENDIMENTO HR_ATENDIMENTO,
       PACIENTE.NM_SOCIAL_PACIENTE AS NM_SOCIAL_PACIENTE,
       DECODE(ATENDIME.TP_ATENDIMENTO,
              'A',
              'AMBULATORIAL',
              'I',
              'INTERNAÇÃO',
              'E',
              'EXTERNO',
              'U',
              'URGÊNCIA',
              'B',
              'BUSCA ATIVA',
              'H',
              'HOME CARE',
              'S',
              'SUS AIH') TP_ATENDIMENTO,
       prestador.cd_prestador cd_prestador,
       PRESTADOR.NM_PRESTADOR NM_PRESTADOR,
       CASOS_ATD.DS_HDA DS_HDA,
       CASOS_ATD.DS_EX_FIS DS_EX_FIS,
       CASOS_ATD.DS_DIAGNOSTICO DS_DIAGNOSTICO,
       CASOS_ATD.DS_TRATAMENTO DS_TRATAMENTO,
       EVO_CASOS.DS_EVOLUCAO DS_EVOLUCAO
  FROM DBAMV.ATENDIME, DBAMV.PRESTADOR, DBAMV.CASOS_ATD, DBAMV.EVO_CASOS, DBAMV.PACIENTE
 WHERE ATENDIME.CD_PACIENTE = :CAMPO_CD_PACIENTE
 AND PACIENTE.CD_PACIENTE = ATENDIME.CD_PACIENTE
 AND PACIENTE.CD_PACIENTE = :CAMPO_CD_PACIENTE
   AND NVL(CASOS_ATD.CD_PRESTADOR, ATENDIME.CD_PRESTADOR) =
       PRESTADOR.CD_PRESTADOR
   AND ATENDIME.CD_ATENDIMENTO = CASOS_ATD.CD_ATENDIMENTO(+)
   AND ATENDIME.CD_CASOS_ATD = EVO_CASOS.CD_CASOS_ATD(+)
   AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA 
 ORDER BY ATENDIME.DT_ATENDIMENTO
