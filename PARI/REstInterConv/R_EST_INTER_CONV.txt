--R_EST_INTER_CONV
SELECT CONV.CD_CONVENIO,
       CONV.NM_CONVENIO,
       COUNT(*) INTERNACOES
  FROM DBAMV.ATENDIME ATEN,
       DBAMV.CONVENIO CONV,
       DBAMV.LEITO,
       DBAMV.UNID_INT,
       DBAMV.MOV_INT
 WHERE ATEN.DT_ATENDIMENTO BETWEEN Trunc(:CAMPO_DATINI) AND Trunc(:CAMPO_DATFIM)+1-0.00001
   AND ATEN.TP_ATENDIMENTO IN ('I', 'H')
   AND ATEN.CD_ATENDIMENTO_PAI IS NULL
   AND ATEN.CD_CONVENIO = CONV.CD_CONVENIO
   AND ATEN.CD_ATENDIMENTO = MOV_INT.CD_ATENDIMENTO
   AND MOV_INT.TP_MOV = 'I'
   AND MOV_INT.CD_LEITO = LEITO.CD_LEITO(+)
   AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)
   AND ATEN.CD_MULTI_EMPRESA = PKG_MV2000.LE_EMPRESA
 GROUP BY CONV.NM_CONVENIO, CONV.CD_CONVENIO
 ORDER BY INTERNACOES DESC
