--R_FICHAPAC
SELECT PACIENTE.CD_PACIENTE
              ,DBAMV.PKG_PACIENTE.getNomePaciente(PACIENTE.CD_PACIENTE) NM_PACIENTE
              ,PACIENTE.NM_MNEMONICO
              ,PACIENTE.DS_ENDERECO
              ,NVL(to_char(PACIENTE.NR_ENDERECO), ' ') NR_ENDERECO
              ,PACIENTE.DS_COMPLEMENTO
              ,CIDADE.NM_CIDADE
              ,CIDADE.CD_UF
              ,PACIENTE.NM_BAIRRO
              ,PACIENTE.NR_DDD_FONE
              ,PACIENTE.NR_FONE
              ,PACIENTE.NR_DDD_CELULAR
              ,PACIENTE.NR_CELULAR
              ,PACIENTE.NR_CEP
              ,NVL(PACIENTE.NM_PAI,' ') NM_PAI
              ,NVL(PACIENTE.NM_MAE,' ') NM_MAE
              ,PACIENTE.NM_CONJUGE
              ,PACIENTE.DT_CADASTRO
              ,PACIENTE.DT_NASCIMENTO
              ,PACIENTE.NR_DOCUMENTO
              ,to_char(PACIENTE.NR_RG_NASC) NR_RG_NASC
              ,PACIENTE.NR_IDENTIDADE
              ,PACIENTE.DS_OM_IDENTIDADE
              ,PACIENTE.HR_CADASTRO
              ,PACIENTE.DS_TRABALHO
              ,PACIENTE.TP_SANGUINEO
              ,PACIENTE.DS_CHECAPAC
              ,PACIENTE.NM_USUARIO
              ,PACIENTE.NR_CNS
              ,PACIENTE.NR_CPF
              ,PACIENTE.CD_DIS_SAN
              ,PACIENTE.CD_NATURALIDADE
              ,CIDADE_NAT.NM_CIDADE       CIDADE_NATAL
              ,CIDADE_NAT.CD_UF                 UF_NATAL
              ,DIS_SAN.NM_DIS_SAN
              ,PACIENTE.CD_CLA_ECO
              ,CLA_ECO.DS_CLA_ECO
              ,PACIENTE.CD_TIP_MOR
              ,TIP_MOR.DS_TIP_MOR
              ,PACIENTE.CD_TIP_RES
              ,TIP_RES.DS_TIP_RES
              ,PACIENTE.CD_GRAU_INS
              ,GRAU_INS.DS_GRAU_INS
              ,PACIENTE.CD_RELIGIAO
              ,PACIENTE.DS_OBSERVACAO
              ,RELIGIAO.DS_RELIGIAO
              ,PACIENTE.CD_PROFISSAO
              ,PROFISSAO.NM_PROFISSAO
              ,DECODE(PACIENTE.SN_DOADOR,'S','SIM','N')   SN_DOADOR
              ,DECODE(PACIENTE.TP_SITUACAO,'O', 'ÓBITO', 'NORMAL') TP_SITUACAO
              ,DECODE(PACIENTE.TP_SEXO,'M','MASCULINO','F','FEMININO','INDEFINIDO')    TP_SEXO
              ,DBAMV.FNC_ESTADO_CIVIL(PACIENTE.TP_ESTADO_CIVIL) TP_ESTADO_CIVIL1
              ,DECODE(PACIENTE.TP_COR,
                             'B', 'BRANCA',
                             'P','PRETA',
                             'A','AMARELA',
                             'I','INDIGENA',
                             'R','PARDA',
                             'Sem Informação') TP_COR,
        HOSPITAL.DS_RAZAO
        ,DECODE(PACIENTE.SN_UTIL_IDENTIDADE_GENERO,
                      'S',DECODE(
                          (
                          SELECT MAX(1)
                          FROM   DBASGU.CONFIG_ITEM_MODULO
                          WHERE  CD_MODULO = 'CAD_PAC'
                          AND    SN_VISIBLE = 'S'
                          AND    CD_ITEM = 'DSP_DS_IDENTIDADE_GENERO'
                          AND    CD_BLOCO = 'PACIENTE'),
                         1,'Id.Gênero: '||IDENTIDADE_GENERO.DS_IDENTIDADE_GENERO,
                         NULL),
                    NULL) AS IDENTIDADE_GENERO ,
                      DECODE(PACIENTE.SN_UTIL_ORIENTACAO_SEXUAL,
                      'S',DECODE(
                          (
                          SELECT MAX(1)
                          FROM   DBASGU.CONFIG_ITEM_MODULO
                          WHERE  CD_MODULO = 'CAD_PAC'
                          AND    SN_VISIBLE = 'S'
                          AND    CD_ITEM = 'DSP_DS_IDENTIDADE_GENERO'
                          AND    CD_BLOCO = 'PACIENTE'),
                         1,'Orient.Sexual: '||ORIENTACAO_SEXUAL.DS_ORIENTACAO_SEXUAL,
                         NULL),
                    NULL) AS ORIENTACAO_SEXUAL
FROM   DBAMV.PACIENTE
            ,DBAMV.CIDADE
            ,DBAMV.DIS_SAN
            ,DBAMV.CLA_ECO
            ,DBAMV.TIP_MOR
            ,DBAMV.TIP_RES
            ,DBAMV.GRAU_INS
            ,DBAMV.RELIGIAO
            ,DBAMV.PROFISSAO
            ,DBAMV.CIDADE  CIDADE_NAT
            ,DBAMV.HOSPITAL
            ,DBAMV.ORIENTACAO_SEXUAL
            ,DBAMV.IDENTIDADE_GENERO
WHERE  PACIENTE.CD_PACIENTE = :CAMPO_CD_PACIENTE
AND        PACIENTE.CD_CIDADE = CIDADE.CD_CIDADE(+)
AND        PACIENTE.CD_DIS_SAN = DIS_SAN.CD_DIS_SAN(+)
AND        PACIENTE.CD_CLA_ECO = CLA_ECO.CD_CLA_ECO(+)
AND        PACIENTE.CD_TIP_MOR = TIP_MOR.CD_TIP_MOR(+)
AND        PACIENTE.CD_TIP_RES = TIP_RES.CD_TIP_RES(+)
AND        PACIENTE.CD_GRAU_INS = GRAU_INS.CD_GRAU_INS(+)
AND        PACIENTE.CD_RELIGIAO = RELIGIAO.CD_RELIGIAO(+)
AND        PACIENTE.CD_PROFISSAO = PROFISSAO.CD_PROFISSAO(+)
AND        PACIENTE.CD_NATURALIDADE = CIDADE_NAT.CD_CIDADE(+)
AND        HOSPITAL.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
AND        PACIENTE.CD_IDENTIDADE_GENERO = IDENTIDADE_GENERO.CD_IDENTIDADE_GENERO(+)
AND        PACIENTE.CD_ORIENTACAO_SEXUAL = ORIENTACAO_SEXUAL.CD_ORIENTACAO_SEXUAL(+)
