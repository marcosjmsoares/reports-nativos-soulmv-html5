--R_CENSO_RETRO860
SELECT * FROM (
SELECT  LEITO.CD_UNID_INT
               ,UNID_INT.DS_UNID_INT
               ,CONVENIO.CD_CONVENIO
               ,ESPEC_SUS.CD_ESPEC_SUS
               ,NVL(ESPEC_SUS.DS_ESPEC_SUS,'NÃO INFORMADA') DS_ESPEC_SUS
               ,LEITO.DS_RESUMO
               ,LEITO.DS_ENFERMARIA
               ,NVL( DBAMV.VER_DT_LIBERACAO(:CAMPO_DT_EMISSAO, MOV_INT.DT_LIB_MOV, MOV_INT.TP_MOV), 'V')    TP_OCUPACAO
               ,DECODE(DBAMV.VER_DT_LIBERACAO(:CAMPO_DT_EMISSAO, MOV_INT.DT_LIB_MOV, MOV_INT.TP_MOV),  'V', NULL, PACIENTE.NM_PACIENTE)           NM_PACIENTE
               ,DECODE(DBAMV.VER_DT_LIBERACAO(:CAMPO_DT_EMISSAO, MOV_INT.DT_LIB_MOV, MOV_INT.TP_MOV),  'V', NULL, PACIENTE.DT_NASCIMENTO)      DT_NASCIMENTO
               ,NVL(DECODE(DBAMV.VER_DT_LIBERACAO(:CAMPO_DT_EMISSAO, MOV_INT.DT_LIB_MOV, MOV_INT.TP_MOV),  'V', NULL, CONVENIO.NM_CONVENIO),'NÃO INFORMADO')        NM_CONVENIO
               ,DECODE(DBAMV.VER_DT_LIBERACAO(:CAMPO_DT_EMISSAO, MOV_INT.DT_LIB_MOV, MOV_INT.TP_MOV),  'V', NULL, PRESTADOR.NM_PRESTADOR)  NM_PRESTADOR
               ,DECODE(DBAMV.VER_DT_LIBERACAO(:CAMPO_DT_EMISSAO, MOV_INT.DT_LIB_MOV, MOV_INT.TP_MOV),  'V', TO_DATE(NULL), ATENDIME.DT_ATENDIMENTO)  DT_ATENDIMENTO
               ,DECODE(DBAMV.VER_DT_LIBERACAO(:CAMPO_DT_EMISSAO, MOV_INT.DT_LIB_MOV, MOV_INT.TP_MOV),  'V', NULL, ATENDIME.CD_ATENDIMENTO)  CD_ATENDIMENTO
               ,LEITO.CD_LEITO
               ,NVL(TRUNC( :CAMPO_DT_EMISSAO - DECODE(DBAMV.VER_DT_LIBERACAO(:CAMPO_DT_EMISSAO, MOV_INT.DT_LIB_MOV, MOV_INT.TP_MOV),  'V', TO_DATE(NULL), ATENDIME.DT_ATENDIMENTO) ),0) DIFER_DIAS
FROM    DBAMV.MOV_INT
              ,DBAMV.LEITO
              ,DBAMV.UNID_INT
              ,DBAMV.ATENDIME
              ,DBAMV.PACIENTE
              ,DBAMV.CONVENIO
              ,DBAMV.PRESTADOR
              ,DBAMV.PRO_FAT
              ,DBAMV.ESPEC_SUS
              , (SELECT MAX(MOV_INT.CD_MOV_INT) CD_MOV_INT, LEITO.CD_LEITO
                FROM   DBAMV.MOV_INT MOV_INT,
                       DBAMV.LEITO LEITO
                WHERE  TRUNC(DT_MOV_INT)  <= :CAMPO_DT_EMISSAO
                AND    LEITO.CD_LEITO = MOV_INT.CD_LEITO
                GROUP BY LEITO.CD_LEITO) mov
WHERE   MOV.CD_LEITO(+) = LEITO.CD_LEITO
AND         MOV_INT.CD_ATENDIMENTO = ATENDIME.CD_ATENDIMENTO(+)
AND         ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE (+)
AND         ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO (+)
AND         ATENDIME.CD_PRESTADOR  = PRESTADOR.CD_PRESTADOR(+)
AND         LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT
AND         ATENDIME.CD_PRO_INT  =  PRO_FAT.CD_PRO_FAT(+)
AND         PRO_FAT.CD_ESPEC_SUS  =  ESPEC_SUS.CD_ESPEC_SUS(+)
AND         MOV_INT.CD_MOV_INT = MOV.CD_MOV_INT
AND        ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
union all
(select  LEITO.CD_UNID_INT
               ,UNID_INT.DS_UNID_INT
               ,to_number(null) CD_CONVENIO
               ,null CD_ESPEC_SUS
               ,'NÃO INFORMADA' DS_ESPEC_SUS
               ,LEITO.DS_RESUMO
               ,LEITO.DS_ENFERMARIA
               ,'V' TP_OCUPACAO
               ,null NM_PACIENTE
               ,null DT_NASCIMENTO
               ,'NÃO INFORMADO' NM_CONVENIO
               ,null NM_PRESTADOR
               ,TO_DATE(null) DT_ATENDIMENTO
               ,null CD_ATENDIMENTO
               ,LEITO.CD_LEITO
               ,0 DIFER_DIAS
from   dbamv.leito leito
      ,dbamv.unid_int unid_int
      ,(select distinct mov.cd_leito
       from dbamv.mov_int mov) MOV
where  LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT
AND    LEITO.CD_LEITO = MOV.CD_LEITO(+)
AND    MOV.CD_LEITO IS NULL
:CAMPO_CF_IMP_VAGO2
)
:CAMPO_CF_ORDEM
)
ORDER BY DS_UNID_INT,CD_UNID_INT,
DS_ESPEC_SUS,CD_ESPEC_SUS,
NM_CONVENIO,CD_CONVENIO,CD_ATENDIMENTO
