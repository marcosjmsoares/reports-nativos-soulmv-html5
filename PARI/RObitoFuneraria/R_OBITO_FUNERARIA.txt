--R_OBITO_FUNERARIA
SELECT * FROM
			  (SELECT atendime.cd_paciente cd_paciente,
			          paciente.nm_paciente nm_paciente,
			     (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND ATENDIME.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE,
			          atendime.dt_atendimento dt_atendimento,
			          to_date(to_char(nvl(atendime.dt_alta_medica, atendime.dt_alta), 'dd/mm/yyyy')||' '||TO_CHAR(nvl(atendime.hr_alta_medica, atendime.dt_alta), 'hh24'), 'dd/mm/yyyy hh24') dthr_alta_medica,
			          funeraria.cd_funeraria cd_funeraria,
			          funeraria.nm_funeraria nm_funeraria,
			          cad_obito.nr_doc_obito nr_doc_obito,
			          atendime.cd_atendimento cd_atendimento
			   FROM dbamv.atendime atendime,
			        dbamv.paciente paciente,
			        dbamv.mot_alt mot_alt,
			        dbamv.cad_obito cad_obito,
			        dbamv.funeraria funeraria
			   WHERE trunc(atendime.dt_alta) BETWEEN :CAMPO_DTINIC AND 
			     AND atendime.cd_paciente = paciente.cd_paciente
			     AND atendime.cd_mot_alt = mot_alt.cd_mot_alt
			     AND atendime.dt_alta IS NOT NULL
			     AND mot_alt.tp_mot_alta = 'O'
			     AND atendime.cd_atendimento = cad_obito.cd_atendimento
			     AND cad_obito.cd_funeraria = funeraria.cd_funeraria
			     AND atendime.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA 
			   ORDER BY atendime.dt_alta,
			            paciente.nm_paciente)
			ORDER BY cd_funeraria,
			         nm_paciente
