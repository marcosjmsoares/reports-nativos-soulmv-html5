              WHERE CD_TIP_RES = ATENDIME.CD_TIP_RES
                AND TIP_RES.SN_OBITO =  'S'
             )
     )
 Group  By
 UNID_INT.CD_UNID_INT,
 UNID_INT.DS_UNID_INT,
 CONVENIO.TP_CONVENIO
)
Group By
  CD_UNID_INT,
  DS_UNIDADE,
  tp_convenio
ORDER BY DS_UNIDADE
)
ORDER BY CV_CD_UNID_INT,CV_DS_UNIDADE
