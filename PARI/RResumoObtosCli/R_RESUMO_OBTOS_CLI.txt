--R_RESUMO_OBTOS_CLI
SELECT * FROM (
SELECT
 ATENDIME.CD_ORI_ATE
,ORI_ATE.DS_ORI_ATE
,ATENDIME.CD_ESPECIALID
,ESPECIALID.DS_ESPECIALID
,SUM( DECODE(  TRUNC(TO_DATE( TO_CHAR(ATENDIME.DT_ALTA,'DD/MM/YYYY') || TO_CHAR(HR_ALTA,'HH24'),'DD/MM/YYYY HH24' )  -
                             TO_DATE( TO_CHAR(ATENDIME.DT_ATENDIMENTO,'DD/MM/YYYY') || TO_CHAR(HR_ATENDIMENTO,'HH24'),'DD/MM/YYYY HH24')),  0, 1, 0) )   ATE_24
,SUM( DECODE(  TRUNC(TO_DATE( TO_CHAR(ATENDIME.DT_ALTA,'DD/MM/YYYY') || TO_CHAR(HR_ALTA,'HH24'),'DD/MM/YYYY HH24' )  -
                             TO_DATE( TO_CHAR(ATENDIME.DT_ATENDIMENTO,'DD/MM/YYYY') || TO_CHAR(HR_ATENDIMENTO,'HH24'),'DD/MM/YYYY HH24')),  0, 0, 1) )   DEPOIS_24
FROM      DBAMV.ATENDIME
               ,DBAMV.ORI_ATE
               ,DBAMV.ESPECIALID
               ,DBAMV.MOT_ALT
WHERE TRUNC(ATENDIME.DT_ALTA)  BETWEEN  :CAMPO_DT_INICIAL  AND  
AND       ATENDIME.CD_ORI_ATE  =  ORI_ATE.CD_ORI_ATE
AND       ATENDIME.CD_ESPECIALID  =  ESPECIALID.CD_ESPECIALID(+)
AND       ATENDIME.DT_ALTA  IS  NOT NULL
AND       ATENDIME.CD_MOT_ALT  =  MOT_ALT.CD_MOT_ALT
AND       MOT_ALT.TP_MOT_ALTA  =  'O'
AND       ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA  
GROUP BY  ATENDIME.CD_ORI_ATE
                   ,ORI_ATE.DS_ORI_ATE
                   ,ATENDIME.CD_ESPECIALID
                   ,ESPECIALID.DS_ESPECIALID
ORDER BY  ORI_ATE.DS_ORI_ATE
                    ,ESPECIALID.DS_ESPECIALID
)
ORDER BY DS_ORI_ATE,CD_ORI_ATE
