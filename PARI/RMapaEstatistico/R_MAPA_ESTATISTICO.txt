--R_MAPA_ESTATISTICO
SELECT
			DESCRICAO,
			MES_01,
			MES_02,
			MES_03,
			MES_04,
			MES_05,
			MES_06,
			MES_07,
			MES_08,
			MES_09,
			MES_10,
			MES_11,
			MES_12,
			TOT_ANO,
			(
				SELECT COUNT(*)
				FROM      DBAMV.V_MAPA_ESTATISTICO    V
				WHERE  TO_CHAR( V.DTA_ATENDIMENTO, 'YYYY' ) =  :CAMPO_ANO
				:CAMPO_COND_CD_CONVENIO
			) TOT_ANO_GERAL,
			NVL(CALC_QTD_DIARIAS(:CAMPO_ANO, '01', DESCRICAO, :CAMPO_TP_COLUNA, :CAMPO_CF_TP_ATENDE), 0) QTD_DIARIAS_M01,
			NVL(CALC_QTD_DIARIAS(:CAMPO_ANO, '02', DESCRICAO, :CAMPO_TP_COLUNA, :CAMPO_CF_TP_ATENDE), 0) QTD_DIARIAS_M02,
			NVL(CALC_QTD_DIARIAS(:CAMPO_ANO, '03', DESCRICAO, :CAMPO_TP_COLUNA, :CAMPO_CF_TP_ATENDE), 0) QTD_DIARIAS_M03,
			NVL(CALC_QTD_DIARIAS(:CAMPO_ANO, '04', DESCRICAO, :CAMPO_TP_COLUNA, :CAMPO_CF_TP_ATENDE), 0) QTD_DIARIAS_M04,
			NVL(CALC_QTD_DIARIAS(:CAMPO_ANO, '05', DESCRICAO, :CAMPO_TP_COLUNA, :CAMPO_CF_TP_ATENDE), 0) QTD_DIARIAS_M05,
			NVL(CALC_QTD_DIARIAS(:CAMPO_ANO, '06', DESCRICAO, :CAMPO_TP_COLUNA, :CAMPO_CF_TP_ATENDE), 0) QTD_DIARIAS_M06,
			NVL(CALC_QTD_DIARIAS(:CAMPO_ANO, '07', DESCRICAO, :CAMPO_TP_COLUNA, :CAMPO_CF_TP_ATENDE), 0) QTD_DIARIAS_M07,
			NVL(CALC_QTD_DIARIAS(:CAMPO_ANO, '08', DESCRICAO, :CAMPO_TP_COLUNA, :CAMPO_CF_TP_ATENDE), 0) QTD_DIARIAS_M08,
			NVL(CALC_QTD_DIARIAS(:CAMPO_ANO, '09', DESCRICAO, :CAMPO_TP_COLUNA, :CAMPO_CF_TP_ATENDE), 0) QTD_DIARIAS_M09,
			NVL(CALC_QTD_DIARIAS(:CAMPO_ANO, '10', DESCRICAO, :CAMPO_TP_COLUNA, :CAMPO_CF_TP_ATENDE), 0) QTD_DIARIAS_M10,
			NVL(CALC_QTD_DIARIAS(:CAMPO_ANO, '11', DESCRICAO, :CAMPO_TP_COLUNA, :CAMPO_CF_TP_ATENDE), 0) QTD_DIARIAS_M11,
			NVL(CALC_QTD_DIARIAS(:CAMPO_ANO, '12', DESCRICAO, :CAMPO_TP_COLUNA, :CAMPO_CF_TP_ATENDE), 0) QTD_DIARIAS_M12
		FROM
		(
			SELECT  NVL(DESCRICAO,'NÃO INFORMADO') DESCRICAO
	               ,SUM(MES_01)  MES_01
	               ,SUM(MES_02)  MES_02
	               ,SUM(MES_03)  MES_03
	               ,SUM(MES_04)  MES_04
	               ,SUM(MES_05)  MES_05
	               ,SUM(MES_06)  MES_06
	               ,SUM(MES_07)  MES_07
	               ,SUM(MES_08)  MES_08
	               ,SUM(MES_09)  MES_09
	               ,SUM(MES_10)  MES_10
	               ,SUM(MES_11)  MES_11
	               ,SUM(MES_12)  MES_12
	               ,SUM(TOT_ANO) TOT_ANO
			FROM
			(
				SELECT                                                           DESCRICAO,
				                  COUNT( DECODE( TO_CHAR( V.DTA_ATENDIMENTO, 'MM' ), 01, 1, NULL ) ) MES_01,
				                  COUNT( DECODE( TO_CHAR( V.DTA_ATENDIMENTO, 'MM' ), 02, 1, NULL ) ) MES_02,
				                  COUNT( DECODE( TO_CHAR( V.DTA_ATENDIMENTO, 'MM' ), 03, 1, NULL ) ) MES_03,
				                  COUNT( DECODE( TO_CHAR( V.DTA_ATENDIMENTO, 'MM' ), 04, 1, NULL ) ) MES_04,
				                  COUNT( DECODE( TO_CHAR( V.DTA_ATENDIMENTO, 'MM' ), 05, 1, NULL ) ) MES_05,
				                  COUNT( DECODE( TO_CHAR( V.DTA_ATENDIMENTO, 'MM' ), 06, 1, NULL ) ) MES_06,
				                  COUNT( DECODE( TO_CHAR( V.DTA_ATENDIMENTO, 'MM' ), 07, 1, NULL ) ) MES_07,
				                  COUNT( DECODE( TO_CHAR( V.DTA_ATENDIMENTO, 'MM' ), 08, 1, NULL ) ) MES_08,
				                  COUNT( DECODE( TO_CHAR( V.DTA_ATENDIMENTO, 'MM' ), 09, 1, NULL ) ) MES_09,
				                  COUNT( DECODE( TO_CHAR( V.DTA_ATENDIMENTO, 'MM' ), 10, 1, NULL ) ) MES_10,
				                  COUNT( DECODE( TO_CHAR( V.DTA_ATENDIMENTO, 'MM' ), 11, 1, NULL ) ) MES_11,
				                  COUNT( DECODE( TO_CHAR( V.DTA_ATENDIMENTO, 'MM' ), 12, 1, NULL ) ) MES_12,
				                  COUNT(*)                                                           TOT_ANO
				FROM      DBAMV.V_MAPA_ESTATISTICO    V
				WHERE  TO_CHAR( V.DTA_ATENDIMENTO, 'YYYY' ) =  :CAMPO_ANO
				GROUP BY  
				UNION ALL
				SELECT  DISTINCT
				                     DESCRICAO
				               ,0                               MES_01
				               ,0                               MES_02
				               ,0                               MES_03
				               ,0                               MES_04
				               ,0                               MES_05
				               ,0                               MES_06
				               ,0                               MES_07
				               ,0                               MES_08
				               ,0                               MES_09
				               ,0                               MES_10
				               ,0                               MES_11
				               ,0                               MES_12
				               ,0                               TOT_ANO
				FROM     DBAMV.V_MAPA_ESTATISTICO V
				WHERE  V.TIPO_ATEND = 'I'
				AND        ( (V.DTA_ALTA IS NULL) OR  (TO_CHAR(DTA_ALTA,'YYYY/MM') >= (:CAMPO_ANO || '/01')) )
			)
			GROUP BY  NVL(DESCRICAO,'NÃO INFORMADO')
			ORDER BY  TOT_ANO  DESC
		)

               ,SUM(MES_01_O)       MES_01_O

--R_MAPA_ESTATISTICO_R_17_Q_3
SELECT   DESCRICAO    DESC_DIARIA_M01_0
               ,SUM( (DTA_ALTA - DTA_ATENDIMENTO) + 1 )    QTD_DIARIAS_M01_0
FROM (
                 SELECT  V.DTA_ATENDIMENTO
                                ,NVL(V.DTA_ALTA,  SYSDATE)   DTA_ALTA
                                ,:CAMPO_TP_COLUNA                  DESCRICAO
                 FROM    DBAMV.V_MAPA_ESTATISTICO V
                 WHERE  V.TIPO_ATEND = 'I'
                 AND    TO_CHAR( V.DTA_ATENDIMENTO, 'YYYY/MM' ) = (:CAMPO_ANO || '/01' )
                 AND    NVL(TO_CHAR(V.DTA_ALTA,'YYYY/MM'),TO_CHAR(SYSDATE,'YYYY/MM')) = (:CAMPO_ANO || '/01' )
                 UNION ALL 
                SELECT  V.DTA_ATENDIMENTO
                              ,LAST_DAY(TO_DATE( '01/' || :CAMPO_ANO,'MM/YYYY'))    DTA_ALTA
                              ,:CAMPO_TP_COLUNA                  DESCRICAO
                FROM    DBAMV.V_MAPA_ESTATISTICO V
                WHERE  V.TIPO_ATEND = 'I'
                AND    TO_CHAR( V.DTA_ATENDIMENTO, 'YYYY/MM' ) = (:CAMPO_ANO || '/01' )
                AND    NVL(TO_CHAR(V.DTA_ALTA,'YYYY/MM'),TO_CHAR(SYSDATE,'YYYY/MM')) > (:CAMPO_ANO || '/01' )
                UNION ALL
               SELECT  TO_DATE( ('01/01' || :CAMPO_ANO) ,'DD/MM/YYYY')    DTA_ATENDIMENTO
                              ,NVL(V.DTA_ALTA,  SYSDATE)   DTA_ALTA
                              ,:CAMPO_TP_COLUNA                  DESCRICAO
                FROM    DBAMV.V_MAPA_ESTATISTICO V
                WHERE  V.TIPO_ATEND = 'I'
                AND    TO_CHAR( V.DTA_ATENDIMENTO, 'YYYY/MM' ) < (:CAMPO_ANO || '/01' )
                AND    NVL(TO_CHAR(V.DTA_ALTA,'YYYY/MM'),TO_CHAR(SYSDATE,'YYYY/MM')) = (:CAMPO_ANO || '/01' )
                UNION ALL
                SELECT  TO_DATE( ('01/01' || :CAMPO_ANO) ,'DD/MM/YYYY')           DTA_ATENDIMENTO
                               ,LAST_DAY(TO_DATE( '01/' || :CAMPO_ANO,'MM/YYYY'))     DTA_ALTA
                               ,:CAMPO_TP_COLUNA                  DESCRICAO
                FROM    DBAMV.V_MAPA_ESTATISTICO V
                WHERE  V.TIPO_ATEND = 'I'
                AND    TO_CHAR( V.DTA_ATENDIMENTO, 'YYYY/MM' ) < (:CAMPO_ANO || '/01' )
                AND    NVL(TO_CHAR(V.DTA_ALTA,'YYYY/MM'),TO_CHAR(SYSDATE,'YYYY/MM')) > (:CAMPO_ANO || '/01' )
                :CAMPO_CF_TP_ATENDE
 )
GROUP BY  DESCRICAO
