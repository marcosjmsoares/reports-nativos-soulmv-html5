--R_EST_ATEND_USUARIO
SELECT * FROM (
SELECT  TRUNC(ATENDIME.DT_ATENDIMENTO)     DT_ATENDIMENTO
               ,ATENDIME.NM_USUARIO                            NM_USUARIO
               ,COUNT(ATENDIME.CD_ATENDIMENTO)                                                        QT_ATENDE
FROM    DBAMV.ATENDIME
WHERE  TRUNC(ATENDIME.DT_ATENDIMENTO)  BETWEEN  :CAMPO_DT_INIC  AND 
AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
GROUP BY TRUNC(ATENDIME.DT_ATENDIMENTO)
                   ,ATENDIME.NM_USUARIO
ORDER BY  QT_ATENDE  DESC
)
ORDER BY DT_ATENDIMENTO
