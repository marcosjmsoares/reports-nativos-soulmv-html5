--R_PAC_INT_ACOMOD_RF_UNIDADE1_Q_ATEND_ACOMOD
SELECT UNID_INT.CD_UNID_INT ,
       UNID_INT.DS_UNID_INT DS_UNID_INT ,
       ATENDIME.CD_ATENDIMENTO CD_ATENDIMENTO ,
       PACIENTE.CD_PACIENTE CD_PACIENTE ,
   (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
       PACIENTE.NM_PACIENTE NM_PACIENTE ,
       DBAMV.PACK_INTERNAMENTO.RETORNA_IDADE(PACIENTE.DT_NASCIMENTO, sysdate) IDADE_PAC ,
       LEITO.DS_RESUMO DS_RESUMO ,
       LEITO.DS_ENFERMARIA DS_ENFERMARIA ,
       TIP_ACOM.DS_TIP_ACOM DS_TIP_ACOM ,
       ATENDIME.DT_ATENDIMENTO DT_ATENDIMENTO ,
       ATENDIME.HR_ATENDIMENTO HR_ATENDIMENTO ,
       CONVENIO.CD_CONVENIO ,
       CONVENIO.NM_CONVENIO ,
       CON_PLA.DS_CON_PLA
FROM DBAMV.ATENDIME ,
     DBAMV.PACIENTE ,
     DBAMV.LEITO ,
     DBAMV.TIP_ACOM ,
     DBAMV.UNID_INT ,
     DBAMV.CON_PLA ,
     DBAMV.CONVENIO
WHERE ATENDIME.DT_ALTA IS NULL
  AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
  AND ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
  AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)
  AND ATENDIME.CD_TIP_ACOM = TIP_ACOM.CD_TIP_ACOM
  AND ATENDIME.CD_CONVENIO = CONVENIO.CD_CONVENIO
  AND ATENDIME.CD_CONVENIO = CON_PLA.CD_CONVENIO
  AND ATENDIME.CD_CON_PLA = CON_PLA.CD_CON_PLA
  AND ATENDIME.TP_ATENDIMENTO = 'I'
  AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA    
ORDER BY
