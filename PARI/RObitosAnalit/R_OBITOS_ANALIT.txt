--R_OBITOS_ANALIT
SELECT ATENDIME.CD_ATENDIMENTO CD_ATENDIMENTO ,
			       PACIENTE.CD_PACIENTE CD_PACIENTE ,
			       PACIENTE.NM_PACIENTE NM_PACIENTE ,
			   (SELECT P.NM_SOCIAL_PACIENTE FROM DBAMV.PACIENTE P WHERE NVL(SN_UTILIZA_NOME_SOCIAL, 'N') = 'S' AND PACIENTE.CD_PACIENTE =  P.CD_PACIENTE) NM_SOCIAL_PACIENTE ,
			       ATENDIME.DT_ATENDIMENTO DT_ATENDIMENTO ,
			       ATENDIME.HR_ATENDIMENTO HR_ATENDIMENTO ,
			       nvl(atendime.dt_alta_medica, atendime.dt_alta) DT_ALTA ,
			       nvl(atendime.hr_alta_medica, atendime.hr_alta) HR_ALTA ,
			       PACIENTE.DT_NASCIMENTO DT_NASCIMENTO ,
			       PACIENTE.TP_SEXO TP_SEXO ,
			       LOC_TRANS.DS_LOC_TRANS DS_LOC_TRANS ,
			       UNID_INT.CD_UNID_INT CD_UNID_INT ,
			       UNID_INT.DS_UNID_INT DS_UNID_INT1 ,
			       (Trunc(DT_ALTA) - Trunc(DT_ATENDIMENTO) || ' dd') PERMANENCIA ,
			       dbamv.FN_IDADE(PACIENTE.DT_NASCIMENTO, 'a A', ATENDIME.DT_ALTA) IDADE
			FROM DBAMV.ATENDIME ,
			     DBAMV.PACIENTE ,
			     DBAMV.LEITO ,
			     DBAMV.MOT_ALT ,
			     DBAMV.UNID_INT ,
			     DBAMV.LOC_TRANS ,
			     DBAMV.FAIXA_ETARIA_RELATORIO
			WHERE ATENDIME.TP_ATENDIMENTO = 'I'
			  AND ATENDIME.DT_ALTA IS NOT NULL
			  AND ATENDIME.DT_ALTA BETWEEN To_Date(Concat(To_Char(:CAMPO_DT_INI,'dd/mm/yyyy'), ' 00'), 'dd/mm/yyyy hh24') AND To_Date(Concat(To_Char(:CAMPO_DT_FIM,'dd/mm/yyyy'), ' 23'), 'dd/mm/yyyy hh24')
			  AND ATENDIME.CD_PACIENTE = PACIENTE.CD_PACIENTE
			  AND ATENDIME.CD_LEITO = LEITO.CD_LEITO(+)
			  AND LEITO.CD_UNID_INT = UNID_INT.CD_UNID_INT(+)
			  AND ATENDIME.CD_MOT_ALT = MOT_ALT.CD_MOT_ALT(+)
			  AND ATENDIME.CD_LOC_TRANS = LOC_TRANS.CD_LOC_TRANS(+)
			  AND MOT_ALT.TP_MOT_ALTA = 'O'
			  AND DBAMV.RETORNA_FAIXA_ETARIA_RELATORIO(PACIENTE.DT_NASCIMENTO, 'C', ATENDIME.DT_ALTA) = DBAMV.FAIXA_ETARIA_RELATORIO.CD_FAIXA_ETARIA_RELATORIO
			  AND ATENDIME.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA       
			ORDER BY DT_ALTA ASC
