--R_CASO_POSITIVO_CITO
SELECT  SISCO_CITO.CD_SISCO_CITO CD_SISCO_CITO
       , SISCO_CITO.CD_PACIENTE CD_PACIENTE
       ,PACIENTE.NM_PACIENTE NM_PACIENTE
       ,PACIENTE.DS_ENDERECO DS_ENDERECO
       ,PACIENTE.DS_COMPLEMENTO DS_COMPLEMENTO
       ,PACIENTE.NR_ENDERECO NR_ENDERECO
       ,PACIENTE.NM_BAIRRO NM_BAIRRO
       ,CIDADE_PAC.CD_CIDADE CD_CIDADE_PAC
       ,CIDADE_PAC.NM_CIDADE NM_CIDADE_PAC
       ,CIDADE_PAC.CD_UF CD_UF_PAC
      ,SISCO_CITO.TP_ALT_CEL_ESC TP_ALT_CEL_ESC
       ,SISCO_CITO.TP_ALT_CEL_GLA TP_ALT_CEL_GLA
        , SISCO_CITO.SN_ATI_ESC SN_ATI_ESC
       , SISCO_CITO.SN_ATI_GLA SN_ATI_GLA
        , SISCO_CITO.ds_outros3   DS_OUTROS3
       ,SISCO_CITO.SN_COMP_HPV SN_COMP_HPV
       ,MULTI_EMPRESAS.DS_MULTI_EMPRESA DS_MULTI_EMPRESA
       ,MULTI_EMPRESAS.CD_CIDADE CD_CIDADE_HOS
       ,CIDADE_HOS.NM_CIDADE NM_CIDADE_HOS
       ,CIDADE_HOS.CD_IBGE CD_IBGE_HOS
       ,CIDADE_HOS.CD_UF CD_UF_HOS
       ,MULTI_EMPRESAS.CD_CGC CD_CGC_HOS
       ,DECODE(SISCO_UNI_SER.CD_UNI_SER, NULL, HOSPITAL.CD_UNIDADE, SISCO_UNI_SER.CD_UNI_SER) CD_UNI_SER
       ,DECODE(SISCO_UNI_SER.DS_UNI_SER, NULL, MULTI_EMPRESAS.DS_MULTI_EMPRESA, SISCO_UNI_SER.DS_UNI_SER) DS_UNI_SER
 FROM DBAMV.SISCO_CITO SISCO_CITO
      ,DBAMV.PACIENTE PACIENTE
      ,DBAMV.CIDADE CIDADE_PAC
      ,DBAMV.MULTI_EMPRESAS MULTI_EMPRESAS
      ,DBAMV.CIDADE CIDADE_HOS
      ,DBAMV.SISCO_UNI_SER SISCO_UNI_SER
      ,DBAMV.HOSPITAL HOSPITAL
WHERE SISCO_CITO.CD_FAT_SIA = :CAMPO_FAT_SIA
  AND SISCO_CITO.CD_PACIENTE = PACIENTE.CD_PACIENTE
  AND PACIENTE.CD_CIDADE = CIDADE_PAC.CD_CIDADE
  AND MULTI_EMPRESAS.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
  AND MULTI_EMPRESAS.CD_CIDADE = CIDADE_HOS.CD_CIDADE
  AND SISCO_CITO.CD_UNI_SER = SISCO_UNI_SER.CD_UNI_SER(+)
  AND Hospital.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA

--R_CASO_POSITIVO_CITO_R_4_Q_HISTOPATOLOGICO
SELECT * FROM (
SELECT  SISCO_HISTO.CD_SISCO_HISTO
       , SISCO_HISTO.CD_PACIENTE CD_PACIENTE1
       ,PACIENTE.NM_PACIENTE NM_PACIENTE1
       ,PACIENTE.DS_ENDERECO DS_ENDERECO1
       ,PACIENTE.DS_COMPLEMENTO DS_COMPLEMENTO1
       ,PACIENTE.NR_ENDERECO NR_ENDERECO1
       ,PACIENTE.NM_BAIRRO NM_BAIRRO1
       ,CIDADE_PAC.CD_CIDADE CD_CIDADE_PAC1
       ,CIDADE_PAC.NM_CIDADE NM_CIDADE_PAC1
       ,CIDADE_PAC.CD_UF CD_UF_PAC1
       ,SISCO_HISTO.TP_LESAO_NEO TP_LESAO_NEO
       ,SISCO_HISTO.TP_LESAO_NEO2 TP_LESAO_NEO2
       , SISCO_HISTO.DS_OUT_LESAO_NEO DS_OUT_LESAO_NEO
       , SISCO_HISTO.SN_MET_ESC SN_MET_ESC
       , SISCO_HISTO.SN_POL_END SN_POL_END
       , SISCO_HISTO.SN_CER_CRO SN_CER_CRO
       , SISCO_HISTO.SN_ALT_CITO SN_ALT_CITO
       ,MULTI_EMPRESAS.DS_MULTI_EMPRESA DS_MULTI_EMPRESA1
       ,MULTI_EMPRESAS.CD_CIDADE CD_CIDADE_HOS1
       ,CIDADE_HOS.NM_CIDADE NM_CIDADE_HOS1
       ,CIDADE_HOS.CD_IBGE CD_IBGE_HOS1
       ,CIDADE_HOS.CD_UF CD_UF_HOS1
       ,MULTI_EMPRESAS.CD_CGC CD_CGC_HOS1
       ,DECODE(SISCO_UNI_SER.CD_UNI_SER, NULL, HOSPITAL.CD_UNIDADE, SISCO_UNI_SER.CD_UNI_SER) CD_UNI_SER1
       ,DECODE(SISCO_UNI_SER.DS_UNI_SER, NULL, MULTI_EMPRESAS.DS_MULTI_EMPRESA, SISCO_UNI_SER.DS_UNI_SER) DS_UNI_SER1
 FROM DBAMV.SISCO_HISTO SISCO_HISTO
      ,DBAMV.PACIENTE PACIENTE
      ,DBAMV.CIDADE CIDADE_PAC
      ,DBAMV.MULTI_EMPRESAS MULTI_EMPRESAS
      ,DBAMV.CIDADE CIDADE_HOS
      ,DBAMV.SISCO_UNI_SER SISCO_UNI_SER
      ,DBAMV.HOSPITAL HOSPITAL
WHERE SISCO_HISTO.CD_FAT_SIA = :CAMPO_FAT_SIA
  AND SISCO_HISTO.CD_PACIENTE = PACIENTE.CD_PACIENTE
  AND PACIENTE.CD_CIDADE = CIDADE_PAC.CD_CIDADE
  AND MULTI_EMPRESAS.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
  AND MULTI_EMPRESAS.CD_CIDADE = CIDADE_HOS.CD_CIDADE
  AND SISCO_HISTO.CD_UNI_SER = SISCO_UNI_SER.CD_UNI_SER(+)
  AND Hospital.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
)
ORDER BY CD_UNI_SER1,DS_UNI_SER1
