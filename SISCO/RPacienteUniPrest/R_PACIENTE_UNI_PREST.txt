--R_PACIENTE_UNI_PREST_R_HISTO_Q_HISTO
SELECT * FROM (
SELECT MULTI_EMPRESAS.CD_MULTI_EMPRESA
     , MULTI_EMPRESAS.DS_MULTI_EMPRESA
     , MULTI_EMPRESAS.NR_CNES
     , MULTI_EMPRESAS.CD_CGC
     , CIDADE.NM_CIDADE
     , CIDADE.CD_UF
     , SISCO_UNI_SER.CD_UNI_SER
     , SISCO_UNI_SER.DS_UNI_SER
     , CIDADE_UNIDADE.NM_CIDADE NM_CIDADE_unid
     , SISCO_UNI_SER.NR_CNES NR_CNES_unid
     , CIDADE_UNIDADE.CD_UF CD_UF_unid
     , SISCO_HISTO.CD_SISCO_HISTO
     , PACIENTE.CD_PACIENTE
     , PACIENTE.NM_PACIENTE
     , PACIENTE.DT_NASCIMENTO
     , SISCO_HISTO.TP_PROC_CIRU
     , SISCO_HISTO.TP_LESAO_NEO
     , SISCO_HISTO.TP_LESAO_NEO2
     , SISCO_HISTO.DS_OUT_LESAO_NEO
     , SISCO_HISTO.SN_INSATISFATORIO
     , SISCO_HISTO.DS_INSATISFATORIO
  FROM DBAMV.MULTI_EMPRESAS
     , DBAMV.CIDADE
     , DBAMV.PACIENTE
     , DBAMV.SISCO_UNI_SER
     , DBAMV.CIDADE CIDADE_UNIDADE
     , DBAMV.SISCO_HISTO
     , DBAMV.FAT_SIA 
 WHERE CIDADE.CD_CIDADE                = MULTI_EMPRESAS.CD_CIDADE
   AND SISCO_UNI_SER.CD_CIDADE         = CIDADE_UNIDADE.CD_CIDADE
   AND MULTI_EMPRESAS.CD_MULTI_EMPRESA = SISCO_HISTO.CD_MULTI_EMPRESA
   AND PACIENTE.CD_PACIENTE            = SISCO_HISTO.CD_PACIENTE
   AND SISCO_UNI_SER.CD_UNI_SER        = SISCO_HISTO.CD_UNI_SER
   AND MULTI_EMPRESAS.CD_MULTI_EMPRESA = PKG_MV2000.LE_EMPRESA
   AND SISCO_HISTO.CD_FAT_SIA = FAT_SIA.CD_FAT_SIA
   AND to_char(FAT_SIA.DT_COMPETENCIA, 'MMYYYY') = :CAMPO_COMPETENCIA
 ORDER BY CIDADE.NM_CIDADE ASC
        , SISCO_UNI_SER.DS_UNI_SER ASC
)
ORDER BY CD_MULTI_EMPRESA,DS_MULTI_EMPRESA,NR_CNES,CD_CGC,NM_CIDADE,CD_UF,CD_UNI_SER,DS_UNI_SER,NM_CIDADE_UNID,NR_CNES_UNID,CD_UF_UNID
