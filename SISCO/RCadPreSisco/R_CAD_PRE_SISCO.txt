--R_CAD_PRE_SISCO
SELECT especialid.CD_ESPECIALID
     , especialid.DS_ESPECIALID
     , prestador.CD_PRESTADOR
     , prestador.NM_PRESTADOR
     , prestador.NR_CPF_CGC
     , prestador.DS_CODIGO_CONSELHO
  FROM DBAMV.ESP_MED esp_med
     , DBAMV.ESPECIALID especialid
     , DBAMV.PRESTADOR prestador
 WHERE esp_med.CD_PRESTADOR       = prestador.CD_PRESTADOR
   AND esp_med.CD_ESPECIALID      = especialid.CD_ESPECIALID
   AND prestador.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
 ORDER BY especialid.DS_ESPECIALID
        , prestador.NM_PRESTADOR
