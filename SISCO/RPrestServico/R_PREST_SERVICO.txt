--R_PREST_SERVICO
SELECT MULTI_EMPRESAS.CD_MULTI_EMPRESA
       ,DS_MULTI_EMPRESA
       ,HOSPITAL.CD_UNIDADE
       ,MULTI_EMPRESAS.CD_CGC
       ,' ' NUMERO
       ,MULTI_EMPRESAS.DS_RAZAO_SOCIAL
       ,MULTI_EMPRESAS.NM_BAIRRO
       ,MULTI_EMPRESAS.DS_ENDERECO
       ,MULTI_EMPRESAS.CD_UF
       ,' ' COMPLEMENTO
       ,MULTI_EMPRESAS.NR_DDD_EMPRESA
       ,MULTI_EMPRESAS.NR_TELEFONE_EMPRESA
       ,CIDADE.NM_CIDADE
       ,MULTI_EMPRESAS.NR_CEP
       ,CIDADE.CD_IBGE
      ,MULTI_EMPRESAS.nr_cnes
   FROM  dbamv.MULTI_EMPRESAS , dbamv.HOSPITAL , dbamv.CIDADE
  WHERE MULTI_EMPRESAS.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
    AND Hospital.Cd_Multi_Empresa = :CAMPO_CD_MULTI_EMPRESA
    AND MULTI_EMPRESAS.CD_MULTI_EMPRESA = HOSPITAL.CD_MULTI_EMPRESA
    AND MULTI_EMPRESAS.CD_CIDADE = CIDADE.CD_CIDADE
