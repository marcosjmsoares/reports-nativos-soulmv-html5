--R_CAD_PAC_SISCO
SELECT paciente.CD_PACIENTE
     , paciente.NM_PACIENTE
     , paciente.NM_MNEMONICO
     , paciente.NM_MAE
     , paciente.NR_IDENTIDADE
     , paciente.DS_OM_IDENTIDADE
     , paciente.NR_CPF
     , paciente.DT_NASCIMENTO
     , dbamv.PACK_SISCOLO.Retorna_Idade(DT_NASCIMENTO, sysdate) IDADE
     , paciente.DS_ENDERECO
     , paciente.DS_COMPLEMENTO
     , paciente.NR_ENDERECO
     , paciente.NM_BAIRRO
     , paciente.NR_CEP
     , cidade.NM_CIDADE
     , cidade.CD_UF
     , paciente.DS_OBSERVACAO
     , paciente.NR_FONE
     , grau_ins.DS_GRAU_INS
  FROM DBAMV.PACIENTE paciente
     , DBAMV.CIDADE cidade
     , DBAMV.GRAU_INS grau_ins
 WHERE paciente.CD_CIDADE        = cidade.CD_CIDADE
   AND paciente.CD_GRAU_INS      = grau_ins.CD_GRAU_INS
   AND paciente.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
 ORDER BY paciente.NM_PACIENTE ASC
