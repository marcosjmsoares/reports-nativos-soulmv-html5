SELECT decode(sisco_uni_ser.ds_uni_ser, null, multi_empresas.DS_MULTI_EMPRESA, sisco_uni_ser.ds_uni_ser) DS_UNI_SER
     , decode(cida_uni.NM_CIDADE      , null, cidade.NM_CIDADE               , cida_uni.NM_CIDADE      ) CIDA_UNI
     , decode(cida_uni.CD_UF          , null, cidade.CD_UF                   , cida_uni.CD_UF          ) UF_UNI
     , sisco_uni_ser.NR_CNES
     , cidade.NM_CIDADE
     , cidade.CD_UF
     , '0'||sisco_histo.CD_SISCO_HISTO CD_SISCO_HISTO     , sisco_histo.CD_PACIENTE
     , paciente.NM_PACIENTE
     , paciente.NM_MAE
     , paciente.DT_NASCIMENTO
     , paciente.DS_ENDERECO
     , paciente.NR_ENDERECO
     , paciente.DS_COMPLEMENTO
     , paciente.NM_BAIRRO
     , cida_pac.NM_CIDADE CIDA_PAC
     , cida_pac.CD_UF UF_PAC
     , paciente.NR_FONE
     , paciente.DS_OBSERVACAO
     , multi_empresas.DS_MULTI_EMPRESA
     , multi_empresas.CD_CGC
     , sisco_histo.DT_EXAME
     , sisco_histo.DT_RECEBIDO
     , sisco_histo.DT_LIB_RESULTADO
     , prestador.NM_PRESTADOR
     , prestador.DS_CODIGO_CONSELHO
     , cida_pre.CD_UF UF_PRE
     , sisco_histo.TP_PROC_CIRU
     , sisco_histo.SN_BIOPSIA
     , sisco_histo.QT_FRAG
     , sisco_histo.DS_MACROSCOPIA
     , sisco_histo.SN_MET_ESC
     , sisco_histo.SN_POL_END
     , sisco_histo.SN_CER_CRO
     , sisco_histo.SN_ALT_CITO
     , sisco_histo.TP_LESAO_NEO
     , sisco_histo.TP_LESAO_NEO2
     , sisco_histo.DS_OUT_LESAO_NEO
     , sisco_histo.TP_GRAU_DIF
     , sisco_histo.NR_PROF_INV
     , sisco_histo.SN_VASCULAR
     , sisco_histo.SN_PERINEURAL
     , sisco_histo.SN_PARAMETRIAL
     , sisco_histo.SN_CORPO_UTE
     , sisco_histo.SN_VAGINA
     , sisco_histo.TP_MARGEM_CIRU
     , sisco_histo.DS_DIAG_DESC
     , sisco_histo.NR_FRAG
     , sisco_histo.NR_BLOCOS
     , sisco_histo.NR_DIM_TU_COM
     , sisco_histo.NR_DIM_TU_LAR
     , sisco_histo.NR_DIST_MARGEM
     , sisco_histo.TP_LOCALIZACAO
     ,sisco_histo.SN_INSATISFATORIO
     ,sisco_histo.DS_INSATISFATORIO
     ,sisco_histo.SN_COL_CAF
     ,sisco_histo.SN_RES_BIOP
     ,sisco_histo.SN_RES_PECA
     ,sisco_histo.SN_CIT_ESC
     ,sisco_histo.SN_CIT_GLA
     ,sisco_histo.SN_CIT_IND
     ,sisco_histo.SN_CIT_ESCA
  FROM DBAMV.PRESTADOR prestador
     , DBAMV.PACIENTE paciente
     , DBAMV.SISCO_HISTO sisco_histo
     , DBAMV.CIDADE cida_pac
     , DBAMV.CIDADE cida_pre
     , DBAMV.MULTI_EMPRESAS multi_empresas
     , DBAMV.CIDADE cidade
     , DBAMV.SISCO_UNI_SER sisco_uni_ser
     , DBAMV.CIDADE cida_uni
 WHERE sisco_histo.CD_FAT_SIA  IS NOT NULL
   AND sisco_histo.CD_MULTI_EMPRESA = multi_empresas.CD_MULTI_EMPRESA
   AND multi_empresas.CD_CIDADE           = cidade.CD_CIDADE
   AND sisco_histo.CD_PACIENTE             = paciente.CD_PACIENTE
   AND paciente.CD_CIDADE                     = cida_pac.CD_CIDADE
   AND sisco_histo.CD_PRESTADOR         = prestador.CD_PRESTADOR
   AND prestador.CD_CIDADE                     = cida_pre.CD_CIDADE
   AND sisco_histo.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
   AND prestador.CD_MULTI_EMPRESA   = :CAMPO_CD_MULTI_EMPRESA
   AND sisco_uni_ser.cd_uni_ser (+)           = sisco_histo.cd_uni_ser
   AND sisco_uni_ser.CD_CIDADE             = cida_uni.CD_CIDADE (+)
   ORDER BY SISCO_HISTO.CD_SISCO_HISTO
