--R_UNIDADE_SAUDE
SELECT SISCO_UNI_SER.CD_UNI_SER
      ,SISCO_UNI_SER.DS_UNI_SER
      ,SISCO_UNI_SER.CD_CIDADE
      ,CIDADE.CD_UF
      ,CIDADE.NM_CIDADE
 FROM DBAMV.SISCO_UNI_SER , DBAMV.CIDADE
WHERE SISCO_UNI_SER.CD_CIDADE = CIDADE.CD_CIDADE
  AND SISCO_UNI_SER.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
