SELECT 
       sisco_uni_ser.NR_CNES
     , decode(sisco_uni_ser.ds_uni_ser, null, multi_empresas.DS_MULTI_EMPRESA, sisco_uni_ser.ds_uni_ser) DS_UNI_SER
     , sisco_cito.DT_COLETA
     , decode(cida_uni.NM_CIDADE      , null, cidade.NM_CIDADE               , cida_uni.NM_CIDADE      ) CIDA_UNI
     , decode(cida_uni.CD_UF          , null, cidade.CD_UF                   , cida_uni.CD_UF          ) UF_UNI
     , sisco_cito.CD_SISCO_CITO
     , sisco_cito.CD_PACIENTE
     , paciente.NM_PACIENTE
     , paciente.DT_NASCIMENTO
     , paciente.NM_MAE
     , paciente.DS_ENDERECO
     , paciente.NR_ENDERECO
     , paciente.DS_COMPLEMENTO
     , paciente.NM_BAIRRO
     , cida_pac.NM_CIDADE CIDA_PAC
     , cida_pac.CD_UF UF_PAC
     , paciente.DS_OBSERVACAO
     , paciente.NR_FONE
     , multi_empresas.DS_MULTI_EMPRESA
     , multi_empresas.CD_CGC
     , cidade.NM_CIDADE
     , cidade.CD_UF
     , sisco_cito.DT_RECEBIDO
     , sisco_cito.SN_REJ_AUS
     , sisco_cito.SN_REJ_DAN
     , sisco_cito.SN_REJ_ALH
     , sisco_cito.DS_REJ_AESP
     , sisco_cito.SN_REJ_OUT
     , sisco_cito.DS_REJ_OESP
     , sisco_cito.TP_ADEQUA_MAT
     , sisco_cito.SN_ADQ_ACEL
     , sisco_cito.SN_ADQ_SANG
     , sisco_cito.SN_ADQ_PIO
     , sisco_cito.SN_ADQ_ART
     , sisco_cito.SN_ADQ_CONT
     , sisco_cito.SN_ADQ_INT
     , sisco_cito.SN_ADQ_OUT
     , sisco_cito.SN_EPI_ESC
     , sisco_cito.SN_EPI_GLA
     , sisco_cito.SN_EPI_MET
     , sisco_cito.SN_DENTRO_NORMAL
     , sisco_cito.SN_INFLAMACAO
     , sisco_cito.SN_META_ESCAMOSA
     , sisco_cito.SN_REPARACAO
     , sisco_cito.SN_ATROFIA
     , sisco_cito.SN_RADIACAO
     , sisco_cito.DS_OUTROS1
     , sisco_cito.SN_LACTOBACILOS
     , sisco_cito.SN_COCOS
     , sisco_cito.SN_BACILOS
     , sisco_cito.SN_MIC_OUT
     , sisco_cito.SN_CLAMIDIA
     , sisco_cito.SN_ACTINOMYCES
     , sisco_cito.SN_CANDIDA
     , sisco_cito.SN_TRICHOMONAS
     , sisco_cito.SN_VIRUS_HERPES
     , sisco_cito.SN_GARDNERELLA
     , sisco_cito.DS_OUTROS2
     , sisco_cito.SN_ATI_ESC
     , sisco_cito.SN_ATI_GLA
     , sisco_cito.SN_ATI_IND
     , sisco_cito.TP_ALT_CEL_ESC
     , sisco_cito.TP_ALT_CEL_GLA
     , sisco_cito.DS_OUTROS3
     , sisco_cito.SN_CEL_ENDOMETRIA
     , substr(sisco_cito.DS_OBS_GERAL, 1, 100) DS_OBS_GERAL
     , sisco_cito.CD_PRESTADOR
     , sisco_cito.DT_LIBERACAO
     , prestador.NM_PRESTADOR
     , prestador.DS_CODIGO_CONSELHO
     , cida_pre.CD_UF UF_PRE
  FROM DBAMV.PRESTADOR prestador
     , DBAMV.PACIENTE paciente
     , DBAMV.SISCO_CITO sisco_cito
     , DBAMV.CIDADE cida_pac
     , DBAMV.CIDADE cida_pre
     , DBAMV.MULTI_EMPRESAS multi_empresas
     , DBAMV.CIDADE cidade
     , DBAMV.SISCO_UNI_SER sisco_uni_ser
     , DBAMV.CIDADE cida_uni
     ,DBAMV.EVE_SIASUS eve_siasus
 WHERE sisco_cito.CD_FAT_SIA     IS NOT NULL
   AND sisco_cito.CD_MULTI_EMPRESA = multi_empresas.CD_MULTI_EMPRESA
   AND multi_empresas.CD_CIDADE    = cidade.CD_CIDADE
   AND sisco_cito.CD_PACIENTE      = paciente.CD_PACIENTE
   AND paciente.CD_CIDADE          = cida_pac.CD_CIDADE
   AND sisco_cito.CD_PRESTADOR     = prestador.CD_PRESTADOR
   AND prestador.CD_CIDADE         = cida_pre.CD_CIDADE
   AND sisco_cito.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
   AND prestador.CD_MULTI_EMPRESA  = :CAMPO_CD_MULTI_EMPRESA
   AND sisco_uni_ser.cd_uni_ser (+) = sisco_cito.cd_uni_ser
   AND sisco_uni_ser.CD_CIDADE      = cida_uni.CD_CIDADE (+)
   AND sisco_cito.CD_SISCO_CITO = eve_siasus.CD_SISCO_CITO
   AND sisco_cito.CD_FAT_SIA = eve_siasus.CD_FAT_SIA
  ORDER BY SISCO_CITO.CD_SISCO_CITO
