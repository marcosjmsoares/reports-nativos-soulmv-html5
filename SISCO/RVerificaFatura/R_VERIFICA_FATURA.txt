--R_VERIFICA_FATURA
SELECT sisco_cito.CD_MULTI_EMPRESA
     , '12011010 CITOPATOLOGIA-COLO' PROC
     , fat_sia.DT_COMPETENCIA COMPETENCIA
     , count(*) QTD
  FROM DBAMV.FAT_SIA fat_sia
     , DBAMV.SISCO_CITO sisco_cito
 WHERE sisco_cito.CD_FAT_SIA       = fat_sia.CD_FAT_SIA
   AND sisco_cito.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
  AND sisco_cito.cd_sisco_cito not in (Select eve_siasus.cd_sisco_cito
                                                              From dbamv.eve_siasus,
                                                                       dbamv.fat_sia
                                                              Where eve_siasus.cd_fat_sia = fat_sia.cd_fat_sia
                                                                And eve_siasus.nr_idade < 61
                                      )
 GROUP BY sisco_cito.CD_MULTI_EMPRESA
        , 2
        , fat_sia.DT_COMPETENCIA
union all
SELECT sisco_histo.CD_MULTI_EMPRESA
     , '12012033 HISTOPATOLÓGICO-COLO' PROC
     , fat_sia.DT_COMPETENCIA COMPETENCIA
     , count(*) QTD
  FROM DBAMV.FAT_SIA fat_sia
     , DBAMV.SISCO_HISTO sisco_histo
 WHERE sisco_histo.CD_FAT_SIA       = fat_sia.CD_FAT_SIA
   AND sisco_histo.CD_MULTI_EMPRESA = :CAMPO_CD_MULTI_EMPRESA
  AND sisco_histo.cd_sisco_histo not in (Select eve_siasus.cd_sisco_histo
                                                              From dbamv.eve_siasus,
                                                                       dbamv.fat_sia
                                                              Where eve_siasus.cd_fat_sia = fat_sia.cd_fat_sia
                                                                And eve_siasus.nr_idade < 61
                                        )
 GROUP BY sisco_histo.CD_MULTI_EMPRESA
        , 2
        , fat_sia.DT_COMPETENCIA
ORDER BY CD_MULTI_EMPRESA,
PROC,
COMPETENCIA desc
