--R_STATUS_SOLIC_ANALITICO
SELECT S.CD_SOLICITACAO
       , S.NM_PACIENTE
       , (SELECT PS.DS_PROBLEMA_SAUDE FROM DBAMV.PROBLEMA_SAUDE_SIDRA PS WHERE PS.CD_PROBLEMA_SAUDE = S.CD_PROBLEMA_SAUDE) PROBLEMA_SAUDE
       , (SELECT P.CD_PATOLOGIA_GES || ' - ' || P.DS_PATOLOGIA_GES FROM DBAMV.PATOLOGIA_GES P WHERE P.CD_PATOLOGIA_GES = S.CD_PATOLOGIA_GES) PATOLOGIA_GES
       , (SELECT C.CD_CID || ' - ' || C.DS_CID FROM DBAMV.CID C WHERE C.CD_CID = S.CD_CID) CID
       , (SELECT EO.DS_ESTABELECIMENTO_SAUDE FROM DBAMV.ESTABELECIMENTO_SAUDE_SIDRA EO WHERE EO.CD_ESTABELECIMENTO_SAUDE = S.CD_ESTABELECIMENTO_ORIGEM) ESTABELECIMENTO_ORIGEM
       , (SELECT ED.DS_ESTABELECIMENTO_SAUDE FROM DBAMV.ESTABELECIMENTO_SAUDE_SIDRA ED WHERE ED.CD_ESTABELECIMENTO_SAUDE = S.CD_ESTABELECIMENTO_DESTINO) ESTABELECIMENTO_DESTINO
       , S.DT_SOLICITACAO
       , S.DT_CANCELAMENTO
       , S.DT_CADASTRO
       , (SELECT DS_MOTIVO_CANCELAMENTO FROM DBAMV.MOTIVO_CANCELAMENTO_SIDRA M WHERE M.CD_MOTIVO_CANCELAMENTO = S.CD_MOTIVO_CANCELAMENTO) DS_MOTIVO_CANCELAMENTO
       , S.DT_CLASSIFICACAO
       , S.DT_ATENDIMENTO
       , S.TP_STATUS_SOLIC
       , (SELECT P.NM_SOCIAL_PACIENTE
			FROM DBAMV.PACIENTE P
			WHERE P.CD_PACIENTE = S.CD_PACIENTE
				AND NVL(P.SN_UTILIZA_NOME_SOCIAL, 'N') = 'S') NM_SOCIAL_PACIENTE
FROM DBAMV.SOLICITACAO_ATENDIMENTO_SIDRA S
WHERE S.CD_SOLICITACAO IS NOT NULL
ORDER BY S.CD_SOLICITACAO
