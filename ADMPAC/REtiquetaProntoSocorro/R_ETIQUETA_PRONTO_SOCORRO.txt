--R_ETIQUETA_PRONTO_SOCORRO
SELECT
     'ATEND.: '||VDIC_TI_PACIENTE_BOLETIM_PS.CD_ATENDIMENTO cd_atendimento,
     VDIC_TI_PACIENTE_BOLETIM_PS.CD_BARRA_ATENDIMENTO,
     'AT.: '||to_char(VDIC_TI_PACIENTE_BOLETIM_PS.DT_ATENDIMENTO,'dd/mm/yyyy')||' '||VDIC_TI_PACIENTE_BOLETIM_PS.HR_ATENDIMENTO dt_atendimento,
     VDIC_TI_PACIENTE_BOLETIM_PS.NOME_PRESTADOR,
     'Matricula: '||VDIC_TI_PACIENTE_BOLETIM_PS.NR_CARTEIRA nr_carteira,
     VDIC_TI_PACIENTE_BOLETIM_PS.NOME_PACIENTE,
     'Nasc.: '||to_char(VDIC_TI_PACIENTE_BOLETIM_PS.DATA_NASCIMENTO,'dd/mm/yyyy') data_nascimento,
     VDIC_TI_PACIENTE_BOLETIM_PS.IDADE_ANO_MES_DIA,
     VDIC_TI_PACIENTE_BOLETIM_PS.descricao_convenio,
     'SX.: '||VDIC_TI_PACIENTE_BOLETIM_PS.TIPO_SEXO tipo_sexo,
     VDIC_TI_PACIENTE_BOLETIM_PS.CODIGO_PACIENTE
FROM
     DBAMV.VDIC_TI_PACIENTE_BOLETIM_PS VDIC_TI_PACIENTE_BOLETIM_PS
WHERE
 DBAMV.VDIC_TI_PACIENTE_BOLETIM_PS.cd_atendimento = :CAMPO_CD_ATENDIMENTO
