SELECT 'Atend.: '||VDIC_TI_PACIENTE_AMB_URG.CD_ATENDIMENTO cd_atendimento,
       'PRONT.: '||VDIC_TI_PACIENTE_AMB_URG.CODIGO_PACIENTE CODIGO_PACIENTE,
       VDIC_TI_PACIENTE_AMB_URG.NOME_PACIENTE,
       'SX.: '||VDIC_TI_PACIENTE_AMB_URG.TIPO_SEXO tipo_sexo,
       'NASC.: '||to_char(VDIC_TI_PACIENTE_AMB_URG.DATA_NASCIMENTO,'dd/mm/yyyy') data_nascimento,
       VDIC_TI_PACIENTE_AMB_URG.IDADE_ANO_MES_DIA,
       'DT: '||to_char(VDIC_TI_PACIENTE_AMB_URG.DT_ATENDIMENTO,'dd/mm/yyyy')||' - '||VDIC_TI_PACIENTE_AMB_URG.HR_ATENDIMENTO DT_ATENDIMENTO,
       VDIC_TI_PACIENTE_AMB_URG.cd_barra_atendimento
FROM
     DBAMV.VDIC_TI_PACIENTE_AMB_URG VDIC_TI_PACIENTE_AMB_URG
WHERE
      VDIC_TI_PACIENTE_AMB_URG.CD_ATENDIMENTO = :CAMPO_CD_ATENDIMENTO
