--R_SOLICITACAO_AMBULANCIA
SELECT solicitacao_ambulancia.CD_SOLICITACAO_AMBULANCIA
       , setor.NM_SETOR
       ,  trajeto.DS_TRAJETO
       , posicao_paciente_ambulancia.DS_POSICAO_PACIENTE_AMBULANCIA
       , CD_USUARIO_APROVACAO
       , solicitacao_ambulancia.NM_PACIENTE
       , NM_BAIRRO
       , CID.CD_CID
       , CID.DS_CID
       , DS_SERVICO
       , NR_DDD_FONE
       , NR_TELEFONE
       ,DS_ENDERECO
       ,NR_ENDERECO AS NUMERO
       ,DS_COMPLEMENTO
       ,NR_CEP
       ,CIDADE.NM_CIDADE AS CIDADE
       , To_CHAR(DT_HR_SOLICITACAO,'DD/MM/YYYY') DT_SOLICITACAO
       , solicitacao_ambulancia.NR_CARTEIRA
       , Decode(sn_titular,'S','Titular', 'N', 'Dependente',NULL) TIPO_CARTEIRA
       , solicitacao_ambulancia.DT_HR_SAIDA_AMBULANCIA_HOSP
       , solicitacao_ambulancia.NR_QUILOMETRAGEM_CHEGADA_INI
       , To_Char(solicitacao_ambulancia.DT_HR_FIM_TRAJETO,'HH24') HORA_FIM
       , solicitacao_ambulancia.NR_QUILOMETRAGEM_FIM_TRAJETO
       , To_Char(solicitacao_ambulancia.DT_HR_AMBULANCIA,'HH24') HORA_AMBULANCIA
       , To_Char(DT_HR_SAIDA_AMBULANCIA_HOSP,'HH24') HORA_SAIDA
       , To_Char(solicitacao_ambulancia.DT_HR_AMBULANCIA,'DD/MM/YYYY') DATA_AMBULANCIA
     ,Decode (TP_ATENDIMENTO, 'H', 'Home Care','I', 'Internação', 'A','Ambulátorio','E', 'Externo', 'U', 'Urgência', 'B', 'Busca Ativa', 'S', 'SUS') TIPO_ATENDIMENTO
    FROM dbamv.solicitacao_ambulancia
       , dbamv.setor
       , dbamv.trajeto
       , dbamv.posicao_paciente_ambulancia
       , dbamv.atendime
       , dbamv.cid
       , dbamv.servico
       , dbamv.carteira
       ,dbamv.cidade
    WHERE solicitacao_ambulancia.cd_setor = setor.cd_setor (+)
     AND solicitacao_ambulancia.cd_cidade = cidade.cd_cidade  (+)	
     AND solicitacao_ambulancia.cd_atendimento = atendime.cd_atendimento (+)
     AND atendime.cd_cid = cid.cd_cid (+)
     AND atendime.cd_servico = servico.cd_servico (+)
     AND solicitacao_ambulancia.cd_trajeto = trajeto.cd_trajeto (+)
     AND solicitacao_ambulancia.cd_posicao_paciente_ambulancia = posicao_paciente_ambulancia.cd_posicao_paciente_ambulancia (+)
     AND  solicitacao_ambulancia.cd_paciente = carteira.cd_paciente (+)
     AND  solicitacao_ambulancia.cd_convenio = carteira.cd_convenio (+)
     AND  solicitacao_ambulancia.cd_con_pla = carteira.cd_con_pla (+)
     AND  solicitacao_ambulancia.nr_carteira = carteira.nr_carteira (+)
