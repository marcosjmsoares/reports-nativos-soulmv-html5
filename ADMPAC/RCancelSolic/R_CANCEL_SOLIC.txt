--R_CANCEL_SOLIC
SELECT (SELECT E.DS_ESTABELECIMENTO_SAUDE FROM DBAMV.ESTABELECIMENTO_SAUDE_SIDRA E WHERE E.CD_ESTABELECIMENTO_SAUDE = :CAMPO_TP_ESTABELECIMENTO) DS_ESTABELECIMENTO_SAUDE,
        Count(:CAMPO_TP_ESTABELECIMENTO) QTDE,
       (SELECT M.DS_MOTIVO_CANCELAMENTO FROM DBAMV.MOTIVO_CANCELAMENTO_SIDRA M WHERE M.CD_MOTIVO_CANCELAMENTO = S.CD_MOTIVO_CANCELAMENTO) DS_MOTIVO_CANCELAMENTO
FROM DBAMV.SOLICITACAO_ATENDIMENTO_SIDRA S
WHERE S.CD_MOTIVO_CANCELAMENTO IS NOT NULL
GROUP BY :CAMPO_TP_ESTABELECIMENTO, S.CD_MOTIVO_CANCELAMENTO
