--R_STATUS_SOLIC_SINTETICO
SELECT
       CD_ESTABELECIMENTO_SAUDE    CD_ESTABELECIMENTO_SAUDE,
       CD_ESTABELECIMENTO_SAUDE || ' - ' || DS_ESTABELECIMENTO_SAUDE ESTABELECIMENTO_SAUDE,
       TO_CHAR(DT_SOLICITACAO, 'MM/YYYY') COMPETENCIA ,
       TP_STATUS_SOLIC TIPO
FROM
  (SELECT CD_SOLICITACAO,
          CD_ESTABELECIMENTO_SAUDE,
          DS_ESTABELECIMENTO_SAUDE,
          CD_ESTABELECIMENTO_ORIGEM,
          CD_ESTABELECIMENTO_DESTINO,
          TP_STATUS_SOLIC,
          DT_SOLICITACAO
   FROM DBAMV.SOLICITACAO_ATENDIMENTO_SIDRA S
       ,DBAMV.ESTABELECIMENTO_SAUDE_SIDRA E
   :CAMPO_COND_PROBLEMA_SAUDE
  )
ORDER BY ESTABELECIMENTO_SAUDE, COMPETENCIA

--R_STATUS_SOLIC_SINTETICO_QTD_TIPO
SELECT
(
SELECT
       COUNT(CD_ESTABELECIMENTO_SAUDE) QUANTIDADE
FROM
  (
   SELECT CD_ESTABELECIMENTO_SAUDE,
          TO_CHAR(DT_SOLICITACAO, 'MM/YYYY') COMPETENCIA
   FROM DBAMV.SOLICITACAO_ATENDIMENTO_SIDRA S
       ,DBAMV.ESTABELECIMENTO_SAUDE_SIDRA E
   AND  TP_STATUS_SOLIC = 1 
   AND CD_ESTABELECIMENTO_SAUDE = :CAMPO_CD_ESTABELECIMENTO_SAUDE
   AND TO_CHAR(DT_SOLICITACAO, 'MM/YYYY') =  ''
  )
) QTD_PENDENTES,
(
SELECT
       COUNT(CD_ESTABELECIMENTO_SAUDE) QUANTIDADE
FROM
  (
   SELECT CD_ESTABELECIMENTO_SAUDE,
          TO_CHAR(DT_SOLICITACAO, 'MM/YYYY') COMPETENCIA
   FROM DBAMV.SOLICITACAO_ATENDIMENTO_SIDRA S
       ,DBAMV.ESTABELECIMENTO_SAUDE_SIDRA E
   AND  TP_STATUS_SOLIC = 2 
   AND CD_ESTABELECIMENTO_SAUDE = :CAMPO_CD_ESTABELECIMENTO_SAUDE
   AND TO_CHAR(DT_SOLICITACAO, 'MM/YYYY') =  ''
  )
) QTD_CLASSIFICADOS,
(
SELECT
       COUNT(CD_ESTABELECIMENTO_SAUDE) QUANTIDADE
FROM
  (
   SELECT CD_ESTABELECIMENTO_SAUDE,
          TO_CHAR(DT_SOLICITACAO, 'MM/YYYY') COMPETENCIA
   FROM DBAMV.SOLICITACAO_ATENDIMENTO_SIDRA S
       ,DBAMV.ESTABELECIMENTO_SAUDE_SIDRA E
   AND  TP_STATUS_SOLIC = 3 
   AND CD_ESTABELECIMENTO_SAUDE = :CAMPO_CD_ESTABELECIMENTO_SAUDE
   AND TO_CHAR(DT_SOLICITACAO, 'MM/YYYY') =  ''
  )
) QTD_AGENDADOS,
(
SELECT
       COUNT(CD_ESTABELECIMENTO_SAUDE) QUANTIDADE
FROM
  (
   SELECT CD_ESTABELECIMENTO_SAUDE,
          TO_CHAR(DT_SOLICITACAO, 'MM/YYYY') COMPETENCIA
   FROM DBAMV.SOLICITACAO_ATENDIMENTO_SIDRA S
       ,DBAMV.ESTABELECIMENTO_SAUDE_SIDRA E
   AND  TP_STATUS_SOLIC = 4 
   AND CD_ESTABELECIMENTO_SAUDE = :CAMPO_CD_ESTABELECIMENTO_SAUDE
   AND TO_CHAR(DT_SOLICITACAO, 'MM/YYYY') =  ''
  )
) QTD_ATENDIDOS,
(
SELECT
       COUNT(CD_ESTABELECIMENTO_SAUDE) QUANTIDADE
FROM
  (
   SELECT CD_ESTABELECIMENTO_SAUDE,
          TO_CHAR(DT_SOLICITACAO, 'MM/YYYY') COMPETENCIA
   FROM DBAMV.SOLICITACAO_ATENDIMENTO_SIDRA S
       ,DBAMV.ESTABELECIMENTO_SAUDE_SIDRA E
   AND  (TP_STATUS_SOLIC = 5 OR TP_STATUS_SOLIC = 6) 
   AND CD_ESTABELECIMENTO_SAUDE = :CAMPO_CD_ESTABELECIMENTO_SAUDE
   AND TO_CHAR(DT_SOLICITACAO, 'MM/YYYY') =  ''
   )
) QTD_CANCELADOS
FROM DUAL
