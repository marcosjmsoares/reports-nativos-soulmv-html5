--R_SAC_EST_ORIGEM_TIPO
select sac_solicitacao.cd_origem  cd_origem_chamado,
         origem_sac_solicitacao.nm_origem  nm_origem,
         sac_solicitacao.cd_chamado cd_tp_chamado,
         tipo_sac_chamado.nm_chamado nm_tp_chamado,
         sac_solicitacao.cd_nivel   cd_tp_nivel,
         tipo_sac_chamado_nivel.nm_nivel  nm_nivel,
         count(sac_solicitacao.cd_origem ||sac_solicitacao.cd_nivel) tot_nivel
    from dbamv.sac_solicitacao,
         dbamv.tipo_sac_chamado_nivel,
         dbamv.tipo_sac_chamado,
         dbamv.origem_sac_solicitacao
   where sac_solicitacao.cd_origem     = origem_sac_solicitacao.cd_origem
     and sac_solicitacao.cd_nivel = tipo_sac_chamado_nivel.cd_nivel
     and sac_solicitacao.cd_nivel = tipo_sac_chamado_nivel.cd_nivel
     and sac_solicitacao.cd_chamado = tipo_sac_chamado.cd_chamado
     and sac_solicitacao.cd_multi_empresa = nvl(dbamv.pkg_mv2000.le_empresa, sac_solicitacao.cd_multi_empresa)
     and dt_solicidacao >= decode(:CAMPO_DATAINICIO,null,dt_solicidacao,:CAMPO_DATAINICIO)
     and dt_solicidacao <= decode(:CAMPO_DATAFIM,null,dt_solicidacao,:CAMPO_DATAFIM)
group by sac_solicitacao.cd_origem,
         origem_sac_solicitacao.nm_origem,
         sac_solicitacao.cd_chamado,
         tipo_sac_chamado.nm_chamado,
         sac_solicitacao.cd_nivel,
         tipo_sac_chamado_nivel.nm_nivel
order by sac_solicitacao.cd_origem
