--R_SAC_EST_NIVEL
select  tipo_sac_chamado.cd_chamado,
             nm_chamado,
             tipo_sac_chamado_nivel.cd_nivel   cd_tp_nivel,
             nm_nivel,
             count(tipo_sac_chamado.cd_chamado ||tipo_sac_chamado_nivel.cd_nivel) tot_nivel
    from dbamv.sac_solicitacao,
            dbamv.tipo_sac_chamado_nivel,
            dbamv.tipo_sac_chamado
   where  sac_solicitacao.cd_nivel = tipo_sac_chamado_nivel.cd_nivel
              and sac_solicitacao.cd_chamado = tipo_sac_chamado.cd_chamado
              and sac_solicitacao.cd_multi_empresa = nvl(dbamv.pkg_mv2000.le_empresa, sac_solicitacao.cd_multi_empresa)
              and dt_solicidacao >= decode(:CAMPO_DATAINICIO,null,dt_solicidacao,:CAMPO_DATAINICIO)
              and dt_solicidacao <= decode(:CAMPO_DATAFIM,null,dt_solicidacao,:CAMPO_DATAFIM)
group by tipo_sac_chamado.cd_chamado,
           nm_chamado,
           tipo_sac_chamado_nivel.cd_nivel,
           nm_nivel
order by cd_chamado
