--R_SAIDA_BENEFICIO
SELECT
  F.CD_TIPO_DESTINATARIO_BENEFICIO  CD_TIPO_DESTINATARIO_BENEFICIO
	,F.DS_TIPO_DESTINATARIO_BENEFICIO  DS_TIPO_DESTINATARIO_BENEFICIO
  ,TRUNC(MV.DT_MOVIMENTO_BENEFICIO) DT_MOVIMENTO_BENEFICIO
  ,IM.CD_TIPO_BENEFICIO CD_TIPO_BENEFICIO
  ,TB.DS_TIPO_BENEFICIO DS_TIPO_BENEFICIO
  ,SUM(IM.QT_ENTRADA_ITEM) QT_ENTRADA_ITEM
FROM DBAMV.MOVIMENTO_BENEFICIO MV
    ,DBAMV.TIPO_DESTINATARIO_BENEFICIO F
    ,DBAMV.ITEM_MOVIMENTO_BENEFICIO IM
    ,DBAMV.TIPO_BENEFICIO TB
WHERE F.CD_TIPO_DESTINATARIO_BENEFICIO = MV.CD_TIPO_DESTINATARIO_BENEFICIO(+)
  AND IM.CD_MOVIMENTO_BENEFICIO = MV.CD_MOVIMENTO_BENEFICIO
  AND IM.CD_TIPO_BENEFICIO = TB.CD_TIPO_BENEFICIO
  AND TRUNC(MV.DT_MOVIMENTO_BENEFICIO) BETWEEN :CAMPO_DT_INICIAL GROUP BY
	  F.CD_TIPO_DESTINATARIO_BENEFICIO
	 ,F.DS_TIPO_DESTINATARIO_BENEFICIO
 	 ,IM.CD_TIPO_BENEFICIO
   ,TB.DS_TIPO_BENEFICIO
   ,TRUNC(MV.DT_MOVIMENTO_BENEFICIO)
ORDER BY
   F.CD_TIPO_DESTINATARIO_BENEFICIO
  , F.DS_TIPO_DESTINATARIO_BENEFICIO
  ,TRUNC(MV.DT_MOVIMENTO_BENEFICIO)
  ,IM.CD_TIPO_BENEFICIO
  ,TB.DS_TIPO_BENEFICIO
