--R_PESQUISA_SATISFACAO
SELECT  TRUNC(QTD_1_SIM/DECODE(QTD_1_SIM+QTD_1_NAO, 0, 1, QTD_1_SIM+QTD_1_NAO ),2)*100||'%' AS QTD_1_SIM,
        TRUNC(QTD_1_NAO/DECODE(QTD_1_SIM+QTD_1_NAO, 0, 1, QTD_1_SIM+QTD_1_NAO),2)*100||'%' AS QTD_1_NAO,
        TRUNC(QTD_2_SIM/DECODE(QTD_2_SIM+QTD_2_NAO, 0, 1, QTD_2_SIM+QTD_2_NAO),2)*100||'%' AS QTD_2_SIM,
        TRUNC(QTD_2_NAO/DECODE(QTD_2_SIM+QTD_2_NAO, 0, 1, QTD_2_SIM+QTD_2_NAO),2)*100||'%' AS QTD_2_NAO,
        TRUNC(QTD_3_SIM/DECODE(QTD_3_SIM+QTD_3_NAO, 0, 1, QTD_3_SIM+QTD_3_NAO),2)*100||'%' AS QTD_3_SIM,
        TRUNC(QTD_3_NAO/DECODE(QTD_3_SIM+QTD_3_NAO, 0, 1, QTD_3_SIM+QTD_3_NAO),2)*100||'%' AS QTD_3_NAO,
        TRUNC(QTD_4_SIM/DECODE(QTD_4_SIM+QTD_4_NAO, 0, 1, QTD_4_SIM+QTD_4_NAO),2)*100||'%' AS QTD_4_SIM,
        TRUNC(QTD_4_NAO/DECODE(QTD_4_SIM+QTD_4_NAO, 0, 1, QTD_4_SIM+QTD_4_NAO),2)*100||'%' AS QTD_4_NAO,
        TRUNC(QTD_5_SIM/DECODE(QTD_5_SIM+QTD_5_NAO, 0, 1, QTD_5_SIM+QTD_5_NAO),2)*100||'%' AS QTD_5_SIM,
        TRUNC(QTD_5_NAO/DECODE(QTD_5_SIM+QTD_5_NAO, 0, 1, QTD_5_SIM+QTD_5_NAO),2)*100||'%' AS QTD_5_NAO
FROM (
SELECT  NVL(SUM(CASE WHEN QST_1 = 'S' THEN 1 ELSE 0 END),0) QTD_1_SIM,
        NVL(SUM(CASE WHEN QST_1 = 'S' THEN 0 ELSE 1 END),0) QTD_1_NAO,
	NVL(SUM(CASE WHEN QST_2 = 'S' THEN 1 ELSE 0 END),0) QTD_2_SIM,
	NVL(SUM(CASE WHEN QST_2 = 'S' THEN 0 ELSE 1 END),0) QTD_2_NAO,
	NVL(SUM(CASE WHEN QST_3 = 'S' THEN 1 ELSE 0 END),0) QTD_3_SIM,
	NVL(SUM(CASE WHEN QST_3 = 'S' THEN 0 ELSE 1 END),0) QTD_3_NAO,
	NVL(SUM(CASE WHEN QST_4 = 'S' THEN 1 ELSE 0 END),0) QTD_4_SIM,
	NVL(SUM(CASE WHEN QST_4 = 'S' THEN 0 ELSE 1 END),0) QTD_4_NAO,
	NVL(SUM(CASE WHEN QST_5 = 'S' THEN 1 ELSE 0 END),0) QTD_5_SIM,
	NVL(SUM(CASE WHEN QST_5 = 'S' THEN 0 ELSE 1 END),0) QTD_5_NAO
FROM 	DBAMV.PESQUISA_SATISFACAO
WHERE  	TRUNC(DT_PESQUISA) BETWEEN :CAMPO_DATAINICIO  AND 
AND   	CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA)
