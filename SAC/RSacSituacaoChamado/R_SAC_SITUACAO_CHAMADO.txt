--R_SAC_SITUACAO_CHAMADO
SELECT * FROM (
SELECT  SAC_SOLICITACAO.CD_SOLICITACAO
               ,SAC_SOLICITACAO.HR_SOLICITACAO
               ,SAC_SOLICITACAO.DT_SOLICIDACAO
               ,decode (SAC_SOLICITACAO.tp_situacao
,'A',:CAMPO_TP_SITUACAO_A
 					                           ,'G',:CAMPO_TP_SITUACAO_G
					                           ,'E',:CAMPO_TP_SITUACAO_E
					                           ,'M',:CAMPO_TP_SITUACAO_M
					                           ,'O',:CAMPO_TP_SITUACAO_O
					                           ,'F',:CAMPO_TP_SITUACAO_F
					                           ,'C',:CAMPO_TP_SITUACAO_C
					                           ,'R',:CAMPO_TP_SITUACAO_R
					                           ,'P',:CAMPO_TP_SITUACAO_P) tp_situacao
FROM    DBAMV.SAC_SOLICITACAO
             ,DBAMV.SAC_ALTERA_SITUACAO
WHERE  SAC_SOLICITACAO.DT_SOLICIDACAO >= DECODE(:CAMPO_DATAINICIO, NULL, SAC_SOLICITACAO.DT_SOLICIDACAO, :CAMPO_DATAINICIO)
AND        SAC_SOLICITACAO.DT_SOLICIDACAO <=  DECODE(:CAMPO_DATAFIM, NULL, SAC_SOLICITACAO.DT_SOLICIDACAO, :CAMPO_DATAFIM)
AND  SAC_SOLICITACAO.CD_SOLICITACAO = SAC_ALTERA_SITUACAO.CD_SOLICITACAO
AND SAC_SOLICITACAO.CD_MULTI_EMPRESA = NVL(DBAMV.PKG_MV2000.LE_EMPRESA, SAC_SOLICITACAO.CD_MULTI_EMPRESA)
and   sac_altera_situacao.cd_alt_situacao = (select max(alt.cd_alt_situacao)
                                    from   dbamv.sac_altera_situacao alt
                                    where  alt.cd_solicitacao =  sac_solicitacao.cd_solicitacao)
GROUP BY  SAC_SOLICITACAO.DT_SOLICIDACAO
                    ,SAC_SOLICITACAO.HR_SOLICITACAO
                   ,SAC_SOLICITACAO.CD_CHAMADO
                   ,SAC_SOLICITACAO.CD_NIVEL
                   ,SAC_SOLICITACAO.TP_SITUACAO
                   ,SAC_SOLICITACAO.CD_SOLICITACAO
ORDER BY DT_SOLICIDACAO
)
ORDER BY DT_SOLICIDACAO
