--R_SAC_SOLIC_TIPO
SELECT * FROM (
SELECT SETOR.CD_SETOR
      ,SETOR.NM_SETOR
      ,SAC_CHAMADO_SETOR.CD_SAC_PARECER_SETOR
      ,TIPO_SAC_CHAMADO.CD_CHAMADO
      ,TIPO_SAC_CHAMADO.NM_CHAMADO
      ,SAC_MOTIVO.CD_SAC_MOTIVO
      ,SAC_MOTIVO.DS_SAC_MOTIVO
      ,SAC_SOLICITACAO.CD_SOLICITACAO
      ,SAC_SOLICITACAO.DT_SOLICIDACAO
      ,SAC_SOLICITACAO.HR_SOLICITACAO
      ,(TO_CHAR(SAC_SOLICITACAO.DT_SOLICIDACAO,'DD/MM/YYYY')||' '||TO_CHAR(SAC_SOLICITACAO.HR_SOLICITACAO,'HH24')) DT_HR_SOLIC
      ,DECODE(SAC_SOLICITACAO.SN_RETORNA_CONTATO,'S','SIM','NÃO') SN_RETORNA_CONTATO
      ,DECODE(SAC_SOLICITACAO.TP_SITUACAO,'A','ABERTO'
   			   	                 ,'G','AGUARDANDO RETORNO'
					 ,'E','EM ANÁLISE'
                                                                                 ,'M','EM ACOMPANHAMENTO'
					 ,'O','AGUARDANDO CORREÇÃO'
					 ,'F','FINALIZADO'
					 ,'C','RETORNO AO CLIENTE'
					 ,'R','RETORNADO') TP_SITUACAO
      ,DECODE(SAC_SOLICITACAO.SN_SATISFATORIO,'S','SIM','NÃO') SN_SATISFATORIO
      ,DECODE(SAC_SOLICITACAO.SN_FIDELIDADE,'S','SIM','NÃO') SN_FIDELIDADE
      ,SAC_SOLICITACAO.DESCR_CHAMADO
      ,USUARIOS.NM_USUARIO
      ,SAC_CHAMADO_SETOR.DS_RETORNO
FROM DBAMV.SAC_SOLICITACAO
    ,DBAMV.SAC_CHAMADO_SETOR
    ,DBAMV.TIPO_SAC_CHAMADO
    ,DBAMV.SETOR
    ,DBAMV.SAC_MOTIVO_SETOR
    ,DBAMV.SAC_MOTIVO
    ,DBASGU.USUARIOS
WHERE SAC_SOLICITACAO.CD_SOLICITACAO = SAC_CHAMADO_SETOR.CD_SOLICITACAO(+)
AND SAC_SOLICITACAO.CD_CHAMADO = TIPO_SAC_CHAMADO.CD_CHAMADO
AND SAC_CHAMADO_SETOR.CD_SETOR = SETOR.CD_SETOR(+)
AND SAC_MOTIVO_SETOR.CD_SAC_PARECER_SETOR(+) = SAC_CHAMADO_SETOR.CD_SAC_PARECER_SETOR
AND SAC_MOTIVO.CD_SAC_MOTIVO(+) = SAC_MOTIVO_SETOR.CD_SAC_MOTIVO
AND SAC_MOTIVO_SETOR.CD_USUARIO = USUARIOS.CD_USUARIO(+)
AND SAC_SOLICITACAO.CD_MULTI_EMPRESA = NVL(DBAMV.PKG_MV2000.LE_EMPRESA, SAC_SOLICITACAO.CD_MULTI_EMPRESA)
AND SAC_SOLICITACAO.DT_SOLICIDACAO BETWEEN :CAMPO_DT_INICIAL ORDER BY SETOR.NM_SETOR, TIPO_SAC_CHAMADO.NM_CHAMADO, SAC_SOLICITACAO.CD_SOLICITACAO
)
ORDER BY NM_SETOR,CD_SETOR,
NM_CHAMADO,CD_CHAMADO,
CD_SOLICITACAO,CD_SAC_PARECER_SETOR,DS_RETORNO,DT_SOLICIDACAO,HR_SOLICITACAO,SN_RETORNA_CONTATO,TP_SITUACAO,SN_SATISFATORIO,SN_FIDELIDADE,DESCR_CHAMADO
