--R_SAC_EST_COMPR
SELECT * FROM (
SELECT SAC_CHAMADO_SETOR.CD_SETOR
      ,SETOR.NM_SETOR
	,SAC_CHAMADO_SETOR.CD_SAC_PARECER_SETOR
      ,initcap(TIPO_SAC_CHAMADO.NM_CHAMADO) NM_CHAMADO
      ,SAC_SOLICITACAO.CD_SOLICITACAO
      ,TO_DATE(TO_CHAR(SAC_SOLICITACAO.DT_SOLICIDACAO,'DD/MM/YYYY')||' '||TO_CHAR(SAC_SOLICITACAO.HR_SOLICITACAO,'HH24'),'DD/MM/YYYY HH24') DT_ABERTURA
      ,SAC_CHAMADO_SETOR.DT_RETORNO
      ,SAC_SOLICITACAO.DT_ENCERRAMENTO
      ,(trunc(TO_DATE(TO_CHAR(nvl(SAC_CHAMADO_SETOR.DT_RETORNO,NVL(SAC_SOLICITACAO.DT_ENCERRAMENTO, SYSDATE)),'DD/MM/YYYY HH24'),'DD/MM/YYYY HH24')-TO_DATE(TO_CHAR(SAC_SOLICITACAO.DT_SOLICIDACAO,'DD/MM/YYYY')||' '||TO_CHAR(SAC_SOLICITACAO.HR_SOLICITACAO,'HH24'),'DD/MM/YYYY HH24'))) dia
      ,TRUNC(((TO_DATE(TO_CHAR(nvl(SAC_CHAMADO_SETOR.DT_RETORNO,NVL(SAC_SOLICITACAO.DT_ENCERRAMENTO, SYSDATE)),'DD/MM/YYYY HH24'),'DD/MM/YYYY HH24')-TO_DATE(TO_CHAR(SAC_SOLICITACAO.DT_SOLICIDACAO,'DD/MM/YYYY')||' '||TO_CHAR(SAC_SOLICITACAO.HR_SOLICITACAO,'HH24'),'DD/MM/YYYY HH24'))
       - (trunc(TO_DATE(TO_CHAR(nvl(SAC_CHAMADO_SETOR.DT_RETORNO,NVL(SAC_SOLICITACAO.DT_ENCERRAMENTO, SYSDATE)),'DD/MM/YYYY HH24'),'DD/MM/YYYY HH24')-TO_DATE(TO_CHAR(SAC_SOLICITACAO.DT_SOLICIDACAO,'DD/MM/YYYY')||' '||TO_CHAR(SAC_SOLICITACAO.HR_SOLICITACAO,'HH24'),'DD/MM/YYYY HH24')))) * 24) horas
      ,ABS(((TRUNC(((TO_DATE(TO_CHAR(nvl(SAC_CHAMADO_SETOR.DT_RETORNO,NVL(SAC_SOLICITACAO.DT_ENCERRAMENTO, SYSDATE)),'DD/MM/YYYY HH24'),'DD/MM/YYYY HH24')-TO_DATE(TO_CHAR(SAC_SOLICITACAO.DT_SOLICIDACAO,'DD/MM/YYYY')||' '||TO_CHAR(SAC_SOLICITACAO.HR_SOLICITACAO,'HH24'),'DD/MM/YYYY HH24'))
		- (trunc(TO_DATE(TO_CHAR(nvl(SAC_CHAMADO_SETOR.DT_RETORNO,NVL(SAC_SOLICITACAO.DT_ENCERRAMENTO, SYSDATE)),'DD/MM/YYYY HH24'),'DD/MM/YYYY HH24')-TO_DATE(TO_CHAR(SAC_SOLICITACAO.DT_SOLICIDACAO,'DD/MM/YYYY')||' '||TO_CHAR(SAC_SOLICITACAO.HR_SOLICITACAO,'HH24'),'DD/MM/YYYY HH24')))) * 24) *60)
	    -
		((((TO_DATE(TO_CHAR(nvl(SAC_CHAMADO_SETOR.DT_RETORNO,NVL(SAC_SOLICITACAO.DT_ENCERRAMENTO, SYSDATE)),'DD/MM/YYYY HH24'),'DD/MM/YYYY HH24')-TO_DATE(TO_CHAR(SAC_SOLICITACAO.DT_SOLICIDACAO,'DD/MM/YYYY')||' '||TO_CHAR(SAC_SOLICITACAO.HR_SOLICITACAO,'HH24'),'DD/MM/YYYY HH24'))
		- (trunc(TO_DATE(TO_CHAR(nvl(SAC_CHAMADO_SETOR.DT_RETORNO,NVL(SAC_SOLICITACAO.DT_ENCERRAMENTO, SYSDATE)),'DD/MM/YYYY HH24'),'DD/MM/YYYY HH24')-TO_DATE(TO_CHAR(SAC_SOLICITACAO.DT_SOLICIDACAO,'DD/MM/YYYY')||' '||TO_CHAR(SAC_SOLICITACAO.HR_SOLICITACAO,'HH24'),'DD/MM/YYYY HH24')))) * 24)*60))) minutos
      ,DECODE(SAC_CHAMADO_SETOR.TP_SITUACAO,'A','Aberto'
   			   	                         ,'G','Aguardando Retorno'
					                     ,'E','Em Análise'
                                                                                                     ,'M','Em Acompanhamento'
					                     ,'O','Aguardando Correção'
					                     ,'F','Finalizado'
					                     ,'C','Retorno ao Cliente'
					                     ,'R','Retornado'
                                                                                                     ,'P','Respondido') TP_SITUACAO
      ,(TO_DATE(TO_CHAR(nvl(SAC_CHAMADO_SETOR.DT_RETORNO,NVL(SAC_SOLICITACAO.DT_ENCERRAMENTO, SYSDATE)),'DD/MM/YYYY HH24'),'DD/MM/YYYY HH24')-TO_DATE(TO_CHAR(SAC_SOLICITACAO.DT_SOLICIDACAO,'DD/MM/YYYY')||' '||TO_CHAR(SAC_SOLICITACAO.HR_SOLICITACAO,'HH24'),'DD/MM/YYYY HH24')) SUBTRACAO_DATA
FROM DBAMV.SAC_SOLICITACAO
    ,DBAMV.SAC_CHAMADO_SETOR
    ,DBAMV.TIPO_SAC_CHAMADO
    ,DBAMV.SETOR
WHERE SAC_SOLICITACAO.CD_SOLICITACAO = SAC_CHAMADO_SETOR.CD_SOLICITACAO
AND SAC_SOLICITACAO.CD_CHAMADO = TIPO_SAC_CHAMADO.CD_CHAMADO
AND SAC_SOLICITACAO.CD_MULTI_EMPRESA = NVL(DBAMV.PKG_MV2000.LE_EMPRESA, SAC_SOLICITACAO.CD_MULTI_EMPRESA)
AND SAC_CHAMADO_SETOR.CD_SETOR = SETOR.CD_SETOR
AND NVL(SAC_SOLICITACAO.SN_RETORNA_CONTATO,'N') = 'S'  
AND SAC_SOLICITACAO.DT_SOLICIDACAO BETWEEN :CAMPO_DT_INICIAL AND 
:CAMPO_CF_CD_CHAMADO
)
ORDER BY CD_SETOR,NM_SETOR
