--R_SAC_EST_ORIGEM
select origem_sac_solicitacao.cd_origem,
             origem_sac_solicitacao.nm_origem,
            count(origem_sac_solicitacao.cd_origem)      tot_origem
    from dbamv.sac_solicitacao,
           dbamv.origem_sac_solicitacao
   where  sac_solicitacao.cd_origem     = origem_sac_solicitacao.cd_origem
              and sac_solicitacao.cd_multi_empresa = nvl(dbamv.pkg_mv2000.le_empresa, sac_solicitacao.cd_multi_empresa)
              and dt_solicidacao >= decode(:CAMPO_DATAINICIO,null,dt_solicidacao,:CAMPO_DATAINICIO)
              and dt_solicidacao <= decode(:CAMPO_DATAFIM,null,dt_solicidacao,:CAMPO_DATAFIM)
    group by origem_sac_solicitacao.cd_origem,
              origem_sac_solicitacao.nm_origem
     order by cd_origem
