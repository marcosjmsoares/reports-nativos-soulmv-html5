--R_CIDADE_TIPO_LOCAL
select ctl.cd_cidade cd_cidade
     , ctl.cd_tipo_local cd_tipo_local
     , tp.ds_tipo_local ds_tipo_local
     , ctl.ds_cidade_tipo_local ds_cidade_tipo_local
     , ctl.nr_fone_contato1 nr_fone_contato1
     , ctl.nr_fone_contato2 nr_fone_contato2
     , ctl.nr_fone_contato3 nr_fone_contato3
     , ctl.nr_ramal_contato1 nr_ramal_contato1
     , ctl.nr_ramal_contato2 nr_ramal_contato2
     , ctl.nr_ramal_contato3 nr_ramal_contato3
     , ctl.nr_fax nr_fax
     , ctl.ds_nome_contato ds_nome_contato
     , ctl.ds_email ds_email
     , ctl.ds_obs ds_obs
  from dbamv.cidade_tipo_local ctl
      ,dbamv.cidade c
      ,dbamv.tipo_local tp
where c.cd_cidade = ctl.cd_cidade
  and tp.cd_tipo_local = ctl.cd_tipo_local
