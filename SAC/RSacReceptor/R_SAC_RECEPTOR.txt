--R_SAC_RECEPTOR
SELECT  REC.CD_RECEPTOR
               ,REC.NM_RECEPTOR
               ,REC.CD_SETOR
               ,SETOR.NM_SETOR
FROM  DBAMV.SAC_RECEPTOR REC
           ,DBAMV.SETOR
WHERE REC.CD_SETOR = SETOR.CD_SETOR
ORDER BY
