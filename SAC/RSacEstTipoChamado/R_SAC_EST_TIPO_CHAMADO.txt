--R_SAC_EST_TIPO_CHAMADO
SELECT TIPO_SAC_CHAMADO.NM_CHAMADO
	  ,COUNT(*) QTDE
FROM DBAMV.TIPO_SAC_CHAMADO
    ,DBAMV.SAC_SOLICITACAO
WHERE TIPO_SAC_CHAMADO.CD_CHAMADO = SAC_SOLICITACAO.CD_CHAMADO
AND SAC_SOLICITACAO.CD_MULTI_EMPRESA = DBAMV.PKG_MV2000.LE_EMPRESA
AND SAC_SOLICITACAO.DT_SOLICIDACAO BETWEEN :CAMPO_DT_INICIAL GROUP BY TIPO_SAC_CHAMADO.NM_CHAMADO
ORDER BY TIPO_SAC_CHAMADO.NM_CHAMADO
