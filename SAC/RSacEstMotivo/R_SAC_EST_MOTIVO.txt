--R_SAC_EST_MOTIVO
SELECT SAC_MOTIVO.DS_SAC_MOTIVO
	  ,COUNT(*) QTDE
FROM DBAMV.SAC_MOTIVO
	,DBAMV.SAC_MOTIVO_SETOR
    ,DBAMV.SAC_CHAMADO_SETOR
    ,DBAMV.SAC_SOLICITACAO
WHERE SAC_MOTIVO.CD_SAC_MOTIVO = SAC_MOTIVO_SETOR.CD_SAC_MOTIVO
AND SAC_MOTIVO_SETOR.CD_SAC_PARECER_SETOR = SAC_CHAMADO_SETOR.CD_SAC_PARECER_SETOR
AND sac_solicitacao.cd_multi_empresa     = nvl(dbamv.pkg_mv2000.le_empresa, sac_solicitacao.cd_multi_empresa)
AND SAC_SOLICITACAO.CD_SOLICITACAO = SAC_CHAMADO_SETOR.CD_SOLICITACAO
AND SAC_SOLICITACAO.DT_SOLICIDACAO BETWEEN :CAMPO_DT_INICIAL GROUP BY SAC_MOTIVO.DS_SAC_MOTIVO
ORDER BY SAC_MOTIVO.DS_SAC_MOTIVO
